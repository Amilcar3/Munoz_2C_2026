267void lv_dropdown_add_option(lv_obj_t * obj, const char * option, uint32_t pos)
268{
269 LV_CHECK_OBJ(obj, MY_CLASS, return);
270 LV_CHECK_ARG(option != NULL, return);
271
272 lv_dropdown_t * dropdown = (lv_dropdown_t *)obj;
273 /*Fast path: adding an item to empty list is the same as just setting its options*/
274 if(dropdown->option_cnt == 0) {
275 lv_dropdown_set_options(obj, option);
276 return;
277 }
278
279 size_t old_len = lv_strlen(dropdown->options);
280 const char * ins_buf;
281#if LV_USE_ARABIC_PERSIAN_CHARS == 0
282 size_t ins_len = lv_strlen(option) + 1;
283 ins_buf = option;
284#else
285 size_t ins_len = lv_text_ap_strlen(option) + 1;
286
287 /*The option to insert needs to be Arabic preprocessed*/
288 char * ap_tmp_buf = lv_malloc(ins_len);
289 LV_ASSERT_MALLOC(ap_tmp_buf);
290 if(ap_tmp_buf == NULL) {
291 LV_LOG_WARN("Couldn't allocate buffer for the Arabic conversion");
292 return;
293 }
294 lv_text_ap_proc(option, ap_tmp_buf);
295 ins_buf = ap_tmp_buf;
296
297#endif
298
299 /*Allocate space for the new option*/
300 size_t new_len = ins_len + old_len + 2; /*+2 for terminating NULL and possible \n*/
301 char * new_options;
302 if(dropdown->static_options) {
303 LV_LOG_INFO("Converting static options to dynamic");
304 new_options = lv_malloc(new_len);
305 if(new_options) lv_strcpy(new_options, dropdown->options);
306 }
307 else {
308 new_options = lv_realloc(dropdown->options, new_len);
309 }
310
311 LV_ASSERT_MALLOC(new_options);
312 if(new_options == NULL) {
313 LV_LOG_WARN("Couldn't reallocate the buffer for the options");
314#if LV_USE_ARABIC_PERSIAN_CHARS
315 lv_free(ap_tmp_buf);
316#endif
317 return;
318 }
319
320 /*Insert to the beginning*/
321 if(pos == 0) {
322 lv_text_ins(new_options, 0, ins_buf);
323 lv_text_ins(new_options, lv_text_get_encoded_length(ins_buf), "\n");
324 }
325 /*Insert to the end*/
326 else if(pos == LV_DROPDOWN_POS_LAST || pos >= dropdown->option_cnt) {
327 new_options[old_len] = '\n';
328 new_options[old_len + 1] = '\0';
329 lv_strcat(new_options, ins_buf);
330 }
331 /*Insert to the middle*/
332 else {
333 /*Find the insert character position after a '\n'*/
334 uint32_t ins_byte_pos;
335 uint32_t option_cnt = 0;
336 for(ins_byte_pos = 0; new_options[ins_byte_pos] != '\0'; ins_byte_pos++) {
337 if(option_cnt == pos) break;
338 if(new_options[ins_byte_pos] == '\n') option_cnt++;
339 }
340
341 uint32_t ins_char_pos = lv_text_encoded_get_char_id(new_options, ins_byte_pos);
342 lv_text_ins(new_options, ins_char_pos, ins_buf);
343 ins_char_pos += lv_text_get_encoded_length(ins_buf);
344 lv_text_ins(new_options, ins_char_pos, "\n");
345 }
346
347 dropdown->options = new_options;
348 dropdown->option_cnt++;
349 dropdown->static_options = 0;
350
351#if LV_USE_TRANSLATION
352 if(dropdown->options_translation_tag) {
353 LV_LOG_WARN("Adding an option to translated options removed the translation tag");
354 remove_options_translation_tag(obj);
355 }
356#endif
357
358#if LV_USE_ARABIC_PERSIAN_CHARS
359 lv_free(ap_tmp_buf);
360#endif
361
362 refresh_size(obj);
363 if(dropdown->list) lv_obj_invalidate(dropdown->list);
364}

508void lv_dropdown_get_selected_str(const lv_obj_t * obj, char * buf, uint32_t buf_size)
509{
510 LV_CHECK_OBJ(obj, MY_CLASS, return);
511 LV_CHECK_ARG(buf != NULL, return);
512
513 lv_dropdown_t * dropdown = (lv_dropdown_t *)obj;
514
515 uint32_t i;
516 uint32_t line = 0;
517 size_t txt_len;
518
519 if(dropdown->options) {
520 txt_len = lv_strlen(dropdown->options);
521 }
522 else {
523 buf[0] = '\0';
524 return;
525 }
526
527 for(i = 0; i < txt_len && line != dropdown->sel_opt_id_orig; i++) {
528 if(dropdown->options[i] == '\n') line++;
529 }
530
531 uint32_t c;
532 for(c = 0; i < txt_len && dropdown->options[i] != '\n'; c++, i++) {
533 if(buf_size && c >= buf_size - 1) {
534 LV_LOG_WARN("the buffer was too small");
535 break;
536 }
537 buf[c] = dropdown->options[i];
538 }
539
540 buf[c] = '\0';
541}

543int32_t lv_dropdown_get_option_index(lv_obj_t * obj, const char * option)
544{
545 LV_CHECK_OBJ(obj, MY_CLASS, return 0);
546 LV_CHECK_ARG(option != NULL, return 0);
547
548 const char * opts = lv_dropdown_get_options(obj);
549 uint32_t char_i = 0;
550 uint32_t opt_i = 0;
551 const char * start = opts;
552 const size_t option_len = lv_strlen(option); /*avoid recomputing this multiple times in the loop*/
553
554 while(start[0] != '\0') {
555 for(char_i = 0; (start[char_i] != '\n') && (start[char_i] != '\0'); char_i++);
556
557 if(option_len == char_i && lv_memcmp(start, option, LV_MIN(option_len, char_i)) == 0) {
558 return opt_i;
559 }
560
561 start = &start[char_i];
562 if(start[0] == '\n') start++;
563 char_i = 0;
564 opt_i++;
565 }
566
567 return -1;
568}

595void lv_dropdown_open(lv_obj_t * dropdown_obj)
596{
597 LV_CHECK_OBJ(dropdown_obj, MY_CLASS, return);
598
599 lv_dropdown_t * dropdown = (lv_dropdown_t *)dropdown_obj;
600
601 lv_obj_add_state(dropdown_obj, LV_STATE_CHECKED);
602 lv_obj_set_parent(dropdown->list, lv_obj_get_screen(dropdown_obj));
603 lv_obj_move_to_index(dropdown->list, -1);
604 lv_obj_set_hidden(dropdown->list, false);
605
606 /*To allow styling the list*/
607 lv_obj_send_event(dropdown_obj, LV_EVENT_READY, NULL);
608
609 lv_obj_t * label = get_label(dropdown_obj);
610 lv_label_set_text_static(label, dropdown->options);
611 lv_obj_set_width(dropdown->list, LV_SIZE_CONTENT);
612
613 lv_obj_update_layout(label);
614 /*Set smaller width to the width of the button*/
615 if(lv_obj_get_width(dropdown->list) <= lv_obj_get_width(dropdown_obj) &&
616 (dropdown->dir == LV_DIR_TOP || dropdown->dir == LV_DIR_BOTTOM)) {
617 lv_obj_set_width(dropdown->list, lv_obj_get_width(dropdown_obj));
618 }
619
620 int32_t label_h = lv_obj_get_height(label);
621 int32_t border_width = lv_obj_get_style_border_width_internal(dropdown->list, LV_PART_MAIN);
622 int32_t top = lv_obj_get_style_pad_top_internal(dropdown->list, LV_PART_MAIN) + border_width;
623 int32_t bottom = lv_obj_get_style_pad_bottom_internal(dropdown->list, LV_PART_MAIN) + border_width;
624
625 int32_t list_fit_h = label_h + top + bottom;
626 int32_t list_h = list_fit_h;
627
628 lv_dir_t dir = dropdown->dir;
629 /*No space on the bottom? See if top is better.*/
630 if(dropdown->dir == LV_DIR_BOTTOM) {
631 if(dropdown_obj->coords.y2 + list_h > LV_VER_RES) {
632 if(dropdown_obj->coords.y1 > LV_VER_RES - dropdown_obj->coords.y2) {
633 /*There is more space on the top, so make it drop up*/
634 dir = LV_DIR_TOP;
635 list_h = dropdown_obj->coords.y1 - 1;
636 }
637 else {
638 list_h = LV_VER_RES - dropdown_obj->coords.y2 - 1 ;
639 }
640 }
641 }
642 /*No space on the top? See if bottom is better.*/
643 else if(dropdown->dir == LV_DIR_TOP) {
644 if(dropdown_obj->coords.y1 - list_h < 0) {
645 if(dropdown_obj->coords.y1 < LV_VER_RES - dropdown_obj->coords.y2) {
646 /*There is more space on the top, so make it drop up*/
647 dir = LV_DIR_BOTTOM;
648 list_h = LV_VER_RES - dropdown_obj->coords.y2;
649 }
650 else {
651 list_h = dropdown_obj->coords.y1;
652 }
653 }
654 }
655
656 if(list_h > list_fit_h) list_h = list_fit_h;
657 lv_obj_set_height(dropdown->list, list_h);
658
659 /*Update the layout to know the real size of the list. It can be smaller than `list_h`
660 *e.g. if a `max_height` style is applied on it.*/
661 lv_obj_update_layout(dropdown->list);
662
663 position_to_selected(dropdown_obj, LV_ANIM_OFF);
664
665 if(dir == LV_DIR_BOTTOM) dropdown_list_align_bidi(dropdown, LV_DIR_BOTTOM);
666 else if(dir == LV_DIR_TOP) dropdown_list_align_bidi(dropdown, LV_DIR_TOP);
667 else if(dir == LV_DIR_LEFT) lv_obj_align_to(dropdown->list, dropdown_obj, LV_ALIGN_OUT_LEFT_TOP, 0, 0);
668 else if(dir == LV_DIR_RIGHT) lv_obj_align_to(dropdown->list, dropdown_obj, LV_ALIGN_OUT_RIGHT_TOP, 0, 0);
669
670 lv_obj_update_layout(dropdown->list);
671
672 if(dropdown->dir == LV_DIR_LEFT || dropdown->dir == LV_DIR_RIGHT) {
673 int32_t y1 = lv_obj_get_y(dropdown->list);
674 int32_t y2 = lv_obj_get_y2(dropdown->list);
675 if(y2 >= LV_VER_RES) {
676 lv_obj_set_y(dropdown->list, y1 - (y2 - LV_VER_RES) - 1);
677 }
678 }
679
680 lv_text_align_t align = lv_obj_calculate_style_text_align(label, LV_PART_MAIN, dropdown->options);
681
682 switch(align) {
683 default:
684 case LV_TEXT_ALIGN_LEFT:
685 lv_obj_align(label, LV_ALIGN_TOP_LEFT, 0, 0);
686 break;
687 case LV_TEXT_ALIGN_RIGHT:
688 lv_obj_align(label, LV_ALIGN_TOP_RIGHT, 0, 0);
689 break;
690 case LV_TEXT_ALIGN_CENTER:
691 lv_obj_align(label, LV_ALIGN_TOP_MID, 0, 0);
692 break;
693
694 }

833static void lv_dropdown_event(const lv_obj_class_t * class_p, lv_event_t * e)
834{
835 LV_UNUSED(class_p);
836 LV_ASSERT(e != NULL);
837
838 lv_result_t res;
839
840 /*Call the ancestor's event handler*/
841 res = lv_obj_event_base(MY_CLASS, e);
842 if(res != LV_RESULT_OK) return;
843
844 lv_event_code_t code = lv_event_get_code(e);
845 lv_obj_t * obj = lv_event_get_current_target(e);
846 lv_dropdown_t * dropdown = (lv_dropdown_t *)obj;
847
848 if(code == LV_EVENT_FOCUSED) {
849 lv_group_t * g = lv_obj_get_group(obj);
850 const bool editing = g && lv_group_get_editing(g);
851 lv_indev_t * indev = lv_indev_active();
852 lv_indev_type_t indev_type = indev != NULL ? lv_indev_get_type(indev) : LV_INDEV_TYPE_NONE;
853
854 /*Encoders need special handling*/
855 if(indev_type == LV_INDEV_TYPE_ENCODER) {
856 /*Open the list if editing*/
857 if(editing) {
858 lv_dropdown_open(obj);
859 }
860 /*Close the list if navigating*/
861 else {
862 dropdown->sel_opt_id = dropdown->sel_opt_id_orig;
863 lv_dropdown_close(obj);
864 }
865 }
866 }
867 else if(code == LV_EVENT_DEFOCUSED || code == LV_EVENT_LEAVE) {
868 lv_dropdown_close(obj);
869 }
870 else if(code == LV_EVENT_RELEASED) {
871 res = btn_release_handler(obj);
872 if(res != LV_RESULT_OK)
873 return;
874 }
875 else if(code == LV_EVENT_VALUE_CHANGED) {
876 refresh_size(obj);
877 }
878 else if(code == LV_EVENT_STYLE_CHANGED) {
879 lv_obj_refresh_self_size(obj);
880 }
881 else if(code == LV_EVENT_SIZE_CHANGED) {
882 lv_obj_refresh_self_size(obj);
883 }
884 else if(code == LV_EVENT_GET_SELF_SIZE) {
885 const lv_font_t * font = lv_obj_get_style_text_font_internal(obj, LV_PART_MAIN);
886
887 lv_point_t size;
888 size.x = 0;
889 size.y = 0;
890
891 /* Calculate the symbol width */
892 int32_t symbol_w = 0;
893 if(dropdown->symbol) {
894 lv_draw_label_dsc_t symbol_dsc;
895 lv_draw_label_dsc_init(&symbol_dsc);
896
897 lv_image_src_t symbol_type = lv_image_src_get_type(dropdown->symbol);
898 if(symbol_type == LV_IMAGE_SRC_SYMBOL) {
899 lv_point_t text_size;
900
901 lv_text_get_size_internal(&text_size,
902 dropdown->symbol,
903 symbol_dsc.font,
904 symbol_dsc.letter_space,
905 symbol_dsc.line_space,
906 LV_COORD_MAX,
907 symbol_dsc.flag);
908 symbol_w = text_size.x;
909 }
910 else {
911 lv_image_header_t header;
912 lv_result_t decoder_res = lv_image_decoder_get_info(dropdown->symbol, &header);
913 if(decoder_res == LV_RESULT_OK) {
914 symbol_w = header.w;
915 }
916 else {
917 symbol_w = 0;
918 }
919 }
920 size.x += symbol_w;
921 size.x += lv_obj_get_style_pad_column_internal(obj, LV_PART_MAIN);
922 }
923
924 /* Calculate the text width */
925 const char * opt_txt;
926 char buf[128];
927 if(dropdown->text)
928 opt_txt = dropdown->text;
929 else {
930 lv_dropdown_get_selected_str(obj, buf, 128);
931 opt_txt = buf;
932 }
933
934 if(opt_txt == NULL) {
935 size.y = LV_MAX(size.y, lv_font_get_line_height_internal(font));
936 }
937 else {
938 lv_draw_label_dsc_t dsc;
939 lv_draw_label_dsc_init(&dsc);
940
941 lv_point_t text_size;
942 int32_t max_width = lv_obj_calc_dynamic_width(obj, LV_STYLE_MAX_WIDTH) - size.x;
943 lv_text_get_size_internal(&text_size, opt_txt, font, dsc.letter_space, dsc.line_space, max_width, dsc.flag);
944
945 size.x += text_size.x;
946 size.y = LV_MAX(size.y, text_size.y);
947 }
948
949 lv_point_t * p = lv_event_get_param(e);
950 p->x = LV_MAX(p->x, size.x);
951 p->y = LV_MAX(p->y, size.y);
952 }
953 else if(code == LV_EVENT_KEY) {
954 uint32_t c = lv_event_get_key(e);
955 if(c == LV_KEY_RIGHT || c == LV_KEY_DOWN) {
956 if(!lv_dropdown_is_open(obj)) {
957 lv_dropdown_open(obj);
958 }
959 else if(dropdown->sel_opt_id + 1 < dropdown->option_cnt) {
960 dropdown->sel_opt_id++;
961 position_to_selected(obj, LV_ANIM_ON);
962 }
963 }
964 else if(c == LV_KEY_LEFT || c == LV_KEY_UP) {
965
966 if(!lv_dropdown_is_open(obj)) {
967 lv_dropdown_open(obj);
968 }
969 else if(dropdown->sel_opt_id > 0) {
970 dropdown->sel_opt_id--;
971 position_to_selected(obj, LV_ANIM_ON);
972 }
973 }
974 else if(c == LV_KEY_ESC) {
975 dropdown->sel_opt_id = dropdown->sel_opt_id_orig;
976 lv_dropdown_close(obj);
977 }
978 else if(c == LV_KEY_ENTER) {
979 /* Handle the ENTER key only if it was send by another object.
980 * Do no process it if ENTER is sent by the dropdown because it's handled in LV_EVENT_RELEASED */
981 lv_obj_t * indev_obj = lv_indev_get_active_obj();
982 if(indev_obj != obj) {
983 res = btn_release_handler(obj);
984 if(res != LV_RESULT_OK)
985 return;
986 }
987 }
988 }
989 else if(code == LV_EVENT_ROTARY) {
990 if(!lv_dropdown_is_open(obj)) {
991 lv_dropdown_open(obj);
992 }
993 else {
994 int32_t r = lv_event_get_rotary_diff(e);
995 int32_t new_id = dropdown->sel_opt_id + r;
996 new_id = LV_CLAMP(0, new_id, (int32_t)dropdown->option_cnt - 1);
997
998 dropdown->sel_opt_id = new_id;
999 position_to_selected(obj, LV_ANIM_ON);
1000 }
1001 }
1002 else if(code == LV_EVENT_DRAW_MAIN) {
1003 draw_main(e);
1004 }
1005#if LV_USE_TRANSLATION
1006 else if(code == LV_EVENT_TRANSLATION_LANGUAGE_CHANGED) {
1007 if(dropdown->text_translation_tag) {
1008 lv_result_t text_res = update_text(obj, lv_tr(dropdown->text_translation_tag));
1009 if(text_res == LV_RESULT_INVALID) {
1010 LV_LOG_WARN("Couldn't update the translated text");
1011 }
1012 else {
1013 refresh_size(obj);
1014 }
1015 }
1016 if(dropdown->options_translation_tag) {
1017 /*Keep the selected option on language change*/
1018 uint32_t prev_sel = dropdown->sel_opt_id;
1019 lv_result_t opts_res = update_options(obj, lv_tr(dropdown->options_translation_tag));
1020 if(opts_res == LV_RESULT_INVALID) {
1021 LV_LOG_WARN("Couldn't update the translated options");
1022 }
1023 else {
1024 refresh_size(obj);
1025
1026 dropdown->sel_opt_id = LV_MIN(prev_sel, dropdown->option_cnt - 1);
1027 dropdown->sel_opt_id_orig = dropdown->sel_opt_id;
1028
1029 if(dropdown->list) lv_obj_invalidate(dropdown->list);
1030 }
1031 }
1032 }
1033#endif /*LV_USE_TRANSLATION*/
1034}

1073static void draw_main(lv_event_t * e)
1074{
1075 LV_ASSERT(e != NULL);
1076 lv_obj_t * obj = lv_event_get_current_target(e);
1077 lv_dropdown_t * dropdown = (lv_dropdown_t *)obj;
1078 lv_layer_t * layer = lv_event_get_layer(e);
1079
1080 int32_t border_width = lv_obj_get_style_border_width_internal(obj, LV_PART_MAIN);
1081 int32_t left = lv_obj_get_style_pad_left_internal(obj, LV_PART_MAIN) + border_width;
1082 int32_t right = lv_obj_get_style_pad_right_internal(obj, LV_PART_MAIN) + border_width;
1083
1084 lv_text_attributes_t attributes = {0};
1085 lv_draw_label_dsc_t symbol_dsc;
1086 lv_draw_label_dsc_init(&symbol_dsc);
1087 symbol_dsc.base.layer = layer;
1088 lv_obj_init_draw_label_dsc(obj, LV_PART_INDICATOR, &symbol_dsc);
1089
1090 /*If no text specified use the selected option*/
1091 const char * opt_txt;
1092 char buf[128];
1093 if(dropdown->text) opt_txt = dropdown->text;
1094 else {
1095 lv_dropdown_get_selected_str(obj, buf, 128);
1096 opt_txt = buf;
1097 }
1098
1099 bool symbol_to_left = false;
1100 if(dropdown->dir == LV_DIR_LEFT) symbol_to_left = true;
1101 if(lv_obj_get_style_base_dir_internal(obj, LV_PART_MAIN) == LV_BASE_DIR_RTL) symbol_to_left = true;
1102
1103 int32_t symbol_w = -1;
1104 if(dropdown->symbol) {
1105 lv_image_src_t symbol_type = lv_image_src_get_type(dropdown->symbol);
1106 int32_t symbol_h;
1107 if(symbol_type == LV_IMAGE_SRC_SYMBOL) {
1108 lv_point_t size;
1109
1110 attributes.letter_space = symbol_dsc.letter_space;
1111 attributes.line_space = symbol_dsc.line_space;
1112 attributes.max_width = LV_COORD_MAX;
1113 attributes.text_flags = symbol_dsc.flag;
1114
1115 lv_text_get_size_attributes(&size, dropdown->symbol, symbol_dsc.font, &attributes);
1116 symbol_w = size.x;
1117 symbol_h = size.y;
1118 }
1119 else {
1120 lv_image_header_t header;
1121 lv_result_t res = lv_image_decoder_get_info(dropdown->symbol, &header);
1122 if(res == LV_RESULT_OK) {
1123 symbol_w = header.w;
1124 symbol_h = header.h;
1125 }
1126 else {
1127 symbol_w = -1;
1128 symbol_h = -1;
1129 }
1130 }
1131
1132 lv_area_t symbol_area;
1133 symbol_area.y1 = obj->coords.y1;
1134 symbol_area.y2 = symbol_area.y1 + symbol_h - 1;
1135 symbol_area.x1 = obj->coords.x1;
1136 symbol_area.x2 = symbol_area.x1 + symbol_w - 1;
1137 if(symbol_to_left) {
1138 lv_area_align(&obj->coords, &symbol_area, LV_ALIGN_LEFT_MID, left, 0);
1139 }
1140 else {
1141 lv_area_align(&obj->coords, &symbol_area, LV_ALIGN_RIGHT_MID, -right, 0);
1142 }
1143
1144 if(symbol_type == LV_IMAGE_SRC_SYMBOL) {
1145 symbol_dsc.text = dropdown->symbol;
1146 lv_draw_label(layer, &symbol_dsc, &symbol_area);
1147 }
1148 else {
1149 lv_draw_image_dsc_t img_dsc;
1150 lv_draw_image_dsc_init(&img_dsc);
1151 img_dsc.base.layer = layer;
1152 lv_obj_init_draw_image_dsc(obj, LV_PART_INDICATOR, &img_dsc);
1153 lv_point_set(&img_dsc.pivot, symbol_w / 2, symbol_h / 2);
1154 img_dsc.rotation = lv_obj_get_style_transform_rotation_internal(obj, LV_PART_INDICATOR);
1155 img_dsc.src = dropdown->symbol;
1156 lv_draw_image(layer, &img_dsc, &symbol_area);
1157 }
1158 }
1159
1160 lv_draw_label_dsc_t label_dsc;
1161 lv_draw_label_dsc_init(&label_dsc);
1162 label_dsc.base.layer = layer;
1163 lv_obj_init_draw_label_dsc(obj, LV_PART_MAIN, &label_dsc);
1164 label_dsc.flag |= LV_TEXT_FLAG_EXPAND;
1165
1166 attributes.letter_space = label_dsc.letter_space;
1167 attributes.max_width = LV_COORD_MAX;
1168 attributes.line_space = label_dsc.line_space;
1169 attributes.text_flags = label_dsc.flag;
1170
1171 lv_point_t size;
1172 lv_text_get_size_attributes(&size, opt_txt, label_dsc.font, &attributes);
1173
1174 lv_area_t txt_area;
1175 txt_area.x1 = obj->coords.x1 + left;
1176 txt_area.x2 = obj->coords.x2 - right;
1177 txt_area.y1 = obj->coords.y1;
1178 txt_area.y2 = txt_area.y1 + size.y - 1;
1179 lv_area_align(&obj->coords, &txt_area, LV_ALIGN_CENTER, 0, 0);
1180
1181 /*Center align the text if no symbol*/
1182 if(dropdown->symbol == NULL && label_dsc.align == LV_TEXT_ALIGN_AUTO) {
1183 label_dsc.align = LV_TEXT_ALIGN_CENTER;
1184 }
1185 else {
1186 /*Add some space between the label and symbol*/
1187 symbol_w += lv_obj_get_style_pad_column_internal(obj, LV_PART_MAIN);
1188
1189 /*Text to the right*/
1190 if(symbol_to_left) {
1191 if(label_dsc.align == LV_TEXT_ALIGN_AUTO) label_dsc.align = LV_TEXT_ALIGN_RIGHT;
1192 txt_area.x1 += symbol_w;
1193 lv_area_align(&obj->coords, &txt_area, LV_ALIGN_RIGHT_MID, -right, 0);
1194 }
1195 else {
1196 if(label_dsc.align == LV_TEXT_ALIGN_AUTO) label_dsc.align = LV_TEXT_ALIGN_LEFT;
1197 txt_area.x2 -= symbol_w;
1198 lv_area_align(&obj->coords, &txt_area, LV_ALIGN_LEFT_MID, left, 0);
1199 }
1200 }
1201
1202 label_dsc.text = opt_txt;
1203 if(dropdown->text == NULL) {
1204 label_dsc.text_local = true;
1205 }
1206
1207 lv_draw_label(layer, &label_dsc, &txt_area);
1208}

1218static void draw_list(lv_event_t * e)
1219{
1220 LV_ASSERT(e != NULL);
1221 lv_obj_t * list_obj = lv_event_get_current_target(e);
1222 lv_dropdown_list_t * list = (lv_dropdown_list_t *)list_obj;
1223 lv_obj_t * dropdown_obj = list->dropdown;
1224 lv_dropdown_t * dropdown = (lv_dropdown_t *)dropdown_obj;
1225 lv_layer_t * layer = lv_event_get_layer(e);
1226
1227 /* Clip area might be too large too to shadow but
1228 * the selected option can be drawn on only the background*/
1229 lv_area_t clip_area_core;
1230 bool has_common;
1231 has_common = lv_area_intersect(&clip_area_core, &layer->_clip_area, &dropdown->list->coords);
1232 if(has_common) {
1233 const lv_area_t clip_area_ori = layer->_clip_area;
1234 layer->_clip_area = clip_area_core;
1235 if(dropdown->selected_highlight) {
