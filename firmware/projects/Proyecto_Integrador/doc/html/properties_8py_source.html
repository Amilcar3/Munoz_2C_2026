59 def match_properties(file_path):
60 pattern = r'^\s*LV_PROPERTY_ID2?\‍((\w+),\s*(\w+),\s*(\w+)(,\s*(\w+))?,\s*(\d+)\‍)'
61 with open(file_path, 'r', encoding='utf-8') as file:
62 for line in file.readlines():

64 if match:
65 id = f"LV_PROPERTY_{match.group(1).upper()}_{match.group(2).upper()}"

71 def match_styles(file_path):
72 pattern_with_value = r'^\s+LV_STYLE_(\w+)\s*=\s*(\d+),'
73 pattern_name_only = r'^\s+LV_STYLE_(\w+)\s*,'
74 last_value = 0
75 process = False
76 with open(file_path, 'r', encoding='utf-8') as file:
77 for line in file.readlines():
78 if re.match("enum _lv_style_id_t", line):
79 process = True
80 continue
81
82 if process and re.match("};", line):
83 return
84
85 if process == False: continue
86 match = re.match(pattern_with_value, line)
87 name = ""
88 if match:
89 name = match.group(1).upper()
90 last_value = int(match.group(2))
91 else:
92 match = re.match(pattern_name_only, line)
93 if match:
94 name = match.group(1).upper()
95 last_value += 1
96 if name:
97 id = f"LV_PROPERTY_STYLE_{name}"
98 yield Property("style",
99 match.group(1).lower(), "style",
100 last_value, id)
