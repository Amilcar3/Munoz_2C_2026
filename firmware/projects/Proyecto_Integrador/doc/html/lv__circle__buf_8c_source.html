235uint32_t lv_circle_buf_fill(lv_circle_buf_t * circle_buf, uint32_t count, lv_circle_buf_fill_cb_t fill_cb,
236 void * user_data)
237{
238 LV_ASSERT_NULL(circle_buf);
239 LV_ASSERT_NULL(fill_cb);
240
241 uint32_t filled = 0;
242 while(count > 0 && !lv_circle_buf_is_full(circle_buf)) {
243 void * data = lv_circle_buf_tail(circle_buf);
244 if(fill_cb(data, circle_buf->array.element_size, (int32_t)filled, user_data) == LV_RESULT_OK) {
245 circle_buf->tail++;
246 filled++;
247 }
248 else break;
249
250 count--;
251 }
252
253 return filled;
254}

279lv_result_t lv_circle_buf_peek_at(const lv_circle_buf_t * circle_buf, const uint32_t index, void * data)
280{
281 LV_ASSERT_NULL(circle_buf);
282 LV_ASSERT_NULL(data);
283 LV_ASSERT_MSG(lv_circle_buf_size(circle_buf) > 0, "can't peek into an empty buffer");
284
285 const uint32_t real_index = (index % lv_circle_buf_size(circle_buf) + circle_buf->head) % lv_circle_buf_capacity(
286 circle_buf);
287 lv_memcpy(data, lv_array_at(&circle_buf->array, real_index), circle_buf->array.element_size);
288
289 return LV_RESULT_OK;
290}
