102lv_result_t lv_fs_frogfs_register_blob(const void * blob, const char * path_prefix)
103{
104 LV_CHECK_ARG(blob != NULL, return LV_RESULT_INVALID);
105 LV_CHECK_ARG(path_prefix != NULL, return LV_RESULT_INVALID);
106
107 if(path_prefix[0] == '\0') {
108 LV_LOG_WARN("path prefix should not be zero-length");
109 return LV_RESULT_INVALID;
110 }
111
112 lv_fs_drv_t * fs_drv_p = frogfs_fs_drv;
113 fs_drv_data_t * data = fs_drv_p->user_data;
114
115 frogfs_config_t frogfs_config = {
116 .addr = blob,
117 };
118
119 frogfs_fs_t * blob_fs = frogfs_init(&frogfs_config);
120 if(blob_fs == NULL) {
121 LV_LOG_WARN("Could not register frogfs blob 0x%p", blob);
122 return LV_RESULT_INVALID;
123 }
124
125 blob_t * hdl = lv_ll_ins_head(&data->blob_ll);
126 LV_ASSERT_MALLOC(hdl);
127
128 hdl->path_prefix = lv_strdup(path_prefix);
129 LV_ASSERT_MALLOC(hdl->path_prefix);
130
131 hdl->blob_fs = blob_fs;
132
133 return LV_RESULT_OK;
134}

136void lv_fs_frogfs_unregister_blob(const char * path_prefix)
137{
138 LV_CHECK_ARG(path_prefix != NULL, return);
139
140 lv_fs_drv_t * fs_drv_p = frogfs_fs_drv;
141 fs_drv_data_t * data = fs_drv_p->user_data;
142
143 blob_t * blob_handle;
144 LV_LL_READ(&data->blob_ll, blob_handle) {
145 if(lv_streq(path_prefix, blob_handle->path_prefix)) {
146 break;
147 }
148 }
149 if(blob_handle == NULL) {
150 LV_LOG_WARN("No frogfs blob with path prefix '%s' to unregister", path_prefix);
151 return;
152 }
153
154 destroy_blob(blob_handle);
155}

161static bool get_blob_and_entry(const char * path, blob_t ** blob_dst, const frogfs_entry_t ** entry_dst)
162{
163 LV_ASSERT(path != NULL);
164 LV_ASSERT(blob_dst != NULL);
165 LV_ASSERT(entry_dst != NULL);
166 lv_fs_drv_t * fs_drv_p = frogfs_fs_drv;
167 fs_drv_data_t * data = fs_drv_p->user_data;
168
169 blob_t * blob;
170 size_t path_prefix_length;
171 LV_LL_READ(&data->blob_ll, blob) {
172 path_prefix_length = lv_strlen(blob->path_prefix);
173 if(0 == lv_strncmp(path, blob->path_prefix, path_prefix_length)
174 && (blob->path_prefix[path_prefix_length - 1] == '/'
175 || blob->path_prefix[path_prefix_length - 1] == '\\'
176 || path[path_prefix_length] == '\0'
177 || path[path_prefix_length] == '/'
178 || path[path_prefix_length] == '\\')) {
179 break;
180 }
181 }
182 if(blob == NULL) {
183 LV_LOG_WARN("Path '%s' does not have a prefix that matches any of the registered frogfs blobs", path);
184 return false;
185 }
186
187 path += path_prefix_length;
188 if(path[0] == '/' || path[0] == '\\') path++;
189
190 const frogfs_entry_t * entry = frogfs_get_entry(blob->blob_fs, path);
191 if(entry == NULL) {
192 LV_LOG_WARN("No entry '%s' in frogfs blob registered under '%s'", path, blob->path_prefix);
193 return false;
194 }
195
196 *blob_dst = blob;
197 *entry_dst = entry;
198 return true;
199}

214static void * fs_open(lv_fs_drv_t * drv, const char * path, lv_fs_mode_t mode)
215{
216 LV_ASSERT(path != NULL);
217 LV_UNUSED(drv);
218
219 if(mode & LV_FS_MODE_WR) {
220 LV_LOG_WARN("Cannot open files for writing with frogfs");
221 return NULL;
222 }
223
224 blob_t * blob;
225 const frogfs_entry_t * entry;
226 if(!get_blob_and_entry(path, &blob, &entry)) {
227 return NULL;
228 }
229
230 if(frogfs_is_dir(entry)) {
231 LV_LOG_WARN("Cannot open directory as file with frogfs");
232 return NULL;
233 }
234
235 frogfs_fh_t * fh = frogfs_open(blob->blob_fs, entry, 0);
236 if(fh == NULL) {
237 LV_LOG_WARN("Could not open '%s' even though the entry exists in frogfs", path);
238 return NULL;
239 }
240
241 return fh;
242}

324static lv_fs_res_t fs_dir_read(lv_fs_drv_t * drv, void * dir_p, char * fn, uint32_t fn_len)
325{
326 LV_ASSERT(dir_p != NULL);
327 LV_ASSERT(fn != NULL);
328 LV_ASSERT(fn_len > 0);
329 LV_UNUSED(drv);
330
331 const frogfs_entry_t * entry = frogfs_readdir(dir_p);
332 if(entry == NULL) {
333 fn[0] = '\0';
334 return LV_FS_RES_OK;
335 }
336
337 char * name = frogfs_get_name(entry);
338
339 if(frogfs_is_dir(entry)) {
340 lv_snprintf(fn, fn_len, "/%s", name);
341 }
342 else {
343 lv_strlcpy(fn, name, fn_len);
344 }
345
346 lv_free(name); /* frogfs `malloc`d it */
347
348 return LV_FS_RES_OK;
349}
