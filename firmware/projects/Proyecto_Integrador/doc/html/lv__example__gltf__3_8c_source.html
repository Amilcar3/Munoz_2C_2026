29static void clicked_event_cb(lv_event_t * e)
30{
31 lv_obj_t * gltf = lv_event_get_target_obj(e);
32 lv_event_code_t code = lv_event_get_code(e);
33
34 if(code == LV_EVENT_RELEASED) {
35 hide_node(cursor_node);
36 return;
37 }
38
39 lv_indev_t * indev = (lv_indev_t *)lv_event_get_param(e);
40 lv_point_t point;
41 lv_indev_get_point(indev, &point);
42
43 /* Create a ray starting from the 2d point in the same direction as the camera view */
44 lv_3dray_t ray = lv_gltf_get_ray_from_2d_coordinate(gltf, &point);
45 /* Create the ground plane */
46 lv_3dplane_t plane = lv_get_ground_plane(0.f);
47
48 /* Check if the ray intersects with the ground plane*/
49 lv_3dpoint_t collision_point;
50 lv_result_t res = lv_intersect_ray_with_plane(&ray, &plane, &collision_point);
51 if(res != LV_RESULT_OK) {
52 /* No hit*/
53 hide_node(cursor_node);
54 return;
55 }
56
57 if(code == LV_EVENT_PRESSED) {
58 /* Get the selected plant if the collision point is near enough
59 * and also compute its offset point so the plant doesn't suddenly
60 * jump to collision point*/
61 selected_plant.plant = get_plant(&collision_point, &selected_plant.offset_point);
62 if(!selected_plant.plant) {
63 /* No plant near hit*/
64 hide_node(cursor_node);
65 return;
66 }
67 }
68 if(selected_plant.plant) {
69 /* Calculate the offset based on when it first was pressed
70 * so that the plant doesn't jump */
71 collision_point.x -= selected_plant.offset_point.x;
72 collision_point.y -= selected_plant.offset_point.y;
73 collision_point.z -= selected_plant.offset_point.z;
74
75 move_plant(selected_plant.plant, &collision_point);
76 rotate_plant(selected_plant.plant);
77 }
78}

80static plant_t * get_plant(const lv_3dpoint_t * point, lv_3dpoint_t * d)
81{
82 for(size_t i = 0; i < PLANT_COUNT; ++i) {
83 lv_3dpoint_t * plant_position = &plant_positions[i].position;
84 d->x = point->x - plant_position->x;
85 d->y = point->y - plant_position->y;
86 d->z = point->z - plant_position->z;
87 float dist_sq = d->x * d->x + d->y * d->y + d->z * d->z;
88 if(dist_sq < .1) {
89 return &plant_positions[i];
90 }
91 }
92 return NULL;
93}

149void lv_example_gltf_3(void)
150{
151 lv_obj_t * gltf = lv_gltf_create(lv_screen_active());
152 lv_obj_set_size(gltf, LV_PCT(100), LV_PCT(100));
153 lv_gltf_set_pitch(gltf, -22.5f);
154 lv_gltf_set_distance(gltf, 5);
155 static const float plant_x[] = {-1.0, -1.0, 0.0, 1.0, 1.0};
156 static const float plant_z[] = {1.0, -1.0, 0.0, 1.0, -1.0};
157
158 for(size_t i = 0; i < PLANT_COUNT; ++i) {
159 lv_gltf_model_t * plant_model = lv_gltf_load_model_from_file(gltf,
160 "A:lvgl/examples/libs/gltf/webp_diffuse_transmission_plant.glb");
161 plant_positions[i].root = lv_gltf_model_node_get_by_numeric_path(plant_model, ".0");
162
163 /* Register an event so that we can get the plant position when it's updated*/
164 /* For the plant we are interested in its world position and its rotation
165 * but since we're using the root node of the plant, its local position
166 * will be the same as the world position so we can use it instead of the world position
167 * to avoid computation overhead*/
168 lv_gltf_model_node_add_event_cb(plant_positions[i].root, plant_position_cb,
169 LV_EVENT_VALUE_CHANGED, NULL);
170
171 lv_gltf_model_node_set_position_x(plant_positions[i].root, plant_x[i]);
172 lv_gltf_model_node_set_position_z(plant_positions[i].root, plant_z[i]);
173 }
174
175 lv_gltf_model_t * grid_model = lv_gltf_load_model_from_file(gltf, "A:lvgl/examples/libs/gltf/support_assets.glb");
176
177 cursor_node = lv_gltf_model_node_get_by_path(grid_model, "/cursor");
178
179 hide_node(cursor_node);
180 lv_obj_add_event_cb(gltf, clicked_event_cb, LV_EVENT_PRESSED, NULL);
181 lv_obj_add_event_cb(gltf, clicked_event_cb, LV_EVENT_PRESSING, NULL);
182 lv_obj_add_event_cb(gltf, clicked_event_cb, LV_EVENT_RELEASED, NULL);
183}
