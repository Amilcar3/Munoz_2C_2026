122void EVE_polar_cartesian(uint16_t length, uint16_t angle, int16_t *p_xc0, int16_t *p_yc0)
123{
124 uint16_t anglev;
125 anglev = angle % 360U;
126
127 if (p_xc0 != NULL)
128 {
129 int32_t calc = (int16_t) length;

132 }
133

135 {

139 int32_t calc = (int16_t) length;

141 *p_yc0 = (int16_t) calc;
142 }
143}
