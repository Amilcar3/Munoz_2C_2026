229void test_freetype_kerning(void)
230{
231 /* Create a font with kerning enabled using font_info */
232 lv_font_info_t font_info;
233 lv_freetype_init_font_info(&font_info);
234 font_info.name = "./src/test_files/fonts/noto/NotoSansSC-Regular.ttf";
235 font_info.size = 32;
236 font_info.render_mode = LV_FREETYPE_FONT_RENDER_MODE_BITMAP;
237 font_info.style = LV_FREETYPE_FONT_STYLE_NORMAL;
238 font_info.kerning = LV_FONT_KERNING_NORMAL;
239
240 lv_font_t * font_kerning = lv_freetype_font_create_with_info(&font_info);
241 TEST_ASSERT_NOT_NULL(font_kerning);
242
243 /* Create a font with kerning disabled */
244 font_info.kerning = LV_FONT_KERNING_NONE;
245 lv_font_t * font_no_kerning = lv_freetype_font_create_with_info(&font_info);
246 TEST_ASSERT_NOT_NULL(font_no_kerning);
247
248 /* Test glyph width with kerning - exercises the FT_Set_Pixel_Sizes code path */
249 uint16_t width_kerning = lv_font_get_glyph_width(font_kerning, 'A', 'V');
250 uint16_t width_no_kerning = lv_font_get_glyph_width(font_no_kerning, 'A', 'V');
251
252 /* Both should return valid widths */
253 TEST_ASSERT_GREATER_THAN(0, width_kerning);
254 TEST_ASSERT_GREATER_THAN(0, width_no_kerning);
255
256 /* Clean up */
257 lv_freetype_font_delete(font_kerning);
258 lv_freetype_font_delete(font_no_kerning);
259}

265void test_freetype_kerning_scalable_sizes(void)
266{
267 /* Test with multiple sizes to ensure pixel size is set correctly */
268 const uint32_t sizes[] = {16, 24, 32, 48, 64};
269
270 for(int i = 0; i < 5; i++) {
271 lv_font_info_t font_info;
272 lv_freetype_init_font_info(&font_info);
273 font_info.name = "./src/test_files/fonts/noto/NotoSansSC-Regular.ttf";
274 font_info.size = sizes[i];
275 font_info.render_mode = LV_FREETYPE_FONT_RENDER_MODE_BITMAP;
276 font_info.style = LV_FREETYPE_FONT_STYLE_NORMAL;
277 font_info.kerning = LV_FONT_KERNING_NORMAL;
278
279 lv_font_t * font = lv_freetype_font_create_with_info(&font_info);
280 TEST_ASSERT_NOT_NULL(font);
281
282 /* Test getting glyph width which triggers kerning lookup */
283 /* This exercises the FT_IS_SCALABLE and FT_Set_Pixel_Sizes path */
284 uint16_t width_V = lv_font_get_glyph_width(font, 'V', 'A');
285 uint16_t width_A = lv_font_get_glyph_width(font, 'A', 'V');
286 uint16_t width_T = lv_font_get_glyph_width(font, 'T', 'o');
287
288 /* Width should be non-zero for valid glyphs */
289 TEST_ASSERT_GREATER_THAN(0, width_V);
290 TEST_ASSERT_GREATER_THAN(0, width_A);
291 TEST_ASSERT_GREATER_THAN(0, width_T);
292
293 lv_freetype_font_delete(font);
294 }
