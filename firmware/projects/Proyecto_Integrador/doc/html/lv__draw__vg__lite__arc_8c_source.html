52void lv_draw_vg_lite_arc(lv_draw_task_t * t, const lv_draw_arc_dsc_t * dsc,
53 const lv_area_t * coords)
54{
55 lv_draw_vg_lite_unit_t * u = (lv_draw_vg_lite_unit_t *)t->draw_unit;
56
57 lv_area_t clip_area;
58 if(!lv_area_intersect(&clip_area, coords, &t->clip_area)) {
59 /*Fully clipped, nothing to do*/
60 return;
61 }
62
63 float start_angle = dsc->start_angle;
64 float end_angle = dsc->end_angle;
65 float sweep_angle = end_angle - start_angle;
66
67 while(sweep_angle < 0) {
68 sweep_angle += 360;
69 }
70
71 while(sweep_angle > 360) {
72 sweep_angle -= 360;
73 }
74
75 /*If the angles are the same then there is nothing to draw*/
76 if(math_zero(sweep_angle)) {
77 return;
78 }
79
80 LV_PROFILER_DRAW_BEGIN;
81
82 lv_vg_lite_path_t * path = lv_vg_lite_path_get(u, VG_LITE_FP32);
83 lv_vg_lite_path_set_bounding_box_area(path, &clip_area);
84
85 float radius_out = dsc->radius;
86 float radius_in = dsc->radius - dsc->width;
87 float cx = dsc->center.x;
88 float cy = dsc->center.y;
89
90 vg_lite_fill_t fill = VG_LITE_FILL_NON_ZERO;
91
92 if(math_equal(sweep_angle, 360)) {
93 lv_vg_lite_path_append_circle(path, cx, cy, radius_out, radius_out);
94
95 /* radius_in <= 0, normal fill circle */
96 if(radius_in > 0) {
97 lv_vg_lite_path_append_circle(path, cx, cy, radius_in, radius_in);
98 }
99 fill = VG_LITE_FILL_EVEN_ODD;
100 }
101 else {
102 float start_angle_rad = MATH_RADIANS(start_angle);
103 float end_angle_rad = MATH_RADIANS(end_angle);
104
105 if(radius_in > 0) {
106 /* radius_out start point */
107 float start_x = radius_out * MATH_COSF(start_angle_rad) + cx;
108 float start_y = radius_out * MATH_SINF(start_angle_rad) + cy;
109
110 /* radius_in start point */
111 float end_x = radius_in * MATH_COSF(end_angle_rad) + cx;
112 float end_y = radius_in * MATH_SINF(end_angle_rad) + cy;
113
114 lv_vg_lite_path_move_to(path, start_x, start_y);
115
116 /* radius_out arc */
117 lv_vg_lite_path_append_arc(path,
118 cx, cy,
119 radius_out,
120 start_angle,
121 sweep_angle,
122 false);
123
124 /* line to radius_in */
125 lv_vg_lite_path_line_to(path, end_x, end_y);
126
127 /* radius_in arc */
128 lv_vg_lite_path_append_arc(path,
129 cx, cy,
130 radius_in,
131 end_angle,
132 -sweep_angle,
133 false);
134
135 /* close arc */
136 lv_vg_lite_path_close(path);
137 }
138 else {
139 /* draw a normal arc pie shape */
140 lv_vg_lite_path_append_arc(path, cx, cy, radius_out, start_angle, sweep_angle, true);
141 }
142
143 /* draw round */
144 if(dsc->rounded && dsc->width > 0) {
145 float round_radius = radius_out > dsc->width ? dsc->width / 2.0f : radius_out / 2.0f;
146 float round_center = radius_out - round_radius;
147 float rcx1 = cx + round_center * MATH_COSF(end_angle_rad);
148 float rcy1 = cy + round_center * MATH_SINF(end_angle_rad);
149 lv_vg_lite_path_append_circle(path, rcx1, rcy1, round_radius, round_radius);
150
151 float rcx2 = cx + round_center * MATH_COSF(start_angle_rad);
152 float rcy2 = cy + round_center * MATH_SINF(start_angle_rad);
153 lv_vg_lite_path_append_circle(path, rcx2, rcy2, round_radius, round_radius);
154 }
155 }
156
157 lv_vg_lite_path_end(path);
158
159 vg_lite_matrix_t matrix = u->global_matrix;
160
161 if(dsc->img_src) {
162 vg_lite_buffer_t src_buf;
163 lv_image_decoder_dsc_t decoder_dsc;
164 if(lv_vg_lite_buffer_open_image(&src_buf, &decoder_dsc, dsc->img_src, false, true)) {
165
166 vg_lite_color_t img_color = 0;
167 if(dsc->opa < LV_OPA_COVER) {
168 /* normal image opa */
169 src_buf.image_mode = VG_LITE_MULTIPLY_IMAGE_MODE;
170 lv_memset(&img_color, dsc->opa, sizeof(img_color));
171 }
172
173 vg_lite_matrix_t path_matrix = u->global_matrix;
174
175 /* move image to center */
176 float img_half_w = decoder_dsc.decoded->header.w / 2.0f;
177 float img_half_h = decoder_dsc.decoded->header.h / 2.0f;
178 vg_lite_translate(cx - img_half_w, cy - img_half_h, &matrix);
179
180 lv_vg_lite_draw_pattern(
181 &u->target_buffer,
182 lv_vg_lite_path_get_path(path),
183 fill,
184 &path_matrix,
185 &src_buf,
186 &matrix,
187 VG_LITE_BLEND_SRC_OVER,
188 VG_LITE_PATTERN_COLOR,
189 0,
190 img_color,
191 lv_vg_lite_matrix_get_filter(&matrix));
192
193 lv_vg_lite_pending_add(u->image_dsc_pending, &decoder_dsc);
194 }
195 }
196 else {
197 /* normal color fill */
198 lv_vg_lite_draw(
199 &u->target_buffer,
200 lv_vg_lite_path_get_path(path),
201 fill,
202 &matrix,
203 VG_LITE_BLEND_SRC_OVER,
204 lv_vg_lite_color(dsc->color, dsc->opa, true));
205 }
206
207 lv_vg_lite_path_drop(u, path);
208 LV_PROFILER_DRAW_END;
209}
