70void lv_epic_init(void)
71{
72 if(epic_initialized) {
73 return;
74 }
75
76 memset(&epic_handle, 0, sizeof(epic_handle));
77 epic_handle.Instance = LV_SIFLI_EPIC_INSTANCE;
78
79#ifdef HAL_EZIP_MODULE_ENABLED
80 memset(&ezip_handle, 0, sizeof(ezip_handle));
81 ezip_handle.Instance = EZIP;
82 epic_handle.hezip = &ezip_handle;
83
84 if(HAL_EZIP_Init(epic_handle.hezip) != HAL_OK) {
85 return;
86 }
87#endif
88
89 if(HAL_EPIC_Init(&epic_handle) != HAL_OK) {
90 return;
91 }
92
93 if(lv_epic_osa_init() != LV_RESULT_OK) {
94 memset(&epic_handle, 0, sizeof(epic_handle));
95#ifdef HAL_EZIP_MODULE_ENABLED
96 memset(&ezip_handle, 0, sizeof(ezip_handle));
97#endif
98 return;
99 }
100
101 epic_initialized = true;
102}

263static void epic_xfer_cplt_callback(EPIC_HandleTypeDef * epic)
264{
265 if(epic_irq_log_cnt < 8) {
266 LV_LOG_WARN("EPIC irq done[%u]: state=%d",
267 (unsigned int)epic_irq_log_cnt,
268 (int)epic->State);
269 epic_irq_log_cnt++;
270 }
271
272 lv_epic_osa_thread_sync_signal_isr();
273
274 if(epic_async_cb != NULL) {
275 lv_epic_cplt_cbk cb = epic_async_cb;
276 epic_async_cb = NULL;
277 cb(epic);
278 }
279}

281static HAL_StatusTypeDef epic_wait_async_result(HAL_StatusTypeDef start_status)
282{
283 if(start_status != HAL_OK) {
284 LV_LOG_WARN("EPIC start failed: status=%d", (int)start_status);
285 epic_start_failed_cleanup();
286 return start_status;
287 }
288
289 lv_epic_wait();
290 return HAL_OK;
291}

293static HAL_StatusTypeDef epic_prepare_start(EPIC_HandleTypeDef * epic,
294 lv_epic_cplt_cbk cb)
295{
296 if(epic == NULL) {
297 return HAL_ERROR;
298 }
299
300 if(lv_epic_is_busy()) {
301 lv_epic_wait();
302 }
303
304 epic_async_cb = cb;
305 epic->XferCpltCallback = epic_xfer_cplt_callback;
306 lv_epic_run();
307 if(epic_hw_start_log_cnt < 8) {
308 LV_LOG_WARN("EPIC hw start[%u]: state=%d",
309 (unsigned int)epic_hw_start_log_cnt,
310 (int)epic->State);
311 epic_hw_start_log_cnt++;
312 }
313 return HAL_OK;
314}
