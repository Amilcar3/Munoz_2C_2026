437lv_span_t * lv_spangroup_get_child(const lv_obj_t * obj, int32_t id)
438{
439 LV_CHECK_OBJ(obj, MY_CLASS, return NULL);
440 lv_spangroup_t * spans = (lv_spangroup_t *)obj;
441 lv_ll_t * linked_list = &spans->child_ll;
442
443 bool traverse_forwards = (id >= 0);
444 int32_t cur_idx = 0;
445 lv_ll_node_t * cur_node = linked_list->head;
446
447 /*If using a negative index, start from the tail and use cur -1 to indicate the end*/
448 if(!traverse_forwards) {
449 cur_idx = -1;
450 cur_node = linked_list->tail;
451 }
452
453 while(cur_node != NULL) {
454 if(cur_idx == id) {
455 return (lv_span_t *) cur_node;
456 }
457 if(traverse_forwards) {
458 cur_node = (lv_ll_node_t *) lv_ll_get_next(linked_list, cur_node);
459 cur_idx++;
460 }
461 else {
462 cur_node = (lv_ll_node_t *) lv_ll_get_prev(linked_list, cur_node);
463 cur_idx--;
464 }
465 }
466
467 return NULL;
468}

548uint32_t lv_spangroup_get_expand_width(lv_obj_t * obj, uint32_t max_width)
549{
550 LV_CHECK_OBJ(obj, MY_CLASS, return 0);
551 lv_spangroup_t * spans = (lv_spangroup_t *)obj;
552
553 if(lv_ll_get_head(&spans->child_ll) == NULL) {
554 return 0;
555 }
556
557 uint32_t width = LV_COORD_IS_PCT(spans->indent) ? 0 : spans->indent;
558 int32_t letter_space = 0;
559 lv_span_t * cur_span;
560 LV_LL_READ(&spans->child_ll, cur_span) {
561 uint32_t letter;
562 uint32_t letter_next;
563 const lv_font_t * font = lv_span_get_style_text_font(obj, cur_span);
564
565 letter_space = lv_span_get_style_text_letter_space(obj, cur_span);
566 uint32_t j = 0;
567 const char * cur_txt = cur_span->txt;
568 SPAN_TEXT_CHECK(&cur_txt);
569 while(cur_txt[j] != '\0') {
570 if(max_width > 0 && width >= max_width) {
571 return max_width;
572 }
573 letter = lv_text_encoded_next(cur_txt, &j);
574 letter_next = lv_text_encoded_next(&cur_txt[j], NULL);
575 uint32_t letter_w = lv_font_get_glyph_width(font, letter, letter_next);
576 width = width + letter_w + letter_space;
577 }
578 }
579
580 return width - letter_space;
581}

583int32_t lv_spangroup_get_expand_height(lv_obj_t * obj, int32_t width)
584{
585 LV_CHECK_OBJ(obj, MY_CLASS, return 0);
586 lv_spangroup_t * spans = (lv_spangroup_t *)obj;
587 if(lv_ll_get_head(&spans->child_ll) == NULL || width <= 0) {
588 return 0;
589 }
590
591 /* init draw variable */
592 lv_text_flag_t txt_flag = LV_TEXT_FLAG_NONE;
593 int32_t line_space = lv_obj_get_style_text_line_space_internal(obj, LV_PART_MAIN);
594 int32_t max_width = width;
595 int32_t indent = convert_indent_pct(obj, max_width);
596 int32_t max_w = max_width - indent; /* first line need minus indent */
597
598 /* coords of draw span-txt */
599 lv_point_t txt_pos;
600 lv_point_set(&txt_pos, indent, 0); /* first line need add indent */
601
602 lv_span_t * cur_span = lv_ll_get_head(&spans->child_ll);
603 const char * cur_txt = cur_span->txt;
604 SPAN_TEXT_CHECK(&cur_txt);
605 uint32_t cur_txt_ofs = 0;
606 lv_snippet_t snippet; /* use to save cur_span info and push it to stack */
607 lv_memset(&snippet, 0, sizeof(snippet));
608
609 lv_span_t * prev_span = cur_span;
610 int32_t line_cnt = 0;
611 int32_t lines = spans->lines < 0 ? INT32_MAX : spans->lines;
612 /* the loop control how many lines need to draw */
613 while(cur_span) {
614 int snippet_cnt = 0;
615 int32_t max_line_h = 0; /* the max height of span-font when a line have a lot of span */
616
617 /* the loop control to find a line and push the relevant span info into stack */
618 while(1) {
619 /* switch to the next span when current is end */
620 if(cur_txt[cur_txt_ofs] == '\0') {
621 cur_span->trailing_pos = txt_pos;
622
623 cur_span = lv_ll_get_next(&spans->child_ll, cur_span);
624 if(cur_span == NULL) break;
625 cur_txt = cur_span->txt;
626 SPAN_TEXT_CHECK(&cur_txt);
627 cur_txt_ofs = 0;
628 /* maybe also cur_txt[cur_txt_ofs] == '\0' */
629 continue;
630 }
631
632 /* init span info to snippet. */
633 if(cur_txt_ofs == 0) {
634 snippet.span = cur_span;
635 snippet.font = lv_span_get_style_text_font(obj, cur_span);
636 snippet.letter_space = lv_span_get_style_text_letter_space(obj, cur_span);
637 snippet.line_h = lv_font_get_line_height_internal(snippet.font) + line_space;
638 }
639
640 /* get current span text line info */
641 uint32_t next_ofs = 0;
642 int32_t use_width = 0;
643 bool isfill = lv_text_get_snippet(&cur_txt[cur_txt_ofs], snippet.font, snippet.letter_space,
644 max_w, txt_flag, &use_width, &next_ofs);
645 if(isfill) txt_pos.x = 0;
646 else txt_pos.x += use_width;
647
648 /* break word deal width */
649 if(isfill && next_ofs > 0 && snippet_cnt > 0) {
650 int32_t drawn_width = use_width;
651 if(lv_ll_get_next(&spans->child_ll, cur_span) == NULL) {
652 drawn_width -= snippet.letter_space;
653 }
654 if(max_w < drawn_width) {
655 break;
656 }
657
658 uint32_t tmp_ofs = next_ofs;
659 uint32_t letter = lv_text_encoded_prev(&cur_txt[cur_txt_ofs], &tmp_ofs);
660 uint32_t letter_next = lv_text_encoded_next(&cur_txt[cur_txt_ofs + next_ofs], NULL);
661 if(!(letter == '\0' || letter == '\n' || letter == '\r' || lv_text_is_break_char(letter) ||
662 lv_text_is_a_word(letter) || lv_text_is_a_word(letter_next))) {
663 if(!(letter_next == '\0' || letter_next == '\n' || letter_next == '\r' || lv_text_is_break_char(letter_next))) {
664 break;
665 }
666 }
667 }
668
669 snippet.txt = &cur_txt[cur_txt_ofs];
670 snippet.bytes = next_ofs;
671 snippet.txt_w = use_width;
672 cur_txt_ofs += next_ofs;
673 if(max_line_h < snippet.line_h) {
674 max_line_h = snippet.line_h;
675 }
676 snippet_cnt ++;
677 max_w = max_w - use_width;
678 if(isfill || max_w <= 0) {
679 break;
680 }
681 }
682
683 /* next line init */
684 txt_pos.y += max_line_h;
685
686 /* iterate all the spans in the current line and set the trailing height to the max line height */
687 for(lv_span_t * tmp_span = prev_span;
688 tmp_span && tmp_span != cur_span;
689 tmp_span = lv_ll_get_next(&spans->child_ll, tmp_span))
690 tmp_span->trailing_height = max_line_h;
691
692 prev_span = cur_span;
693
694 max_w = max_width;
695 line_cnt += 1;
696 if(line_cnt >= lines) {
697 break;
698 }
699 }
700 txt_pos.y -= line_space;
701
702 return txt_pos.y;
703}

871{
872 LV_UNUSED(class_p);
873
874 LV_ASSERT(e != NULL);
875 /* Call the ancestor's event handler */
876 if(lv_obj_event_base(MY_CLASS, e) != LV_RESULT_OK) return;
877
878 lv_event_code_t code = lv_event_get_code(e);
879 lv_obj_t * obj = lv_event_get_current_target(e);
880 LV_ASSERT(obj != NULL);
881 lv_spangroup_t * spans = (lv_spangroup_t *)obj;
882
883 if(code == LV_EVENT_DRAW_MAIN) {
884 draw_main(e);
885 }
886 else if(code == LV_EVENT_STYLE_CHANGED) {
887 lv_spangroup_refresh(obj);
888 }
889 else if(code == LV_EVENT_SIZE_CHANGED) {
890 lv_spangroup_refresh(obj);
891 }
892 else if(code == LV_EVENT_GET_SELF_SIZE) {
893 int32_t width = 0;
894 int32_t height = 0;
895 lv_point_t * self_size = lv_event_get_param(e);
896
897 lv_span_mode_t mode = lv_spangroup_get_mode(obj);
898 if(mode == LV_SPAN_MODE_EXPAND) {
899 if(spans->refresh) {
900 spans->cache_w = (int32_t)lv_spangroup_get_expand_width(obj, 0);
901 spans->cache_h = lv_spangroup_get_max_line_height(obj);
902 spans->refresh = 0;
903 }
904 width = spans->cache_w;
905 height = spans->cache_h;
906 }
907 else if(mode == LV_SPAN_MODE_BREAK) {
908 width = lv_obj_get_content_width(obj);
909 if(self_size->y >= 0) {
910 if(width != spans->cache_w || spans->refresh) {
911 height = lv_spangroup_get_expand_height(obj, width);
912 spans->cache_w = width;
913 spans->cache_h = height;
914 spans->refresh = 0;
915 }
916 else {
917 height = spans->cache_h;
918 }
919 }
920 }
921 else if(mode == LV_SPAN_MODE_FIXED) {
922 width = self_size->x >= 0 ? lv_obj_get_content_width(obj) : 0;
923 height = self_size->y >= 0 ? lv_obj_get_content_height(obj) : 0;
924 }
925 self_size->x = LV_MAX(self_size->x, width);
926 self_size->y = LV_MAX(self_size->y, height);
927 }
928}

1108static void lv_draw_span(lv_obj_t * obj, lv_layer_t * layer)
1109{
1110 LV_ASSERT(obj != NULL);
1111 LV_ASSERT(layer != NULL);
1112 lv_area_t coords;
1113 lv_obj_get_content_coords(obj, &coords);
1114
1115 lv_spangroup_t * spans = (lv_spangroup_t *)obj;
1116
1117 /* return if not span */
1118 if(lv_ll_get_head(&spans->child_ll) == NULL) {
1119 return;
1120 }
1121
1122 /* return if no draw area */
1123 lv_area_t clip_area;
1124 if(!lv_area_intersect(&clip_area, &coords, &layer->_clip_area)) return;
1125 const lv_area_t clip_area_ori = layer->_clip_area;
1126 layer->_clip_area = clip_area;
1127
1128 /* init draw variable */
1129 lv_text_flag_t txt_flag = LV_TEXT_FLAG_NONE;
1130 int32_t line_space = lv_obj_get_style_text_line_space_internal(obj, LV_PART_MAIN);
1131 int32_t max_width = lv_area_get_width(&coords);
1132 int32_t indent = convert_indent_pct(obj, max_width);
1133 int32_t max_w = max_width - indent; /* first line need minus indent */
1134 lv_opa_t obj_opa = lv_obj_get_style_opa_recursive(obj, LV_PART_MAIN);
1135
1136 /* coords of draw span-txt */
1137 lv_point_t txt_pos;
1138 txt_pos.y = coords.y1;
1139 txt_pos.x = coords.x1 + indent; /* first line need add indent */
1140
1141 lv_span_t * cur_span = lv_ll_get_head(&spans->child_ll);
1142 const char * cur_txt = cur_span->txt;
1143 SPAN_TEXT_CHECK(&cur_txt);
1144
1145 lv_text_align_t align = lv_obj_get_style_text_align_internal(obj, LV_PART_MAIN);
1146#if LV_USE_BIDI
1147 lv_base_dir_t base_dir = lv_obj_get_style_base_dir_internal(obj, LV_PART_MAIN);
1148 if(base_dir == LV_BASE_DIR_AUTO) {
1149 base_dir = lv_bidi_detect_base_dir(cur_txt) == LV_BASE_DIR_RTL ? LV_BASE_DIR_RTL : LV_BASE_DIR_AUTO;
1150 }
1151
1152 while(cur_span) {
1153 cur_span = lv_ll_get_next(&spans->child_ll, cur_span);
1154 if(cur_span == NULL) break;
1155 cur_txt = cur_span->txt;
1156 SPAN_TEXT_CHECK(&cur_txt);
1157
1158 if(base_dir == LV_BASE_DIR_AUTO) {
1159 base_dir = lv_bidi_detect_base_dir(cur_txt) == LV_BASE_DIR_RTL ? LV_BASE_DIR_RTL : LV_BASE_DIR_AUTO;
1160 }
1161 }
1162 cur_span = lv_ll_get_head(&spans->child_ll);
1163 cur_txt = cur_span->txt;
1164#endif
1165
1166 uint32_t cur_txt_ofs = 0;
1167 lv_snippet_t snippet; /* use to save cur_span info and push it to stack */
1168 lv_memzero(&snippet, sizeof(snippet));
1169
1170 lv_draw_label_dsc_t label_draw_dsc;
1171 lv_draw_label_dsc_init(&label_draw_dsc);
1172
1173 bool is_first_line = true;
1174 /* the loop control how many lines need to draw */
1175 while(cur_span) {
1176 bool is_end_line = false;
1177 bool ellipsis_valid = false;
1178 int32_t max_line_h = 0; /* the max height of span-font when a line have a lot of span */
1179 int32_t max_baseline = 0; /*baseline of the highest span*/
1180 lv_snippet_clear();
1181
1182 /* the loop control to find a line and push the relevant span info into stack */
1183 while(1) {
1184 /* switch to the next span when current is end */
1185 if(cur_txt[cur_txt_ofs] == '\0') {
1186 cur_span = lv_ll_get_next(&spans->child_ll, cur_span);
1187 if(cur_span == NULL) break;
1188 cur_txt = cur_span->txt;
1189 SPAN_TEXT_CHECK(&cur_txt);
1190 cur_txt_ofs = 0;
1191 /* maybe also cur_txt[cur_txt_ofs] == '\0' */
1192 continue;
1193 }
1194
1195 /* init span info to snippet. */
1196 if(cur_txt_ofs == 0) {
1197 snippet.span = cur_span;
1198 snippet.font = lv_span_get_style_text_font(obj, cur_span);
1199 snippet.letter_space = lv_span_get_style_text_letter_space(obj, cur_span);
1200 snippet.line_h = lv_font_get_line_height_internal(snippet.font) + line_space;
1201 }
1202
1203 /* get current span text line info */
1204 uint32_t next_ofs = 0;
1205 int32_t use_width = 0;
1206 bool isfill = lv_text_get_snippet(&cur_txt[cur_txt_ofs], snippet.font, snippet.letter_space,
1207 max_w, txt_flag, &use_width, &next_ofs);
1208
1209 if(isfill) {
1210 if(next_ofs > 0 && lv_get_snippet_count() > 0) {
1211 int32_t drawn_width = use_width;
1212 if(lv_ll_get_next(&spans->child_ll, cur_span) == NULL) {
1213 drawn_width -= snippet.letter_space;
1214 }
1215 /* To prevent infinite loops, the lv_text_get_next_line() may return incomplete words, */
1216 /* This phenomenon should be avoided when lv_get_snippet_count() > 0 */
1217 if(max_w < drawn_width) {
1218 break;
1219 }

1221 uint32_t letter = lv_text_encoded_prev(&cur_txt[cur_txt_ofs], &tmp_ofs);
