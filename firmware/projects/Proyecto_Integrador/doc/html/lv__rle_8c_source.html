37uint32_t lv_rle_decompress(const uint8_t * input,
38 uint32_t input_buff_len, uint8_t * output,
39 uint32_t output_buff_len, uint8_t blk_size)
40{
41 uint32_t ctrl_byte;
42 uint32_t rd_len = 0;
43 uint32_t wr_len = 0;
44
45 while(rd_len < input_buff_len) {
46 ctrl_byte = input[0];
47 rd_len++;
48 input++;
49 if(rd_len > input_buff_len)
50 return 0;
51
52 if(ctrl_byte & 0x80) {
53 /* copy directly from input to output */
54 uint32_t bytes = blk_size * (ctrl_byte & 0x7f);
55 rd_len += bytes;
56 if(rd_len > input_buff_len)
57 return 0;
58
59 wr_len += bytes;
60 if(wr_len > output_buff_len) {
61 if(wr_len > output_buff_len + blk_size)
62 return 0; /* Error */
63 lv_memcpy(output, input, output_buff_len - (wr_len - bytes));
64 return output_buff_len;
65 }
66
67 lv_memcpy(output, input, bytes);
68 output += bytes;
69 input += bytes;
70 }
71 else {
72 rd_len += blk_size;
73 if(rd_len > input_buff_len)
74 return 0;
75
76 wr_len += blk_size * ctrl_byte;
77 if(wr_len > output_buff_len) {
78 if(wr_len > output_buff_len + blk_size)
79 return 0; /* Error happened */
80
81 /* Skip the last pixel, which could overflow output buffer.*/
82 for(uint32_t i = 0; i < ctrl_byte - 1; i++) {
83 lv_memcpy(output, input, blk_size);
84 output += blk_size;
85 }
86 return output_buff_len;
87 }
88
89 if(blk_size == 1) {
90 /* optimize the most common case. */
91 lv_memset(output, input[0], ctrl_byte);
92 output += ctrl_byte;
93 }
94 else {
95 for(uint32_t i = 0; i < ctrl_byte; i++) {
96 lv_memcpy(output, input, blk_size);
97 output += blk_size;
98 }
99 }
100 input += blk_size;
101 }
102 }
103
104 return wr_len;
105}
