42def parse_kconfig_content(path):
43 with open(path, "r") as f:
44 lines = f.readlines()
45
46 SOURCE_RE = re.compile(r'^(?!.*#.*\brsource\b)\s*rsource\s+"([^"]+)"')
47
48 content = []
49 for line in lines:
50 m = re.match(SOURCE_RE, line)
51 if not m:
52 content.append(line)
53 continue
54 relpath = m.group(1)
55 dir_name = os.path.dirname(path)
56 sourced_path = os.path.join(dir_name, relpath)
57 content.extend(parse_kconfig_content(sourced_path))
58
59 return content
