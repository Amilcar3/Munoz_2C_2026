24def _collect_simple(lvgl, dict_key: str, accessor: str | None, label: str) -> list:
25 """Collect a simple subsystem using the registry entry.
26
27 For entries with accessor != None: [x.snapshot().as_dict() for x in lvgl.<accessor>()]
28 For entries with accessor == None: [s.as_dict() for s in lvgl.<dict_key>().snapshots()]
29 """
30
31 @safe_collect(label)
32 def _inner():
33 if accessor is not None:
34 return [x.snapshot().as_dict() for x in getattr(lvgl, accessor)()]
35 return [s.as_dict() for s in getattr(lvgl, dict_key)().snapshots()]
36
37 return _inner()

67 """Try to read LVGL version from macros."""

77def _buf_to_dict(draw_buf) -> dict | None:
78 """Convert an LVDrawBuf to a dict with base64 PNG image."""
79 if draw_buf is None:
80 return None
81 try:
82 cf_info = draw_buf.color_format_info()
83 header = draw_buf.super_value("header")
84 stride = int(header["stride"])
85 height = int(header["h"])
86 bpp = cf_info["bpp"]
87 width = (stride * 8) // bpp if bpp else 0
88
89 png_bytes = draw_buf.to_png_bytes()
90 image_b64 = base64.b64encode(png_bytes).decode("ascii") if png_bytes else None
91
92 return {
93 "addr": hex(int(draw_buf)),
94 "width": width,
95 "height": height,
96 "color_format": cf_info["name"],
97 "data_size": int(draw_buf.super_value("data_size")),
98 "image_base64": image_b64,
99 }
100 except Exception:
101 return None

195def _collect_subjects_from_obj(obj, seen, result):
196 """Recursively collect subjects from an object's event list."""
197 from lvglgdb.lvgl.core.lv_observer import LVSubject
198
199 event_list = obj.event_list
200 if event_list:
201 for dsc in event_list:
202 try:
203 user_data = dsc.user_data
204 if not int(user_data):
205 continue
206 subject = LVSubject(user_data)
207 addr = int(subject)
208 if addr not in seen:
209 seen.add(addr)
210 result.append(subject.snapshot().as_dict())
211 except Exception:
212 # Per-dsc failure: skip this descriptor, continue others
213 continue
214
215 try:
216 for child in obj.children:
217 try:
218 _collect_subjects_from_obj(child, seen, result)
219 except Exception:
220 # Per-child failure: skip this child, continue siblings
221 continue
222 except CorruptedError:
223 # Children pointer unreadable: stop traversal for this subtree
224 pass

229 import json as _json
230 from pathlib import Path

232 specs_path = Path(__file__).parent.parent.parent.parent / "scripts" / "generators" / "widget_specs.json"
233 if not specs_path.exists():

237 all_specs = _json.loads(specs_path.read_text())
238 except Exception:
239 return {}

241 # Collect all class_names from object trees
242 seen = set()
243
244 def _walk(node):
245 cn = node.get("class_name", "")
246 if cn:
247 seen.add(cn)
248 for child in node.get("children", []):
249 _walk(child)
250
251 for tree in object_trees:
252 for screen in tree.get("screens", []):
253 _walk(screen)

255 # Build output: merge _auto + enums into flat fields spec

273 spec["summary_tpl"] = raw["summary_tpl"]
274 if raw.get("primary"):
275 spec["primary"] = raw["primary"]
276 if fields:
277 spec["fields"] = fields
