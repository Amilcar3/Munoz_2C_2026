91int32_t nema_sys_init(void)
92{
93 int error_code = 0;
94
95 lv_thread_sync_init(&sync);
96
97 /* Setup GPU2D Callback */
98#if (USE_HAL_GPU2D_REGISTER_CALLBACKS == 1)
99 /* Register Command List Complete Callback */
100 HAL_GPU2D_RegisterCommandListCpltCallback(&hgpu2d, GPU2D_CommandListCpltCallback);
101#endif
102
103 /* Initialise Mem Space */
104 error_code = tsi_malloc_init_pool_aligned(0, (void *)nemagfx_pool_mem, (uintptr_t)nemagfx_pool_mem,
105 NEMAGFX_MEM_POOL_SIZE, 1, 8);
106 LV_ASSERT(error_code == 0);
107
108 /* Allocate ring_buffer memory */
109 ring_buffer_str.bo = nema_buffer_create(RING_SIZE);
110 LV_ASSERT(ring_buffer_str.bo.base_virt);
111
112 /* Initialize Ring Buffer */
113 error_code = nema_rb_init(&ring_buffer_str, 1);
114 if(error_code < 0) {
115 return error_code;
116 }
117
118 /* Reset last_cl_id counter */
119 last_cl_id = 0;
120
121 return error_code;
122}
