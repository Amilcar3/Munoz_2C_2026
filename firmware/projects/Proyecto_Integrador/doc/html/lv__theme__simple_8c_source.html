145lv_theme_t * lv_theme_simple_init(lv_display_t * disp)
146{
147 /*This trick is required only to avoid the garbage collection of
148 *styles' data if LVGL is used in a binding (e.g. MicroPython)
149 *In a general case styles could be in a simple `static lv_style_t my_style...` variables*/
150 if(!lv_theme_simple_is_inited()) {
151 theme_def = lv_malloc_zeroed(sizeof(my_theme_t));
152 LV_ASSERT_MALLOC(theme_def);
153 }
154
155 my_theme_t * theme = theme_def;
156
157 theme->base.disp = disp;
158 theme->base.font_small = LV_FONT_DEFAULT;
159 theme->base.font_normal = LV_FONT_DEFAULT;
160 theme->base.font_large = LV_FONT_DEFAULT;
161 theme->base.apply_cb = theme_apply;
162#if LV_USE_EXT_DATA
163 theme->base.ext_data.free_cb = NULL;
164 theme->base.ext_data.data = NULL;
165#endif
166
167 style_init(theme);
168
169 if(disp == NULL || lv_display_get_theme(disp) == (lv_theme_t *)theme) {
170 lv_obj_report_style_change(NULL);
171 }
172
173 theme->inited = true;
174
175 return (lv_theme_t *)theme_def;
176}

178bool lv_theme_simple_is_inited(void)
179{
180 my_theme_t * theme = theme_def;
181 if(theme == NULL) return false;
182 return theme->inited;
183}

194void lv_theme_simple_deinit(void)
195{
196 my_theme_t * theme = theme_def;
197 if(theme) {
198 if(theme->inited) {
199 lv_style_t * theme_styles = (lv_style_t *)(&(theme->styles));
200 uint32_t i;
201 for(i = 0; i < sizeof(my_theme_styles_t) / sizeof(lv_style_t); i++) {
202 lv_style_reset(theme_styles + i);
203 }
204 }
205#if LV_USE_EXT_DATA
206 if(theme->base.ext_data.free_cb) {
207 theme->base.ext_data.free_cb(theme->base.ext_data.data);
208 theme->base.ext_data.data = NULL;
209 }
210#endif
211 lv_free(theme_def);
212 theme_def = NULL;
213 }
