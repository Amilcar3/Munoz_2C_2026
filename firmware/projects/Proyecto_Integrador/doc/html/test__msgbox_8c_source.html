154void test_msgbox_content_auto_height(void)
155{
156 /* If parent is NULL the message box will be modal*/
157 msgbox = lv_msgbox_create(NULL);
158 lv_msgbox_add_title(msgbox, "The title");
159 lv_msgbox_add_text(msgbox, "The text");
160 lv_msgbox_add_text_fmt(msgbox, "The %s text", "fmt");
161 lv_msgbox_add_footer_button(msgbox, "Apply");
162 lv_msgbox_add_footer_button(msgbox, "Close");
163 lv_msgbox_add_header_button(msgbox, LV_SYMBOL_AUDIO);
164 lv_msgbox_add_close_button(msgbox);
165
166 /* Test1 : msgbox's height is LV_SIZE_CONTENT by default */
167 bool is_height_size_content = (lv_obj_get_style_height(msgbox, LV_PART_MAIN) == LV_SIZE_CONTENT);
168 TEST_ASSERT_EQUAL(is_height_size_content, 1);
169
170 lv_obj_update_layout(msgbox);
171 lv_obj_t * header = lv_msgbox_get_header(msgbox);
172 lv_obj_t * footer = lv_msgbox_get_footer(msgbox);
173 lv_obj_t * content = lv_msgbox_get_content(msgbox);
174
175 int32_t h_header = (header == NULL) ? 0 : lv_obj_get_height(header);
176 int32_t h_footer = (footer == NULL) ? 0 : lv_obj_get_height(footer);
177 int32_t h_content = lv_obj_get_height(content);
178
179 int32_t h_obj_content = lv_obj_get_content_height(msgbox);
180 int32_t h_msgbox_element_sum = h_header + h_footer + h_content;
181 /* Default Size : The height of the msgbox's obj-content should be equal to the total height of the msgbox's element. */
182 TEST_ASSERT_EQUAL(h_obj_content, h_msgbox_element_sum);
183
184 /* Test2 : Now change size of msgbox manually*/
185 lv_obj_set_size(msgbox, lv_pct(80), lv_pct(80));
186
187 is_height_size_content = (lv_obj_get_style_height(msgbox, LV_PART_MAIN) == LV_SIZE_CONTENT);
188 TEST_ASSERT_EQUAL(is_height_size_content, 0);
189
190 lv_obj_update_layout(msgbox);
191 h_header = (header == NULL) ? 0 : lv_obj_get_height(header);
192 h_footer = (footer == NULL) ? 0 : lv_obj_get_height(footer);
193 h_content = lv_obj_get_height(content);
194
195 h_obj_content = lv_obj_get_content_height(msgbox);
196 h_msgbox_element_sum = h_header + h_footer + h_content;
197 /* Manual Size : The height of the msgbox's obj-content should also be equal to the total height of the msgbox's element. */
198 TEST_ASSERT_EQUAL(h_obj_content, h_msgbox_element_sum);
199}
