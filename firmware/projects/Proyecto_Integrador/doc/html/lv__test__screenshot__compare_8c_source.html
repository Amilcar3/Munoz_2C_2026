90lv_test_screenshot_result_t lv_test_screenshot_compare_core(const char * fn_ref)
91{
92 LV_CHECK_ARG(fn_ref != NULL, return LV_TEST_SCREENSHOT_RESULT_FAILED);
93
94 char fn_ref_full[256];
95 lv_snprintf(fn_ref_full, sizeof(fn_ref_full), "%s%s", REF_IMGS_PATH, fn_ref);
96
97#if LV_TEST_SCREENSHOT_CREATE_REFERENCE_IMAGE
98 create_folders_if_needed(fn_ref_full);
99#endif
100
101 lv_draw_buf_t * draw_buf = lv_display_get_buf_active(lv_display_get_default());
102
103 uint8_t * screen_buf_xrgb8888 = lv_malloc(draw_buf->header.w * draw_buf->header.h * 4);
104 if(!screen_buf_xrgb8888) {
105 LV_LOG_ERROR("Not enough memory to compare display with screenshot");
106 return LV_TEST_SCREENSHOT_RESULT_FAILED;
107 }
108
109 buf_to_xrgb8888(draw_buf, screen_buf_xrgb8888);
110
111 lv_draw_buf_t * ref_draw_buf = NULL;
112 unsigned ref_img_width = 0;
113 unsigned ref_img_height = 0;
114 unsigned res = read_png_file(&ref_draw_buf, &ref_img_width, &ref_img_height, fn_ref_full);
115 if(res) {
116 lv_test_screenshot_result_t comp_res;
117#if LV_TEST_SCREENSHOT_CREATE_REFERENCE_IMAGE
118 LV_LOG_WARN("%s%s", fn_ref_full, " was not found, creating it now from the rendered screen");
119 write_png_file(screen_buf_xrgb8888, draw_buf->header.w, draw_buf->header.h, fn_ref_full);
120 comp_res = LV_TEST_SCREENSHOT_RESULT_PASSED;
121#else
122 comp_res = LV_TEST_SCREENSHOT_RESULT_NO_REFERENCE_IMAGE;
123#endif
124 lv_free(screen_buf_xrgb8888);
125 if(ref_draw_buf) lv_draw_buf_destroy(ref_draw_buf);
126 return comp_res;
127 }
128
129 if(ref_img_width != draw_buf->header.w || ref_img_height != draw_buf->header.h) {
130 LV_LOG_WARN("The dimensions of the rendered and the %s reference image don't match", fn_ref);
131 lv_free(screen_buf_xrgb8888);
132 if(ref_draw_buf) lv_draw_buf_destroy(ref_draw_buf);
133 return LV_TEST_SCREENSHOT_RESULT_FAILED;
134 }
135
136
137 unsigned x, y;
138 bool err = false;
139 for(y = 0; y < ref_img_height; y++) {
140 uint8_t * screen_buf_tmp = screen_buf_xrgb8888 + draw_buf->header.w * 4 * y;
141 uint8_t * ref_row = (uint8_t *)ref_draw_buf->data + y * ref_draw_buf->header.stride;
142 for(x = 0; x < ref_img_width; x++) {
143 uint8_t * ptr_ref = &(ref_row[x * 4]);
144 uint8_t * ptr_act = &screen_buf_tmp[x * 4];
145
146 if(LV_ABS((int32_t) ptr_act[0] - (int32_t) ptr_ref[0]) > REF_IMG_TOLERANCE ||
147 LV_ABS((int32_t) ptr_act[1] - (int32_t) ptr_ref[1]) > REF_IMG_TOLERANCE ||
148 LV_ABS((int32_t) ptr_act[2] - (int32_t) ptr_ref[2]) > REF_IMG_TOLERANCE) {
149 err = true;
150#if LV_USE_LOG
151 uint32_t act_px = (ptr_act[2] << 16) + (ptr_act[1] << 8) + (ptr_act[0] << 0);
152 uint32_t ref_px = 0;
153 memcpy(&ref_px, ptr_ref, 3);
154 LV_LOG("\nScreenshot compare error\n"
155 " - File: %s\n"
156 " - At x:%d, y:%d.\n"
157 " - Expected: %X\n"
158 " - Actual: %X\n"
159 " - Tolerance: %d\n",
160 fn_ref_full, x, y, ref_px, act_px, REF_IMG_TOLERANCE);
161#endif
162 break;
163 }
164 }
165 if(err) break;
166 }
167
168 if(err) {
169 char fn_ref_no_ext[128];
170 lv_strlcpy(fn_ref_no_ext, fn_ref, sizeof(fn_ref_no_ext));
171 fn_ref_no_ext[strlen(fn_ref_no_ext) - 4] = '\0';
172
173 char fn_err_full[256];
174 lv_snprintf(fn_err_full, sizeof(fn_err_full), "%s%s_err.png", REF_IMGS_PATH, fn_ref_no_ext);
175
176 write_png_file(screen_buf_xrgb8888, draw_buf->header.w, draw_buf->header.h, fn_err_full);
177 }
178
179 fflush(stdout);
180 lv_free(screen_buf_xrgb8888);
181 if(ref_draw_buf) lv_draw_buf_destroy(ref_draw_buf);
182 return err ? LV_TEST_SCREENSHOT_RESULT_FAILED : LV_TEST_SCREENSHOT_RESULT_PASSED;
183
184}

211static void buf_to_xrgb8888(const lv_draw_buf_t * draw_buf, uint8_t * buf_out)
212{
213 LV_ASSERT(draw_buf != NULL);
214 LV_ASSERT(buf_out != NULL);
215 uint32_t stride = draw_buf->header.stride;
216 lv_color_format_t cf_in = draw_buf->header.cf;
217 const uint8_t * buf_in = draw_buf->data;
218
219 if(cf_in == LV_COLOR_FORMAT_RGB565 || cf_in == LV_COLOR_FORMAT_RGB565_SWAPPED) {
220 if(cf_in == LV_COLOR_FORMAT_RGB565_SWAPPED) {
221 lv_draw_rgb565_swap(draw_buf->data, draw_buf->header.w * draw_buf->header.h);
222 }
223 uint32_t y;
224 for(y = 0; y < draw_buf->header.h; y++) {
225
226 uint32_t x;
227 for(x = 0; x < draw_buf->header.w; x++) {
228 const lv_color16_t * c16 = (const lv_color16_t *)&buf_in[x * 2];
229
230 buf_out[x * 4 + 3] = 0xff;
231 buf_out[x * 4 + 2] = (c16->blue * 2106) >> 8; /*To make it rounded*/
232 buf_out[x * 4 + 1] = (c16->green * 1037) >> 8;
233 buf_out[x * 4 + 0] = (c16->red * 2106) >> 8;
234 }
235
236 buf_in += stride;
237 buf_out += draw_buf->header.w * 4;
238 }
239 }
240 else if(cf_in == LV_COLOR_FORMAT_ARGB8888 || cf_in == LV_COLOR_FORMAT_XRGB8888 ||
241 cf_in == LV_COLOR_FORMAT_ARGB8888_PREMULTIPLIED) {
242 uint32_t y;
243 for(y = 0; y < draw_buf->header.h; y++) {
244 uint32_t x;
245 for(x = 0; x < draw_buf->header.w; x++) {
246 buf_out[x * 4 + 3] = buf_in[x * 4 + 3];
247 buf_out[x * 4 + 2] = buf_in[x * 4 + 0];
248 buf_out[x * 4 + 1] = buf_in[x * 4 + 1];
249 buf_out[x * 4 + 0] = buf_in[x * 4 + 2];
250 }
251
252 buf_in += stride;
253 buf_out += draw_buf->header.w * 4;
254 }
255 }
256 else if(cf_in == LV_COLOR_FORMAT_RGB888) {
257 uint32_t y;
258 for(y = 0; y < draw_buf->header.h; y++) {
259 uint32_t x;
260 for(x = 0; x < draw_buf->header.w; x++) {
261 buf_out[x * 4 + 3] = 0xff;
262 buf_out[x * 4 + 2] = buf_in[x * 3 + 0];
263 buf_out[x * 4 + 1] = buf_in[x * 3 + 1];
264 buf_out[x * 4 + 0] = buf_in[x * 3 + 2];
265 }
266
267 buf_in += stride;
268 buf_out += draw_buf->header.w * 4;
269 }
270 }
271 else if(cf_in == LV_COLOR_FORMAT_L8) {
272 uint32_t y;
273 for(y = 0; y < draw_buf->header.h; y++) {
274 uint32_t x;
275 for(x = 0; x < draw_buf->header.w; x++) {
276 buf_out[x * 4 + 3] = 0xff;
277 buf_out[x * 4 + 2] = buf_in[x];
278 buf_out[x * 4 + 1] = buf_in[x];
279 buf_out[x * 4 + 0] = buf_in[x];
280 }
281
282 buf_in += stride;
283 buf_out += draw_buf->header.w * 4;
284 }
285 }
286 else if(cf_in == LV_COLOR_FORMAT_AL88) {
287 uint32_t y;
288 for(y = 0; y < draw_buf->header.h; y++) {
289 uint32_t x;
290 for(x = 0; x < draw_buf->header.w; x++) {
291 buf_out[x * 4 + 3] = buf_in[x * 2 + 1];
292 buf_out[x * 4 + 2] = buf_in[x * 2 + 0];
293 buf_out[x * 4 + 1] = buf_in[x * 2 + 0];
294 buf_out[x * 4 + 0] = buf_in[x * 2 + 0];
295 }
296
297 buf_in += stride;
298 buf_out += draw_buf->header.w * 4;
299 }
300 }
301 else if(cf_in == LV_COLOR_FORMAT_I1) {
302 buf_in += 8;
303 uint32_t y;
304 for(y = 0; y < draw_buf->header.h; y++) {
305 uint32_t x;
306 for(x = 0; x < draw_buf->header.w; x++) {
307 const uint8_t byte = buf_in[x / 8] ;
308 const uint8_t bit_pos = x % 8;
309 const uint8_t pixel = (byte >> (7 - bit_pos)) & 0x01;
310
311 buf_out[x * 4 + 3] = 0xff;
312 buf_out[x * 4 + 2] = pixel ? 0xff : 0x00;
313 buf_out[x * 4 + 1] = pixel ? 0xff : 0x00;
314 buf_out[x * 4 + 0] = pixel ? 0xff : 0x00;
315 }
316
317 buf_in += stride;
318 buf_out += draw_buf->header.w * 4;
319 }
320 }
321}

323static void create_folders_if_needed(const char * path)
324{
325 LV_ASSERT(path != NULL);
326 char * ptr;
327 char * path_copy = lv_strdup(path);
328 if(path_copy == NULL) {
329 LV_LOG_ERROR("Error duplicating path");
330 exit(EXIT_FAILURE);
331 }
332
333 char * token = strtok_r(path_copy, "/", &ptr);
334 char current_path[1024] = {'\0'}; /* Adjust the size as needed */
335
336 while(token && ptr && *ptr != '\0') {
337 lv_strcat(current_path, token);
338 lv_strcat(current_path, "/");
339
340 int mkdir_retval = mkdir(current_path, 0777);
341 if(mkdir_retval == 0) {
342 LV_LOG_INFO("Created folder: %s\n", current_path);
343 }
344 else if(errno != EEXIST) {
345 perror("Error creating folder");
346 lv_free(path_copy);
347 exit(EXIT_FAILURE);
348 }
349
350 token = strtok_r(NULL, "/", &ptr);
351 }
352
353 lv_free(path_copy);
354}
