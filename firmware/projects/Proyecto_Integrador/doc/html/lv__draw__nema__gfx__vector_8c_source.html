48void lv_draw_nema_gfx_vector(lv_draw_task_t * t, const lv_draw_vector_dsc_t * dsc,
49 const lv_area_t * coords)
50{
51 ctx_t c;
52 c.u = (lv_draw_nema_gfx_unit_t *) t->draw_unit;
53
54 lv_layer_t * layer = t->target_layer;
55 lv_area_t rel_clip_area;
56 rel_clip_area = t->clip_area;
57 lv_area_move(&rel_clip_area, -layer->buf_area.x1, -layer->buf_area.y1);
58
59 c.rel_translate_x = -layer->buf_area.x1;
60 c.rel_translate_y = -layer->buf_area.y1;
61
62 nema_set_clip(rel_clip_area.x1, rel_clip_area.y1, lv_area_get_width(&rel_clip_area),
63 lv_area_get_height(&rel_clip_area));
64
65 lv_color_format_t dst_cf = layer->draw_buf->header.cf;
66 uint32_t dst_nema_cf = lv_nemagfx_cf_to_nema(dst_cf);
67
68 /* the stride should be computed internally for NEMA_TSC images and images missing a stride value */
69 int32_t stride = (dst_cf >= LV_COLOR_FORMAT_NEMA_TSC_START && dst_cf <= LV_COLOR_FORMAT_NEMA_TSC_END) ?
70 -1 : lv_area_get_width(&(layer->buf_area)) * lv_color_format_get_size(dst_cf);
71
72 nema_bind_dst_tex((uintptr_t)NEMA_VIRT2PHYS(layer->draw_buf->data), lv_area_get_width(&(layer->buf_area)),
73 lv_area_get_height(&(layer->buf_area)), dst_nema_cf, stride);
74
75 nema_vg_set_blend(NEMA_BL_SRC_OVER | NEMA_BLOP_SRC_PREMULT);
76
77 lv_vector_for_each_destroy_tasks(dsc->task_list, task_draw_cb, &c);
78}

84static void task_draw_cb(void * ctx, const lv_vector_path_t * path, const lv_vector_path_ctx_t * dsc)
85{
86 ctx_t * c = ctx;
87 lv_draw_nema_gfx_unit_t * u = c->u;
88
89 if(!path) return;
90
91 nema_vg_path_clear(u->path);
92 nema_vg_paint_clear(u->paint);
93
94 nema_vg_paint_set_type(u->paint, NEMA_VG_PAINT_COLOR);
95
96 lv_matrix_t matrix = dsc->matrix;
97 matrix.m[0][2] += c->rel_translate_x;
98 matrix.m[1][2] += c->rel_translate_y;
99 nema_vg_path_set_matrix(c->u->path, (void *) &matrix);
100
101 /* the path ops array needs to be translated to nema's opcodes */
102 lv_vector_path_op_t * ops = lv_array_front(&path->ops);
103 uint32_t op_count = lv_array_size(&path->ops);
104 uint8_t * nema_ops = lv_malloc(op_count * sizeof(*nema_ops));
105 LV_ASSERT_MALLOC(nema_ops);
106 for(uint32_t i = 0; i < op_count; i++) {
107 nema_ops[i] = ops[i] == LV_VECTOR_PATH_OP_MOVE_TO ? NEMA_VG_PRIM_MOVE :
108 ops[i] == LV_VECTOR_PATH_OP_LINE_TO ? NEMA_VG_PRIM_LINE :
109 ops[i] == LV_VECTOR_PATH_OP_QUAD_TO ? NEMA_VG_PRIM_BEZIER_QUAD :
110 ops[i] == LV_VECTOR_PATH_OP_CUBIC_TO ? NEMA_VG_PRIM_BEZIER_CUBIC :
111 /*LV_VECTOR_PATH_OP_CLOSE*/ NEMA_VG_PRIM_CLOSE;
112 }
113
114 /* the path points array is in the right format for nema to use as-is */
115 uint32_t point_count = lv_array_size(&path->points);
116 float * points = lv_array_front(&path->points);
117
118 nema_vg_path_set_shape(u->path, op_count, nema_ops, point_count, points);
119
120 nema_vg_set_quality(path->quality == LV_VECTOR_PATH_QUALITY_LOW ? NEMA_VG_QUALITY_FASTER :
121 path->quality == LV_VECTOR_PATH_QUALITY_MEDIUM ? NEMA_VG_QUALITY_BETTER :
122 /*LV_VECTOR_PATH_QUALITY_HIGH*/ NEMA_VG_QUALITY_MAXIMUM);
123
124 if(dsc->fill_dsc.opa) {
125 nema_vg_set_fill_rule(dsc->fill_dsc.fill_rule == LV_VECTOR_FILL_NONZERO ? NEMA_VG_FILL_NON_ZERO :
126 /*LV_VECTOR_FILL_EVENODD*/ NEMA_VG_FILL_EVEN_ODD);
127
128 uint32_t nema_dsc_color = nema_rgba(dsc->fill_dsc.color.red, dsc->fill_dsc.color.green,
129 dsc->fill_dsc.color.blue, dsc->fill_dsc.color.alpha);
130 nema_vg_paint_set_paint_color(u->paint, nema_dsc_color);
131
132 nema_vg_draw_path(u->path, u->paint);
133 }
134
135 if(dsc->stroke_dsc.opa) {
136 nema_vg_set_fill_rule(NEMA_VG_STROKE);
137 nema_vg_stroke_set_width(dsc->stroke_dsc.width);
138 uint8_t cap = dsc->stroke_dsc.cap == LV_VECTOR_STROKE_CAP_BUTT ? NEMA_VG_CAP_BUTT :
139 dsc->stroke_dsc.cap == LV_VECTOR_STROKE_CAP_SQUARE ? NEMA_VG_CAP_SQUARE :
140 /*LV_VECTOR_STROKE_CAP_ROUND*/ NEMA_VG_CAP_ROUND;
141 nema_vg_stroke_set_cap_style(cap, cap);
142 uint8_t join = dsc->stroke_dsc.join == LV_VECTOR_STROKE_JOIN_MITER ? NEMA_VG_JOIN_MITER :
143 dsc->stroke_dsc.join == LV_VECTOR_STROKE_JOIN_BEVEL ? NEMA_VG_JOIN_BEVEL :
144 /*LV_VECTOR_STROKE_JOIN_ROUND*/ NEMA_VG_JOIN_ROUND;
145 nema_vg_stroke_set_join_style(join);
146 nema_vg_stroke_set_miter_limit(dsc->stroke_dsc.miter_limit);
147
148 uint32_t nema_dsc_color = nema_rgba(dsc->stroke_dsc.color.red, dsc->stroke_dsc.color.green,
149 dsc->stroke_dsc.color.blue, dsc->stroke_dsc.color.alpha);
150 nema_vg_paint_set_paint_color(u->paint, nema_dsc_color);
151
152 nema_vg_draw_path(u->path, u->paint);
153 }
154
155 nema_cl_submit(&c->u->cl);
156 nema_cl_wait(&u->cl);
157
158 lv_free(nema_ops);
159}
