34# Section order, and the commit types each one collects.
35SECTIONS = [
36 ("Breaking Changes", None), # populated by the breaking flag, not by type
37 ("Architectural", ["arch"]),
38 ("New Features", ["feat"]),
39 ("Fixes", ["fix"]),
40 ("Performance", ["perf"]),
41 ("Examples", ["example", "examples"]),
42 ("Docs", ["doc", "docs"]),
43 ("Build", ["build"]),
44 ("Refactoring", ["refactor", "style"]),

66def confirm_overwrite(path, force):
67 """The page is meant to be hand-edited after generation, so never clobber it silently."""
68 if force or not path.exists():
69 return
70 if not sys.stdin.isatty():
71 sys.exit(f"error: {display(path)} already exists; pass --force to overwrite")
72 answer = input(f"{display(path)} already exists. Overwrite? [y/N] ").strip().lower()
73 if answer not in ("y", "yes"):
74 sys.exit("aborted")

77def escape_mdx(text):
78 """Neutralise characters MDX would read as JSX or as an expression."""
79 text = text.replace("&", "&amp;")
80 text = text.replace("<", "&lt;").replace(">", "&gt;")
81 return text.replace("{", "&#123;").replace("}", "&#125;")

94def previous_version_tag():
95 """Latest vX.Y.Z tag reachable from HEAD."""
96 tags = git("tag", "--list", "--merged", "HEAD", "--sort=-v:refname").split()
97 for tag in tags:

103def parse_subject(subject):
104 """Split a commit subject into (type, breaking, text). type is None if unparseable."""
105 match = SUBJECT_WITH_COLON.match(subject) or SUBJECT_NO_COLON.match(subject)
106 if not match:
107 return None, False, subject
108 breaking = bool(match.group("bang1") or match.group("bang2"))
109 return match.group("type").lower(), breaking, match.group("desc")

112def collect(revision_range):
113 """Return {section: [(hash, subject)]} for every commit in the range."""
114 # %x00/%x01 are expanded by git itself; NUL cannot be passed in argv.
115 log = git(
116 "log",
117 revision_range,
118 "--no-merges",
119 "--reverse",
120 "--format=%H%x00%s%x00%b%x01",
121 )
122
123 sections = {name: [] for name, _ in SECTIONS}
124 for record in log.split("\x01"):
125 record = record.strip("\n")
126 if not record:
127 continue
128 commit_hash, subject, body = record.split("\x00", 2)
129
130 # Subjects sometimes carry a stray leading space, and the squash suffix
131 # "(#1234)" is redundant once we link the commit itself.
132 subject = TRAILING_PR.sub("", subject.strip())
133 if not subject or IGNORED.match(subject):
134 continue
135
136 commit_type, bang, _ = parse_subject(subject)
137 breaking = bang or bool(
138 re.search(r"^BREAKING[ -]CHANGE:\s*", body, re.MULTILINE)
139 )
140 if breaking:
141 section = "Breaking Changes"
142 else:
143 section = TYPE_TO_SECTION.get(commit_type, CATCH_ALL)
144 sections[section].append((commit_hash, subject))
145
146 return sections

149def render(version, when, sections, repo_url, previous_tag):
150 lines = [
151 "---",
152 f"title: {version} Change Log",
153 f"description: Summary of the changes of LVGL {version}",
154 "---",
155 "",
156 f"## {version} ({when})",
157 "",
158 ]
159
160 for name, _ in SECTIONS:
161 entries = sections[name]
162 if not entries:
163 continue
164 lines.append(f"### {name}")
165 lines.append("")
166 for commit_hash, subject in entries:
167 link = f"{repo_url}/commit/{commit_hash}"
168 lines.append(f"- [{commit_hash[:7]}]({link}) - {escape_mdx(subject)}")
169 lines.append("")
170
171 compare = f"{repo_url}/compare/{previous_tag}...{version}"
172 lines.append(f"<sub>Full changes [here]({compare})</sub>")
173 lines.append("")
174 return "\n".join(lines)
