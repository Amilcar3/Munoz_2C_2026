27static void styles_init(void)
28{
29 static bool inited = false;
30 if(inited) return;
31 inited = true;
32
33 /*List background: flatten the base object's card into a tight, edge-clipped column.*/
34 lv_style_init(&style_list);
35 lv_style_set_pad_ver(&style_list, 0);
36 lv_style_set_pad_gap(&style_list, 0);
37 lv_style_set_clip_corner(&style_list, true);
38
39 /*Section header: a full-width grey bar with dark text. The transform makes the
40 *grey reach the list edges despite the list's horizontal padding.*/
41 lv_style_init(&style_header);
42 lv_style_set_bg_opa(&style_header, LV_OPA_COVER);
43 lv_style_set_bg_color(&style_header, lv_palette_lighten(LV_PALETTE_GREY, 2));
44 lv_style_set_text_color(&style_header, lv_palette_darken(LV_PALETTE_GREY, 4));
45 lv_style_set_transform_width(&style_header, 16);
46
47 /*List button: a flat white row with a grey bottom divider, overriding the
48 *default blue button look.*/
49 lv_style_init(&style_button);
50 lv_style_set_radius(&style_button, 0);
51 lv_style_set_shadow_width(&style_button, 0);
52 lv_style_set_bg_opa(&style_button, LV_OPA_COVER);
53 lv_style_set_bg_color(&style_button, lv_color_white());
54 lv_style_set_text_color(&style_button, lv_palette_darken(LV_PALETTE_GREY, 4));
55 lv_style_set_border_width(&style_button, 1);
56 lv_style_set_border_color(&style_button, lv_palette_lighten(LV_PALETTE_GREY, 2));
57 lv_style_set_border_side(&style_button, LV_BORDER_SIDE_BOTTOM);
58 lv_style_set_pad_all(&style_button, 8);
59 lv_style_set_pad_column(&style_button, 8);
60
61 /*Pressed feedback: a subtle dark recolor.*/
62 lv_style_init(&style_button_pressed);
63 lv_style_set_recolor(&style_button_pressed, lv_color_black());
64 lv_style_set_recolor_opa(&style_button_pressed, LV_OPA_20);
65}

93static lv_obj_t * flex_list_add_button(lv_obj_t * list, const void * icon, const char * txt)
94{
95 lv_obj_t * btn = lv_button_create(list);
96 lv_obj_set_size(btn, lv_pct(100), LV_SIZE_CONTENT);
97 lv_obj_set_flex_flow(btn, LV_FLEX_FLOW_ROW);
98 lv_obj_add_style(btn, &style_button, 0);
99 lv_obj_add_style(btn, &style_button_pressed, LV_STATE_PRESSED);
100
101#if LV_USE_IMAGE == 1
102 if(icon) {
103 lv_obj_t * img = lv_image_create(btn);
104 lv_image_set_src(img, icon);
105 }
106#endif
107
108 if(txt) {
109 lv_obj_t * label = lv_label_create(btn);
110 lv_label_set_text(label, txt);
111 lv_label_set_long_mode(label, LV_LABEL_LONG_MODE_SCROLL_CIRCULAR);
112 lv_obj_set_flex_grow(label, 1);
113 }
114
115 return btn;
116}
