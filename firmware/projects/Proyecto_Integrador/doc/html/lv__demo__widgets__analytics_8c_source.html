380static void scale2_timer_cb(lv_timer_t * timer)
381{
382 LV_UNUSED(timer);
383
384 static bool down1 = false;
385 static bool down2 = false;
386 static bool down3 = false;
387
388 if(down1) {
389 session_desktop -= 137;
390 if(session_desktop < 1400) down1 = false;
391 }
392 else {
393 session_desktop += 116;
394 if(session_desktop > 4500) down1 = true;
395 }
396
397 if(down2) {
398 session_tablet -= 3;
399 if(session_tablet < 1400) down2 = false;
400 }
401 else {
402 session_tablet += 9;
403 if(session_tablet > 4500) down2 = true;
404 }
405
406 if(down3) {
407 session_mobile -= 57;
408 if(session_mobile < 1400) down3 = false;
409 }
410 else {
411 session_mobile += 76;
412 if(session_mobile > 4500) down3 = true;
413 }
414
415 uint32_t all = session_desktop + session_tablet + session_mobile;
416 uint32_t angle1 = (session_desktop * 354) / all;
417 uint32_t angle2 = (session_tablet * 354) / all;
418
419 lv_arc_set_angles(lv_obj_get_child(scale2, 0), 0, angle1);
420 lv_arc_set_angles(lv_obj_get_child(scale2, 1), angle1 + 2, angle1 + 2 + angle2);
421 lv_arc_set_angles(lv_obj_get_child(scale2, 2), angle1 + 2 + angle2 + 2, 358);
422
423 lv_obj_t * card = lv_obj_get_parent(scale2);
424 lv_obj_t * label;
425
426 label = lv_obj_get_child(card, -5);
427 lv_label_set_text_fmt(label, "Desktop: %"LV_PRIu32, session_desktop);
428
429 label = lv_obj_get_child(card, -3);

431

593static void chart_event_cb(lv_event_t * e)
594{
595 lv_event_code_t code = lv_event_get_code(e);
596 lv_obj_t * obj = lv_event_get_target(e);
597
598 if(code == LV_EVENT_PRESSED || code == LV_EVENT_RELEASED) {
599 lv_obj_invalidate(obj); /*To make the value boxes visible*/
600 }
601 else if(code == LV_EVENT_DRAW_TASK_ADDED) {
602 lv_draw_task_t * draw_task = lv_event_get_param(e);
603 lv_draw_dsc_base_t * base_dsc = lv_draw_task_get_draw_dsc(draw_task);

607 lv_area_t obj_coords;
608 lv_obj_get_coords(obj, &obj_coords);
609 const lv_chart_series_t * ser = lv_chart_get_series_next(obj, NULL);
610 if(base_dsc->id1 == 1) ser = lv_chart_get_series_next(obj, ser);
611 lv_point_precise_t p1;
612 lv_point_precise_t p2;
613 int32_t i;
614 for(i = 0; i < draw_line_dsc->point_cnt - 1; i++) {
615 p1 = draw_line_dsc->points[i];
616 p2 = draw_line_dsc->points[i + 1];
617 lv_draw_triangle_dsc_t tri_dsc;
618 lv_draw_triangle_dsc_init(&tri_dsc);
619 tri_dsc.p[0].x = (int32_t)p1.x;
620 tri_dsc.p[0].y = (int32_t)p1.y;
621 tri_dsc.p[1].x = (int32_t)p2.x;
622 tri_dsc.p[1].y = (int32_t)p2.y;
623 tri_dsc.p[2].x = (int32_t)(p1.y < p2.y ? p1.x : p2.x);
624 tri_dsc.p[2].y = (int32_t)LV_MAX(p1.y, p2.y);
625 tri_dsc.grad.dir = LV_GRAD_DIR_VER;
626
627 int32_t full_h = lv_obj_get_height(obj);
628 int32_t fract_upper = (int32_t)(LV_MIN(p1.y, p2.y) - obj_coords.y1) * 255 / full_h;
629 int32_t fract_lower = (int32_t)(LV_MAX(p1.y, p2.y) - obj_coords.y1) * 255 / full_h;
630 tri_dsc.grad.stops[0].color = lv_chart_get_series_color(obj, ser);
631 tri_dsc.grad.stops[0].opa = 255 - fract_upper;
632 tri_dsc.grad.stops[0].frac = 0;
633 tri_dsc.grad.stops[1].color = lv_chart_get_series_color(obj, ser);
634 tri_dsc.grad.stops[1].opa = 255 - fract_lower;
635 tri_dsc.grad.stops[1].frac = 255;
636
637 lv_draw_triangle(base_dsc->layer, &tri_dsc);
638
639 lv_draw_rect_dsc_t rect_dsc;
640 lv_draw_rect_dsc_init(&rect_dsc);
641 rect_dsc.bg_grad.dir = LV_GRAD_DIR_VER;
642 rect_dsc.bg_grad.stops[0].color = lv_chart_get_series_color(obj, ser);
643 rect_dsc.bg_grad.stops[0].frac = 0;
644 rect_dsc.bg_grad.stops[0].opa = 255 - fract_lower;
645 rect_dsc.bg_grad.stops[1].color = lv_chart_get_series_color(obj, ser);
646 rect_dsc.bg_grad.stops[1].frac = 255;
647 rect_dsc.bg_grad.stops[1].opa = 0;
648
649 lv_area_t rect_area;
650 rect_area.x1 = (int32_t)p1.x;
651 rect_area.x2 = (int32_t)p2.x;
652 rect_area.y1 = (int32_t)LV_MAX(p1.y, p2.y);
653 rect_area.y2 = (int32_t)obj_coords.y2;
654 lv_draw_rect(base_dsc->layer, &rect_dsc, &rect_area);
655 }
656 }
657
658 bool add_value = false;
659 if(base_dsc->part == LV_PART_INDICATOR && lv_chart_get_pressed_point(obj) == base_dsc->id2) {
660 if(lv_chart_get_type(obj) == LV_CHART_TYPE_LINE) {
661 lv_draw_rect_dsc_t outline_dsc;
662 lv_draw_rect_dsc_init(&outline_dsc);
663 outline_dsc.bg_opa = LV_OPA_TRANSP;

667 lv_area_t draw_task_area;
668 lv_draw_task_get_area(draw_task, &draw_task_area);
669 lv_draw_rect(base_dsc->layer, &outline_dsc, &draw_task_area);
670 add_value = true;
671 }
672 }
673 if(base_dsc->part == LV_PART_ITEMS && lv_chart_get_pressed_point(obj) == base_dsc->id2) {
674 const lv_chart_series_t * ser = lv_chart_get_series_next(obj, NULL);
675 if(base_dsc->id1 == 1) ser = lv_chart_get_series_next(obj, ser);

677 if(lv_chart_get_type(obj) == LV_CHART_TYPE_BAR) {

679 lv_draw_rect_dsc_t shadow_dsc;
680 lv_draw_rect_dsc_init(&shadow_dsc);
681 shadow_dsc.radius = fill_dsc->radius;
682 shadow_dsc.bg_opa = LV_OPA_TRANSP;
683 shadow_dsc.shadow_color = lv_chart_get_series_color(obj, ser);

688 add_value = true;
689 }
690 }
691
692 if(add_value) {
693 const lv_chart_series_t * ser = lv_chart_get_series_next(obj, NULL);
694 if(base_dsc->id1 == 1) {
695 ser = lv_chart_get_series_next(obj, ser);
696 }
697
698 char buf[8];
699 lv_snprintf(buf, sizeof(buf), "%"LV_PRId32, lv_chart_get_series_y_array(obj, (lv_chart_series_t *)ser)[base_dsc->id2]);

709 txt_area.x1 = draw_task_area.x1 + (lv_area_get_width(&draw_task_area) - text_size.x) / 2;
710 txt_area.x2 = txt_area.x1 + text_size.x;

719 lv_draw_rect_dsc_init(&rect_dsc);
720 rect_dsc.bg_color = lv_chart_get_series_color(obj, ser);
721 rect_dsc.radius = LV_DPX(5);
722 lv_draw_rect(base_dsc->layer, &rect_dsc, &bg_area);
723
724 lv_draw_label_dsc_t label_dsc;
725 lv_draw_label_dsc_init(&label_dsc);
726 label_dsc.color = lv_color_white();
727 label_dsc.font = font_normal;
728 label_dsc.text = buf;
729 label_dsc.text_local = true;
730 lv_draw_label(base_dsc->layer, &label_dsc, &txt_area);
731 }
