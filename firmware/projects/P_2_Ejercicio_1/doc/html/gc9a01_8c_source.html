180static bool ClipRect(int32_t *x0, int32_t *y0, int32_t *x1, int32_t *y1){
181 int32_t aux;
182 if (*x0 > *x1){ aux = *x0; *x0 = *x1; *x1 = aux; }
183 if (*y0 > *y1){ aux = *y0; *y0 = *y1; *y1 = aux; }
184 if (*x1 < 0 || *y1 < 0 || *x0 >= GC9A01_WIDTH || *y0 >= GC9A01_HEIGHT){
185 return false;
186 }
187 if (*x0 < 0) *x0 = 0;
188 if (*y0 < 0) *y0 = 0;
189 if (*x1 >= GC9A01_WIDTH) *x1 = GC9A01_WIDTH - 1;
190 if (*y1 >= GC9A01_HEIGHT) *y1 = GC9A01_HEIGHT - 1;
191 return true;
192}

197static void FillRect(int32_t x0, int32_t y0, int32_t x1, int32_t y1, uint16_t color){
198 uint32_t pixels, chunk, i;
199 if (!ClipRect(&x0, &y0, &x1, &y1)){
200 return;
201 }
202 pixels = (uint32_t)(x1 - x0 + 1) * (uint32_t)(y1 - y0 + 1);
203 SetWindow(x0, y0, x1, y1);
204 WriteCmd(CMD_RAMWR, NULL, 0);
205 /* Se carga el buffer una sola vez con el color y se envía repetidas veces */
206 chunk = (pixels < TX_BUF_PIXELS) ? pixels : TX_BUF_PIXELS;
207 for (i = 0; i < chunk; i++){
208 tx_buf[2 * i] = color >> 8;
209 tx_buf[2 * i + 1] = color & 0xFF;
210 }
211 tx_len = 0;
212 GPIOOn(lcd_dc);
213 while (pixels > 0){
214 chunk = (pixels < TX_BUF_PIXELS) ? pixels : TX_BUF_PIXELS;
215 SpiWrite(lcd_spi, tx_buf, chunk * 2);
216 pixels -= chunk;
217 }
218}

269static void DrawGlyph(uint16_t x, uint16_t y, uint16_t w, uint16_t h, const uint8_t *bits, uint16_t fg, uint16_t bg){
270 uint16_t stride = (w + 7) / 8;
271 uint16_t vis_w, vis_h, row, col;
272 const uint8_t *line;
273 if (w == 0 || h == 0 || x >= GC9A01_WIDTH || y >= GC9A01_HEIGHT){
274 return;
275 }
276 /* Si no entra completo se recorta (se sigue leyendo la fila completa pero se envían solo los píxeles visibles) */
277 vis_w = (x + w > GC9A01_WIDTH) ? (GC9A01_WIDTH - x) : w;
278 vis_h = (y + h > GC9A01_HEIGHT) ? (GC9A01_HEIGHT - y) : h;
279 StreamBegin(x, y, x + vis_w - 1, y + vis_h - 1);
280 for (row = 0; row < vis_h; row++){
281 line = bits + (uint32_t)row * stride;
282 for (col = 0; col < vis_w; col++){
283 StreamPixel((line[col >> 3] & (0x80 >> (col & 7))) ? fg : bg);
284 }
285 }
286 StreamFlush();
287}

290uint8_t GC9A01Init(spi_dev_t spi_dev, uint8_t gpio_dc, uint8_t gpio_rst){
291 uint8_t params[INIT_PARAM_MAX];
292 const uint8_t *p = init_seq;
293 uint8_t cmd, info, n, i;
294
295 /* Configuración del puerto SPI (modo 0). Se inicializa una sola vez */
296 spi_mcu_config_t spi_conf = {
297 .device = spi_dev,
298 .clk_mode = MODE0,
299 .bitrate = GC9A01_SPI_BR,
300 .transfer_mode = SPI_POLLING,
301 .func_p = NULL,
302 .param_p = NULL
303 };
304 lcd_spi = spi_dev;
305 SpiInit(&spi_conf);
306
307 /* Configuración de los GPIO */
308 lcd_dc = gpio_dc;
309 GPIOInit(lcd_dc, GPIO_OUTPUT);
310 GPIOOn(lcd_dc);
311
312 if (gpio_rst != GC9A01_NO_RST){
313 /* Reset por hardware */
314 lcd_rst = gpio_rst;
315 GPIOInit(lcd_rst, GPIO_OUTPUT);
316 GPIOOn(lcd_rst);
317 DelayMs(5);
318 GPIOOff(lcd_rst);
319 DelayMs(20);
320 GPIOOn(lcd_rst);
321 DelayMs(INIT_DELAY_MS);
322 } else{
323 /* Sin pin de reset: reset por software */
324 WriteCmd(CMD_SWRESET, NULL, 0);
325 DelayMs(INIT_DELAY_MS);
326 }
327
328 /* Se envía la secuencia de inicialización (los parámetros se copian a RAM antes de enviarlos) */
329 while ((cmd = *p++) != 0x00){
330 info = *p++;
331 n = info & ~DELAY_FLAG;
332 for (i = 0; i < n && i < INIT_PARAM_MAX; i++){
333 params[i] = p[i];
334 }
335 WriteCmd(cmd, params, n);
336 p += n;
337 if (info & DELAY_FLAG){
338 DelayMs(INIT_DELAY_MS);
339 }
340 }
341
342 /* Salida del modo sleep, pantalla en negro y encendido del display */
343 WriteCmd(CMD_SLPOUT, NULL, 0);
344 DelayMs(INIT_DELAY_MS);
345 GC9A01Fill(GC9A01_BLACK);
346 WriteCmd(CMD_DISPON, NULL, 0);
347 DelayMs(20);
348 return 1;
349}

421void GC9A01DrawInt(uint16_t x, uint16_t y, uint32_t num, uint8_t dig, const Font_t* font, uint16_t foreground, uint16_t background){
422 uint8_t i, d;
423 char digit;
424 int32_t lcd_x;
425 uint16_t cell = 0, w;
426
427 /* Los dígitos de las fuentes tienen anchos distintos (el '1' es más angosto que el '0'), por eso
428 * cada dígito se dibuja en una celda de ancho fijo: el del dígito más ancho más 1 píxel de separación */
429 for (d = 0; d < 10; d++){
430 w = font->info['0' + d - ' '].width;
431 if (w > cell){
432 cell = w;
433 }
434 }
435 cell += 1;
436 for (i = 0; i < dig; i++){
437 digit = num % 10 + '0';
438 lcd_x = x + cell * (dig - 1 - i);
439 GC9A01DrawChar(lcd_x, y, digit, font, foreground, background);
440 /* Se pinta con el color de fondo lo que sobra de la celda, para no dejar restos de un dígito anterior más ancho */
441 w = font->info[digit - ' '].width;
442 FillRect(lcd_x + w, y, lcd_x + cell - 1, (int32_t)y + font->font_height - 1, background);
443 num = num / 10;
444 }
445}

480static void DrawLine(int32_t x0, int32_t y0, int32_t x1, int32_t y1, uint16_t color){
481 int32_t dx = ABS(x1 - x0);
482 int32_t dy = ABS(y1 - y0);
483 int32_t sx = (x0 < x1) ? 1 : -1;
484 int32_t sy = (y0 < y1) ? 1 : -1;
485 int32_t err, run, x, y;
486 bool last;
487
488 /* Bresenham agrupando en "tramos": cada tramo de píxeles consecutivos sobre el eje mayor
489 * se dibuja con una sola escritura, en lugar de un píxel por vez. Es mucho más rápido por SPI */
490 if (dx >= dy){
491 err = dx / 2;
492 run = x0;
493 y = y0;
494 for (x = x0; ; x += sx){
495 err -= dy;
496 last = (x == x1);
497 if (err < 0 || last){
498 FillRect(run, y, x, y, color);
499 run = x + sx;
500 y += sy;
501 err += dx;
502 }
503 if (last){
504 break;
505 }
506 }
507 } else{
508 err = dy / 2;
509 run = y0;
510 x = x0;
511 for (y = y0; ; y += sy){
512 err -= dx;
513 last = (y == y1);
514 if (err < 0 || last){
515 FillRect(x, run, x, y, color);
516 run = y + sy;
517 x += sx;
518 err += dy;
519 }
520 if (last){
521 break;
522 }
523 }
524 }
525}

553 PutPixel(x0 - y, y0 + x, color);
554 PutPixel(x0 + y, y0 - x, color);
555 PutPixel(x0 - y, y0 - x, color);
556 if (d < 0){
557 d += 2 * x + 3;
558 } else{
559 d += 2 * (x - y) + 5;

561 }

594void GC9A01DrawFilledTriangle(int16_t x0, int16_t y0, int16_t x1, int16_t y1, int16_t x2, int16_t y2, uint16_t color){

599 /* Se ordenan los vértices por y creciente: (ax,ay) arriba, (cx,cy) abajo */
600 if (ay > by){ aux = ay; ay = by; by = aux; aux = ax; ax = bx; bx = aux; }
601 if (ay > cy){ aux = ay; ay = cy; cy = aux; aux = ax; ax = cx; cx = aux; }
602 if (by > cy){ aux = by; by = cy; cy = aux; aux = bx; bx = cx; cx = aux; }
603
604 total_h = cy - ay;
605 if (total_h == 0){
606 /* Triángulo degenerado: los 3 vértices están en la misma fila */
607 xa = ax; xb = ax;
608 if (bx < xa) xa = bx;
609 if (cx < xa) xa = cx;
610 if (bx > xb) xb = bx;
611 if (cx > xb) xb = cx;
612 FillRect(xa, ay, xb, ay, color);
613 return;
614 }
615 /* Se recorre fila por fila calculando dónde cortan los lados del triángulo */
616 for (y = ay; y <= cy; y++){
617 second_half = (y > by) || (by == ay);
618 seg_h = second_half ? (cy - by) : (by - ay);
619 xa = ax + (cx - ax) * (y - ay) / total_h;
620 if (second_half){
621 xb = bx + (cx - bx) * (y - by) / seg_h;
622 } else{
623 xb = ax + (bx - ax) * (y - ay) / seg_h;
624 }
625 FillRect(xa, y, xb, y, color);
626 }
627}

629void GC9A01DrawPicture(uint16_t x, uint16_t y, uint16_t width, uint16_t height, const uint8_t* pic){
630 uint16_t vis_w, vis_h, row;
631 uint32_t n, chunk, i;
632 const uint8_t *src;
633 if (width == 0 || height == 0 || x >= GC9A01_WIDTH || y >= GC9A01_HEIGHT){
634 return;
635 }
636 vis_w = (x + width > GC9A01_WIDTH) ? (GC9A01_WIDTH - x) : width;
637 vis_h = (y + height > GC9A01_HEIGHT) ? (GC9A01_HEIGHT - y) : height;
638 StreamBegin(x, y, x + vis_w - 1, y + vis_h - 1);
639 for (row = 0; row < vis_h; row++){
640 /* Se copian los bytes de la fila al buffer (la imagen suele estar en flash, que no sirve para DMA) */
641 src = pic + (uint32_t)row * width * 2;
642 n = (uint32_t)vis_w * 2;
643 while (n > 0){
644 chunk = TX_BUF_SIZE - tx_len;
645 if (chunk > n){
646 chunk = n;
647 }
648 for (i = 0; i < chunk; i++){
649 tx_buf[tx_len + i] = src[i];
650 }
651 tx_len += chunk;
652 src += chunk;
653 n -= chunk;
654 if (tx_len >= TX_BUF_SIZE){
655 StreamFlush();
656 }
657 }
658 }
659 StreamFlush();
660}
