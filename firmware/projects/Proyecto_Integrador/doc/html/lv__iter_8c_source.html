113lv_result_t lv_iter_next(lv_iter_t * iter, void * elem)
114{
115 LV_ASSERT_NULL(iter);
116 if(iter == NULL) return LV_RESULT_INVALID;
117
118 lv_circle_buf_t * c_buf = iter->peek_buf;
119 if(c_buf != NULL && !lv_circle_buf_is_empty(c_buf)) {
120 if(elem) lv_circle_buf_read(c_buf, elem);
121 else lv_circle_buf_skip(c_buf);
122 iter->peek_offset = 0;
123 return LV_RESULT_OK;
124 }
125
126 const lv_result_t iter_res = iter->next_cb(iter->instance, iter->context, elem);
127 if(iter_res == LV_RESULT_INVALID) return LV_RESULT_INVALID;
128
129 if(c_buf != NULL) iter->peek_offset = 0;
130
131 return iter_res;
132}

134lv_result_t lv_iter_peek(lv_iter_t * iter, void * elem)
135{
136 LV_ASSERT_NULL(iter);
137 if(iter == NULL) return LV_RESULT_INVALID;
138
139 lv_circle_buf_t * c_buf = iter->peek_buf;
140 if(c_buf == NULL) return LV_RESULT_INVALID;
141
142 const uint32_t peek_count = lv_circle_buf_size(c_buf);
143 if(iter->peek_offset >= peek_count) {
144 const uint32_t required = iter->peek_offset + 1 - peek_count;
145 const uint32_t filled = lv_circle_buf_fill(c_buf, required, peek_fill_cb, iter);
146 if(filled != required) return LV_RESULT_INVALID;
147 }
148
149 lv_circle_buf_peek_at(c_buf, iter->peek_offset, elem);
150
151 return LV_RESULT_OK;
152}

154lv_result_t lv_iter_peek_advance(lv_iter_t * iter)

176void lv_iter_inspect(lv_iter_t * iter, const lv_iter_inspect_cb inspect_cb)
177{
178 LV_ASSERT_NULL(iter);
179 if(iter == NULL) return;
180
181 void * elem = lv_malloc_zeroed(iter->elem_size);
182 LV_ASSERT_MALLOC(elem);
183
184 if(elem == NULL) {
185 LV_LOG_ERROR("Could not allocate memory for element");
186 return;
187 }
188
189 while(lv_iter_next(iter, elem) == LV_RESULT_OK) {
190 inspect_cb(elem);
191 }
192
193 lv_free(elem);
194}
