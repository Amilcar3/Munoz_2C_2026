506void lv_obj_set_hidden(lv_obj_t * obj, bool en)
507{
508 LV_CHECK_OBJ(obj, MY_CLASS, return);
509 if(obj->hidden == en) return;
510
511 bool was_on_layout = lv_obj_is_layout_positioned(obj);
512
513 /*Invalidate while still visible*/
514 if(!obj->hidden) lv_obj_invalidate(obj);
515
516 obj->hidden = en;
517
518 /*Invalidate when visible again*/
519 if(!obj->hidden) lv_obj_invalidate(obj);
520
521 /*Focus the next next item when this one get hidden*/
522 if(obj->hidden && has_state(obj, LV_STATE_FOCUSED)) {
523 lv_group_t * group = lv_obj_get_group(obj);
524 if(group != NULL) {
525 lv_group_focus_next(group);
526 lv_obj_t * next_obj = lv_group_get_focused(group);
527 if(next_obj != NULL) {
528 lv_obj_invalidate(next_obj);
529 }
530 }
531 }
532

1207{

1209 LV_CHECK_ARG(child != NULL, return LV_RESULT_INVALID);
1210

1212

1214 new_child_cnt * (sizeof(lv_obj_t *)));
1215 if(!children) {
1216 return LV_RESULT_INVALID;
1217 }
1218 children[new_child_cnt - 1] = child;
1219
1220 parent->spec_attr->child_cnt = new_child_cnt;
1221 parent->spec_attr->children = children;
1222 return LV_RESULT_OK;

1225void lv_obj_remove_child(lv_obj_t * parent, lv_obj_t * child)
1226{
1227 LV_CHECK_OBJ(parent, MY_CLASS, return);
1228 LV_CHECK_OBJ(child, MY_CLASS, return);
1229
1230 for(int32_t i = lv_obj_get_index(child); i < (int32_t)parent->spec_attr->child_cnt - 1; i++) {
1231 parent->spec_attr->children[i] = parent->spec_attr->children[i + 1];
1232 }
1233
1234 /* No more children*/
1235 if(parent->spec_attr->child_cnt == 1) {
1236 lv_free(parent->spec_attr->children);
1237 parent->spec_attr->children = NULL;
1238 parent->spec_attr->child_cnt = 0;
1239 return;
1240 }
1241 parent->spec_attr->child_cnt--;
1242 lv_obj_t ** new_children = lv_realloc(parent->spec_attr->children, parent->spec_attr->child_cnt * (sizeof(lv_obj_t *)));
1243 /* Reallocating a smaller size should never fail but just in case it's implemented as malloc + memcpy + free*/
1244 if(!new_children) {
1245 LV_LOG_INFO("Failed to shrink children array");
1246 return;
1247 }
1248 parent->spec_attr->children = new_children;

1566static void lv_obj_draw(lv_event_t * e)
1567{
1568 lv_event_code_t code = lv_event_get_code(e);
1569 lv_obj_t * obj = lv_event_get_current_target(e);
1570 if(code == LV_EVENT_COVER_CHECK) {
1571 lv_cover_check_info_t * info = lv_event_get_param(e);
1572 if(info->res == LV_COVER_RES_MASKED) return;
1573 if(lv_obj_get_style_clip_corner_internal(obj, LV_PART_MAIN)) {
1574 info->res = LV_COVER_RES_MASKED;
1575 return;
1576 }
1577
1578 /*Most trivial test. Is the mask fully IN the object? If no it surely doesn't cover it*/
1579 int32_t r = lv_obj_get_style_radius_internal(obj, LV_PART_MAIN);
1580 int32_t w = lv_obj_get_style_transform_width_internal(obj, LV_PART_MAIN);
1581 int32_t h = lv_obj_get_style_transform_height_internal(obj, LV_PART_MAIN);
1582 lv_area_t coords;
1583 coords = obj->coords;
1584 lv_area_increase(&coords, w, h);
1585
1586 if(lv_area_is_in(info->area, &coords, r) == false) {
1587 info->res = LV_COVER_RES_NOT_COVER;
1588 return;
1589 }
1590
1591 if(lv_obj_get_style_bg_opa_internal(obj, LV_PART_MAIN) < LV_OPA_MAX) {
1592 info->res = LV_COVER_RES_NOT_COVER;
1593 return;
1594 }
1595
1596 if(lv_obj_get_style_opa_internal(obj, LV_PART_MAIN) < LV_OPA_MAX) {
1597 info->res = LV_COVER_RES_NOT_COVER;
1598 return;
1599 }
1600
1601 if(lv_obj_get_style_bg_grad_dir_internal(obj, LV_PART_MAIN) != LV_GRAD_DIR_NONE) {
1602 if(lv_obj_get_style_bg_grad_opa_internal(obj, LV_PART_MAIN) < LV_OPA_MAX ||
1603 lv_obj_get_style_bg_main_opa_internal(obj, LV_PART_MAIN) < LV_OPA_MAX) {
1604 info->res = LV_COVER_RES_NOT_COVER;
1605 return;
1606 }
1607 }
1608 const lv_grad_dsc_t * grad_dsc = lv_obj_get_style_bg_grad_internal(obj, LV_PART_MAIN);
1609 if(grad_dsc) {
1610 uint32_t i;
1611 for(i = 0; i < grad_dsc->stops_count; i++) {
1612 if(grad_dsc->stops[i].opa < LV_OPA_MAX) {
1613 info->res = LV_COVER_RES_NOT_COVER;
1614 return;
1615 }
1616 }
1617 }
1618 info->res = LV_COVER_RES_COVER;
1619 }
1620 else if(code == LV_EVENT_DRAW_MAIN) {
1621 lv_layer_t * layer = lv_event_get_layer(e);
1622
1623 lv_draw_rect_dsc_t draw_dsc;
1624 lv_draw_rect_dsc_init(&draw_dsc);
1625 draw_dsc.base.layer = layer;
1626
1627 int32_t w = lv_obj_get_style_transform_width_internal(obj, LV_PART_MAIN);
1628 int32_t h = lv_obj_get_style_transform_height_internal(obj, LV_PART_MAIN);
1629 lv_area_t coords;
1630 coords = obj->coords;
1631 lv_area_increase(&coords, w, h);
1632
1633 bool backdrop_blur = lv_obj_get_style_blur_backdrop_internal(obj, LV_PART_MAIN);
1634 if(backdrop_blur) {
1635 lv_draw_blur_dsc_t blur_dsc;
1636 lv_draw_blur_dsc_init(&blur_dsc);
1637 lv_obj_init_draw_blur_dsc(obj, LV_PART_MAIN, &blur_dsc);
1638 blur_dsc.corner_radius = draw_dsc.radius;
1639 blur_dsc.base.layer = layer;
1640 lv_draw_blur(layer, &blur_dsc, &coords);
1641 }
1642
1643 lv_obj_init_draw_rect_dsc(obj, LV_PART_MAIN, &draw_dsc);
1644 /*If the border is drawn later disable loading its properties*/
1645 if(lv_obj_get_style_border_post_internal(obj, LV_PART_MAIN)) {
1646 draw_dsc.border_post = 1;
1647 }
1648
1649
1650 lv_draw_rect(layer, &draw_dsc, &coords);
1651 }
1652 else if(code == LV_EVENT_DRAW_MAIN_END) {
1653 /*Draw the non backdrop blur when the main content is rendered the the children are not yet */
1654 lv_layer_t * layer = lv_event_get_layer(e);
1655 bool backdrop_blur = lv_obj_get_style_blur_backdrop_internal(obj, LV_PART_MAIN);
1656 if(!backdrop_blur) {
1657 int32_t w = lv_obj_get_style_transform_width_internal(obj, LV_PART_MAIN);
1658 int32_t h = lv_obj_get_style_transform_height_internal(obj, LV_PART_MAIN);
1659 lv_area_t coords;
1660 coords = obj->coords;
1661 lv_area_increase(&coords, w, h);
1662
1663 lv_draw_blur_dsc_t blur_dsc;
1664 lv_draw_blur_dsc_init(&blur_dsc);
1665 lv_obj_init_draw_blur_dsc(obj, LV_PART_MAIN, &blur_dsc);
1666 blur_dsc.corner_radius = lv_obj_get_style_radius_internal(obj, LV_PART_MAIN);
1667 blur_dsc.base.layer = layer;
1668 lv_draw_blur(layer, &blur_dsc, &coords);
1669 }
1670 }
1671 else if(code == LV_EVENT_DRAW_POST) {
1672 lv_layer_t * layer = lv_event_get_layer(e);
1673 draw_scrollbar(obj, layer);
1674
1675 /*If the border is drawn later disable loading other properties*/

1792static void lv_obj_event(const lv_obj_class_t * class_p, lv_event_t * e)
1793{
1794 LV_UNUSED(class_p);
1795
1796 lv_event_code_t code = lv_event_get_code(e);
1797 lv_obj_t * obj = lv_event_get_current_target(e);
1798 if(code == LV_EVENT_PRESSED) {
1799 lv_obj_add_state(obj, LV_STATE_PRESSED);
1800 }
1801 else if(code == LV_EVENT_RELEASED) {
1802 lv_obj_remove_state(obj, LV_STATE_PRESSED);
1803 lv_indev_t * indev = lv_event_get_indev(e);
1804 /*Go the checked state if enabled*/
1805 if((indev == NULL || lv_indev_get_scroll_obj(indev) == NULL) && lv_obj_is_checkable(obj)) {
1806
1807 bool was_checked = has_state(obj, LV_STATE_CHECKED);
1808 if(!(lv_obj_get_state(obj) & LV_STATE_CHECKED)) {
1809 lv_obj_add_state(obj, LV_STATE_CHECKED);
1810 }
1811 /*Radio buttons can't be checked off directly*/
1812 else if(!lv_obj_is_radio_button(obj)) {
1813 lv_obj_remove_state(obj, LV_STATE_CHECKED);
1814 }
1815 if(was_checked != has_state(obj, LV_STATE_CHECKED)) {
1816 lv_result_t res = lv_obj_send_event(obj, LV_EVENT_VALUE_CHANGED, NULL);
1817 if(res != LV_RESULT_OK) return;
1818
1819 res = lv_obj_send_event(obj, was_checked ? LV_EVENT_UNCHECKED : LV_EVENT_CHECKED, NULL);
1820 if(res != LV_RESULT_OK) return;
1821 }
1822 }
1823 }
1824 else if(code == LV_EVENT_VALUE_CHANGED) {
1825 if(lv_obj_is_radio_button(obj) && has_state(obj, LV_STATE_CHECKED)) {
1826 lv_obj_t * parent = lv_obj_get_parent(obj);
1827 if(parent) {
1828 uint32_t child_cnt = lv_obj_get_child_count(parent);
1829 uint32_t i;
1830 for(i = 0; i < child_cnt; i++) {
1831 lv_obj_t * sibling = lv_obj_get_child(parent, i);
1832 if(obj == sibling) continue;
1833
1834 if(lv_obj_is_radio_button(sibling) && has_state(sibling, LV_STATE_CHECKED)) {
1835 lv_obj_remove_state(sibling, LV_STATE_CHECKED);
1836 lv_result_t res = lv_obj_send_event(sibling, LV_EVENT_VALUE_CHANGED, NULL);
1837 if(res != LV_RESULT_OK) return;
1838
1839 res = lv_obj_send_event(sibling, LV_EVENT_UNCHECKED, NULL);
1840 if(res != LV_RESULT_OK) return;
1841 }
1842 }
1843 }
1844 }
1845 }
1846 else if(code == LV_EVENT_PRESS_LOST) {
1847 lv_obj_remove_state(obj, LV_STATE_PRESSED);
1848 }
1849 else if(code == LV_EVENT_STYLE_CHANGED) {
1850 uint32_t child_cnt = lv_obj_get_child_count(obj);
1851 for(uint32_t i = 0; i < child_cnt; i++) {
1852 lv_obj_t * child = obj->spec_attr->children[i];
1853 lv_obj_mark_layout_as_dirty(child);
1854 }
1855 }
1856 else if(code == LV_EVENT_KEY) {
1857 if(lv_obj_is_checkable(obj)) {
1858 uint32_t c = lv_event_get_key(e);
1859 bool was_checked = has_state(obj, LV_STATE_CHECKED);
1860 if(c == LV_KEY_RIGHT || c == LV_KEY_UP) {
1861 lv_obj_add_state(obj, LV_STATE_CHECKED);
1862 }
1863 else if(c == LV_KEY_LEFT || c == LV_KEY_DOWN) {
1864 /*Radio buttons can't be checked off directly*/
1865 if(!lv_obj_is_radio_button(obj)) {
1866 lv_obj_remove_state(obj, LV_STATE_CHECKED);
1867 }
1868 }
1869
1870 /*With Enter LV_EVENT_RELEASED will send VALUE_CHANGE event*/
1871 if(c != LV_KEY_ENTER && was_checked != has_state(obj, LV_STATE_CHECKED)) {
1872 lv_result_t res = lv_obj_send_event(obj, LV_EVENT_VALUE_CHANGED, NULL);
1873 if(res != LV_RESULT_OK) return;
1874
1875 res = lv_obj_send_event(obj, was_checked ? LV_EVENT_UNCHECKED : LV_EVENT_CHECKED, NULL);
1876 if(res != LV_RESULT_OK) return;
1877 }
1878 }
1879 else if(lv_obj_is_scrollable(obj) && lv_obj_is_scroll_with_arrow(obj) && !lv_obj_is_editable(obj)) {
1880 /*scroll by keypad or encoder*/
1881 lv_anim_enable_t anim_enable = LV_ANIM_OFF;
1882 int32_t sl = lv_obj_get_scroll_left(obj);
1883 int32_t sr = lv_obj_get_scroll_right(obj);
1884 uint32_t c = lv_event_get_key(e);
1885 if(c == LV_KEY_DOWN) {
1886 /*use scroll_to_x/y functions to enforce scroll limits*/
1887 lv_obj_scroll_to_y(obj, lv_obj_get_scroll_y(obj) + lv_obj_get_height(obj) / 4, anim_enable);
1888 }
1889 else if(c == LV_KEY_UP) {
1890 lv_obj_scroll_to_y(obj, lv_obj_get_scroll_y(obj) - lv_obj_get_height(obj) / 4, anim_enable);
1891 }
1892 else if(c == LV_KEY_RIGHT) {
1893 /*If the object can't be scrolled horizontally then scroll it vertically*/
1894 if(!((lv_obj_get_scroll_dir(obj) & LV_DIR_HOR) && (sl > 0 || sr > 0)))
1895 lv_obj_scroll_to_y(obj, lv_obj_get_scroll_y(obj) + lv_obj_get_height(obj) / 4, anim_enable);
1896 else
1897 lv_obj_scroll_to_x(obj, lv_obj_get_scroll_x(obj) + lv_obj_get_width(obj) / 4, anim_enable);
1898 }
1899 else if(c == LV_KEY_LEFT) {
1900 /*If the object can't be scrolled horizontally then scroll it vertically*/
1901 if(!((lv_obj_get_scroll_dir(obj) & LV_DIR_HOR) && (sl > 0 || sr > 0)))
1902 lv_obj_scroll_to_y(obj, lv_obj_get_scroll_y(obj) - lv_obj_get_height(obj) / 4, anim_enable);
1903 else
1904 lv_obj_scroll_to_x(obj, lv_obj_get_scroll_x(obj) - lv_obj_get_width(obj) / 4, anim_enable);
1905 }
1906 }
1907 }
1908 else if(code == LV_EVENT_FOCUSED) {
1909 if(lv_obj_is_scroll_on_focus(obj)) {
1910 lv_obj_scroll_to_view_recursive(obj, LV_ANIM_ON);
1911 }
1912
1913 lv_group_t * group = lv_obj_get_group(obj);
1914 const bool editing = group && lv_group_get_editing(group);
1915 lv_state_t state = LV_STATE_FOCUSED;
1916
1917 /* Use the indev for then indev handler.
1918 * But if the obj was focused manually it returns NULL so try to
1919 * use the indev from the event*/
1920 lv_indev_t * indev = lv_indev_active();
1921 if(indev == NULL) indev = lv_event_get_indev(e);
1922
1923 lv_indev_type_t indev_type = indev != NULL ? lv_indev_get_type(indev) : LV_INDEV_TYPE_NONE;
1924 if(indev_type == LV_INDEV_TYPE_KEYPAD || indev_type == LV_INDEV_TYPE_ENCODER) state |= LV_STATE_FOCUS_KEY;
1925 if(editing) {
1926 state |= LV_STATE_EDITED;
1927 lv_obj_add_state(obj, state);
1928 }
1929 else {
1930 lv_obj_add_state(obj, state);
1931 lv_obj_remove_state(obj, LV_STATE_EDITED);
1932 }
1933 }
1934 else if(code == LV_EVENT_SCROLL_BEGIN) {
1935 lv_obj_add_state(obj, LV_STATE_SCROLLED);
1936 }
1937 else if(code == LV_EVENT_SCROLL_END) {
1938 lv_obj_remove_state(obj, LV_STATE_SCROLLED);
1939 if(lv_obj_get_scrollbar_mode(obj) == LV_SCROLLBAR_MODE_ACTIVE) {
1940 lv_area_t hor_area, ver_area;
1941 lv_obj_get_scrollbar_area(obj, &hor_area, &ver_area);
1942 lv_obj_invalidate_area(obj, &hor_area);
1943 lv_obj_invalidate_area(obj, &ver_area);
1944 }
1945 }
1946 else if(code == LV_EVENT_DEFOCUSED) {
1947 lv_obj_remove_state(obj, LV_STATE_FOCUSED | LV_STATE_EDITED | LV_STATE_FOCUS_KEY);
1948 }
1949 else if(code == LV_EVENT_SIZE_CHANGED) {
1950 int32_t align = lv_obj_get_style_align_internal(obj, LV_PART_MAIN);
1951 uint16_t layout = lv_obj_get_style_layout_internal(obj, LV_PART_MAIN);
1952 if(layout || align) {
1953 lv_obj_mark_layout_as_dirty(obj);
1954 }
1955
1956 uint32_t i;
1957 uint32_t child_cnt = lv_obj_get_child_count(obj);
1958 for(i = 0; i < child_cnt; i++) {
1959 lv_obj_t * child = obj->spec_attr->children[i];
1960 lv_obj_mark_layout_as_dirty(child);
1961 }
1962 }
1963 else if(code == LV_EVENT_CHILD_CHANGED) {
1964 int32_t align = lv_obj_get_style_align_internal(obj, LV_PART_MAIN);
1965 uint16_t layout = lv_obj_get_style_layout_internal(obj, LV_PART_MAIN);
1966 if(layout || align || lv_obj_is_style_any_width_content(obj) || lv_obj_is_style_any_height_content(obj)) {
1967 lv_obj_mark_layout_as_dirty(obj);
1968 }
1969 }
1970 else if(code == LV_EVENT_CHILD_DELETED) {
1971 obj->readjust_scroll_after_layout = 1;
1972 lv_obj_mark_layout_as_dirty(obj);
1973 }
1974 else if(code == LV_EVENT_REFR_EXT_DRAW_SIZE) {
1975 int32_t d = lv_obj_calculate_ext_draw_size(obj, LV_PART_MAIN);
1976 lv_event_set_ext_draw_size(e, d);
1977 }
1978 else if(code == LV_EVENT_DRAW_MAIN || code == LV_EVENT_DRAW_POST || code == LV_EVENT_DRAW_MAIN_END ||
1979 code == LV_EVENT_COVER_CHECK) {
1980 lv_obj_draw(e);
1981 }
1982 else if(code == LV_EVENT_INDEV_RESET) {
1983 lv_obj_remove_state(obj, LV_STATE_PRESSED);
1984 lv_obj_remove_state(obj, LV_STATE_SCROLLED);
1985 }
1986 else if(code == LV_EVENT_HOVER_OVER) {
1987 lv_obj_add_state(obj, LV_STATE_HOVERED);
1988 }
1989 else if(code == LV_EVENT_HOVER_LEAVE) {
1990 lv_obj_remove_state(obj, LV_STATE_HOVERED);
1991 }
1992}

2000static void update_obj_state(lv_obj_t * obj, lv_state_t new_state)
2001{
2002 LV_ASSERT(obj != NULL);
2003
2004 if(obj->state == new_state) return;
2005
2006
2007 lv_state_t prev_state = obj->state;
2008
2009 lv_part_t changed_part = LV_PART_ANY;
2010 lv_style_state_cmp_t cmp_res = lv_obj_style_state_compare(obj, prev_state, new_state, &changed_part);
2011 /*If there is no difference in styles there is nothing else to do*/
2012 if(cmp_res == LV_STYLE_STATE_CMP_SAME) {
2013 obj->state = new_state;
2014 lv_obj_send_event(obj, LV_EVENT_STATE_CHANGED, &prev_state);
2015 return;
2016 }
2017
2018 /*Invalidate the object in their current state*/
2019 lv_obj_invalidate(obj);
2020
2021 obj->state = new_state;
2022 lv_obj_update_layer_type(obj);
2023 lv_obj_update_blur_status(obj);
2024
2025 /*Skip transitions if the widget is not rendered yet. */
2026 if(!obj->rendered) {
2027 lv_obj_invalidate(obj);
2028 if(cmp_res == LV_STYLE_STATE_CMP_DIFF_DRAW_PAD) {
2029 lv_obj_refresh_ext_draw_size(obj);
2030 }
2031
2032 lv_obj_send_event(obj, LV_EVENT_STATE_CHANGED, &prev_state);
2033 return;
2034 }
2035
2036 obj_transition_states(obj, prev_state, new_state);
2037
2038 if(cmp_res == LV_STYLE_STATE_CMP_DIFF_REDRAW && changed_part == LV_PART_SCROLLBAR) {
2039 /*A redraw-only change confined to the scrollbar: refresh just that part.
2040 *lv_obj_refresh_style() skips the child-subtree refresh for
2041 *LV_PART_SCROLLBAR, avoiding a full-subtree style cascade. This is safe
2042 *because the scrollbar is drawn by lv_obj and has no styled children.*/
2043 lv_obj_refresh_style(obj, LV_PART_SCROLLBAR, LV_STYLE_PROP_ANY);
2044 }
2045 else if(cmp_res == LV_STYLE_STATE_CMP_DIFF_REDRAW) {
2046 /*Invalidation is not enough, e.g. layer type needs to be updated too*/
2047 lv_obj_refresh_style(obj, LV_PART_ANY, LV_STYLE_PROP_ANY);
2048 }
2049 else if(cmp_res == LV_STYLE_STATE_CMP_DIFF_LAYOUT) {
2050 lv_obj_refresh_style(obj, LV_PART_ANY, LV_STYLE_PROP_ANY);
2051 }
2052 else if(cmp_res == LV_STYLE_STATE_CMP_DIFF_DRAW_PAD) {
2053 lv_obj_invalidate(obj);
2054 lv_obj_refresh_ext_draw_size(obj);
2055 }
2056
2057 lv_obj_send_event(obj, LV_EVENT_STATE_CHANGED, &prev_state);
2058}

2060static void obj_transition_states(lv_obj_t * obj, lv_state_t prev_state, lv_state_t new_state)
2061{
2062
2063 lv_obj_style_transition_dsc_t * ts = lv_malloc(sizeof(lv_obj_style_transition_dsc_t) * STYLE_TRANSITION_MAX);
2064 LV_ASSERT_MALLOC(ts);
2065 if(!ts) {
2066 LV_LOG_WARN("No memory to compute state transitions");
2067 return;
2068 }
2069 uint32_t tsi = 0;
2070 uint32_t i;
2071 for(i = 0; i < obj->style_cnt && tsi < STYLE_TRANSITION_MAX; i++) {
2072 lv_obj_style_t * obj_style = &obj->styles[i];
2073 lv_state_t state_act = lv_obj_style_get_selector_state(obj->styles[i].selector);
2074 lv_part_t part_act = lv_obj_style_get_selector_part(obj->styles[i].selector);
2075 if(state_act & (~new_state)) continue; /*Skip unrelated styles*/
2076 if(obj_style->is_trans) continue;
2077
2078 lv_style_value_t v;
2079 if(lv_style_get_prop_internal(obj_style->style, LV_STYLE_TRANSITION, &v) != LV_STYLE_RES_FOUND) continue;
2080 const lv_style_transition_dsc_t * tr = v.ptr;
2081
2082 /*Add the props to the set if not added yet or added but with smaller weight*/
2083 uint32_t j;
2084 for(j = 0; tr->props[j] != 0 && tsi < STYLE_TRANSITION_MAX; j++) {
2085 uint32_t t;
2086 for(t = 0; t < tsi; t++) {
2087 lv_style_selector_t selector = ts[t].selector;
2088 lv_state_t state_ts = lv_obj_style_get_selector_state(selector);
2089 lv_part_t part_ts = lv_obj_style_get_selector_part(selector);
2090 if(ts[t].prop == tr->props[j] && part_ts == part_act && state_ts >= state_act) break;
2091 }
2092
2093 /*If not found add it*/
2094 if(t == tsi) {
2095 ts[tsi].time = tr->time;
2096 ts[tsi].delay = tr->delay;
2097 ts[tsi].path_cb = tr->path_xcb;
2098 ts[tsi].prop = tr->props[j];
2099 ts[tsi].user_data = tr->user_data;
2100 ts[tsi].selector = obj_style->selector;
2101 tsi++;
2102 }
2103 }
2104 }
2105
2106 for(i = 0; i < tsi; i++) {
2107 lv_part_t part_act = lv_obj_style_get_selector_part(ts[i].selector);
2108 lv_obj_style_create_transition(obj, part_act, prev_state, new_state, &ts[i]);
2109 }
2110
2111 lv_free(ts);
2112}
