89lv_draw_sw_mask_res_t LV_ATTRIBUTE_FAST_MEM lv_draw_sw_mask_apply(void * masks[], lv_opa_t * mask_buf, int32_t abs_x,
90 int32_t abs_y,
91 int32_t len)
92{
93 bool changed = false;
94 lv_draw_sw_mask_common_dsc_t * dsc;
95
96 uint32_t i;
97 for(i = 0; masks[i]; i++) {
98 dsc = masks[i];
99 lv_draw_sw_mask_res_t res = LV_DRAW_SW_MASK_RES_FULL_COVER;
100 res = dsc->cb(mask_buf, abs_x, abs_y, len, masks[i]);
101 if(res == LV_DRAW_SW_MASK_RES_TRANSP) return LV_DRAW_SW_MASK_RES_TRANSP;
102 else if(res == LV_DRAW_SW_MASK_RES_CHANGED) changed = true;
103 }
104
105 return changed ? LV_DRAW_SW_MASK_RES_CHANGED : LV_DRAW_SW_MASK_RES_FULL_COVER;
106}

127void lv_draw_sw_mask_line_points_init(lv_draw_sw_mask_line_param_t * param, int32_t p1x, int32_t p1y,
128 int32_t p2x,
129 int32_t p2y, lv_draw_sw_mask_line_side_t side)
130{
131 lv_memzero(param, sizeof(lv_draw_sw_mask_line_param_t));
132
133 if(p1y == p2y && side == LV_DRAW_SW_MASK_LINE_SIDE_BOTTOM) {
134 p1y--;
135 p2y--;
136 }
137
138 if(p1y > p2y) {
139 int32_t t;
140 t = p2x;
141 p2x = p1x;
142 p1x = t;
143
144 t = p2y;
145 p2y = p1y;
146 p1y = t;
147 }
148
149 lv_point_set(&param->cfg.p1, p1x, p1y);
150 lv_point_set(&param->cfg.p2, p2x, p2y);
151 param->cfg.side = side;
152
153 lv_point_set(&param->origo, p1x, p1y);
154 param->flat = (LV_ABS(p2x - p1x) > LV_ABS(p2y - p1y)) ? 1 : 0;
155 param->yx_steep = 0;
156 param->xy_steep = 0;
157 param->dsc.cb = lv_draw_mask_line;
158 param->dsc.type = LV_DRAW_SW_MASK_TYPE_LINE;
159
160 int32_t dx = p2x - p1x;
161 int32_t dy = p2y - p1y;
162
163 if(param->flat) {
164 /*Normalize the steep. Delta x should be relative to delta x = 1024*/
165 int32_t m;
166
167 if(dx) {
168 m = (1L << 20) / dx; /*m is multiplier to normalize y (upscaled by 1024)*/
169 param->yx_steep = (m * dy) >> 10;
170 }
171
172 if(dy) {
173 m = (1L << 20) / dy; /*m is multiplier to normalize x (upscaled by 1024)*/
174 param->xy_steep = (m * dx) >> 10;
175 }
176 param->steep = param->yx_steep;
177 }
178 else {
179 /*Normalize the steep. Delta y should be relative to delta x = 1024*/
180 int32_t m;
181
182 if(dy) {
183 m = (1L << 20) / dy; /*m is multiplier to normalize x (upscaled by 1024)*/
184 param->xy_steep = (m * dx) >> 10;
185 }
186
187 if(dx) {
188 m = (1L << 20) / dx; /*m is multiplier to normalize x (upscaled by 1024)*/
189 param->yx_steep = (m * dy) >> 10;
190 }
191 param->steep = param->xy_steep;
192 }
193
194 if(param->cfg.side == LV_DRAW_SW_MASK_LINE_SIDE_LEFT) param->inv = 0;
195 else if(param->cfg.side == LV_DRAW_SW_MASK_LINE_SIDE_RIGHT) param->inv = 1;
196 else if(param->cfg.side == LV_DRAW_SW_MASK_LINE_SIDE_TOP) {
197 if(param->steep > 0) param->inv = 1;
198 else param->inv = 0;
199 }
200 else if(param->cfg.side == LV_DRAW_SW_MASK_LINE_SIDE_BOTTOM) {
201 if(param->steep > 0) param->inv = 0;
202 else param->inv = 1;
203 }
204
205 param->spx = param->steep >> 2;
206 if(param->steep < 0) param->spx = -param->spx;
207}

209void lv_draw_sw_mask_line_angle_init(lv_draw_sw_mask_line_param_t * param, int32_t px, int32_t py, int16_t angle,
210 lv_draw_sw_mask_line_side_t side)
211{
212 /*Find an optimal degree.
213 *lv_mask_line_points_init will swap the points to keep the smaller y in p1
214 *Theoretically a line with `angle` or `angle+180` is the same only the points are swapped
215 *Find the degree which keeps the origo in place*/
216 if(angle > 180) angle -= 180; /*> 180 will swap the origo*/
217
218 int32_t p2x;
219 int32_t p2y;
220
221 p2x = (lv_trigo_sin(angle + 90) >> 5) + px;
222 p2y = (lv_trigo_sin(angle) >> 5) + py;
223
224 lv_draw_sw_mask_line_points_init(param, px, py, p2x, p2y, side);
225}

227void lv_draw_sw_mask_angle_init(lv_draw_sw_mask_angle_param_t * param, int32_t vertex_x, int32_t vertex_y,
228 int32_t start_angle, int32_t end_angle)
229{
230 lv_draw_sw_mask_line_side_t start_side;
231 lv_draw_sw_mask_line_side_t end_side;
232
233 /*Constrain the input angles*/
234 if(start_angle < 0)
235 start_angle = 0;
236 else if(start_angle > 359)
237 start_angle = 359;
238
239 if(end_angle < 0)
240 end_angle = 0;
241 else if(end_angle > 359)
242 end_angle = 359;
243
244 if(end_angle < start_angle) {
245 param->delta_deg = 360 - start_angle + end_angle;
246 }
247 else {
248 param->delta_deg = LV_ABS(end_angle - start_angle);
249 }
250
251 param->cfg.start_angle = start_angle;
252 param->cfg.end_angle = end_angle;
253 lv_point_set(&param->cfg.vertex_p, vertex_x, vertex_y);
254 param->dsc.cb = lv_draw_mask_angle;
255 param->dsc.type = LV_DRAW_SW_MASK_TYPE_ANGLE;
256
257 LV_ASSERT_MSG(start_angle >= 0 && start_angle <= 360, "Unexpected start angle");
258
259 if(start_angle >= 0 && start_angle < 180) {
260 start_side = LV_DRAW_SW_MASK_LINE_SIDE_LEFT;
261 }
262 else
263 start_side = LV_DRAW_SW_MASK_LINE_SIDE_RIGHT; /*silence compiler*/
264
265 LV_ASSERT_MSG(end_angle >= 0 && start_angle <= 360, "Unexpected end angle");
266
267 if(end_angle >= 0 && end_angle < 180) {
268 end_side = LV_DRAW_SW_MASK_LINE_SIDE_RIGHT;
269 }
270 else if(end_angle >= 180 && end_angle < 360) {
271 end_side = LV_DRAW_SW_MASK_LINE_SIDE_LEFT;
272 }
273 else
274 end_side = LV_DRAW_SW_MASK_LINE_SIDE_RIGHT; /*silence compiler*/
275
276 lv_draw_sw_mask_line_angle_init(&param->start_line, vertex_x, vertex_y, start_angle, start_side);
277 lv_draw_sw_mask_line_angle_init(&param->end_line, vertex_x, vertex_y, end_angle, end_side);
278}

280lv_result_t lv_draw_sw_mask_radius_init(lv_draw_sw_mask_radius_param_t * param, const lv_area_t * rect, int32_t radius,
281 bool inv)
282{
283 lv_memset(param, 0, sizeof(*param));
284 lv_result_t res = LV_RESULT_OK;
285 int32_t w = lv_area_get_width(rect);
286 int32_t h = lv_area_get_height(rect);
287 int32_t short_side = LV_MIN(w, h);
288 if(radius > short_side >> 1) radius = short_side >> 1;
289 if(radius < 0) radius = 0;
290
291 param->cfg.rect = *rect;
292 param->cfg.radius = radius;
293 param->cfg.outer = inv ? 1 : 0;
294 param->dsc.cb = lv_draw_mask_radius;
295 param->dsc.type = LV_DRAW_SW_MASK_TYPE_RADIUS;
296
297 if(radius == 0) {
298 return res;
299 }
300
301 lv_draw_sw_mask_radius_circle_dsc_t * circle = acquire_mask_radius_from_cache(radius);
302 if(!circle) {
303 circle = create_non_cached_mask_radius(radius);
304 }
305
306 if(!circle) {
307 LV_LOG_WARN("Failed to allocate memory for circle mask");
308 return LV_RESULT_INVALID;
309 }
310
311 param->circle = circle;
312 return res;
313}

340static lv_draw_sw_mask_res_t LV_ATTRIBUTE_FAST_MEM lv_draw_mask_line(lv_opa_t * mask_buf, int32_t abs_x,
341 int32_t abs_y, int32_t len,
342 void * param)
343{
344 lv_draw_sw_mask_line_param_t * p = param;
345 /*Make to points relative to the vertex*/
346 abs_y -= p->origo.y;
347 abs_x -= p->origo.x;
348
349 /*Handle special cases*/
350 if(p->steep == 0) {
351 /*Horizontal*/
352 if(p->flat) {
353 /*Non sense: Can't be on the right/left of a horizontal line*/
354 if(p->cfg.side == LV_DRAW_SW_MASK_LINE_SIDE_LEFT ||
355 p->cfg.side == LV_DRAW_SW_MASK_LINE_SIDE_RIGHT) return LV_DRAW_SW_MASK_RES_FULL_COVER;
356 else if(p->cfg.side == LV_DRAW_SW_MASK_LINE_SIDE_TOP && abs_y < 0) return LV_DRAW_SW_MASK_RES_FULL_COVER;
357 else if(p->cfg.side == LV_DRAW_SW_MASK_LINE_SIDE_BOTTOM && abs_y > 0) return LV_DRAW_SW_MASK_RES_FULL_COVER;
358 else {
359 return LV_DRAW_SW_MASK_RES_TRANSP;
360 }
361 }
362 /*Vertical*/
363 else {
364 /*Non sense: Can't be on the top/bottom of a vertical line*/
365 if(p->cfg.side == LV_DRAW_SW_MASK_LINE_SIDE_TOP ||
366 p->cfg.side == LV_DRAW_SW_MASK_LINE_SIDE_BOTTOM) return LV_DRAW_SW_MASK_RES_FULL_COVER;
367 else if(p->cfg.side == LV_DRAW_SW_MASK_LINE_SIDE_RIGHT && abs_x > 0) return LV_DRAW_SW_MASK_RES_FULL_COVER;
368 else if(p->cfg.side == LV_DRAW_SW_MASK_LINE_SIDE_LEFT) {
369 if(abs_x + len < 0) return LV_DRAW_SW_MASK_RES_FULL_COVER;
370 else {
371 int32_t k = - abs_x;
372 if(k < 0) return LV_DRAW_SW_MASK_RES_TRANSP;
373 if(k >= 0 && k < len) lv_memzero(&mask_buf[k], len - k);
374 return LV_DRAW_SW_MASK_RES_CHANGED;
375 }
376 }
377 else {
378 if(abs_x + len < 0) return LV_DRAW_SW_MASK_RES_TRANSP;
379 else {
380 int32_t k = - abs_x;
381 if(k < 0) k = 0;
382 if(k >= len) return LV_DRAW_SW_MASK_RES_TRANSP;
383 else if(k >= 0 && k < len) lv_memzero(&mask_buf[0], k);
384 return LV_DRAW_SW_MASK_RES_CHANGED;
385 }
386 }
387 }
388 }
389
390 lv_draw_sw_mask_res_t res;
391 if(p->flat) {
392 res = line_mask_flat(mask_buf, abs_x, abs_y, len, p);
393 }
394 else {
395 res = line_mask_steep(mask_buf, abs_x, abs_y, len, p);
396 }
397
398 return res;
399}

401static lv_draw_sw_mask_res_t LV_ATTRIBUTE_FAST_MEM line_mask_flat(lv_opa_t * mask_buf, int32_t abs_x,
402 int32_t abs_y,
403 int32_t len,
404 lv_draw_sw_mask_line_param_t * p)
405{
406
407 int32_t y_at_x;
408 y_at_x = (int32_t)((int32_t)p->yx_steep * abs_x) >> 10;
409
410 if(p->yx_steep > 0) {
411 if(y_at_x > abs_y) {
412 if(p->inv) {
413 return LV_DRAW_SW_MASK_RES_FULL_COVER;
414 }
415 else {
416 return LV_DRAW_SW_MASK_RES_TRANSP;
417 }
418 }
419 }
420 else {
421 if(y_at_x < abs_y) {
422 if(p->inv) {
423 return LV_DRAW_SW_MASK_RES_FULL_COVER;
424 }
425 else {
426 return LV_DRAW_SW_MASK_RES_TRANSP;
427 }
428 }
429 }
430
431 /*At the end of the mask if the limit line is smaller than the mask's y.
432 *Then the mask is in the "good" area*/
433 y_at_x = (int32_t)((int32_t)p->yx_steep * (abs_x + len)) >> 10;
434 if(p->yx_steep > 0) {
435 if(y_at_x < abs_y) {
436 if(p->inv) {
437 return LV_DRAW_SW_MASK_RES_TRANSP;
438 }
439 else {
440 return LV_DRAW_SW_MASK_RES_FULL_COVER;
441 }
442 }
443 }
444 else {
445 if(y_at_x > abs_y) {
446 if(p->inv) {
447 return LV_DRAW_SW_MASK_RES_TRANSP;
448 }
449 else {
450 return LV_DRAW_SW_MASK_RES_FULL_COVER;
451 }
452 }
453 }
454
455 int32_t xe;
456 if(p->yx_steep > 0) xe = ((abs_y * 256) * p->xy_steep) >> 10;
457 else xe = (((abs_y + 1) * 256) * p->xy_steep) >> 10;
458
459 int32_t xei = xe >> 8;
460 int32_t xef = xe & 0xFF;
461
462 int32_t px_h;
463 if(xef == 0) px_h = 255;
464 else px_h = 255 - (((255 - xef) * p->spx) >> 8);
465 int32_t k = xei - abs_x;
466 lv_opa_t m;
467
468 if(xef) {
469 if(k >= 0 && k < len) {
470 m = 255 - (((255 - xef) * (255 - px_h)) >> 9);
471 if(p->inv) m = 255 - m;
472 mask_buf[k] = mask_mix(mask_buf[k], m);
473 }
474 k++;
475 }
476
477 while(px_h > p->spx) {
478 if(k >= 0 && k < len) {
479 m = px_h - (p->spx >> 1);
480 if(p->inv) m = 255 - m;
481 mask_buf[k] = mask_mix(mask_buf[k], m);
482 }
483 px_h -= p->spx;
484 k++;
485 if(k >= len) break;
486 }
487
488 if(k < len && k >= 0) {
489 int32_t x_inters = (px_h * p->xy_steep) >> 10;
490 m = (x_inters * px_h) >> 9;
491 if(p->yx_steep < 0) m = 255 - m;
492 if(p->inv) m = 255 - m;
493 mask_buf[k] = mask_mix(mask_buf[k], m);
494 }
495
496 if(p->inv) {
497 k = xei - abs_x;
498 if(k > len) {
499 return LV_DRAW_SW_MASK_RES_TRANSP;
500 }
501 if(k >= 0) {
502 lv_memzero(&mask_buf[0], k);
503 }
504 }
505 else {
506 k++;
507 if(k < 0) {
508 return LV_DRAW_SW_MASK_RES_TRANSP;
509 }
510 if(k < len) {
511 lv_memzero(&mask_buf[k], len - k);
512 }
513 }
514
515 return LV_DRAW_SW_MASK_RES_CHANGED;
516}

518static lv_draw_sw_mask_res_t LV_ATTRIBUTE_FAST_MEM line_mask_steep(lv_opa_t * mask_buf, int32_t abs_x,
519 int32_t abs_y,
520 int32_t len,
521 lv_draw_sw_mask_line_param_t * p)
522{
523 int32_t k;
524 int32_t x_at_y;
525 /*At the beginning of the mask if the limit line is greater than the mask's y.
526 *Then the mask is in the "wrong" area*/
527 x_at_y = (int32_t)((int32_t)p->xy_steep * abs_y) >> 10;
528 if(p->xy_steep > 0) x_at_y++;
529 if(x_at_y < abs_x) {
530 if(p->inv) {
531 return LV_DRAW_SW_MASK_RES_FULL_COVER;
532 }
533 else {
534 return LV_DRAW_SW_MASK_RES_TRANSP;
535 }
536 }
537
538 /*At the end of the mask if the limit line is smaller than the mask's y.
539 *Then the mask is in the "good" area*/
540 x_at_y = (int32_t)((int32_t)p->xy_steep * (abs_y)) >> 10;
541 if(x_at_y > abs_x + len) {
542 if(p->inv) {
543 return LV_DRAW_SW_MASK_RES_TRANSP;
544 }
545 else {
546 return LV_DRAW_SW_MASK_RES_FULL_COVER;
547 }
548 }
549
550 /*X start*/
551 int32_t xs = ((abs_y * 256) * p->xy_steep) >> 10;
552 int32_t xsi = xs >> 8;
553 int32_t xsf = xs & 0xFF;
554
555 /*X end*/
556 int32_t xe = (((abs_y + 1) * 256) * p->xy_steep) >> 10;
557 int32_t xei = xe >> 8;
558 int32_t xef = xe & 0xFF;
559
560 lv_opa_t m;
561
562 k = xsi - abs_x;
563 if(xsi != xei && (p->xy_steep < 0 && xsf == 0)) {
564 xsf = 0xFF;
565 xsi = xei;
566 k--;
567 }
568
569 if(xsi == xei) {
570 if(k >= 0 && k < len) {
571 m = (xsf + xef) >> 1;
572 if(p->inv) m = 255 - m;
573 mask_buf[k] = mask_mix(mask_buf[k], m);
574 }
575 k++;
576
577 if(p->inv) {
578 k = xsi - abs_x;
579 if(k >= len) {
580 return LV_DRAW_SW_MASK_RES_TRANSP;
581 }
582 if(k >= 0) lv_memzero(&mask_buf[0], k);
583
584 }
585 else {
586 if(k > len) k = len;
587 if(k == 0) return LV_DRAW_SW_MASK_RES_TRANSP;
588 else if(k > 0) lv_memzero(&mask_buf[k], len - k);
589 }
590
591 }
592 else {
593 int32_t y_inters;
594 if(p->xy_steep < 0) {
595 y_inters = (xsf * (-p->yx_steep)) >> 10;
596 if(k >= 0 && k < len) {
597 m = (y_inters * xsf) >> 9;
598 if(p->inv) m = 255 - m;
599 mask_buf[k] = mask_mix(mask_buf[k], m);
600 }
601 k--;
602
603 int32_t x_inters = ((255 - y_inters) * (-p->xy_steep)) >> 10;
604
605 if(k >= 0 && k < len) {
606 m = 255 - (((255 - y_inters) * x_inters) >> 9);
607 if(p->inv) m = 255 - m;
608 mask_buf[k] = mask_mix(mask_buf[k], m);
609 }
610
611 k += 2;
612
613 if(p->inv) {
614 k = xsi - abs_x - 1;
615
616 if(k > len) k = len;
617 else if(k > 0) lv_memzero(&mask_buf[0], k);
618
619 }
620 else {
621 if(k > len) return LV_DRAW_SW_MASK_RES_FULL_COVER;
622 if(k >= 0) lv_memzero(&mask_buf[k], len - k);
623 }
624
625 }
626 else {
627 y_inters = ((255 - xsf) * p->yx_steep) >> 10;
628 if(k >= 0 && k < len) {
629 m = 255 - ((y_inters * (255 - xsf)) >> 9);
630 if(p->inv) m = 255 - m;
631 mask_buf[k] = mask_mix(mask_buf[k], m);
632 }
633
634 k++;
635
636 int32_t x_inters = ((255 - y_inters) * p->xy_steep) >> 10;
637 if(k >= 0 && k < len) {
638 m = ((255 - y_inters) * x_inters) >> 9;
639 if(p->inv) m = 255 - m;
640 mask_buf[k] = mask_mix(mask_buf[k], m);
641 }
642 k++;
643
644 if(p->inv) {
645 k = xsi - abs_x;
646 if(k > len) return LV_DRAW_SW_MASK_RES_TRANSP;
647 if(k >= 0) lv_memzero(&mask_buf[0], k);
648
649 }
650 else {
651 if(k > len) k = len;
652 if(k == 0) return LV_DRAW_SW_MASK_RES_TRANSP;
653 else if(k > 0) lv_memzero(&mask_buf[k], len - k);
654 }
655 }
656 }
657
658 return LV_DRAW_SW_MASK_RES_CHANGED;
659}

661static lv_draw_sw_mask_res_t LV_ATTRIBUTE_FAST_MEM lv_draw_mask_angle(lv_opa_t * mask_buf, int32_t abs_x,

687 else if(p->cfg.end_angle > 0 && p->cfg.end_angle <= 90 && start_angle_last < 0) start_angle_last = 0;
688 else if(p->cfg.end_angle > 90 && p->cfg.end_angle < 270 && start_angle_last > 0) start_angle_last = 0;
689
690 int32_t dist = (end_angle_first - start_angle_last) >> 1;

692 lv_draw_sw_mask_res_t res1 = LV_DRAW_SW_MASK_RES_FULL_COVER;

703
704 if(tmp > len) tmp = len;
705 if(tmp < 0) tmp = 0;
706 res2 = lv_draw_mask_line(&mask_buf[tmp], abs_x + tmp, abs_y, len - tmp, &p->end_line);
707 if(res2 == LV_DRAW_SW_MASK_RES_TRANSP) {
708 lv_memzero(&mask_buf[tmp], len - tmp);
709 }
710 if(res1 == res2) return res1;
711 else return LV_DRAW_SW_MASK_RES_CHANGED;
712 }
713 else if(p->cfg.start_angle > 180 && p->cfg.end_angle > 180 && p->cfg.start_angle > p->cfg.end_angle) {
714
715 if(abs_y > p->cfg.vertex_p.y) {
716 return LV_DRAW_SW_MASK_RES_FULL_COVER;
717 }
718
719 /*Start angle mask can work only from the end of end angle mask*/
720 int32_t end_angle_first = (rel_y * p->end_line.xy_steep) >> 10;
721 int32_t start_angle_last = ((rel_y + 1) * p->start_line.xy_steep) >> 10;
722
723 /*Do not let the line end cross the vertex else it will affect the opposite part*/
724 if(p->cfg.start_angle > 270 && p->cfg.start_angle <= 359 && start_angle_last < 0) start_angle_last = 0;
725 else if(p->cfg.start_angle > 0 && p->cfg.start_angle <= 90 && start_angle_last < 0) start_angle_last = 0;

728 if(p->cfg.end_angle > 270 && p->cfg.end_angle <= 359 && start_angle_last < 0) start_angle_last = 0;
729 else if(p->cfg.end_angle > 0 && p->cfg.end_angle <= 90 && start_angle_last < 0) start_angle_last = 0;
730 else if(p->cfg.end_angle > 90 && p->cfg.end_angle < 270 && start_angle_last > 0) start_angle_last = 0;
731
732 int32_t dist = (end_angle_first - start_angle_last) >> 1;
733
734 lv_draw_sw_mask_res_t res1 = LV_DRAW_SW_MASK_RES_FULL_COVER;
735 lv_draw_sw_mask_res_t res2 = LV_DRAW_SW_MASK_RES_FULL_COVER;
736
737 int32_t tmp = start_angle_last + dist - rel_x;
738 if(tmp > len) tmp = len;
739 if(tmp > 0) {
740 res1 = lv_draw_mask_line(&mask_buf[0], abs_x, abs_y, tmp, (lv_draw_sw_mask_line_param_t *)&p->end_line);
741 if(res1 == LV_DRAW_SW_MASK_RES_TRANSP) {
742 lv_memzero(&mask_buf[0], tmp);
743 }
744 }
745
746 if(tmp > len) tmp = len;
747 if(tmp < 0) tmp = 0;
748 res2 = lv_draw_mask_line(&mask_buf[tmp], abs_x + tmp, abs_y, len - tmp, (lv_draw_sw_mask_line_param_t *)&p->start_line);
749 if(res2 == LV_DRAW_SW_MASK_RES_TRANSP) {
750 lv_memzero(&mask_buf[tmp], len - tmp);
751 }
752 if(res1 == res2) return res1;
753 else return LV_DRAW_SW_MASK_RES_CHANGED;
754 }
755 else {
756
757 lv_draw_sw_mask_res_t res1 = LV_DRAW_SW_MASK_RES_FULL_COVER;
758 lv_draw_sw_mask_res_t res2 = LV_DRAW_SW_MASK_RES_FULL_COVER;

762 else res1 = LV_DRAW_SW_MASK_RES_UNKNOWN;

770 res1 = LV_DRAW_SW_MASK_RES_UNKNOWN;
771 }
772 else {
773 res1 = lv_draw_mask_line(mask_buf, abs_x, abs_y, len, &p->start_line);
774 }
775
776 if(p->cfg.end_angle == 180) {
777 if(abs_y < p->cfg.vertex_p.y) res2 = LV_DRAW_SW_MASK_RES_UNKNOWN;
778 else res2 = LV_DRAW_SW_MASK_RES_FULL_COVER;

780 else if(p->cfg.end_angle == 0) {
781 if(abs_y < p->cfg.vertex_p.y) res2 = LV_DRAW_SW_MASK_RES_FULL_COVER;
782 else res2 = LV_DRAW_SW_MASK_RES_UNKNOWN;
783 }
784 else if((p->cfg.end_angle < 180 && abs_y < p->cfg.vertex_p.y) ||
785 (p->cfg.end_angle > 180 && abs_y >= p->cfg.vertex_p.y)) {
786 res2 = LV_DRAW_SW_MASK_RES_UNKNOWN;
787 }
788 else {
789 res2 = lv_draw_mask_line(mask_buf, abs_x, abs_y, len, &p->end_line);
790 }

792 if(res1 == LV_DRAW_SW_MASK_RES_TRANSP || res2 == LV_DRAW_SW_MASK_RES_TRANSP) return LV_DRAW_SW_MASK_RES_TRANSP;
793 else if(res1 == LV_DRAW_SW_MASK_RES_UNKNOWN && res2 == LV_DRAW_SW_MASK_RES_UNKNOWN) return LV_DRAW_SW_MASK_RES_TRANSP;
794 else if(res1 == LV_DRAW_SW_MASK_RES_FULL_COVER &&
795 res2 == LV_DRAW_SW_MASK_RES_FULL_COVER) return LV_DRAW_SW_MASK_RES_FULL_COVER;
796 else return LV_DRAW_SW_MASK_RES_CHANGED;
797 }

800static lv_draw_sw_mask_res_t LV_ATTRIBUTE_FAST_MEM lv_draw_mask_radius(lv_opa_t * mask_buf, int32_t abs_x,
801 int32_t abs_y, int32_t len,
802 void * param)
803{
804 lv_draw_sw_mask_radius_param_t * p = param;
805 bool outer = p->cfg.outer;
806 int32_t radius = p->cfg.radius;
807 lv_area_t rect;
808 rect = p->cfg.rect;
809
810 if(outer == false) {
811 if((abs_y < rect.y1 || abs_y > rect.y2)) {
812 return LV_DRAW_SW_MASK_RES_TRANSP;
813 }
814 }
815 else {
816 if(abs_y < rect.y1 || abs_y > rect.y2) {
817 return LV_DRAW_SW_MASK_RES_FULL_COVER;
818 }
819 }
820
821 if((abs_x >= rect.x1 + radius && abs_x + len <= rect.x2 - radius) ||
822 (abs_y >= rect.y1 + radius && abs_y <= rect.y2 - radius)) {
823 if(outer == false) {
824 /*Remove the edges*/
825 int32_t last = rect.x1 - abs_x;
826 if(last > len) return LV_DRAW_SW_MASK_RES_TRANSP;
827 if(last >= 0) {
828 lv_memzero(&mask_buf[0], last);
829 }
830
831 int32_t first = rect.x2 - abs_x + 1;
832 if(first <= 0) return LV_DRAW_SW_MASK_RES_TRANSP;
833 else if(first < len) {
834 lv_memzero(&mask_buf[first], len - first);
835 }
836 if(last == 0 && first == len) return LV_DRAW_SW_MASK_RES_FULL_COVER;
837 else return LV_DRAW_SW_MASK_RES_CHANGED;
838 }
839 else {
840 int32_t first = rect.x1 - abs_x;
841 if(first < 0) first = 0;
842 if(first <= len) {
843 int32_t last = rect.x2 - abs_x - first + 1;
844 if(first + last > len) last = len - first;
845 if(last >= 0) {
846 lv_memzero(&mask_buf[first], last);
847 }
848 }
849 }
850 return LV_DRAW_SW_MASK_RES_CHANGED;
851 }
852
853 int32_t k = rect.x1 - abs_x; /*First relevant coordinate on the of the mask*/
854 int32_t w = lv_area_get_width(&rect);
855 int32_t h = lv_area_get_height(&rect);
856 abs_x -= rect.x1;
857 abs_y -= rect.y1;
858
859 int32_t aa_len;
860 int32_t x_start;
861 int32_t cir_y;
862 if(abs_y < radius) {
863 cir_y = radius - abs_y - 1;
864 }
865 else {
866 cir_y = abs_y - (h - radius);
867 }
868 lv_opa_t * aa_opa = get_next_line(p->circle, cir_y, &aa_len, &x_start);
869 int32_t cir_x_right = k + w - radius + x_start;
870 int32_t cir_x_left = k + radius - x_start - 1;
871 int32_t i;
872
873 if(outer == false) {
874 for(i = 0; i < aa_len; i++) {
875 lv_opa_t opa = aa_opa[aa_len - i - 1];
876 if(cir_x_right + i >= 0 && cir_x_right + i < len) {
877 mask_buf[cir_x_right + i] = mask_mix(opa, mask_buf[cir_x_right + i]);
878 }
879 if(cir_x_left - i >= 0 && cir_x_left - i < len) {
880 mask_buf[cir_x_left - i] = mask_mix(opa, mask_buf[cir_x_left - i]);
881 }
882 }
883
884 /*Clean the right side*/
885 cir_x_right = LV_CLAMP(0, cir_x_right + i, len);
886 lv_memzero(&mask_buf[cir_x_right], len - cir_x_right);
887
888 /*Clean the left side*/
889 cir_x_left = LV_CLAMP(0, cir_x_left - aa_len + 1, len);
890 lv_memzero(&mask_buf[0], cir_x_left);
891 }
892 else {
893 for(i = 0; i < aa_len; i++) {
894 lv_opa_t opa = 255 - (aa_opa[aa_len - 1 - i]);
895 if(cir_x_right + i >= 0 && cir_x_right + i < len) {
896 mask_buf[cir_x_right + i] = mask_mix(opa, mask_buf[cir_x_right + i]);
897 }
898 if(cir_x_left - i >= 0 && cir_x_left - i < len) {
899 mask_buf[cir_x_left - i] = mask_mix(opa, mask_buf[cir_x_left - i]);
900 }
901 }
902
903 int32_t clr_start = LV_CLAMP(0, cir_x_left + 1, len);
904 int32_t clr_len = LV_CLAMP(0, cir_x_right - clr_start, len - clr_start);
905 lv_memzero(&mask_buf[clr_start], clr_len);
906 }
907
908 return LV_DRAW_SW_MASK_RES_CHANGED;
909}

999static void circ_init(lv_point_t * c, int32_t * tmp, int32_t radius)
1000{
1001 c->x = radius;
1002 c->y = 0;
1003 *tmp = 1 - radius;
1004}
1005
1011static bool circ_cont(lv_point_t * c)
1012{
1013 return c->y <= c->x;
1014}
1015
1021static void circ_next(lv_point_t * c, int32_t * tmp)
1022{
1023
1024 if(*tmp <= 0) {
1025 (*tmp) += 2 * c->y + 3; /*Change in decision criterion for y -> y+1*/
1026 }
1027 else {
1028 (*tmp) += 2 * (c->y - c->x) + 5; /*Change for y -> y+1, x -> x-1*/
1029 c->x--;
1030 }
1031 c->y++;
1032}
1033
1034static lv_result_t circ_calc_aa4(lv_draw_sw_mask_radius_circle_dsc_t * c, int32_t radius)
1035{
1036 LV_ASSERT_MSG(radius > 0, "caller should handle radius == 0 fast path");
1037
1038 uint8_t * new_buf = lv_realloc(c->buf, radius * 6 + 6);
1039 LV_ASSERT_MALLOC(new_buf);
1040 if(!new_buf) {
1041 LV_LOG_WARN("Failed to allocate memory for circle mask (radius %" LV_PRId32 ")", radius);
1042 return LV_RESULT_INVALID;
1043 }
1044
1045 c->cir_opa = c->buf = new_buf;
1046 c->radius = radius;
1047 c->opa_start_on_y = (uint16_t *)(c->buf + 2 * radius + 2);
1048 c->x_start_on_y = (uint16_t *)(c->buf + 4 * radius + 4);
1049
1050 /*Special case, handle manually*/
1051 if(radius == 1) {
1052 c->cir_opa[0] = 180;
1053 c->opa_start_on_y[0] = 0;
1054 c->opa_start_on_y[1] = 1;
1055 c->x_start_on_y[0] = 0;
1056 return LV_RESULT_OK;
1057 }
1058
1059 const size_t cir_xy_size = (radius + 1) * 2 * 2 * sizeof(int32_t);
1060 int32_t * cir_x = lv_malloc_zeroed(cir_xy_size);
1061 LV_ASSERT_MALLOC(cir_x);
1062 if(!cir_x) {
1063 LV_LOG_WARN("Failed to allocate memory for circle points (radius %" LV_PRId32 ")", radius);
1064 lv_free(c->buf);
1065 c->buf = c->cir_opa = NULL;
1066 c->opa_start_on_y = c->x_start_on_y = NULL;
1067 return LV_RESULT_INVALID;
1068 }
1069 int32_t * cir_y = &cir_x[(radius + 1) * 2];
1070
1071 uint32_t y_8th_cnt = 0;
1072 lv_point_t cp;
1073 int32_t tmp;
1074 circ_init(&cp, &tmp, radius * 4); /*Upscale by 4*/
1075 int32_t i;
1076
1077 uint32_t x_int[4];
1078 uint32_t x_fract[4];
1079 int32_t cir_size = 0;
1080 x_int[0] = cp.x >> 2;
1081 x_fract[0] = 0;
1082
1083 /*Calculate an 1/8 circle*/
1084 while(circ_cont(&cp)) {
1085 /*Calculate 4 point of the circle */
1086 for(i = 0; i < 4; i++) {
1087 circ_next(&cp, &tmp);
1088 if(circ_cont(&cp) == false) break;
1089 x_int[i] = cp.x >> 2;
1090 x_fract[i] = cp.x & 0x3;
1091 }
1092 if(i != 4) break;
1093
1094 /*All lines on the same x when downscaled*/
1095 if(x_int[0] == x_int[3]) {
1096 cir_x[cir_size] = x_int[0];
1097 cir_y[cir_size] = y_8th_cnt;
1098 c->cir_opa[cir_size] = x_fract[0] + x_fract[1] + x_fract[2] + x_fract[3];
1099 c->cir_opa[cir_size] *= 16;
1100 cir_size++;
1101 }
1102 /*Second line on new x when downscaled*/
1103 else if(x_int[0] != x_int[1]) {
1104 cir_x[cir_size] = x_int[0];
1105 cir_y[cir_size] = y_8th_cnt;
1106 c->cir_opa[cir_size] = x_fract[0];
1107 c->cir_opa[cir_size] *= 16;
1108 cir_size++;
1109
1110 cir_x[cir_size] = x_int[0] - 1;

1190 lv_free(cir_x);
