565/* The open list has to touch the button, no matter the base direction of the screen
566 * or whether it opens downwards or upwards.
567 * See https://github.com/lvgl/lvgl/issues/10254 */
568static void dropdown_list_position_test(lv_base_dir_t base_dir, int32_t x, int32_t y)
569{
570 lv_obj_t * screen = lv_screen_active();
571 lv_obj_set_style_base_dir(screen, base_dir, LV_PART_MAIN);
572
573 lv_obj_t * dd = lv_dropdown_create(screen);
574 lv_dropdown_set_options(dd, "Option 1\nA considerably longer option");
575 lv_obj_set_ignore_layout(dd, true);
576 lv_obj_set_width(dd, 100);
577 lv_obj_set_pos(dd, x, y);
578
579 lv_dropdown_open(dd);
580 lv_obj_update_layout(screen);
581
582 lv_area_t dd_coords;
583 lv_area_t list_coords;
584 lv_obj_get_coords(dd, &dd_coords);
585 lv_obj_get_coords(lv_dropdown_get_list(dd), &list_coords);
586
587 /*The list is wider than the button, so the two aligned edges are the only matching ones*/
588 TEST_ASSERT_GREATER_THAN_INT32(lv_area_get_width(&dd_coords), lv_area_get_width(&list_coords));
589
590 if(base_dir == LV_BASE_DIR_RTL) {
591 TEST_ASSERT_EQUAL_INT32(dd_coords.x2, list_coords.x2);
592 }
593 else {
594 TEST_ASSERT_EQUAL_INT32(dd_coords.x1, list_coords.x1);
595 }
596
597 /*The list has to touch the button, no matter if it's opened downwards or upwards*/
598 if(list_coords.y1 > dd_coords.y1) {
599 TEST_ASSERT_EQUAL_INT32(dd_coords.y2 + 1, list_coords.y1);
600 }
601 else {
602 TEST_ASSERT_EQUAL_INT32(dd_coords.y1 - 1, list_coords.y2);
603 }
604}
