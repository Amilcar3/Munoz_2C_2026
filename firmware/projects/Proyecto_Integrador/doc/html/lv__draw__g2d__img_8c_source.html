72void lv_draw_g2d_img(lv_draw_task_t * t)
73{
74 lv_draw_image_dsc_t * dsc = t->draw_dsc;
75
76 if(dsc->opa <= (lv_opa_t)LV_OPA_MIN)
77 return;
78
79 lv_area_t * coords = &t->area;
80
81 bool is_tiled = (dsc->tile != 0);
82
83 if(is_tiled)
84 lv_draw_image_tiled_helper(t, dsc, coords, _g2d_draw_core_cb, NULL);
85 else
86 lv_draw_image_normal_helper(t, dsc, coords, _g2d_draw_core_cb, NULL);
87}

93static void _g2d_draw_core_cb(lv_draw_task_t * t, const lv_draw_image_dsc_t * draw_dsc,
94 const lv_image_decoder_dsc_t * decoder_dsc, lv_draw_image_sup_t * sup, const lv_area_t * img_coords,
95 const lv_area_t * clipped_img_area)
96{
97 LV_UNUSED(sup);
98 lv_draw_buf_t * draw_buf = t->target_layer->draw_buf;
99
100 const lv_draw_buf_t * decoded = decoder_dsc->decoded;
101
102 lv_area_t rel_clip_area;
103 rel_clip_area = *clipped_img_area;
104 lv_area_move(&rel_clip_area, -img_coords->x1, -img_coords->y1);
105
106 lv_area_t rel_img_coords;
107 rel_img_coords = *img_coords;
108 lv_area_move(&rel_img_coords, -img_coords->x1, -img_coords->y1);
109
110 lv_area_t src_area;
111 if(!lv_area_intersect(&src_area, &rel_clip_area, &rel_img_coords))
112 return;
113
114 lv_color_format_t src_cf = draw_dsc->header.cf;
115
116 /* G2D takes stride in pixels. */
117 const uint8_t pixel_size = lv_color_format_get_size(src_cf);
118
119 uint32_t src_stride = draw_dsc->header.stride == 0 ?
120 lv_color_format_get_size(draw_dsc->header.cf) * draw_dsc->header.w :
121 draw_dsc->header.stride;
122 LV_ASSERT(pixel_size != 0);
123 src_stride /= pixel_size;
124
125 /* Source image */
126 struct g2d_buf * src_buf = _g2d_handle_src_buf(decoded);
127
128 /* Destination buffer */
129 struct g2d_buf * dst_buf = g2d_search_buf_map(draw_buf->data);
130
131
132 void * handle = g2d_get_handle();
133
134 struct g2d_surface src_surf;
135 struct g2d_surface dst_surf;
136
137 _g2d_set_src_surf(&src_surf, src_buf, &src_area, src_stride, src_cf, draw_dsc);
138 _g2d_set_dst_surf(&dst_surf, dst_buf, clipped_img_area, draw_buf);
139
140 bool has_rotation = (draw_dsc->rotation != 0);
141
142 if(has_rotation) {
147 struct g2d_buf * tmp_buf = g2d_alloc(lv_area_get_width(&src_area) * lv_area_get_height(
148 &src_area) * lv_color_format_get_size(src_cf), 1);
149 G2D_ASSERT_MSG(tmp_buf, "Failed to alloc temporary buffer.");
150 struct g2d_surface tmp_surf;
151 _g2d_set_tmp_surf(&tmp_surf, tmp_buf, &src_area, src_cf);
152 _g2d_blit_two_steps(handle, &dst_surf, &src_surf, &tmp_surf);
153 g2d_free(tmp_buf);
154 }
155 else {
156 // If rotation is not involved, blit in one step.
157 _g2d_blit(handle, &dst_surf, &src_surf);
158 }
159}

161static struct g2d_buf * _g2d_handle_src_buf(const lv_draw_buf_t * img_dsc)
162{
163 struct g2d_buf * src_buf = g2d_search_buf_map((void *)img_dsc->data);
164
165 if(src_buf == NULL) {
166 src_buf = g2d_alloc(img_dsc->data_size, 1);
167 G2D_ASSERT_MSG(src_buf, "Failed to alloc source buffer.");
168 memcpy((uint8_t *)src_buf->buf_vaddr, img_dsc->data, img_dsc->data_size);
169 g2d_cache_op(src_buf, G2D_CACHE_FLUSH);
170 g2d_insert_buf_map((void *)img_dsc->data, src_buf);
171 }
172
173 return src_buf;
174}

176static void _g2d_set_src_surf(struct g2d_surface * src_surf, struct g2d_buf * buf, const lv_area_t * area,
177 int32_t stride, lv_color_format_t cf, const lv_draw_image_dsc_t * dsc)
178{
179 src_surf->format = g2d_get_buf_format(cf);
180
181 int32_t src_w = lv_area_get_width(area);
182 int32_t src_h = lv_area_get_height(area);
183
184 bool has_rotation = (dsc->rotation != 0);
185 enum g2d_rotation g2d_angle = G2D_ROTATION_0;
186
187 if(has_rotation) {
188 switch(dsc->rotation) {
189 case 0:
190 g2d_angle = G2D_ROTATION_0;
191 break;
192 case 900:
193 g2d_angle = G2D_ROTATION_270;
194 break;
195 case 1800:
196 g2d_angle = G2D_ROTATION_180;
197 break;
198 case 2700:
199 g2d_angle = G2D_ROTATION_90;
200 break;
201 default:
202 g2d_angle = G2D_ROTATION_0;
203 }
204 }
205
206 src_surf->left = area->x1;
207 src_surf->top = area->y1;
208 src_surf->right = area->x1 + src_w;
209 src_surf->bottom = area->y1 + src_h;
210 src_surf->stride = stride;
211 src_surf->width = src_w;
212 src_surf->height = src_h;
213
214 src_surf->planes[0] = buf->buf_paddr;
215 src_surf->rot = g2d_angle;
216
217 src_surf->clrcolor = g2d_rgba_to_u32(lv_color_black());
218 src_surf->global_alpha = dsc->opa;
219 src_surf->blendfunc = G2D_ONE | G2D_PRE_MULTIPLIED_ALPHA;
220}

275static void _g2d_blit(void * handle, struct g2d_surface * dst_surf, struct g2d_surface * src_surf)

285static void _g2d_blit_two_steps(void * handle, struct g2d_surface * dst_surf, struct g2d_surface * src_surf,
286 struct g2d_surface * tmp_surf)
287{

292 g2d_blit(handle, src_surf, tmp_surf);
293 g2d_finish(handle);
294 g2d_disable(handle, G2D_GLOBAL_ALPHA);
295 g2d_disable(handle, G2D_BLEND);
296
300 tmp_surf->blendfunc = G2D_ONE | G2D_PRE_MULTIPLIED_ALPHA;
301 tmp_surf->global_alpha = 0xFF;
302
303 g2d_enable(handle, G2D_BLEND);
304 g2d_enable(handle, G2D_GLOBAL_ALPHA);
305 g2d_blit(handle, tmp_surf, dst_surf);
306 g2d_finish(handle);
307 g2d_disable(handle, G2D_GLOBAL_ALPHA);
