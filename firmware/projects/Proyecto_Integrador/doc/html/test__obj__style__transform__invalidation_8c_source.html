7/* Tests that changing a style property which affects the drawn area invalidates
8 * the *old* area too.
9 *
10 * A transformed widget is drawn outside its coordinates, and the area to
11 * invalidate is computed from the current style values. So if the new value is
12 * written first and the widget is invalidated only afterwards, the area the
13 * widget used to occupy is never redrawn and the old pixels stay on the screen.
14 * Every path that can change such a property therefore has to invalidate before
15 * the change.
16 *
17 * The tests shrink the drawn area (so the new area can't cover the old one) and
18 * check through LV_EVENT_INVALIDATE_AREA that the part of the old area which
19 * the widget no longer covers was invalidated. */
20

56/* Record every area the display is asked to invalidate while capturing. */
57static void invalidate_area_cb(lv_event_t * e)
58{
59 TEST_ASSERT_LESS_THAN_UINT32(MAX_CAPTURED, captured_cnt);
60 lv_area_t * area = lv_event_get_param(e);
61 captured[captured_cnt++] = *area;
62}
63
64static void start_capture(void)
65{
66 captured_cnt = 0;
67 lv_display_add_event_cb(disp, invalidate_area_cb, LV_EVENT_INVALIDATE_AREA, NULL);
68}
69
70static void stop_capture(void)
71{
72 lv_display_remove_event_cb_with_user_data(disp, invalidate_area_cb, NULL);
73}

75/* True if any captured invalidated area covers the whole given area. */
76static bool area_was_invalidated(const lv_area_t * area)
77{
78 for(uint32_t i = 0; i < captured_cnt; i++) {
79 if(lv_area_is_in(area, &captured[i], 0)) return true;
80 }
81 return false;
82}

117/* Assert that the strip the widget dropped when it shrank from `before` to its
118 * current area was invalidated. The default pivot is the top left corner, so a
119 * smaller scale_y releases a strip below the new area. Only an invalidation done
120 * before the change can cover it. */
121static void assert_released_strip_was_invalidated(lv_obj_t * obj, const lv_area_t * before)
122{
123 lv_area_t after = drawn_area(obj);
124
125 /*The widget really shrank, otherwise the new area alone could cover the old*/
126 TEST_ASSERT_LESS_THAN_INT32(before->y2 - 4, after.y2);
127
128 lv_area_t strip = {before->x1, after.y2 + 2, before->x2, before->y2};
129 TEST_ASSERT_TRUE(area_was_invalidated(&strip));
130}

217/* Start a scale_y transition from 4x down to 1x and stop half way, so a
218 * transition is running and the widget is drawn larger than its coordinates. */
219static lv_obj_t * create_obj_with_running_transition(void)
220{
221 static const lv_style_prop_t props[] = {LV_STYLE_TRANSFORM_SCALE_Y, LV_STYLE_PROP_INV};
222
223 lv_obj_t * obj = lv_obj_create(lv_screen_active());
224 lv_obj_remove_style_all(obj);
225 lv_obj_set_pos(obj, 100, 100);
226 lv_obj_set_size(obj, 60, 40);
227 lv_obj_set_style_bg_opa(obj, LV_OPA_COVER, 0);
228
229 /*4x while pressed, 1x otherwise*/
230 lv_obj_add_style(obj, &style_big, LV_STATE_PRESSED);
231
232 lv_style_transition_dsc_init(&trans_dsc, props, lv_anim_path_linear, 400, 0, NULL);
233 lv_obj_set_style_transition(obj, &trans_dsc, 0);
234
235 /*A transition is only started for a widget that was already rendered*/
236 lv_refr_now(NULL);
237
238 lv_obj_add_state(obj, LV_STATE_PRESSED);
239 lv_test_wait(400);
240 TEST_ASSERT_EQUAL_INT32(4 * 256, lv_obj_get_style_transform_scale_y(obj, LV_PART_MAIN));
241
242 /*Transition back to 1x, stopped half way*/
243 lv_obj_remove_state(obj, LV_STATE_PRESSED);
244 lv_test_wait(200);
245 int32_t scale = lv_obj_get_style_transform_scale_y(obj, LV_PART_MAIN);
246 TEST_ASSERT_GREATER_THAN_INT32(256 + 32, scale);
247 TEST_ASSERT_LESS_THAN_INT32(4 * 256, scale);
248
249 return obj;
250}
