182void lv_draw_buf_clear_ex(lv_draw_buf_t * draw_buf, const lv_area_t * a, lv_layer_t * layer)
183{
184 LV_CHECK_ARG(draw_buf != NULL, return);
185 LV_PROFILER_DRAW_BEGIN;
186
187 /* TODO(v10): remove the layer check here once `draw_buf_clear` is standardized properly */
188 if(layer != NULL && draw_buf->handlers && draw_buf->handlers->buf_clear_cb) {
189 draw_buf->handlers->buf_clear_cb(draw_buf, a, layer);
190 LV_PROFILER_DRAW_END;
191 return;
192 }
193
194 const lv_image_header_t * header = &draw_buf->header;
195 uint32_t stride = header->stride;
196
197 if(a == NULL) {
198 uint8_t * buf = lv_draw_buf_goto_xy(draw_buf, 0, 0);
199 lv_memzero(buf, header->h * stride);
200 lv_draw_buf_flush_cache(draw_buf, a);
201 LV_PROFILER_DRAW_END;
202 return;
203 }
204
205 lv_area_t a_draw_buf;
206 a_draw_buf.x1 = 0;
207 a_draw_buf.y1 = 0;
208 a_draw_buf.x2 = draw_buf->header.w - 1;
209 a_draw_buf.y2 = draw_buf->header.h - 1;
210
211 lv_area_t a_clipped;
212 if(!lv_area_intersect(&a_clipped, a, &a_draw_buf)) {
213 LV_PROFILER_DRAW_END;
214 return;
215 }
216
217 if(lv_area_get_width(&a_clipped) <= 0) {
218 LV_PROFILER_DRAW_END;
219 return;
220 }
221
222 if(lv_area_get_height(&a_clipped) <= 0) {
223 LV_PROFILER_DRAW_END;
224 return;
225 }
226
227 uint8_t * buf = lv_draw_buf_goto_xy(draw_buf, a_clipped.x1, a_clipped.y1);
228 uint8_t bpp = lv_color_format_get_bpp(header->cf);
229 uint32_t line_length = (lv_area_get_width(&a_clipped) * bpp + 7) >> 3;
230 int32_t y;
231 for(y = a_clipped.y1; y <= a_clipped.y2; y++) {
232 lv_memzero(buf, line_length);
233 buf += stride;
234 }
235 lv_draw_buf_flush_cache(draw_buf, a);
236 LV_PROFILER_DRAW_END;
237}

239lv_result_t lv_draw_buf_init(lv_draw_buf_t * draw_buf, uint32_t w, uint32_t h, lv_color_format_t cf, uint32_t stride,
240 void * data, uint32_t data_size)
241{
242
243 LV_CHECK_ARG(draw_buf != NULL, return LV_RESULT_INVALID);
244 LV_CHECK_ARG(data != NULL || (w * h == 0), return LV_RESULT_INVALID);
245 lv_memzero(draw_buf, sizeof(lv_draw_buf_t));
246
247 if(stride == LV_STRIDE_AUTO) stride = lv_draw_buf_width_to_stride(w, cf);
248
249 LV_CHECK_ARG_FORMAT_MSG(data_size >= stride * h,
250 return LV_RESULT_INVALID,
251 "Data size too small, required: %" LV_PRId32 ", provided: %" LV_PRId32,
252 stride * h, data_size);
253
254 lv_image_header_t * header = &draw_buf->header;
255 header->w = w;
256 header->h = h;
257 header->cf = cf;
258 header->stride = stride;
259 header->flags = 0;
260 header->magic = LV_IMAGE_HEADER_MAGIC;
261
262 draw_buf->data = data;
263 draw_buf->unaligned_data = data;
264 draw_buf->handlers = &default_handlers;
265 draw_buf->data_size = data_size;
266 if(data != NULL && lv_draw_buf_align(data, cf) != draw_buf->unaligned_data) {
267 LV_LOG_INFO("Data is not aligned, ignored");
268 }
269 return LV_RESULT_OK;
270}

277lv_draw_buf_t * lv_draw_buf_create_ex(const lv_draw_buf_handlers_t * handlers, uint32_t w, uint32_t h,
278 lv_color_format_t cf, uint32_t stride)
279{
280 LV_CHECK_ARG(handlers != NULL, return NULL);
281
282 LV_PROFILER_DRAW_BEGIN;
283 lv_draw_buf_t * draw_buf = lv_malloc_zeroed(sizeof(lv_draw_buf_t));
284 LV_ASSERT_MALLOC(draw_buf);
285 if(draw_buf == NULL) {
286 LV_PROFILER_DRAW_END;
287 return NULL;
288 }
289 if(stride == LV_STRIDE_AUTO) stride = lv_draw_buf_width_to_stride(w, cf);
290
291 uint32_t size = _calculate_draw_buf_size(w, h, cf, stride);
292
293 void * buf = draw_buf_malloc(handlers, size, cf);
294 /*Do not assert here as LVGL or the app might just want to try creating a draw_buf*/
295 if(buf == NULL) {
296 LV_LOG_WARN("No memory: %"LV_PRIu32"x%"LV_PRIu32", cf: %d, stride: %"LV_PRIu32", %"LV_PRIu32"Byte, ",
297 w, h, cf, stride, size);
298 lv_free(draw_buf);
299 LV_PROFILER_DRAW_END;
300 return NULL;
301 }
302
303 draw_buf->header.w = w;
304 draw_buf->header.h = h;
305 draw_buf->header.cf = cf;
306 draw_buf->header.flags = LV_IMAGE_FLAGS_MODIFIABLE | LV_IMAGE_FLAGS_ALLOCATED;
307 draw_buf->header.stride = stride;
308 draw_buf->header.magic = LV_IMAGE_HEADER_MAGIC;
309 draw_buf->data = lv_draw_buf_align_ex(handlers, buf, cf);
310 draw_buf->unaligned_data = buf;
311 draw_buf->data_size = size;
312 draw_buf->handlers = handlers;
313 LV_PROFILER_DRAW_END;
314 return draw_buf;
315}

324lv_draw_buf_t * lv_draw_buf_dup_ex(const lv_draw_buf_handlers_t * handlers, const lv_draw_buf_t * draw_buf)
325{
326 LV_CHECK_ARG(handlers != NULL, return NULL);
327 LV_CHECK_ARG(draw_buf != NULL, return NULL);
328
329 LV_PROFILER_DRAW_BEGIN;
330 const lv_image_header_t * header = &draw_buf->header;
331 lv_draw_buf_t * new_buf = lv_draw_buf_create_ex(handlers, header->w, header->h, header->cf, header->stride);
332 if(new_buf == NULL) {
333 LV_PROFILER_DRAW_END;
334 return NULL;
335 }
336
337 lv_draw_buf_set_flag(new_buf, draw_buf->header.flags | LV_IMAGE_FLAGS_MODIFIABLE | LV_IMAGE_FLAGS_ALLOCATED);
338
339 /*Choose the smaller size to copy*/
340 uint32_t size = LV_MIN(draw_buf->data_size, new_buf->data_size);
341 if(size > 0) {
342 /*Copy image data*/
343 lv_memcpy(new_buf->data, draw_buf->data, size);
344 }
345
346 LV_PROFILER_DRAW_END;
347 return new_buf;
348}

350lv_draw_buf_t * lv_draw_buf_reshape(lv_draw_buf_t * draw_buf, lv_color_format_t cf, uint32_t w, uint32_t h,
351 uint32_t stride)
352{
353 if(!draw_buf) return NULL;
354 LV_PROFILER_DRAW_BEGIN;
355 /*If color format is unknown, keep using the original color format.*/
356 if(cf == LV_COLOR_FORMAT_UNKNOWN) cf = draw_buf->header.cf;
357 if(stride == LV_STRIDE_AUTO) stride = lv_draw_buf_width_to_stride(w, cf);
358
359 uint32_t size = _calculate_draw_buf_size(w, h, cf, stride);
360
361 if(size > draw_buf->data_size) {
362 LV_LOG_TRACE("Draw buf too small for new shape");
363 LV_PROFILER_DRAW_END;
364 return NULL;
365 }
366
367 draw_buf->header.cf = cf;
368 draw_buf->header.w = w;
369 draw_buf->header.h = h;
370 draw_buf->header.stride = stride;
371
372 LV_PROFILER_DRAW_END;
373 return draw_buf;
374}

376void lv_draw_buf_destroy(lv_draw_buf_t * draw_buf)
377{
378 if(!draw_buf) {
379 return;
380 }
381 LV_PROFILER_DRAW_BEGIN;
382 if(lv_draw_buf_has_flag(draw_buf, LV_IMAGE_FLAGS_ALLOCATED)) {
383 LV_ASSERT_NULL(draw_buf->handlers);
384 LV_ASSERT_FORMAT_MSG(draw_buf->header.magic == LV_IMAGE_HEADER_MAGIC,
385 "Invalid draw buf magic: 0x%02X", draw_buf->header.magic);
386
387 const lv_draw_buf_handlers_t * handlers = draw_buf->handlers;
388 draw_buf_free(handlers, draw_buf->unaligned_data);
389 draw_buf->unaligned_data = NULL;
390 draw_buf->data = NULL;
391
392 /*Poison the magic before freeing so UAF access can be detected*/
393 draw_buf->header.magic = LV_IMAGE_HEADER_DEADBEEF;
394 lv_free(draw_buf);
395 }
396 else {
397 LV_LOG_ERROR("draw buffer is not allocated, ignored");
398 }
399 LV_PROFILER_DRAW_END;
400}

422void * lv_draw_buf_goto_xy(const lv_draw_buf_t * buf, uint32_t x, uint32_t y)
423{
424 LV_CHECK_ARG(buf != NULL, return NULL);
425 LV_CHECK_ARG_FORMAT_MSG(lv_draw_buf_is_position_valid(buf, x, y),
426 return NULL,
427 "coordinates out of range, x: %" LV_PRIu32 ", y: %"LV_PRIu32", w: %"LV_PRIu32", h: %"LV_PRIu32,
428 x, y, (uint32_t)buf->header.w, (uint32_t)buf->header.h);
429 LV_CHECK_ARG(buf->data != NULL, return NULL);
430
431 uint8_t * data = buf->data;
432
433 /*Skip palette*/
434 data += LV_COLOR_INDEXED_PALETTE_SIZE(buf->header.cf) * sizeof(lv_color32_t);
435 data += buf->header.stride * y;
436
437 if(x == 0) return data;
438
439 return data + x * lv_color_format_get_bpp(buf->header.cf) / 8;
440}

442lv_result_t lv_draw_buf_adjust_stride(lv_draw_buf_t * src, uint32_t stride)
443{
444 LV_CHECK_ARG(src != NULL, return LV_RESULT_INVALID);
445 LV_CHECK_ARG(src->data != NULL, return LV_RESULT_INVALID);
446 LV_PROFILER_DRAW_BEGIN;
447
448 const lv_image_header_t * header = &src->header;
449 uint32_t w = header->w;
450 uint32_t h = header->h;
451
452 if(!lv_draw_buf_has_flag(src, LV_IMAGE_FLAGS_MODIFIABLE)) {
453 LV_PROFILER_DRAW_END;
454 return LV_RESULT_INVALID;
455 }
456
457 /*Use global stride*/
458 if(stride == 0) stride = lv_draw_buf_width_to_stride(w, header->cf);
459
460 /*Check if stride already match*/
461 if(header->stride == stride) {
462 LV_PROFILER_DRAW_END;
463 return LV_RESULT_OK;
464 }
465
466 /*Calculate the minimal stride allowed from bpp*/
467 uint32_t bpp = lv_color_format_get_bpp(header->cf);
468 uint32_t min_stride = (w * bpp + 7) >> 3;
469 if(stride < min_stride) {
470 LV_LOG_WARN("New stride is too small. min: %" LV_PRId32, min_stride);
471 LV_PROFILER_DRAW_END;
472 return LV_RESULT_INVALID;
473 }
474
475 /*Check if buffer has enough space. */
476 uint32_t new_size = _calculate_draw_buf_size(w, h, header->cf, stride);
477 if(new_size > src->data_size) {
478 LV_PROFILER_DRAW_END;
479 return LV_RESULT_INVALID;
480 }
481
482 uint32_t offset = LV_COLOR_INDEXED_PALETTE_SIZE(header->cf) * 4;
483
484 if(stride > header->stride) {
485 /*Copy from the last line to the first*/
486 uint8_t * src_data = src->data + offset + header->stride * (h - 1);
487 uint8_t * dst_data = src->data + offset + stride * (h - 1);
488 for(uint32_t y = 0; y < h; y++) {
489 lv_memmove(dst_data, src_data, min_stride);
490 src_data -= header->stride;
491 dst_data -= stride;
492 }
493 }
494 else {
495 /*Copy from the first line to the last*/
496 uint8_t * src_data = src->data + offset;
497 uint8_t * dst_data = src->data + offset;
498 for(uint32_t y = 0; y < h; y++) {
499 lv_memmove(dst_data, src_data, min_stride);
500 src_data += header->stride;
501 dst_data += stride;
502 }
503 }
504
505 src->header.stride = stride;
506
507 LV_PROFILER_DRAW_END;
508 return LV_RESULT_OK;
509}

511lv_result_t lv_draw_buf_premultiply(lv_draw_buf_t * draw_buf)
512{
513 LV_CHECK_ARG(draw_buf != NULL, return LV_RESULT_INVALID);

533
534 LV_PROFILER_DRAW_END;

628static void buf_copy(lv_draw_buf_t * dest, const lv_area_t * dest_area,
629 const lv_draw_buf_t * src, const lv_area_t * src_area)
630{
631 LV_PROFILER_DRAW_BEGIN;
632 uint8_t * dest_bufc;
633 uint8_t * src_bufc;
634 int32_t line_width;
635
636 LV_ASSERT(dest != NULL);
637 LV_ASSERT(dest->data != NULL);
638 LV_ASSERT(src != NULL);
639 LV_ASSERT(src->data != NULL);
640
641 /*Source and dest color format must be same. Color conversion is not supported yet.*/
642 LV_ASSERT_FORMAT_MSG(dest->header.cf == src->header.cf, "Color format mismatch: %d != %d",
643 dest->header.cf, src->header.cf);
644
645 if(dest_area == NULL) line_width = dest->header.w;
646 else line_width = lv_area_get_width(dest_area);
647
648 /* For indexed image, copy the palette if we are copying full image area*/
649 if(dest_area == NULL || src_area == NULL) {
650 if(LV_COLOR_FORMAT_IS_INDEXED(dest->header.cf)) {
651 lv_memcpy(dest->data, src->data, LV_COLOR_INDEXED_PALETTE_SIZE(dest->header.cf) * sizeof(lv_color32_t));
652 }
653 }
654
655 /*Check source and dest area have same width*/
656 if((src_area == NULL && line_width != src->header.w) || \
657 (src_area != NULL && line_width != lv_area_get_width(src_area))) {
658 LV_ASSERT_MSG(0, "Source and destination areas have different width");
659 LV_PROFILER_DRAW_END;
660 return;
661 }
662
663 if(src_area) src_bufc = lv_draw_buf_goto_xy(src, src_area->x1, src_area->y1);
664 else src_bufc = lv_draw_buf_goto_xy(src, 0, 0);
665
666 if(dest_area) dest_bufc = lv_draw_buf_goto_xy(dest, dest_area->x1, dest_area->y1);
667 else dest_bufc = lv_draw_buf_goto_xy(dest, 0, 0);
668
669 int32_t start_y, end_y;
670 if(dest_area) {
671 start_y = dest_area->y1;
672 end_y = dest_area->y2;
673 }
674 else {
675 start_y = 0;
676 end_y = dest->header.h - 1;
677 }
678
679 uint32_t dest_stride = dest->header.stride;
680 uint32_t src_stride = src->header.stride;
681 uint32_t line_bytes = (line_width * lv_color_format_get_bpp(dest->header.cf) + 7) >> 3;
682
683 for(; start_y <= end_y; start_y++) {
684 lv_memcpy(dest_bufc, src_bufc, line_bytes);
685 dest_bufc += dest_stride;
686 src_bufc += src_stride;
687 }
688 LV_PROFILER_DRAW_END;
689}
