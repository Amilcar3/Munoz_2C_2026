10def safe_string(obj, field_name):
11 """Read a char* field as string, corrupted marker, or None (NULL/missing)."""
12 from lvglgdb.value import CorruptedValue
13 val = obj.safe_field(field_name)
14 if val is None:
15 return None
16 if not getattr(val, 'is_ok', True):
17 return str(val)
18 addr = int(val)
19 if not addr:
20 return None
21 return val.string(fallback=str(CorruptedValue(addr, MemoryError("unreadable"))))

59def safe_wrapper(obj, field_name, module_path, class_name):
60 """Read a struct field using its known Value wrapper, return snapshot dict.
61
62 Not every wrapper has a snapshot(); LVArray does not. Value forwards an
63 unknown attribute to gdb.Value.__getitem__, so asking for one raises
64 gdb.error rather than AttributeError, and the caller would otherwise lose
65 the whole widget - and its subtree - over a single field.
66 """
67 import gdb
68
69 val = obj.safe_field(field_name)
70 if val is None or not getattr(val, 'is_ok', True):
71 return None
72 import importlib
73 try:
74 mod = importlib.import_module(module_path)
75 cls = getattr(mod, class_name)
76 wrapper = cls(val)
77 if getattr(type(wrapper), "snapshot", None) is None:
78 # An embedded struct - lv_scale_t.needles - is not convertible to an
79 # int, so the address has to come from the field itself.
80 if val.type.strip_typedefs().code != gdb.TYPE_CODE_PTR:
81 val = val.address
82 return {"addr": hex(int(val)), "type": class_name}
83 return wrapper.snapshot().as_dict()
84 except (gdb.error, gdb.MemoryError, ImportError, AttributeError, TypeError):
85 return None
