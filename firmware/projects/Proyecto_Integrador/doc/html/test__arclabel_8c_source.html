208 /* Use fixed-point LUT for gamma-corrected opacity (base perceived brightness from 1.0 to 0.5)
209 * Values are pre-calculated: (pow(1.0 - (i/15.0)*0.5, 1/2.2) * 255) */
210 static const uint8_t opa_gamma_lut[] = {
211 255, 251, 247, 242, 238, 234, 229, 225, 220, 215, 210, 205, 200, 195, 189, 184
212 };
213
214 int32_t opa_gamma = opa_gamma_lut[i];
215
216 /* Apply light attenuation: final_opa = opa_gamma / (1 + (i*k)^2)
217 * With k = 0.25 (1/4), the formula becomes: opa_gamma * 16 / (16 + i*i) */
218 int32_t final_opa = (opa_gamma * 16) / (16 + i * i);
219 if(final_opa < 0) final_opa = 0;
220 if(final_opa > LV_OPA_COVER) final_opa = LV_OPA_COVER;
221
222 lv_obj_set_style_opa(arclabel, final_opa, LV_PART_MAIN);
223 }
