90bool LottieLoader::header()
91{
92 //A single thread doesn't need to perform intensive tasks.
93 if (TaskScheduler::threads() == 0) {
94 LoadModule::read();
95 run(0);
96 if (comp) {
97 w = static_cast<float>(comp->w);
98 h = static_cast<float>(comp->h);
99 frameDuration = comp->duration();
100 frameCnt = comp->frameCnt();
101 frameRate = comp->frameRate;
102 return true;
103 } else {
104 return false;
105 }
106 }
107
108 //Quickly validate the given Lottie file without parsing in order to get the animation info.
109 auto startFrame = 0.0f;
110 auto endFrame = 0.0f;
111 uint32_t depth = 0;
112
113 auto p = content;
114
115 while (*p != '\0') {
116 if (*p == '{') {
117 ++depth;
118 ++p;
119 continue;
120 }
121 if (*p == '}') {
122 --depth;
123 ++p;
124 continue;
125 }
126 if (depth != 1) {
127 ++p;
128 continue;
129 }
130 //version.
131 if (!strncmp(p, "\"v\":", 4)) {
132 p += 4;
133 continue;
134 }
135
136 //framerate
137 if (!strncmp(p, "\"fr\":", 5)) {
138 p += 5;
139 auto e = strstr(p, ",");
140 if (!e) e = strstr(p, "}");
141 frameRate = strToFloat(p, nullptr);
142 p = e;
143 continue;
144 }
145
146 //start frame
147 if (!strncmp(p, "\"ip\":", 5)) {
148 p += 5;
149 auto e = strstr(p, ",");
150 if (!e) e = strstr(p, "}");
151 startFrame = strToFloat(p, nullptr);
152 p = e;
153 continue;
154 }
155
156 //end frame
157 if (!strncmp(p, "\"op\":", 5)) {
158 p += 5;
159 auto e = strstr(p, ",");
160 if (!e) e = strstr(p, "}");
161 endFrame = strToFloat(p, nullptr);
162 p = e;
163 continue;
164 }
165
166 //width
167 if (!strncmp(p, "\"w\":", 4)) {
168 p += 4;
169 auto e = strstr(p, ",");
170 if (!e) e = strstr(p, "}");
171 w = strToFloat(p, nullptr);
172 p = e;
173 continue;
174 }
175 //height
176 if (!strncmp(p, "\"h\":", 4)) {
177 p += 4;
178 auto e = strstr(p, ",");
179 if (!e) e = strstr(p, "}");
180 h = strToFloat(p, nullptr);
181 p = e;
182 continue;
183 }
184 ++p;
185 }
186
187 if (frameRate < FLOAT_EPSILON) {
188 TVGLOG("LOTTIE", "Not a Lottie file? Frame rate is 0!");
189 return false;
190 }
191
192 frameCnt = (endFrame - startFrame);
193 frameDuration = frameCnt / frameRate;
194
195 TVGLOG("LOTTIE", "info: frame rate = %f, duration = %f size = %f x %f", frameRate, frameDuration, w, h);
196
197 return true;
198}

220bool LottieLoader::open(const string& path)
221{
222 auto f = fopen(path.c_str(), "r");
223 if (!f) return false;
224
225 fseek(f, 0, SEEK_END);
226
227 size = ftell(f);
228 if (size == 0) {
229 fclose(f);
230 return false;
231 }
232
233 auto content = (char*)(lv_malloc(sizeof(char) * size + 1));
234 LV_ASSERT_MALLOC(content);
235 fseek(f, 0, SEEK_SET);
236 auto ret = fread(content, sizeof(char), size, f);
237 if (ret < size) {
238 fclose(f);
239 return false;
240 }
241 content[size] = '\0';
242
243 fclose(f);
244
245 this->dirName = strDirname(path.c_str());
246 this->content = content;
247 this->copy = true;
248
249 return header();
250}

294bool LottieLoader::override(const char* slot)
295{
296 if (!ready() || comp->slots.count == 0) return false;
297
298 auto success = true;
299
300 //override slots
301 if (slot) {
302 //Copy the input data because the JSON parser will encode the data immediately.
303 auto temp = lv_strdup(slot);
304
305 //parsing slot json
306 LottieParser parser(temp, dirName);
307 parser.comp = comp;
308
309 auto idx = 0;
310 while (auto sid = parser.sid(idx == 0)) {
311 for (auto s = comp->slots.begin(); s < comp->slots.end(); ++s) {
312 if (strcmp((*s)->sid, sid)) continue;
313 if (!parser.apply(*s)) success = false;
314 break;
315 }
316 ++idx;
317 }
318
319 if (idx < 1) success = false;
320 lv_free(temp);
321 rebuild = overridden = success;
322 //reset slots
323 } else if (overridden) {
324 for (auto s = comp->slots.begin(); s < comp->slots.end(); ++s) {
325 (*s)->reset();
326 }
327 overridden = false;
328 rebuild = true;
329 }
330 return success;
331}

334bool LottieLoader::frame(float no)
335{
336 auto frameNo = no + startFrame();
337
338 //This ensures that the target frame number is reached.
339 frameNo *= 10000.0f;
340 frameNo = nearbyintf(frameNo);
341 frameNo *= 0.0001f;
342
343 //Skip update if frame diff is too small.
344 if (fabsf(this->frameNo - frameNo) <= 0.0009f) return false;
345
346 this->done();
347
348 this->frameNo = frameNo;
349
350 TaskScheduler::request(this);
351
352 return true;
353}

403bool LottieLoader::segment(const char* marker, float& begin, float& end)
404{
405 if (!ready() || comp->markers.count == 0) return false;
406
407 for (auto m = comp->markers.begin(); m < comp->markers.end(); ++m) {
408 if (!strcmp(marker, (*m)->name)) {
409 begin = (*m)->time / frameCnt;
410 end = ((*m)->time + (*m)->duration) / frameCnt;
411 return true;
412 }
413 }
414 return false;
415}
