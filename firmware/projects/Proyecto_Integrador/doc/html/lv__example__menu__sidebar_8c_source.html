149static void switch_handler(lv_event_t * e)
150{
151 lv_event_code_t code = lv_event_get_code(e);
152 lv_obj_t * menu = (lv_obj_t *)lv_event_get_user_data(e);
153 lv_obj_t * obj = lv_event_get_target_obj(e);
154 if(code == LV_EVENT_VALUE_CHANGED) {
155 if(lv_obj_has_state(obj, LV_STATE_CHECKED)) {
156 lv_menu_set_page(menu, NULL);
157 lv_menu_set_sidebar_page(menu, root_page);
158 lv_obj_send_event(lv_obj_get_child(lv_obj_get_child(lv_menu_get_cur_sidebar_page(menu), 0), 0), LV_EVENT_CLICKED,
159 NULL);
160 }
161 else {
162 lv_menu_set_sidebar_page(menu, NULL);
163 lv_menu_clear_history(menu); /* Clear history because we will be showing the root page later */
164 lv_menu_set_page(menu, root_page);
165 }
166 }
