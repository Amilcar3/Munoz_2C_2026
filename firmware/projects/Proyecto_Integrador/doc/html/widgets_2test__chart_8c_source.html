115static void chart_event_cb(lv_event_t * e)
116{
117 lv_event_code_t code = lv_event_get_code(e);
118 lv_obj_t * obj = lv_event_get_target(e);
119
120 if(code == LV_EVENT_DRAW_TASK_ADDED) {
121 lv_draw_task_t * draw_task = lv_event_get_param(e);
122 lv_draw_dsc_base_t * base_dsc = draw_task->draw_dsc;
123
124 lv_chart_type_t chart_type = lv_chart_get_type(obj);
125 if(chart_type == LV_CHART_TYPE_LINE && base_dsc->part != LV_PART_INDICATOR) return;
126 if(chart_type == LV_CHART_TYPE_BAR && base_dsc->part != LV_PART_ITEMS) return;
127
128 const lv_chart_series_t * ser = NULL;
129 for(uint32_t i = 0; i < base_dsc->id1 + 1; i++) {
130 ser = lv_chart_get_series_next(obj, ser);
131 }
132
133 char buf[8];
134 lv_snprintf(buf, sizeof(buf), "%"LV_PRIu32, ser->y_points[base_dsc->id2]);
135
136 lv_point_t text_size;
137 lv_text_attributes_t attributes = {0};
138
139 attributes.letter_space = 0;
140 attributes.line_space = 0;
141 attributes.max_width = LV_COORD_MAX;
142 attributes.text_flags = LV_TEXT_FLAG_NONE;
143
144 lv_text_get_size_attributes(&text_size, buf, LV_FONT_DEFAULT, &attributes);
145
146 lv_area_t txt_area;
147 txt_area.x1 = draw_task->area.x1 + lv_area_get_width(&draw_task->area) / 2 - text_size.x / 2;
148 txt_area.x2 = txt_area.x1 + text_size.x;
149 txt_area.y2 = draw_task->area.y1 - LV_DPX(15);
150 txt_area.y1 = txt_area.y2 - text_size.y;
151
152 lv_area_t bg_area;
153 bg_area.x1 = txt_area.x1 - LV_DPX(8);
154 bg_area.x2 = txt_area.x2 + LV_DPX(8);
155 bg_area.y1 = txt_area.y1 - LV_DPX(8);
156 bg_area.y2 = txt_area.y2 + LV_DPX(8);
157
158 lv_draw_rect_dsc_t rect_dsc;
159 lv_draw_rect_dsc_init(&rect_dsc);
160 rect_dsc.bg_color = ser->color;
161 rect_dsc.radius = LV_DPX(5);
162 lv_draw_rect(base_dsc->layer, &rect_dsc, &bg_area);
163
164 lv_draw_label_dsc_t label_dsc;
165 lv_draw_label_dsc_init(&label_dsc);
166 label_dsc.color = lv_color_white();
167 label_dsc.font = LV_FONT_DEFAULT;
168 label_dsc.text = buf;
169 label_dsc.text_local = true;
170 lv_draw_label(base_dsc->layer, &label_dsc, &txt_area);
171 }

174void test_draw_task_hooking(void)
175{
176 lv_obj_t * chart_wrapper = lv_obj_create(lv_screen_active());
177 lv_obj_set_size(chart_wrapper, 600, 300);

183 lv_chart_set_div_line_count(chart, 0, 21);
184 lv_chart_set_point_count(chart, 21);
185 lv_obj_add_event_cb(chart, chart_event_cb, LV_EVENT_DRAW_TASK_ADDED, NULL);
186
187 lv_chart_series_t * chart_set1 = lv_chart_add_series(chart, lv_palette_main(LV_PALETTE_YELLOW), 0);

190 lv_chart_set_axis_range(chart, LV_CHART_AXIS_PRIMARY_Y, 0, 700);
191
192 int32_t points[21] = {0, 31, 59, 81, 95, 100, 95, 81, 59, 31, 0, -31, -59, -81, -95, -100, -95, -81, -59, -31, 0};
193
194 for(uint32_t i = 0; i < 21; i++) {
195 lv_chart_set_next_value(chart, chart_set1, points[i] + 100);
196 lv_chart_set_next_value(chart, chart_set2, points[i] + 300);
197 lv_chart_set_next_value(chart, chart_set3, points[i] + 500);
198 }
199
200 lv_obj_scroll_to_x(chart_wrapper, 300, LV_ANIM_OFF);
201
202 lv_chart_set_type(chart, LV_CHART_TYPE_LINE);
203 TEST_ASSERT_EQUAL_SCREENSHOT("widgets/chart_line_draw_hook.png");
204
205 lv_chart_set_type(chart, LV_CHART_TYPE_BAR);
