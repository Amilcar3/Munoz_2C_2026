147static void draw_letter_cb(lv_draw_task_t * t, lv_draw_glyph_dsc_t * glyph_draw_dsc,
148 lv_draw_fill_dsc_t * fill_draw_dsc, const lv_area_t * fill_area)
149{
150 lv_draw_vg_lite_unit_t * u = (lv_draw_vg_lite_unit_t *)t->draw_unit;
151 if(glyph_draw_dsc) {
152 switch(glyph_draw_dsc->format) {
153 case LV_FONT_GLYPH_FORMAT_A1:
154 case LV_FONT_GLYPH_FORMAT_A2:
155 case LV_FONT_GLYPH_FORMAT_A3:
156 case LV_FONT_GLYPH_FORMAT_A4:
157 case LV_FONT_GLYPH_FORMAT_A8: {
158 const lv_font_t * resolved_font = glyph_draw_dsc->g->resolved_font;
159 vg_lite_buffer_t src_buf;
160 if(lv_font_has_static_bitmap_internal(resolved_font) && init_buffer_from_glyph_dsc(&src_buf, glyph_draw_dsc->g)) {
161 }
162 else {
163 if(resolved_font->release_glyph) {
164 /* For dynamic fonts, its internal implementation already supports cache management. */
165 glyph_draw_dsc->glyph_data = lv_font_get_glyph_bitmap_internal(glyph_draw_dsc->g, glyph_draw_dsc->_draw_buf);
166 }
167 else {
168 /* For non-cached unaligned fonts, we need to manage the cache manually. */
169 glyph_draw_dsc->glyph_data = lv_vg_lite_bitmap_font_cache_get(u, glyph_draw_dsc->g);
170 }
171
172 if(!glyph_draw_dsc->glyph_data) {
173 return;
174 }
175
176 lv_vg_lite_buffer_from_draw_buf(&src_buf, glyph_draw_dsc->glyph_data);
177 }
178
179 draw_letter_bitmap(t, glyph_draw_dsc, &src_buf);
180 }
181 break;
182
183 case LV_FONT_GLYPH_FORMAT_VECTOR:
184 draw_letter_outline(t, glyph_draw_dsc);
185
186 break;
187
188 case LV_FONT_GLYPH_FORMAT_IMAGE: {
189 glyph_draw_dsc->glyph_data = lv_font_get_glyph_bitmap_internal(glyph_draw_dsc->g, glyph_draw_dsc->_draw_buf);
190 if(!glyph_draw_dsc->glyph_data) {
191 return;
192 }
193
194 lv_draw_image_dsc_t image_dsc;
195 lv_draw_image_dsc_init(&image_dsc);
196 image_dsc.opa = glyph_draw_dsc->opa;
197 image_dsc.src = glyph_draw_dsc->glyph_data;
198 image_dsc.rotation = glyph_draw_dsc->rotation;
199 image_dsc.pivot = glyph_draw_dsc->pivot;
200 lv_draw_vg_lite_img(t, &image_dsc, glyph_draw_dsc->letter_coords, false);
201 }
202 break;
203
204#if LV_USE_FONT_PLACEHOLDER
205 case LV_FONT_GLYPH_FORMAT_NONE: {
206 if(glyph_draw_dsc->bg_coords == NULL) break;
207 /* Draw a placeholder rectangle*/
208 lv_draw_border_dsc_t border_draw_dsc;
209 lv_draw_border_dsc_init(&border_draw_dsc);
210 border_draw_dsc.opa = glyph_draw_dsc->opa;
211 border_draw_dsc.color = glyph_draw_dsc->color;
212 border_draw_dsc.width = 1;
213 lv_draw_vg_lite_border(t, &border_draw_dsc, glyph_draw_dsc->bg_coords);
214 }
215 break;
216#endif /* LV_USE_FONT_PLACEHOLDER */
217
218 default:
219 break;
220 }
221 }
222
223 if(fill_draw_dsc && fill_area) {
224 lv_draw_vg_lite_fill(t, fill_draw_dsc, fill_area);
225 }
226
227 /* Flush in time to avoid accumulation of drawing commands */
228 u->letter_count++;
229 if(u->letter_count > PATH_FLUSH_COUNT_MAX) {
230 lv_vg_lite_flush(u);
231 }
232}

234static inline bool init_buffer_from_glyph_dsc(vg_lite_buffer_t * buffer, lv_font_glyph_dsc_t * g_dsc)
235{
236 const void * glyph_bitmap = lv_font_get_glyph_static_bitmap_internal(g_dsc);
237 if(!glyph_bitmap) {
238 return false;
239 }
240
241 if(!LV_VG_LITE_IS_ALIGNED(glyph_bitmap, 16)) {
242 LV_LOG_WARN("Glyph data %p is not aligned to 16 bytes", glyph_bitmap);
243 return false;
244 }
245
246 if(g_dsc->stride == 0 || !LV_VG_LITE_IS_ALIGNED(g_dsc->stride, 16)) {
247 LV_LOG_WARN("Glyph stride %" LV_PRIu32 " is not aligned to 16 bytes", g_dsc->stride);
248 return false;
249 }
250
251 lv_vg_lite_buffer_init(buffer, glyph_bitmap, g_dsc->box_w, g_dsc->box_h, g_dsc->stride, VG_LITE_A8, false);
252 return true;
253}

256static inline void convert_letter_matrix(vg_lite_matrix_t * matrix, const lv_draw_glyph_dsc_t * dsc)
257{
258 vg_lite_translate(dsc->letter_coords->x1, dsc->letter_coords->y1, matrix);
259
260 if(!dsc->rotation) {
261 return;
262 }
263
264 const lv_point_t pivot = {
265 .x = dsc->pivot.x,
266 .y = dsc->g->box_h + dsc->g->ofs_y
267 };
268 vg_lite_translate(pivot.x, pivot.y, matrix);
269 vg_lite_rotate(dsc->rotation / 10.0f, matrix);
270 vg_lite_translate(-pivot.x, -pivot.y, matrix);
271}

273static bool draw_letter_clip_areas(lv_draw_task_t * t, const lv_draw_glyph_dsc_t * dsc, lv_area_t * letter_area,
274 lv_area_t * clipped_area)
275{
276 *letter_area = *dsc->letter_coords;
277
278 if(dsc->rotation) {
279 const lv_point_t pivot = {
280 .x = dsc->pivot.x,
281 .y = dsc->g->box_h + dsc->g->ofs_y
282 };
283
284 lv_image_buf_get_transformed_area(
285 letter_area,
286 lv_area_get_width(dsc->letter_coords),
287 lv_area_get_height(dsc->letter_coords),
288 dsc->rotation,
289 LV_SCALE_NONE,
290 LV_SCALE_NONE,
291 &pivot);
292 lv_area_move(letter_area, dsc->letter_coords->x1, dsc->letter_coords->y1);
293 }
294
295 if(!lv_area_intersect(clipped_area, &t->clip_area, letter_area)) {
296 return false;
297 }
298
299 return true;
300}

302static void draw_letter_bitmap(lv_draw_task_t * t, const lv_draw_glyph_dsc_t * dsc, vg_lite_buffer_t * src_buf)
303{
304 LV_PROFILER_DRAW_BEGIN;
305
306 lv_area_t image_area;
307 lv_area_t clip_area;
308 if(!draw_letter_clip_areas(t, dsc, &image_area, &clip_area)) {
309 LV_PROFILER_DRAW_END;
310 return;
311 }
312
313 lv_draw_vg_lite_unit_t * u = (lv_draw_vg_lite_unit_t *)t->draw_unit;
314
315 vg_lite_matrix_t matrix = u->global_matrix;
316 convert_letter_matrix(&matrix, dsc);
317
318 const vg_lite_color_t color = lv_vg_lite_color(dsc->color, dsc->opa, true);
319
320 const int32_t clip_offset_x = clip_area.x1 - image_area.x1;
321 const int32_t clip_offset_y = clip_area.y1 - image_area.y1;
322
323 /* If rotation is not required, blit directly */
324 if(!dsc->rotation
325#if LV_VG_LITE_DISABLE_BLIT_RECT_OFFSET
330 && (clip_offset_x == 0 && clip_offset_y == 0)
331#endif
332 ) {
333 vg_lite_rectangle_t rect = {
334 .x = clip_offset_x,
335 .y = clip_offset_y,
336 .width = lv_area_get_width(&clip_area),
337 .height = lv_area_get_height(&clip_area)
338 };
339
340 /* add offset for clipped area */
341 if(rect.x || rect.y) {
342 vg_lite_translate(rect.x, rect.y, &matrix);
343 }
344
345 lv_vg_lite_blit_rect(
346 &u->target_buffer,
347 src_buf,
348 &rect,
349 &matrix,
350 VG_LITE_BLEND_SRC_OVER,
351 color,
352 lv_vg_lite_matrix_get_filter(&matrix));
353 }
354 else {
355 lv_vg_lite_path_t * path = lv_vg_lite_path_get(u, VG_LITE_S16);
356 lv_vg_lite_path_append_rect(
357 path,
358 image_area.x1, image_area.y1,
359 lv_area_get_width(&image_area), lv_area_get_height(&image_area),
360 0);
361 lv_vg_lite_path_end(path);
362 lv_vg_lite_path_set_bounding_box_area(path, &clip_area);
363
364 vg_lite_matrix_t path_matrix = u->global_matrix;
365
366 lv_vg_lite_draw_pattern(
367 &u->target_buffer,
368 lv_vg_lite_path_get_path(path),
369 VG_LITE_FILL_EVEN_ODD,
370 &path_matrix,
371 src_buf,
372 &matrix,
373 VG_LITE_BLEND_SRC_OVER,
374 VG_LITE_PATTERN_COLOR,
375 0,
376 color,
377 lv_vg_lite_matrix_get_filter(&matrix));
378
379 lv_vg_lite_path_drop(u, path);
380 }
381
382 /* Check if the data has cache and add it to the pending list */
383 if(dsc->g->entry) {
384 /* Increment the cache reference count */
385 lv_cache_entry_acquire_data(dsc->g->entry);
386 lv_vg_lite_pending_add(u->letter_pending, dsc->g);
387 }
388
389 LV_PROFILER_DRAW_END;
390}

400static void draw_letter_outline(lv_draw_task_t * t, const lv_draw_glyph_dsc_t * dsc)
401{
402 LV_PROFILER_DRAW_BEGIN;
403
404 lv_area_t letter_area;
405 lv_area_t path_clip_area;
406 if(!draw_letter_clip_areas(t, dsc, &letter_area, &path_clip_area)) {
407 LV_PROFILER_DRAW_END;
408 return;
409 }
410
411 lv_draw_vg_lite_unit_t * u = (lv_draw_vg_lite_unit_t *)t->draw_unit;
412
413 /* vg-lite bounding_box will crop the pixels on the edge, so +1px is needed here */
414 path_clip_area.x2++;
415 path_clip_area.y2++;
416
417 lv_vg_lite_path_t * outline = (void *)dsc->glyph_data;
418 const lv_point_t glyph_pos = {
419 dsc->letter_coords->x1 - dsc->g->ofs_x,
420 dsc->letter_coords->y1 + dsc->g->box_h + dsc->g->ofs_y
421 };
422 /* scale size */
423 float scale = 1.0;
424#if LV_USE_FREETYPE
425 if(lv_freetype_is_outline_font(dsc->g->resolved_font)) {
426 scale = FT_F26DOT6_TO_PATH_SCALE(lv_freetype_outline_get_scale(dsc->g->resolved_font));
427 }
428#endif
429 const bool has_rotation_with_clipped = dsc->rotation && !lv_area_is_in(&letter_area, &t->clip_area, false);
430
431 /* calc convert matrix */
432 vg_lite_matrix_t matrix;
433 vg_lite_identity(&matrix);
434
435 if(!has_rotation_with_clipped && dsc->rotation) {
436 vg_lite_translate(glyph_pos.x + dsc->pivot.x, glyph_pos.y, &matrix);
437 vg_lite_rotate(dsc->rotation / 10.0f, &matrix);
438 vg_lite_translate(-dsc->pivot.x, 0, &matrix);
439 }
440 else {
441 vg_lite_translate(glyph_pos.x, glyph_pos.y, &matrix);
442 }
443
444 vg_lite_scale(scale, scale, &matrix);
445
446 /* matrix for drawing, different from matrix for calculating the bounding box */
447 vg_lite_matrix_t draw_matrix = u->global_matrix;
448 lv_vg_lite_matrix_multiply(&draw_matrix, &matrix);
449
450 /* calc inverse matrix */
451 vg_lite_matrix_t result;
452 if(!lv_vg_lite_matrix_inverse(&result, &matrix)) {
453 LV_LOG_ERROR("no inverse matrix");
454 lv_vg_lite_matrix_dump_info(&matrix);
455 LV_PROFILER_DRAW_END;
456 return;
457 }
458
459 const int32_t stroke_width_scaled = (int32_t)(dsc->outline_stroke_width / scale);
460 const lv_point_precise_t p1 = { path_clip_area.x1 - stroke_width_scaled, path_clip_area.y1 - stroke_width_scaled };
461 const lv_point_precise_t p1_res = lv_vg_lite_matrix_transform_point(&result, &p1);
462 const lv_point_precise_t p2 = { path_clip_area.x2 + stroke_width_scaled, path_clip_area.y2 + stroke_width_scaled };
463 const lv_point_precise_t p2_res = lv_vg_lite_matrix_transform_point(&result, &p2);
464
465 if(has_rotation_with_clipped) {
470 vg_lite_matrix_t internal_matrix;
471 vg_lite_identity(&internal_matrix);
472 const float pivot_x = (dsc->pivot.x + dsc->g->ofs_x) / scale;
473 const float pivot_y = dsc->g->box_h + dsc->g->ofs_y;
474 vg_lite_translate(pivot_x, pivot_y, &internal_matrix);
475 vg_lite_rotate(dsc->rotation / 10.0f, &internal_matrix);
476 vg_lite_translate(-pivot_x, -pivot_y, &internal_matrix);
477
478 lv_vg_lite_path_t * outline_transformed = lv_vg_lite_path_get(u, VG_LITE_FP32);
479 lv_vg_lite_path_set_transform(outline_transformed, &internal_matrix);
480 lv_vg_lite_path_for_each_data(lv_vg_lite_path_get_path(outline), outline_iter_cb, outline_transformed);
481 lv_vg_lite_path_set_bounding_box(outline_transformed, p1_res.x, p2_res.y, p2_res.x, p1_res.y);
482
483 lv_vg_lite_draw(
484 &u->target_buffer,
485 lv_vg_lite_path_get_path(outline_transformed),
486 VG_LITE_FILL_NON_ZERO,
487 &draw_matrix,
488 VG_LITE_BLEND_SRC_OVER,
489 lv_vg_lite_color(dsc->color, dsc->opa, true));
490
491 lv_vg_lite_path_drop(u, outline_transformed);
492
493 LV_PROFILER_DRAW_END;
494 return;
495 }
496
497 if(dsc->rotation) {
498 /* The bounding rectangle before scaling relative to the original coordinates of the path */
499 lv_area_t box_area;
500 box_area.x1 = dsc->g->ofs_x - stroke_width_scaled;
501 box_area.y1 = -dsc->g->box_h - dsc->g->ofs_y - stroke_width_scaled;
502 lv_area_set_width(&box_area, dsc->g->box_w + stroke_width_scaled * 2);
503 lv_area_set_height(&box_area, dsc->g->box_h + stroke_width_scaled * 2);
504
505 /* Workaround for loss of rotation precision */
506 lv_area_increase(&box_area, 5, 5);
507
508 /* Scale the path area to fit the original path data */
509 lv_vg_lite_path_set_bounding_box(outline,
510 box_area.x1 / scale,
511 box_area.y1 / scale,
512 box_area.x2 / scale,
513 box_area.y2 / scale);
514 }
515 else {
516 lv_vg_lite_path_set_bounding_box(outline, p1_res.x, p2_res.y, p2_res.x, p1_res.y);
517 }
518
519
520 vg_lite_path_t * vg_path = lv_vg_lite_path_get_path(outline);
521 if(stroke_width_scaled > 0) {
522 /*Set the stroke and update the path type */
523 LV_VG_LITE_CHECK_ERROR(vg_lite_set_stroke(vg_path,
524 VG_LITE_CAP_ROUND, VG_LITE_JOIN_ROUND, stroke_width_scaled, 1.0,
525 NULL, 0, 0,
526 lv_vg_lite_color(dsc->outline_stroke_color, dsc->outline_stroke_opa, true)), {});
527
528 LV_VG_LITE_CHECK_ERROR(vg_lite_set_path_type(vg_path, VG_LITE_DRAW_STROKE_PATH), {});
529 LV_VG_LITE_CHECK_ERROR(vg_lite_update_stroke(vg_path), {});
530
531 /*Draw the stroke (transparent fill)*/
532 lv_vg_lite_draw(
533 &u->target_buffer,
534 vg_path,
535 VG_LITE_FILL_NON_ZERO,
536 &draw_matrix,
537 VG_LITE_BLEND_SRC_OVER,
538 lv_vg_lite_color(dsc->color, 0, true));
539
540 /*Restore the path to fill mode, it will release the local memory allocated for the stroke op */
541 LV_VG_LITE_CHECK_ERROR(vg_lite_set_draw_path_type(vg_path, VG_LITE_DRAW_FILL_PATH), {});
542
543 }
544
545 LV_VG_LITE_CHECK_ERROR(vg_lite_set_path_type(lv_vg_lite_path_get_path(outline), VG_LITE_DRAW_FILL_PATH), {});
546 lv_vg_lite_draw(
547 &u->target_buffer,
548 lv_vg_lite_path_get_path(outline),
549 VG_LITE_FILL_NON_ZERO,
550 &draw_matrix,
551 VG_LITE_BLEND_SRC_OVER,
552 lv_vg_lite_color(dsc->color, dsc->opa, true));
553
554
555 LV_PROFILER_DRAW_END;
556}

558static void outline_iter_cb(void * user_data, uint8_t op_code, const float * data, uint32_t len)
559{
560 LV_UNUSED(len);
561 typedef struct {
562 float x;
563 float y;
564 } point_t;
565
566 lv_vg_lite_path_t * path = user_data;
567 const point_t * pt = (point_t *)data;
568
569 switch(op_code) {
570 case VLC_OP_MOVE:
571 lv_vg_lite_path_move_to(path, pt->x, pt->y);
572 break;
573 case VLC_OP_LINE:
574 lv_vg_lite_path_line_to(path, pt->x, pt->y);
575 break;
576 case VLC_OP_QUAD:
577 lv_vg_lite_path_quad_to(path, pt[0].x, pt[0].y, pt[1].x, pt[1].y);
578 break;
579 case VLC_OP_CUBIC:
580 lv_vg_lite_path_cubic_to(path, pt[0].x, pt[0].y, pt[1].x, pt[1].y, pt[2].x, pt[2].y);

636static void freetype_outline_event_cb(lv_event_t * e)
637{
638 LV_PROFILER_DRAW_BEGIN;
639 lv_event_code_t code = lv_event_get_code(e);
640 lv_freetype_outline_event_param_t * param = lv_event_get_param(e);
641 switch(code) {
642 case LV_EVENT_CREATE: {
643 lv_vg_lite_path_t * path = lv_vg_lite_path_create(PATH_DATA_COORD_FORMAT);
644 param->outline = path;
645
646 /* Pre-allocate path memory using sizes info to avoid realloc during decompose */
647 if(param->sizes.segments_size > 0) {
648 const uint8_t fmt_len = lv_vg_lite_path_format_len(PATH_DATA_COORD_FORMAT);
649 /* Each segment has 1 op code, plus all coordinate data, plus 1 end op */
650 const size_t total_size = (param->sizes.segments_size + param->sizes.data_size + 1) * fmt_len;
651 lv_vg_lite_path_reserve_space(path, total_size);
652 }
653 }
654 break;
655 case LV_EVENT_DELETE:
656 if(param->outline) lv_vg_lite_path_destroy(param->outline);
657 break;
658 case LV_EVENT_INSERT:
659 vg_lite_outline_push(param);
660 break;
661 default:
662 LV_LOG_WARN("unknown event code: %d", code);
663 break;
664 }
665 LV_PROFILER_DRAW_END;
666}
