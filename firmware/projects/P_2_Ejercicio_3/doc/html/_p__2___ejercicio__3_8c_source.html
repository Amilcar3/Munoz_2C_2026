240void FuncUart(void* param){
241
242 uint8_t c;
243 UartReadByte(UART_PC, &c);
244 switch(c){
245 case 'O':
246 case 'o':
247 button_1 = !button_1;
248 break;
249 case 'H':
250 case 'h':
251 button_2 = !button_2;
252 break;
253 case 'I':
254 case 'i':
255 switch (unit_inches) {
256 case true:
257 unit_inches = !unit_inches;
258 distance = (uint16_t)(distance * 2.54); // convertir a cm
259 max_distance = (uint16_t)(max_distance * 2.54); // convertir a cm
260 held_distance = (uint16_t)(held_distance * 2.54);
261 break;
262 case false:
263 unit_inches = !unit_inches;
264 distance = (uint16_t)(distance / 2.54); // convertir a pulgadas
265 max_distance = (uint16_t)(max_distance / 2.54); // convertir a pulgadas
266 held_distance = (uint16_t)(held_distance / 2.54);
267 break;
268
269 }
270 break;
271 case 'M':
272 case 'm':
273 show_max = !show_max;
274 break;
275 case 'F':
276 case 'f':
277 if (measure_period > 100000){ // evitá que llegue a 0 o se vuelva negativo
278 measure_period -= 100000;
279 TimerUpdatePeriod(TIMER_A, measure_period);
280 }
281 break;
282 case 'S':
283 case 's':
284 measure_period += 100000;
285 TimerUpdatePeriod(TIMER_A, measure_period);
286 break;
287 }
288
289}

299void app_main(void){
300
301 SwitchesInit(); /* Inicialización de switches */
302 LedsInit(); /* Inicialización de LEDs */
303 HcSr04Init(GPIO_3, GPIO_2); /* inicialización del sensor HC-SR04 */
304 LcdItsE0803Init();
305
306 /* Utilozando la funcion SwitchActivInt*/
307 SwitchActivInt(SWITCH_1, switch1Action, NULL);
308 SwitchActivInt(SWITCH_2, switch2Action, NULL);
309
310
311 /* Inicialización de timers */
312 timer_config_t timer_measure = {
313 .timer = TIMER_A,
314 .period = CONFIG_TASK_MEASURE_PERIOD,
315 .func_p = FuncTimerA,
316 .param_p = NULL
317 };
318 TimerInit(&timer_measure);
319
320 timer_config_t timer_display = {
321 .timer = TIMER_B,
322 .period = CONFIG_TASK_DISPLAY_PERIOD,
323 .func_p = FuncTimerB,
324 .param_p = NULL
325 };
326 TimerInit(&timer_display);
327 /*Estructura de UART*/
328 serial_config_t my_uart = {
329 .port = UART_PC, // UART_PC o UART_CONNECTOR
330 .baud_rate = 115200, // tiene que coincidir con el otro extremo
331 .func_p = FuncUart, // callback de recepción: se invoca al llegar un byte
332 .param_p = NULL
333 };
334 UartInit(&my_uart);
335
336 xTaskCreate(&measureTask, "Measure", 512, NULL, 5, &measure_task_handle);
337 xTaskCreate(&displayTask, "Display", 512, NULL, 5, &display_task_handle);
338 /* Inicialización del conteo de timers */
339 TimerStart(timer_measure.timer);
340 TimerStart(timer_display.timer);
341}
