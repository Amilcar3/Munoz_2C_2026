116void lv_obj_add_style(lv_obj_t * obj, const lv_style_t * style, lv_style_selector_t selector)
117{
118 LV_CHECK_ARG(obj != NULL, return);
119 LV_CHECK_ARG(style != NULL, return);
120 LV_CHECK_ARG_MSG(obj->style_cnt < 63, return,
121 "obj->style_cnt is restricted to 6 bits, so we can't store more than 63 styles");
122
123 lv_part_t part = lv_obj_style_get_selector_part(selector);
124
125 /*If the draw size changes invalidate the old area first*/
126 if(part == LV_PART_MAIN && style_has_flag(style, DRAW_AREA_FLAGS)) {
127 lv_obj_invalidate(obj);
128 }
129
130 bool trans_removed = remove_trans_styles(obj, selector, LV_STYLE_PROP_ANY, NULL);
131
132 /*Try removing the style first to be sure it won't be added twice*/
133 lv_obj_remove_style(obj, style, selector);
134
135 uint32_t i;
136 /*Go after the transition and local styles*/
137 for(i = 0; i < obj->style_cnt; i++) {
138 if(obj->styles[i].is_trans) continue;
139 if(obj->styles[i].is_local) continue;
140 break;
141 }
142
143 /*Now `i` is at the first normal style. Insert the new style before this*/
144
145 /*Allocate space for the new style and shift the rest of the style to the end*/
146 obj->style_cnt++;
147 LV_ASSERT(obj->style_cnt != 0);
148 obj->styles = lv_realloc(obj->styles, obj->style_cnt * sizeof(lv_obj_style_t));
149 LV_ASSERT_MALLOC(obj->styles);
150
151 uint32_t j;
152 for(j = obj->style_cnt - 1; j > i ; j--) {
153 obj->styles[j] = obj->styles[j - 1];
154 }
155
156 lv_memzero(&obj->styles[i], sizeof(lv_obj_style_t));
157 obj->styles[i].style = style;
158 obj->styles[i].selector = selector;
159
160 if(trans_removed) {
161 full_cache_refresh(obj, LV_PART_ANY);
162 }
163#if LV_OBJ_STYLE_CACHE
164 else {
165 uint32_t * prop_is_set = part == LV_PART_MAIN ? &obj->style_main_prop_is_set : &obj->style_other_prop_is_set;
166 if(lv_style_is_const_internal(style)) {
167 lv_style_const_prop_t * props = style->values_and_props;
168 for(i = 0; props[i].prop != LV_STYLE_PROP_INV; i++) {
169 (*prop_is_set) |= STYLE_PROP_SHIFTED(props[i].prop);
170 }
171 }
172 else {
173 lv_style_prop_t * props = (lv_style_prop_t *)style->values_and_props + style->prop_cnt * sizeof(lv_style_value_t);
174 for(i = 0; i < style->prop_cnt; i++) {
175 (*prop_is_set) |= STYLE_PROP_SHIFTED(props[i]);
176 }
177 }
178 }
179#endif
180
181 lv_obj_refresh_style(obj, selector, LV_STYLE_PROP_ANY);
182}

184bool lv_obj_replace_style(lv_obj_t * obj, const lv_style_t * old_style, const lv_style_t * new_style,
185 lv_style_selector_t selector)
186{
187 LV_CHECK_ARG(obj != NULL, return false);
188 LV_CHECK_ARG(old_style != NULL, return false);
189 LV_CHECK_ARG(new_style != NULL, return false);
190 LV_CHECK_ARG(old_style != new_style, return false);
191
192 lv_state_t state = lv_obj_style_get_selector_state(selector);
193 lv_part_t part = lv_obj_style_get_selector_part(selector);
194
195 /*If the draw size changes invalidate the old area first*/
196 if((part == LV_PART_MAIN || part == LV_PART_ANY) &&
197 (style_has_flag(old_style, DRAW_AREA_FLAGS) || style_has_flag(new_style, DRAW_AREA_FLAGS))) {
198 lv_obj_invalidate(obj);
199 }
200
201 /*Similar to lv_obj_add_style, delete transition*/
202 bool trans_removed = remove_trans_styles(obj, selector, LV_STYLE_PROP_ANY, NULL);
203
204 bool replaced = false;
205 uint32_t i;
206 for(i = 0; i < obj->style_cnt; i++) {
207 lv_state_t state_act = lv_obj_style_get_selector_state(obj->styles[i].selector);
208 lv_part_t part_act = lv_obj_style_get_selector_part(obj->styles[i].selector);
209
210 /*Skip local styles and transitions*/
211 if(obj->styles[i].is_local || obj->styles[i].is_trans) {
212 continue;
213 }
214
215 /*Skip non-matching styles*/
216 if((state != LV_STATE_ANY && state_act != state) ||
217 (part != LV_PART_ANY && part_act != part) ||
218 (old_style != obj->styles[i].style)) {
219 continue;
220 }
221
222 lv_memzero(&obj->styles[i], sizeof(lv_obj_style_t));
223 obj->styles[i].style = new_style;
224 obj->styles[i].selector = selector;
225
226 replaced = true;
227 /*Don't break and continue replacing other occurrences*/
228 }
229 if(replaced || trans_removed) {
230 full_cache_refresh(obj, part);
231 lv_obj_refresh_style(obj, part, LV_STYLE_PROP_ANY);
232 }
233 return replaced;
234}

271void lv_obj_refresh_style(lv_obj_t * obj, lv_part_t part, lv_style_prop_t prop)
272{
273 LV_CHECK_OBJ(obj, MY_CLASS, return);
274
275 if(!style_refr) return;
276
277 LV_PROFILER_STYLE_BEGIN;
278
279 lv_obj_invalidate(obj);
280
281 bool is_layout_refr = lv_style_prop_has_flag(prop, LV_STYLE_PROP_FLAG_LAYOUT_UPDATE);
282 bool is_ext_draw = lv_style_prop_has_flag(prop, LV_STYLE_PROP_FLAG_EXT_DRAW_UPDATE);
283 bool is_inheritable = lv_style_prop_has_flag(prop, LV_STYLE_PROP_FLAG_INHERITABLE);
284 bool is_layer_refr = lv_style_prop_has_flag(prop, LV_STYLE_PROP_FLAG_LAYER_UPDATE);
285
286 if(is_layout_refr) {
287 if(part == LV_PART_ANY ||
288 part == LV_PART_MAIN ||
289 lv_obj_get_style_height_internal(obj, LV_PART_MAIN) == LV_SIZE_CONTENT ||
290 lv_obj_get_style_width_internal(obj, LV_PART_MAIN) == LV_SIZE_CONTENT) {
291 lv_obj_send_event(obj, LV_EVENT_STYLE_CHANGED, NULL);
292 lv_obj_mark_layout_as_dirty(obj);
293 }
294 }
295 if((part == LV_PART_ANY || part == LV_PART_MAIN) && (prop == LV_STYLE_PROP_ANY || is_layout_refr)) {
296 lv_obj_t * parent = lv_obj_get_parent(obj);
297 if(parent) lv_obj_mark_layout_as_dirty(parent);
298 }
299
300 /*Cache the layer type*/
301 if((part == LV_PART_ANY || part == LV_PART_MAIN) && is_layer_refr) {
302 lv_obj_update_layer_type(obj);
303 }
304
305 /*Cache whether the widget has blur or drop shadow*/
306 if(prop == LV_STYLE_PROP_ANY || prop == LV_STYLE_BLUR_RADIUS || prop == LV_STYLE_DROP_SHADOW_OPA) {
307 lv_obj_update_blur_status(obj);
308 }
309
310 if(prop == LV_STYLE_PROP_ANY || is_ext_draw) {
311 lv_obj_refresh_ext_draw_size(obj);
312 }
313 lv_obj_invalidate(obj);
314
315 if(prop == LV_STYLE_PROP_ANY || (is_inheritable && (is_ext_draw || is_layout_refr))) {
316 if(part != LV_PART_SCROLLBAR) {
317 refresh_children_style(obj);
318 }
319 }
320
321 LV_PROFILER_STYLE_END;
322}

513void lv_obj_style_create_transition(lv_obj_t * obj, lv_part_t part, lv_state_t prev_state, lv_state_t new_state,
514 const lv_obj_style_transition_dsc_t * tr_dsc)
515{
516 LV_CHECK_ARG(obj != NULL, return);
517 LV_CHECK_ARG(tr_dsc != NULL, return);
518
519 trans_t * tr;
520
521 /*Get the previous and current values*/
522 obj->skip_trans = 1;
523 obj->state = prev_state;
524 lv_style_value_t v1 = lv_obj_get_style_prop(obj, part, tr_dsc->prop);
525 obj->state = new_state;
526 lv_style_value_t v2 = lv_obj_get_style_prop(obj, part, tr_dsc->prop);
527 obj->skip_trans = 0;
528
529 if(v1.ptr == v2.ptr && v1.num == v2.num && lv_color_eq(v1.color, v2.color)) return;
530 obj->state = prev_state;
531 v1 = lv_obj_get_style_prop(obj, part, tr_dsc->prop);
532 obj->state = new_state;
533
534 lv_obj_style_t * style_trans = get_trans_style(obj, part);
535 lv_style_set_prop((lv_style_t *)style_trans->style, tr_dsc->prop, v1); /*Be sure `trans_style` has a valid value*/
536 lv_obj_refresh_style(obj, tr_dsc->selector, tr_dsc->prop);
537
538 if(tr_dsc->prop == LV_STYLE_RADIUS) {
539 if(v1.num == LV_RADIUS_CIRCLE || v2.num == LV_RADIUS_CIRCLE) {
540 int32_t whalf = lv_obj_get_width(obj) / 2;
541 int32_t hhalf = lv_obj_get_height(obj) / 2;
542 if(v1.num == LV_RADIUS_CIRCLE) v1.num = LV_MIN(whalf + 1, hhalf + 1);
543 if(v2.num == LV_RADIUS_CIRCLE) v2.num = LV_MIN(whalf + 1, hhalf + 1);
544 }
545 }
546
547 tr = lv_ll_ins_head(style_trans_ll_p);
548 LV_ASSERT_MALLOC(tr);
549 if(tr == NULL) return;
550 tr->start_value = v1;
551 tr->end_value = v2;
552 tr->obj = obj;
553 tr->prop = tr_dsc->prop;
554 tr->selector = part;
555
556 lv_anim_t a;
557 lv_anim_init(&a);
558 lv_anim_set_var(&a, tr);
559 lv_anim_set_exec_cb(&a, trans_anim_cb);
560 lv_anim_set_start_cb(&a, trans_anim_start_cb);
561 lv_anim_set_completed_cb(&a, trans_anim_completed_cb);
562 lv_anim_set_values(&a, 0x00, 0xFF);
563 lv_anim_set_duration(&a, tr_dsc->time);
564 lv_anim_set_delay(&a, tr_dsc->delay);
565 lv_anim_set_path_cb(&a, tr_dsc->path_cb);
566 lv_anim_set_early_apply(&a, false);
567 lv_anim_set_user_data(&a, tr_dsc->user_data);
568 lv_anim_start(&a);

577lv_style_state_cmp_t lv_obj_style_state_compare(lv_obj_t * obj, lv_state_t state1, lv_state_t state2,
578 lv_part_t * changed_part)
579{
580 LV_CHECK_ARG(obj != NULL, return LV_STYLE_STATE_CMP_SAME);
581
582 lv_style_state_cmp_t res = LV_STYLE_STATE_CMP_SAME;
583
584 /*Accumulate which part the differing styles belong to: the lone differing
585 *part, or LV_PART_ANY once styles from more than one part differ. Reported
586 *via changed_part so the caller can scope the refresh to a single part.
587 *diff_part_found is needed as a separate flag because a style's part can
588 *itself be LV_PART_ANY, so LV_PART_ANY can't double as "none seen yet".*/
589 lv_part_t diff_part = LV_PART_ANY;
590 bool diff_part_found = false;
591
592 /*Are there any new styles for the new state?*/
593 uint32_t i;
594 for(i = 0; i < obj->style_cnt; i++) {
595 if(obj->styles[i].is_trans) continue;
596
597 lv_state_t state_act = lv_obj_style_get_selector_state(obj->styles[i].selector);
598 /*The style is valid for a state but not the other*/
599 bool valid1 = state_act & (~state1) ? false : true;
600 bool valid2 = state_act & (~state2) ? false : true;
601 if(valid1 != valid2) {
602 /*Record this differing style's part: keep it if it's the only one
603 *seen so far, otherwise collapse to LV_PART_ANY (more than one part).*/
604 lv_part_t part_act = lv_obj_style_get_selector_part(obj->styles[i].selector);
605 if(!diff_part_found) {
606 diff_part = part_act;
607 diff_part_found = true;
608 }
609 else if(part_act != diff_part) {
610 diff_part = LV_PART_ANY;
611 }
612
613 const lv_style_t * style = obj->styles[i].style;
614 lv_style_value_t v;
615 /*If there is layout difference on the main part, return immediately. There is no more serious difference*/
616 bool layout_diff = false;
617 if(lv_style_get_prop_internal(style, LV_STYLE_PAD_TOP, &v))layout_diff = true;
618 else if(lv_style_get_prop_internal(style, LV_STYLE_PAD_BOTTOM, &v)) layout_diff = true;
619 else if(lv_style_get_prop_internal(style, LV_STYLE_PAD_LEFT, &v)) layout_diff = true;
620 else if(lv_style_get_prop_internal(style, LV_STYLE_PAD_RIGHT, &v)) layout_diff = true;
621 else if(lv_style_get_prop_internal(style, LV_STYLE_PAD_COLUMN, &v)) layout_diff = true;
622 else if(lv_style_get_prop_internal(style, LV_STYLE_PAD_ROW, &v)) layout_diff = true;
623 else if(lv_style_get_prop_internal(style, LV_STYLE_LAYOUT, &v)) layout_diff = true;
624 else if(lv_style_get_prop_internal(style, LV_STYLE_TRANSLATE_X, &v)) layout_diff = true;
625 else if(lv_style_get_prop_internal(style, LV_STYLE_TRANSLATE_Y, &v)) layout_diff = true;
626 else if(lv_style_get_prop_internal(style, LV_STYLE_WIDTH, &v)) layout_diff = true;
627 else if(lv_style_get_prop_internal(style, LV_STYLE_HEIGHT, &v)) layout_diff = true;
628 else if(lv_style_get_prop_internal(style, LV_STYLE_MIN_WIDTH, &v)) layout_diff = true;
629 else if(lv_style_get_prop_internal(style, LV_STYLE_MAX_WIDTH, &v)) layout_diff = true;
630 else if(lv_style_get_prop_internal(style, LV_STYLE_MIN_HEIGHT, &v)) layout_diff = true;
631 else if(lv_style_get_prop_internal(style, LV_STYLE_MAX_HEIGHT, &v)) layout_diff = true;
632 else if(lv_style_get_prop_internal(style, LV_STYLE_BORDER_WIDTH, &v)) layout_diff = true;
633
634 if(layout_diff) {
635 /*Returning early, so the full set of changed parts is unknown:
636 *report LV_PART_ANY rather than a possibly-incomplete part.*/
637 if(changed_part) *changed_part = LV_PART_ANY;
638 return LV_STYLE_STATE_CMP_DIFF_LAYOUT;
639 }
640
641 /*Check for draw pad changes*/
642 if(lv_style_get_prop_internal(style, LV_STYLE_TRANSFORM_WIDTH, &v)) res = LV_STYLE_STATE_CMP_DIFF_DRAW_PAD;
643 else if(lv_style_get_prop_internal(style, LV_STYLE_TRANSFORM_HEIGHT, &v)) res = LV_STYLE_STATE_CMP_DIFF_DRAW_PAD;
644 else if(lv_style_get_prop_internal(style, LV_STYLE_TRANSFORM_ROTATION, &v)) res = LV_STYLE_STATE_CMP_DIFF_DRAW_PAD;
645 else if(lv_style_get_prop_internal(style, LV_STYLE_TRANSFORM_SCALE_X, &v)) res = LV_STYLE_STATE_CMP_DIFF_DRAW_PAD;
646 else if(lv_style_get_prop_internal(style, LV_STYLE_TRANSFORM_SCALE_Y, &v)) res = LV_STYLE_STATE_CMP_DIFF_DRAW_PAD;
647 else if(lv_style_get_prop_internal(style, LV_STYLE_OUTLINE_OPA, &v)) res = LV_STYLE_STATE_CMP_DIFF_DRAW_PAD;
648 else if(lv_style_get_prop_internal(style, LV_STYLE_OUTLINE_PAD, &v)) res = LV_STYLE_STATE_CMP_DIFF_DRAW_PAD;
649 else if(lv_style_get_prop_internal(style, LV_STYLE_OUTLINE_WIDTH, &v)) res = LV_STYLE_STATE_CMP_DIFF_DRAW_PAD;
650 else if(lv_style_get_prop_internal(style, LV_STYLE_SHADOW_WIDTH, &v)) res = LV_STYLE_STATE_CMP_DIFF_DRAW_PAD;
651 else if(lv_style_get_prop_internal(style, LV_STYLE_SHADOW_OPA, &v)) res = LV_STYLE_STATE_CMP_DIFF_DRAW_PAD;
652 else if(lv_style_get_prop_internal(style, LV_STYLE_SHADOW_OFFSET_X, &v)) res = LV_STYLE_STATE_CMP_DIFF_DRAW_PAD;
653 else if(lv_style_get_prop_internal(style, LV_STYLE_SHADOW_OFFSET_Y, &v)) res = LV_STYLE_STATE_CMP_DIFF_DRAW_PAD;
654 else if(lv_style_get_prop_internal(style, LV_STYLE_SHADOW_SPREAD, &v)) res = LV_STYLE_STATE_CMP_DIFF_DRAW_PAD;
655 else if(lv_style_get_prop_internal(style, LV_STYLE_LINE_WIDTH, &v)) res = LV_STYLE_STATE_CMP_DIFF_DRAW_PAD;
656 else if(res == LV_STYLE_STATE_CMP_SAME) res = LV_STYLE_STATE_CMP_DIFF_REDRAW;
657 }
658 }
659
660 if(changed_part) *changed_part = diff_part;
661 return res;
662}

701{
702 LV_CHECK_ARG(obj != NULL, return LV_OPA_TRANSP);
703
704 LV_PROFILER_STYLE_BEGIN;
705 lv_opa_t opa_obj = lv_obj_get_style_opa_internal(obj, part);

707 LV_PROFILER_STYLE_END;

716 if(part != LV_PART_MAIN) {
717 part = LV_PART_MAIN;
718 }
719 else {
720 obj = lv_obj_get_parent(obj);
721 }
722
723 while(obj) {
724 opa_obj = lv_obj_get_style_opa_internal(obj, part);
725 if(opa_obj <= LV_OPA_MIN) {
726 LV_PROFILER_STYLE_END;
727 return LV_OPA_TRANSP;
728 }
729 if(opa_obj < LV_OPA_MAX) {
730 opa_final = LV_OPA_MIX2(opa_final, opa_obj);
731 }
732
733 obj = lv_obj_get_parent(obj);
734 }
735
736 if(opa_final <= LV_OPA_MIN) {
737 LV_PROFILER_STYLE_END;
738 return LV_OPA_TRANSP;
739 }
740
741 if(opa_final >= LV_OPA_MAX) {
742 LV_PROFILER_STYLE_END;
743 return LV_OPA_COVER;
744 }
745
746 LV_PROFILER_STYLE_END;
747 return opa_final;
748}

1005static lv_style_t * get_local_style(lv_obj_t * obj, lv_style_selector_t selector)
1006{
1007 uint32_t i;
1008 for(i = 0; i < obj->style_cnt; i++) {
1009 if(obj->styles[i].is_local &&
1010 obj->styles[i].selector == selector) {
1011 return (lv_style_t *)obj->styles[i].style;
1012 }
1013 }
1014
1015 /*Allocate the individual style first so obj->styles stays valid on failure*/
1016 lv_style_t * new_style = lv_malloc_zeroed(sizeof(lv_style_t));
1017 if(new_style == NULL) {
1018 LV_LOG_WARN("couldn't allocate local style");
1019 return NULL;
1020 }
1021
1022 obj->style_cnt++;
1023 LV_ASSERT(obj->style_cnt != 0);
1024 lv_obj_style_t * new_styles = lv_realloc(obj->styles, obj->style_cnt * sizeof(lv_obj_style_t));
1025 if(new_styles == NULL) {
1026 obj->style_cnt--;
1027 lv_free(new_style);
1028 LV_LOG_WARN("couldn't allocate styles");
1029 return NULL;
1030 }
1031 obj->styles = new_styles;
1032
1033 for(i = obj->style_cnt - 1; i > 0 ; i--) {
1034 /*Copy only normal styles (not local and transition).
1035 *The new local style will be added as the last local style*/
1036 if(obj->styles[i - 1].is_local || obj->styles[i - 1].is_trans) break;
1037 obj->styles[i] = obj->styles[i - 1];
1038 }
1039
1040 lv_memzero(&obj->styles[i], sizeof(lv_obj_style_t));
1041 obj->styles[i].style = new_style;
1042 lv_style_init((lv_style_t *)obj->styles[i].style);
1043
1044 obj->styles[i].is_local = 1;
1045 obj->styles[i].selector = selector;
1046 return (lv_style_t *)obj->styles[i].style;
1047}

1056static lv_obj_style_t * get_trans_style(lv_obj_t * obj, lv_style_selector_t selector)
1057{
1058 uint32_t i;
1059 for(i = 0; i < obj->style_cnt; i++) {
1060 if(obj->styles[i].is_trans && obj->styles[i].selector == selector) break;
1061 }
1062
1063 /*Already have a transition style for it*/
1064 if(i != obj->style_cnt) return &obj->styles[i];
1065

1067 lv_style_t * new_style = lv_malloc(sizeof(lv_style_t));

1098 lv_style_value_t * v)
1099{
1100
1101 const uint32_t group = (uint32_t)1 << lv_style_get_prop_group(prop);
1102 const lv_part_t part = lv_obj_style_get_selector_part(selector);
1103 const bool skip_trans = (const bool) obj->skip_trans;
1104 int32_t weight = -1;
1105 lv_style_res_t found;
1106 uint32_t i;
1107 /*Handle the transition styles first. They are at the beginning of the style list
1108 *as the have the highest precedence. It's a different for loop as in case of transitions
1109 *the state doesn't need to be checked. */
1110 for(i = 0; i < obj->style_cnt; i++) {
1111 lv_obj_style_t * obj_style = &obj->styles[i];
1112 if(obj_style->is_trans == false) break;
1113 if(skip_trans) continue;
1114
1115 lv_part_t part_act = lv_obj_style_get_selector_part(obj->styles[i].selector);
1116
1117 if(part_act != part) continue;
1118 if((obj_style->style->has_group & group) == 0) continue;
1119 found = lv_style_get_prop_internal(obj_style->style, prop, v);
1120 if(found == LV_STYLE_RES_FOUND) {
1121 return LV_STYLE_RES_FOUND;
1122 }
1123 }
1124
1125 /*Check the normal local and global styles that are assigned to given state*/
1126 const lv_state_t state = lv_obj_style_get_selector_state(selector);
1127 const lv_state_t state_inv = ~state;
1128 for(; i < obj->style_cnt; i++) {
1129 if((obj->styles[i].style->has_group & group) == 0) continue;
1130 if(obj->styles[i].is_disabled) continue;
1131 lv_obj_style_t * obj_style = &obj->styles[i];
1132 lv_part_t part_act = lv_obj_style_get_selector_part(obj->styles[i].selector);
1133 if(part_act != part) continue;
1134
1135 /*Be sure the style not specifies other state than the requested.
1136 *E.g. For HOVER+PRESS object state, HOVER style only is OK, but HOVER+FOCUS style is not*/
1137 lv_state_t state_act = lv_obj_style_get_selector_state(obj->styles[i].selector);
1138 if((state_act & state_inv)) continue;
1139
1140 /*Check only better candidates*/
1141 if((int32_t)state_act <= weight) continue;
1142
1143 found = lv_style_get_prop_internal(obj_style->style, prop, v);
1144 if(found == LV_STYLE_RES_FOUND) {
1145 if(state_act == state) {
1146 return LV_STYLE_RES_FOUND;
1147 }
1148 weight = state_act;
1149 }
1150 }
1151
1152 if(weight >= 0) return LV_STYLE_RES_FOUND;
1153 else return LV_STYLE_RES_NOT_FOUND;
1154}

1208static bool remove_trans_styles(lv_obj_t * obj, lv_part_t part, lv_style_prop_t prop, trans_t * tr_limit)
1209{
1210 trans_t * tr;
1211 trans_t * tr_prev;
1212 bool removed = false;
1213 tr = lv_ll_get_tail(style_trans_ll_p);
1214 while(tr != NULL) {
1215 if(tr == tr_limit) break;
1216
1217 /*'tr' might be deleted, so get the next object while 'tr' is valid*/
1218 tr_prev = lv_ll_get_prev(style_trans_ll_p, tr);
1219
1220 if(tr->obj == obj && (part == tr->selector || part == LV_PART_ANY) && (prop == tr->prop || prop == LV_STYLE_PROP_ANY)) {
1221 /*Dropping the transitioned value changes the drawn area right away,
1222 *so invalidate the old area first*/
1223 if(lv_obj_style_get_selector_part(tr->selector) == LV_PART_MAIN &&
1224 lv_style_prop_has_flag(tr->prop, DRAW_AREA_FLAGS)) {
1225 lv_obj_invalidate(obj);
1226 }
1227
1228 /*Remove any transitioned properties from the trans. style
1229 *to allow changing it by normal styles*/
1230 uint32_t i;
1231 for(i = 0; i < obj->style_cnt; i++) {
1232 if(obj->styles[i].is_trans && (part == LV_PART_ANY || obj->styles[i].selector == part)) {
1233 lv_style_remove_prop((lv_style_t *)obj->styles[i].style, tr->prop);
1234 }
1235 }
1236
1237 /*Free the transition descriptor too*/
1238 lv_anim_delete(tr, NULL);
1239 lv_ll_remove(style_trans_ll_p, tr);
1240 lv_free(tr);
1241 removed = true;
1242
1243 }
1244 tr = tr_prev;
1245 }
1246
1247 return removed;

1250static void trans_anim_cb(void * _tr, int32_t v)
1251{
1252 trans_t * tr = _tr;
1253 lv_obj_t * obj = tr->obj;
1254
1255 uint32_t i;
1256 /*Find the transition style of of the descriptor*/
1257 for(i = 0; i < obj->style_cnt; i++) {
1258 if(obj->styles[i].is_trans == 0 || obj->styles[i].selector != tr->selector) continue;
1259
1260 lv_style_value_t value_final = {0};
1261 switch(tr->prop) {
1262
1263 case LV_STYLE_BORDER_SIDE:
1264 case LV_STYLE_BORDER_POST:
1265 case LV_STYLE_BLEND_MODE:
1266 if(v < 255) value_final.num = tr->start_value.num;
1267 else value_final.num = tr->end_value.num;
1268 break;
1269 case LV_STYLE_TRANSITION:
1270 case LV_STYLE_TEXT_FONT:
1271 if(v < 255) value_final.ptr = tr->start_value.ptr;
1272 else value_final.ptr = tr->end_value.ptr;
1273 break;
1274 case LV_STYLE_COLOR_FILTER_DSC:
1275 if(tr->start_value.ptr == NULL) value_final.ptr = tr->end_value.ptr;
1276 else if(tr->end_value.ptr == NULL) value_final.ptr = tr->start_value.ptr;
1277 else if(v < 128) value_final.ptr = tr->start_value.ptr;
1278 else value_final.ptr = tr->end_value.ptr;
1279 break;
1280 case LV_STYLE_RECOLOR:
1281 case LV_STYLE_BG_COLOR:
1282 case LV_STYLE_BG_GRAD_COLOR:
1283 case LV_STYLE_BORDER_COLOR:
1284 case LV_STYLE_TEXT_COLOR:
1285 case LV_STYLE_SHADOW_COLOR:
1286 case LV_STYLE_OUTLINE_COLOR:
1287 case LV_STYLE_IMAGE_RECOLOR:
1288 if(v <= 0) value_final.color = tr->start_value.color;
1289 else if(v >= 255) value_final.color = tr->end_value.color;
1290 else value_final.color = lv_color_mix(tr->end_value.color, tr->start_value.color, v);
1291 break;
1292
1293 default:
1294 if(v == 0) value_final.num = tr->start_value.num;
1295 else if(v == 255) value_final.num = tr->end_value.num;
1296 else value_final.num = tr->start_value.num + ((int32_t)((int32_t)(tr->end_value.num - tr->start_value.num) * v) >> 8);
1297 break;
1298 }
1299
1300 /*Don't do anything is the prop didn't changed*/
1301 lv_style_value_t old_value = {0};
1302 if(lv_style_get_prop_internal(obj->styles[i].style, tr->prop, &old_value) == LV_STYLE_RES_FOUND) {
1303 if(value_final.ptr == old_value.ptr &&
1304 lv_color_eq(value_final.color, old_value.color) &&
1305 value_final.num == old_value.num) {
1306 break;
1307 }
1308 }
1309
1310 /*If the draw size changes invalidate the old area first*/
1311 lv_part_t part = lv_obj_style_get_selector_part(tr->selector);
1312 if(part == LV_PART_MAIN && lv_style_prop_has_flag(tr->prop, DRAW_AREA_FLAGS)) {
1313 lv_obj_invalidate(tr->obj);
1314 }
1315 lv_style_set_prop((lv_style_t *)obj->styles[i].style, tr->prop, value_final);
1316
1317#if LV_OBJ_STYLE_CACHE
1318 uint32_t prop_shifted = STYLE_PROP_SHIFTED(tr->prop);
1319 if(part == LV_PART_MAIN) {
1320 obj->style_main_prop_is_set |= prop_shifted;
1321 }
1322 else {
1323 obj->style_other_prop_is_set |= prop_shifted;
1324 }
1325#endif
1326
1327 lv_obj_refresh_style(tr->obj, tr->selector, tr->prop);
1328 break;
1329 }

1427/*True if any style attached to the widget enables blur or a drop shadow in the
1428 *widget's current state. The per-style has_group bitmask makes the common

1431{

1433 const uint32_t group_dropshadow = (uint32_t)1 << lv_style_get_prop_group(LV_STYLE_DROP_SHADOW_OPA);

1446 lv_style_get_prop(obj_style->style, LV_STYLE_BLUR_RADIUS, &v)) {

1450 lv_style_get_prop(obj_style->style, LV_STYLE_DROP_SHADOW_OPA, &v)) {

1453 }

1544static lv_style_res_t get_selector_style_prop(const lv_obj_t * obj, lv_style_selector_t selector, lv_style_prop_t prop,
1545 lv_style_value_t * value_act)
1546{
1547 lv_style_res_t found;
1548 lv_part_t part = lv_obj_style_get_selector_part(selector);
1549
1550 /*The happy path*/
1551#if LV_OBJ_STYLE_CACHE
1552 const uint32_t prop_shifted = STYLE_PROP_SHIFTED(prop);
1553 if((part == LV_PART_MAIN ? obj->style_main_prop_is_set : obj->style_other_prop_is_set) & prop_shifted)
1554#endif
1555 {
1556 found = get_prop_core(obj, selector, prop, value_act);
1557 if(found == LV_STYLE_RES_FOUND) return LV_STYLE_RES_FOUND;
1558 }
1559
1560 extern const uint8_t lv_style_builtin_prop_flag_lookup_table[];
1561 bool inheritable = false;
1562 if(prop < LV_STYLE_NUM_BUILT_IN_PROPS) {
1563 inheritable = lv_style_builtin_prop_flag_lookup_table[prop] & LV_STYLE_PROP_FLAG_INHERITABLE;
1564 }
1565 else {
1566 if(_style_custom_prop_flag_lookup_table != NULL) {
1567 inheritable = _style_custom_prop_flag_lookup_table[prop - LV_STYLE_NUM_BUILT_IN_PROPS] &
1568 LV_STYLE_PROP_FLAG_INHERITABLE;
1569 }
1570 }
1571
1572 if(inheritable) {
1573 /*If not found, check the `MAIN` style first, if already on the MAIN part go to the parent*/
1574 if(part != LV_PART_MAIN) part = LV_PART_MAIN;
1575 else obj = obj->parent;
1576
1577 while(obj) {
1578#if LV_OBJ_STYLE_CACHE
1579 if(obj->style_main_prop_is_set & prop_shifted)
1580#endif
1581 {
1582 selector = part | obj->state;
1583 found = get_prop_core(obj, selector, prop, value_act);
1584 if(found == LV_STYLE_RES_FOUND) return LV_STYLE_RES_FOUND;
1585 }
1586 /*Check the parent too.*/
1587 obj = obj->parent;
1588 }
1589 }
1590 else {
1591 /*Get the width and height from the class.
1592 * WIDTH and HEIGHT are not inherited so add them in the `else` to skip checking them for inherited properties */
1593 if(part == LV_PART_MAIN && (prop == LV_STYLE_WIDTH || prop == LV_STYLE_HEIGHT)) {
1594 const lv_obj_class_t * cls = obj->class_p;
1595 while(cls) {
1596 if(prop == LV_STYLE_WIDTH) {
1597 if(cls->width_def != 0) {
1598 value_act->num = cls->width_def;
1599 return LV_STYLE_RES_FOUND;
1600 }
1601 }
1602 else {
1603 if(cls->height_def != 0) {
1604 value_act->num = cls->height_def;
1605 return LV_STYLE_RES_FOUND;
1606 }
1607 }
1608 cls = cls->base_class;
1609 }
1610 }
1611 }
1612
1613 return LV_STYLE_RES_NOT_FOUND;
1614}

1616static void remove_style_core(lv_obj_t * obj, const lv_style_t * style, lv_style_selector_t selector, bool theme_only)
1617{
1618 LV_ASSERT(obj != NULL);
1619 lv_state_t state = lv_obj_style_get_selector_state(selector);
1620 lv_part_t part = lv_obj_style_get_selector_part(selector);
1621
1622 /*If the draw size changes invalidate the old area first.
1623 *`style == NULL` means "any style" (e.g. `lv_obj_remove_style_all()`) so it can't be checked for flags.*/
1624 if((part == LV_PART_MAIN || part == LV_PART_ANY) &&
1625 (style == NULL || style_has_flag(style, DRAW_AREA_FLAGS))) {
1626 lv_obj_invalidate(obj);
1627 }
1628
1629 uint32_t style_count = 0;
1630 bool deleted = false;
1631
1632 for(uint32_t i = 0; i < obj->style_cnt; i++) {
1633 lv_state_t state_act = lv_obj_style_get_selector_state(obj->styles[i].selector);
1634 lv_part_t part_act = lv_obj_style_get_selector_part(obj->styles[i].selector);
1635
1636 bool should_remove = true;
1637 if(theme_only && !obj->styles[i].is_theme) {
1638 should_remove = false;
1639 }
1640 else if((state != LV_STATE_ANY && state_act != state) ||
1641 (part != LV_PART_ANY && part_act != part) ||
