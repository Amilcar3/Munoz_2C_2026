174static void BleReadCb(uint8_t *data, uint8_t length){
175 if(length == 0) return;
176 switch(data[0]){
177 case 'N':
178 case '0':
179 case '1':
180 case '2': {
181 display_msg_t msg = { .type = MSG_KEY, .key = data[0] };

183 break;
184 }
185 default:

247static void readTask(void *pvParameter){
248 sample_t s;
249 TickType_t last = xTaskGetTickCount();
250
251 MAX3010X_begin();
252 MAX3010X_setup(LED_POWER, 2, 2, 100, 411, 4096);
253
254 while(true){
255 vTaskDelayUntil(&last, pdMS_TO_TICKS(READ_POLL_MS));
256
257 MAX3010X_check(); /* trae lo nuevo de la FIFO */
258 while(MAX3010X_available()){ /* y se vacía por completo */
259 s.red = MAX3010X_getFIFORed();

263 samples_lost++; /* cola llena: contamos, no bloqueamos */

280static void ProcessTask(void *pvParameter){
281 sample_t s;
282 display_msg_t msg;
283 uint16_t new_count = 0;
284 uint16_t i;
285 static float ppg_span = PPG_INITIAL_SPAN;
286 static float win_min = 1e6f, win_max = -1e6f;
287 static uint8_t ppg_decim = 0; /* decimador: 1 de cada PPG_DECIM muestras al display */
288
289 HiPassInit(SAMPLE_FREQ, 0.5, ORDER_4);
290 LowPassInit(SAMPLE_FREQ, 4.0, ORDER_4);
291
292 while(true){
293 xQueueReceive(xQueueSamples, &s, portMAX_DELAY);
294
295 redBuffer[bufferLength - SAMPLE_FREQ + new_count] = s.red;
296 irBuffer[bufferLength - SAMPLE_FREQ + new_count] = s.ir;
297 new_count++;
298
299 /* Detector de dedo y filtrado de la PPG */
300 bool finger = (s.ir >= FINGER_THRESHOLD);
301 if(finger){
302 dato = (float)s.red;
303 HiPassFilter(&dato, &dato_hp, 1);
304 LowPassFilter(&dato_hp, &dato_lp, 1);
305 dato_lp = -dato_lp; /* Inversión: mayor absorción sístólica -> pico positivo */
306 if(dato_lp < win_min) win_min = dato_lp;
307 if(dato_lp > win_max) win_max = dato_lp;
308 } else {
309 dato_lp = 0.0f;
310 }
311
312 if(finger && checkForBeat((int32_t)s.red)){
313 display_msg_t beat_msg = { .type = MSG_BEAT };
314 xQueueSend(xQueueDisplay, &beat_msg, 0); /*Se detectan y envian latidos para la funcion anillo*/
315 }
316
317 float v;
318 if(finger && ppg_span > 10.0f){
319 float scale = PPG_TARGET_HEIGHT / ppg_span;
320 v = PPG_CENTER + dato_lp * scale;
321 } else {
322 v = PPG_CENTER;
323 }
324 if(v < 50.0f) v = 50.0f;
325 if(v > 950.0f) v = 950.0f;
326 g_ppg_ble = (int16_t)v; /* actualizacion continua para BLE (~50 Hz) */
327 /* Decimacion: solo 1 de cada PPG_DECIM muestras va al chart (~16 Hz) */
328 if(++ppg_decim >= PPG_DECIM){
329 ppg_decim = 0;
330 msg.type = MSG_PPG_SAMPLE;
331 msg.ppg = (int16_t)v;
332 xQueueSend(xQueueDisplay, &msg, 0);
333 }
334
335 /* Cada 50 muestras nuevas se recalcula HR y SpO2 */
336 if(new_count >= SAMPLE_FREQ){
337 maxim_heart_rate_and_oxygen_saturation(irBuffer, bufferLength, redBuffer,
338 &spo2, &validSPO2, &heartRate, &validHeartRate);
339
340 if(validHeartRate && (heartRate < 40 || heartRate > 200)){ validHeartRate = 0; }
341 if(validSPO2 && (spo2 < 70 || spo2 > 100)){ validSPO2 = 0; }
342
343 if(validHeartRate){
344 hr_buffer[hr_idx % HR_AVG_SIZE] = heartRate;
345 hr_idx++;
346 int32_t sum = 0;
347 uint8_t count = (hr_idx < HR_AVG_SIZE) ? hr_idx : HR_AVG_SIZE;
348 for(uint8_t j = 0; j < count; j++) sum += hr_buffer[j];
349 hr_display = sum / count;
350 }
351 if(!finger){ hr_display = 0; } /* sin dedo, no mostrar un valor viejo */
352
353 msg.type = MSG_VITALS;
354 msg.vitals.hr = (uint8_t)hr_display;
355 msg.vitals.spo2 = (validSPO2 && finger) ? (uint8_t)spo2 : 0;
356 xQueueSend(xQueueDisplay, &msg, pdMS_TO_TICKS(10));
357
358 /* Actualiza variables compartidas para que BleTask las envíe */
359 g_hr_ble = msg.vitals.hr;
360 g_spo2_ble = msg.vitals.spo2;
361
362 /* Se corre la ventana: las últimas 150 pasan al inicio */
363 for(i = SAMPLE_FREQ; i < bufferLength; i++){
364 redBuffer[i - SAMPLE_FREQ] = redBuffer[i];
365 irBuffer[i - SAMPLE_FREQ] = irBuffer[i];
366 }
367 /* Actualización de la escala dinámica entre ventanas */
368 if(finger && (win_max > win_min)){
369 float current_span = win_max - win_min;
370 if(current_span > 20.0f){
371 /* Suavizado entre ventana y ventana */
372 ppg_span = 0.7f * ppg_span + 0.3f * current_span;
373 }
374 } else if(!finger){
375 ppg_span = PPG_INITIAL_SPAN;
376 }
377 win_min = 1e6f;
378 win_max = -1e6f;
379
380 new_count = 0;
381 }
382 }
383}

393static void DisplayTask(void *pvParameter){
394 uint32_t wait_ms;
395 display_msg_t msg;
396
397 DisplayInit();
398 all_in_one_create(NULL);
399 all_in_one_set_beat_auto(false); /* el anillo lo dispara checkForBeat() */
400
401 while(true){
402 while(xQueueReceive(xQueueDisplay, &msg, 0) == pdTRUE){ /* no bloqueante */
403 switch(msg.type){
404 case MSG_PPG_SAMPLE:
405 all_in_one_ppg_push(msg.ppg);
406 break;
407 case MSG_VITALS:
408 all_in_one_set_hr(msg.vitals.hr);
409 all_in_one_set_spo2(msg.vitals.spo2);
410 break;
411 case MSG_KEY:
412 /* Cambio de vista desde tecla fisica (SWITCH_1) o BLE */
413 if(msg.key == 'N'){
414 all_in_one_next_view();
415 } else if(msg.key >= '0' && msg.key <= '2'){
416 all_in_one_set_view((all_in_one_view_t)(msg.key - '0'));
417 }
418 break;
419 case MSG_BEAT:
420 all_in_one_beat();
421 break;
422 }
423 }
424 wait_ms = lv_timer_handler();
425 if (wait_ms < 10) wait_ms = 10;
426 if (wait_ms > 20) wait_ms = 20;
427 vTaskDelay(pdMS_TO_TICKS(wait_ms));
428 }
429}

449static void BleTask(void *pvParameter){

451 static const ble_config_t ble_cfg = {
452 .device_name = "ESP_EDU_1",
453 .func_p = BleReadCb,
454 };
455 BleInit((ble_config_t *)&ble_cfg);
456
457 while(true){
458 if(BleStatus() == BLE_CONNECTED){
459 snprintf(msg, sizeof(msg), "*H%u**S%u**G%d*", g_hr_ble, g_spo2_ble, g_ppg_ble);
460 BleSendString(msg);
461 }
462 vTaskDelay(pdMS_TO_TICKS(50)); /* envio cada 50 ms (~20 Hz) para graficar onda en vivo */
463 }
464}
