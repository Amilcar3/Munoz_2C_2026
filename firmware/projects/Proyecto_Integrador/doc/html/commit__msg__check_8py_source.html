3"""
4LVGL Commit Message Style Checker
5
6Format: type(scope): description
7
8Valid types: feat, fix, arch, test, perf, example, refactor, revert, docs, style, chore, ci, build
9Scope: required (except chore/docs/ci), letters/digits/_/-/ allowed, e.g. (draw), (obj)

11
12Usage:
13 commit_msg_check.py # Check commits between base branch and HEAD
14 commit_msg_check.py --base <branch> # Specify base branch
15 commit_msg_check.py --check-title "msg" # Check a single PR title / commit message
16 commit_msg_check.py --self-test # Run self-tests
17
18See: https://docs.lvgl.io/master/contributing/pull_requests.html#commit-message-format
19"""

43# Common typos/aliases -> correct type
44TYPE_TYPOS = {
45 "feature": "feat",
46 "fea": "feat",
47 "refact": "refactor",
48 "doc": "docs",
49 "tests": "test",
50 "bugfix": "fix",
51 "hotfix": "fix",
52 "perf_opt": "perf",

56VALID_TYPES_RE = "|".join(VALID_TYPES)

58# type(scope): description (chore/docs/ci allow omitting scope)
59FULL_PATTERN = re.compile(rf"^({VALID_TYPES_RE})\‍(([a-zA-Z0-9_/-]+)\‍)(!?): (.+)$")

61# Types that allow omitting scope
62SCOPE_OPTIONAL_TYPES = {"chore", "docs", "ci"}

64# type: description (no scope, for scope-optional types)
65NO_SCOPE_PATTERN = re.compile(rf"^({'|'.join(SCOPE_OPTIONAL_TYPES)})(!?): (.+)$")

67# type( or type:
68TYPE_ONLY_PATTERN = re.compile(r"^([a-zA-Z_]+)")

83def check_commit_msg(msg):
84 """Check a single commit message subject line.
85
86 Returns a list of error strings. Empty list means OK.
87 """
88 errors = []
89
90 # Strip leading/trailing whitespace (common in git log output)
91 msg = msg.strip()
92
93 if not msg:
94 errors.append("Commit message is empty")
95 return errors
96
97 # Allow Revert commits
98 if msg.startswith('Revert "'):
99 return errors
100
101 # Allow merge commits
102 if msg.startswith("Merge "):
103 return errors
104
105 # Check for Chinese punctuation (common mistake)
106 cn_punctuation = {
107 "\uff08": "(", # （ -> (
108 "\uff09": ")", # ） -> )
109 "\uff1a": ":", # ： -> :
110 "\u3002": ".", # 。 -> .
111 "\uff0c": ",", # ， -> ,
112 "\uff1b": ";", # ； -> ;
113 }
114 found_cn = [
115 f"'{ch}'(U+{ord(ch):04X}) -> '{en}'"
116 for ch, en in cn_punctuation.items()
117 if ch in msg
118 ]
119 if found_cn:
120 errors.append(
121 f"Chinese punctuation detected: {', '.join(found_cn)}. "
122 "Please use English punctuation only"
123 )
124 return errors
125
126 # Extract type word
127 m = TYPE_ONLY_PATTERN.match(msg)
128 type_word = m.group(1) if m else ""
129 type_lower = type_word.lower()
130
131 # Check common typos first
132 if type_lower in TYPE_TYPOS:
133 correct = TYPE_TYPOS[type_lower]
134 errors.append(f"Type '{type_word}' is not standard, did you mean '{correct}'?")
135 return errors
136
137 # Check type is valid and followed by '('
138 type_with_paren = re.compile(rf"^({VALID_TYPES_RE})\‍(")
139 if not type_with_paren.match(msg):
140 # type: desc (missing scope)
141 type_with_colon = re.compile(rf"^({VALID_TYPES_RE})!?:")
142 if type_with_colon.match(msg):
143 # Allow scope-optional types (chore, docs, ci) without scope
144 if type_lower in SCOPE_OPTIONAL_TYPES:
145 no_scope_match = NO_SCOPE_PATTERN.match(msg)
146 if no_scope_match:
147 desc = no_scope_match.group(3)
148 if desc and desc[0].isupper():
149 errors.append(
150 f"Description should start with lowercase: '{desc[:30]}...'"
151 )
152 if desc.endswith("."):
153 errors.append("Description should not end with a period")
154 return errors
155 errors.append(
156 "Missing scope. Use 'type(scope): description' "
157 "instead of 'type: description'"
158 )
159 elif msg and msg[0].isupper():
160 errors.append(
161 "Do not start with a capital letter. "
162 "Use conventional commit format: type(scope): description"
163 )
164 else:
165 errors.append(
166 f"Invalid type '{type_word}'. "
167 f"Allowed types: {', '.join(VALID_TYPES)}"
168 )
169 return errors
170
171 # Full format check
172 full = FULL_PATTERN.match(msg)
173 if not full:
174 # Diagnose specific issues
175 empty_scope = re.compile(rf"^({VALID_TYPES_RE})\‍(\‍)")
176 no_space = re.compile(rf"^({VALID_TYPES_RE})\‍([^)]*\‍)!?:[^ ]")
177 no_colon = re.compile(rf"^({VALID_TYPES_RE})\‍([^)]*\‍)!?[^:!]")
178
179 if empty_scope.match(msg):
180 errors.append("Scope cannot be empty")
181 elif no_space.match(msg):
182 errors.append("Missing space after colon. Use 'type(scope): description'")
183 elif no_colon.match(msg):
184 errors.append("Missing colon after scope. Use 'type(scope): description'")
185 else:
186 # Check if scope contains filename or PR reference
187 scope_match = re.match(rf"^({VALID_TYPES_RE})\‍(([^)]+)\‍)!?: .+", msg)
188 if scope_match:
189 scope = scope_match.group(2)
190 if re.search(r"\.[a-zA-Z]+$", scope):
191 errors.append(
192 f"Scope '{scope}' looks like a filename. "
193 "Use a module name instead, e.g. 'obj' not 'lv_obj.h'"
194 )
195 elif "#" in scope:
196 errors.append(
197 f"Scope '{scope}' should be a module name, "
198 "not a PR reference. Put PR number in description, "
199 "e.g. 'fix(module): description (#1234)'"
200 )
201 else:
202 errors.append("Invalid format. Expected: type(scope): description")
203 else:
204 errors.append("Invalid format. Expected: type(scope): description")
205 return errors
206
207 # Validate description
208 desc = full.group(4)
209
210 if desc and desc[0].isupper():
211 errors.append(f"Description should start with lowercase: '{desc[:30]}...'")
212
213 if desc.endswith("."):
214 errors.append("Description should not end with a period")
215
216 return errors

225def find_base_branch():
226 """Find the base branch for comparison."""
227 candidates = ["origin/master", "origin/main", "upstream/master", "upstream/main"]
228 for branch in candidates:
229 _, rc = git_run("rev-parse", "--verify", branch)
230 if rc == 0:
231 return branch
232 return None

235def check_commits(base_branch=None, last_n=None):
236 """Check all commits between base branch and HEAD."""
237 if last_n:
238 log_output, _ = git_run("log", "--oneline", f"-{last_n}")
239 if not log_output:
240 print("No commits to check.")
241 return 0
242 print(f"Checking commit message style (last {last_n} commits)...")
243 else:
244 if not base_branch:
245 base_branch = find_base_branch()
246 if not base_branch:
247 print(
248 "Warning: Could not determine base branch, "
249 "skipping commit message style check."
250 )
251 return 0
252
253 merge_base, rc = git_run("merge-base", base_branch, "HEAD")
254 if rc != 0 or not merge_base:
255 print(
256 "Warning: Could not determine merge base, "
257 "skipping commit message style check."
258 )
259 return 0
260
261 head, _ = git_run("rev-parse", "HEAD")
262 if head == merge_base:
263 print("No new commits to check, skipping commit message style check.")
264 return 0
265
266 log_output, _ = git_run("log", "--oneline", f"{merge_base}..HEAD")
267 if not log_output:
268 print("No new commits to check.")
269 return 0
270
271 print(f"Checking commit message style (base: {base_branch})...")
272
273 error_count = 0
274 total_count = 0
275 for line in log_output.splitlines():
276 parts = line.split(" ", 1)
277 if len(parts) < 2:
278 continue
279 hash_str, msg = parts[0], parts[1]
280 total_count += 1
281 errors = check_commit_msg(msg)
282 if errors:
283 error_count += 1
284 print(f"\n Commit: {hash_str[:12]} {msg}")
285 for e in errors:
286 print(f" ✗ {e}")
287 else:
288 print(f" ✓ {hash_str[:12]} {msg}")
289
290 if error_count > 0:
291 print(
292 f"""
293==========================================
294 Commit message style check FAILED
295 {error_count} commit(s) with bad format
296==========================================
297
298Expected format: type(scope): description
299
300 Valid types: {', '.join(VALID_TYPES)}
301 Scope: required (except chore/docs/ci), e.g. (draw), (obj), (style)
302 Description: lowercase start, no trailing period
303
304Good examples:
305 feat(draw): add new gradient support
306 fix(obj): fix crash when object is deleted
307 test(cache): add complete cache test cases
308 perf(style): optimize style removal performance
309
310See: {COMMIT_MSG_DOC_URL}
311
312Use 'git commit --amend' or 'git rebase -i' to fix your commit messages."""
313 )
314 return 1
315
316 print("Commit message style check passed")
317 return 0
