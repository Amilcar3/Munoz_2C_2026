87void lv_sdl_mousewheel_handler(SDL_Event * event)
88{
89 uint32_t win_id = UINT32_MAX;
90 switch(event->type) {
91 case SDL_MOUSEWHEEL:
92 win_id = event->wheel.windowID;
93 break;
94 case SDL_MOUSEBUTTONDOWN:
95 case SDL_MOUSEBUTTONUP:
96 win_id = event->button.windowID;
97 break;
98 default:
99 return;
100 }
101
102 lv_display_t * disp = lv_sdl_get_disp_from_win_id(win_id);
103
104 /*Find a suitable indev*/
105 lv_indev_t * indev = lv_indev_get_next(NULL);
106 while(indev) {
107 if(lv_indev_get_read_cb(indev) == sdl_mousewheel_read) {
108 /*If disp is NULL for any reason use the first indev with the correct type*/
109 if(disp == NULL || lv_indev_get_display(indev) == disp) break;
110 }
111 indev = lv_indev_get_next(indev);
112 }
113
114 if(indev == NULL) return;
115 lv_sdl_mousewheel_t * dsc = lv_indev_get_driver_data(indev);
116
117 switch(event->type) {
118 case SDL_MOUSEWHEEL:
119#ifdef __EMSCRIPTEN__
120 /*Emscripten scales it wrong*/
121 if(event->wheel.y < 0) dsc->diff++;
122 if(event->wheel.y > 0) dsc->diff--;
123#else
124 dsc->diff = -event->wheel.y;
125#endif
126 break;
127 case SDL_MOUSEBUTTONDOWN:
128 if(event->button.button == SDL_BUTTON_MIDDLE) {
129 dsc->state = LV_INDEV_STATE_PRESSED;
130 }
131 break;
132 case SDL_MOUSEBUTTONUP:
133 if(event->button.button == SDL_BUTTON_MIDDLE) {
134 dsc->state = LV_INDEV_STATE_RELEASED;
135 }
136 break;
137 default:
138 break;
139 }
140 lv_indev_read(indev);
141}
