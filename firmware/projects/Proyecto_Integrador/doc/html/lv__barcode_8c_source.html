302static void lv_barcode_event(const lv_obj_class_t * class_p, lv_event_t * e)
303{
304 LV_UNUSED(class_p);
305
306 lv_event_code_t code = lv_event_get_code(e);
307
308 /*Call the ancestor's event handler*/
309 lv_result_t res = lv_obj_event_base(MY_CLASS, e);
310 if(res != LV_RESULT_OK) return;
311
312 lv_obj_t * obj = lv_event_get_current_target(e);
313 lv_barcode_t * barcode = (lv_barcode_t *)obj;
314
315 if(code == LV_EVENT_SIZE_CHANGED) {
316 /*Refitting reallocates the canvas, which is itself a content size change and comes
317 *back here: `fitting` catches that echo within the call, barcode_fit_needed() the
318 *one a layout pass later. An out of date bitmap is always refitted, so a size that
319 *could not be fitted before gets another go.*/
320 if(!barcode->fitting && (barcode->needs_update || barcode_fit_needed(obj))) {
321 barcode_mark_dirty(obj);
322 }
323 }
324 else if(code == LV_EVENT_DRAW_MAIN_BEGIN) {
325 /*A failed state is not retried here; only a change makes the Widget try again*/
326 if(barcode->needs_update && barcode->render_valid) {
327 LV_ASSERT(barcode->update_mode == LV_BARCODE_UPDATE_MODE_DEFERRED);
328
329 /*Safe: the setter already resized the canvas, so nothing is reallocated here*/
330 LV_LOG_WARN("filling in the barcode during the redraw because lv_barcode_render() "
331 "was not called after the property changes; this adds the work to the "
332 "refresh and its result cannot be reported");
333
334 if(barcode_fill(obj) != LV_RESULT_OK) {
335 barcode->render_valid = false;
336 LV_LOG_ERROR("the barcode could not be filled in during the redraw "
337 "(scale %d, %s); the bitmap is left blank",
338 (int)barcode->scale, barcode->direction == LV_DIR_VER ? "vertical" : "horizontal");
339 }
340 }
341 }
342}

344static bool lv_barcode_change_buf_size(lv_obj_t * obj, int32_t w, int32_t h)
345{
346 LV_ASSERT_NULL(obj);
347 if(w <= 0 || h <= 0) {
348 LV_LOG_WARN("invalid size: %" LV_PRId32 " x %" LV_PRId32, w, h);
349 return false;
350 }
351
352 lv_draw_buf_t * old_buf = lv_canvas_get_draw_buf(obj);
353
354 /*Reuse the buffer at an unchanged geometry, to avoid the invalidation and relayout a
355 *reallocation triggers*/
356 if(old_buf != NULL &&
357 (int32_t)old_buf->header.w == w &&
358 (int32_t)old_buf->header.h == h &&
359 old_buf->header.cf == LV_COLOR_FORMAT_I1) {
360 return true;
361 }
362
363 lv_draw_buf_t * new_buf = lv_draw_buf_create(w, h, LV_COLOR_FORMAT_I1, LV_STRIDE_AUTO);
364 if(new_buf == NULL) {
365 LV_LOG_ERROR("malloc failed for canvas buffer");
366 return false;
367 }
368
369 lv_canvas_set_draw_buf(obj, new_buf);
370 LV_LOG_INFO("set canvas buffer: %p, width = %" LV_PRId32, (void *)new_buf, w);
371
372 if(old_buf != NULL) lv_draw_buf_destroy(old_buf);
373 return true;
374}

389static bool barcode_store_data(lv_barcode_t * barcode, const char * data)
390{
391 const size_t len = lv_strlen(data);
392
393 /*Assign only on success, so a failed realloc leaves the previous data owned by
394 *`barcode`. lv_realloc(NULL, n) allocates, so the first call needs no special case.*/
395 char * new_data = lv_realloc(barcode->data, len + 1);
396 LV_ASSERT_MALLOC(new_data);
397 if(new_data == NULL) return false;
398
399 lv_memcpy(new_data, data, len);
400 new_data[len] = '\0';
401
402 barcode->data = new_data;
403
404 barcode_drop_encoded(barcode);
405 return true;
406}

422static uint8_t * barcode_encode(const lv_barcode_t * barcode, int32_t * bar_count)
423{
424 const char * data = barcode->data;
425 LV_ASSERT_NULL(data);
426
427 size_t len = code128_estimate_len(data);
428 LV_LOG_INFO("data: %s, len = %zu", data, len);
429
430 uint8_t * pattern = lv_malloc(len);
431 LV_ASSERT_MALLOC(pattern);
432 if(pattern == NULL) {
433 LV_LOG_ERROR("malloc failed for the bar pattern");
434 return NULL;
435 }
436
437 int32_t w = 0;
438 switch(barcode->encoding) {
439 case LV_BARCODE_ENCODING_CODE128_GS1:
440 w = (int32_t) code128_encode_gs1(data, (char *)pattern, len);
441 break;
442 case LV_BARCODE_ENCODING_CODE128_RAW:
443 w = (int32_t) code128_encode_raw(data, (char *)pattern, len);
444 break;
445 default:
446 LV_ASSERT(false);
447 break;
448 }
449 LV_LOG_INFO("barcode width = %" LV_PRId32, w);
450
451 if(w <= 0) {
452 LV_LOG_WARN("the data could not be encoded");
453 lv_free(pattern);
454 return NULL;
455 }
456
457 *bar_count = w;
458 return pattern;
459}

461static bool barcode_fit(lv_obj_t * obj)
462{
463 lv_barcode_t * barcode = (lv_barcode_t *)obj;
464
465 if(barcode->bar_count == 0) {
466 barcode->pattern = barcode_encode(barcode, &barcode->bar_count);
467 if(barcode->pattern == NULL) return false;
468 }
469
470 const int32_t bar_count = barcode->bar_count;
471 const int32_t scale = barcode->scale;
472
473 if(bar_count > INT32_MAX / scale) {
474 LV_LOG_WARN("%" LV_PRId32 " bars at scale %" LV_PRId32 " do not fit an int32_t",
475 bar_count, scale);
476 return false;
477 }
478 const int32_t bars_px = bar_count * scale;
479
480 int32_t buf_w;
481 int32_t buf_h;
482
483 if(barcode->tiled) {
484 buf_w = (barcode->direction == LV_DIR_HOR) ? bars_px : 1;
485 buf_h = (barcode->direction == LV_DIR_VER) ? bars_px : 1;
486 }
487 else {
488 lv_obj_update_layout(obj);
489 buf_w = (barcode->direction == LV_DIR_HOR) ? bars_px : lv_obj_get_width(obj);
490 buf_h = (barcode->direction == LV_DIR_VER) ? bars_px : lv_obj_get_height(obj);
491 }
492
493 /*Reallocating changes the content size, which comes straight back as SIZE_CHANGED*/
494 barcode->fitting = true;
495 bool ok = lv_barcode_change_buf_size(obj, buf_w, buf_h);
496 barcode->fitting = false;
497
498 return ok;
499}

501static bool barcode_fit_needed(lv_obj_t * obj)
502{
503 lv_barcode_t * barcode = (lv_barcode_t *)obj;
504
505 if(barcode->data == NULL) return false;
506
507 if(barcode->tiled) return false;
508
509 lv_draw_buf_t * draw_buf = lv_canvas_get_draw_buf(obj);
510 if(draw_buf == NULL) return true;
511
512 const bool hor = (barcode->direction == LV_DIR_HOR);
513 const int32_t have = hor ? (int32_t)draw_buf->header.h : (int32_t)draw_buf->header.w;
514 const int32_t want = hor ? lv_obj_get_height(obj) : lv_obj_get_width(obj);
515 return have != want;
516}

518static bool barcode_fit_only(lv_obj_t * obj)

532static lv_result_t barcode_fill(lv_obj_t * obj)
533{
534 lv_barcode_t * barcode = (lv_barcode_t *)obj;
535
536 if(barcode->data == NULL) return LV_RESULT_INVALID;
537
538 lv_draw_buf_t * draw_buf = lv_canvas_get_draw_buf(obj);
539 if(draw_buf == NULL) return LV_RESULT_INVALID;
540
541 uint8_t * pattern = barcode->pattern;
542 barcode->pattern = NULL;
543 if(pattern == NULL) {
544 pattern = barcode_encode(barcode, &barcode->bar_count);
545 if(pattern == NULL) return LV_RESULT_INVALID;
546 }
547
548 const int32_t bar_count = barcode->bar_count;
549 const int32_t scale = barcode->scale;
550 const int32_t buf_w = (int32_t)draw_buf->header.w;
551 const int32_t buf_h = (int32_t)draw_buf->header.h;
552
553 const int32_t avail = (barcode->direction == LV_DIR_HOR) ? buf_w : buf_h;
554 if(bar_count > INT32_MAX / scale || bar_count * scale > avail) {
555 LV_LOG_WARN("the bars (%" LV_PRId32 " x %" LV_PRId32 " px) do not fit the %"
556 LV_PRId32 " px canvas", bar_count, scale, avail);
557 lv_free(pattern);
558 return LV_RESULT_INVALID;
559 }
560
561 /*Temporarily disable invalidation to improve the efficiency of lv_canvas_set_px*/
562 lv_display_enable_invalidation(lv_obj_get_display(obj), false);
563
564 uint32_t stride = draw_buf->header.stride;
565 const lv_color_t color = lv_color_hex(1);
566
567 lv_draw_buf_clear(draw_buf, NULL);
568
569 /*Set the palette on the draw buffer, not via lv_canvas_set_palette(), to avoid an
570 *invalidation here - the caller or the draw pass refreshes the object*/
571 lv_draw_buf_set_palette(draw_buf, 0, lv_color_to_32(barcode->light_color, LV_OPA_COVER));
572 lv_draw_buf_set_palette(draw_buf, 1, lv_color_to_32(barcode->dark_color, LV_OPA_COVER));
573
574 for(int32_t x = 0; x < bar_count; x++) {
575 /*skip empty data*/
576 if(pattern[x] == 0) {
577 continue;
578 }
579
580 for(int32_t i = 0; i < scale; i++) {
581 int32_t offset = x * scale + i;
582 if(barcode->direction == LV_DIR_HOR) {
583 lv_canvas_set_px(obj, offset, 0, color, LV_OPA_COVER);
584 }
585 else { /*LV_DIR_VER*/
586 if(barcode->tiled) {
587 lv_canvas_set_px(obj, 0, offset, color, LV_OPA_COVER);
588 }
589 else {
590 uint8_t * dest = lv_draw_buf_goto_xy(draw_buf, 0, offset);
591 lv_memset(dest, 0xFF, stride);
592 }
593 }
594 }
595 }
596
597 if(!barcode->tiled && barcode->direction == LV_DIR_HOR && buf_h > 1) {
598 int32_t h = buf_h - 1;
599 const uint8_t * src = lv_draw_buf_goto_xy(draw_buf, 0, 0);
600 uint8_t * dest = lv_draw_buf_goto_xy(draw_buf, 0, 1);
601 while(h--) {
602 lv_memcpy(dest, src, stride);
603 dest += stride;
604 }
605 }
606
607 lv_display_enable_invalidation(lv_obj_get_display(obj), true);
608 lv_image_cache_drop(draw_buf);
609
610 lv_free(pattern);
611
612 barcode->needs_update = false;
613 barcode->render_valid = true;
614 return LV_RESULT_OK;
615}

617static lv_result_t barcode_render(lv_obj_t * obj)
618{
619 lv_barcode_t * barcode = (lv_barcode_t *)obj;
620
621 /*Start invalid so no early return can forget to record a failure; only barcode_fill()'s
622 *success path sets `render_valid`. `needs_update` survives a failure on purpose - the
623 *bitmap still does not match the properties - and a cleared `render_valid` keeps the
624 *draw hook from retrying a known-bad state every frame. Nothing is logged here: the
625 *result is returned, and the caller reports it if nothing else will.*/
626 barcode->render_valid = false;
627 barcode->needs_update = true;
628
629 if(barcode->data == NULL) return LV_RESULT_INVALID;
