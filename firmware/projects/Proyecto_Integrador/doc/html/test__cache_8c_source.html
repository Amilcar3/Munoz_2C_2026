34static lv_cache_compare_res_t compare_cb(const test_data_t * lhs,
35 const test_data_t * rhs)
36{
37 if(lhs->key1 != rhs->key1) {
38 return lhs->key1 > rhs->key1 ? 1 : -1;
39 }
40 if(lhs->key2 != rhs->key2) {
41 return lhs->key2 > rhs->key2 ? 1 : -1;
42 }
43 return 0;
44}

63void test_cache_lru_rb_1(void)
64{
65 lv_cache_t * cache = create_cache(&lv_cache_class_lru_rb_size, CACHE_SIZE_BYTES);
66 TEST_ASSERT_NOT_NULL(cache);
67 void * record_data_ptr = NULL;
68
69 // create many node unless cache is full
70 uint32_t curr_mem_size = 8;
71 uint32_t curr_total_mem_size = 0;
72 while(curr_total_mem_size < CACHE_SIZE_BYTES) {
73 test_data_t search_key = { .slot.size = curr_mem_size,
74
75 .key1 = (int32_t)curr_mem_size,
76 .key2 = (int32_t)curr_mem_size + 1
77 };
78
79 // acquire cache first
80 lv_cache_entry_t * entry =
81 lv_cache_acquire(cache, &search_key, NULL);
82 if(entry != NULL) {
83 continue;
84 }
85
86 // if cache miss then add cache
87 entry = lv_cache_add(cache, &search_key, NULL);
88 TEST_ASSERT_NOT_NULL(entry);
89
90 test_data_t * data = lv_cache_entry_get_data(entry);
91 TEST_ASSERT_NOT_NULL(data);
92
93 data->data = lv_malloc(data->slot.size);
94
95 // record data ptr when {key1 = 32, key2 = 33}.
96 if(search_key.key1 == 32 && search_key.key2 == 33) {
97 record_data_ptr = data->data;
98 }
99
100 lv_cache_release(cache, entry, NULL);
101
102 curr_total_mem_size += curr_mem_size;
103 curr_mem_size *= 2;
104
105 TEST_PRINTF("cache free: %d, allocated: %d",
106 lv_cache_get_free_size(cache, NULL), curr_mem_size);
107 }
108
109 /*
110 * allocated = 8 + 16 + 32 + 64 + 128 + 256 - 8 - 16 + 512 = 992
111 * free = 1000 - 992 = 8
112 * The last node size should be 1024, but the cache's max size is 1000. So new entry will be allocated failed and
113 * the loop will break.
114 * */
115 TEST_ASSERT_EQUAL(8, lv_cache_get_free_size(cache, NULL));
116
117 /*
118 * Search entry {key1 = 32, key2 = 33}
119 */
120 test_data_t search_key32 = { .key1 = 32, .key2 = 33 };
121 lv_cache_entry_t * entry_key32 =
122 lv_cache_acquire(cache, &search_key32, NULL);
123
124 test_data_t * cached_data_key32 = lv_cache_entry_get_data(entry_key32);
125 TEST_ASSERT_EQUAL(record_data_ptr, cached_data_key32->data);
126
127 /*
128 * Now drop the cache {key1 = 32, key2 = 33}. However, this entry is acquired once without release, so `drop`
129 * will not release the memory allocated by this entry.
130 */
131 size_t mem_curr_free = lv_test_get_free_mem();
132 lv_cache_drop(cache, &search_key32, NULL);
133 /*
134 * Though it doesn't release the data, the entry and other structure has been freed.
135 * lv_rb_note_t (4 ptr + 1 int32 may align to 8 bit on 64 bit machine) + lv_ll (2 ptr + node_size).
136 * Also, the def heap has some other aligned attributes. It'll also affect the final result.
137 */
138 TEST_ASSERT_MEM_LEAK_LESS_THAN(
139 mem_curr_free,
140 sizeof(lv_rb_node_t) + sizeof(void *) +
141 (sizeof(lv_ll_node_t *) + sizeof(lv_ll_node_t *)) +
142 32); // the last 32 is an error in memory allocating
143 mem_curr_free = lv_test_get_free_mem();
144 lv_cache_release(cache, entry_key32, NULL);
145 TEST_ASSERT_MEM_LEAK_LESS_THAN(
146 mem_curr_free, lv_cache_entry_get_size(sizeof(test_data_t)) +
147 sizeof(void *) + 32 + 32);
148
149 // Now the freed cache size should be 8 + 32 = 40
150 TEST_ASSERT_EQUAL(40, lv_cache_get_free_size(cache, NULL));
151
152 lv_cache_destroy(cache, NULL);
153}
