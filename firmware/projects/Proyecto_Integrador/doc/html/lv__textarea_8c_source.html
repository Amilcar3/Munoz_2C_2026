194void lv_textarea_add_text(lv_obj_t * obj, const char * txt)
195{
196 LV_CHECK_OBJ(obj, MY_CLASS, return);
197 LV_CHECK_ARG(txt != NULL, return);
198
199 lv_textarea_t * ta = (lv_textarea_t *)obj;
200
201 if(ta->pwd_mode) pwd_char_hider(obj); /*Make sure all the current text contains only '*'*/
202
203 /*Add the character one-by-one if not all characters are accepted or there is character limit.*/
204 if(lv_textarea_get_accepted_chars(obj) || lv_textarea_get_max_length(obj)) {
205 if(add_text(obj, txt)) {
206 lv_obj_send_event(obj, LV_EVENT_VALUE_CHANGED, NULL);
207 }
208 return;
209 }
210
211 lv_result_t res = insert_handler(obj, txt);
212 if(res != LV_RESULT_OK) return;
213
214 /*If the textarea is empty, invalidate it to hide the placeholder*/
215 if(ta->placeholder_txt) {
216 const char * txt_act = lv_label_get_text(ta->label);
217 if(txt_act[0] == '\0') lv_obj_invalidate(obj);
218 }
219
220 /*Insert the text*/
221 lv_label_ins_text(ta->label, ta->cursor.pos, txt);
222 lv_textarea_clear_selection(obj);
223
224 if(ta->pwd_mode) {
225 size_t realloc_size = lv_strlen(ta->pwd_tmp) + lv_strlen(txt) + 1;
226 char * new_buf = lv_realloc(ta->pwd_tmp, realloc_size);
227 LV_ASSERT_MALLOC(new_buf);
228 if(!new_buf) return;
229 ta->pwd_tmp = new_buf;
230
231 lv_text_ins(ta->pwd_tmp, ta->cursor.pos, txt);
232
233 /*Auto hide characters*/
234 auto_hide_characters(obj);
235 }
236
237 /*Move the cursor after the new text*/
238 lv_textarea_set_cursor_pos(obj, lv_textarea_get_cursor_pos(obj) + lv_text_get_encoded_length(txt));
239
240 lv_obj_send_event(obj, LV_EVENT_VALUE_CHANGED, NULL);
241}

243void lv_textarea_delete_char(lv_obj_t * obj)
244{
245 LV_CHECK_OBJ(obj, MY_CLASS, return);
246
247 lv_textarea_t * ta = (lv_textarea_t *)obj;
248 uint32_t cur_pos = ta->cursor.pos;
249
250 if(cur_pos == 0) return;
251
252 char del_buf[2] = {LV_KEY_DEL, '\0'};
253
254 lv_result_t res = insert_handler(obj, del_buf);
255 if(res != LV_RESULT_OK) return;
256
257 char * label_txt = lv_label_get_text(ta->label);
258
259 /*Delete a character*/
260 lv_text_cut(label_txt, ta->cursor.pos - 1, 1);
261
262 /*Refresh the label*/
263 lv_label_set_text(ta->label, label_txt);
264 lv_textarea_clear_selection(obj);
265
266 /*If the textarea became empty, invalidate it to hide the placeholder*/
267 if(ta->placeholder_txt) {
268 const char * txt = lv_label_get_text(ta->label);
269 if(txt[0] == '\0') lv_obj_invalidate(obj);
270 }
271
272 if(ta->pwd_mode) {
273 lv_text_cut(ta->pwd_tmp, ta->cursor.pos - 1, 1);
274
275 char * new_buf = lv_realloc(ta->pwd_tmp, lv_strlen(ta->pwd_tmp) + 1);
276 LV_ASSERT_MALLOC(new_buf);
277 if(!new_buf) return;
278 ta->pwd_tmp = new_buf;
279 }
280
281 /*Move the cursor to the place of the deleted character*/
282 lv_textarea_set_cursor_pos(obj, ta->cursor.pos - 1);
283
284 lv_obj_send_event(obj, LV_EVENT_VALUE_CHANGED, NULL);
285

301void lv_textarea_set_text(lv_obj_t * obj, const char * txt)
302{
303 LV_CHECK_OBJ(obj, MY_CLASS, return);
304 LV_CHECK_ARG(txt != NULL, return);
305
306 lv_textarea_t * ta = (lv_textarea_t *)obj;
307
308 /*Clear the existing selection*/
309 lv_textarea_clear_selection(obj);
310
311 /*Add the character one-by-one if not all characters are accepted or there is character limit.*/
312 bool text_changed = false;
313 if(lv_textarea_get_accepted_chars(obj) || lv_textarea_get_max_length(obj)) {
314 lv_label_set_text(ta->label, "");
315 lv_textarea_set_cursor_pos(obj, LV_TEXTAREA_CURSOR_LAST);
316 if(ta->pwd_mode) {
317 ta->pwd_tmp[0] = '\0'; /*Clear the password too*/
318 }
319 text_changed = add_text(obj, txt);
320 }
321 else {
322 lv_label_set_text(ta->label, txt);
323 lv_textarea_set_cursor_pos(obj, LV_TEXTAREA_CURSOR_LAST);
324 text_changed = true;
325 }
326
327 /*If the textarea is empty, invalidate it to hide the placeholder*/
328 if(ta->placeholder_txt) {
329 const char * txt_act = lv_label_get_text(ta->label);
330 if(txt_act[0] == '\0') lv_obj_invalidate(obj);
331 }
332
333 if(ta->pwd_mode) {
334 lv_free(ta->pwd_tmp);
335 ta->pwd_tmp = lv_strdup(txt);
336 LV_ASSERT_MALLOC(ta->pwd_tmp);
337 if(ta->pwd_tmp == NULL) return;
338
339 pwd_char_hider(obj);
340 }
341
342 if(text_changed) {
343 lv_obj_send_event(obj, LV_EVENT_VALUE_CHANGED, NULL);
344 }
345}

347void lv_textarea_set_placeholder_text(lv_obj_t * obj, const char * txt)
348{
349 LV_CHECK_OBJ(obj, MY_CLASS, return);
350
351 lv_textarea_t * ta = (lv_textarea_t *)obj;
352 const bool has_text = txt && txt[0] != '\0';
353
354 if(!has_text) {
355 if(ta->placeholder_txt) {
356 lv_free(ta->placeholder_txt);
357 ta->placeholder_txt = NULL;
358 }
359 }
360 else {
361 size_t txt_len = lv_strlen(txt);
362 /*Allocate memory for the placeholder_txt text*/
363 /*NOTE: Using special realloc behavior, malloc-like when data_p is NULL*/
364 ta->placeholder_txt = lv_realloc(ta->placeholder_txt, txt_len + 1);
365 LV_ASSERT_MALLOC(ta->placeholder_txt);
366 if(ta->placeholder_txt == NULL) {
367 LV_LOG_ERROR("couldn't allocate memory for placeholder");
368 return;
369 }
370
371 lv_strcpy(ta->placeholder_txt, txt);
372 ta->placeholder_txt[txt_len] = '\0';
373
374 calc_placeholder_text_size(obj);
375 }
376
377 lv_obj_invalidate(obj);
378}

888static void lv_textarea_event(const lv_obj_class_t * class_p, lv_event_t * e)
889{
890 LV_UNUSED(class_p);
891
892 lv_result_t res;
893 /*Call the ancestor's event handler*/
894 res = lv_obj_event_base(MY_CLASS, e);
895 if(res != LV_RESULT_OK) return;
896
897 lv_event_code_t code = lv_event_get_code(e);
898 lv_obj_t * obj = lv_event_get_current_target(e);
899
900 if(code == LV_EVENT_FOCUSED) {
901 start_cursor_blink(obj);
902 }
903 else if(code == LV_EVENT_KEY) {
904 uint32_t c = *((uint32_t *)lv_event_get_param(e)); /*uint32_t because can be UTF-8*/
905 if(c == LV_KEY_RIGHT)
906 lv_textarea_cursor_right(obj);
907 else if(c == LV_KEY_LEFT)
908 lv_textarea_cursor_left(obj);
909 else if(c == LV_KEY_UP)
910 lv_textarea_cursor_up(obj);
911 else if(c == LV_KEY_DOWN)
912 lv_textarea_cursor_down(obj);
913 else if(c == LV_KEY_BACKSPACE)
914 lv_textarea_delete_char(obj);
915 else if(c == LV_KEY_DEL)
916 lv_textarea_delete_char_forward(obj);
917 else if(c == LV_KEY_HOME)
918 lv_textarea_set_cursor_pos(obj, 0);
919 else if(c == LV_KEY_END)
920 lv_textarea_set_cursor_pos(obj, LV_TEXTAREA_CURSOR_LAST);
921 else if(c == LV_KEY_ENTER && lv_textarea_get_one_line(obj))
922 lv_obj_send_event(obj, LV_EVENT_READY, NULL);
923 else {
924 lv_textarea_add_char(obj, c);
925 }
926 }
927 else if(code == LV_EVENT_PRESSED || code == LV_EVENT_PRESSING || code == LV_EVENT_PRESS_LOST ||
928 code == LV_EVENT_RELEASED) {
929 update_cursor_position_on_click(e);
930 }
931 else if(code == LV_EVENT_DRAW_MAIN) {
932 draw_placeholder(e);
933 }
934 else if(code == LV_EVENT_DRAW_POST) {
935 draw_cursor(e);
936 }
937 else if(code == LV_EVENT_SIZE_CHANGED) {
938 lv_textarea_t * ta = (lv_textarea_t *)obj;
939 lv_textarea_scroll_to_cursor_pos(obj, ta->cursor.pos);
940 }
941 else if(code == LV_EVENT_STYLE_CHANGED) {
942 refr_cursor_area(obj);
943 }
944}

1090static void refr_cursor_area(lv_obj_t * obj)
1091{
1092 lv_textarea_t * ta = (lv_textarea_t *)obj;
1093
1094 const lv_font_t * font = lv_obj_get_style_text_font_internal(obj, LV_PART_MAIN);
1095 int32_t line_space = lv_obj_get_style_text_line_space_internal(obj, LV_PART_MAIN);
1096
1097 uint32_t cur_pos = lv_textarea_get_cursor_pos(obj);
1098 const char * txt = lv_label_get_text(ta->label);
1099
1100 uint32_t byte_pos = lv_text_encoded_get_byte_id(txt, cur_pos);
1101 uint32_t letter = lv_text_encoded_next(&txt[byte_pos], NULL);
1102
1103 /* Letter height and width */
1104 const int32_t letter_h = lv_font_get_line_height_internal(font);
1105 /*Set letter_w (set not 0 on non printable but valid chars)*/
1106 uint32_t letter_space = letter;
1107 if(is_valid_but_non_printable_char(letter)) {
1108 letter_space = ' ';
1109 }
1110 int32_t letter_w = lv_font_get_glyph_width(font, letter_space, IGNORE_KERNING);
1111
1112 lv_point_t letter_pos;
1113 lv_label_get_letter_pos(ta->label, cur_pos, &letter_pos);
1114
1115 lv_text_align_t align = lv_obj_calculate_style_text_align_internal(ta->label, LV_PART_MAIN,
1116 lv_label_get_text(ta->label));
1117
1118 /*If the cursor is out of the text (most right) draw it to the next line*/
1119 if(((letter_pos.x + ta->label->coords.x1) + letter_w > ta->label->coords.x2) &&
1120 (ta->one_line == 0 && align != LV_TEXT_ALIGN_RIGHT)) {
1121
1122 letter_pos.x = 0;
1123 letter_pos.y += letter_h + line_space;
1124
1125 if(letter != '\0') {
1126 byte_pos += lv_text_encoded_size(&txt[byte_pos]);
1127 letter = lv_text_encoded_next(&txt[byte_pos], NULL);
1128 }
1129
1130 uint32_t tmp = letter;
1131 if(is_valid_but_non_printable_char(letter)) {
1132 /*If non printable get the letter_w of the space char*/
1133 tmp = ' ';
1134 }
1135 letter_w = lv_font_get_glyph_width(font, tmp, IGNORE_KERNING);
1136 }
1137
1138 /*Save the byte position. It is required to draw `LV_CURSOR_BLOCK`*/
1139 ta->cursor.txt_byte_pos = byte_pos;
1140
1141 /*Calculate the cursor according to its type*/
1142 int32_t border_width = lv_obj_get_style_border_width_internal(obj, LV_PART_CURSOR);
1143 int32_t top = lv_obj_get_style_pad_top_internal(obj, LV_PART_CURSOR) + border_width;
1144 int32_t bottom = lv_obj_get_style_pad_bottom_internal(obj, LV_PART_CURSOR) + border_width;
1145 int32_t left = lv_obj_get_style_pad_left_internal(obj, LV_PART_CURSOR) + border_width;
1146 int32_t right = lv_obj_get_style_pad_right_internal(obj, LV_PART_CURSOR) + border_width;
1147 int32_t letter_space_w = lv_obj_get_style_text_letter_space_internal(ta->label, LV_PART_MAIN);
1148
1149 lv_area_t cur_area;
1150 cur_area.x1 = letter_pos.x - left - letter_space_w / 2;
1151 cur_area.y1 = letter_pos.y - top;
1152 cur_area.x2 = letter_pos.x + right + letter_w - 1 + (letter_space_w + 1) / 2;
1153 cur_area.y2 = letter_pos.y + bottom + letter_h - 1;
1154
1155 /*Save the new area*/
1156 lv_area_t area_tmp;
1157 area_tmp = ta->cursor.area;
1158 area_tmp.x1 += ta->label->coords.x1;
1159 area_tmp.y1 += ta->label->coords.y1;
1160 area_tmp.x2 += ta->label->coords.x1;
1161 area_tmp.y2 += ta->label->coords.y1;
1162 lv_obj_invalidate_area(obj, &area_tmp);
1163
1164 ta->cursor.area = cur_area;
1165
1166 area_tmp = ta->cursor.area;
1167 area_tmp.x1 += ta->label->coords.x1;
1168 area_tmp.y1 += ta->label->coords.y1;
1169 area_tmp.x2 += ta->label->coords.x1;
1170 area_tmp.y2 += ta->label->coords.y1;
1171 lv_obj_invalidate_area(obj, &area_tmp);
1172}

1174static void update_cursor_position_on_click(lv_event_t * e)
1175{
1176 lv_indev_t * click_source = lv_indev_active();
1177 if(click_source == NULL) return;
1178
1179 lv_obj_t * obj = lv_event_get_current_target(e);
1180 lv_textarea_t * ta = (lv_textarea_t *)obj;
1181 if(ta->cursor.click_pos == 0) return;
1182
1183 if(lv_indev_get_type(click_source) == LV_INDEV_TYPE_KEYPAD ||
1184 lv_indev_get_type(click_source) == LV_INDEV_TYPE_ENCODER) {
1185 return;
1186 }
1187
1188 lv_area_t label_coords;
1189 lv_obj_get_coords(ta->label, &label_coords);
1190
1191 lv_point_t point_act, vect_act;
1192 lv_indev_get_point(click_source, &point_act);
1193 lv_indev_get_vect(click_source, &vect_act);
1194
1195 if(point_act.x < 0 || point_act.y < 0) return; /*Ignore event from keypad*/
1196 lv_point_t rel_pos;
1197 rel_pos.x = point_act.x - label_coords.x1;
1198 rel_pos.y = point_act.y - label_coords.y1;
1199
1200 const lv_event_code_t code = lv_event_get_code(e);
1201
1202 int32_t label_width = lv_obj_get_width(ta->label);
1203 uint32_t char_id_at_click = 0;
1204
1205#if LV_LABEL_TEXT_SELECTION
1206 lv_label_t * label_data = (lv_label_t *)ta->label;
1207 bool click_outside_label = false;
1208 /*Check if the click happened on the left side of the area outside the label*/
1209 if(rel_pos.x < 0) {
1210 char_id_at_click = 0;
1211 click_outside_label = true;
1212 }
1213 /*Check if the click happened on the right side of the area outside the label*/
1214 else if(rel_pos.x >= label_width) {
1215 char_id_at_click = LV_TEXTAREA_CURSOR_LAST;
1216 click_outside_label = true;
1217 }
1218 else {
1219 char_id_at_click = lv_label_get_letter_on(ta->label, &rel_pos, true);
1220 click_outside_label = !lv_label_is_char_under_pos(ta->label, &rel_pos);
1221 }
1222
1223 if(ta->text_sel_en) {
1224 if(!ta->text_sel_in_prog && !click_outside_label && code == LV_EVENT_PRESSED) {
1225 /*Input device just went down. Store the selection start position*/
1226 ta->sel_start = char_id_at_click;
1227 ta->sel_end = LV_LABEL_TEXT_SELECTION_OFF;
1228 ta->text_sel_in_prog = 1;
1229 lv_obj_set_scroll_chain(obj, false);
1230 }
1231 else if(ta->text_sel_in_prog && code == LV_EVENT_PRESSING) {
1232 /*Input device may be moving. Store the end position*/
1233 ta->sel_end = char_id_at_click;
1234 }
1235 else if(ta->text_sel_in_prog && (code == LV_EVENT_PRESS_LOST || code == LV_EVENT_RELEASED)) {
1236 /*Input device is released. Check if anything was selected.*/
1237 lv_obj_set_scroll_chain(obj, true);
1238 }
1239 }
1240
1241 if(ta->text_sel_in_prog || code == LV_EVENT_PRESSED) lv_textarea_set_cursor_pos(obj, char_id_at_click);
1242
1243 if(ta->text_sel_in_prog) {
