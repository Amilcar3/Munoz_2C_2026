57
58 if(gLvEfiBS->AllocatePool(EfiBootServicesData, size_with_header, (void **)&ptr) != EFI_SUCCESS) return NULL;

65void * lv_realloc_core(void * p, size_t new_size)
66{
67 mem_header_t * p_header = NULL;
68 uintptr_t p_address = (uintptr_t)p;
69 void * p_new = NULL;
70
71 if(p == NULL) return lv_malloc_core(new_size);
72 // Check for invalid pointers
73 if(p_address < sizeof(mem_header_t)) return NULL;
74
75 p_address -= sizeof(mem_header_t);
76 p_header = (mem_header_t *) p_address;
77
78 // UEFI supports no realloc, if the size grows a new memory block has to be allocated
79 if(p_header->size > new_size) return p;
80
81 p_new = lv_malloc_core(new_size);
82 lv_memcpy(p_new, p, p_header->size);
83 lv_free_core(p);
84
85 return p_new;
86}
