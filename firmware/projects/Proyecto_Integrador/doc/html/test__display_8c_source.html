144static void test_matrix_transform_area(lv_display_t * display)
145{
146 lv_area_t ori_area = { 0 };
147 lv_area_set_width(&ori_area, 100);
148 lv_area_set_height(&ori_area, 100);
149
150 lv_matrix_t matrix;
151 lv_matrix_identity(&matrix);
152
153 const int32_t hor_res = lv_display_get_original_horizontal_resolution(display);
154 const int32_t ver_res = lv_display_get_original_vertical_resolution(display);
155
156 switch(lv_display_get_rotation(display)) {
157 case LV_DISPLAY_ROTATION_0:
158 break;
159 case LV_DISPLAY_ROTATION_90:
160 lv_matrix_rotate(&matrix, 270);
161 lv_matrix_translate(&matrix, -ver_res, 0);
162 break;
163 case LV_DISPLAY_ROTATION_180:
164 lv_matrix_rotate(&matrix, 180);
165 lv_matrix_translate(&matrix, -hor_res, -ver_res);
166 break;
167 case LV_DISPLAY_ROTATION_270:
168 lv_matrix_rotate(&matrix, 90);
169 lv_matrix_translate(&matrix, 0, -hor_res);
170 break;
171 }
172
173 lv_area_t disp_area = ori_area;
174 lv_display_rotate_area(display, &disp_area);
175
176 lv_area_t matrix_area = lv_matrix_transform_area(&matrix, &ori_area);
177
178 TEST_ASSERT_TRUE(lv_area_is_equal(&disp_area, &matrix_area));
179}

300static void test_display_resolution_full_rotation(
301 lv_display_t * disp,
302 int32_t ori_hor_res, int32_t ori_ver_res,
303 int32_t ori_physical_hor_res, int32_t ori_physical_ver_res,
304 int32_t ori_offset_x, int32_t ori_offset_y)
305{
306 const lv_display_rotation_t rotations[] = {
307 LV_DISPLAY_ROTATION_0,
308 LV_DISPLAY_ROTATION_90,
309 LV_DISPLAY_ROTATION_180,
310 LV_DISPLAY_ROTATION_270,
311 LV_DISPLAY_ROTATION_0,
312 };
313
314 for(size_t i = 0; i < sizeof(rotations) / sizeof(rotations[0]); i++) {
315 lv_display_set_rotation(disp, rotations[i]);
316
317 if(!disp && !lv_display_get_default()) {
318 TEST_ASSERT_EQUAL(LV_DISPLAY_ROTATION_0, lv_display_get_rotation(disp));
319 }
320 else {
321 TEST_ASSERT_EQUAL(rotations[i], lv_display_get_rotation(disp));
322 }
323
324 /* Original resolution should be the same */
325 TEST_ASSERT_EQUAL_INT32(ori_hor_res, lv_display_get_original_horizontal_resolution(disp));
326 TEST_ASSERT_EQUAL_INT32(ori_ver_res, lv_display_get_original_vertical_resolution(disp));
327
328 /* verify resolution */
329 switch(lv_display_get_rotation(disp)) {
330 case LV_DISPLAY_ROTATION_0:
331 case LV_DISPLAY_ROTATION_180:
332 TEST_ASSERT_EQUAL_INT32(ori_physical_hor_res, lv_display_get_physical_horizontal_resolution(disp));
333 TEST_ASSERT_EQUAL_INT32(ori_physical_ver_res, lv_display_get_physical_vertical_resolution(disp));
334 TEST_ASSERT_EQUAL_INT32(ori_hor_res, lv_display_get_horizontal_resolution(disp));
335 TEST_ASSERT_EQUAL_INT32(ori_ver_res, lv_display_get_vertical_resolution(disp));
336 break;
337 case LV_DISPLAY_ROTATION_90:
338 case LV_DISPLAY_ROTATION_270:
339 TEST_ASSERT_EQUAL_INT32(ori_physical_ver_res, lv_display_get_physical_horizontal_resolution(disp));
340 TEST_ASSERT_EQUAL_INT32(ori_physical_hor_res, lv_display_get_physical_vertical_resolution(disp));
341 TEST_ASSERT_EQUAL_INT32(ori_ver_res, lv_display_get_horizontal_resolution(disp));
342 TEST_ASSERT_EQUAL_INT32(ori_hor_res, lv_display_get_vertical_resolution(disp));
343 break;
344 default:
345 TEST_FAIL();
346 break;
347 }
348
349 /* verify offset */
350 switch(lv_display_get_rotation(disp)) {
351 case LV_DISPLAY_ROTATION_0:
352 TEST_ASSERT_EQUAL_INT32(ori_offset_x, lv_display_get_offset_x(disp));
353 TEST_ASSERT_EQUAL_INT32(ori_offset_y, lv_display_get_offset_y(disp));
354 break;
355 case LV_DISPLAY_ROTATION_180:
356 TEST_ASSERT_EQUAL_INT32(ori_physical_hor_res - ori_offset_x, lv_display_get_offset_x(disp));
357 TEST_ASSERT_EQUAL_INT32(ori_physical_ver_res - ori_offset_y, lv_display_get_offset_y(disp));
358 break;
359 case LV_DISPLAY_ROTATION_90:
360 TEST_ASSERT_EQUAL_INT32(ori_offset_y, lv_display_get_offset_x(disp));
361 TEST_ASSERT_EQUAL_INT32(ori_offset_x, lv_display_get_offset_y(disp));
362 break;
363 case LV_DISPLAY_ROTATION_270:
364 TEST_ASSERT_EQUAL_INT32(ori_physical_ver_res - ori_offset_y, lv_display_get_offset_x(disp));
365 TEST_ASSERT_EQUAL_INT32(ori_physical_hor_res - ori_offset_x, lv_display_get_offset_y(disp));
366 break;
367 default:
368 TEST_FAIL();
369 break;
370 }
371 }
