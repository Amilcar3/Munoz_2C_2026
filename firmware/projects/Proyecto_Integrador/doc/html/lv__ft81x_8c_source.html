89lv_display_t * lv_ft81x_create(const lv_ft81x_parameters_t * params, void * partial_buf, uint32_t buf_size,
90 lv_ft81x_spi_cb_t spi_cb, void * user_data)
91{
92 LV_CHECK_ARG(spi_cb != NULL, return NULL);
93 LV_CHECK_ARG(params != NULL, return NULL);
94 LV_CHECK_ARG(partial_buf != NULL, return NULL);
95
96 lv_display_t * disp = lv_display_create(params->hor_res, params->ver_res);
97
98 lv_ft81x_driver_data_t * drv = lv_malloc_zeroed(sizeof(lv_ft81x_driver_data_t));
99 LV_ASSERT_MALLOC(drv);
100 drv->spi_cb = spi_cb;
101 drv->user_data = user_data;
102 drv->cmd_offset = 0;
103 lv_display_set_driver_data(disp, drv);
104 lv_display_set_flush_cb(disp, flush_cb);
105 lv_display_set_color_format(disp, LV_COLOR_FORMAT_RGB565);
106 lv_display_set_buffers(disp, partial_buf, NULL, buf_size, LV_DISPLAY_RENDER_MODE_PARTIAL);
107 lv_display_add_event_cb(disp, delete_cb, LV_EVENT_DELETE, NULL);
108
109 lv_result_t init_res = initialize(disp, params);
110 if(init_res != LV_RESULT_OK) {
111 lv_display_delete(disp);
112 return NULL;
113 }
114
115 return disp;
116}

130static lv_result_t initialize(lv_display_t * disp, const lv_ft81x_parameters_t * params)
131{
132 lv_ft81x_cmd(disp, params->has_crystal ? EVE_CLKEXT : EVE_CLKINT, 0);
133 if(params->is_bt81x) lv_ft81x_cmd(disp, EVE_CLKSEL, 0x46);
134 lv_ft81x_cmd(disp, EVE_ACTIVE, 0);
135
136 /* at least 40 ms is needed for EVE to become ready. */
137 lv_delay_ms(40);
138
139 uint32_t start_millis = lv_tick_get();
140 while(lv_ft81x_read_8(disp, REG_ID) != 0x7c) {
141 if(lv_tick_get() - start_millis > 1000) {
142 LV_LOG_ERROR("failed to read 0x7C from the ID register 1000 ms after activation.");
143 return LV_RESULT_INVALID;
144 }
145 lv_delay_ms(1);
146 };
147
148 start_millis = lv_tick_get();
149 while(lv_ft81x_read_8(disp, REG_CPURESET) & 0x03) {
150 if(lv_tick_get() - start_millis > 1000) {
151 LV_LOG_ERROR("CPURESET register coprocessor and touch engines not in \"working status\" 1000 ms after activation.");
152 return LV_RESULT_INVALID;
153 }
154 lv_delay_ms(1);
155 };
156
157 if(params->is_bt81x) lv_ft81x_write_32(disp, REG_FREQUENCY, 72000000);
158
159 lv_ft81x_write_8(disp, REG_PWM_DUTY, PWM_DUTY_BACKLIGHT_ON);
160
161 lv_ft81x_write_16(disp, REG_HSIZE, params->hor_res); /* active display width */
162 lv_ft81x_write_16(disp, REG_HCYCLE, params->hcycle); /* total number of clocks per line, incl front/back porch */
163 lv_ft81x_write_16(disp, REG_HOFFSET, params->hoffset); /* start of active line */
164 lv_ft81x_write_16(disp, REG_HSYNC0, params->hsync0); /* start of horizontal sync pulse */
165 lv_ft81x_write_16(disp, REG_HSYNC1, params->hsync1); /* end of horizontal sync pulse */
166 lv_ft81x_write_16(disp, REG_VSIZE, params->ver_res); /* active display height */
167 lv_ft81x_write_16(disp, REG_VCYCLE, params->vcycle); /* total number of lines per screen, including pre/post */
168 lv_ft81x_write_16(disp, REG_VOFFSET, params->voffset); /* start of active screen */
169 lv_ft81x_write_16(disp, REG_VSYNC0, params->vsync0); /* start of vertical sync pulse */
170 lv_ft81x_write_16(disp, REG_VSYNC1, params->vsync1); /* end of vertical sync pulse */
171 lv_ft81x_write_8(disp, REG_SWIZZLE, params->swizzle); /* FT8xx output to LCD - pin order */
172 lv_ft81x_write_8(disp, REG_PCLK_POL, params->pclkpol); /* LCD data is clocked in on this PCLK edge */
173 lv_ft81x_write_8(disp, REG_CSPREAD,
174 params->cspread); /* helps with noise, when set to 1 fewer signals are changed simultaneously, reset-default: 1 */
175
176 /* write a basic display-list to get things started */
177 lv_ft81x_write_32(disp, EVE_RAM_DL, DL_CLEAR_RGB);
178 lv_ft81x_write_32(disp, EVE_RAM_DL + 4, (DL_CLEAR | CLR_COL | CLR_STN | CLR_TAG));
179 lv_ft81x_write_32(disp, EVE_RAM_DL + 8, DL_DISPLAY); /* end of display list */
180 lv_ft81x_write_32(disp, REG_DLSWAP, EVE_DLSWAP_FRAME);
181
182 /* nothing is being displayed yet... the pixel clock is still 0x00 */
183 lv_ft81x_write_8(disp, REG_GPIO,
184 0x80); /* enable the DISP signal to the LCD panel, it is set to output in REG_GPIO_DIR by default */
185 lv_ft81x_write_8(disp, REG_PCLK, params->pclk); /* now start clocking data to the LCD panel */
186
187 LV_ASSERT(lv_ft81x_read_16(disp, REG_CMD_READ) != 0xfff);
188
189 LV_ASSERT(0 == lv_ft81x_read_16(disp, REG_CMD_WRITE));
190
191
192 lv_ft81x_cmd_list_t cmd_list;
193 lv_ft81x_cmd_list_init(disp, &cmd_list);
194 lv_ft81x_cmd_list_add_32(disp, &cmd_list, CMD_MEMSET);
195 lv_ft81x_cmd_list_add_32(disp, &cmd_list, 0); /* address */
196 lv_ft81x_cmd_list_add_32(disp, &cmd_list, 0x00); /* val */
197 lv_ft81x_cmd_list_add_32(disp, &cmd_list, 2 * params->hor_res * params->ver_res); /* count */
198 lv_ft81x_cmd_list_send(disp, &cmd_list);
199
200 lv_ft81x_cmd_list_init(disp, &cmd_list);
201
202 lv_ft81x_cmd_list_add_32(disp, &cmd_list, CMD_DLSTART);
203
204 lv_ft81x_cmd_list_add_32(disp, &cmd_list, DL_CLEAR_RGB | 0); /* clear to black */
205 lv_ft81x_cmd_list_add_32(disp, &cmd_list, DL_CLEAR | CLR_COL | CLR_STN | CLR_TAG);
206
207 lv_ft81x_cmd_list_add_32(disp, &cmd_list, TAG(0));
208
209 lv_ft81x_cmd_list_add_32(disp, &cmd_list, TAG(20));
210 lv_ft81x_cmd_list_add_32(disp, &cmd_list, CMD_SETBITMAP);
211 lv_ft81x_cmd_list_add_32(disp, &cmd_list, 0); /* address */
212 lv_ft81x_cmd_list_add_16(disp, &cmd_list, EVE_RGB565);
213 lv_ft81x_cmd_list_add_16(disp, &cmd_list, params->hor_res);
214 lv_ft81x_cmd_list_add_16(disp, &cmd_list, params->ver_res);
215 lv_ft81x_cmd_list_add_16(disp, &cmd_list, 0);
216
217 lv_ft81x_cmd_list_add_32(disp, &cmd_list, DL_BEGIN | EVE_BITMAPS);
218 lv_ft81x_cmd_list_add_32(disp, &cmd_list, VERTEX2F(0, 0));
219 lv_ft81x_cmd_list_add_32(disp, &cmd_list, DL_END);
220
221 lv_ft81x_cmd_list_add_32(disp, &cmd_list, TAG(0));
222
223 lv_ft81x_cmd_list_add_32(disp, &cmd_list, DL_DISPLAY);
224
225 lv_ft81x_cmd_list_add_32(disp, &cmd_list, CMD_SWAP);
226
227 lv_ft81x_cmd_list_send(disp, &cmd_list);
228
229
230 return LV_RESULT_OK;
231}

233static void flush_cb(lv_display_t * disp, const lv_area_t * area, uint8_t * px_map)
234{
235 lv_ft81x_driver_data_t * drv = lv_display_get_driver_data(disp);
236
237 if(drv->spi_cb == NULL) {
238 LV_LOG_ERROR("The SPI callback is NULL.");
239 return;
240 }
241
242 int32_t hor_res = lv_display_get_horizontal_resolution(disp);
243 uint32_t disp_row_bytes = hor_res * 2;
244
245 uint32_t address = disp_row_bytes * area->y1 + area->x1 * 2;
246 uint8_t encoded_address[3];
247
248 if(lv_area_get_width(area) == hor_res) {
249 lv_ft81x_encode_write_address(encoded_address, address);
250 drv->spi_cb(disp, LV_FT81X_SPI_OPERATION_CS_ASSERT, NULL, 0);
251 drv->spi_cb(disp, LV_FT81X_SPI_OPERATION_SEND, encoded_address, sizeof(encoded_address));
252 drv->spi_cb(disp, LV_FT81X_SPI_OPERATION_SEND, px_map, lv_area_get_size(area) * 2);
253 drv->spi_cb(disp, LV_FT81X_SPI_OPERATION_CS_DEASSERT, NULL, 0);
254 }
255 else {
256 uint32_t area_height = lv_area_get_height(area);
257 uint32_t area_row_bytes = lv_area_get_width(area) * 2;
258 for(uint32_t i = 0; i < area_height; i++) {
259 lv_ft81x_encode_write_address(encoded_address, address);
260 drv->spi_cb(disp, LV_FT81X_SPI_OPERATION_CS_ASSERT, NULL, 0);
261 drv->spi_cb(disp, LV_FT81X_SPI_OPERATION_SEND, encoded_address, sizeof(encoded_address));
262 drv->spi_cb(disp, LV_FT81X_SPI_OPERATION_SEND, px_map, area_row_bytes);
263 drv->spi_cb(disp, LV_FT81X_SPI_OPERATION_CS_DEASSERT, NULL, 0);
264 address += disp_row_bytes;
265 px_map += area_row_bytes;
266 }
267 }
268
269 lv_display_flush_ready(disp);
270}
