79void lv_opengl_shader_manager_init(lv_opengl_shader_manager_t * manager, const lv_opengl_shader_t * sources,
80 size_t len, const char * vert_src,
81 const char * frag_src)
82{
83
84 manager->sources_map = create_shader_map(sources, len);
85 if(vert_src != NULL) {
86 lv_opengl_shader_source_t entry = { { "__MAIN__.vert", lv_strdup(vert_src) }, 1 };
87 lv_rb_node_t * node = lv_rb_insert(&manager->sources_map, &entry);
88 LV_ASSERT_MSG(node,
89 "Failed to insert shader source to source map");
90 lv_memcpy(node->data, &entry, sizeof(entry));
91 }
92 if(frag_src != NULL) {
93 lv_opengl_shader_source_t entry = { { "__MAIN__.frag", lv_strdup(frag_src) }, 1 };
94 lv_rb_node_t * node = lv_rb_insert(&manager->sources_map, &entry);
95 LV_ASSERT_MSG(node, "Failed to insert shader to shader map");
96 lv_memcpy(node->data, &entry, sizeof(entry));
97 }
98
99 lv_rb_init(&manager->compiled_shaders_map,
100 (lv_rb_compare_t)compiled_shader_compare_cb,
101 sizeof(lv_opengl_compiled_shader_t));
102
103 /* Textures and compiled shaders share the same compare function */
104 lv_rb_init(&manager->textures_map,
105 (lv_rb_compare_t)compiled_shader_compare_cb,
106 sizeof(lv_opengl_shader_texture_t));
107
108 lv_rb_init(&manager->programs_map,
109 (lv_rb_compare_t)shader_program_compare_cb,
110 sizeof(lv_opengl_program_map_key_t));
111
112 manager->bg_index_buf = 0;
113 manager->bg_vertex_buf = 0;
114 manager->bg_vao = 0;
115 manager->bg_program = 0;
116}

118uint32_t lv_opengl_shader_hash(const char * value)
119{
120 uint32_t hash = 0;
121 const size_t len = lv_strlen(value);
122 if(len == 0) {
123 return hash;
124 }
125 for(size_t i = 0; i < len; i++) {
126 hash = ((hash << 5) - hash) + value[i];
127 }
128 return hash;
129}

159lv_result_t lv_opengl_shader_manager_select_shader(lv_opengl_shader_manager_t * shader, const char * shader_identifier,
160 const lv_opengl_shader_define_t * permutations, size_t permutations_len,
161 lv_opengl_glsl_version_t glsl_version, uint32_t * out_hash)
162{
163 /* First check that the shader identifier exists */
164 lv_opengl_shader_t key = { shader_identifier, NULL };
165 lv_rb_node_t * source_node = lv_rb_find(&shader->sources_map, &key);
166 LV_LOG_TRACE("Select shader '%s'", shader_identifier);
167
168 if(!source_node) {
169 LV_LOG_WARN("Couldn't find shader %s", shader_identifier);
170 return LV_RESULT_INVALID;
171 }
172
173 /* Then hash the name with the permutations and see if we already compiled it */
174 char define[512];
175 uint32_t hash = lv_opengl_shader_hash(shader_identifier);
176 for(size_t i = 0; i < permutations_len; ++i) {
177 LV_ASSERT(permutations[i].name != NULL);
178 if(permutations[i].value) {
179 lv_snprintf(define, sizeof(define), "%s%s", permutations[i].name, permutations[i].value);
180 }
181 else {
182 lv_snprintf(define, sizeof(define), "%s", permutations[i].name);
183 }
184 hash ^= lv_opengl_shader_hash(define);
185 }
186 /* hash the version so that the same shader with different versions produces a different hash*/
187 hash ^= lv_opengl_shader_hash(lv_opengles_glsl_version_to_string(glsl_version));
188
189 lv_opengl_compiled_shader_t shader_map_key = { hash, 0 };
190 lv_rb_node_t * shader_map_node =
191 lv_rb_find(&shader->compiled_shaders_map, &shader_map_key);
192
193 /* Fast path. Shader already compiled */
194 if(shader_map_node != NULL) {
195 LV_LOG_INFO("Shader '%s' with hash %u found. Id: %d", shader_identifier, hash,
196 ((lv_opengl_compiled_shader_t *)shader_map_node->data)->id);
197 *out_hash = hash;
198 return LV_RESULT_OK;
199 }
200
201 /* New shader requested, construct and compile it */
202 bool is_vertex = string_ends_with(shader_identifier, ".vert");
203 const char * original_shader_source = ((lv_opengl_shader_source_t *)source_node->data)->data.source;
204
205 char * shader_source = construct_shader(original_shader_source,
206 permutations, permutations_len, glsl_version);
207
208 shader_map_key.id = compile_shader(shader_source, is_vertex);
209 lv_free(shader_source);
210
211 if(!is_shader_compiled(shader_map_key.id)) {
212 GLchar info_log[512];
213 GL_CALL(glGetShaderInfoLog(shader_map_key.id, sizeof(info_log), NULL, info_log));
214 LV_LOG_WARN("Failed to compile shader for glsl version '%s': %s", lv_opengles_glsl_version_to_string(glsl_version),
215 info_log);
216 GL_CALL(glDeleteShader(shader_map_key.id));
217 return LV_RESULT_INVALID;
218 }
219
220 LV_LOG_TRACE("Compiled %s shader %s to %d Hash %u", is_vertex ? "V" : "F", shader_identifier, shader_map_key.id, hash);
221 lv_rb_node_t * node = lv_rb_insert(&shader->compiled_shaders_map, &shader_map_key);
222 LV_ASSERT_MSG(node, "Failed to insert shader to shader map");
223 lv_memcpy(node->data, &shader_map_key, sizeof(shader_map_key));
224
225 *out_hash = hash;
226
227 return LV_RESULT_OK;
228}

230lv_opengl_shader_program_t *
231lv_opengl_shader_manager_get_program(lv_opengl_shader_manager_t * manager,
232 uint32_t fragment_shader_hash,
233 uint32_t vertex_shader_hash)
234{
235 lv_opengl_program_map_key_t key = {
236 .vertex_shader_hash = vertex_shader_hash,
237 .fragment_shader_hash = fragment_shader_hash
238 };
239
240 lv_rb_node_t * node = lv_rb_find(&manager->programs_map, &key);
241 if(node) {
242 return ((lv_opengl_program_map_key_t *)node->data)->program;
243 }
244
245 lv_opengl_compiled_shader_t shader_key = { .hash = vertex_shader_hash };
246 lv_rb_node_t * vertex_node =
247 lv_rb_find(&manager->compiled_shaders_map, &shader_key);
248 shader_key.hash = fragment_shader_hash;
249 lv_rb_node_t * fragment_node =
250 lv_rb_find(&manager->compiled_shaders_map, &shader_key);
251
252 LV_ASSERT_FORMAT_MSG(
253 vertex_node,
254 "Unable to find to find vertex shader with hash %d",
255 vertex_shader_hash);
256 LV_ASSERT_FORMAT_MSG(
257 fragment_node,
258 "Unable to find to find fragment shader with hash %d",
259 fragment_shader_hash);
260
261 const GLuint vertex_shader_id =
262 ((lv_opengl_compiled_shader_t *)vertex_node->data)->id;
263 const GLuint fragment_shader_id =
264 ((lv_opengl_compiled_shader_t *)fragment_node->data)->id;
265
266 GLuint program_id = link_program(vertex_shader_id, fragment_shader_id);
267 bool is_linked = is_program_linked(program_id);
268 if(!is_linked) {
269 GLchar info_log[512];
270 GL_CALL(glGetProgramInfoLog(program_id, sizeof(info_log), NULL,
271 info_log));
272 LV_LOG_WARN("Failed to link program: %s", info_log);
273 GL_CALL(glDeleteProgram(program_id));
274 return NULL;
275 }
276 LV_LOG_TRACE("Linking program with shaders V: %d F:%d P: %d", vertex_shader_id, fragment_shader_id, program_id);
277
278 lv_opengl_shader_program_t * program =
279 lv_opengl_shader_program_create(program_id);
280
281 LV_ASSERT_MSG(program, "Failed to create program");
282
283 lv_opengl_program_map_key_t prog_key = {
284 .program = program,
285 .fragment_shader_hash = fragment_shader_hash,
286 .vertex_shader_hash = vertex_shader_hash
287 };
288 node = lv_rb_insert(&manager->programs_map, &prog_key);
289 LV_ASSERT_MSG(node, "Failed to store program in cache");
290 lv_memcpy(node->data, &prog_key, sizeof(prog_key));
291 return program;
292}

294lv_opengl_shader_program_t * lv_opengl_shader_manager_compile_program(lv_opengl_shader_manager_t * manager,
295 const lv_opengl_shader_params_t * frag_shader,
296 const lv_opengl_shader_params_t * vert_shader,
297 lv_opengl_glsl_version_t version)
298{
299 uint32_t frag_shader_hash;
300 uint32_t vert_shader_hash;
301
302 lv_result_t res = lv_opengl_shader_manager_select_shader(manager, frag_shader->name, frag_shader->permutations,
303 frag_shader->permutations_len,
304 version, &frag_shader_hash);
305 if(res != LV_RESULT_OK) {
306 LV_LOG_WARN("Failed to compile shader for glsl version %s", lv_opengles_glsl_version_to_string(version));
307 return NULL;
308 }
309
310 res = lv_opengl_shader_manager_select_shader(manager, vert_shader->name, vert_shader->permutations,
311 vert_shader->permutations_len,
312 version, &vert_shader_hash);
313 if(res != LV_RESULT_OK) {
314 LV_LOG_WARN("Failed to compile shader for glsl version %s", lv_opengles_glsl_version_to_string(version));
315 return NULL;
316 }
317 lv_opengl_shader_program_t * program =
318 lv_opengl_shader_manager_get_program(manager, frag_shader_hash, vert_shader_hash);
319
320 if(!program) {
321 LV_LOG_WARN("Failed to link program for glsl version %s", lv_opengles_glsl_version_to_string(version));
322 }
323
324 return program;
325}

343void lv_opengl_shader_manager_deinit(lv_opengl_shader_manager_t * manager)
344{
345 LV_LOG_INFO("Destroying shader cache");
346
347 lv_rb_destroy(&manager->textures_map);
348
349 lv_rb_node_t * node;
350
351 while((node = manager->sources_map.root)) {
352 lv_opengl_shader_source_t * shader = node->data;
353 if(shader->src_allocated) {
354 lv_free((void *)shader->data.source);
355 }
356 lv_rb_drop_node(&manager->sources_map, node);
357 }
358 lv_rb_destroy(&manager->sources_map);
359
360 while((node = manager->compiled_shaders_map.root)) {
361 lv_opengl_compiled_shader_t * shader = node->data;
362 GL_CALL(glDeleteShader(shader->id));
363 lv_rb_drop_node(&manager->compiled_shaders_map, node);
364 }
365
366 lv_rb_destroy(&manager->compiled_shaders_map);
367 while((node = manager->programs_map.root)) {
368 lv_opengl_program_map_key_t * program_key = node->data;
369 lv_opengl_shader_program_destroy(program_key->program);
370 lv_rb_drop_node(&manager->programs_map, node);
371 }
372 lv_rb_destroy(&manager->programs_map);
373
374}

393char * lv_opengl_shader_manager_process_includes(const char * c_src, const lv_opengl_shader_t * src_includes,
394 size_t num_items)
395{
396 if(!c_src || !src_includes) {
397 return NULL;
398 }
399
400 char * rep = lv_strdup(c_src);
401 if(!rep) {
402 return NULL;
403 }
404 char search_str[255];
405
406 for(size_t i = 0; i < num_items; i++) {
407 lv_snprintf(search_str, sizeof(search_str), "\n#include <%s>", src_includes[i].name);
408
409 char * new_rep = replace_word(rep, search_str, src_includes[i].source);
410 lv_free(rep);
411 if(!new_rep) {
412 return NULL;
413 }
414 rep = new_rep;
415 }
416
417 return rep;
418}

424static char * replace_word(const char * source, const char * f, const char * r)
425{
426 if(!source || !f || !r || strlen(f) == 0 || strcmp(f, r) == 0 || !strstr(source, f)) {
427 return lv_strdup(source);
428 }
429
430 size_t s_len = strlen(source);
431 size_t f_len = strlen(f);
432 size_t r_len = strlen(r);
433
434 size_t count = 0;
435 const char * temp = source;
436 while((temp = strstr(temp, f)) != NULL) {
437 count++;
438 temp += f_len;
439 }
440
441 size_t new_size = s_len + count * (r_len - f_len) + 1;
442 char * result = lv_malloc(new_size);
443 LV_ASSERT_MALLOC(result);
444
445 char * dest = result;
446 const char * src = source;
447 const char * pos;
448
449 while((pos = strstr(src, f)) != NULL) {
450 size_t prefix_len = pos - src;
451 memcpy(dest, src, prefix_len);
452 dest += prefix_len;
453
454 memcpy(dest, r, r_len);
455 dest += r_len;
456
457 src = pos + f_len;
458 }
459
460 strcpy(dest, src);
461
462 return result;
463}

502static GLuint compile_shader(const char * shader_source, bool is_vertex_shader)
503{
504 GLuint shader_id;
505 if(is_vertex_shader) {
506 GL_CALL(shader_id = glCreateShader(GL_VERTEX_SHADER));
507 }
508 else {
509 GL_CALL(shader_id = glCreateShader(GL_FRAGMENT_SHADER));
510 }

515
516 return shader_id;

537static char * construct_shader(const char * source,
538 const lv_opengl_shader_define_t * permutations,
539 size_t permutations_len, lv_opengl_glsl_version_t glsl_version)
540{
541 const char * defines = lv_opengles_glsl_version_to_string(glsl_version);
542 const char * prefix = "#define ";
543
544 const size_t prefix_len = lv_strlen(prefix);
545 size_t shader_source_size = lv_strlen(defines) + lv_strlen(source);
546
547 /* First calculate the necessary size */
548 for(size_t i = 0; i < permutations_len; ++i) {
549 shader_source_size += prefix_len;
550 if(!permutations[i].name) {
551 LV_LOG_WARN("Name is NULL for permutation # %zu", i);
552 continue;
553 }
554 shader_source_size +=
555 lv_strlen(permutations[i].name) + 1; /* ' ' */
556 if(permutations[i].value) {
557 shader_source_size +=
558 lv_strlen(permutations[i].value) + 1; /* '\n' */
559 }
560 }
561
562 /* Allocate enough for memory with calculated size*/
563 char * result = (char *)lv_malloc(shader_source_size + 1);
564 if(!result) {
565 LV_LOG_ERROR(
566 "Failed to allocate enough space for shader changes");
567 return 0;
568 }
569
570 /* Construct shader */
571 size_t curr_index = 0;
572 append_to_shader(result, defines, &curr_index);
573 for(size_t i = 0; i < permutations_len; ++i) {
574 if(!permutations[i].name) {
575 continue;
576 }
577 append_to_shader(result, prefix, &curr_index);
578 append_to_shader(result, permutations[i].name, &curr_index);
579 if(permutations[i].value) {
580 result[curr_index++] = ' ';
581 append_to_shader(result, permutations[i].value,
582 &curr_index);
583 }
584 result[curr_index++] = '\n';
585 }
586 append_to_shader(result, source, &curr_index);
587 LV_ASSERT(curr_index == shader_source_size);
588 result[shader_source_size] = '\0';
589 return result;
590}

592static char * replace_include(const char * source, const char * pattern,
593 const char * replacement)
594{
595 const char * pos = strstr(source, pattern);
596 LV_ASSERT(pos);
597
598 const size_t source_len = lv_strlen(source);
599 const size_t pattern_len = lv_strlen(pattern);
600 const size_t replacement_len = lv_strlen(replacement);
601
602 const size_t new_len = source_len - pattern_len + replacement_len;
603
604 char * result = (char *)lv_malloc(new_len + 1);
605 if(!result) {
606 return NULL;
607 }
608
609 const size_t before_len = pos - source;
610 lv_memcpy(result, source, before_len);
611 lv_memcpy(result + before_len, replacement, replacement_len);
612 lv_strcpy(result + before_len + replacement_len, pos + pattern_len);
613
614 /* Replace other patterns with whitespaces */
615 while((pos = strstr(result, pattern))) {
616 lv_memset((void *)pos, ' ', pattern_len);
617 }
618 result[new_len] = '\0';
619
620 return result;

622static lv_rb_t create_shader_map(const lv_opengl_shader_t * shaders, size_t len)
623{
624 lv_rb_t map;
625 lv_rb_init(&map, (lv_rb_compare_t)shader_source_compare_cb,
626 sizeof(lv_opengl_shader_source_t));
627
628 char pattern[256];
629 for(size_t i = 0; i < len; i++) {
630 if(strlen(shaders[i].source) == 0) {
631 LV_LOG_WARN("Shader %s at index %zu is empty\n",
632 shaders[i].name, i);
633 continue;
634 }
635 lv_opengl_shader_source_t value = {
636 .data = {
637 .name = shaders[i].name,
638 .source = shaders[i].source
639 },
640 .src_allocated = false
641 };
642
643 for(size_t j = 0; j < len; j++) {
644 const char * source = value.data.source;
645 const char * include_name = shaders[j].name;
646 const char * include_source = shaders[j].source;
647 const size_t include_name_len = strlen(include_name);
648 const size_t include_source_len =
649 strlen(include_source);
650 if(include_name_len == 0 || include_source_len == 0) {
651 continue;
652 }
653 lv_snprintf(pattern, sizeof(pattern), "#include <%s>",
654 include_name);
655 const char * include_pattern = strstr(source, pattern);
656 if(include_pattern == NULL) {
657 continue;
658 }
659 LV_LOG_TRACE("Replacing %s", pattern);
660 char * new_source = replace_include(source, pattern,
661 include_source);
662 LV_ASSERT_MSG(
663 new_source,
664 "Failed to allocate memory to replace shader include with source code");
665
666 if(value.src_allocated) {
667 lv_free((void *)value.data.source);
668 }
669 value.data.source = new_source;
670 value.src_allocated = true;
671 }
672 if(strstr(value.data.source, "#include")) {
673 LV_LOG_ERROR(
674 "Couldn't replace every include in shader %s. Last result:\n%s",
675 value.data.name, value.data.source);
676 LV_ASSERT_MSG(
677 0,
678 "Refusing to continue execution with incorrect shaders");
679 }
680
681 lv_rb_node_t * node = lv_rb_insert(&map, &value);
682 LV_ASSERT_MSG(node,
683 "Failed to allocate memory for shader map entry");
684 lv_memcpy(node->data, &value, sizeof(value));
685 }
686 return map;
687}
