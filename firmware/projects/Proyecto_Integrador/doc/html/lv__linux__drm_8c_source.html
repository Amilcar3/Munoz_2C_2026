133lv_display_t * lv_linux_drm_create(void)
134{
135 lv_display_t * disp;
136
137 lv_tick_set_cb(tick_get_cb);
138
139 drm_dev_t * drm_dev = lv_malloc_zeroed(sizeof(drm_dev_t));
140 LV_ASSERT_MALLOC(drm_dev);
141 if(drm_dev == NULL) return NULL;
142
143 drm_dev->fd = -1;
144
145 disp = lv_display_create(800, 480);
146 if(disp == NULL) {
147 lv_free(drm_dev);
148 return NULL;
149 }
150 lv_display_set_driver_data(disp, drm_dev);
151 lv_display_set_flush_wait_cb(disp, drm_flush_wait);
152 lv_display_set_flush_cb(disp, drm_flush);
153 lv_display_add_event_cb(disp, drm_del_event_cb, LV_EVENT_DELETE, NULL);
154
155 return disp;
156}

158/* Called by LVGL when there is something that needs redrawing
159 * it sets the active buffer. if GBM buffers are used, it issues a DMA_BUF_SYNC
160 * ioctl call to lock the buffer for CPU access, the buffer is unlocked just
161 * before the atomic commit */
162static void drm_dmabuf_set_active_buf(lv_event_t * event)
163{
164 drm_dev_t * drm_dev;
165 lv_display_t * disp;
166 lv_draw_buf_t * act_buf;
167 int i;
168
169 disp = (lv_display_t *) lv_event_get_current_target(event);
170 drm_dev = (drm_dev_t *) lv_display_get_driver_data(disp);
171 act_buf = lv_display_get_buf_active(disp);
172
173 if(drm_dev->act_buf == NULL) {
174
175 for(i = 0; i < BUFFER_CNT; i++) {
176 if(act_buf->unaligned_data == drm_dev->drm_bufs[i].map) {
177 drm_dev->act_buf = &drm_dev->drm_bufs[i];
178 LV_LOG_TRACE("Set active buffer idx: %d", i);
179 break;
180 }
181 }
182
183#if LV_USE_LINUX_DRM_GBM_BUFFERS
184 struct dma_buf_sync sync_req;
185 sync_req.flags = DMA_BUF_SYNC_START | DMA_BUF_SYNC_RW;
186 int res;
187
188 if((res = ioctl(drm_dev->act_buf->handle, DMA_BUF_IOCTL_SYNC, &sync_req)) != 0) {
189 LV_LOG_ERROR("Failed to start DMA-BUF R/W SYNC res: %d", res);
190 }
191#endif
192
193 }
194 else {
195
196 LV_LOG_TRACE("active buffer already set");
197 }
198
199}

201lv_result_t lv_linux_drm_set_file(lv_display_t * disp, const char * file, int64_t connector_id)
202{
203 int ret;
204
205 LV_CHECK_ARG(disp != NULL, return LV_RESULT_INVALID);
206 LV_CHECK_ARG(file != NULL, return LV_RESULT_INVALID);
207 drm_dev_t * drm_dev = lv_display_get_driver_data(disp);
208
209 ret = drm_setup(drm_dev, file, connector_id, DRM_FOURCC);
210 if(ret) {
211 return LV_RESULT_INVALID;
212 }
213
214 int32_t hor_res = drm_dev->width;
215 int32_t ver_res = drm_dev->height;
216
217 ret = drm_setup_buffers(drm_dev);
218 if(ret) {
219 LV_LOG_ERROR("DRM buffer allocation failed");
220 close(drm_dev->fd);
221 drm_dev->fd = -1;
222 return LV_RESULT_INVALID;
223 }
224
225 LV_LOG_INFO("DRM subsystem and buffer mapped successfully");
226
227 int32_t width = drm_dev->mmWidth;
228
229 size_t buf_size = LV_MIN(drm_dev->drm_bufs[1].size, drm_dev->drm_bufs[0].size);
230 uint32_t stride = drm_dev->drm_bufs[0].pitch;
231 /* Resolution must be set first because if the screen is smaller than the size passed
232 * to lv_display_create then the buffers aren't big enough for LV_DISPLAY_RENDER_MODE_DIRECT.
233 */
234 lv_display_set_resolution(disp, hor_res, ver_res);
235 lv_display_set_buffers_with_stride(disp, drm_dev->drm_bufs[1].map, drm_dev->drm_bufs[0].map, buf_size,
236 stride, LV_DISPLAY_RENDER_MODE_DIRECT);
237
238
239 /* Set the handler that is called before a redraw occurs to set the active buffer/plane
240 * when GBM buffers are used the DMA_BUF_SYNC_START is issued there */
241 lv_display_add_event_cb(disp, drm_dmabuf_set_active_buf, LV_EVENT_REFR_START, drm_dev);
242
243 if(width) {
244 lv_display_set_dpi(disp, DIV_ROUND_UP(hor_res * 25400, width * 1000));
245 }
246
247 LV_LOG_INFO("Resolution is set to %" LV_PRId32 "x%" LV_PRId32 " at %" LV_PRId32 "dpi",
248 hor_res, ver_res, lv_display_get_dpi(disp));
249 return LV_RESULT_OK;
250}

442static int drm_dmabuf_set_plane(drm_dev_t * drm_dev, drm_buffer_t * buf)
443{
444 int ret;
445 static int first = 1;
446 uint32_t flags = DRM_MODE_PAGE_FLIP_EVENT | DRM_MODE_ATOMIC_NONBLOCK;
447
448#if LV_USE_LINUX_DRM_GBM_BUFFERS
449
450 struct dma_buf_sync sync_req;
451
452 sync_req.flags = DMA_BUF_SYNC_END | DMA_BUF_SYNC_RW;
453 if(ioctl(buf->handle, DMA_BUF_IOCTL_SYNC, &sync_req) != 0) {
454 LV_LOG_ERROR("Failed to end DMA-BUF R/W SYNC");
455 }
456
457#endif
458
459 drm_dev->req = drmModeAtomicAlloc();
460
461 /* On first Atomic commit, do a modeset */
462 if(first) {
463 drm_add_conn_property(drm_dev, "CRTC_ID", drm_dev->crtc_id);
464
465 drm_add_crtc_property(drm_dev, "MODE_ID", drm_dev->blob_id);
466 drm_add_crtc_property(drm_dev, "ACTIVE", 1);
467
468 flags |= DRM_MODE_ATOMIC_ALLOW_MODESET;
469
470 first = 0;
471 }
472
473 drm_add_plane_property(drm_dev, "FB_ID", buf->fb_handle);
474 drm_add_plane_property(drm_dev, "CRTC_ID", drm_dev->crtc_id);
475 drm_add_plane_property(drm_dev, "SRC_X", 0);
476 drm_add_plane_property(drm_dev, "SRC_Y", 0);
477 drm_add_plane_property(drm_dev, "SRC_W", drm_dev->width << 16);
478 drm_add_plane_property(drm_dev, "SRC_H", drm_dev->height << 16);
479 drm_add_plane_property(drm_dev, "CRTC_X", 0);
480 drm_add_plane_property(drm_dev, "CRTC_Y", 0);
481 drm_add_plane_property(drm_dev, "CRTC_W", drm_dev->width);
482 drm_add_plane_property(drm_dev, "CRTC_H", drm_dev->height);
483
484 ret = drmModeAtomicCommit(drm_dev->fd, drm_dev->req, flags, drm_dev);
485 if(ret) {
486 LV_LOG_ERROR("drmModeAtomicCommit failed: %s (%d)", strerror(errno), errno);
487 drmModeAtomicFree(drm_dev->req);
488 return ret;
489 }
490
491 return 0;
492}

494static int find_plane(drm_dev_t * drm_dev, unsigned int fourcc, uint32_t * plane_id, uint32_t crtc_id,
495 uint32_t crtc_idx)
496{
497 LV_UNUSED(crtc_id);
498 drmModePlaneResPtr planes;
499 drmModePlanePtr plane;
500 unsigned int i;
501 unsigned int j;
502 int ret = 0;
503
504 planes = drmModeGetPlaneResources(drm_dev->fd);
505 if(!planes) {
506 LV_LOG_ERROR("drmModeGetPlaneResources failed");
507 return -1;
508 }
509
510 LV_LOG_TRACE("drm: found planes %u", planes->count_planes);
511
512 for(i = 0; i < planes->count_planes; ++i) {
513 plane = drmModeGetPlane(drm_dev->fd, planes->planes[i]);
514 if(!plane) {
515 LV_LOG_ERROR("drmModeGetPlane failed: %s", strerror(errno));
516 ret = -1;
517 break;
518 }
519
520 if(!(plane->possible_crtcs & (1 << crtc_idx))) {
521 drmModeFreePlane(plane);
522 continue;
523 }
524
525 for(j = 0; j < plane->count_formats; ++j) {
526 if(plane->formats[j] == fourcc)
527 break;
528 }
529
530 if(j == plane->count_formats) {
531 drmModeFreePlane(plane);
532 continue;
533 }
534
535 *plane_id = plane->plane_id;
536 drmModeFreePlane(plane);
537
538 LV_LOG_TRACE("found plane %d", *plane_id);
539
540 /* Success */
541 goto out;
542 }
543
544 if(i == planes->count_planes)
545 ret = -1;
546out:
547 drmModeFreePlaneResources(planes);
548 return ret;
549}

551static int drm_find_connector(drm_dev_t * drm_dev, int64_t connector_id)
552{
553 drmModeConnector * conn = NULL;
554 drmModeEncoder * enc = NULL;
555 drmModeRes * res;
556 int i;
557 int ret = -1;
558
559 if((res = drmModeGetResources(drm_dev->fd)) == NULL) {
560 LV_LOG_ERROR("drmModeGetResources() failed");
561 goto free_res;
562 }
563
564 if(res->count_crtcs <= 0) {
565 LV_LOG_ERROR("no Crtcs");
566 goto free_res;
567 }
568
569 /* find all available connectors */
570 for(i = 0; i < res->count_connectors; i++) {
571 conn = drmModeGetConnector(drm_dev->fd, res->connectors[i]);
572 if(!conn)
573 continue;
574
575 if(connector_id >= 0 && conn->connector_id != connector_id) {
576 drmModeFreeConnector(conn);
577 continue;
578 }
579
580 if(conn->connection == DRM_MODE_CONNECTED) {
581 LV_LOG_TRACE("drm: connector %d: connected", conn->connector_id);
582 }
583 else if(conn->connection == DRM_MODE_DISCONNECTED) {
584 LV_LOG_TRACE("drm: connector %d: disconnected", conn->connector_id);
585 }
586 else if(conn->connection == DRM_MODE_UNKNOWNCONNECTION) {
587 LV_LOG_TRACE("drm: connector %d: unknownconnection", conn->connector_id);
588 }
589 else {
590 LV_LOG_TRACE("drm: connector %d: unknown", conn->connector_id);
591 }
592
593 if(conn->connection == DRM_MODE_CONNECTED && conn->count_modes > 0)
594 break;
595
596 drmModeFreeConnector(conn);
597 conn = NULL;
598 };
599
600 if(!conn) {
601 LV_LOG_ERROR("suitable connector not found");
602 goto free_res;
603 }
604
605 drm_dev->conn_id = conn->connector_id;
606 LV_LOG_TRACE("conn_id: %d", drm_dev->conn_id);
607 drm_dev->mmWidth = conn->mmWidth;
608 drm_dev->mmHeight = conn->mmHeight;
609
610 lv_memcpy(&drm_dev->mode, &conn->modes[0], sizeof(drmModeModeInfo));
611
612 if(drmModeCreatePropertyBlob(drm_dev->fd, &drm_dev->mode, sizeof(drm_dev->mode),
613 &drm_dev->blob_id)) {
614 LV_LOG_ERROR("error creating mode blob");
615 goto free_res;
616 }
617
618 drm_dev->width = conn->modes[0].hdisplay;
619 drm_dev->height = conn->modes[0].vdisplay;
620
621 for(i = 0 ; i < res->count_encoders; i++) {
622 enc = drmModeGetEncoder(drm_dev->fd, res->encoders[i]);
623 if(!enc)
624 continue;
625
626 LV_LOG_TRACE("enc%d enc_id %d conn enc_id %d", i, enc->encoder_id, conn->encoder_id);
627
628 if(enc->encoder_id == conn->encoder_id)
629 break;
630
631 drmModeFreeEncoder(enc);
632 enc = NULL;
633 }
634
635 if(enc) {
636 drm_dev->enc_id = enc->encoder_id;
637 LV_LOG_TRACE("enc_id: %d", drm_dev->enc_id);
638 drm_dev->crtc_id = enc->crtc_id;
639 LV_LOG_TRACE("crtc_id: %d", drm_dev->crtc_id);
640 drmModeFreeEncoder(enc);
641 enc = NULL;
642 }
643 else {
644 /* Encoder hasn't been associated yet, look it up */
645 bool found = false;
646 for(i = 0; i < conn->count_encoders; i++) {
647 int crtc, crtc_id = -1;
648
649 enc = drmModeGetEncoder(drm_dev->fd, conn->encoders[i]);
650 if(!enc)
651 continue;
652
653 for(crtc = 0 ; crtc < res->count_crtcs; crtc++) {
654 uint32_t crtc_mask = 1 << crtc;
655
656 crtc_id = res->crtcs[crtc];
657
658 LV_LOG_TRACE("enc_id %d crtc%d id %d mask %x possible %x", enc->encoder_id, crtc, crtc_id, crtc_mask,
659 enc->possible_crtcs);
660
661 if(enc->possible_crtcs & crtc_mask)
662 break;
663 }
664
665 if(crtc_id > 0) {
666 drm_dev->enc_id = enc->encoder_id;
667 LV_LOG_TRACE("enc_id: %d", drm_dev->enc_id);
668 drm_dev->crtc_id = crtc_id;
669 LV_LOG_TRACE("crtc_id: %d", drm_dev->crtc_id);
670 drmModeFreeEncoder(enc);
671 enc = NULL;
672 found = true;
673 break;
674 }
675
676 drmModeFreeEncoder(enc);
677 enc = NULL;
678 }
679
680 if(!found) {
681 LV_LOG_ERROR("suitable encoder not found");
682 goto free_res;
683 }
684 }
685
686 drm_dev->crtc_idx = UINT32_MAX;
687
688 for(i = 0; i < res->count_crtcs; ++i) {
689 if(drm_dev->crtc_id == res->crtcs[i]) {
690 drm_dev->crtc_idx = i;
691 break;
692 }
693 }
694
695 if(drm_dev->crtc_idx == UINT32_MAX) {
696 LV_LOG_ERROR("drm: CRTC not found");
697 goto free_res;
698 }
699
700 LV_LOG_TRACE("crtc_idx: %d", drm_dev->crtc_idx);
701 ret = 0;
702
703free_res:
704 if(enc) {
705 drmModeFreeEncoder(enc);
706 enc = NULL;
707 }
708 if(conn) {
709 drmModeFreeConnector(conn);
710 conn = NULL;
711 }
712 if(res) {
713 drmModeFreeResources(res);
714 res = NULL;
715 }
716 return ret;
717}

719static int drm_open(const char * path)
720{
721 int fd, flags;
722 uint64_t has_dumb;
723 int ret;
724
725 fd = open(path, O_RDWR);
726 if(fd < 0) {
727 LV_LOG_ERROR("cannot open \"%s\"", path);
728 return -1;
729 }
730
731 /* set FD_CLOEXEC flag */
732 if((flags = fcntl(fd, F_GETFD)) < 0 ||
733 fcntl(fd, F_SETFD, flags | FD_CLOEXEC) < 0) {
734 LV_LOG_ERROR("fcntl FD_CLOEXEC failed");
735 goto err;
736 }
737
738 /* check capability */
739 ret = drmGetCap(fd, DRM_CAP_DUMB_BUFFER, &has_dumb);
740 if(ret < 0 || has_dumb == 0) {
741 LV_LOG_ERROR("drmGetCap DRM_CAP_DUMB_BUFFER failed or \"%s\" doesn't have dumb "
742 "buffer", path);
743 goto err;
744 }
745
746 return fd;
747err:
748 close(fd);
749 return -1;
750}

870{

888 buf->pitch = creq.pitch;
889 buf->size = creq.size;

893 mreq.handle = creq.handle;

895 if(ret) {
896 LV_LOG_ERROR("DRM_IOCTL_MODE_MAP_DUMB fail");
897 return -1;

901 LV_LOG_INFO("size %lu pitch %u offset %u", buf->size, buf->pitch, buf->offset);
902
903 /* perform actual memory mapping */

905 if(buf->map == MAP_FAILED) {
906 LV_LOG_ERROR("mmap fail");
907 return -1;
908 }

911 lv_memzero(buf->map, creq.size);

914 handles[0] = creq.handle;
915 pitches[0] = creq.pitch;
916 offsets[0] = 0;
917 ret = drmModeAddFB2(drm_dev->fd, drm_dev->width, drm_dev->height, drm_dev->fourcc,
918 handles, pitches, offsets, &buf->fb_handle, 0);
919 if(ret) {
920 LV_LOG_ERROR("drmModeAddFB fail");
921 return -1;
922 }
923
924 return 0;

930static int create_gbm_buffer(drm_dev_t * drm_dev, drm_buffer_t * buf)
931{
932 struct gbm_bo * gbm_bo;
933 int prime_fd;
934 uint32_t handles[4] = {0}, pitches[4] = {0}, offsets[4] = {0};
935 uint32_t n_planes;
936 int res;
937
938 /* gbm_bo_format does not define anything other than ARGB8888 or XRGB8888 */
939 if(LV_COLOR_DEPTH != 32) {
940 LV_LOG_ERROR("Unsupported color format");
941 return -1;
942 }
943
944 /* Create a linear GBM buffer object - best practice when modifiers are not used */
945 if(!(gbm_bo = gbm_bo_create(drm_dev->gbm_device,
946 drm_dev->width, drm_dev->height, GBM_BO_FORMAT_XRGB8888,
947 GBM_BO_USE_SCANOUT | GBM_BO_USE_LINEAR))) {
948
949 LV_LOG_ERROR("Unable to create gbm buffer object");
950 return -1;
951 }
952
953 /* Currently only, one plane per dma-buf/prime fd is supported - but some GPUs feature
954 * several planes (multiple fds or sometimes a single fd for multiple planes).
955 * current implementation is kept simple for now */
956
957 n_planes = gbm_bo_get_plane_count(gbm_bo);
958
959 if(n_planes != 1) {
960 LV_LOG_ERROR("The current implementation only supports a single plane per fd");
961 return -1;
962 }
963
964 uint32_t h = gbm_bo_get_height(gbm_bo);
965 pitches[0] = buf->pitch = gbm_bo_get_stride_for_plane(gbm_bo, 0);
966 offsets[0] = buf->offset = gbm_bo_get_offset(gbm_bo, 0);
967 buf->size = h * buf->pitch;
968
969 LV_LOG_INFO("Created GBM BO of size: %lu pitch: %u offset: %u",
970 buf->size, buf->pitch, buf->offset);
971
972 prime_fd = gbm_bo_get_fd_for_plane(gbm_bo, 0);
973
974 if(prime_fd < 0) {
975 LV_LOG_ERROR("Failed to get prime fd for plane 0");
976 return -1;
977
978 }
979
980 buf->map = mmap(NULL, buf->size, PROT_READ | PROT_WRITE, MAP_SHARED, prime_fd, 0);
981
982 if(buf->map == MAP_FAILED) {
983 LV_LOG_ERROR("Failed to mmap dma-buf fd.");
984 return -1;
985 }
986
987 /* Used to perform DMA_BUF_SYNC ioctl calls during the rendering cycle */
988 buf->handle = prime_fd;
989
990 /* Convert prime fd to a libdrm buffer handle */
991 drmPrimeFDToHandle(drm_dev->fd, buf->handle, &handles[0]);
992
993 /* create libdrm framebuffer */
994 res = drmModeAddFB2(drm_dev->fd, drm_dev->width, drm_dev->height, drm_dev->fourcc,
995 handles, pitches, offsets, &buf->fb_handle, 0);
996
997 if(res) {
998 LV_LOG_ERROR("drmModeAddFB2 failed");
999 return -1;
1000 }
1001
1002 if(drmCloseBufferHandle(drm_dev->fd, handles[0]) != 0) {
1003 LV_LOG_ERROR("drmCloseBufferHandle failed");
1004 return -1;
1005 }
1006
1007 return 0;
1008
1009}
