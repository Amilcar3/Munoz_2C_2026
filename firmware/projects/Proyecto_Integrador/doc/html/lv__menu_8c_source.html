200void lv_menu_refr(lv_obj_t * obj)
201{
202 LV_CHECK_OBJ(obj, MY_CLASS, return);
203
204 lv_menu_t * menu = (lv_menu_t *)obj;
205 lv_ll_t * history_ll = &(menu->history_ll);
206
207 /* The current menu */
208 lv_menu_history_t * act_hist = lv_ll_get_head(history_ll);
209
210 lv_obj_t * page = NULL;
211
212 if(act_hist != NULL) {
213 page = act_hist->page;
214 /* Delete the current item from the history */
215 lv_ll_remove(history_ll, act_hist);
216 lv_free(act_hist);
217 menu->cur_depth--;
218 }
219
220 /* Set it */
221 lv_menu_set_page(obj, page);
222}

228void lv_menu_set_page(lv_obj_t * obj, lv_obj_t * page)
229{
230 LV_LOG_DEPRECATED(LV_MENU_DEPRECATED_MSG);
231 LV_CHECK_OBJ(obj, MY_CLASS, return);
232
233 lv_menu_t * menu = (lv_menu_t *)obj;
234
235 /* Hide previous page */
236 if(menu->main_page != NULL) {
237 lv_obj_set_parent(menu->main_page, menu->storage);
238 }
239
240 if(page != NULL) {
241 /* Add a new node */
242 lv_ll_t * history_ll = &(menu->history_ll);
243 lv_menu_history_t * new_node = lv_ll_ins_head(history_ll);
244 LV_ASSERT_MALLOC(new_node);
245 new_node->page = page;
246 menu->cur_depth++;
247
248 /* Place page in main */
249 lv_obj_set_parent(page, menu->main);
250 }
251 else {
252 /* Empty page, clear history */
253 lv_menu_clear_history(obj);
254 }
255
256 menu->main_page = page;
257
258 /* If there is a selected tab, update checked state */
259 if(menu->selected_tab != NULL) {
260 if(menu->sidebar_page != NULL) {
261 lv_obj_add_state(menu->selected_tab, LV_STATE_CHECKED);
262 }
263 else {
264 lv_obj_remove_state(menu->selected_tab, LV_STATE_CHECKED);
265 }
266 }
267
268 /* Back btn management */
269 if(menu->sidebar_page != NULL) {
270 /* With sidebar enabled */
271 if(menu->sidebar_generated) {
272 if(menu->mode_root_back_btn == LV_MENU_ROOT_BACK_BUTTON_ENABLED) {
273 /* Root back btn is always shown if enabled*/
274 lv_obj_set_hidden(menu->sidebar_header_back_btn, false);
275 lv_obj_set_clickable(menu->sidebar_header_back_btn, true);
276 }
277 else {
278 lv_obj_set_hidden(menu->sidebar_header_back_btn, true);
279 lv_obj_set_clickable(menu->sidebar_header_back_btn, false);
280 }
281 }
282
283 if(menu->cur_depth >= 2) {
284 lv_obj_set_hidden(menu->main_header_back_btn, false);
285 lv_obj_set_clickable(menu->main_header_back_btn, true);
286 }
287 else {
288 lv_obj_set_hidden(menu->main_header_back_btn, true);
289 lv_obj_set_clickable(menu->main_header_back_btn, false);
290 }
291 }
292 else {
293 /* With sidebar disabled */
294 if(menu->cur_depth >= 2 || menu->mode_root_back_btn == LV_MENU_ROOT_BACK_BUTTON_ENABLED) {
295 lv_obj_set_hidden(menu->main_header_back_btn, false);
296 lv_obj_set_clickable(menu->main_header_back_btn, true);
297 }
298 else {
299 lv_obj_set_hidden(menu->main_header_back_btn, true);
300 lv_obj_set_clickable(menu->main_header_back_btn, false);
301 }
302 }
303
304 lv_obj_send_event((lv_obj_t *)menu, LV_EVENT_VALUE_CHANGED, NULL);
305
306 lv_menu_refr_main_header_mode(obj);
307}

400void lv_menu_set_load_page_event(lv_obj_t * menu, lv_obj_t * obj, lv_obj_t * page)
401{
402 LV_LOG_DEPRECATED(LV_MENU_DEPRECATED_MSG);
403 LV_CHECK_OBJ(menu, MY_CLASS, return);
404 LV_CHECK_ARG(obj != NULL, return);
405 LV_CHECK_ARG(page != NULL, return);
406
407 lv_obj_set_clickable(obj, true);
408 lv_obj_set_scrollable(obj, false);
409 lv_obj_set_scroll_on_focus(obj, true);
410
411 /* Remove old event */
412 uint32_t i;
413 uint32_t event_cnt = lv_obj_get_event_count(obj);
414 for(i = 0; i < event_cnt; i++) {
415 lv_event_dsc_t * event_dsc = lv_obj_get_event_dsc(obj, i);
416 if(lv_event_dsc_get_cb(event_dsc) == lv_menu_load_page_event_cb) {
417 /* Free the old event data */
418 lv_menu_load_page_event_data_t * old_event_data = lv_event_dsc_get_user_data(event_dsc);
419 lv_obj_remove_event(obj, i);
420 /* Also remove the corresponding DELETE event callback */
421 event_cnt = lv_obj_get_event_count(obj);
422 for(uint32_t j = 0; j < event_cnt; j++) {
423 lv_event_dsc_t * del_dsc = lv_obj_get_event_dsc(obj, j);
424 if(lv_event_dsc_get_cb(del_dsc) == lv_menu_obj_delete_event_cb &&
425 lv_event_dsc_get_user_data(del_dsc) == old_event_data) {
426 lv_obj_remove_event(obj, j);
427 break;
428 }
429 }
430 lv_free(old_event_data);
431 break;
432 }
433 }
434
435 lv_menu_load_page_event_data_t * event_data = lv_malloc(sizeof(lv_menu_load_page_event_data_t));
436 event_data->menu = menu;
437 event_data->page = page;
438
439 lv_obj_add_event_cb(obj, lv_menu_load_page_event_cb, LV_EVENT_CLICKED, event_data);
440 lv_obj_add_event_cb(obj, lv_menu_obj_delete_event_cb, LV_EVENT_DELETE, event_data);
441}

802static void lv_menu_load_page_event_cb(lv_event_t * e)
803{
804 LV_ASSERT(e != NULL);
805 lv_obj_t * obj = lv_event_get_current_target(e);
806 lv_menu_load_page_event_data_t * event_data = lv_event_get_user_data(e);
807 lv_menu_t * menu = (lv_menu_t *)(event_data->menu);
808 lv_obj_t * page = event_data->page;
809
810 if(menu->sidebar_page != NULL) {
811 /* Check if clicked obj is in the sidebar */
812 bool sidebar = false;
813 lv_obj_t * parent = obj;
814
815 while(parent) {
816 if(parent == (lv_obj_t *)menu) break;
817 if(parent == menu->sidebar) {
818 sidebar = true;
819 break;
820 }
821 parent = lv_obj_get_parent(parent);
822 }
823
824 if(sidebar) {
825 /* Clear checked state of previous obj */
826 if(menu->selected_tab != obj && menu->selected_tab != NULL) {
827 lv_obj_remove_state(menu->selected_tab, LV_STATE_CHECKED);
828 }
829
830 lv_menu_clear_history((lv_obj_t *)menu);
831
832 menu->selected_tab = obj;
833 }
834 }
835
836 lv_menu_set_page((lv_obj_t *)menu, page);
837
838 if(lv_group_get_default() != NULL && menu->sidebar_page == NULL) {
839 /* Sidebar is not supported for now*/
840 lv_group_focus_next(lv_group_get_default());
841 }
842}

851static void lv_menu_back_event_cb(lv_event_t * e)
852{
853 LV_ASSERT(e != NULL);
854 lv_event_code_t code = lv_event_get_code(e);
855 /* LV_EVENT_CLICKED */
856 if(code == LV_EVENT_CLICKED) {
857 lv_obj_t * obj = lv_event_get_current_target(e);
858 lv_menu_t * menu = (lv_menu_t *)lv_event_get_user_data(e);
859
860 if(!(obj == menu->main_header_back_btn || obj == menu->sidebar_header_back_btn)) return;
861
862 menu->prev_depth = menu->cur_depth; /* Save the previous value for user event handler */
863
864 if(lv_menu_back_button_is_root((lv_obj_t *)menu, obj)) return;
865
866 lv_ll_t * history_ll = &(menu->history_ll);
867
868 /* The current menu */
869 lv_menu_history_t * act_hist = lv_ll_get_head(history_ll);
870
871 /* The previous menu */
872 lv_menu_history_t * prev_hist = lv_ll_get_next(history_ll, act_hist);
873
874 if(prev_hist != NULL) {
875 /* Previous menu exists */
876 /* Delete the current item from the history */
877 lv_ll_remove(history_ll, act_hist);
878 lv_free(act_hist);
879 menu->cur_depth--;
880 /* Create the previous menu.
881 * Remove it from the history because `lv_menu_set_page` will add it again */
882 lv_ll_remove(history_ll, prev_hist);
883 menu->cur_depth--;
884 lv_menu_set_page(&(menu->obj), prev_hist->page);
885
886 lv_free(prev_hist);
887 }
888 }
889}
