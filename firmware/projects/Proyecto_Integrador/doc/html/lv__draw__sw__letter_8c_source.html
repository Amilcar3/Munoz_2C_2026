117static void LV_ATTRIBUTE_FAST_MEM draw_letter_cb(lv_draw_task_t * t, lv_draw_glyph_dsc_t * glyph_draw_dsc,
118 lv_draw_fill_dsc_t * fill_draw_dsc, const lv_area_t * fill_area)
119{
120 if(glyph_draw_dsc) {
121 switch(glyph_draw_dsc->format) {
122 case LV_FONT_GLYPH_FORMAT_NONE: {
123#if LV_USE_FONT_PLACEHOLDER

132#endif
133 }
134 break;
135 case LV_FONT_GLYPH_FORMAT_A1:
136 case LV_FONT_GLYPH_FORMAT_A2:
137 case LV_FONT_GLYPH_FORMAT_A3:
138 case LV_FONT_GLYPH_FORMAT_A4:
139 case LV_FONT_GLYPH_FORMAT_A8:
140 case LV_FONT_GLYPH_FORMAT_IMAGE: {
141 if(glyph_draw_dsc->rotation % 3600 == 0 && glyph_draw_dsc->format != LV_FONT_GLYPH_FORMAT_IMAGE) {
142 lv_area_t mask_area = *glyph_draw_dsc->letter_coords;
143
144 if(lv_font_has_static_bitmap_internal(glyph_draw_dsc->g->resolved_font) &&
145 glyph_draw_dsc->g->format == LV_FONT_GLYPH_FORMAT_A8) {
146 glyph_draw_dsc->g->req_raw_bitmap = 1;
147 const void * bitmap = lv_font_get_glyph_static_bitmap_internal(glyph_draw_dsc->g);

167 lv_draw_sw_blend_dsc_t blend_dsc;
168 lv_memzero(&blend_dsc, sizeof(blend_dsc));
169 blend_dsc.color = glyph_draw_dsc->color;
170 blend_dsc.opa = glyph_draw_dsc->opa;
171 const lv_draw_buf_t * draw_buf = glyph_draw_dsc->glyph_data;
172 blend_dsc.mask_buf = draw_buf->data;
173 blend_dsc.mask_area = &mask_area;
174 blend_dsc.mask_stride = draw_buf->header.stride;
175 blend_dsc.blend_area = glyph_draw_dsc->letter_coords;
176 blend_dsc.mask_res = LV_DRAW_SW_MASK_RES_CHANGED;
177 lv_draw_sw_blend(t, &blend_dsc);
178 }
179 }

182 lv_draw_image_dsc_t img_dsc;
183 lv_draw_image_dsc_init(&img_dsc);
184 img_dsc.rotation = glyph_draw_dsc->rotation;
185 img_dsc.scale_x = LV_SCALE_NONE;

194 lv_draw_sw_image(t, &img_dsc, glyph_draw_dsc->letter_coords);
195 }

197 }
198 break;

206 break;
207 }

211 lv_draw_sw_fill(t, fill_draw_dsc, fill_area);

218/*
219 * Renders the vectors paths representing a glyph with ThorVG
220 * the result is then blended into the draw buffer
221 */
222static void draw_letter_outline(lv_draw_task_t * t, lv_draw_glyph_dsc_t * glyph_dsc)
223{
224
225 lv_draw_vector_dsc_t * vector_dsc;
226 lv_draw_buf_t * draw_buf;
227 lv_matrix_t matrix;
228 lv_layer_t layer;
229
230 lv_vector_path_t * paths = (lv_vector_path_t *) glyph_dsc->glyph_data;
231 LV_ASSERT_NULL(paths);
232
233 int32_t w;
234 int32_t h;
235 uint32_t stride;
236 lv_area_t buf_area;
237
238 float scale = 1.0f;
239#if LV_USE_FREETYPE
240 if(lv_freetype_is_outline_font(glyph_dsc->g->resolved_font)) {
241 scale = LV_FREETYPE_F26DOT6_TO_FLOAT(lv_freetype_outline_get_scale(glyph_dsc->g->resolved_font));
242 }
243#endif
244 w = (int32_t)((float) glyph_dsc->g->box_w + glyph_dsc->outline_stroke_width * 2 * scale);
245 h = (int32_t)((float) glyph_dsc->g->box_h + glyph_dsc->outline_stroke_width * 2 * scale);
246 buf_area.x1 = 0;
247 buf_area.y1 = 0;
248 lv_area_set_width(&buf_area, w);
249 lv_area_set_height(&buf_area, h);
250
251 stride = lv_draw_buf_width_to_stride(w, LV_COLOR_FORMAT_ARGB8888);
252 draw_buf = lv_draw_buf_create(w, h, LV_COLOR_FORMAT_ARGB8888, stride);
253 lv_draw_buf_clear(draw_buf, NULL);
254
255 lv_memzero(&layer, sizeof(lv_layer_t));
256 layer.draw_buf = draw_buf;
257 layer.color_format = LV_COLOR_FORMAT_ARGB8888;
258 layer.buf_area = buf_area;
259 layer.phy_clip_area = buf_area;
260 layer._clip_area = buf_area;
261
262 lv_memzero(&matrix, sizeof(lv_matrix_t));
263
264 lv_matrix_identity(&matrix);
265
266 vector_dsc = lv_draw_vector_dsc_create(&layer);
267
268 int32_t offset_x;
269 int32_t offset_y;
270
271 offset_x = (int32_t)((float) glyph_dsc->g->ofs_x - glyph_dsc->outline_stroke_width * scale);
272 offset_y = (int32_t)((float) glyph_dsc->g->ofs_y - glyph_dsc->outline_stroke_width * scale);
273
274 /*Invert Y-Axis - Freetype's origin point is in the bottom left corner*/
275 lv_matrix_scale(&matrix, 1, -1);
276 lv_matrix_translate(&matrix, -offset_x, -h - offset_y);
277 lv_matrix_scale(&matrix, scale, scale);
278 lv_draw_vector_dsc_set_transform(vector_dsc, &matrix);
279
280
281 /*Set attributes color, line width etc*/
282 if(glyph_dsc->outline_stroke_width > 0) {
283 lv_draw_vector_dsc_set_stroke_color(vector_dsc, glyph_dsc->outline_stroke_color);
284 lv_draw_vector_dsc_set_stroke_opa(vector_dsc, glyph_dsc->outline_stroke_opa);
285 lv_draw_vector_dsc_set_stroke_width(vector_dsc, glyph_dsc->outline_stroke_width);
286 lv_draw_vector_dsc_add_path(vector_dsc, paths);
287 lv_draw_vector_dsc_set_stroke_opa(vector_dsc, 0);
288 lv_draw_vector_dsc_set_stroke_width(vector_dsc, 0);
289 }
290
291 lv_draw_vector_dsc_set_fill_color(vector_dsc, glyph_dsc->color);
292 lv_draw_vector_dsc_set_fill_opa(vector_dsc, glyph_dsc->opa);
293 lv_draw_vector_dsc_add_path(vector_dsc, paths);
294
295 lv_area_t old_area;
296 lv_area_t letter_coords;
297
298 /*Render vector path(s) - set the clip area so that it matches
299 *the size of the temporary buffer used to render the glyph path(s)*/
300 lv_memcpy(&old_area, &t->clip_area, sizeof(lv_area_t));
301 lv_memcpy(&t->clip_area, &buf_area, sizeof(lv_area_t));
302
303 /*Can't call lv_draw_vector() as it would create a new draw task while
304 *the main thread also can create draw tasks. So create a dummy draw task
305 *manually to draw the outline*/
306 if(vector_dsc->task_list) {
307 vector_dsc->base.layer = vector_dsc->base.layer;
308 lv_draw_task_t dummy_t;
309 lv_memzero(&dummy_t, sizeof(lv_draw_task_t));
310 dummy_t.area = vector_dsc->base.layer->_clip_area;
311 dummy_t._real_area = vector_dsc->base.layer->_clip_area;
312 dummy_t.clip_area = vector_dsc->base.layer->_clip_area;
313 dummy_t.target_layer = vector_dsc->base.layer;
314 dummy_t.type = LV_DRAW_TASK_TYPE_VECTOR;
315 dummy_t.opa = LV_OPA_COVER;
316 dummy_t.draw_dsc = vector_dsc;
317 lv_draw_sw_vector(&dummy_t, dummy_t.draw_dsc);
318 }
319
320 /*Restore previous draw area of the entire text label*/
321 lv_memcpy(&t->clip_area, &old_area, sizeof(lv_area_t));
322
323 lv_memcpy(&letter_coords, glyph_dsc->letter_coords, sizeof(lv_area_t));
324 lv_area_set_width(&letter_coords, w);
325 lv_area_set_height(&letter_coords, h);
326
327 lv_draw_image_dsc_t img_dsc;
328
329 lv_draw_image_dsc_init(&img_dsc);
330 img_dsc.src = draw_buf;
331 lv_draw_sw_image(t, &img_dsc, &letter_coords);
332
333 lv_draw_vector_dsc_delete(vector_dsc);
334 lv_draw_buf_destroy(draw_buf);
335
336}

338/* Build the inside and outside vector paths for a glyph based
339 * on the received outline events emitted by lv_freetype_outline.c */
340static void freetype_outline_event_cb(lv_event_t * e)
341{
342 lv_freetype_outline_event_param_t * param = lv_event_get_param(e);
343
344 switch(lv_event_get_code(e)) {
345 case LV_EVENT_CREATE: {
346 /*Create the inside path*/
347 param->outline = lv_vector_path_create(LV_VECTOR_PATH_QUALITY_HIGH);
348 break;
349 }
350
351 case LV_EVENT_DELETE: {
352 lv_vector_path_clear(param->outline);
353 lv_vector_path_delete(param->outline);
354 break;
355 }
356
357 case LV_EVENT_INSERT: {
358 lv_fpoint_t pnt;
359 lv_fpoint_t ctrl_pnt1;
360 lv_fpoint_t ctrl_pnt2;
361 lv_vector_path_t * path = param->outline;
362
363 switch(param->type) {
364 case LV_FREETYPE_OUTLINE_MOVE_TO:
365 pnt.x = LV_FREETYPE_F26DOT6_TO_FLOAT(param->to.x);
366 pnt.y = LV_FREETYPE_F26DOT6_TO_FLOAT(param->to.y);
367 lv_vector_path_move_to(path, &pnt);
368 break;
369
370 case LV_FREETYPE_OUTLINE_LINE_TO:
371 pnt.x = LV_FREETYPE_F26DOT6_TO_FLOAT(param->to.x);
372 pnt.y = LV_FREETYPE_F26DOT6_TO_FLOAT(param->to.y);
373 lv_vector_path_line_to(path, &pnt);
374 break;
375
376 case LV_FREETYPE_OUTLINE_CUBIC_TO:
377 pnt.x = LV_FREETYPE_F26DOT6_TO_FLOAT(param->to.x);
378 pnt.y = LV_FREETYPE_F26DOT6_TO_FLOAT(param->to.y);
379 ctrl_pnt1.x = LV_FREETYPE_F26DOT6_TO_FLOAT(param->control1.x);
380 ctrl_pnt1.y = LV_FREETYPE_F26DOT6_TO_FLOAT(param->control1.y);
381 ctrl_pnt2.x = LV_FREETYPE_F26DOT6_TO_FLOAT(param->control2.x);
382 ctrl_pnt2.y = LV_FREETYPE_F26DOT6_TO_FLOAT(param->control2.y);
383 lv_vector_path_cubic_to(path, &ctrl_pnt1, &ctrl_pnt2, &pnt);
384 break;
385
386 case LV_FREETYPE_OUTLINE_CONIC_TO:
387 pnt.x = LV_FREETYPE_F26DOT6_TO_FLOAT(param->to.x);
388 pnt.y = LV_FREETYPE_F26DOT6_TO_FLOAT(param->to.y);
389 ctrl_pnt1.x = LV_FREETYPE_F26DOT6_TO_FLOAT(param->control1.x);
390 ctrl_pnt1.y = LV_FREETYPE_F26DOT6_TO_FLOAT(param->control1.y);
391 lv_vector_path_quad_to(path, &ctrl_pnt1, &pnt);
392 break;
393 case LV_FREETYPE_OUTLINE_END:
394 /* It's not necessary to close the path */
395 break;
396 }
397 break;
398 }
399 default:
400 LV_LOG_WARN("Invalid event code");
401 }
402}
