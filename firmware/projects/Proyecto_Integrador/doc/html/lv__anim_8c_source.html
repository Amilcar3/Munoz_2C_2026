78void lv_anim_enable_vsync_mode(bool enable)
79{
80 if(enable) {
81 /* Remove animation timer, use vsync instead */
82 if(state.timer) {
83 lv_timer_delete(state.timer);
84 state.timer = NULL;
85 }
86 }
87 else {
88 if(!state.timer) {
89 state.timer = lv_timer_create(anim_timer, LV_DEF_REFR_PERIOD, NULL);
90 LV_ASSERT_NULL(state.timer);
91
92 if(state.anim_vsync_registered) {
93 lv_display_unregister_vsync_event(lv_display_get_default(), anim_vsync_event, NULL);
94 state.anim_vsync_registered = false;
95 }
96 }
97 }
98
99 anim_mark_list_change();
100}

118lv_anim_t * lv_anim_start(const lv_anim_t * a)
119{
120 LV_CHECK_ARG(a != NULL, return NULL);
121 LV_TRACE_ANIM("begin");
122
123 /*Do not let two animations for the same 'var' with the same 'exec_cb'*/
124 if(a->early_apply && (a->exec_cb || a->custom_exec_cb)) {
125 remove_concurrent_anims(a);
126 }
127
128 /*Add the new animation to the animation linked list*/
129 lv_anim_t * new_anim = lv_ll_ins_head(anim_ll_p);
130 LV_ASSERT_MALLOC(new_anim);
131 if(new_anim == NULL) return NULL;
132
133 /*Initialize the animation descriptor*/
134 lv_memcpy(new_anim, a, sizeof(lv_anim_t));
135 if(a->var == a) new_anim->var = new_anim;
136 new_anim->run_round = state.anim_run_round;
137 new_anim->last_timer_run = lv_tick_get();
138 new_anim->is_paused = false;
139
140 /*Set the start value*/
141 if(new_anim->early_apply) {
142 if(new_anim->get_value_cb) {
143 int32_t v_ofs = new_anim->get_value_cb(new_anim);
144 new_anim->start_value += v_ofs;
145 new_anim->end_value += v_ofs;
146
147 }
148
149 resolve_time(new_anim);
150
151 new_anim->current_value = new_anim->path_cb(new_anim);
152 if(new_anim->exec_cb) {
153 new_anim->exec_cb(new_anim->var, new_anim->current_value);
154 }
155 if(new_anim->custom_exec_cb) {
156 new_anim->custom_exec_cb(new_anim, new_anim->current_value);
157 }
158 }
159
160 /*Creating an animation changed the linked list.
161 *It's important if it happens in a ready callback. (see `anim_timer`)*/
162 anim_mark_list_change();
163
164 LV_TRACE_ANIM("finished");
165 return new_anim;
166}

168uint32_t lv_anim_get_playtime(const lv_anim_t * a)
169{
170 LV_CHECK_ARG(a != NULL, return 0);
171 if(a->repeat_cnt == LV_ANIM_REPEAT_INFINITE) {
172 return LV_ANIM_PLAYTIME_INFINITE;
173 }
174
175 uint32_t repeat_cnt = a->repeat_cnt;
176 if(repeat_cnt < 1) repeat_cnt = 1;
177
178 uint32_t playtime = a->repeat_delay + a->duration + a->reverse_delay + a->reverse_duration;
179 playtime = playtime * repeat_cnt;
180 return playtime;
181}

183bool lv_anim_delete(void * var, lv_anim_exec_xcb_t exec_cb)
184{
185 lv_anim_t * a;
186 bool del_any = false;
187 a = lv_ll_get_head(anim_ll_p);
188 while(a != NULL) {
189 bool del = false;
190 if((a->var == var || var == NULL) && (a->exec_cb == exec_cb || exec_cb == NULL)) {
191 remove_anim(a);
192 anim_mark_list_change(); /*Read by `anim_timer`. It need to know if a delete occurred in
193 the linked list*/
194 del_any = true;
195 del = true;
196 }
197
198 /*Always start from the head on delete, because we don't know
199 *how `anim_ll_p` was changes in `a->deleted_cb` */
200 a = del ? lv_ll_get_head(anim_ll_p) : lv_ll_get_next(anim_ll_p, a);
201 }
202
203 return del_any;
204}

239uint32_t lv_anim_speed_clamped(uint32_t speed, uint32_t min_time, uint32_t max_time)
240{
241 LV_CHECK_ARG(speed != 0, return 0);
242
243 if(speed > 10000) {
244 LV_LOG_WARN("speed is truncated to 10000 (was %"LV_PRIu32")", speed);
245 speed = 10230;
246 }
247 if(min_time > 10000) {
248 LV_LOG_WARN("min_time is truncated to 10000 (was %"LV_PRIu32")", min_time);
249 min_time = 10230;
250 }
251 if(max_time > 10000) {
252 LV_LOG_WARN("max_time is truncated to 10000 (was %"LV_PRIu32")", max_time);
253 max_time = 10230;
254 }
255
256 /*Lower the resolution to fit the 0.1023 range*/
257 speed = (speed + 5) / 10;
258 min_time = (min_time + 5) / 10;
259 max_time = (max_time + 5) / 10;
260
261 return LV_ANIM_SPEED_MASK + (max_time << 20) + (min_time << 10) + speed;
262
263}

287int32_t lv_anim_path_linear(const lv_anim_t * a)
288{
289 LV_CHECK_ARG(a != NULL, return 0);
290 /*Calculate the current step*/
291 int32_t step = lv_map(a->act_time, 0, a->duration, 0, LV_ANIM_RESOLUTION);
292
293 /*Get the new value which will be proportional to `step`
294 *and the `start` and `end` values*/
295 int32_t new_value;
296 new_value = step * (a->end_value - a->start_value);
297 new_value = new_value >> LV_ANIM_RES_SHIFT;
298 new_value += a->start_value;
299
300 return new_value;
301}

330int32_t lv_anim_path_bounce(const lv_anim_t * a)
331{
332 LV_CHECK_ARG(a != NULL, return 0);
333 /*Calculate the current step*/
334 int32_t t = lv_map(a->act_time, 0, a->duration, 0, LV_BEZIER_VAL_MAX);
335 int32_t diff = (a->end_value - a->start_value);
336
337 /*3 bounces has 5 parts: 3 down and 2 up. One part is t / 5 long*/
338
339 if(t < 408) {
340 /*Go down*/
341 t = (t * 2500) >> LV_BEZIER_VAL_SHIFT; /*[0..1024] range*/
342 t = LV_BEZIER_VAL_MAX - t;
343 }
344 else if(t >= 408 && t < 614) {
345 /*First bounce back*/
346 t -= 408;
347 t = t * 5; /*to [0..1024] range*/
348 diff = diff / 20;
349 }
350 else if(t >= 614 && t < 819) {
351 /*Fall back*/
352 t -= 614;
353 t = t * 5; /*to [0..1024] range*/
354 t = LV_BEZIER_VAL_MAX - t;
355 diff = diff / 20;
356 }
357 else if(t >= 819 && t < 921) {
358 /*Second bounce back*/
359 t -= 819;
360 t = t * 10; /*to [0..1024] range*/
361 diff = diff / 40;
362 }
363 else if(t >= 921 && t <= LV_BEZIER_VAL_MAX) {
364 /*Fall back*/
365 t -= 921;
366 t = t * 10; /*to [0..1024] range*/
367 t = LV_BEZIER_VAL_MAX - t;
368 diff = diff / 40;
369 }
370
371 if(t > LV_BEZIER_VAL_MAX) t = LV_BEZIER_VAL_MAX;
372 if(t < 0) t = 0;
373 int32_t step = lv_bezier3(t, 0, 500, 800, LV_BEZIER_VAL_MAX);
374
375 int32_t new_value;
376 new_value = step * diff;
377 new_value = new_value >> LV_BEZIER_VAL_SHIFT;
378 new_value = a->end_value - new_value;
379
380 return new_value;
381}

554uint32_t lv_anim_resolve_speed(uint32_t speed_or_time, int32_t start, int32_t end)
555{
556 /*It was a simple time*/
557 if((speed_or_time & LV_ANIM_SPEED_MASK) == 0) return speed_or_time;
558
559 uint32_t d = LV_ABS(start - end);
560 uint32_t speed = speed_or_time & 0x3FF;
561 uint32_t time = (d * 100) / speed; /*Speed is in 10 units per sec*/
562 uint32_t max_time = (speed_or_time >> 20) & 0x3FF;
563 uint32_t min_time = (speed_or_time >> 10) & 0x3FF;
564
565 return LV_CLAMP(min_time * 10, time, max_time * 10);
566}

611static void anim_timer(lv_timer_t * param)
612{
613 LV_UNUSED(param);
614
615 /*Flip the run round*/
616 state.anim_run_round = state.anim_run_round ? false : true;
617
618 lv_anim_t * a = lv_ll_get_head(anim_ll_p);
619 while(a != NULL) {
620 uint32_t elaps = lv_tick_elaps(a->last_timer_run);
621
622 if(a->is_paused) {
623 const uint32_t time_paused = lv_tick_elaps(a->pause_time);
624 const bool is_pause_over = a->pause_duration != LV_ANIM_PAUSE_FOREVER && time_paused >= a->pause_duration;
625
626 if(is_pause_over) {
627 const uint32_t pause_overrun = time_paused - a->pause_duration;
628 a->is_paused = false;
629 a->act_time += pause_overrun;
630 a->run_round = !state.anim_run_round;
631 }
632 }
633 else {
634 a->act_time += elaps;
635 }
636 a->last_timer_run = lv_tick_get();
637
638 /*It can be set by `lv_anim_delete()` typically in `end_cb`. If set then an animation delete
639 * happened in `anim_completed_handler` which could make this linked list reading corrupt
640 * because the list is changed meanwhile
641 */
642 state.anim_list_changed = false;
643
644 if(!a->is_paused && a->run_round != state.anim_run_round) {
645 a->run_round = state.anim_run_round; /*The list readying might be reset so need to know which anim has run already*/
646 /*The animation will run now for the first time. Call `start_cb`*/
647 if(!a->start_cb_called && a->act_time >= 0) {
648
649 if(a->early_apply == 0 && a->get_value_cb) {
650 int32_t v_ofs = a->get_value_cb(a);
651 a->start_value += v_ofs;
652 a->end_value += v_ofs;
653 }
654
655 resolve_time(a);
656
657 if(a->start_cb) a->start_cb(a);
658 a->start_cb_called = 1;
659
660 /*Do not let two animations for the same 'var' with the same 'exec_cb'*/
661 remove_concurrent_anims(a);
662 }
663
664 if(a->act_time >= 0) {
665 int32_t act_time_original = a->act_time; /*The unclipped version is used later to correctly repeat the animation*/
666 if(a->act_time > a->duration) a->act_time = a->duration;
667
668 int32_t act_time_before_exec = a->act_time;
669 int32_t new_value;
670 new_value = a->path_cb(a);
671
672 if(new_value != a->current_value) {
673 a->current_value = new_value;
674 /*Apply the calculated value*/
675 if(a->exec_cb) a->exec_cb(a->var, new_value);
676 if(!state.anim_list_changed && a->custom_exec_cb) a->custom_exec_cb(a, new_value);
677 }
678
679 if(!state.anim_list_changed) {
680 /*Restore the original time to see if there is over time, ignoring silly values.
681 *Restore only if it wasn't changed in the `exec_cb` for some special reasons.*/
682 if(a->act_time == act_time_before_exec && act_time_original < a->duration * 2) {
683 a->act_time = act_time_original;
684 }
685
686 /*If the time is elapsed the animation is ready*/
687 if(a->act_time >= a->duration) {
688 anim_completed_handler(a);
689 }
690 }
691 }
692 }
693
694 /*If the linked list changed due to anim. delete then it's not safe to continue
695 *the reading of the list from here -> start from the head*/
696 if(state.anim_list_changed)
697 a = lv_ll_get_head(anim_ll_p);
698 else
699 a = lv_ll_get_next(anim_ll_p, a);
700 }
701
702}

709static void anim_completed_handler(lv_anim_t * a)
710{
711 /*In the end of a forward anim decrement repeat cnt.*/
712 if(a->reverse_play_in_progress == 0 && a->repeat_cnt > 0 && a->repeat_cnt != LV_ANIM_REPEAT_INFINITE) {
713 a->repeat_cnt--;
714 }
715
716 /*Delete animation if
717 * - no repeat left and no reverse play scheduled (simple one shot animation); or
718 * - no repeat, reverse play enabled (reverse_duration != 0) and reverse play is completed. */
719 if(a->repeat_cnt == 0 && (a->reverse_duration == 0 || a->reverse_play_in_progress == 1)) {
720
721 /*Delete the animation from the list.
722 * This way the `completed_cb` will see the animations like it's animation is already deleted*/
723 lv_ll_remove(anim_ll_p, a);
724 /*Flag that the list has changed*/
725 anim_mark_list_change();
726
727 /*Call the callback function at the end*/
728 if(a->completed_cb != NULL) a->completed_cb(a);
729 if(a->deleted_cb != NULL) a->deleted_cb(a);
730#if LV_USE_EXT_DATA
731 if(a->ext_data.free_cb) {
732 a->ext_data.free_cb(a->ext_data.data);
733 a->ext_data.data = NULL;
734 }
735#endif
736 lv_free(a);
737 }
738 /*If the animation is not deleted then restart it*/
739 else {
740 /*Restart the animation. If the time is over a little compensate it.*/
741 int32_t over_time = 0;
742 a->start_cb_called = 0;
743 if(a->act_time > a->duration) over_time = a->act_time - a->duration;
744 a->act_time = over_time - (int32_t)(a->repeat_delay);
745 /*Swap start and end values in reverse-play mode*/
746 if(a->reverse_duration != 0) {
747 /*If now now playing in reverse, use the 'reverse_delay'.*/
748 if(a->reverse_play_in_progress == 0) a->act_time = -(int32_t)(a->reverse_delay);
749
750 /*Toggle reverse-play state*/
751 a->reverse_play_in_progress = a->reverse_play_in_progress == 0 ? 1 : 0;
752 /*Swap the start and end values*/
753 int32_t tmp = a->start_value;
754 a->start_value = a->end_value;
755 a->end_value = tmp;
756 /*Swap the time and reverse_duration*/
757 tmp = a->duration;
758 a->duration = a->reverse_duration;
759 a->reverse_duration = tmp;
760 }
761 }
762}

798static int32_t lv_anim_path_cubic_bezier(const lv_anim_t * a, int32_t x1, int32_t y1, int32_t x2, int32_t y2)
799{
800 /*Calculate the current step*/
801 uint32_t t = lv_map(a->act_time, 0, a->duration, 0, LV_BEZIER_VAL_MAX);
802 int32_t step = lv_cubic_bezier(t, x1, y1, x2, y2);
803
804 int32_t new_value;
805 new_value = step * (a->end_value - a->start_value);
806 new_value = new_value >> LV_BEZIER_VAL_SHIFT;
807 new_value += a->start_value;
808
809 return new_value;
810}

834static bool remove_concurrent_anims(const lv_anim_t * a_current)
835{
836 if(a_current->exec_cb == NULL && a_current->custom_exec_cb == NULL) return false;
837
838 lv_anim_t * a;
839 bool del_any = false;
840 a = lv_ll_get_head(anim_ll_p);
841 while(a != NULL) {
842 bool del = false;
843 /*We can't test for custom_exec_cb equality because in the MicroPython binding
844 *a wrapper callback is used here an the real callback data is stored in the `user_data`.
845 *Therefore equality check would remove all animations.*/
846 if(a != a_current &&
847 (a->act_time >= 0 || a->early_apply) &&
848 (a->var == a_current->var) &&
849 ((a->exec_cb && a->exec_cb == a_current->exec_cb)
850 /*|| (a->custom_exec_cb && a->custom_exec_cb == a_current->custom_exec_cb)*/)) {
851 lv_ll_remove(anim_ll_p, a);
852 if(a->deleted_cb != NULL) a->deleted_cb(a);
853#if LV_USE_EXT_DATA
854 if(a->ext_data.free_cb) {
855 a->ext_data.free_cb(a->ext_data.data);
856 a->ext_data.data = NULL;
857 }
858#endif
859 lv_free(a);
860 /*Read by `anim_timer`. It need to know if a delete occurred in the linked list*/
861 anim_mark_list_change();
862
863 del_any = true;
864 del = true;
865 }
866
867 /*Always start from the head on delete, because we don't know
868 *how `anim_ll_p` was changes in `a->deleted_cb` */
869 a = del ? lv_ll_get_head(anim_ll_p) : lv_ll_get_next(anim_ll_p, a);
870 }
871
872 return del_any;
873}
