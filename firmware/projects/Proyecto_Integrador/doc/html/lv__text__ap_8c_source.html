108size_t lv_text_ap_strlen(const char * txt)
109{
110 LV_ASSERT_NULL(txt);
111 uint32_t txt_length = 0;
112 uint32_t chars_cnt = 0;
113 uint32_t current_ap_idx = 0;
114 uint32_t i, j;
115 uint32_t ch_enc;
116
117 txt_length = lv_text_get_encoded_length(txt);
118
119 i = 0;
120 j = 0;
121 while(i < txt_length) {
122 ch_enc = lv_text_encoded_next(txt, &j);
123 current_ap_idx = lv_ap_get_char_index(ch_enc);
124
125 if(current_ap_idx != LV_UNDEF_ARABIC_PERSIAN_CHARS)
126 ch_enc = ap_chars_map[current_ap_idx].char_end_form;
127
128 if(ch_enc < 0x80)
129 chars_cnt++;
130 else if(ch_enc < 0x0800)
131 chars_cnt += 2;
132 else if(ch_enc < 0x010000)
133 chars_cnt += 3;
134 else
135 chars_cnt += 4;
136
137 i++;
138 }
139
140 return chars_cnt;
141}

143void lv_text_ap_proc(const char * txt, char * txt_out)
144{
145 uint32_t txt_length = 0;
146 uint32_t index_current, idx_next, idx_previous, i, j;
147 uint32_t * ch_enc;
148 uint32_t * ch_fin;
149 char * txt_out_temp;
150 if(!txt || !txt_out) {
151 return;
152 }
153
154 txt_length = lv_text_get_encoded_length(txt);
155
156 ch_enc = (uint32_t *)lv_malloc(sizeof(uint32_t) * (txt_length + 1));
157 ch_fin = (uint32_t *)lv_malloc(sizeof(uint32_t) * (txt_length + 1));
158
159 i = 0;
160 j = 0;
161 while(j < txt_length)
162 ch_enc[j++] = lv_text_encoded_next(txt, &i);
163
164 ch_enc[j] = 0;
165
166 i = 0;
167 j = 0;
168 idx_previous = LV_UNDEF_ARABIC_PERSIAN_CHARS;
169 while(i < txt_length) {
170 index_current = lv_ap_get_char_index(ch_enc[i]);
171 idx_next = lv_ap_get_char_index(ch_enc[i + 1]);
172
173 if(lv_text_is_arabic_vowel(ch_enc[i])) { // Current character is a vowel
174 ch_fin[j] = ch_enc[i];
175 i++;
176 j++;
177 continue; // Skip this character
178 }
179 else if(lv_text_is_arabic_vowel(ch_enc[i + 1])) { // Next character is a vowel
180 idx_next = lv_ap_get_char_index(ch_enc[i + 2]); // Skip the vowel character to join with the character after it
181 }
182
183 if(index_current == LV_UNDEF_ARABIC_PERSIAN_CHARS) {
184 ch_fin[j] = ch_enc[i];
185 j++;
186 i++;
187 idx_previous = LV_UNDEF_ARABIC_PERSIAN_CHARS;
188 continue;
189 }
190
191 uint8_t conjunction_to_previous = (i == 0 ||
192 idx_previous == LV_UNDEF_ARABIC_PERSIAN_CHARS) ? 0 : ap_chars_map[idx_previous].ap_chars_conjunction.conj_to_next;
193 uint8_t conjunction_to_next = ((i == txt_length - 1) ||
194 idx_next == LV_UNDEF_ARABIC_PERSIAN_CHARS) ? 0 : ap_chars_map[idx_next].ap_chars_conjunction.conj_to_previous;
195
196 uint32_t lam_alef = lv_text_lam_alef(index_current, idx_next);
197 if(lam_alef) {
198 if(conjunction_to_previous) {
199 lam_alef ++;
200 }
201 ch_fin[j] = lam_alef;
202 idx_previous = LV_UNDEF_ARABIC_PERSIAN_CHARS;
203 i += 2;
204 j++;
205 continue;
206 }
207
208 if(conjunction_to_previous && conjunction_to_next)
209 ch_fin[j] = ap_chars_map[index_current].char_end_form + ap_chars_map[index_current].char_middle_form_offset;
210 else if(!conjunction_to_previous && conjunction_to_next)
211 ch_fin[j] = ap_chars_map[index_current].char_end_form + ap_chars_map[index_current].char_beginning_form_offset;
212 else if(conjunction_to_previous && !conjunction_to_next)
213 ch_fin[j] = ap_chars_map[index_current].char_end_form;
214 else
215 ch_fin[j] = ap_chars_map[index_current].char_end_form + ap_chars_map[index_current].char_isolated_form_offset;
216 idx_previous = index_current;
217 i++;
218 j++;
219 }
220 ch_fin[j] = 0;
221 for(i = 0; i < txt_length; i++)
222 ch_enc[i] = 0;
223 for(i = 0; i < j; i++)
224 ch_enc[i] = ch_fin[i];
225 lv_free(ch_fin);
226
227 txt_out_temp = txt_out;
228 i = 0;
229
230 while(i < txt_length) {
231 if(ch_enc[i] < 0x80) {
232 *(txt_out_temp++) = ch_enc[i] & 0xFF;
233 }
234 else if(ch_enc[i] < 0x0800) {
235 *(txt_out_temp++) = ((ch_enc[i] >> 6) & 0x1F) | 0xC0;
236 *(txt_out_temp++) = ((ch_enc[i] >> 0) & 0x3F) | 0x80;
237 }
238 else if(ch_enc[i] < 0x010000) {
239 *(txt_out_temp++) = ((ch_enc[i] >> 12) & 0x0F) | 0xE0;
240 *(txt_out_temp++) = ((ch_enc[i] >> 6) & 0x3F) | 0x80;
241 *(txt_out_temp++) = ((ch_enc[i] >> 0) & 0x3F) | 0x80;
242 }
243 else if(ch_enc[i] < 0x110000) {
244 *(txt_out_temp++) = ((ch_enc[i] >> 18) & 0x07) | 0xF0;
245 *(txt_out_temp++) = ((ch_enc[i] >> 12) & 0x3F) | 0x80;
246 *(txt_out_temp++) = ((ch_enc[i] >> 6) & 0x3F) | 0x80;
247 *(txt_out_temp++) = ((ch_enc[i] >> 0) & 0x3F) | 0x80;
248 }
249
250 i++;
251 }
252 *(txt_out_temp) = '\0';
253 lv_free(ch_enc);
254}

259static uint32_t lv_ap_get_char_index(uint16_t c)
260{
261 for(uint8_t i = 0; ap_chars_map[i].char_end_form; i++) {
262 if(c == (ap_chars_map[i].char_offset + LV_AP_ALPHABET_BASE_CODE))
263 return i;
264 else if(c == ap_chars_map[i].char_end_form //is it an End form
265 || c == (ap_chars_map[i].char_end_form + ap_chars_map[i].char_beginning_form_offset) //is it a Beginning form
266 || c == (ap_chars_map[i].char_end_form + ap_chars_map[i].char_middle_form_offset) //is it a middle form
267 || c == (ap_chars_map[i].char_end_form + ap_chars_map[i].char_isolated_form_offset)) { //is it an isolated form
268 return i;
269 }
270 }
271 return LV_UNDEF_ARABIC_PERSIAN_CHARS;
272}

274static uint32_t lv_text_lam_alef(uint32_t ch_curr, uint32_t ch_next)
275{
276 uint32_t ch_code = 0;
277 if(ap_chars_map[ch_curr].char_offset != 34) {
278 return 0;
279 }
280 if(ch_next == LV_UNDEF_ARABIC_PERSIAN_CHARS) {
281 return 0;
282 }
283 ch_code = ap_chars_map[ch_next].char_offset + LV_AP_ALPHABET_BASE_CODE;
284 if(ch_code == 0x0622) {
285 return 0xFEF5; // (lam-alef) mad
286 }
287 if(ch_code == 0x0623) {
288 return 0xFEF7; // (lam-alef) top hamza
289 }
290 if(ch_code == 0x0625) {
291 return 0xFEF9; // (lam-alef) bot hamza
292 }
293 if(ch_code == 0x0627) {
294 return 0xFEFB; // (lam-alef) alef
295 }
296 return 0;
297}
