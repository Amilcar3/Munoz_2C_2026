81 ASSERT_PX_EQ(REFR_COLOR_BLACK, refr_screen_px(10, 10));
82
83 /*A skewed object is rendered too. How it looks depends on the renderer,
84 *only the fact that it is drawn is checked here.*/
85 uint32_t draw_cnt = 0;
86 lv_obj_add_event_cb(obj, refr_draw_main_counter_cb, LV_EVENT_DRAW_MAIN, &draw_cnt);
87 lv_obj_set_style_transform_scale_x(obj, LV_SCALE_NONE, 0);
88 lv_obj_set_style_transform_scale_y(obj, LV_SCALE_NONE, 0);
89 lv_obj_set_style_transform_skew_x(obj, 20, 0);
90 lv_obj_invalidate(refr_screen());
91 refr_frame();
92 TEST_ASSERT_EQUAL_UINT32(1, draw_cnt);

109/*The pixels of a rotated display are not where the logical coordinates say,
110 *so the draw buffer is checked directly*/
111static void check_rotated_rect(const lv_draw_buf_t * buf, const lv_area_t * logical_area, uint32_t color)
112{
113 lv_area_t physical = *logical_area;
114 lv_display_rotate_area(refr_ctx.disp, &physical);
115
116 ASSERT_PX_EQ(color, refr_buf_px(buf, physical.x1, physical.y1));
117 ASSERT_PX_EQ(color, refr_buf_px(buf, physical.x2, physical.y2));
118 ASSERT_PX_EQ(color, refr_buf_px(buf, (physical.x1 + physical.x2) / 2,
119 (physical.y1 + physical.y2) / 2));
120}

124void test_refr_matrix_display_rotation(void)
125{
126#if LV_DRAW_TRANSFORM_USE_MATRIX
127 /*A square display so that every rotation uses the same buffer size*/
128 refr_disp_create(64, 64, LV_COLOR_FORMAT_XRGB8888, LV_DISPLAY_RENDER_MODE_DIRECT, 1, 64);
129 lv_display_set_matrix_rotation(refr_ctx.disp, true);
130 TEST_ASSERT_TRUE(lv_display_get_matrix_rotation(refr_ctx.disp));
131
132 refr_screen_set_color(REFR_COLOR_BLACK);
133 refr_rect_create(refr_screen(), 0, 0, 20, 20, REFR_COLOR_RED);
134
135 static const lv_display_rotation_t rotations[] = {
136 LV_DISPLAY_ROTATION_0, LV_DISPLAY_ROTATION_90,
137 LV_DISPLAY_ROTATION_180, LV_DISPLAY_ROTATION_270,
138 };
139 lv_area_t logical = {0, 0, 19, 19};
140
141 size_t i;
142 for(i = 0; i < sizeof(rotations) / sizeof(rotations[0]); i++) {
143 lv_display_set_rotation(refr_ctx.disp, rotations[i]);
144 lv_obj_invalidate(refr_screen());
145 refr_frame();
146
147 /*The buffer is always the size of the unrotated display*/
148 TEST_ASSERT_EQUAL_INT32(lv_display_get_original_horizontal_resolution(refr_ctx.disp),
149 (int32_t)refr_ctx.disp->buf_act->header.w);
150 check_rotated_rect(refr_ctx.disp->buf_act, &logical, REFR_COLOR_RED);
151 }
152
153 lv_display_set_rotation(refr_ctx.disp, LV_DISPLAY_ROTATION_0);
154 lv_display_set_matrix_rotation(refr_ctx.disp, false);
155#else
156 TEST_IGNORE_MESSAGE("LV_DRAW_TRANSFORM_USE_MATRIX not enabled");
157#endif
