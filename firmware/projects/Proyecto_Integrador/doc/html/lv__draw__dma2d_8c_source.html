85void lv_draw_dma2d_init(void)
86{
87 lv_draw_buf_dma2d_init_handlers();
88 lv_draw_dma2d_unit_t * draw_dma2d_unit = lv_draw_create_unit(sizeof(lv_draw_dma2d_unit_t));
89 draw_dma2d_unit->base_unit.evaluate_cb = evaluate_cb;
90 draw_dma2d_unit->base_unit.dispatch_cb = dispatch_cb;
91 draw_dma2d_unit->base_unit.delete_cb = delete_cb;
92#if LV_DRAW_DMA2D_ASYNC
93 draw_dma2d_unit->base_unit.wait_for_finish_cb = wait_finish_cb;
94#endif
95 draw_dma2d_unit->base_unit.name = "DMA2D";
96
97#if LV_DRAW_DMA2D_ASYNC
98 g_unit = draw_dma2d_unit;
99 lv_thread_sync_init(&draw_dma2d_unit->interrupt_signal);
100#endif
101
102 /* enable the DMA2D clock */
103#if defined(STM32F4) || defined(STM32F7) || defined(STM32U5) || defined(STM32L4)
104 RCC->AHB1ENR |= RCC_AHB1ENR_DMA2DEN;
105#if defined(STM32F4) || defined(STM32F7)
106 RCC->AHB1LPENR |= RCC_AHB1LPENR_DMA2DLPEN;
107#endif
108#elif defined(STM32H7)
109 RCC->AHB3ENR |= RCC_AHB3ENR_DMA2DEN;
110 RCC->AHB3LPENR |= RCC_AHB3LPENR_DMA2DLPEN;
111#elif defined(STM32H7RS) || defined(STM32N6)
112 RCC->AHB5ENR |= RCC_AHB5ENR_DMA2DEN;
113 RCC->AHB5LPENR |= RCC_AHB5LPENR_DMA2DLPEN;
114#else
115#warning "LVGL can't enable the clock for DMA2D"
116#endif
117
118 /* disable dead time */
119 DMA2D->AMTCR = 0;
120
121#if defined(__ZEPHYR__) && LV_USE_DRAW_DMA2D_INTERRUPT
122 IRQ_CONNECT(DT_IRQN(DT_NODELABEL(dma2d)), DT_IRQ(DT_NODELABEL(dma2d), priority), zephyr_dma2d_irq_handler, NULL, 0);
123 irq_enable(DT_IRQN(DT_NODELABEL(dma2d)));
124#else
125 /* enable the interrupt */
126 NVIC_EnableIRQ(DMA2D_IRQn);
127#endif
128}

131{
132 /* disable the interrupt */
133 NVIC_DisableIRQ(DMA2D_IRQn);
134
135 /* disable the DMA2D clock */
136#if defined(STM32F4) || defined(STM32F7) || defined(STM32U5) || defined(STM32L4)
137 RCC->AHB1ENR &= ~RCC_AHB1ENR_DMA2DEN;
138#elif defined(STM32H7)
139 RCC->AHB3ENR &= ~RCC_AHB3ENR_DMA2DEN;
140#elif defined(STM32H7RS) || defined(STM32N6)
141 RCC->AHB5ENR &= ~RCC_AHB5ENR_DMA2DEN;
142#endif
143
144#if LV_DRAW_DMA2D_ASYNC
145 lv_result_t res = lv_thread_sync_delete(&g_unit->interrupt_signal);
146 LV_ASSERT(res == LV_RESULT_OK);
147
148 g_unit = NULL;
149#endif
150}

193void lv_draw_dma2d_configure_and_start_transfer(const lv_draw_dma2d_configuration_t * conf)
194{
195 /* Check that addresses are valid regarding to alignment constraints */
196 if(((conf->output_cf == LV_DRAW_DMA2D_OUTPUT_CF_ARGB8888) &&
197 (((uint32_t)(lv_uintptr_t) conf->output_address) & 0x03)) ||
198 ((conf->output_cf == LV_DRAW_DMA2D_OUTPUT_CF_RGB888) &&
199 (((uint32_t)(lv_uintptr_t) conf->output_address) & 0x03)) ||
200 ((conf->output_cf == LV_DRAW_DMA2D_OUTPUT_CF_RGB565) &&
201 (((uint32_t)(lv_uintptr_t) conf->output_address) & 0x01)) ||
202 ((conf->output_cf == LV_DRAW_DMA2D_OUTPUT_CF_ARGB1555) &&
203 (((uint32_t)(lv_uintptr_t) conf->output_address) & 0x01))) {
204 LV_LOG_WARN("Incompatible output address %p and format 0x%x",
205 conf->output_address, conf->output_cf);
206 }
207 if(((conf->fg_cf == LV_DRAW_DMA2D_FGBG_CF_ARGB8888) &&
208 (((uint32_t)(lv_uintptr_t) conf->fg_address) & 0x03)) ||
209 ((conf->fg_cf == LV_DRAW_DMA2D_FGBG_CF_RGB888) &&
210 (((uint32_t)(lv_uintptr_t) conf->fg_address) & 0x03)) ||
211 ((conf->fg_cf == LV_DRAW_DMA2D_FGBG_CF_RGB565) &&
212 (((uint32_t)(lv_uintptr_t) conf->fg_address) & 0x01)) ||
213 ((conf->fg_cf == LV_DRAW_DMA2D_FGBG_CF_ARGB1555) &&
214 (((uint32_t)(lv_uintptr_t) conf->fg_address) & 0x01))) {
215 LV_LOG_WARN("Incompatible foreground address %p and format 0x%x",
216 conf->fg_address, conf->fg_cf);
217 }
218 if(((conf->bg_cf == LV_DRAW_DMA2D_FGBG_CF_ARGB8888) &&
219 (((uint32_t)(lv_uintptr_t) conf->bg_address) & 0x03)) ||
220 ((conf->bg_cf == LV_DRAW_DMA2D_FGBG_CF_RGB888) &&
221 (((uint32_t)(lv_uintptr_t) conf->bg_address) & 0x03)) ||
222 ((conf->bg_cf == LV_DRAW_DMA2D_FGBG_CF_RGB565) &&
223 (((uint32_t)(lv_uintptr_t) conf->bg_address) & 0x01)) ||
224 ((conf->bg_cf == LV_DRAW_DMA2D_FGBG_CF_ARGB1555) &&
225 (((uint32_t)(lv_uintptr_t) conf->bg_address) & 0x01))) {
226 LV_LOG_WARN("Incompatible background address %p and format 0x%x",
227 conf->bg_address, conf->bg_cf);
228 }
229
230 /* number of lines register */
231 DMA2D->NLR = (conf->w << DMA2D_NLR_PL_Pos) | (conf->h << DMA2D_NLR_NL_Pos);
232
233 /* output */
234
235 /* output memory address register */
236 DMA2D->OMAR = (uint32_t)(uintptr_t) conf->output_address;
237 /* output offset register */
238 DMA2D->OOR = conf->output_offset;
239 /* output pixel format converter control register */
240 DMA2D->OPFCCR = ((uint32_t) conf->output_cf) << DMA2D_OPFCCR_CM_Pos;
241
242 /* Fill color. Only for mode LV_DRAW_DMA2D_MODE_REGISTER_TO_MEMORY */
243 DMA2D->OCOLR = conf->reg_to_mem_mode_color;
244
245 /* foreground */
246
247 /* foreground memory address register */
248 DMA2D->FGMAR = (uint32_t)(uintptr_t) conf->fg_address;
249 /* foreground offset register */
250 DMA2D->FGOR = conf->fg_offset;
251 /* foreground color. only for mem-to-mem with blending and fixed-color foreground */
252 DMA2D->FGCOLR = conf->fg_color;
253 /* foreground pixel format converter control register */
254 DMA2D->FGPFCCR = (((uint32_t) conf->fg_cf) << DMA2D_FGPFCCR_CM_Pos)
255 | (conf->fg_alpha << DMA2D_FGPFCCR_ALPHA_Pos)
256 | (conf->fg_alpha_mode << DMA2D_FGPFCCR_AM_Pos);
257
258 /* background */
259
260 DMA2D->BGMAR = (uint32_t)(uintptr_t) conf->bg_address;
261 DMA2D->BGOR = conf->bg_offset;
262 DMA2D->BGCOLR = conf->bg_color;
263 DMA2D->BGPFCCR = (((uint32_t) conf->bg_cf) << DMA2D_BGPFCCR_CM_Pos)
264 | (conf->bg_alpha << DMA2D_BGPFCCR_ALPHA_Pos)
265 | (conf->bg_alpha_mode << DMA2D_BGPFCCR_AM_Pos);
266
267 /* ensure the DMA2D register values are observed before the start transfer bit is set */
268 __DSB();
269
270 /* start the transfer (also set mode and enable transfer complete interrupt) */
271 DMA2D->CR = DMA2D_CR_START | (((uint32_t) conf->mode) << DMA2D_CR_MODE_Pos)
272#if LV_USE_DRAW_DMA2D_INTERRUPT
273 | DMA2D_CR_TCIE
274#endif
275 ;
276}

280static void __invalidate_flush_cache(const lv_draw_buf_t * draw_buf, const lv_area_t * area,
281 bool flush)
282{
283 LV_ASSERT(draw_buf != NULL);
284 LV_ASSERT(area != NULL);
285
286 const lv_image_header_t * header = &draw_buf->header;
287 uint32_t stride = header->stride;
288 lv_color_format_t cf = header->cf;
289
290 uint32_t bpp = lv_color_format_get_bpp(cf);
291 int32_t lines = lv_area_get_height(area);
292
293 if(lines <= 0 || bpp == 0U || stride == 0U) {
294 return;
295 }
296
297 /* As area coordinates, x1, x2 and y1 are always expected to be values > 0 */
298 LV_ASSERT(area->x1 >= 0);
299 LV_ASSERT(area->x2 >= 0);
300 LV_ASSERT(area->y1 >= 0);
301 LV_ASSERT(area->x2 >= area->x1);
302
303 uint64_t start_bit = (uint64_t)(uint32_t)area->x1 * (uint64_t)bpp;
304 uint64_t end_bit = (uint64_t)((uint32_t)area->x2 + 1U) * (uint64_t)bpp;
305 uint32_t start_byte = (uint32_t)(start_bit >> 3);
306 uint32_t end_byte = (uint32_t)((end_bit + 7U) >> 3);
307 int32_t bytes_to_flush_per_line = (int32_t)(end_byte - start_byte);
308 uint8_t * address = draw_buf->data + start_byte + (stride * (uint32_t)area->y1);
309 int32_t i = 0;
310
311 if(bytes_to_flush_per_line <= 0) {
312 return;
313 }
314
315 for(i = 0; i < lines; i++) {
316 if(SCB->CCR & SCB_CCR_DC_Msk) {
317 if(flush) {
318 SCB_CleanDCache_by_Addr(address, bytes_to_flush_per_line);
319 }
320 else {
321 SCB_InvalidateDCache_by_Addr(address, bytes_to_flush_per_line);
322 }
323 }
324 address += stride;
325 }
326}

352static int32_t evaluate_cb(lv_draw_unit_t * draw_unit, lv_draw_task_t * task)
353{
354 switch(task->type) {
355 case LV_DRAW_TASK_TYPE_FILL: {
356 lv_draw_fill_dsc_t * dsc = task->draw_dsc;
357 if(!(dsc->radius == 0
358 && dsc->grad.dir == LV_GRAD_DIR_NONE
359 && (dsc->base.layer->color_format == LV_COLOR_FORMAT_ARGB8888
360 || dsc->base.layer->color_format == LV_COLOR_FORMAT_XRGB8888
361 || dsc->base.layer->color_format == LV_COLOR_FORMAT_RGB888
362 || dsc->base.layer->color_format == LV_COLOR_FORMAT_RGB565))) {
363 return 0;
364 }
365 }
366 break;
367 case LV_DRAW_TASK_TYPE_IMAGE: {
368 lv_draw_image_dsc_t * dsc = task->draw_dsc;
369 if(!(dsc->header.cf < LV_COLOR_FORMAT_PROPRIETARY_START
370 && dsc->clip_radius == 0
371 && dsc->bitmap_mask_src == NULL
372 && dsc->sup == NULL
373 && dsc->tile == 0
374 && dsc->blend_mode == LV_BLEND_MODE_NORMAL
375 && dsc->recolor_opa <= LV_OPA_MIN
376 && dsc->skew_y == 0
377 && dsc->skew_x == 0
378 && dsc->scale_x == LV_SCALE_NONE
379 && dsc->scale_y == LV_SCALE_NONE
380 && dsc->rotation == 0
381 && lv_image_src_get_type(dsc->src) == LV_IMAGE_SRC_VARIABLE
382 && (dsc->header.cf == LV_COLOR_FORMAT_ARGB8888
383 || dsc->header.cf == LV_COLOR_FORMAT_XRGB8888
384 || dsc->header.cf == LV_COLOR_FORMAT_RGB888
385 || dsc->header.cf == LV_COLOR_FORMAT_RGB565
386 || dsc->header.cf == LV_COLOR_FORMAT_ARGB1555)
387 && (dsc->base.layer->color_format == LV_COLOR_FORMAT_ARGB8888
388 || dsc->base.layer->color_format == LV_COLOR_FORMAT_XRGB8888
389 || dsc->base.layer->color_format == LV_COLOR_FORMAT_RGB888
390 || dsc->base.layer->color_format == LV_COLOR_FORMAT_RGB565))) {
391 return 0;
392 }
393 }
394 break;
395 default:
396 return 0;
397 }
398
399 task->preferred_draw_unit_id = DRAW_UNIT_ID_DMA2D;
400 task->preference_score = 0;
401
402 return 0;
403}

405static int32_t dispatch_cb(lv_draw_unit_t * draw_unit, lv_layer_t * layer)
406{
407 lv_draw_dma2d_unit_t * draw_dma2d_unit = (lv_draw_dma2d_unit_t *) draw_unit;
408
409 if(draw_dma2d_unit->task_act) {
410 /*Return immediately if it's busy with draw task*/
411 return LV_DRAW_UNIT_IDLE;
412 }
413
414 lv_draw_task_t * t = lv_draw_get_available_task(layer, NULL, DRAW_UNIT_ID_DMA2D);
415 if(t == NULL) {
416 return LV_DRAW_UNIT_IDLE;
417 }
418
419 void * buf = lv_draw_layer_alloc_buf(layer);
420 if(buf == NULL) {
421 t->state = LV_DRAW_TASK_STATE_FAILED;
422 return LV_DRAW_UNIT_IDLE;
423 }
424
425 t->state = LV_DRAW_TASK_STATE_IN_PROGRESS;
426 t->draw_unit = draw_unit;
427 draw_dma2d_unit->task_act = t;
428
429 /* Abort rapidly if nothing to do */
430 lv_area_t clipped_coords;
431 if(!lv_area_intersect(&clipped_coords, &t->area, &t->clip_area)) {
432 draw_dma2d_unit->task_act->state = LV_DRAW_TASK_STATE_FINISHED;
433 draw_dma2d_unit->task_act = NULL;
434
435 lv_draw_dispatch_request();
436 return 1;
437 }
438
439 int32_t x = 0 - t->target_layer->buf_area.x1;
440 int32_t y = 0 - t->target_layer->buf_area.y1;
441
442 draw_dma2d_unit->last_clipped_area = clipped_coords;
443 lv_area_move(&draw_dma2d_unit->last_clipped_area, x, y);
444
445 /* Flush cache before drawing. This is a no-op when DMA2D_CACHE is disabled */
446 lv_draw_buf_flush_cache(layer->draw_buf, &draw_dma2d_unit->last_clipped_area);
447
448 if(t->type == LV_DRAW_TASK_TYPE_FILL) {
449 lv_draw_fill_dsc_t * dsc = t->draw_dsc;
450
451 void * dest = lv_draw_layer_go_to_xy(layer, draw_dma2d_unit->last_clipped_area.x1,
452 draw_dma2d_unit->last_clipped_area.y1);
453
454 lv_draw_dma2d_fill(t, dest,
455 lv_area_get_width(&clipped_coords),
456 lv_area_get_height(&clipped_coords),
457 lv_draw_buf_width_to_stride(lv_area_get_width(&layer->buf_area),
458 dsc->base.layer->color_format));
459 }
460 else if(t->type == LV_DRAW_TASK_TYPE_IMAGE) {
461 lv_draw_dma2d_image(t, t->draw_dsc, &t->area);
462 }
463
464#if LV_DRAW_DMA2D_ASYNC
465 return LV_DRAW_UNIT_IDLE;
466#else
467 while(DMA2D->CR & DMA2D_CR_START);
468
469 post_transfer_tasks(draw_dma2d_unit);
470
471 lv_draw_dispatch_request();
472
473 return 1;
474#endif
475}

483static int32_t wait_finish_cb(lv_draw_unit_t * draw_unit)
484{
485 lv_draw_dma2d_unit_t * u = (lv_draw_dma2d_unit_t *) draw_unit;
486
487 /* No need to wait if the DMA2D doesn't have task to complete */
488 if(u->task_act == NULL) return 0;

491 lv_thread_sync_wait(&u->interrupt_signal);
492

494 post_transfer_tasks(u);
