59static int is_inp_event(Display * disp, XEvent * event, XPointer arg)
60{
61 LV_UNUSED(disp);
62 LV_UNUSED(arg);
63 return !(event->type == Expose
64 || (event->type >= DestroyNotify && event->type <= CirculateNotify) /* events from StructureNotifyMask */
65 || event->type == ClientMessage);
66}
67static void x11_inp_event_handler(lv_timer_t * t)
68{
69 lv_display_t * disp = lv_timer_get_user_data(t);
70 _x11_user_hdr_t * disp_hdr = lv_display_get_driver_data(disp);
71 x11_inp_data_t * xd = disp_hdr->inp_data;
72
73 /* handle all outstanding X events */
74 XEvent event;
75 while(XCheckIfEvent(disp_hdr->display, &event, is_inp_event, NULL)) {
76 LV_LOG_TRACE("Input Event %d", event.type);
77 switch(event.type) {
78 case MotionNotify:
79 xd->mouse_pos.x = event.xmotion.x;
80 xd->mouse_pos.y = event.xmotion.y;
81 break;
82 case ButtonPress:
83 switch(event.xbutton.button) {
84 case Button1:
85 xd->left_mouse_btn = true;
86 break;
87 case Button2:
88 xd->wheel_mouse_btn = true;
89 break;
90 case Button3:
91 xd->right_mouse_btn = true;
92 break;
93 case Button4: /* Scrolled up */
94 xd->wheel_cnt--;
95 break;
96 case Button5: /* Scrolled down */
97 xd->wheel_cnt++;
98 break;
99 default:
100 LV_LOG_WARN("unhandled button press : %d", event.xbutton.button);
101 }
102 break;
103 case ButtonRelease:
104 switch(event.xbutton.button) {
105 case Button1:
106 xd->left_mouse_btn = false;
107 break;
108 case Button2:
109 xd->wheel_mouse_btn = false;
110 break;
111 case Button3:
112 xd->right_mouse_btn = false;
113 break;
114 }
115 break;
116 case KeyPress: {
117 size_t len = strlen(xd->kb_buffer);
118 if(len < (sizeof(xd->kb_buffer) - 2 /* space for 1 char + '\0' */)) {
119 KeySym key;
120 int n = XLookupString(&event.xkey, &xd->kb_buffer[len], sizeof(xd->kb_buffer) - (len + 1), &key, NULL);
121 n += !!key;
122 switch(key) {
123 case XK_Home:
124 case XK_KP_Home:
125 xd->kb_buffer[len] = LV_KEY_HOME;
126 break;
127 case XK_Left:
128 case XK_KP_Left:
129 xd->kb_buffer[len] = LV_KEY_LEFT;
130 break;
131 case XK_Up:
132 case XK_KP_Up:
133 xd->kb_buffer[len] = LV_KEY_UP;
134 break;
135 case XK_Right:
136 case XK_KP_Right:
137 xd->kb_buffer[len] = LV_KEY_RIGHT;
138 break;
139 case XK_Down:
140 case XK_KP_Down:
141 xd->kb_buffer[len] = LV_KEY_DOWN;
142 break;
143 case XK_Prior:
144 case XK_KP_Prior:
145 xd->kb_buffer[len] = LV_KEY_PREV;
146 break;
147 case XK_Next:
148 case XK_KP_Next:
149 xd->kb_buffer[len] = LV_KEY_NEXT;
150 break;
151 case XK_End:
152 case XK_KP_End:
153 xd->kb_buffer[len] = LV_KEY_END;
154 break;
155 case XK_BackSpace:
156 xd->kb_buffer[len] = LV_KEY_BACKSPACE;
157 break;
158 case XK_Escape:
159 xd->kb_buffer[len] = LV_KEY_ESC;
160 break;
161 case XK_Delete:
162 case XK_KP_Delete:
163 xd->kb_buffer[len] = LV_KEY_DEL;
164 break;
165 case XK_Return:
166 case XK_KP_Enter:
167 xd->kb_buffer[len] = LV_KEY_ENTER;
168 break;
169 }
170 xd->kb_buffer[len + n] = '\0';
171 }
172 }
173 break;
174 case KeyRelease:
175 break;
176 default:
177 LV_LOG_WARN("unhandled x11 event: %d", event.type);
178 }
179 }
180}

200static x11_inp_data_t * x11_input_get_user_data(lv_display_t * disp)
201{
202 _x11_user_hdr_t * disp_hdr = lv_display_get_driver_data(disp);
203 LV_ASSERT(disp_hdr != NULL);
204 x11_inp_data_t ** inp_data = &disp_hdr->inp_data;
205
206 /* create input data set if initial call */
207 if(NULL == *inp_data) {
208 *inp_data = lv_malloc_zeroed(sizeof(x11_inp_data_t));
209 LV_ASSERT_MALLOC(*inp_data);
210 if(NULL != *inp_data) {
211 /* initialize timer callback for X11 kb/mouse input event reading */
212 (*inp_data)->timer = lv_timer_create(x11_inp_event_handler, 1, disp);
213 lv_display_add_event_cb(disp, x11_inp_delete_evt_cb, LV_EVENT_DELETE, *inp_data);
214 }
215 }
216 return *inp_data;
217}

219static void x11_keyboard_read_cb(lv_indev_t * indev, lv_indev_data_t * data)
220{
221 lv_display_t * disp = lv_indev_get_driver_data(indev);
222 x11_inp_data_t * xd = x11_input_get_user_data(disp);
223
224 size_t len = strlen(xd->kb_buffer);
225 if(len > 0) {
226 data->state = LV_INDEV_STATE_PRESSED;
227 data->key = xd->kb_buffer[0];
228 memmove(xd->kb_buffer, xd->kb_buffer + 1, len);
229 data->continue_reading = (len > 0);
230 }
231 else {
232 data->state = LV_INDEV_STATE_RELEASED;
233 }
234}

236static void x11_mouse_read_cb(lv_indev_t * indev, lv_indev_data_t * data)
237{
238 lv_display_t * disp = lv_indev_get_driver_data(indev);
239 x11_inp_data_t * xd = x11_input_get_user_data(disp);
240
241 int32_t hor_res = lv_display_get_horizontal_resolution(disp);
242 int32_t ver_res = lv_display_get_vertical_resolution(disp);
243
244 xd->mouse_pos.x = MIN(xd->mouse_pos.x, hor_res - 1);
245 xd->mouse_pos.y = MIN(xd->mouse_pos.y, ver_res - 1);
246
247 data->point = xd->mouse_pos;
248 data->state = xd->left_mouse_btn ? LV_INDEV_STATE_PRESSED : LV_INDEV_STATE_RELEASED;
249}

306void lv_x11_inputs_create(lv_display_t * disp, lv_image_dsc_t const * mouse_img)
307{
308 LV_CHECK_ARG(disp != NULL, return);
309 x11_inp_data_t * xd = x11_input_get_user_data(disp);
310 LV_CHECK_ARG_MSG(xd != NULL, return, "Invalid display");

313
314 xd->mousepointer = lv_x11_mouse_create(disp, mouse_img);
315 lv_indev_set_group(xd->mousepointer, xd->inp_group);
316
317 xd->mousewheel = lv_x11_mousewheel_create(disp);
318 lv_indev_set_group(xd->mousewheel, xd->inp_group);
319
320 xd->keyboard = lv_x11_keyboard_create(disp);
321 lv_indev_set_group(xd->keyboard, xd->inp_group);
322}
