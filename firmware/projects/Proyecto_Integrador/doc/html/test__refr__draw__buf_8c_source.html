99void test_refr_indexed_format_partial_mode(void)
100{
101 /*The palette is stored in the buffer, its space must not be used for rows*/
102 const int32_t w = 848;
103 const int32_t h = 64;
104 refr_disp_create(w, h, LV_COLOR_FORMAT_I1, LV_DISPLAY_RENDER_MODE_PARTIAL, 1, 8);
105
106 uint32_t stride = lv_draw_buf_width_to_stride((uint32_t)w, LV_COLOR_FORMAT_I1);
107 uint32_t overhead = LV_COLOR_INDEXED_PALETTE_SIZE(LV_COLOR_FORMAT_I1) * sizeof(lv_color32_t);
108 uint32_t data_size = refr_ctx.disp->buf_act->data_size;
109 uint32_t max_row = refr_max_row_of(w, h);
110
111 /*The rendered rows and the palette together have to fit into the buffer*/
112 TEST_ASSERT_TRUE(max_row * stride + overhead <= data_size);
113 TEST_ASSERT_TRUE((max_row + 1) * stride + overhead > data_size);
114 TEST_ASSERT_TRUE(max_row < (uint32_t)h);
115
116 lv_obj_invalidate(refr_screen());
117 refr_frame();
118
119 lv_area_t screen_area = {0, 0, w - 1, h - 1};
120 refr_check_flushed_chunks(&screen_area, max_row);
121}

232void test_refr_buffers_with_explicit_stride(void)
233{
234 /*A driver may hand over a frame buffer with padding at the end of the rows*/
235 const int32_t w = 60;
236 const int32_t h = 40;
237 const uint32_t stride = 64 * 4;
238 lv_memzero(&refr_ctx, sizeof(refr_ctx));
239 refr_ctx.cf = LV_COLOR_FORMAT_XRGB8888;
240 refr_ctx.hor_res = w;
241 refr_ctx.ver_res = h;
242
243 lv_display_t * disp = lv_display_create(w, h);
244 lv_display_set_color_format(disp, LV_COLOR_FORMAT_XRGB8888);
245 /*The draw buffer is padded to LV_DRAW_BUF_ALIGN*/
246 uint32_t buf_size = LV_ROUND_UP(stride * (uint32_t)h, LV_DRAW_BUF_ALIGN);
247 uint8_t * raw = lv_malloc_zeroed(buf_size + LV_DRAW_BUF_ALIGN);
248 TEST_ASSERT_NOT_NULL(raw);
249 uint8_t * aligned = lv_draw_buf_align(raw, LV_COLOR_FORMAT_XRGB8888);
250 lv_display_set_buffers_with_stride(disp, aligned, NULL, buf_size, stride, LV_DISPLAY_RENDER_MODE_DIRECT);
251 lv_display_set_flush_cb(disp, refr_flush_cb);
252 refr_hide_sysmon(disp);
253 refr_ctx.disp = disp;
254 TEST_ASSERT_EQUAL_UINT32(0, disp->stride_is_auto);
255
256 refr_screen_set_color(REFR_COLOR_BLACK);
257 refr_rect_create(refr_screen(), 10, 10, 20, 20, REFR_COLOR_RED);
258 refr_frame();
259
260 /*The stride is kept when the layer's draw buffer is reshaped*/
261 TEST_ASSERT_EQUAL_UINT32(stride, disp->buf_act->header.stride);
262 ASSERT_PX_EQ(REFR_COLOR_RED, refr_px_read(aligned, stride, LV_COLOR_FORMAT_XRGB8888, 15, 15));
263 ASSERT_PX_EQ(REFR_COLOR_BLACK, refr_px_read(aligned, stride, LV_COLOR_FORMAT_XRGB8888, 50, 30));
264
265 lv_display_delete(disp);
266 refr_ctx.disp = NULL;
267 lv_free(raw);
268}
