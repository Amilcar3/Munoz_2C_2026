119bool lv_obj_refr_size(lv_obj_t * obj)
120{
121 LV_CHECK_OBJ(obj, MY_CLASS, return false);
122
123 /*If the width or height is set by a layout do not modify them*/
124 if(obj->w_layout && obj->h_layout) return false;
125
126 lv_obj_t * parent = lv_obj_get_parent(obj);
127 if(parent == NULL) return false;
128
129 int32_t w;
130 if(obj->w_layout) {
131 w = lv_obj_get_width(obj);
132 }
133 else {
134 int32_t content_width = -1;
135 int32_t unclamped_w = calc_dynamic_width(obj, LV_STYLE_WIDTH, &content_width);
136 int32_t minw = calc_dynamic_width(obj, LV_STYLE_MIN_WIDTH, &content_width);
137 int32_t maxw = calc_dynamic_width(obj, LV_STYLE_MAX_WIDTH, &content_width);
138 w = LV_CLAMP(minw, unclamped_w, maxw);
139
145 bool w_pct = size_in_effect_is_pct(unclamped_w, minw, maxw,
146 lv_obj_get_style_width_internal(obj, LV_PART_MAIN),
147 lv_obj_get_style_min_width_internal(obj, LV_PART_MAIN),
148 lv_obj_get_style_max_width_internal(obj, LV_PART_MAIN));
149 obj->w_ignore_size =
150 (w_pct && parent->w_layout == 0 && lv_obj_get_style_width_internal(parent, 0) == LV_SIZE_CONTENT);
151 }
152
153 int32_t h;
154 if(obj->h_layout) {
155 h = lv_obj_get_height(obj);
156 }
157 else {
158 int32_t content_height = -1;
159 int32_t unclamped_h = calc_dynamic_height(obj, LV_STYLE_HEIGHT, &content_height);
160 int32_t minh = calc_dynamic_height(obj, LV_STYLE_MIN_HEIGHT, &content_height);
161 int32_t maxh = calc_dynamic_height(obj, LV_STYLE_MAX_HEIGHT, &content_height);
162 h = LV_CLAMP(minh, unclamped_h, maxh);
163
169 bool h_pct = size_in_effect_is_pct(unclamped_h, minh, maxh,
170 lv_obj_get_style_height_internal(obj, LV_PART_MAIN),
171 lv_obj_get_style_min_height_internal(obj, LV_PART_MAIN),
172 lv_obj_get_style_max_height_internal(obj, LV_PART_MAIN));
173
174 obj->h_ignore_size = (h_pct && parent->h_layout == 0 &&
175 lv_obj_get_style_height_internal(parent, 0) == LV_SIZE_CONTENT);
176 }
177
178 /*Do nothing if the size is not changed*/
179 /*It is very important else recursive resizing can occur without size change*/
180 if(lv_obj_get_width(obj) == w && lv_obj_get_height(obj) == h)
181 return false;
182
183 /*Invalidate the original area*/
184 lv_obj_invalidate(obj);
185
186 /*Save the original coordinates*/
187 lv_area_t ori;
188 lv_obj_get_coords(obj, &ori);
189
190 /*Check if the object inside the parent or not*/
191 lv_area_t parent_fit_area;
192 lv_obj_get_content_coords(parent, &parent_fit_area);
193
194 /*If the object is already out of the parent and its position is changes
195 *surely the scrollbars also changes so invalidate them*/
196 bool on1 = lv_area_is_in(&ori, &parent_fit_area, 0);
197 if(!on1)
198 lv_obj_scrollbar_invalidate(parent);
199
200 /*Set the length and height
201 *Be sure the content is not scrolled in an invalid position on the new size*/
202 obj->coords.y2 = obj->coords.y1 + h - 1;
203 if(lv_obj_get_style_base_dir_internal(obj, LV_PART_MAIN) == LV_BASE_DIR_RTL) {
204 obj->coords.x1 = obj->coords.x2 - w + 1;
205 }
206 else {
207 obj->coords.x2 = obj->coords.x1 + w - 1;
208 }
209
210 /*Call the ancestor's event handler to the object with its new coordinates*/
211 lv_obj_send_event(obj, LV_EVENT_SIZE_CHANGED, &ori);
212
213 /*Call the ancestor's event handler to the parent too*/
214 lv_obj_send_event(parent, LV_EVENT_CHILD_CHANGED, obj);
215
216 /*Invalidate the new area*/
217 lv_obj_invalidate(obj);
218
219 obj->readjust_scroll_after_layout = 1;
220
221 /*If the object was out of the parent invalidate the new scrollbar area too.
222 *If it wasn't out of the parent but out now, also invalidate the scrollbars*/
223 bool on2 = lv_area_is_in(&obj->coords, &parent_fit_area, 0);
224 if(on1 || (!on1 && on2))
225 lv_obj_scrollbar_invalidate(parent);
226
227 lv_obj_refresh_ext_draw_size(obj);
228
229 return true;
230}

322void lv_obj_update_layout(const lv_obj_t * obj)
323{
324 LV_CHECK_OBJ(obj, MY_CLASS, return);
325
326 if(update_layout_mutex) {
327 LV_LOG_TRACE("Already running, returning");
328 return;
329 }
330 LV_PROFILER_LAYOUT_BEGIN;
331 update_layout_mutex = true;
332
333 lv_obj_t * scr = lv_obj_get_screen(obj);
334 /*Repeat until there are no more layout invalidations*/
335 uint32_t pass_cnt = 0;
336 while(scr->scr_layout_inv) {
337 if(pass_cnt >= LV_OBJ_LAYOUT_UPDATE_MAX_PASSES) {
338 LV_ASSERT_FORMAT_MSG(false,
339 "Layout of screen %p (class: '%s') didn't settle in %d passes, giving up. Some "
340 "sizes probably depend on each other circularly. Please report it on github",
341 (void *)scr, scr->class_p->name, LV_OBJ_LAYOUT_UPDATE_MAX_PASSES);
342 /*Reached only if the assert handler returns*/
343 scr->scr_layout_inv = 0;
344 break;
345 }
346 pass_cnt++;
347 LV_LOG_TRACE("Layout update begin");
348 scr->scr_layout_inv = 0;
349 layout_update_core(scr);
350 LV_LOG_TRACE("Layout update end");
351 }
352
353 lv_display_t * disp = lv_obj_get_display(scr);
354 lv_display_send_event(disp, LV_EVENT_UPDATE_LAYOUT_COMPLETED, NULL);
355 update_layout_mutex = false;
356 LV_PROFILER_LAYOUT_END;
357}

373void lv_obj_align_to(lv_obj_t * obj, const lv_obj_t * base, lv_align_t align, int32_t x_ofs, int32_t y_ofs)
374{
375 LV_CHECK_OBJ(obj, MY_CLASS, return);
376
377 lv_obj_update_layout(obj);
378 if(base == NULL) base = lv_obj_get_parent(obj);
379
380 if(base == NULL) {
381 LV_LOG_WARN("lv_obj_align_to: base is NULL");
382 return;
383 }
384
385 int32_t x = 0;
386 int32_t y = 0;
387
388 lv_obj_t * parent = lv_obj_get_parent(obj);
389 if(parent == NULL) {
390 LV_LOG_WARN("lv_obj_align_to: parent is NULL");
391 return;
392 }
393
394 int32_t pleft = lv_obj_get_style_space_left_internal(parent, LV_PART_MAIN);
395 int32_t ptop = lv_obj_get_style_space_top_internal(parent, LV_PART_MAIN);
396
397 int32_t bleft = lv_obj_get_style_space_left_internal(base, LV_PART_MAIN);
398 int32_t btop = lv_obj_get_style_space_top_internal(base, LV_PART_MAIN);
399
400 if(align == LV_ALIGN_DEFAULT) {
401 if(lv_obj_get_style_base_dir_internal(base, LV_PART_MAIN) == LV_BASE_DIR_RTL) align = LV_ALIGN_TOP_RIGHT;
402 else align = LV_ALIGN_TOP_LEFT;
403 }
404
405 switch(align) {
406 case LV_ALIGN_CENTER:
407 x = lv_obj_get_content_width(base) / 2 - lv_obj_get_width(obj) / 2 + bleft;
408 y = lv_obj_get_content_height(base) / 2 - lv_obj_get_height(obj) / 2 + btop;
409 break;
410
411 case LV_ALIGN_TOP_LEFT:
412 x = bleft;
413 y = btop;
414 break;
415
416 case LV_ALIGN_TOP_MID:
417 x = lv_obj_get_content_width(base) / 2 - lv_obj_get_width(obj) / 2 + bleft;
418 y = btop;
419 break;
420
421 case LV_ALIGN_TOP_RIGHT:
422 x = lv_obj_get_content_width(base) - lv_obj_get_width(obj) + bleft;
423 y = btop;
424 break;
425
426 case LV_ALIGN_BOTTOM_LEFT:
427 x = bleft;
428 y = lv_obj_get_content_height(base) - lv_obj_get_height(obj) + btop;
429 break;
430 case LV_ALIGN_BOTTOM_MID:
431 x = lv_obj_get_content_width(base) / 2 - lv_obj_get_width(obj) / 2 + bleft;
432 y = lv_obj_get_content_height(base) - lv_obj_get_height(obj) + btop;
433 break;
434
435 case LV_ALIGN_BOTTOM_RIGHT:
436 x = lv_obj_get_content_width(base) - lv_obj_get_width(obj) + bleft;
437 y = lv_obj_get_content_height(base) - lv_obj_get_height(obj) + btop;
438 break;
439
440 case LV_ALIGN_LEFT_MID:
441 x = bleft;
442 y = lv_obj_get_content_height(base) / 2 - lv_obj_get_height(obj) / 2 + btop;
443 break;
444
445 case LV_ALIGN_RIGHT_MID:
446 x = lv_obj_get_content_width(base) - lv_obj_get_width(obj) + bleft;
447 y = lv_obj_get_content_height(base) / 2 - lv_obj_get_height(obj) / 2 + btop;
448 break;
449
450 case LV_ALIGN_OUT_TOP_LEFT:
451 x = 0;
452 y = -lv_obj_get_height(obj);
453 break;
454
455 case LV_ALIGN_OUT_TOP_MID:
456 x = lv_obj_get_width(base) / 2 - lv_obj_get_width(obj) / 2;
457 y = -lv_obj_get_height(obj);
458 break;
459
460 case LV_ALIGN_OUT_TOP_RIGHT:
461 x = lv_obj_get_width(base) - lv_obj_get_width(obj);
462 y = -lv_obj_get_height(obj);
463 break;
464
465 case LV_ALIGN_OUT_BOTTOM_LEFT:
466 x = 0;
467 y = lv_obj_get_height(base);
468 break;
469
470 case LV_ALIGN_OUT_BOTTOM_MID:
471 x = lv_obj_get_width(base) / 2 - lv_obj_get_width(obj) / 2;
472 y = lv_obj_get_height(base);
473 break;
474
475 case LV_ALIGN_OUT_BOTTOM_RIGHT:
476 x = lv_obj_get_width(base) - lv_obj_get_width(obj);
477 y = lv_obj_get_height(base);
478 break;
479
480 case LV_ALIGN_OUT_LEFT_TOP:
481 x = -lv_obj_get_width(obj);
482 y = 0;
483 break;
484
485 case LV_ALIGN_OUT_LEFT_MID:
486 x = -lv_obj_get_width(obj);
487 y = lv_obj_get_height(base) / 2 - lv_obj_get_height(obj) / 2;
488 break;
489
490 case LV_ALIGN_OUT_LEFT_BOTTOM:
491 x = -lv_obj_get_width(obj);
492 y = lv_obj_get_height(base) - lv_obj_get_height(obj);
493 break;
494
495 case LV_ALIGN_OUT_RIGHT_TOP:
496 x = lv_obj_get_width(base);
497 y = 0;
498 break;
499
500 case LV_ALIGN_OUT_RIGHT_MID:
501 x = lv_obj_get_width(base);
502 y = lv_obj_get_height(base) / 2 - lv_obj_get_height(obj) / 2;
503 break;
504
505 case LV_ALIGN_OUT_RIGHT_BOTTOM:
506 x = lv_obj_get_width(base);
507 y = lv_obj_get_height(base) - lv_obj_get_height(obj);
508 break;
509
510 case LV_ALIGN_DEFAULT:
511 break;
512 }
513
514 if(LV_COORD_IS_PCT(x_ofs)) x_ofs = (lv_obj_get_width(base) * LV_COORD_GET_PCT(x_ofs)) / 100;
515 if(LV_COORD_IS_PCT(y_ofs)) y_ofs = (lv_obj_get_height(base) * LV_COORD_GET_PCT(y_ofs)) / 100;
516 if(lv_obj_get_style_base_dir_internal(parent, LV_PART_MAIN) == LV_BASE_DIR_RTL) {
517 x += x_ofs + base->coords.x1 - parent->coords.x1 + lv_obj_get_scroll_right(parent) - pleft;
518 }
519 else {
520 x += x_ofs + base->coords.x1 - parent->coords.x1 + lv_obj_get_scroll_left(parent) - pleft;
521 }
522 y += y_ofs + base->coords.y1 - parent->coords.y1 + lv_obj_get_scroll_top(parent) - ptop;
523 lv_obj_set_style_align(obj, LV_ALIGN_TOP_LEFT, 0);
524 lv_obj_set_pos(obj, x, y);
525
526}

780void lv_obj_refr_pos(lv_obj_t * obj)
781{
782 LV_CHECK_OBJ(obj, MY_CLASS, return);
783
784 if(lv_obj_is_layout_positioned(obj)) return;
785
786 lv_obj_t * parent = lv_obj_get_parent(obj);
787 int32_t x = lv_obj_get_style_x_internal(obj, LV_PART_MAIN);
788 int32_t y = lv_obj_get_style_y_internal(obj, LV_PART_MAIN);
789
790 if(parent == NULL) {
791 lv_obj_move_to(obj, x, y);
792 return;
793 }
794
795 /*Handle percentage value*/
796 int32_t pw = lv_obj_get_content_width(parent);
797 int32_t ph = lv_obj_get_content_height(parent);
798 if(LV_COORD_IS_PCT(x)) {
799 if(lv_obj_get_style_width_internal(parent, LV_PART_MAIN) == LV_SIZE_CONTENT) x = 0; /*Avoid circular dependency*/
800 else x = (pw * LV_COORD_GET_PCT(x)) / 100;
801 }
802
803 if(LV_COORD_IS_PCT(y)) {
804 if(lv_obj_get_style_height_internal(parent, LV_PART_MAIN) == LV_SIZE_CONTENT) y = 0; /*Avoid circular dependency*/
805 y = (ph * LV_COORD_GET_PCT(y)) / 100;
806 }
807
808 /*Handle percentage value of translate*/
809 int32_t tr_x = lv_obj_get_style_translate_x_internal(obj, LV_PART_MAIN);
810 int32_t tr_y = lv_obj_get_style_translate_y_internal(obj, LV_PART_MAIN);
811 int32_t w = lv_obj_get_width(obj);
812 int32_t h = lv_obj_get_height(obj);
813 if(LV_COORD_IS_PCT(tr_x)) tr_x = (w * LV_COORD_GET_PCT(tr_x)) / 100;
814 if(LV_COORD_IS_PCT(tr_y)) tr_y = (h * LV_COORD_GET_PCT(tr_y)) / 100;
815
816 /*Use the translation*/
817 x += tr_x;
818 y += tr_y;
819
820 lv_align_t align = lv_obj_get_style_align_internal(obj, LV_PART_MAIN);
821
822 if(align == LV_ALIGN_DEFAULT) {
823 align = LV_ALIGN_TOP_LEFT;
824 }
825
826 bool rtl = lv_obj_get_style_base_dir_internal(parent, LV_PART_MAIN) == LV_BASE_DIR_RTL;
827
828 if(rtl) {
829 switch(align) {
830 case LV_ALIGN_TOP_LEFT:
831 align = LV_ALIGN_TOP_RIGHT;
832 break;
833 case LV_ALIGN_TOP_RIGHT:
834 align = LV_ALIGN_TOP_LEFT;
835 break;
836 case LV_ALIGN_LEFT_MID:
837 align = LV_ALIGN_RIGHT_MID;
838 break;
839 case LV_ALIGN_RIGHT_MID:
840 align = LV_ALIGN_LEFT_MID;
841 break;
842 case LV_ALIGN_BOTTOM_LEFT:
843 align = LV_ALIGN_BOTTOM_RIGHT;
844 break;
845 case LV_ALIGN_BOTTOM_RIGHT:
846 align = LV_ALIGN_BOTTOM_LEFT;
847 break;
848 default:
849 break;
850 }
851 }
852
853 switch(align) {
854 case LV_ALIGN_TOP_LEFT:
855 break;
856 case LV_ALIGN_TOP_MID:
857 x = rtl ? pw / 2 - w / 2 - x : pw / 2 - w / 2 + x;
858 break;
859 case LV_ALIGN_TOP_RIGHT:
860 x = rtl ? pw - w - x : pw - w + x;
861 break;
862 case LV_ALIGN_LEFT_MID:
863 y += ph / 2 - h / 2;
864 break;
865 case LV_ALIGN_BOTTOM_LEFT:
866 y += ph - h;
867 break;
868 case LV_ALIGN_BOTTOM_MID:
869 x = rtl ? pw / 2 - w / 2 - x : pw / 2 - w / 2 + x;
870 y += ph - h;
871 break;
872 case LV_ALIGN_BOTTOM_RIGHT:
873 x = rtl ? pw - w - x : pw - w + x;
874 y += ph - h;
875 break;
876 case LV_ALIGN_RIGHT_MID:
877 x = rtl ? pw - w - x : pw - w + x;
878 y += ph / 2 - h / 2;
879 break;
880 case LV_ALIGN_CENTER:
881 x = rtl ? pw / 2 - w / 2 - x : pw / 2 - w / 2 + x;
882 y += ph / 2 - h / 2;
883 break;
884 default:
885 break;
886 }
887
888 lv_obj_move_to(obj, x, y);
889}

891void lv_obj_move_to(lv_obj_t * obj, int32_t x, int32_t y)
892{
893 LV_CHECK_OBJ(obj, MY_CLASS, return);
894
895 /*Convert x and y to absolute coordinates*/
896 lv_obj_t * parent = obj->parent;
897
898 if(parent) {
899 if(lv_obj_is_floating(obj)) {
900 x += parent->coords.x1;
901 y += parent->coords.y1;
902 }
903 else {
904 x += parent->coords.x1 - lv_obj_get_scroll_x(parent);
905 y += parent->coords.y1 - lv_obj_get_scroll_y(parent);
906 }
907
908 x += lv_obj_get_style_space_left_internal(parent, LV_PART_MAIN);
909 y += lv_obj_get_style_space_top_internal(parent, LV_PART_MAIN);
910 }
911
912 /*Calculate and set the movement*/
913 lv_point_t diff;
914 diff.x = x - obj->coords.x1;
915 diff.y = y - obj->coords.y1;
916
917 /*Do nothing if the position is not changed*/
918 /*It is very important else recursive positioning can
919 *occur without position change*/
920 if(diff.x == 0 && diff.y == 0) return;
921
922 /*Invalidate the original area*/
923 lv_obj_invalidate(obj);
924
925 /*Save the original coordinates*/
926 lv_area_t ori;
927 lv_obj_get_coords(obj, &ori);
928
929 /*Check if the object inside the parent or not*/
930 lv_area_t parent_fit_area;
931 bool on1 = false;
932 if(parent) {
933 lv_obj_get_content_coords(parent, &parent_fit_area);
934
935 /*If the object is already out of the parent and its position is changes
936 *surely the scrollbars also changes so invalidate them*/
937 on1 = lv_area_is_in(&ori, &parent_fit_area, 0);
938 if(!on1) lv_obj_scrollbar_invalidate(parent);
939 }
940
941 obj->coords.x1 += diff.x;
942 obj->coords.y1 += diff.y;
943 obj->coords.x2 += diff.x;
944 obj->coords.y2 += diff.y;
945
946 lv_obj_move_children_by(obj, diff.x, diff.y, false);
947
948 /*Call the ancestor's event handler to the parent too*/
949 if(parent) lv_obj_send_event(parent, LV_EVENT_CHILD_CHANGED, obj);
950
951 /*Invalidate the new area*/
952 lv_obj_invalidate(obj);
953
954 /*If the object was out of the parent invalidate the new scrollbar area too.
955 *If it wasn't out of the parent but out now, also invalidate the scrollbars*/
956 if(parent) {
957 bool on2 = lv_area_is_in(&obj->coords, &parent_fit_area, 0);
958 if(on1 || (!on1 && on2)) lv_obj_scrollbar_invalidate(parent);
959 }
960}

962void lv_obj_move_children_by(lv_obj_t * obj, int32_t x_diff, int32_t y_diff, bool ignore_floating)
963{
964 LV_CHECK_OBJ(obj, MY_CLASS, return);
965
966 uint32_t i;
967 uint32_t child_cnt = lv_obj_get_child_count(obj);
968 for(i = 0; i < child_cnt; i++) {
969 lv_obj_t * child = obj->spec_attr->children[i];
970 if(ignore_floating && lv_obj_is_floating(child)) continue;
971 child->coords.x1 += x_diff;
972 child->coords.y1 += y_diff;
973 child->coords.x2 += x_diff;
974 child->coords.y2 += y_diff;
975
976 lv_obj_move_children_by(child, x_diff, y_diff, false);
977 }

1029/*
1030 * Deferred blur invalidation expansion. A blur object samples the pixels
1031 * behind it, so when anything behind it changes the blur object must be
1032 * redrawn too. Once per frame, after all invalidations are collected, walk
1033 * the tree and add the full extent of any blur object overlapping an
1034 * invalidated area. The walk repeats until a pass adds no new areas, which
1035 * catches transitive cases (a blur object overlapping another blur object
1036 * that overlaps an invalidated area).
1037 *
1038 * Each widget's blur status is cached in obj->has_blur and counted globally in
1039 * blur_obj_cnt (see lv_obj_update_blur_status()), so the walk is skipped
1040 * entirely while no widget has blur and costs one bit test per widget when it
1041 * does run.
1042 */
1043
1044static lv_obj_tree_walk_res_t blur_expand_walk_cb(lv_obj_t * obj, void * user_data)
1045{
1046 if(!obj->has_blur) return LV_OBJ_TREE_WALK_NEXT;
1047
1048 /*invalidate_area_core() expects untransformed coordinates and applies the
1049 *transform itself*/
1050 lv_area_t obj_coords;
1051 int32_t ext_size = lv_obj_get_ext_draw_size(obj);
1052 obj_coords = obj->coords;
1053 lv_area_increase(&obj_coords, ext_size, ext_size);
1054
1055 /*The invalidated areas are in screen coordinates so check the overlap
1056 *against the transformed area*/
1057 lv_area_t scr_coords;
1058 scr_coords = obj_coords;
1059 if(is_transformed(obj)) {
1060 lv_obj_get_transformed_area(obj, &scr_coords, LV_OBJ_POINT_TRANSFORM_FLAG_RECURSIVE);
1061 }
1062
1063 lv_display_t * disp = user_data;
1064 uint32_t i;
1065 for(i = 0; i < disp->inv_p; i++) {
1066 /*The join pass runs after this and its state is reset each frame*/
1067 LV_ASSERT(!disp->inv_area_joined[i]);
1068 if(lv_area_is_on(&disp->inv_areas[i], &scr_coords)) {
1069 invalidate_area_core(obj, &obj_coords);
1070
1071 /*No need to check the children as the widget is already invalidated
1072 *which will redraw the children too*/
1073 return LV_OBJ_TREE_WALK_SKIP_CHILDREN;
1074 }
1075 }
1076
1077 return LV_OBJ_TREE_WALK_NEXT;
1078}
1079
1080void lv_obj_invalidate_expand_blur(lv_display_t * disp)
1081{
1082 if(disp->inv_p == 0) return;
1083
1084 /*There is nothing to expand if no widget has blur or drop shadow*/
1085 if(LV_GLOBAL_DEFAULT()->blur_obj_cnt == 0) return;
1086
1087 uint32_t prev_inv_p;
1088 do {
1089 prev_inv_p = disp->inv_p;
1090
1091 lv_obj_tree_walk(disp->act_scr, blur_expand_walk_cb, disp);
1092 if(disp->prev_scr) lv_obj_tree_walk(disp->prev_scr, blur_expand_walk_cb, disp);
1093 lv_obj_tree_walk(disp->sys_layer, blur_expand_walk_cb, disp);
1094 lv_obj_tree_walk(disp->top_layer, blur_expand_walk_cb, disp);
1095 lv_obj_tree_walk(disp->bottom_layer, blur_expand_walk_cb, disp);
1096
1097 /*Repeat while new areas keep being added. An overflow in lv_inv_area()
1098 *can instead shrink inv_p (it collapses to a single whole-screen area)
1099 *and the loop exits, which is correct because the whole screen covers
1100 *every blur object.*/
1101 } while(disp->inv_p > prev_inv_p);
1102}

1150 if(obj_scr != lv_display_get_screen_active(disp) &&
1151 obj_scr != lv_display_get_screen_prev(disp) &&
1152 obj_scr != lv_display_get_layer_bottom(disp) &&
1153 obj_scr != lv_display_get_layer_top(disp) &&
1154 obj_scr != lv_display_get_layer_sys(disp)) {
1155 return false;
1156 }
1157
1158 /*Truncate the area to the object*/
1159 lv_area_t obj_coords;
1160 int32_t ext_size = lv_obj_get_ext_draw_size(obj);
1161 obj_coords = obj->coords;
1162 lv_area_increase(&obj_coords, ext_size, ext_size);
1163
1164 /*The area is not on the object*/
1165 if(!lv_area_intersect(area, area, &obj_coords)) return false;
1166
1167 if(is_transformed(obj)) {
1168 lv_obj_get_transformed_area(obj, area, LV_OBJ_POINT_TRANSFORM_FLAG_RECURSIVE);
1169 }
1170
1171 /*Truncate recursively to the parents*/
1172 lv_obj_t * parent = lv_obj_get_parent(obj);
1173 while(parent != NULL) {
1174 /*If the parent is hidden then the child is hidden and won't be drawn*/
1175 if(lv_obj_is_hidden(parent)) return false;
1176
1177 /*Truncate to the parent and if no common parts break*/
1178 lv_area_t parent_coords = parent->coords;
1179 if(lv_obj_is_overflow_visible(parent)) {
1180 int32_t parent_ext_size = lv_obj_get_ext_draw_size(parent);
1181 lv_area_increase(&parent_coords, parent_ext_size, parent_ext_size);
1182 }
1183
1184 if(is_transformed(parent)) {
1185 lv_obj_get_transformed_area(parent, &parent_coords, LV_OBJ_POINT_TRANSFORM_FLAG_RECURSIVE);
1186 }
1187 if(!lv_area_intersect(area, area, &parent_coords)) return false;
1188
1189 parent = lv_obj_get_parent(parent);
1190 }
1191
1192 return true;
1193}

1381static int32_t calc_content_width(lv_obj_t * obj)

1388

1391
1392 int32_t child_res = LV_COORD_MIN;
1393
1394 if(!lv_layout_get_min_size(obj, &child_res, true)) {

1427 }

1430 }
1431 child_res = LV_MAX(child_res, child_res_tmp + lv_obj_get_style_margin_left_internal(child, LV_PART_MAIN));
1432 }
1433 if(child_res != LV_COORD_MIN) {
1434 child_res += space_left;
1435 }
1436 }
1437 /*Else find the right most coordinate*/
1438 else {
1439 for(i = 0; i < child_cnt; i++) {
1440 int32_t child_res_tmp = LV_COORD_MIN;
1441 lv_obj_t * child = obj->spec_attr->children[i];
1442 if(child->hidden || child->floating) continue;
1443

1451 case LV_ALIGN_TOP_LEFT:
1452 case LV_ALIGN_BOTTOM_LEFT:
1453 case LV_ALIGN_LEFT_MID:
1454 /*Normal left aligns.*/
1455 child_res_tmp = child->coords.x2 - obj->coords.x1 + 1;
1456 break;
1457 default:
1458 /* Consider other cases only if x=0 and use the width of the object.
1459 * With x!=0 circular dependency could occur. */
1460 if(lv_obj_get_style_x_internal(child, LV_PART_MAIN) == 0) {
1461 child_res_tmp = lv_area_get_width(&child->coords) + space_left;
1462 child_res_tmp += lv_obj_get_style_margin_right_internal(child, LV_PART_MAIN);
1463 }
1464 break;
1465 }
1466 }
1467 else {
1468 child_res_tmp = child->coords.x2 - obj->coords.x1 + 1;
1469 }
1470
1471 child_res = LV_MAX(child_res, child_res_tmp + lv_obj_get_style_margin_right_internal(child, LV_PART_MAIN));

1475 child_res += space_right;
1476 }
1477 }
1478 }
1479
1480 if(obj->spec_attr) {
1481 obj->spec_attr->scroll.x = -scroll_x_tmp;
1482 }
1483
1484 if(child_res == LV_COORD_MIN) {

1486 }
1487
1488 return LV_MAX(child_res, self_w);

1583static void transform_point_array(const lv_obj_t * obj, lv_point_t * p, size_t p_count, bool inv)
1584{
1585#if LV_DRAW_TRANSFORM_USE_MATRIX
1586 const lv_matrix_t * obj_matrix = lv_obj_get_transform(obj);
1587 if(obj_matrix) {
1588 lv_matrix_t m;
1589 lv_matrix_identity(&m);
1590 lv_matrix_translate(&m, obj->coords.x1, obj->coords.y1);
1591 lv_matrix_multiply(&m, obj_matrix);
1592 lv_matrix_translate(&m, -obj->coords.x1, -obj->coords.y1);
1593
1594 if(inv) {
1595 lv_matrix_t inv_m;
1596 lv_matrix_inverse(&inv_m, &m);
1597 m = inv_m;
1598 }
1599
1600 for(size_t i = 0; i < p_count; i++) {
1601 lv_point_precise_t p_precise = lv_point_to_precise(&p[i]);
1602 lv_point_precise_t res = lv_matrix_transform_precise_point(&m, &p_precise);
1603 p[i] = lv_point_from_precise(&res);
1604 }
1605
1606 return;
1607 }
1608#endif /* LV_DRAW_TRANSFORM_USE_MATRIX */
1609
1610 int32_t angle = lv_obj_get_style_transform_rotation_internal(obj, LV_PART_MAIN);
1611 int32_t scale_x = lv_obj_get_style_transform_scale_x_safe_internal(obj, LV_PART_MAIN);
1612 int32_t scale_y = lv_obj_get_style_transform_scale_y_safe_internal(obj, LV_PART_MAIN);
1613 if(scale_x == 0) scale_x = 1;
1614 if(scale_y == 0) scale_y = 1;
1615
1616 if(angle == 0 && scale_x == LV_SCALE_NONE && scale_y == LV_SCALE_NONE) return;
1617
1618 lv_point_t pivot = {
1619 .x = lv_obj_get_style_transform_pivot_x_internal(obj, LV_PART_MAIN),
1620 .y = lv_obj_get_style_transform_pivot_y_internal(obj, LV_PART_MAIN)
1621 };
1622
1623 if(LV_COORD_IS_PCT(pivot.x)) {
1624 pivot.x = (LV_COORD_GET_PCT(pivot.x) * lv_area_get_width(&obj->coords)) / 100;
1625 }
1626 if(LV_COORD_IS_PCT(pivot.y)) {
1627 pivot.y = (LV_COORD_GET_PCT(pivot.y) * lv_area_get_height(&obj->coords)) / 100;
1628 }
1629
1630 pivot.x = obj->coords.x1 + pivot.x;
1631 pivot.y = obj->coords.y1 + pivot.y;
1632
1633 if(inv) {
1634 angle = -angle;
1635 scale_x = (256 * 256 + scale_x - 1) / scale_x;
1636 scale_y = (256 * 256 + scale_y - 1) / scale_y;
1637 }
1638
1639 lv_point_array_transform(p, p_count, angle, scale_x, scale_y, &pivot, !inv);
1640}

1751static bool size_in_effect_is_pct(int32_t unclamped, int32_t min, int32_t max, int32_t size_style,
1752 int32_t min_style, int32_t max_style)
1753{
1754 /*If the bounds are inverted the clamp always returns the min size*/
1755 if(min > max) return LV_COORD_IS_PCT(min_style);
1756
1757 /*The size is set by the bound which clamped it*/
1758 if(unclamped < min) return LV_COORD_IS_PCT(min_style);
1759 if(unclamped > max) return LV_COORD_IS_PCT(max_style);
