83static void test_outline_sizes_event_cb(lv_event_t * e)
84{
85 lv_freetype_outline_event_param_t * param = lv_event_get_param(e);
86
87 switch(lv_event_get_code(e)) {
88 case LV_EVENT_CREATE: {
89 /*
90 * At this point, sizes MUST already be calculated.
91 * Assert directly — if this fires, the regression is back.
92 *
93 * Expected values for NotoSansSC-Regular.ttf 'A' at ref_size=128:
94 * segments_size = 18 (16 path segments + 2 contour closes)
95 * data_size = 44 (22 vectors * 2)
96 */
97 TEST_ASSERT_EQUAL_INT32(18, param->sizes.segments_size);
98 TEST_ASSERT_EQUAL_INT32(44, param->sizes.data_size);
99
100 outline_test_obj_t * obj = lv_malloc_zeroed(sizeof(outline_test_obj_t));
101 TEST_ASSERT_NOT_NULL(obj);
102 obj->points = expected_outline_points;
103 param->outline = obj;
104 }
105 break;
106 case LV_EVENT_DELETE: {
107 outline_test_obj_t * obj = param->outline;
108 TEST_ASSERT_NOT_NULL(obj);
109 TEST_ASSERT_EQUAL_PTR(expected_outline_points, obj->points);
110 TEST_ASSERT_EQUAL_UINT32(EXPECTED_OUTLINE_POINTS_COUNT, obj->index);
111 lv_free(obj);
112 param->outline = NULL;
113 }
114 break;
115 case LV_EVENT_INSERT: {
116 outline_test_obj_t * obj = param->outline;
117 TEST_ASSERT_NOT_NULL(obj);
118 TEST_ASSERT_LESS_THAN_UINT32(EXPECTED_OUTLINE_POINTS_COUNT, obj->index);
119 TEST_ASSERT_EQUAL_PTR(expected_outline_points, obj->points);
120
121 const expected_outline_point_t * exp = &obj->points[obj->index];
122 TEST_ASSERT_EQUAL_INT32(exp->type, param->type);
123 TEST_ASSERT_EQUAL_INT32(exp->to.x, param->to.x);
124 TEST_ASSERT_EQUAL_INT32(exp->to.y, param->to.y);
125 TEST_ASSERT_EQUAL_INT32(exp->control1.x, param->control1.x);
126 TEST_ASSERT_EQUAL_INT32(exp->control1.y, param->control1.y);
127 TEST_ASSERT_EQUAL_INT32(exp->control2.x, param->control2.x);
128 TEST_ASSERT_EQUAL_INT32(exp->control2.y, param->control2.y);
129 obj->index++;
130 }
131 break;
132 default:
133 break;
134 }
135}

137void test_freetype_outline_sizes_available_on_create(void)
138{
139 /* Register our test event callback (overrides the draw unit's callback) */
140 lv_freetype_outline_add_event(test_outline_sizes_event_cb, LV_EVENT_ALL, NULL);
141
142 /* Create an outline font - this will trigger the outline event */
143 lv_font_t * font = lv_freetype_font_create("./src/test_files/fonts/noto/NotoSansSC-Regular.ttf",
144 LV_FREETYPE_FONT_RENDER_MODE_OUTLINE,
145 24,
146 LV_FREETYPE_FONT_STYLE_NORMAL);
147 TEST_ASSERT_NOT_NULL(font);
148
149 /*
150 * Trigger outline creation directly via the font glyph API.
151 * Do NOT render (lv_refr_now) because the SW draw unit would try to use
152 * our dummy outline object as a real lv_vector_path_t.
153 */
154 lv_font_glyph_dsc_t g_dsc;
155 bool found = lv_font_get_glyph_dsc(font, &g_dsc, 'A', '\0');
156 TEST_ASSERT_TRUE(found);
157 TEST_ASSERT_EQUAL(LV_FONT_GLYPH_FORMAT_VECTOR, g_dsc.format);
158
159 /* get_glyph_bitmap triggers outline_create -> our event callback */
160 const void * bitmap = lv_font_get_glyph_bitmap(&g_dsc, NULL);
161 TEST_ASSERT_NOT_NULL(bitmap);
162
163 /* Release the glyph (decrements cache ref count) */
164 lv_font_glyph_release_draw_data(&g_dsc);
165
166 /* Deleting the font destroys the cache, triggering LV_EVENT_DELETE */
167 lv_freetype_font_delete(font);
168}
