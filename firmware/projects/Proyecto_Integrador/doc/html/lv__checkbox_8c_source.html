85void lv_checkbox_set_text(lv_obj_t * obj, const char * txt)
86{
87 LV_CHECK_OBJ(obj, MY_CLASS, return);
88
89 lv_checkbox_t * cb = (lv_checkbox_t *)obj;
90
91 if(NULL != txt) {
92 size_t len;
93
94#if LV_USE_ARABIC_PERSIAN_CHARS
95 len = lv_text_ap_strlen(txt) + 1;
96#else
97 len = lv_strlen(txt) + 1;
98#endif
99
100 if(!cb->static_txt) cb->txt = lv_realloc(cb->txt, len);
101 else cb->txt = lv_malloc(len);
102
103 LV_ASSERT_MALLOC(cb->txt);
104 if(NULL == cb->txt) return;
105
106#if LV_USE_ARABIC_PERSIAN_CHARS
107 lv_text_ap_proc(txt, cb->txt);
108#else
109 lv_strcpy(cb->txt, txt);
110#endif
111
112 cb->static_txt = 0;
113 }
114
115 lv_obj_refresh_self_size(obj);
116 lv_obj_invalidate(obj);
117}

189static void lv_checkbox_event(const lv_obj_class_t * class_p, lv_event_t * e)
190{
191 LV_UNUSED(class_p);
192 LV_ASSERT(e != NULL);
193
194 lv_result_t res;
195 /*Call the ancestor's event handler*/
196 res = lv_obj_event_base(MY_CLASS, e);
197 if(res != LV_RESULT_OK) return;
198
199 lv_event_code_t code = lv_event_get_code(e);
200 lv_obj_t * obj = lv_event_get_current_target(e);
201
202 if(code == LV_EVENT_GET_SELF_SIZE) {
203 lv_point_t * p = lv_event_get_param(e);
204 lv_checkbox_t * cb = (lv_checkbox_t *)obj;
205
206 const lv_font_t * font = lv_obj_get_style_text_font_internal(obj, LV_PART_MAIN);
207 int32_t font_h = lv_font_get_line_height_internal(font);
208 lv_text_attributes_t attributes = {0};
209
210 attributes.line_space = lv_obj_get_style_text_line_space_internal(obj, LV_PART_MAIN);
211 attributes.letter_space = lv_obj_get_style_text_letter_space_internal(obj, LV_PART_MAIN);
212 attributes.max_width = LV_COORD_MAX;
213 attributes.text_flags = LV_TEXT_FLAG_NONE;
214
215 lv_point_t txt_size;
216
217 lv_text_get_size_attributes(&txt_size, cb->txt, font, &attributes);
218
219 int32_t bg_colp = lv_obj_get_style_pad_column_internal(obj, LV_PART_MAIN);
220 int32_t marker_leftp = lv_obj_get_style_pad_left_internal(obj, LV_PART_INDICATOR);
221 int32_t marker_rightp = lv_obj_get_style_pad_right_internal(obj, LV_PART_INDICATOR);
222 int32_t marker_topp = lv_obj_get_style_pad_top_internal(obj, LV_PART_INDICATOR);
223 int32_t marker_bottomp = lv_obj_get_style_pad_bottom_internal(obj, LV_PART_INDICATOR);
224 lv_point_t marker_size;
225 marker_size.x = font_h + marker_leftp + marker_rightp;
226 marker_size.y = font_h + marker_topp + marker_bottomp;
227
228 p->x = marker_size.x + txt_size.x + bg_colp;
229 p->y = LV_MAX(marker_size.y, txt_size.y);
230 }
231 else if(code == LV_EVENT_REFR_EXT_DRAW_SIZE) {
232 int32_t * s = lv_event_get_param(e);
233 int32_t m = lv_obj_calculate_ext_draw_size(obj, LV_PART_INDICATOR);
234 *s = LV_MAX(*s, m);
235 }
236 else if(code == LV_EVENT_DRAW_MAIN) {
237 lv_checkbox_draw(e);
238 }
239}
