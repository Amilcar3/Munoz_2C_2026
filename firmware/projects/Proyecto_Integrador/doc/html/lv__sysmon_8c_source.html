237static void perf_monitor_disp_event_cb(lv_event_t * e)
238{
239 LV_ASSERT(e != NULL);
240 lv_display_t * disp = lv_event_get_target(e);
241 LV_ASSERT(disp != NULL);
242 lv_sysmon_perf_info_t * info = &disp->perf_sysmon_info;
243 LV_ASSERT(info != NULL);
244
245 lv_event_code_t code = lv_event_get_code(e);
246
247 switch(code) {
248 case LV_EVENT_REFR_START:
249 info->measured.refr_interval_sum += lv_tick_elaps(info->measured.refr_start);
250 info->measured.refr_start = lv_tick_get();
251 break;
252 case LV_EVENT_REFR_READY:
253 info->measured.refr_elaps_sum += lv_tick_elaps(info->measured.refr_start);
254 info->measured.refr_cnt++;
255 break;
256 case LV_EVENT_RENDER_START:
257 info->measured.render_in_progress = 1;
258 info->measured.render_start = lv_tick_get();
259 break;
260 case LV_EVENT_RENDER_READY:
261 info->measured.render_in_progress = 0;
262 info->measured.render_elaps_sum += lv_tick_elaps(info->measured.render_start);
263 info->measured.render_cnt++;
264 break;
265 case LV_EVENT_FLUSH_START:
266 case LV_EVENT_FLUSH_WAIT_START:
267 if(info->measured.render_in_progress) {
268 info->measured.flush_in_render_start = lv_tick_get();
269 }
270 else {
271 info->measured.flush_not_in_render_start = lv_tick_get();
272 }
273 break;
274 case LV_EVENT_FLUSH_FINISH:
275 case LV_EVENT_FLUSH_WAIT_FINISH:
276 if(info->measured.render_in_progress) {
277 info->measured.flush_in_render_elaps_sum += lv_tick_elaps(info->measured.flush_in_render_start);
278 }
279 else {
280 info->measured.flush_not_in_render_elaps_sum += lv_tick_elaps(info->measured.flush_not_in_render_start);
281 }
282 break;
283 case LV_EVENT_DELETE:
284 lv_timer_delete(disp->perf_sysmon_backend.timer);
285 lv_subject_delete(disp->perf_sysmon_backend.subject);
286 break;
287 default:
288 break;
289 }
290}

292static void perf_dump_info(lv_display_t * disp)
293{
294 LV_ASSERT(disp != NULL);
295 lv_sysmon_perf_info_t * info = &disp->perf_sysmon_info;
296 LV_ASSERT(info != NULL);
297 info->calculated.run_cnt++;
298
299 uint32_t time_since_last_report = lv_tick_elaps(info->measured.last_report_timestamp);
300 lv_timer_t * disp_refr_timer = lv_display_get_refr_timer(lv_display_get_default());
301 uint32_t disp_refr_period = disp_refr_timer ? disp_refr_timer->period : LV_DEF_REFR_PERIOD;
302
303 info->calculated.fps = time_since_last_report ? (1000 * info->measured.refr_cnt / time_since_last_report) : 0;
304 info->calculated.fps = LV_MIN(info->calculated.fps,
305 1000 / disp_refr_period); /*Limit due to possible off-by-one error*/
306
307 info->calculated.cpu = 100 - LV_SYSMON_GET_IDLE();
308#if LV_SYSMON_PROC_IDLE_AVAILABLE
309 info->calculated.cpu_proc = 100 - LV_SYSMON_GET_PROC_IDLE();
310#endif /*LV_SYSMON_PROC_IDLE_AVAILABLE*/
311 info->calculated.refr_avg_time = info->measured.refr_cnt ? (info->measured.refr_elaps_sum / info->measured.refr_cnt) :
312 0;
313
314 info->calculated.flush_avg_time = info->measured.render_cnt ?
315 ((info->measured.flush_in_render_elaps_sum + info->measured.flush_not_in_render_elaps_sum)
316 / info->measured.render_cnt) : 0;
317 /*Flush time was measured in rendering time so subtract it*/
318 info->calculated.render_avg_time = info->measured.render_cnt ? ((info->measured.render_elaps_sum -
319 info->measured.flush_in_render_elaps_sum) /
320 info->measured.render_cnt) : 0;
321
322 info->calculated.cpu_avg_total = ((info->calculated.cpu_avg_total * (info->calculated.run_cnt - 1)) +
323 info->calculated.cpu) / info->calculated.run_cnt;
324 info->calculated.fps_avg_total = ((info->calculated.fps_avg_total * (info->calculated.run_cnt - 1)) +
325 info->calculated.fps) / info->calculated.run_cnt;
326
327 lv_subject_set_pointer(disp->perf_sysmon_backend.subject, info);
328
329 lv_sysmon_perf_info_t prev_info = *info;
330 lv_memzero(info, sizeof(lv_sysmon_perf_info_t));
331 info->measured.refr_start = prev_info.measured.refr_start;
332 info->calculated.cpu_avg_total = prev_info.calculated.cpu_avg_total;
333#if LV_SYSMON_PROC_IDLE_AVAILABLE
334 info->calculated.cpu_proc = prev_info.calculated.cpu_proc;
335#endif /*LV_SYSMON_PROC_IDLE_AVAILABLE*/
336 info->calculated.fps_avg_total = prev_info.calculated.fps_avg_total;
337 info->calculated.run_cnt = prev_info.calculated.run_cnt;
338
339 info->measured.last_report_timestamp = lv_tick_get();
340}

351static void perf_observer_cb(lv_observer_t * observer, lv_subject_t * subject)
352{
353 LV_ASSERT(observer != NULL);
354 LV_ASSERT(subject != NULL);
355 const lv_sysmon_perf_info_t * perf = lv_subject_get_pointer(subject);
356 LV_ASSERT(perf != NULL);
357
358#if LV_USE_PERF_MONITOR_LOG_MODE
359 LV_UNUSED(observer);
360#if LV_SYSMON_PROC_IDLE_AVAILABLE
361 LV_LOG("sysmon: "
362 "%" LV_PRIu32 " FPS (refr_cnt: %" LV_PRIu32 " | redraw_cnt: %" LV_PRIu32"), "
363 "refr %" LV_PRIu32 "ms (render %" LV_PRIu32 "ms | flush %" LV_PRIu32 "ms), "
364 "CPU (total %" LV_PRIu32 "%% proc %" LV_PRIu32 "%%)\n",
365 perf->calculated.fps, perf->measured.refr_cnt, perf->measured.render_cnt,
366 perf->calculated.refr_avg_time, perf->calculated.render_avg_time, perf->calculated.flush_avg_time,
367 perf->calculated.cpu, perf->calculated.cpu_proc);
368#else
369 LV_LOG("sysmon: "
370 "%" LV_PRIu32 " FPS (refr_cnt: %" LV_PRIu32 " | redraw_cnt: %" LV_PRIu32"), "
371 "refr %" LV_PRIu32 "ms (render %" LV_PRIu32 "ms | flush %" LV_PRIu32 "ms), "
372 "CPU %" LV_PRIu32 "%%\n",
373 perf->calculated.fps, perf->measured.refr_cnt, perf->measured.render_cnt,
374 perf->calculated.refr_avg_time, perf->calculated.render_avg_time, perf->calculated.flush_avg_time,
375 perf->calculated.cpu);
376#endif
377#else
378 lv_obj_t * label = lv_observer_get_target(observer);
379 LV_ASSERT(label != NULL);
380#if LV_SYSMON_PROC_IDLE_AVAILABLE
381 lv_label_set_text_fmt(
382 label,
383 "%" LV_PRIu32" FPS | CPU (%" LV_PRIu32 "%% | %" LV_PRIu32 "%%)\n"
384 "%" LV_PRIu32" ms (%" LV_PRIu32" | %" LV_PRIu32")",
385 perf->calculated.fps, perf->calculated.cpu, perf->calculated.cpu_proc,
386 perf->calculated.render_avg_time + perf->calculated.flush_avg_time,
387 perf->calculated.render_avg_time, perf->calculated.flush_avg_time
388 );
389#else
390 lv_label_set_text_fmt(
391 label,
392 "%" LV_PRIu32" FPS, %" LV_PRIu32 "%% CPU\n"
393 "%" LV_PRIu32" ms (%" LV_PRIu32" | %" LV_PRIu32")",
394 perf->calculated.fps, perf->calculated.cpu,
395 perf->calculated.render_avg_time + perf->calculated.flush_avg_time,
396 perf->calculated.render_avg_time, perf->calculated.flush_avg_time
397 );
398#endif /*LV_SYSMON_PROC_IDLE_AVAILABLE*/
399#endif /*LV_USE_PERF_MONITOR_LOG_MODE*/
400}

428static void mem_observer_cb(lv_observer_t * observer, lv_subject_t * subject)
429{
430 LV_ASSERT(observer != NULL);
431 LV_ASSERT(subject != NULL);
432 lv_obj_t * label = lv_observer_get_target(observer);

435 LV_ASSERT(mon != NULL);
436
437 size_t used_size = mon->total_size - mon->free_size;
438 size_t used_kb = used_size / 1024;
439 size_t used_kb_tenth = (used_size - (used_kb * 1024)) / 102;
440 size_t max_used_kb = mon->max_used / 1024;
441 size_t max_used_kb_tenth = (mon->max_used - (max_used_kb * 1024)) / 102;
442 lv_label_set_text_fmt(label,
443 "%zu.%zu kB (%d%%)\n"
444 "%zu.%zu kB max, %d%% frag.",
445 used_kb, used_kb_tenth, mon->used_pct,
446 max_used_kb, max_used_kb_tenth,
447 mon->frag_pct);
448}
