103void LV_ATTRIBUTE_FAST_MEM lv_draw_label(lv_layer_t * layer, const lv_draw_label_dsc_t * dsc,
104 const lv_area_t * coords)
105{
106 LV_CHECK_ARG(layer != NULL, return);
107 LV_CHECK_ARG(dsc != NULL, return);
108 LV_CHECK_ARG(coords != NULL, return);
109
110 if(dsc->opa <= LV_OPA_MIN) return;
111 if(dsc->text == NULL || dsc->text[0] == '\0') return;
112 if(dsc->font == NULL) {
113 LV_LOG_WARN("dsc->font == NULL");
114 return;
115 }
116 if(coords->x1 > coords->x2) {
117 /* Attempting to draw a label too small (negative width), cancel to avoid error */
118 return;
119 }
120 LV_PROFILER_DRAW_BEGIN;
121
122 if(dsc->base.drop_shadow_opa) {
123 lv_layer_t * ds_layer = lv_draw_layer_create_drop_shadow(layer, &dsc->base, coords);
124 LV_ASSERT_NULL(ds_layer);
125 lv_draw_label_dsc_t ds_dsc = *dsc;
126 ds_dsc.base.drop_shadow_opa = 0; /*Disable drop shadow so rendering below will render plain label*/
127 lv_draw_label(ds_layer, &ds_dsc, coords);
128 lv_draw_layer_finish_drop_shadow(ds_layer, &dsc->base);
129 }
130
131
132 lv_draw_task_t * t = lv_draw_add_task(layer, coords, LV_DRAW_TASK_TYPE_LABEL);
133
134 lv_memcpy(t->draw_dsc, dsc, sizeof(*dsc));
135
136 /*The text is stored in a local variable so malloc memory for it*/
137 if(dsc->text_local) {
138 lv_draw_label_dsc_t * new_dsc = t->draw_dsc;
139 new_dsc->text = lv_strndup(dsc->text, dsc->text_length);
140 LV_ASSERT_MALLOC(new_dsc->text);
141 }
142
143 lv_draw_finalize_task_creation(layer, t);
144 LV_PROFILER_DRAW_END;
145}

147void LV_ATTRIBUTE_FAST_MEM lv_draw_character(lv_layer_t * layer, lv_draw_label_dsc_t * dsc,
148 const lv_point_t * point, uint32_t unicode_letter)
149{
150 LV_CHECK_ARG(layer != NULL, return);
151 LV_CHECK_ARG(dsc != NULL, return);
152 LV_CHECK_ARG(point != NULL, return);
153
154 if(dsc->opa <= LV_OPA_MIN) return;
155 if(dsc->font == NULL) {
156 LV_LOG_WARN("dsc->font == NULL");
157 return;
158 }
159
160 if(lv_text_is_marker(unicode_letter)) return;
161
162 LV_PROFILER_DRAW_BEGIN;
163
164 lv_font_glyph_dsc_t g;
165
166 lv_font_get_glyph_dsc_internal(dsc->font, &g, unicode_letter, 0);
167
168 lv_area_t a;
169 a.x1 = point->x;
170 a.y1 = point->y;
171 a.x2 = a.x1 + g.adv_w;
172 a.y2 = a.y1 + lv_font_get_line_height_internal(g.resolved_font ? g.resolved_font : dsc->font);
173
174 /*lv_draw_label needs UTF8 text so convert the Unicode character to an UTF8 string */
175 uint32_t letter_buf[2];
176 letter_buf[0] = lv_text_unicode_to_encoded(unicode_letter);
177 letter_buf[1] = '\0';
178
179 const char * letter_buf_char = (const char *)letter_buf;
180
181#if LV_BIG_ENDIAN_SYSTEM
182 while(*letter_buf_char == 0) ++letter_buf_char;
183#endif
184
185 dsc->text = letter_buf_char;
186 dsc->text_local = 1;
187
188 lv_draw_label(layer, dsc, &a);
189 LV_PROFILER_DRAW_END;
190}

192void LV_ATTRIBUTE_FAST_MEM lv_draw_letter(lv_layer_t * layer, lv_draw_letter_dsc_t * dsc, const lv_point_t * point)
193{
194 LV_CHECK_ARG(layer != NULL, return);
195 LV_CHECK_ARG(dsc != NULL, return);
196 LV_CHECK_ARG(point != NULL, return);
197
198 if(dsc->opa <= LV_OPA_MIN) return;
199 if(dsc->font == NULL) {
200 LV_LOG_WARN("dsc->font == NULL");
201 return;
202 }
203
204 const lv_font_t * font = dsc->font;
205
206 LV_PROFILER_DRAW_BEGIN;
207 lv_font_glyph_dsc_t g;
208
209 lv_font_get_glyph_dsc_internal(font, &g, dsc->unicode, 0);
210
211 font = g.resolved_font ? g.resolved_font : dsc->font;
212
213 lv_area_t a;
214 a.x1 = point->x;
215 a.y1 = point->y;
216 a.x2 = a.x1 + g.adv_w;
217 a.y2 = a.y1 + lv_font_get_line_height_internal(font);
218
219 dsc->pivot.x = g.adv_w / 2 ;
220 dsc->pivot.y = font->line_height - font->base_line;
221
222 lv_draw_task_t * t = lv_draw_add_task(layer, &a, LV_DRAW_TASK_TYPE_LETTER);
223
224 lv_memcpy(t->draw_dsc, dsc, sizeof(*dsc));
225
226 lv_draw_finalize_task_creation(layer, t);
227 LV_PROFILER_DRAW_END;
228}

230void lv_draw_label_iterate_characters(lv_draw_task_t * t, const lv_draw_label_dsc_t * dsc,
231 const lv_area_t * coords,
232 lv_draw_glyph_cb_t cb)
233{
234 LV_CHECK_ARG(t != NULL, return);
235 LV_CHECK_ARG(dsc != NULL, return);
236 LV_CHECK_ARG(coords != NULL, return);
237
238 lv_draw_dsc_base_t * base_dsc = t->draw_dsc;
239 const lv_font_t * font = dsc->font;
240 int32_t w;
241
242 lv_area_t clipped_area;
243 bool clip_ok = lv_area_intersect(&clipped_area, coords, &t->clip_area);
244 if(!clip_ok) return;
245
246 lv_text_align_t align = dsc->align;
247 lv_base_dir_t base_dir = dsc->bidi_dir;
248
249 lv_bidi_calculate_align_internal(&align, &base_dir, dsc->text);
250
251 if((dsc->flag & LV_TEXT_FLAG_EXPAND) == 0) {
252 /*Normally use the label's width as width*/
253 w = lv_area_get_width(coords);
254 }
255 else {
256 /*If EXPAND is enabled then not limit the text's width to the object's width*/
257 if(base_dsc->obj && !lv_obj_is_send_draw_task_events(base_dsc->obj)) {
258 w = dsc->text_size.x;
259 }
260 else {
261 lv_text_attributes_t attributes = {0};
262
263 attributes.letter_space = dsc->letter_space;
264 attributes.line_space = dsc->line_space;
265 attributes.max_width = LV_COORD_MAX;
266 attributes.text_flags = dsc->flag;
267
268 lv_point_t p;
269 lv_text_get_size_attributes(&p, dsc->text, dsc->font, &attributes);
270 w = p.x;
271 }
272 }
273
274 int32_t line_height_font = lv_font_get_line_height_internal(font);
275 int32_t line_height = line_height_font + dsc->line_space;
276
277 /*Init variables for the first line*/
278 int32_t line_width = 0;
279 lv_point_t pos;
280 lv_point_set(&pos, coords->x1, coords->y1);
281
282 int32_t x_ofs = 0;
283 int32_t y_ofs = 0;
284 x_ofs = dsc->ofs_x;
285 y_ofs = dsc->ofs_y;
286 pos.y += y_ofs;
287
288 uint32_t line_start = 0;
289 int32_t last_line_start = -1;
290
291 /*Check the hint to use the cached info*/
292 if(dsc->hint && y_ofs == 0 && coords->y1 < 0) {
293 /*If the label changed too much recalculate the hint.*/
294 if(LV_ABS(dsc->hint->coord_y - coords->y1) > LV_LABEL_HINT_UPDATE_TH - 2 * line_height) {
295 dsc->hint->line_start = -1;
296 }
297 last_line_start = dsc->hint->line_start;
298 }
299
300 /*Use the hint if it's valid*/
301 if(dsc->hint && last_line_start >= 0) {
302 line_start = last_line_start;
303 pos.y += dsc->hint->y;
304 }
305
306 uint32_t remaining_len = dsc->text_length;
307 lv_text_attributes_t attributes = {0};
308 attributes.letter_space = dsc->letter_space;
309 attributes.text_flags = dsc->flag;
310 attributes.max_width = w;
311
312 uint32_t line_end = line_start + lv_text_get_next_line(&dsc->text[line_start], remaining_len, font, NULL, &attributes);
313
314 /*Go the first visible line*/
315 while(pos.y + line_height_font < t->clip_area.y1) {
316 /*Go to next line*/
317 remaining_len -= line_end - line_start;
318 line_start = line_end;
319 line_end += lv_text_get_next_line(&dsc->text[line_start], remaining_len, font, NULL, &attributes);
320 pos.y += line_height;
321
322 /*Save at the threshold coordinate*/
323 if(dsc->hint && pos.y >= -LV_LABEL_HINT_UPDATE_TH && dsc->hint->line_start < 0) {
324 dsc->hint->line_start = line_start;
325 dsc->hint->y = pos.y - coords->y1;
326 dsc->hint->coord_y = coords->y1;
327 }
328
329 if(dsc->text[line_start] == '\0') return;
330 }
331
332 /*Align to middle*/
333 if(align == LV_TEXT_ALIGN_CENTER) {
334 line_width = lv_text_get_line_width(&dsc->text[line_start], line_end - line_start, font, &attributes);
335 pos.x += (lv_area_get_width(coords) - line_width) / 2;
336
337 }
338 /*Align to the right*/
339 else if(align == LV_TEXT_ALIGN_RIGHT) {
340 line_width = lv_text_get_line_width(&dsc->text[line_start], line_end - line_start, font, &attributes);
341 pos.x += lv_area_get_width(coords) - line_width;
342 }
343
344 uint32_t sel_start = dsc->sel_start;
345 uint32_t sel_end = dsc->sel_end;
346 if(sel_start > sel_end) {
347 uint32_t tmp = sel_start;
348 sel_start = sel_end;
349 sel_end = tmp;
350 }
351
352 lv_area_t bg_coords;
353 lv_draw_glyph_dsc_t draw_letter_dsc;
354 lv_font_glyph_dsc_t glyph_dsc;
355 lv_draw_glyph_dsc_init(&draw_letter_dsc);
356 draw_letter_dsc.opa = dsc->opa;
357 draw_letter_dsc.bg_coords = &bg_coords;
358 draw_letter_dsc.color = dsc->color;
359 draw_letter_dsc.rotation = dsc->rotation;
360 draw_letter_dsc.g = &glyph_dsc;
361
362 /* Set letter outline stroke attributes */
363 draw_letter_dsc.outline_stroke_width = dsc->outline_stroke_width;
364 draw_letter_dsc.outline_stroke_opa = dsc->outline_stroke_opa;
365 draw_letter_dsc.outline_stroke_color = dsc->outline_stroke_color;
366
367 lv_draw_fill_dsc_t fill_dsc;
368 lv_draw_fill_dsc_init(&fill_dsc);
369 fill_dsc.opa = dsc->opa;
370 int32_t underline_width = font->underline_thickness ? font->underline_thickness : 1;
371 int32_t line_start_x;
372 int32_t decor_end_x;
373 uint32_t next_char_offset;
374 uint32_t recolor_command_start_index = 0;
375 int32_t letter_w;
376
377 cmd_state_t recolor_cmd_state = RECOLOR_CMD_STATE_WAIT_FOR_PARAMETER;
378 lv_color_t recolor = lv_color_black(); /* Holds the selected color inside the recolor command */
379 uint8_t is_first_space_after_cmd = 0;
380
381 /*Write out all lines*/
382 while(remaining_len && dsc->text[line_start] != '\0') {
383 pos.x += x_ofs;
384 line_start_x = pos.x;
385 decor_end_x = pos.x - 1;
386
387 /*Write all letter of a line*/
388 next_char_offset = 0;
389#if LV_USE_BIDI
390 size_t bidi_size = line_end - line_start;
391 char * bidi_txt = lv_malloc(bidi_size + 1);
392 LV_ASSERT_MALLOC(bidi_txt);
393
397 if(dsc->has_bided) {
398 lv_memcpy(bidi_txt, &dsc->text[line_start], bidi_size);
399 }
400 else {
401 lv_bidi_process_paragraph(dsc->text + line_start, bidi_txt, bidi_size, base_dir, NULL, 0);
402 }
403#else
404 const char * bidi_txt = dsc->text + line_start;
405#endif
406
407 while(next_char_offset < remaining_len && next_char_offset < line_end - line_start) {
408 uint32_t logical_char_pos = 0;
409
410 /* Check if the text selection is enabled */
411 if(sel_start != LV_DRAW_LABEL_NO_TXT_SEL && sel_end != LV_DRAW_LABEL_NO_TXT_SEL) {
412#if LV_USE_BIDI
413 if(dsc->has_bided) {
414 logical_char_pos = lv_text_encoded_get_char_id(dsc->text, line_start + next_char_offset);
415 }
416 else {
417 logical_char_pos = lv_text_encoded_get_char_id(dsc->text, line_start);
418 uint32_t c_idx = lv_text_encoded_get_char_id(bidi_txt, next_char_offset);
419 logical_char_pos += lv_bidi_get_logical_pos(bidi_txt, NULL, line_end - line_start, base_dir, c_idx, NULL);
420 }
421#else
422 logical_char_pos = lv_text_encoded_get_char_id(dsc->text, line_start + next_char_offset);
423#endif
424 }
425
426 uint32_t letter;
427 uint32_t letter_next;
428 lv_text_encoded_letter_next_2(bidi_txt, &letter, &letter_next, &next_char_offset);
429
430 /* If recolor is enabled */
431 if((dsc->flag & LV_TEXT_FLAG_RECOLOR) != 0) {
432
433 if(letter == (uint32_t)LV_TXT_COLOR_CMD[0]) {
434 /* Handle the recolor command marker depending of the current recolor state */
435
436 if(recolor_cmd_state == RECOLOR_CMD_STATE_WAIT_FOR_PARAMETER) {
437 recolor_command_start_index = next_char_offset;
438 recolor_cmd_state = RECOLOR_CMD_STATE_PARAMETER;
439 continue;
440 }
441 /*Other start char in parameter escaped cmd. char*/
442 else if(recolor_cmd_state == RECOLOR_CMD_STATE_PARAMETER) {
443 recolor_cmd_state = RECOLOR_CMD_STATE_WAIT_FOR_PARAMETER;
444 }
445 /* If letter is LV_TXT_COLOR_CMD and we were in the CMD_STATE_IN then the recolor close marked has been found */
446 else if(recolor_cmd_state == RECOLOR_CMD_STATE_TEXT_INPUT) {
447 recolor_cmd_state = RECOLOR_CMD_STATE_WAIT_FOR_PARAMETER;
448 continue;
449 }
450 }
451
452 /* Find the first space (aka ' ') after the recolor command parameter, we need to skip rendering it */
453 if((recolor_cmd_state == RECOLOR_CMD_STATE_PARAMETER) && (letter == ' ') && (is_first_space_after_cmd == 0)) {
454 is_first_space_after_cmd = 1;
455 }
456 else {
457 is_first_space_after_cmd = 0;
458 }
459
460 /* Skip the color parameter and wait the space after it
461 * Once we have reach the space ' ', then we will extract the color information
462 * and store it into the recolor variable */
463 if(recolor_cmd_state == RECOLOR_CMD_STATE_PARAMETER) {
464 /* Not an space? Continue with the next character */
465 if(letter != ' ') {
466 continue;
467 }
468
469 /*Get the recolor parameter*/
470 if((next_char_offset - recolor_command_start_index) == LABEL_RECOLOR_PAR_LENGTH + 1) {
471 /* Temporary buffer to hold the recolor information */
472 char buf[LABEL_RECOLOR_PAR_LENGTH + 1];
473 lv_memcpy(buf, &bidi_txt[recolor_command_start_index], LABEL_RECOLOR_PAR_LENGTH);
474 buf[LABEL_RECOLOR_PAR_LENGTH] = '\0';
475
476 uint8_t r, g, b;
477 r = (hex_char_to_num(buf[0]) << 4) + hex_char_to_num(buf[1]);
478 g = (hex_char_to_num(buf[2]) << 4) + hex_char_to_num(buf[3]);
479 b = (hex_char_to_num(buf[4]) << 4) + hex_char_to_num(buf[5]);
480
481 recolor = lv_color_make(r, g, b);
482 }
483 else {
484 recolor.red = dsc->color.red;
485 recolor.blue = dsc->color.blue;
486 recolor.green = dsc->color.green;
487 }
488
489 /*After the parameter the text is in the command*/
490 recolor_cmd_state = RECOLOR_CMD_STATE_TEXT_INPUT;
491 }
492
493 /* Don't draw the first space after the recolor command */
494 if(is_first_space_after_cmd) {
495 continue;
496 }
497 }
498
499 /* If we're in the CMD_STATE_IN state then we need to subtract the recolor command length */
500 if(((dsc->flag & LV_TEXT_FLAG_RECOLOR) != 0) && (recolor_cmd_state == RECOLOR_CMD_STATE_TEXT_INPUT)) {
501 logical_char_pos -= (LABEL_RECOLOR_PAR_LENGTH + 1);
502 }
503
504 lv_font_get_glyph_dsc_internal(font, &glyph_dsc, letter, letter_next);
505 letter_w = lv_text_is_marker(letter) ? 0 : glyph_dsc.adv_w;
506
507 if(!lv_text_is_hanging_space(letter)) {
508 decor_end_x = pos.x + letter_w - 1;
509 }
510
511 /*Always set the bg_coordinates for placeholder drawing*/
512 bg_coords.x1 = pos.x - dsc->letter_space / 2;
513 bg_coords.y1 = pos.y;
514 bg_coords.x2 = pos.x + letter_w - 1 + (dsc->letter_space + 1) / 2;
515 bg_coords.y2 = pos.y + line_height - 1;
516
517 if(next_char_offset >= line_end - line_start && decor_end_x >= line_start_x) {
518 if(dsc->decor & LV_TEXT_DECOR_UNDERLINE) {
519 lv_area_t fill_area;
520 fill_area.x1 = line_start_x;
521 fill_area.x2 = decor_end_x;
522 fill_area.y1 = pos.y + font->line_height - font->base_line - font->underline_position;
523 fill_area.y2 = fill_area.y1 + underline_width - 1;
524
525 fill_dsc.color = dsc->color;
526 cb(t, NULL, &fill_dsc, &fill_area);
527 }
528 if(dsc->decor & LV_TEXT_DECOR_STRIKETHROUGH) {
529 lv_area_t fill_area;
530 fill_area.x1 = line_start_x;
531 fill_area.x2 = decor_end_x;
532 fill_area.y1 = pos.y + (font->line_height - font->base_line) * 2 / 3 + font->underline_thickness / 2;
533 fill_area.y2 = fill_area.y1 + underline_width - 1;
534
535 fill_dsc.color = dsc->color;
536 cb(t, NULL, &fill_dsc, &fill_area);
537 }
538 }
539
540 /* Handle text selection */
541 if(sel_start != LV_DRAW_LABEL_NO_TXT_SEL && sel_end != LV_DRAW_LABEL_NO_TXT_SEL
542 && logical_char_pos >= sel_start && logical_char_pos < sel_end) {
543 draw_letter_dsc.color = dsc->sel_color;
544 fill_dsc.color = dsc->sel_bg_color;
545 cb(t, NULL, &fill_dsc, &bg_coords);
546 }
547 else if(recolor_cmd_state == RECOLOR_CMD_STATE_TEXT_INPUT) {
548 draw_letter_dsc.color = recolor;
549 }
550 else {
551 draw_letter_dsc.color = dsc->color;
552 }
553
554 lv_draw_unit_draw_letter_internal(t, &draw_letter_dsc, &pos, font, letter, cb);
555
556 if(letter_w > 0) {
557 pos.x += letter_w + dsc->letter_space;
558 }
559 }
560
561#if LV_USE_BIDI
562 lv_free(bidi_txt);
563 bidi_txt = NULL;
564#endif
565
566 lv_text_attributes_t text_attributes = {0};
567 text_attributes.letter_space = dsc->letter_space;
568 text_attributes.text_flags = dsc->flag;
569 text_attributes.max_width = w;
570
571 /*Go to next line*/
572 remaining_len -= line_end - line_start;
573 line_start = line_end;
574 if(remaining_len) {
575 line_end += lv_text_get_next_line(&dsc->text[line_start], remaining_len, font, NULL, &text_attributes);
576 }
577
578 pos.x = coords->x1;
579 /*Align to middle*/
580 if(align == LV_TEXT_ALIGN_CENTER) {
581 line_width =
582 lv_text_get_line_width(&dsc->text[line_start], line_end - line_start, font, &text_attributes);
583

585 }

632void lv_draw_unit_draw_letter_internal(lv_draw_task_t * t, lv_draw_glyph_dsc_t * dsc, const lv_point_t * pos,
633 const lv_font_t * font, uint32_t letter, lv_draw_glyph_cb_t cb)
634{
635 LV_ASSERT(t != NULL);
636 LV_ASSERT(dsc != NULL);
637 LV_ASSERT(pos != NULL);
638 LV_ASSERT(font != NULL);
639 LV_ASSERT(cb != NULL);
640 lv_font_glyph_dsc_t g;
641
642 if(lv_text_is_marker(letter)) /*Markers are valid letters but should not be rendered.*/
643 return;
644
645 LV_PROFILER_DRAW_BEGIN;
646 if(dsc->g == NULL) {
647 /* cppcheck-suppress autoVariables
648 * here we store a stack variable on a pointer that outlives this function
649 * we correctly reset it to NULL at `exit:` but cpp-check chokes on it and warns us*/
650 dsc->g = &g;
651 /*If the glyph dsc is not set then get it from the font*/
652 bool g_ret = lv_font_get_glyph_dsc_internal(font, &g, letter, 0);
653 if(g_ret == false) {
654 /*Add warning if the dsc is not found*/
655 LV_LOG_WARN("lv_draw_letter: glyph dsc. not found for U+%" LV_PRIX32, letter);
656 }
657 }
658 else {
659 /*If the glyph dsc is set then use it*/
660 g = *dsc->g;
661 }
662
663 /*Don't draw anything if the character is empty. E.g. space*/
664 if((g.box_h == 0) || (g.box_w == 0)) {
665 goto exit;
666 }
667
668 lv_area_t letter_coords;
669 letter_coords.x1 = pos->x + g.ofs_x;
670 letter_coords.x2 = letter_coords.x1 + g.box_w - 1;
671 letter_coords.y1 = pos->y + (font->line_height - font->base_line) - g.box_h - g.ofs_y;
672 letter_coords.y2 = letter_coords.y1 + g.box_h - 1;
673 lv_area_move(&letter_coords, -dsc->pivot.x, -dsc->pivot.y);
674
675 /*If the letter is completely out of mask don't draw it*/
676 if(lv_area_is_out(&letter_coords, &t->clip_area, 0) &&
677 dsc->bg_coords &&
678 lv_area_is_out(dsc->bg_coords, &t->clip_area, 0)) {
679 goto exit;
680 }
681
682 if(g.resolved_font) {
683 lv_draw_buf_t * draw_buf = NULL;
684 if(LV_FONT_GLYPH_FORMAT_NONE < g.format && g.format < LV_FONT_GLYPH_FORMAT_IMAGE) {
685 /*Only check draw buf for bitmap glyph*/
686 draw_buf = lv_draw_buf_reshape(dsc->_draw_buf, 0, g.box_w, g.box_h, LV_STRIDE_AUTO);
687 if(draw_buf == NULL) {
688 if(dsc->_draw_buf) {
689 lv_draw_buf_destroy(dsc->_draw_buf);
690 dsc->_draw_buf = NULL;
691 }
692
693 uint32_t h = LV_ROUND_UP(g.box_h, 32); /*Assume a larger size to avoid many reallocations*/
694 draw_buf = lv_draw_buf_create_ex(font_draw_buf_handlers, g.box_w, h, LV_COLOR_FORMAT_A8, LV_STRIDE_AUTO);
695 LV_ASSERT_MALLOC(draw_buf);
696 if(!draw_buf) {
697 LV_LOG_WARN("Failed to allocate memory for glyph draw buffer");
698 goto exit;
699 }
700 draw_buf->header.h = g.box_h;
701 dsc->_draw_buf = draw_buf;
702 }
703 }
704
705 dsc->format = g.format;
706
707 if(g.format == LV_FONT_GLYPH_FORMAT_VECTOR) {
708
709 /*Load the outline of the glyph, even if the function says bitmap*/
710 dsc->glyph_data = (void *) lv_font_get_glyph_bitmap_internal(dsc->g, draw_buf);
711 dsc->format = dsc->glyph_data ? g.format : LV_FONT_GLYPH_FORMAT_NONE;
712 }
713 }
714 else {
715 dsc->format = LV_FONT_GLYPH_FORMAT_NONE;
716 }
717
718 dsc->letter_coords = &letter_coords;
719 cb(t, dsc, NULL, NULL);
720 dsc->letter_coords = NULL;
721
722 lv_font_glyph_release_draw_data_internal(dsc->g);
723
724exit:
725 if(dsc->g == &g) {
726 /* If the glyph was created locally, we don't need to keep it */
727 dsc->g = NULL;
728 }
729 LV_PROFILER_DRAW_END;
730}
