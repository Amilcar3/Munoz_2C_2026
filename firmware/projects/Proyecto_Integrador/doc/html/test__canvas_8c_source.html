449static lv_color_t canvas_convert_c16(lv_color_t color)
450{
451 /* Use the same color conversion algorithm as canvas to keep the results consistent */
452 uint16_t px = lv_color_to_u16(color);
453 lv_color16_t * c16 = (lv_color16_t *) &px;
454 lv_color_t ret;
455 ret.red = (c16->red * 2106) >> 8; /*To make it rounded*/
456 ret.green = (c16->green * 1037) >> 8;
457 ret.blue = (c16->blue * 2106) >> 8;
458 return ret;
459}
