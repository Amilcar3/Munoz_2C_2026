36lv_result_t lv_array_init(lv_array_t * array, uint32_t capacity, uint32_t element_size)
37{
38 LV_ASSERT(array != NULL);
39 lv_memset(array, 0, sizeof(*array));
40
41 void * data = lv_malloc(capacity * element_size);
42 LV_ASSERT_MALLOC(data);
43 if(!data) {
44 return LV_RESULT_INVALID;
45 }
46 lv_array_init_from_buf(array, data, capacity, element_size);
47 array->inner_alloc = true;
48 return LV_RESULT_OK;
49}

75void lv_array_copy(lv_array_t * target, const lv_array_t * source)
76{
77 LV_ASSERT_NULL(target);
78 LV_ASSERT_NULL(source);
79
80 if(target == source) return;
81
82 lv_array_deinit(target);
83
84 /* Guard against multiplication overflow during allocation */
85 if(source->capacity > 0 && source->element_size > 0) {
86 if(source->capacity > UINT32_MAX / source->element_size) {
87 lv_memzero(target, sizeof(lv_array_t));
88 return;
89 }
90
91 void * data = lv_malloc(source->capacity * source->element_size);
92 if(data == NULL) {
93 lv_memzero(target, sizeof(lv_array_t));
94 return;
95 }
96
97 target->data = data;
98 target->capacity = source->capacity;
99 target->element_size = source->element_size;
100 target->inner_alloc = true;
101 target->size = 0;
102 }
103 else {
104 void * data = lv_malloc(0);
105 if(data == NULL) {
106 lv_memzero(target, sizeof(lv_array_t));
107 return;
108 }
109 target->data = data;
110 target->capacity = source->capacity;
111 target->element_size = source->element_size;
112 target->inner_alloc = true;
113 target->size = 0;
114 }
115
116 if(lv_array_is_empty(source)) {
117 return;
118 }
119
120 lv_memcpy(target->data, source->data, source->size * source->element_size);
121 target->size = source->size;
122}

131lv_result_t lv_array_remove(lv_array_t * array, uint32_t index)
132{
133 if(!array) {
134 return LV_RESULT_INVALID;
135 }
136
137 if(index >= array->size) {
138 return LV_RESULT_INVALID;
139 }
140
141 /*Shortcut*/
142 if(index == array->size - 1) {
143 array->size--;
144 lv_array_shrink(array);
145 return LV_RESULT_OK;
146 }
147
148 uint8_t * start = lv_array_at(array, index);
149 uint8_t * remaining = start + array->element_size;
150 uint32_t remaining_size = (array->size - index - 1) * array->element_size;
151 lv_memmove(start, remaining, remaining_size);
152 array->size--;
153 lv_array_shrink(array);
154 return LV_RESULT_OK;
155}

184lv_result_t lv_array_erase(lv_array_t * array, uint32_t start, uint32_t end)
185{
186 if(end > array->size) {
187 end = array->size;
188 }
189
190 if(start >= end) {
191 return LV_RESULT_INVALID;
192 }
193
194 /*Shortcut*/
195 if(end == array->size) {
196 array->size = start;
197 lv_array_shrink(array);
198 return LV_RESULT_OK;
199 }
200
201 uint8_t * start_p = lv_array_at(array, start);
202 uint8_t * remaining = start_p + (end - start) * array->element_size;
203 uint32_t remaining_size = (array->size - end) * array->element_size;
204 lv_memmove(start_p, remaining, remaining_size);
205 array->size -= (end - start);
206 lv_array_shrink(array);
207 return LV_RESULT_OK;
208}

210bool lv_array_resize(lv_array_t * array, uint32_t new_capacity)
211{
212 LV_ASSERT(array);
213
214 if(array->inner_alloc == false) {
215 LV_LOG_WARN("Cannot resize array with external buffer");
216 return false;
217 }
218
219 uint8_t * data = lv_realloc(array->data, new_capacity * array->element_size);
220 if(data == NULL) return false;
221
222 array->data = data;
223 array->capacity = new_capacity;
224 if(array->size > new_capacity) {
225 array->size = new_capacity;
226 }
227 return true;
228}

230lv_result_t lv_array_concat(lv_array_t * array, const lv_array_t * other)
231{
232 LV_ASSERT(array);
233 LV_ASSERT(other);
234
235 if(array->element_size != other->element_size) {
236 LV_LOG_ERROR("Element size mismatch: %"LV_PRIu32" vs %"LV_PRIu32, array->element_size, other->element_size);
237 return LV_RESULT_INVALID;
238 }
239
240 uint32_t size = other->size;
241 if(size == 0) {
242 return LV_RESULT_OK;
243 }
244
245 if(UINT32_MAX - array->size < size) {
246 LV_LOG_ERROR("Array size overflow");
247 return LV_RESULT_INVALID;
248 }
249
250 if(array->size + size > array->capacity) {
251 /*array is full*/
252 if(lv_array_resize(array, array->size + size) == false) {
253 return LV_RESULT_INVALID;
254 }
255 }
256
257 uint8_t * data = array->data + array->size * array->element_size;
258 lv_memcpy(data, other->data, array->element_size * size);
259 array->size += size;
260 return LV_RESULT_OK;
261}

263lv_result_t lv_array_push_back(lv_array_t * array, const void * element)
264{
265 LV_ASSERT_NULL(array->data);
266
267 if(array->size == array->capacity) {
268 /*array is full*/
269 if(lv_array_resize(array, array->capacity + LV_ARRAY_DEFAULT_CAPACITY) == false) {
270 return LV_RESULT_INVALID;
271 }
272 }
273
277 uint8_t * data = array->data + array->size * array->element_size;
278 if(element) lv_memcpy(data, element, array->element_size);
279 else lv_memzero(data, array->element_size);
280
281 array->size++;
282 return LV_RESULT_OK;
283}
