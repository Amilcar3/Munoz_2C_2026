108def get_changed_lines(commit: str, root: str) -> Dict[str, Set[int]]:
109 """
110 Get changed lines in specified commit
111 Returns: {file_path: {set of changed line numbers}}
112 """
113 diff_output = run_git_command(
114 ["diff", f"{commit}^", commit, "--unified=0"], cwd=root
115 )
116
117 changed_lines: Dict[str, Set[int]] = {}
118 current_file = None
119
120 for line in diff_output.split("\n"):
121 if line.startswith("+++ b/"):
122 current_file = line[6:]
123 if current_file not in changed_lines:
124 changed_lines[current_file] = set()
125
126 elif line.startswith("@@"):
127 match = re.search(r"\+(\d+)(?:,(\d+))?", line)
128 if match and current_file:
129 start_line = int(match.group(1))
130 count = int(match.group(2)) if match.group(2) else 1
131 for lineno in range(start_line, start_line + count):
132 changed_lines[current_file].add(lineno)
133
134 return changed_lines

137def get_coverage_data(root: str) -> Dict[str, Dict[int, int]]:
138 """
139 Get coverage data using gcovr
140 Returns: {rel_file_path: {line_number: execution_count}}
141 Notes:
142 - Only lines explicitly present in gcovr JSON are considered "coverable".
143 Lines that are missing from the JSON (e.g. preprocessor directives like
144 #include, comments, whitespace, or excluded lines) are treated as
145 non-coverable and will be ignored by the uncovered check.
146 Raises: subprocess.CalledProcessError if gcovr fails
147 """
148 cmd = [
149 "gcovr",
150 "--gcov-ignore-parse-errors",
151 "--json",
152 "-",
153 "--root",
154 root,
155 ]
156
157 result = subprocess.run(
158 cmd,
159 capture_output=True,
160 text=True,
161 cwd=root,
162 check=False, # Don't raise exception automatically
163 )
164
165 if result.returncode != 0:
166 error_msg = f"gcovr command failed (exit code {result.returncode}):\n"
167 if result.stderr:
168 error_msg += f"Error output:\n{result.stderr}\n"
169 if result.stdout:
170 error_msg += f"Standard output:\n{result.stdout}\n"
171 raise subprocess.CalledProcessError(
172 result.returncode, cmd, result.stdout, result.stderr
173 )
174
175 coverage_json = json.loads(result.stdout)
176 coverage_data: Dict[str, Dict[int, int]] = {}
177
178 for file_info in coverage_json.get("files", []):
179 # Normalize path to be relative to repo root with POSIX separators.
180 filename = file_info["file"]
181 rel = os.path.relpath(filename, root)
182 rel = rel.replace(os.path.sep, "/")
183 coverage_data[rel] = {}
184
185 for line_info in file_info.get("lines", []):
186 line_number = line_info.get("line_number")
187 # Only consider lines explicitly listed by gcovr as coverable.
188 # Some gcovr versions may include additional flags like
189 # "gcovr/noncode" or "excluded"; we simply ignore such lines
190 # by relying on their absence from the JSON or by requiring
191 # a numeric execution count.
192 if line_number is None:
193 continue
194 count = line_info.get("count")
195 if isinstance(count, int):
196 coverage_data[rel][line_number] = count
197
198 return coverage_data

201def check_commit_coverage(
202 commit: str, root: str, coverage_data: Dict[str, Dict[int, int]]
203) -> Tuple[int, int, List[Tuple[str, int]], int]:
204 """
205 Check coverage for a commit or range
206 Returns: (covered_lines, total_new_lines, uncovered_lines, skipped_noncoverable)
207 """
208
209 if "..." in commit:
210 # Handle range format (base...head)
211 base, head = commit.split("...")
212 # Get changes for each commit in the range
213 commits = run_git_command(["rev-list", f"{base}...{head}"], cwd=root).split()
214 else:
215 # Handle single commit
216 commits = [commit]
217
218 print(f"Found {len(commits)} commits in range:")
219 changed_lines = {}
220 for c in commits:
221 print(f" {c}")
222 cl = get_changed_lines(c, root)
223 for f, lines in cl.items():
224 if f not in changed_lines:
225 changed_lines[f] = set()
226 changed_lines[f].update(lines)
227
228 print(f"Found changes in {len(changed_lines)} files (before filtering):")
229 for filename, lines in sorted(changed_lines.items()):
230 print(f" {filename}: {len(lines)} lines changed")
231
232 # Normalize changed file paths to POSIX separators for matching.
233 normalized_changed: Dict[str, Set[int]] = {
234 f.replace(os.path.sep, "/"): lines for f, lines in changed_lines.items()
235 }
236
237 # If desired, we could additionally filter by the gcovr filter pattern.
238 # However, by intersecting with coverage_data keys, we inherently ignore
239 # files that are not part of coverage anyway.
240
241 total_new_lines = 0 # total coverable new lines (per gcovr)
242 covered_lines = 0
243 uncovered_lines: List[Tuple[str, int]] = []
244 skipped_noncoverable = 0 # changed lines that gcovr doesn't consider coverable
245
246 # Build quick lookup for filenames present in coverage.
247 coverage_files: Set[str] = set(coverage_data.keys())
248
249 # Iterate changed files and intersect with gcovr-provided coverable lines.
250 for filename, line_numbers in normalized_changed.items():
251 if filename not in coverage_files:
252 # Entire file has no coverable lines in gcovr output; skip all.
253 skipped_noncoverable += len(line_numbers)
254 continue
255
256 file_coverage = coverage_data[filename]
257 for lineno in line_numbers:
258 if lineno not in file_coverage:
259 skipped_noncoverable += 1
260 continue
261
262 total_new_lines += 1
263 if file_coverage[lineno] > 0:
264 covered_lines += 1
265 else:
266 uncovered_lines.append((filename, lineno))
267
268 return covered_lines, total_new_lines, uncovered_lines, skipped_noncoverable

271def check_path_coverage(
272 path: str, root: str, coverage_data: Dict[str, Dict[int, int]]
273) -> Tuple[int, int, List[Tuple[str, int]]]:
274 """
275 Compute coverage for a specific file or directory.
276 Returns: (covered_lines, total_coverable_lines, uncovered_lines)
277 Notes:
278 - Only lines reported by gcovr as coverable are counted toward totals.
279 - If PATH is a directory, coverage is aggregated over all coverable files within it.
280 - If PATH is a file, coverage is computed only for that file.
281 """
282 # Normalize input path
283 abs_path = os.path.abspath(path)
284
285 # Ensure we operate from repo root to construct relative POSIX paths
286 root = os.path.abspath(root)
287
288 # Build relative POSIX target(s)
289 rel = os.path.relpath(abs_path, root)
290 rel_posix = rel.replace(os.path.sep, "/")
291
292 covered = 0
293 total = 0
294 uncovered: List[Tuple[str, int]] = []
295
296 if os.path.isdir(abs_path):
297 # Directory scope: include all files under this directory
298 scoped_files = {}
299 for f, lines in coverage_data.items():
300 # Convert both to absolute paths for comparison
301 f_abs = os.path.abspath(os.path.join(root, f.replace("/", os.path.sep)))
302 # If the common path of abs_path and f_abs is abs_path, f is under abs_path
303 if os.path.commonpath([abs_path, f_abs]) == abs_path:
304 scoped_files[f] = lines
305
306 print(f"Found {len(scoped_files)} coverable file(s) under: {rel_posix}")
307 for filename, line_map in sorted(scoped_files.items()):
308 for lineno, count in line_map.items():
309 total += 1
310 if count > 0:
311 covered += 1
312 else:
313 uncovered.append((filename, lineno))
314 else:
315 # File scope
316 if rel_posix in coverage_data:
317 line_map = coverage_data[rel_posix]
318 print(f"Found coverable file: {rel_posix}")
319 for lineno, count in line_map.items():
320 total += 1
321 if count > 0:
322 covered += 1
323 else:
324 uncovered.append((rel_posix, lineno))
325 else:
326 print(
327 f"Warning: No coverable lines found for '{rel_posix}' in gcovr output."
328 )
329
330 return covered, total, uncovered

333def overall_coverage(coverage_data: Dict[str, Dict[int, int]]) -> Tuple[int, int]:
334 """
335 Aggregate every coverable line gcovr reported.
336 Returns: (covered_lines, total_coverable_lines)
337 The scope is whatever gcovr.cfg selects, so it matches the patch numbers.
338 """
339 covered = 0
340 total = 0
341 for line_map in coverage_data.values():
342 for count in line_map.values():
343 total += 1
344 if count > 0:
345 covered += 1
346 return covered, total

354def export_overall(covered: int, total: int, root: str, output_path: str) -> None:
355 """Export the overall totals so a later run can use them as a baseline"""
356 data = {
357 "commit": run_git_command(["rev-parse", "HEAD"], cwd=root).strip(),
358 "covered": covered,
359 "total": total,
360 "percent": percent_of(covered, total),
361 }
362 directory = os.path.dirname(os.path.abspath(output_path))
363 os.makedirs(directory, exist_ok=True)
364 with open(output_path, "w", encoding="utf-8") as f:
365 json.dump(data, f, indent=1)
366 f.write("\n")
367 print(f"Exported the overall coverage totals to {output_path}")

370def read_baseline(path: str) -> Optional[Dict]:
371 """
372 Load the totals exported by an earlier --overall-out run.
373 A missing or unusable baseline only costs us the trend, so warn and go on.
374 """
375 try:
376 with open(path, encoding="utf-8") as f:
377 data = json.load(f)
378 except (OSError, json.JSONDecodeError) as e:
379 print(f"Warning: cannot read the coverage baseline '{path}': {e}")
380 return None
381
382 usable = (
383 isinstance(data, dict)
384 and isinstance(data.get("covered"), int)
385 and isinstance(data.get("total"), int)
386 and data["total"] > 0
387 )
388 if not usable:
389 print(f"Warning: the coverage baseline '{path}' has no usable totals")
390 return None
391
392 commit = data.get("commit")
393 data["commit"] = commit if isinstance(commit, str) and commit else "unknown"
394
395 print(
396 f"Baseline: {data['covered']}/{data['total']} = "
397 f"{percent_of(data['covered'], data['total']):.2f}% ({data['commit']})"
398 )
399 return data

402def overall_summary(covered: int, total: int, baseline: Optional[Dict]) -> str:
403 """Render the `overall ...` part of the PR report coverage line"""
404 now = percent_of(covered, total)
405 if now is None:
406 return ""
407
408 before = percent_of(baseline["covered"], baseline["total"]) if baseline else None
409 if before is None:
410 return f"overall {now:.2f}%"
411
412 return f"overall {now:.2f}% (vs master {before:.2f}%, {now - before:+.2f}%)"

415def report_coverage(
416 header: str,
417 total: int,
418 covered: int,
419 uncovered: List[Tuple[str, int]],
420 fail_under: float,
421 *,
422 total_label: str,
423 skipped_noncoverable: Optional[int] = None,
424 min_lines: int = 0,
425) -> int:
426 """
427 Print a standardized coverage report and return exit code (0/1).
428
429 - header: text to show in the report title (e.g., "commit <hash>", "'<path>'")
430 - total_label: label to use for the total line count (e.g.,
431 "New coverable lines (per gcovr)" for commit mode, or
432 "Coverable lines (per gcovr)" for path mode)
433 - skipped_noncoverable: when provided, prints the skipped non-coverable count
434 - min_lines: when > 0, skip --fail-under enforcement if total < min_lines
435 """
436
437 title = f" Coverage analysis results for {header} "
438 separator = "=" * len(title)
439 print(f"\n{separator}\n{title}\n{separator}")
440 print(f"{total_label}: {total}")
441 print(f"Covered lines: {covered}")
442 print(f"Uncovered lines: {len(uncovered)}")
443 if skipped_noncoverable is not None:
444 print(f"Skipped non-coverable changed lines: {skipped_noncoverable}")
445
446 retval = 0
447 if total > 0:
448 coverage_percent = (covered / total) * 100
449 print(f"Coverage: {coverage_percent:.2f}%")
450
451 # Check if this patch is too small to enforce the threshold
452 if min_lines > 0 and fail_under > 0 and total < min_lines:
453 print(
454 f"\nℹ Only {total} new coverable line(s) (< {min_lines}), "
455 f"skipping --fail-under enforcement."
456 )
457 elif coverage_percent < fail_under:
458 print(
459 f"\n✗ Coverage {coverage_percent:.2f}% is below required {fail_under}%"
460 )
461 retval = 1
462
463 if uncovered:
464 print(
465 "\nUncovered lines (explicitly reported by gcovr as coverable with 0 hits):"
466 )
467 for filename, lineno in sorted(uncovered):
468 print(f" {filename}:{lineno}")
469 else:
470 if total == 0:
471 print("\nNo coverable lines found.")
472 else:
473 print(f"\n✓ Code coverage check passed!")
474
475 return retval

478def write_pr_report(covered: int, total: int, fail_under: float,
479 min_lines: int, output_path: str,
480 overall_covered: int = 0, overall_total: int = 0,
481 baseline: Optional[Dict] = None) -> None:
482 """Emit the `Coverage` section of the PR report comment.
483 Reports coverage of the lines the pull request changed, followed by the
484 overall project coverage and, with a baseline, how it moved.
485 The icon still reflects the changed lines only, since that is what gates.
486 """
487
488 if total == 0:
489 icon, summary = "skip", "no coverable lines changed"
490 details = ""
491 else:
492 percent = covered / total * 100.0
493 too_small = 0 < total < min_lines
494 if too_small:
495 icon, summary = "info", (
496 f"{percent:.1f}% of {total} changed line(s) "
497 f"(too few to enforce {fail_under:g}%)"
498 )
499 elif percent < fail_under:
500 icon, summary = "fail", (
501 f"{percent:.1f}% of {total} changed line(s), below {fail_under:g}%"
502 )
503 else:
504 icon, summary = "ok", f"{percent:.1f}% of {total} changed line(s)"
505 details = (f"{covered} of {total} newly coverable line(s) are "
506 f"covered by the test suite.")
507
508 overall = overall_summary(overall_covered, overall_total, baseline)
509 if overall:
510 summary = f"{summary} · {overall}"
511 overall_details = (
512 f"Overall: {overall_covered} of {overall_total} coverable line(s) "
513 f"are covered by the test suite."
514 )
515 if baseline:
516 overall_details += (
517 f" The latest master build, `{baseline['commit'][:9]}`, covered "
518 f"{baseline['covered']} of {baseline['total']}. Commits merged to "
519 f"master since this branch was cut also move that comparison."
520 )
521 details = f"{details}\n\n{overall_details}" if details else overall_details
522
523 write_report(output_path, section="Coverage", icon=icon, summary=summary,
524 details=details)

527def main() -> int:
528 """Main entry point"""
529 parser = create_argument_parser()
530 args = parser.parse_args()
531
532 try:
533 script_dir = os.path.dirname(os.path.abspath(__file__))
534 root = os.path.dirname(script_dir)
535 os.chdir(root)
536 print(f"Current working directory: {root}")
537
538 if args.path and not os.path.exists(os.path.abspath(args.path)):
539 print(
540 "Error: The specified path does not exist: "
541 f"{os.path.abspath(args.path)}",
542 file=sys.stderr,
543 )
544 return 1
545
546 # A single gcovr pass feeds both the overall and the patch numbers.
547 print("Getting coverage data...")
548 coverage_data = get_coverage_data(root)
549
550 overall_covered, overall_total = overall_coverage(coverage_data)
551 overall_percent = percent_of(overall_covered, overall_total)
552 print(
553 f"Overall coverage: {overall_covered}/{overall_total}"
554 + (f" = {overall_percent:.2f}%" if overall_percent is not None else "")
555 )
556
557 if args.overall_out:
558 export_overall(overall_covered, overall_total, root, args.overall_out)
559
560 baseline = read_baseline(args.baseline) if args.baseline else None
561
562 if args.overall_only:
563 return 0
564
565 if args.path:
566 # Path mode: ignore commit, compute coverage for file/dir
567 # --min-lines is not applied here since it targets patch (commit) mode.
568 covered, total, uncovered = check_path_coverage(
569 args.path, root, coverage_data
570 )
571
572 return report_coverage(
573 header=f"'{args.path}'",
574 total=total,
575 covered=covered,
576 uncovered=uncovered,
577 fail_under=args.fail_under,
578 total_label="Coverable lines (per gcovr)",
579 )
580 else:
581 # Commit mode: default behavior
582 covered, total, uncovered, skipped_noncoverable = check_commit_coverage(
583 args.commit, root, coverage_data
584 )
585
586 if args.report:
587 write_pr_report(covered, total, args.fail_under,
588 args.min_lines, args.report,
589 overall_covered, overall_total, baseline)
590
591 return report_coverage(
592 header=f"commit {args.commit}",
593 total=total,
594 covered=covered,
595 uncovered=uncovered,
596 fail_under=args.fail_under,
597 total_label="New coverable lines (per gcovr)",
598 skipped_noncoverable=skipped_noncoverable,
599 min_lines=args.min_lines,
600 )
601
602 except subprocess.CalledProcessError as e:
603 print(f"Error: Command failed - {e}", file=sys.stderr)
604 if e.stderr:
605 print(f"Error details:\n{e.stderr}", file=sys.stderr)
606 if e.stdout:
607 print(f"Command output:\n{e.stdout}", file=sys.stderr)
608 return e.returncode
609 except Exception as e:
610 print(f"Error: {e}", file=sys.stderr)
611 return 2
