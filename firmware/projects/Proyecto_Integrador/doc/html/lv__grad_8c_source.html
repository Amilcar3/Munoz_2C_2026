34void lv_grad_init_stops(lv_grad_dsc_t * grad, const lv_color_t colors[], const lv_opa_t opa[],
35 const uint8_t fracs[], int num_stops)
36{
37 LV_CHECK_ARG(num_stops <= LV_GRADIENT_MAX_STOPS, return);
38 LV_CHECK_ARG(num_stops > 1, return);
39 LV_CHECK_ARG(grad != NULL, return);
40 LV_CHECK_ARG(colors != NULL, return);
41
42 grad->stops_count = num_stops;
43 for(int i = 0; i < num_stops; i++) {
44 grad->stops[i].color = colors[i];
45 grad->stops[i].opa = opa != NULL ? opa[i] : LV_OPA_COVER;
46 grad->stops[i].frac = fracs != NULL ? fracs[i] : 255 * i / (num_stops - 1);
47 }
48}
