43void lv_draw_nema_gfx_border(lv_draw_task_t * t, const lv_draw_border_dsc_t * dsc, const lv_area_t * coords)
44{
45 if(dsc->opa <= LV_OPA_MIN)
46 return;
47 if(dsc->width == 0)
48 return;
49 if(dsc->side == LV_BORDER_SIDE_NONE)
50 return;
51
52 lv_draw_nema_gfx_unit_t * draw_nema_gfx_unit = (lv_draw_nema_gfx_unit_t *)t->draw_unit;
53
54 lv_layer_t * layer = t->target_layer;
55 lv_area_t inward_coords;
56 int32_t width = dsc->width;
57
58 /* Move border inwards to align with software rendered border */
59 inward_coords.x1 = coords->x1 + ceil(width / 2.0f);
60 inward_coords.x2 = coords->x2 - floor(width / 2.0f);
61 inward_coords.y1 = coords->y1 + ceil(width / 2.0f);
62 inward_coords.y2 = coords->y2 - floor(width / 2.0f);
63
64 lv_area_move(&inward_coords, -layer->buf_area.x1, -layer->buf_area.y1);
65
66 lv_area_t clip_area;
67 clip_area = t->clip_area;
68 lv_area_move(&clip_area, -layer->buf_area.x1, -layer->buf_area.y1);
69
70 nema_set_clip(clip_area.x1, clip_area.y1, lv_area_get_width(&clip_area), lv_area_get_height(&clip_area));
71
72 lv_area_t clipped_coords;
73 if(!lv_area_intersect(&clipped_coords, &inward_coords, &clip_area))
74 return; /*Fully clipped, nothing to do*/
75
76 lv_color_format_t dst_cf = layer->draw_buf->header.cf;
77 uint32_t dst_nema_cf = lv_nemagfx_cf_to_nema(dst_cf);
78
79 /* the stride should be computed internally for NEMA_TSC images and images missing a stride value */
80 int32_t stride = (dst_cf >= LV_COLOR_FORMAT_NEMA_TSC_START && dst_cf <= LV_COLOR_FORMAT_NEMA_TSC_END) ?
81 -1 : lv_area_get_width(&(layer->buf_area)) * lv_color_format_get_size(dst_cf);
82
83 nema_bind_dst_tex((uintptr_t)NEMA_VIRT2PHYS(layer->draw_buf->data), lv_area_get_width(&(layer->buf_area)),
84 lv_area_get_height(&(layer->buf_area)), dst_nema_cf, stride);
85
86 /* Recalculate float Dimensions */
87 float x1 = (float)coords->x1 + ((float)width / 2.0f) - (float)layer->buf_area.x1;
88 float x2 = (float)coords->x2 - ((float)width / 2.0f) - (float)layer->buf_area.x1;
89 float y1 = (float)coords->y1 + ((float)width / 2.0f) - (float)layer->buf_area.y1;
90 float y2 = (float)coords->y2 - ((float)width / 2.0f) - (float)layer->buf_area.y1;
91 float coords_bg_w = x2 - x1 + 1;
92 float coords_bg_h = y2 - y1 + 1;
93 int32_t short_side = LV_MIN(coords_bg_w, coords_bg_h);
94 float radius = (float)LV_MIN(dsc->radius, short_side >> 1);
95
96 lv_color32_t col32 = lv_color_to_32(dsc->color, dsc->opa);
97 uint32_t bg_color = nema_rgba(col32.red, col32.green, col32.blue, col32.alpha);
98
99 if(col32.alpha < 255U) {
100 nema_set_blend_fill(NEMA_BL_SRC_OVER);
101 bg_color = nema_premultiply_rgba(bg_color);
102 }
103 else {
104 nema_set_blend_fill(NEMA_BL_SRC);
105 }
106
107 if(radius > 0.0f) {
108 nema_draw_rounded_rect_aa(x1, y1, coords_bg_w, coords_bg_h, radius, width, bg_color);
109 }
110 else {
111 lv_area_t rect_coords = *coords;
112 lv_area_move(&rect_coords, -layer->buf_area.x1, -layer->buf_area.y1);
113 int32_t border_width = lv_area_get_width(&rect_coords);
114 int32_t border_height = lv_area_get_height(&rect_coords);
115
116 if(dsc->side & LV_BORDER_SIDE_TOP) {
117 float x = rect_coords.x1 + width;
118 float y = rect_coords.y1;
119 float w = border_width - 2 * width;
120 float h = width;
121 nema_enable_aa(1, 0, 1, 0);
122 nema_fill_rect_f(x, y, w, h, bg_color);
123 }
124
125 if(dsc->side & LV_BORDER_SIDE_BOTTOM) {
126 float x = rect_coords.x1 + width;
127 float y = rect_coords.y1 + border_height - width;
128 float w = border_width - 2 * width;
129 float h = width;
130 nema_enable_aa(1, 0, 1, 0);
131 nema_fill_rect_f(x, y, w, h, bg_color);
132 }
133
134 if(dsc->side & LV_BORDER_SIDE_LEFT) {
135 float x = rect_coords.x1;
136 float y = rect_coords.y1 + width;
137 float w = width;
138 float h = border_height - 2 * width;
139 nema_enable_aa(0, 1, 0, 1);
140 nema_fill_rect_f(x, y, w, h, bg_color);
141 }
142
143 if(dsc->side & LV_BORDER_SIDE_RIGHT) {
144 float x = rect_coords.x1 + border_width - width;
145 float y = rect_coords.y1 + width;
146 float w = width;
147 float h = border_height - 2 * width;
148 nema_enable_aa(0, 1, 0, 1);
149 nema_fill_rect_f(x, y, w, h, bg_color);
150 }
151
152 /*Draw small corner rectangles
153 Top Left*/
154 if(dsc->side & LV_BORDER_SIDE_TOP || dsc->side & LV_BORDER_SIDE_LEFT) {
155 float x = rect_coords.x1;
156 float y = rect_coords.y1;
157 float w = width;
158 float h = width;
159
160 if(!(dsc->side & LV_BORDER_SIDE_TOP))
161 nema_enable_aa(1, 1, 0, 1);
162 else if(!(dsc->side & LV_BORDER_SIDE_LEFT))
163 nema_enable_aa(1, 0, 1, 1);
164 else
165 nema_enable_aa(1, 0, 0, 1);
166
167 nema_fill_rect_f(x, y, w, h, bg_color);
168 }
169
170 /*Top Right*/
171 if(dsc->side & LV_BORDER_SIDE_TOP || dsc->side & LV_BORDER_SIDE_RIGHT) {
172 float x = rect_coords.x1 + border_width - width;
173 float y = rect_coords.y1;
174 float w = width;
175 float h = width;
176
177 if(!(dsc->side & LV_BORDER_SIDE_TOP))
178 nema_enable_aa(1, 1, 0, 1);
179 else if(!(dsc->side & LV_BORDER_SIDE_RIGHT))
180 nema_enable_aa(1, 1, 1, 0);
181 else
182 nema_enable_aa(1, 1, 0, 0);
183
184 nema_fill_rect_f(x, y, w, h, bg_color);
185 }
186
187 /*Bottom Right*/
188 if(dsc->side & LV_BORDER_SIDE_BOTTOM || dsc->side & LV_BORDER_SIDE_RIGHT) {
189 float x = rect_coords.x1 + border_width - width;
190 float y = rect_coords.y1 + border_height - width;
191 float w = width;
192 float h = width;
193
194 if(!(dsc->side & LV_BORDER_SIDE_BOTTOM))
195 nema_enable_aa(0, 1, 1, 1);
196 else if(!(dsc->side & LV_BORDER_SIDE_RIGHT))
197 nema_enable_aa(1, 1, 1, 0);
198 else
199 nema_enable_aa(0, 1, 1, 0);
200
201 nema_fill_rect_f(x, y, w, h, bg_color);
202 }
203
204 /*Bottom Left*/
205 if(dsc->side & LV_BORDER_SIDE_BOTTOM || dsc->side & LV_BORDER_SIDE_LEFT) {
206 float x = rect_coords.x1;
207 float y = rect_coords.y1 + border_height - width;
208 float w = width;
209 float h = width;
210
211 if(!(dsc->side & LV_BORDER_SIDE_BOTTOM))
212 nema_enable_aa(0, 1, 1, 1);
213 else if(!(dsc->side & LV_BORDER_SIDE_LEFT))
214 nema_enable_aa(1, 0, 1, 1);
215 else
216 nema_enable_aa(0, 0, 1, 1);
217
218 nema_fill_rect_f(x, y, w, h, bg_color);
219 }
220
221 }
222
223 nema_cl_submit(&(draw_nema_gfx_unit->cl));
224
225}
