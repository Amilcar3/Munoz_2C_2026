28def binutil(build_dir, name):
29 """Derive a binutils path from the compiler CMake actually used for this build."""
30 cache = os.path.join(build_dir, "CMakeCache.txt")
31 with open(cache) as f:
32 for line in f:
33 if line.startswith("CMAKE_C_COMPILER:"):
34 compiler = line.split("=", 1)[1].strip()
35 break
36 else:
37 raise RuntimeError(f"no CMAKE_C_COMPILER in {cache}")
38
39 if not compiler.endswith("gcc"):
40 raise RuntimeError(f"unexpected compiler name: {compiler}")
41 return compiler[: -len("gcc")] + name

44def symbol_sizes(build_dir, elf):
45 """Per-symbol sizes from the linked image, so a regression can be localised."""
46 out = run(
47 [binutil(build_dir, "nm"), "--print-size", "--size-sort", "--radix=d", elf]
48 ).stdout
49 sizes = {}
50 for line in out.splitlines():
51 fields = line.split()
52 if len(fields) != 4:
53 continue
54 _, size, _, name = fields
55 sizes[name] = sizes.get(name, 0) + int(size)
56 return sizes

59def library_size(build_dir):
60 """Size of liblvgl.a with nothing dead-stripped."""
61
62 matches = glob.glob(os.path.join(build_dir, "**", "liblvgl.a"), recursive=True)
63 if not matches:
64 raise SystemExit(f"no liblvgl.a under {build_dir}")
65 out = run([binutil(build_dir, "size"), "-t", matches[0]]).stdout
66 text, data, bss = (int(v) for v in out.splitlines()[-1].split()[:3])
67 return {"text": text, "data": data, "bss": bss, "size": text + data}

70def measure(target, config, build_root):
71 build_dir = os.path.join(build_root, f"{target}-{config}")
72 shutil.rmtree(build_dir, ignore_errors=True)
73
74 toolchain = os.path.join(SCRIPT_DIR, "toolchain", f"{target}.cmake")
75 run(
76 [
77 "cmake",
78 "-S",
79 os.path.join(SCRIPT_DIR, config),
80 "-B",
81 build_dir,
82 "-GNinja",
83 f"-DCMAKE_TOOLCHAIN_FILE={toolchain}",
84 ]
85 )
86 run(["cmake", "--build", build_dir])
87
88 elf = os.path.join(build_dir, "main")
89
90 out = run([binutil(build_dir, "size"), elf]).stdout
91 text, data, bss = (int(v) for v in out.splitlines()[-1].split()[:3])
92 sizes = {
93 "text": text,
94 "data": data,
95 "bss": bss,
96 "flash": text + data,
97 "ram": data + bss,
98 "library": library_size(build_dir),
99 }
100 return sizes, symbol_sizes(build_dir, elf)

111def write_results(out, results, all_symbols):
112 out_dir = os.path.dirname(os.path.abspath(out))
113 os.makedirs(out_dir, exist_ok=True)
114 with open(out, "w") as f:
115 json.dump(results, f, indent=2)
116
117 commit = measured_commit()
118 if commit:
119 with open(os.path.join(out_dir, "commit.txt"), "w") as f:
120 f.write(commit + "\n")
121
122 # Kept beside the totals rather than in them: the totals stay readable, and report.py
123 # only needs these for the rows that actually moved.
124 symbols_dir = os.path.join(out_dir, "symbols")
125 os.makedirs(symbols_dir, exist_ok=True)
126 for (target, config), symbols in all_symbols.items():
127 with open(os.path.join(symbols_dir, f"{target}-{config}.json"), "w") as f:
128 json.dump(symbols, f, indent=2, sort_keys=True)
129 return symbols_dir
