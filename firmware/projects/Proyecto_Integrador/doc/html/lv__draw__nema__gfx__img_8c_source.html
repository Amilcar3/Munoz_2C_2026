54static void _draw_nema_gfx_img(lv_draw_task_t * t, const lv_draw_image_dsc_t * dsc,
55 const lv_image_decoder_dsc_t * decoder_dsc, lv_draw_image_sup_t * sup,
56 const lv_area_t * img_coords, const lv_area_t * clipped_img_area)
57{
58 if(dsc->opa <= LV_OPA_MIN) return;
59
60 lv_draw_nema_gfx_unit_t * draw_nema_gfx_unit = (lv_draw_nema_gfx_unit_t *)t->draw_unit;
61
62 lv_layer_t * layer = t->target_layer;
63 const lv_draw_buf_t * decoded = decoder_dsc->decoded;
64 const lv_image_header_t * header = &decoded->header;
65
66 bool masked = dsc->bitmap_mask_src != NULL;
67
68 lv_area_t rel_clip_area;
69 rel_clip_area = *clipped_img_area;
70 lv_area_move(&rel_clip_area, -layer->buf_area.x1, -layer->buf_area.y1);
71
72 bool has_transform = (dsc->rotation != 0 || dsc->scale_x != LV_SCALE_NONE || dsc->scale_y != LV_SCALE_NONE);
73 bool recolor = (dsc->recolor_opa > LV_OPA_MIN);
74
75 uint32_t tex_w = header->w;
76 uint32_t tex_h = header->h;
77
78 nema_set_clip(rel_clip_area.x1, rel_clip_area.y1, lv_area_get_width(&rel_clip_area),
79 lv_area_get_height(&rel_clip_area));
80
81 lv_color_format_t dst_cf = layer->draw_buf->header.cf;
82 uint32_t dst_nema_cf = lv_nemagfx_cf_to_nema(dst_cf);
83
84 const void * src_buf = decoded->data;
85
86 uint32_t blending_mode = lv_nemagfx_blending_mode(dsc->blend_mode);
87
88 lv_color_format_t src_cf = header->cf;
89
90 /*Image contains Alpha*/
91 if(src_cf == LV_COLOR_FORMAT_ARGB8888 || src_cf == LV_COLOR_FORMAT_XRGB8888) {
92 blending_mode |= NEMA_BLOP_SRC_PREMULT;
93 }
94
95 uint32_t src_nema_cf = lv_nemagfx_cf_to_nema(src_cf);
96 /* the stride should be computed internally for NEMA_TSC images and images missing a stride value */
97 int32_t src_stride = (src_cf >= LV_COLOR_FORMAT_NEMA_TSC_START && src_cf <= LV_COLOR_FORMAT_NEMA_TSC_END)
98 || header->stride == 0 ? -1 : (int32_t)header->stride;
99
100 int32_t stride = (dst_cf >= LV_COLOR_FORMAT_NEMA_TSC_START && dst_cf <= LV_COLOR_FORMAT_NEMA_TSC_END) ?
101 -1 : lv_area_get_width(&(layer->buf_area)) * lv_color_format_get_size(dst_cf);
102
103 nema_bind_dst_tex((uintptr_t)NEMA_VIRT2PHYS(layer->draw_buf->data), lv_area_get_width(&(layer->buf_area)),
104 lv_area_get_height(&(layer->buf_area)), dst_nema_cf, stride);
105
106 if(!LV_COLOR_FORMAT_IS_INDEXED(src_cf)) {
107 nema_bind_src_tex((uintptr_t)(src_buf), tex_w, tex_h, src_nema_cf, src_stride,
108 dsc->antialias ? NEMA_FILTER_BL : NEMA_FILTER_PS);
109 }
110 else {
111 nema_bind_lut_tex((uintptr_t)((uint8_t *)src_buf + LV_COLOR_INDEXED_PALETTE_SIZE(src_cf) * 4), tex_w, tex_h,
112 src_nema_cf, src_stride, NEMA_FILTER_PS, (uintptr_t)(src_buf), NEMA_BGRA8888);
113 blending_mode |= NEMA_BLOP_LUT;
114 }
115
116 /*Guard for previous NemaGFX Version*/
117#ifdef NEMA_BLOP_RECOLOR
118 if(recolor) {
119 lv_color32_t col32 = lv_color_to_32(dsc->recolor, LV_OPA_MIX2(dsc->recolor_opa, dsc->opa));
120 uint32_t color = nema_rgba(col32.red, col32.green, col32.blue, col32.alpha);
121 nema_set_recolor_color(color);
122 blending_mode |= NEMA_BLOP_RECOLOR;
123 }
124#endif
125
126 if(dsc->opa < 255) {
127 uint32_t rgba = ((uint32_t)dsc->opa << 24U) | ((uint32_t)dsc->opa << 16U) | ((uint32_t)dsc->opa << 8U) | ((
128 uint32_t)dsc->opa);
129 nema_set_const_color(rgba);
130 blending_mode |= NEMA_BLOP_MODULATE_A;
131 }
132
133 if(!has_transform && masked && !recolor) {
134 if(dsc->bitmap_mask_src->header.cf == LV_COLOR_FORMAT_A8 || dsc->bitmap_mask_src->header.cf == LV_COLOR_FORMAT_L8) {
135 blending_mode |= NEMA_BLOP_STENCIL_TXTY;
136 const lv_image_dsc_t * mask = dsc->bitmap_mask_src;
137 const uint8_t * mask_buf = mask->data;
138
139 const lv_area_t * image_area;
140 lv_area_t mask_area;
141 if(lv_area_get_width(&dsc->image_area) < 0) image_area = img_coords;
142 else image_area = &dsc->image_area;
143
144 lv_area_set(&mask_area, 0, 0, dsc->bitmap_mask_src->header.w - 1, dsc->bitmap_mask_src->header.h - 1);
145 lv_area_align(image_area, &mask_area, LV_ALIGN_CENTER, 0, 0);
146
147 mask_buf += dsc->bitmap_mask_src->header.w * (img_coords->y1 - mask_area.y1) + (img_coords->x1 - mask_area.x1);
148
149 nema_bind_tex(NEMA_TEX3, (uintptr_t)NEMA_VIRT2PHYS(mask_buf), mask->header.w, mask->header.h,
150 lv_nemagfx_mask_cf_to_nema(mask->header.cf),
151 mask->header.stride, NEMA_FILTER_BL);
152 }
153 }
154
155 nema_set_blend_blit(blending_mode);
156
157 if(!has_transform) {
158 nema_blit_rect((img_coords->x1 - layer->buf_area.x1),
159 (img_coords->y1 - layer->buf_area.y1), tex_w, tex_h);
160 }
161 else {
162 /*Calculate the transformed points*/
163 float x0 = (img_coords->x1 - layer->buf_area.x1);
164 float y0 = (img_coords->y1 - layer->buf_area.y1);
165 float x1 = x0 + tex_w ;
166 float y1 = y0;
167 float x2 = x0 + tex_w ;
168 float y2 = y0 + tex_h;
169 float x3 = x0 ;
170 float y3 = y0 + tex_h;
171
172 nema_matrix3x3_t m;
173 nema_mat3x3_load_identity(m);
174 nema_mat3x3_translate(m, -x0, -y0);
175 nema_mat3x3_translate(m, -(float)dsc->pivot.x, -(float)dsc->pivot.y);
176 nema_mat3x3_rotate(m, (dsc->rotation / 10.0f)); /* angle is 1/10 degree */
177 float scale_x = 1.f * dsc->scale_x / LV_SCALE_NONE;
178 float scale_y = 1.f * dsc->scale_y / LV_SCALE_NONE;
179 nema_mat3x3_scale(m, (float)scale_x, (float)scale_y);
180 nema_mat3x3_translate(m, (float)dsc->pivot.x, (float)dsc->pivot.y);
181 nema_mat3x3_translate(m, x0, y0);
182
183 /*Apply Transformation Matrix to Vertices*/
184 nema_mat3x3_mul_vec(m, &x0, &y0);
185 nema_mat3x3_mul_vec(m, &x1, &y1);
186 nema_mat3x3_mul_vec(m, &x2, &y2);
187 nema_mat3x3_mul_vec(m, &x3, &y3);
188
189 nema_blit_quad_fit(x0, y0,
190 x1, y1,
191 x2, y2,
192 x3, y3);
193 }
194
195 nema_cl_submit(&(draw_nema_gfx_unit->cl));
196
197}

199/*NemaGFX does mask operations with A8,A4,A2 and A1 formats*/
200static uint32_t lv_nemagfx_mask_cf_to_nema(lv_color_format_t cf)
201{
202 switch(cf) {
203 case LV_COLOR_FORMAT_A1:
204 return NEMA_A1;
205 case LV_COLOR_FORMAT_A2:
206 return NEMA_A2;
207 case LV_COLOR_FORMAT_A4:
208 return NEMA_A4;
209 case LV_COLOR_FORMAT_A8:
210 case LV_COLOR_FORMAT_L8:
211 default:
212 break;
213 }
214
215 return NEMA_A8;
216}
