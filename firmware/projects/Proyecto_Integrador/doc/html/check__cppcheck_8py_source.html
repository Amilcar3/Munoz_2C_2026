81def run_cppcheck(files: List[str], jobs: int = 1) -> List[Dict]:
82 """Run cppcheck on a list of files and parse results."""
83 if not files:
84 return []
85
86 cmd = (
87 ["cppcheck"]
88 + CPPCHECK_COMMON_ARGS
89 + [
90 f"--template={TEMPLATE}",
91 ]
92 )
93 if jobs > 1:
94 cmd.append(f"-j{jobs}")
95 cmd.extend(files)
96
97 try:
98 result = subprocess.run(cmd, capture_output=True, text=True, timeout=600)
99 except subprocess.TimeoutExpired as exc:
100 print("cppcheck timed out after 600s", file=sys.stderr)
101 raise RuntimeError("cppcheck timed out after 600s") from exc
102
103 # cppcheck returns 0 even when findings exist.
104 # Non-zero typically means internal error (bad args, crash).
105 # Parse stderr regardless, but warn on unexpected exit codes.
106 if result.returncode != 0:
107 print(
108 f"warning: cppcheck exited with code {result.returncode}", file=sys.stderr
109 )
110
111 if "Failed to load library" in result.stderr:
112 raise RuntimeError(
113 f"cppcheck failed to load library configuration:\n{result.stderr.strip()}"
114 )
115
116 issues = []
117 for line in result.stderr.splitlines():
118 parts = line.split("|", 4)
119 if len(parts) == 5:
120 severity, filepath, lineno, checker_id, message = parts
121 if severity in ("error", "warning", "style", "performance", "portability"):
122 issues.append(
123 {
124 "severity": severity,
125 "file": filepath,
126 "line": int(lineno) if lineno.isdigit() else 0,
127 "id": checker_id,
128 "message": message,
129 }
130 )
131
132 # If cppcheck failed AND produced no parseable output, that's a real problem
133 if result.returncode != 0 and not issues and not result.stderr.strip():
134 raise RuntimeError(
135 f"cppcheck exited with code {result.returncode} and no output"
136 )
137
138 return issues

141def get_changed_files(diff_range: str, root: Path) -> List[str]:
142 """Get .c/.h files changed in a git diff range."""
143 cmd = ["git", "diff", "--name-only", "--diff-filter=ACMR", diff_range]
144 result = subprocess.run(cmd, capture_output=True, text=True, cwd=str(root))
145 if result.returncode != 0:
146 stderr = result.stderr.strip() or "unknown error"
147 raise RuntimeError(
148 f"git diff failed for range '{diff_range}' in '{root}': {stderr}"
149 )
150
151 files = []
152 for f in result.stdout.strip().splitlines():
153 f = f.strip()
154 if (
155 f
156 and f.startswith("src/")
157 and (f.endswith(".c") or f.endswith(".h"))
158 and not is_excluded(f)
159 ):
160 full = root / f
161 if full.exists():
162 files.append(str(full))
163 return files

166def get_all_files(root: Path) -> List[str]:
167 """Get all .c/.h files under src/, excluding paths in EXCLUDE_DIRS."""
168 src_dir = root / "src"
169 files = []
170 for ext in ("*.c", "*.h"):
171 for f in src_dir.rglob(ext):
172 rel = str(f.relative_to(root))
173 if not is_excluded(rel):
174 files.append(str(f))
175 return sorted(files)

178def format_github_annotation(issue: Dict, root: Path) -> str:
179 """Format an issue as a GitHub Actions annotation."""
180 try:
181 rel = os.path.relpath(issue["file"], str(root))
182 except ValueError:
183 rel = issue["file"]
184
185 level = "error" if issue["severity"] in BLOCKING_SEVERITIES else "warning"
186
187 line = issue.get("line", 0)
188 attrs = [f"file={rel}"]
189 if line > 0:
190 attrs.append(f"line={line}")
191
192 return f"::{level} {','.join(attrs)}::[cppcheck:{issue['id']}] {issue['message']}"

195def print_summary(issues: List[Dict], root: Path) -> int:
196 """Print results summary. Returns exit code (1 if blocking issues found)."""
197 if not issues:
198 print("✅ cppcheck: no issues found")
199 return 0
200
201 by_severity: Dict[str, List[Dict]] = {}
202 for issue in issues:
203 by_severity.setdefault(issue["severity"], []).append(issue)
204
205 print(f"\ncppcheck found {len(issues)} issue(s):\n")
206
207 for sev in ("error", "warning", "style", "performance", "portability"):
208 group = by_severity.get(sev, [])
209 if group:
210 icon = (
211 "🚫"
212 if sev in BLOCKING_SEVERITIES
213 else "⚠️" if sev in ANNOTATION_SEVERITIES else "📝"
214 )
215 print(f" {icon} {sev}: {len(group)}")
216
217 print()
218
219 for sev in ("error", "warning"):
220 group = by_severity.get(sev, [])
221 for issue in group:
222 try:
223 rel = os.path.relpath(issue["file"], str(root))
224 except ValueError:
225 rel = issue["file"]
226 print(
227 f" {rel}:{issue['line']}: {issue['severity']}: {issue['message']} [{issue['id']}]"
228 )
229
230 ci = os.environ.get("CI") or os.environ.get("GITHUB_ACTIONS")
231 if ci:
232 for issue in issues:
233 if issue["severity"] in BLOCKING_SEVERITIES | ANNOTATION_SEVERITIES:
234 print(format_github_annotation(issue, root))
235
236 errors = by_severity.get("error", [])
237 if errors:
238 print(f"\n🚫 {len(errors)} error(s) found — these must be fixed before merge.")
239 return 1

241 return 0
