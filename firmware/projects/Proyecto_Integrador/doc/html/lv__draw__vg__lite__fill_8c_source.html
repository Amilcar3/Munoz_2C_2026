47void lv_draw_vg_lite_fill(lv_draw_task_t * t, const lv_draw_fill_dsc_t * dsc, const lv_area_t * coords)
48{
49 lv_draw_vg_lite_unit_t * u = (lv_draw_vg_lite_unit_t *)t->draw_unit;
50
51 lv_area_t clip_area;
52 if(!lv_area_intersect(&clip_area, coords, &t->clip_area)) {
53 /*Fully clipped, nothing to do*/
54 return;
55 }
56
57 LV_PROFILER_DRAW_BEGIN;
58
59 vg_lite_matrix_t matrix = u->global_matrix;
60
61 /* Improve GPU rendering efficiency using simpler fill modes */
62 if(dsc->radius == 0 && dsc->opa >= LV_OPA_MAX && dsc->grad.dir == LV_GRAD_DIR_NONE &&
63 lv_matrix_is_identity((lv_matrix_t *)&matrix)) {
64 lv_vg_lite_clear(&u->target_buffer, &clip_area, lv_vg_lite_color(dsc->color, LV_OPA_COVER, false));
65 LV_PROFILER_DRAW_END;
66 return;
67 }
68
69 lv_vg_lite_path_t * path = lv_vg_lite_path_get(u, VG_LITE_FP32);
70 lv_vg_lite_path_set_quality(path, dsc->radius == 0 ? VG_LITE_LOW : VG_LITE_HIGH);
71 lv_vg_lite_path_set_bounding_box_area(path, &clip_area);
72 lv_vg_lite_path_append_rect(path,
73 coords->x1, coords->y1,
74 lv_area_get_width(coords), lv_area_get_height(coords),
75 dsc->radius);
76 lv_vg_lite_path_end(path);
77
78 if(dsc->grad.dir != LV_GRAD_DIR_NONE) {
79#if LV_USE_VECTOR_GRAPHIC
80 lv_vg_lite_draw_grad_helper(
81 u->grad_ctx,
82 &u->target_buffer,
83 lv_vg_lite_path_get_path(path),
84 coords,
85 &dsc->grad,
86 &matrix,
87 VG_LITE_FILL_EVEN_ODD,
88 VG_LITE_BLEND_SRC_OVER);
89#else
90 LV_LOG_WARN("Gradient fill is not supported without VECTOR_GRAPHIC");
91#endif
92 }
93 else {
94 /* normal fill */
95 lv_vg_lite_draw(
96 &u->target_buffer,
97 lv_vg_lite_path_get_path(path),
98 VG_LITE_FILL_EVEN_ODD,
99 &matrix,
100 VG_LITE_BLEND_SRC_OVER,
101 lv_vg_lite_color(dsc->color, dsc->opa, true));
102 }
103
104 lv_vg_lite_path_drop(u, path);
105
106 LV_PROFILER_DRAW_END;
107}
