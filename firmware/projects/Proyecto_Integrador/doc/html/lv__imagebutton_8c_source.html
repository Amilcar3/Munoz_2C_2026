119void lv_imagebutton_set_state(lv_obj_t * obj, lv_imagebutton_state_t state)
120{
121 LV_CHECK_OBJ(obj, MY_CLASS, return);
122
123 lv_state_t obj_state = LV_STATE_DEFAULT;
124 if(state == LV_IMAGEBUTTON_STATE_PRESSED ||
125 state == LV_IMAGEBUTTON_STATE_CHECKED_PRESSED) obj_state |= LV_STATE_PRESSED;
126 if(state == LV_IMAGEBUTTON_STATE_DISABLED ||
127 state == LV_IMAGEBUTTON_STATE_CHECKED_DISABLED) obj_state |= LV_STATE_DISABLED;
128 if(state == LV_IMAGEBUTTON_STATE_CHECKED_DISABLED || state == LV_IMAGEBUTTON_STATE_CHECKED_PRESSED ||
129 state == LV_IMAGEBUTTON_STATE_CHECKED_RELEASED) {
130 obj_state |= LV_STATE_CHECKED;
131 }
132
133 lv_obj_remove_state(obj, LV_STATE_CHECKED | LV_STATE_PRESSED | LV_STATE_DISABLED);
134 lv_obj_add_state(obj, obj_state);
135
136 refr_image(obj);
137}

184static void lv_imagebutton_event(const lv_obj_class_t * class_p, lv_event_t * e)
185{
186 LV_ASSERT(e != NULL);
187 LV_UNUSED(class_p);
188
189 lv_result_t res = lv_obj_event_base(&lv_imagebutton_class, e);
190 if(res != LV_RESULT_OK) return;
191
192 lv_event_code_t code = lv_event_get_code(e);
193 lv_obj_t * obj = lv_event_get_current_target(e);
194 if(code == LV_EVENT_PRESSED || code == LV_EVENT_RELEASED || code == LV_EVENT_PRESS_LOST) {
195 refr_image(obj);
196 }
197 else if(code == LV_EVENT_DRAW_MAIN) {
198 draw_main(e);
199 }
200 else if(code == LV_EVENT_COVER_CHECK) {
201 lv_cover_check_info_t * info = lv_event_get_param(e);
202 if(info->res != LV_COVER_RES_MASKED) info->res = LV_COVER_RES_NOT_COVER;
203 }
204 else if(code == LV_EVENT_GET_SELF_SIZE) {
205 lv_point_t * p = lv_event_get_self_size_info(e);
206 lv_imagebutton_t * imagebutton = (lv_imagebutton_t *)obj;
207 lv_imagebutton_state_t state = suggest_state(obj, get_state(obj));
208 if(imagebutton->src_left[state].img_src == NULL &&
209 imagebutton->src_mid[state].img_src != NULL &&
210 imagebutton->src_right[state].img_src == NULL) {
211 p->x = LV_MAX(p->x, imagebutton->src_mid[state].header.w);
212 }
213 }
214}

216static void draw_main(lv_event_t * e)
217{
218 LV_ASSERT(e != NULL);
219 lv_obj_t * obj = lv_event_get_current_target(e);
220 lv_imagebutton_t * imagebutton = (lv_imagebutton_t *)obj;
221 lv_layer_t * layer = lv_event_get_layer(e);
222
223 /*Just draw_main an image*/
224 lv_imagebutton_state_t state = suggest_state(obj, get_state(obj));
225
226 /*Simply draw the middle src if no tiled*/
227 lv_imagebutton_src_info_t * src_info = &imagebutton->src_left[state];
228
229 int32_t tw = lv_obj_get_style_transform_width_internal(obj, LV_PART_MAIN);
230 int32_t th = lv_obj_get_style_transform_height_internal(obj, LV_PART_MAIN);
231 lv_area_t coords;
232 coords = obj->coords;
233 lv_area_increase(&coords, tw, th);
234
235 lv_draw_image_dsc_t img_dsc;
236 lv_draw_image_dsc_init(&img_dsc);
237 img_dsc.base.layer = layer;
238 lv_obj_init_draw_image_dsc(obj, LV_PART_MAIN, &img_dsc);
239
240 lv_area_t coords_part;
241 int32_t left_w = 0;
242 int32_t right_w = 0;
243
244 if(src_info->img_src) {
245 left_w = src_info->header.w;
246 coords_part.x1 = coords.x1;
247 coords_part.y1 = coords.y1;
248 coords_part.x2 = coords.x1 + src_info->header.w - 1;
249 coords_part.y2 = coords.y1 + src_info->header.h - 1;
250 img_dsc.src = src_info->img_src;
251 lv_draw_image(layer, &img_dsc, &coords_part);
252 }
253
254 src_info = &imagebutton->src_right[state];
255 if(src_info->img_src) {
256 right_w = src_info->header.w;
257 coords_part.x1 = coords.x2 - src_info->header.w + 1;
258 coords_part.y1 = coords.y1;
259 coords_part.x2 = coords.x2;
260 coords_part.y2 = coords.y1 + src_info->header.h - 1;
261 img_dsc.src = src_info->img_src;
262 lv_draw_image(layer, &img_dsc, &coords_part);
263 }
264
265 src_info = &imagebutton->src_mid[state];
266 if(src_info->img_src) {
267 coords_part.x1 = coords.x1 + left_w;
268 coords_part.x2 = coords.x2 - right_w;
269 coords_part.y1 = coords.y1;
270 coords_part.y2 = coords.y2;
271
272 lv_area_t clip_area_center;
273 if(lv_area_intersect(&clip_area_center, &coords_part, &layer->_clip_area)) {
274 lv_area_t clip_area_ori = layer->_clip_area;
275 layer->_clip_area = clip_area_center;
276 img_dsc.src = src_info->img_src;
277 img_dsc.tile = 1;
278 lv_draw_image(layer, &img_dsc, &coords_part);
279 layer->_clip_area = clip_area_ori;
280 }
281
282 }
283}

285static void refr_image(lv_obj_t * obj)
286{
287 LV_ASSERT(obj != NULL);
288 lv_imagebutton_t * imagebutton = (lv_imagebutton_t *)obj;
289 lv_imagebutton_state_t state = suggest_state(obj, get_state(obj));
290
291 const void * src = imagebutton->src_mid[state].img_src;
292 if(src == NULL) return;
293
294 lv_obj_refresh_self_size(obj);
295 lv_obj_set_height(obj, imagebutton->src_mid[state].header.h); /*Keep the user defined width*/
296
297 lv_obj_invalidate(obj);
298}

307static lv_imagebutton_state_t suggest_state(lv_obj_t * obj, lv_imagebutton_state_t state)
308{
309 LV_ASSERT(obj != NULL);
310 lv_imagebutton_t * imagebutton = (lv_imagebutton_t *)obj;
311 if(imagebutton->src_mid[state].img_src == NULL) {
312 switch(state) {
313 case LV_IMAGEBUTTON_STATE_PRESSED:
314 if(imagebutton->src_mid[LV_IMAGEBUTTON_STATE_RELEASED].img_src) return LV_IMAGEBUTTON_STATE_RELEASED;
315 break;
316 case LV_IMAGEBUTTON_STATE_CHECKED_RELEASED:
317 if(imagebutton->src_mid[LV_IMAGEBUTTON_STATE_RELEASED].img_src) return LV_IMAGEBUTTON_STATE_RELEASED;
318 break;
319 case LV_IMAGEBUTTON_STATE_CHECKED_PRESSED:
320 if(imagebutton->src_mid[LV_IMAGEBUTTON_STATE_CHECKED_RELEASED].img_src) return LV_IMAGEBUTTON_STATE_CHECKED_RELEASED;
321 if(imagebutton->src_mid[LV_IMAGEBUTTON_STATE_PRESSED].img_src) return LV_IMAGEBUTTON_STATE_PRESSED;
322 if(imagebutton->src_mid[LV_IMAGEBUTTON_STATE_RELEASED].img_src) return LV_IMAGEBUTTON_STATE_RELEASED;
323 break;
324 case LV_IMAGEBUTTON_STATE_DISABLED:
325 if(imagebutton->src_mid[LV_IMAGEBUTTON_STATE_RELEASED].img_src) return LV_IMAGEBUTTON_STATE_RELEASED;
326 break;
327 case LV_IMAGEBUTTON_STATE_CHECKED_DISABLED:
328 if(imagebutton->src_mid[LV_IMAGEBUTTON_STATE_CHECKED_RELEASED].img_src) return LV_IMAGEBUTTON_STATE_CHECKED_RELEASED;
329 if(imagebutton->src_mid[LV_IMAGEBUTTON_STATE_RELEASED].img_src) return LV_IMAGEBUTTON_STATE_RELEASED;
330 break;
331 default:
332 break;
333 }
334 }
335
336 return state;
337}

339static lv_imagebutton_state_t get_state(const lv_obj_t * imagebutton)
340{
341 LV_ASSERT(imagebutton != NULL);
342
343 lv_state_t obj_state = lv_obj_get_state(imagebutton);
344
345 if(obj_state & LV_STATE_DISABLED) {
346 if(obj_state & LV_STATE_CHECKED) return LV_IMAGEBUTTON_STATE_CHECKED_DISABLED;
347 else return LV_IMAGEBUTTON_STATE_DISABLED;
348 }
349
350 if(obj_state & LV_STATE_CHECKED) {
351 if(obj_state & LV_STATE_PRESSED) return LV_IMAGEBUTTON_STATE_CHECKED_PRESSED;
352 else return LV_IMAGEBUTTON_STATE_CHECKED_RELEASED;
353 }
354 else {
355 if(obj_state & LV_STATE_PRESSED) return LV_IMAGEBUTTON_STATE_PRESSED;
356 else return LV_IMAGEBUTTON_STATE_RELEASED;
357 }
358}

360static void update_src_info(lv_imagebutton_src_info_t * info, const void * src)
361{
362 LV_ASSERT(info != NULL);
363 if(!src) {
364 lv_memzero(info, sizeof(lv_imagebutton_src_info_t));
365 return;
366 }
367
368 lv_result_t res = lv_image_decoder_get_info(src, &info->header);
369 if(res != LV_RESULT_OK) {
370 LV_LOG_WARN("can't get info");
371 return;
372 }
373
374 info->img_src = src;
375}
