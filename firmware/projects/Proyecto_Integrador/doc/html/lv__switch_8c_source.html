147static void lv_switch_event(const lv_obj_class_t * class_p, lv_event_t * e)
148{
149 LV_UNUSED(class_p);
150
151 lv_result_t res;
152
153 /*Call the ancestor's event handler*/
154 res = lv_obj_event_base(MY_CLASS, e);
155 if(res != LV_RESULT_OK) return;
156
157 lv_event_code_t code = lv_event_get_code(e);
158 lv_obj_t * obj = lv_event_get_current_target(e);
159
160 if(code == LV_EVENT_REFR_EXT_DRAW_SIZE) {
161 int32_t knob_left = lv_obj_get_style_pad_left_internal(obj, LV_PART_KNOB);
162 int32_t knob_right = lv_obj_get_style_pad_right_internal(obj, LV_PART_KNOB);
163 int32_t knob_top = lv_obj_get_style_pad_top_internal(obj, LV_PART_KNOB);
164 int32_t knob_bottom = lv_obj_get_style_pad_bottom_internal(obj, LV_PART_KNOB);
165
166 /*The smaller size is the knob diameter*/
167 int32_t knob_size = LV_MAX4(knob_left, knob_right, knob_bottom, knob_top);
168 knob_size += LV_SWITCH_KNOB_EXT_AREA_CORRECTION;
169 knob_size += lv_obj_calculate_ext_draw_size(obj, LV_PART_KNOB);
170
171 int32_t * s = lv_event_get_param(e);
172 *s = LV_MAX(*s, knob_size);
173 *s = LV_MAX(*s, lv_obj_calculate_ext_draw_size(obj, LV_PART_INDICATOR));
174 }
175 else if(code == LV_EVENT_STATE_CHANGED) {
176 lv_state_t prev_state = lv_event_get_prev_state(e);
177 lv_state_t diff = prev_state ^ lv_obj_get_state(obj);
178
179 if(diff & LV_STATE_CHECKED) {
180 lv_switch_trigger_anim(obj);
181 lv_obj_invalidate(obj);
182 }
183 }
184 else if(code == LV_EVENT_DRAW_MAIN) {
185 draw_main(e);
186 }
187}

189static void draw_main(lv_event_t * e)
190{
191 lv_obj_t * obj = lv_event_get_current_target(e);
192 lv_switch_t * sw = (lv_switch_t *)obj;
193
194 lv_layer_t * layer = lv_event_get_layer(e);
195
196 /*Draw the indicator*/
197 lv_area_t indic_area;
198 /*Exclude background's padding*/
199 lv_obj_get_content_coords(obj, &indic_area);
200
201 lv_draw_rect_dsc_t draw_indic_dsc;
202 lv_draw_rect_dsc_init(&draw_indic_dsc);
203 draw_indic_dsc.base.layer = layer;
204 lv_obj_init_draw_rect_dsc(obj, LV_PART_INDICATOR, &draw_indic_dsc);
205 lv_draw_rect(layer, &draw_indic_dsc, &indic_area);
206
207 /*Draw the knob*/
208 lv_area_t knob_area;
209 knob_area = obj->coords;
210
211 int32_t switch_w = lv_area_get_width(&obj->coords);
212 int32_t switch_h = lv_area_get_height(&obj->coords);
213 bool hor = false;
214
215 switch(sw->orientation) {
216 case LV_SWITCH_ORIENTATION_HORIZONTAL:
217 hor = true;
218 break;
219 case LV_SWITCH_ORIENTATION_VERTICAL:
220 hor = false;
221 break;
222 case LV_SWITCH_ORIENTATION_AUTO:
223 default:
224 hor = (switch_w >= switch_h);
225 break;
226 }
227
228 if(hor) {
229 int32_t anim_value_x = 0;
230 int32_t knob_size = lv_obj_get_height(obj);
231 int32_t anim_length = lv_area_get_width(&obj->coords) - knob_size;
232 if(LV_SWITCH_IS_ANIMATING(sw)) {
233 /* Use the animation's coordinate */
234 anim_value_x = (anim_length * sw->anim_state) / LV_SWITCH_ANIM_STATE_END;
235 }
236 else {
237 /* Use LV_STATE_CHECKED to decide the coordinate */
238 bool chk = lv_obj_get_state(obj) & LV_STATE_CHECKED;
239 anim_value_x = chk ? anim_length : 0;
240 }
241
242 if(LV_BASE_DIR_RTL == lv_obj_get_style_base_dir_internal(obj, LV_PART_MAIN)) {
243 anim_value_x = anim_length - anim_value_x;
244 }
245 knob_area.x1 += anim_value_x;
246 knob_area.x2 = knob_area.x1 + (knob_size > 0 ? knob_size - 1 : 0);
247 }
248 else {
249 int32_t anim_value_y = 0;
250 int32_t knob_size = lv_obj_get_width(obj);
251 int32_t anim_length = lv_area_get_height(&obj->coords) - knob_size;
252 if(LV_SWITCH_IS_ANIMATING(sw)) {
253 /* Use the animation's coordinate */
254 anim_value_y = (anim_length * sw->anim_state) / LV_SWITCH_ANIM_STATE_END;
255 }
256 else {
257 /* Use LV_STATE_CHECKED to decide the coordinate */
258 bool chk = lv_obj_get_state(obj) & LV_STATE_CHECKED;
259 anim_value_y = chk ? anim_length : 0;
260 }
261
262 if(LV_BASE_DIR_RTL == lv_obj_get_style_base_dir_internal(obj, LV_PART_MAIN)) {
263 anim_value_y = anim_length - anim_value_y;
264 }
265
266 knob_area.y2 -= anim_value_y;
267 knob_area.y1 = knob_area.y2 - (knob_size > 0 ? knob_size - 1 : 0);
268 }
269
270 int32_t knob_left = lv_obj_get_style_pad_left_internal(obj, LV_PART_KNOB);
271 int32_t knob_right = lv_obj_get_style_pad_right_internal(obj, LV_PART_KNOB);
272 int32_t knob_top = lv_obj_get_style_pad_top_internal(obj, LV_PART_KNOB);
273 int32_t knob_bottom = lv_obj_get_style_pad_bottom_internal(obj, LV_PART_KNOB);
274
275 /*Apply the paddings on the knob area*/
276 knob_area.x1 -= knob_left;
277 knob_area.x2 += knob_right;
278 knob_area.y1 -= knob_top;
279 knob_area.y2 += knob_bottom;
280
281 lv_draw_rect_dsc_t knob_rect_dsc;
282 lv_draw_rect_dsc_init(&knob_rect_dsc);
283 knob_rect_dsc.base.layer = layer;
284 lv_obj_init_draw_rect_dsc(obj, LV_PART_KNOB, &knob_rect_dsc);
285
286 lv_draw_rect(layer, &knob_rect_dsc, &knob_area);
287}

310static void lv_switch_trigger_anim(lv_obj_t * obj)
311{
312 LV_CHECK_OBJ(obj, MY_CLASS, return);
313 /*If the widget is not rendered yet show state changes immediately*/
314 if(!obj->rendered) return;
315
316 lv_switch_t * sw = (lv_switch_t *)obj;
317
318 uint32_t anim_dur_full = lv_obj_get_style_anim_duration_internal(obj, LV_PART_MAIN);
319
320 if(anim_dur_full > 0) {
321 bool chk = lv_obj_get_state(obj) & LV_STATE_CHECKED;
322 int32_t anim_start;
323 int32_t anim_end;
324 /*No animation in progress -> simply set the values*/
325 if(sw->anim_state == LV_SWITCH_ANIM_STATE_INV) {
326 anim_start = chk ? LV_SWITCH_ANIM_STATE_START : LV_SWITCH_ANIM_STATE_END;
327 anim_end = chk ? LV_SWITCH_ANIM_STATE_END : LV_SWITCH_ANIM_STATE_START;
328 }
329 /*Animation in progress. Start from the animation end value*/
330 else {
331 anim_start = sw->anim_state;
332 anim_end = chk ? LV_SWITCH_ANIM_STATE_END : LV_SWITCH_ANIM_STATE_START;
333 }
334 /*Calculate actual animation duration*/
335 uint32_t anim_dur = (anim_dur_full * LV_ABS(anim_start - anim_end)) / LV_SWITCH_ANIM_STATE_END;
336
337 /*Stop the previous animation if it exists*/
338 lv_anim_delete(sw, NULL);
339
340 lv_anim_t a;
341 lv_anim_init(&a);
342 lv_anim_set_var(&a, sw);
343 lv_anim_set_exec_cb(&a, lv_switch_anim_exec_cb);
344 lv_anim_set_values(&a, anim_start, anim_end);
345 lv_anim_set_completed_cb(&a, lv_switch_anim_completed);
346 lv_anim_set_duration(&a, anim_dur);
347 lv_anim_start(&a);
348 }
349}
