142static inline lv_color32_t LV_ATTRIBUTE_FAST_MEM lv_color32_unpremultiply(lv_color32_t c)
143{
144 if(c.alpha == 0) {
145 c.red = 0;
146 c.green = 0;
147 c.blue = 0;
148 }
149 else if(c.alpha != LV_OPA_COVER) {
150 uint32_t reciprocal = (255u * 256u) / c.alpha;
151 uint32_t r = (c.red * reciprocal) >> 8;
152 uint32_t g = (c.green * reciprocal) >> 8;
153 uint32_t b = (c.blue * reciprocal) >> 8;
154 c.red = (uint8_t)(r > 255 ? 255 : r);
155 c.green = (uint8_t)(g > 255 ? 255 : g);
156 c.blue = (uint8_t)(b > 255 ? 255 : b);
157 }
158
159 return c;
160}

187static inline uint16_t LV_ATTRIBUTE_FAST_MEM lv_color_16_16_mix_inlined(uint16_t c1, uint16_t c2, uint8_t mix)
188{
189 if(mix == 255) return c1;
190 if(mix == 0) return c2;
191 if(c1 == c2) return c1;
192
193 /* Source: https://stackoverflow.com/a/50012418/1999969*/
194 uint32_t mix5 = ((uint32_t)mix + 4) >> 3;
195
196 /*0x7E0F81F = 0b00000111111000001111100000011111*/
197 uint32_t bg = (uint32_t)(c2 | ((uint32_t)c2 << 16)) & 0x7E0F81F;
198 uint32_t fg = (uint32_t)(c1 | ((uint32_t)c1 << 16)) & 0x7E0F81F;
199 uint32_t result = ((((fg - bg) * mix5) >> 5) + bg) & 0x7E0F81F;
200
201 return (uint16_t)((result >> 16) | result);
202}

214static inline lv_color32_t LV_ATTRIBUTE_FAST_MEM lv_color_mix32_inlined(lv_color32_t fg, lv_color32_t bg)
215{
216 if(fg.alpha >= LV_OPA_MAX) {
217 fg.alpha = bg.alpha;
218 return fg;
219 }
220 if(fg.alpha <= LV_OPA_MIN) {
221 return bg;
222 }
223
224 uint32_t mix_inv = 255 - fg.alpha;
225 bg.red = (uint8_t)LV_UDIV255((uint32_t)fg.red * fg.alpha + (uint32_t)bg.red * mix_inv);
226 bg.green = (uint8_t)LV_UDIV255((uint32_t)fg.green * fg.alpha + (uint32_t)bg.green * mix_inv);
227 bg.blue = (uint8_t)LV_UDIV255((uint32_t)fg.blue * fg.alpha + (uint32_t)bg.blue * mix_inv);
228 return bg;
229}
