7/* Tests for deferred blur-invalidation expansion.
8 *
9 * A blur object samples the pixels behind it, so when anything behind it
10 * changes it must be redrawn in full. These tests drive the real path: a
11 * normal object behind a blur object is invalidated, a refresh is run (which
12 * runs the once-per-frame expansion), and the areas reported through
13 * LV_EVENT_INVALIDATE_AREA are inspected to confirm the blur object's area was
14 * (or was not) invalidated. */

33/* Record every area the display is asked to invalidate while capturing. */
34static void invalidate_area_cb(lv_event_t * e)
35{
36 TEST_ASSERT_LESS_THAN_UINT32(MAX_CAPTURED, captured_cnt);
37 lv_area_t * area = lv_event_get_param(e);
38 captured[captured_cnt++] = *area;
39}

41/* Start capturing invalidated areas. The callback is only attached while
42 * capturing, so the captured list holds exactly the areas the action under test
43 * produced, not the areas from creating and first rendering the objects. */
44static void start_capture(void)
45{
46 captured_cnt = 0;
47 lv_display_add_event_cb(disp, invalidate_area_cb, LV_EVENT_INVALIDATE_AREA, NULL);
48}

50static void stop_capture(void)
51{
52 lv_display_remove_event_cb_with_user_data(disp, invalidate_area_cb, NULL);
53}

55/* Invalidate an object and run a refresh (which runs the blur expansion),
56 * capturing the invalidated areas it produces. */
57static void invalidate_and_refresh(lv_obj_t * obj)
58{
59 start_capture();
60 lv_obj_invalidate(obj);
61 lv_refr_now(NULL);
62 stop_capture();
63}

65/* True if any captured invalidated area overlaps the given area. */
66static bool area_was_invalidated(const lv_area_t * area)
67{
68 for(uint32_t i = 0; i < captured_cnt; i++) {
69 if(lv_area_is_on(&captured[i], area)) return true;
70 }
71 return false;
72}

74/* True if any captured invalidated area overlaps the object's coordinates. */

108/* Invalidating an object behind a blur object must also invalidate the blur
109 * object's full extent because its background changed. The changed object is
110 * small and in one corner. The assertion checks the opposite corner of the
111 * blur object, which only the full expansion can invalidate. */
112void test_change_behind_blur_invalidates_the_blur_object(void)
113{
114 skip_in_full_render_mode();
115 lv_obj_t * blur = create_blur_obj(100, 100, 80, 80); /* ~100..180 */
116 lv_obj_t * behind = create_obj(105, 105, 10, 10); /* top-left corner, inside blur */
117 lv_refr_now(NULL);
118
119 /* A patch in the blur object's far (bottom-right) corner, away from the
120 * changed object so its own invalidation cannot cover it. */
121 lv_area_t far_corner = {blur->coords.x2 - 10, blur->coords.y2 - 10, blur->coords.x2, blur->coords.y2};
122 TEST_ASSERT_FALSE(lv_area_is_on(&behind->coords, &far_corner));
123
124 invalidate_and_refresh(behind);
125
126 TEST_ASSERT_TRUE(area_was_invalidated(&far_corner));
127}

143/* Transitive chain: blur object b1 sits over the changed object and blur object
144 * b2 overlaps b1's right edge but not the changed object. Expanding b1
145 * invalidates b1's area, and the fixed-point loop must then see b2 overlapping
146 * that area and invalidate b2 too. A single pass would miss b2.
147 *
148 * The assertion checks b2's far-right region, beyond b1 and its blur-expanded
149 * extent, so only b2's own expansion can invalidate it. */
150void test_blur_over_blur_is_invalidated_transitively(void)
151{
152 skip_in_full_render_mode();
153 lv_obj_t * b1 = create_blur_obj(100, 100, 80, 80); /* coords ~100..179 */
154 lv_obj_t * b2 = create_blur_obj(175, 100, 80, 80); /* coords ~175..254, sliver-overlaps b1 */
155 lv_obj_t * behind = create_obj(105, 105, 10, 10); /* inside b1 only, not b2 */
156 lv_refr_now(NULL);
157
158 /* The changed object is inside b1 but not b2, so b2 can only be reached
159 * transitively through b1. */
160 TEST_ASSERT_TRUE(lv_area_is_on(&behind->coords, &b1->coords));
161 TEST_ASSERT_FALSE(lv_area_is_on(&behind->coords, &b2->coords));
162
163 /* A patch at b2's far-right edge, clear of b1 and its blur-expanded extent,
164 * so only b2's own expansion can invalidate it. */
165 lv_area_t b2_far = {b2->coords.x2 - 10, b2->coords.y1, b2->coords.x2, b2->coords.y1 + 10};
166 TEST_ASSERT_TRUE(b2_far.x1 > b1->coords.x2 + 32); /* clear of b1 + any blur expansion */
167
168 invalidate_and_refresh(behind);
169
170 TEST_ASSERT_TRUE(obj_area_was_invalidated(b1)); /* b1: direct overlap with the change */
171 TEST_ASSERT_TRUE(area_was_invalidated(&b2_far)); /* b2: only reachable via transitive expansion */
172}

174/* A blur object that is a descendant of the invalidated object must be expanded
175 * too. The child's full extent must be redrawn even when the change overlaps
176 * only a corner of it. */
177void test_blur_child_of_invalidated_parent_is_expanded(void)

184 lv_obj_set_size(blur_child, 100, 100);
185 lv_obj_set_style_blur_radius(blur_child, 10, 0);
186 lv_refr_now(NULL);

191 };
192
193 /* Far corner of the child, well away from the patch. Only a full expansion

227/* The expansion handles a frame with many invalidations at once, so the inner
228 * loop must scan more than one invalidated area. Two disjoint objects in
229 * opposite corners of a blur object are invalidated in the same frame (with
230 * these coordinates they stay separate rather than being joined), and the blur
231 * object's center, covered by neither, must still be invalidated. Only the
232 * full expansion can do that. */
233void test_multiple_disjoint_changes_behind_blur(void)
234{
235 skip_in_full_render_mode();
236 lv_obj_t * blur = create_blur_obj(100, 100, 120, 120); /* ~100..220 */
237 lv_obj_t * behind_tl = create_obj(105, 105, 10, 10); /* top-left */
238 lv_obj_t * behind_br = create_obj(205, 205, 10, 10); /* bottom-right */
239 lv_refr_now(NULL);
240
241 /* Center patch, covered by neither changed object. */
242 int32_t cx = (blur->coords.x1 + blur->coords.x2) / 2;
243 int32_t cy = (blur->coords.y1 + blur->coords.y2) / 2;
244 lv_area_t center = {cx - 5, cy - 5, cx + 5, cy + 5};
245 TEST_ASSERT_FALSE(lv_area_is_on(&behind_tl->coords, &center));
246 TEST_ASSERT_FALSE(lv_area_is_on(&behind_br->coords, &center));
247
248 start_capture();
249 lv_obj_invalidate(behind_tl);
250 lv_obj_invalidate(behind_br);
251 /* Both invalidations were recorded as separate areas, so the expansion
252 * runs with more than one area to scan. */
253 TEST_ASSERT_GREATER_THAN_UINT32(1, captured_cnt);
254 lv_refr_now(NULL);
255 stop_capture();
256
257 TEST_ASSERT_TRUE(area_was_invalidated(&center));
258}

260/* A transformed blur object renders at its transformed area, not at its
261 * untransformed coordinates. A change behind the transformed area must
262 * invalidate all of it. */
263void test_change_behind_rotated_blur_invalidates_transformed_extent(void)
264{
265 skip_in_full_render_mode();
266 lv_obj_t * blur = create_blur_obj(300, 200, 120, 120);
267 lv_obj_set_style_transform_pivot_x(blur, 0, 0);
268 lv_obj_set_style_transform_pivot_y(blur, 0, 0);
269 lv_obj_set_style_transform_rotation(blur, 900, 0); /* 90 degrees */
270 lv_refr_now(NULL);
271
272 /* Where the blur object actually renders. */
273 lv_area_t tr;
274 tr = blur->coords;
275 lv_obj_get_transformed_area(blur, &tr, LV_OBJ_POINT_TRANSFORM_FLAG_RECURSIVE);
276
277 /* The rotation must move the transformed area almost fully off the
278 * untransformed coordinates, otherwise this test could pass without any
279 * transform handling at all. */
280 lv_area_t overlap;
281 bool on = lv_area_intersect(&overlap, &tr, &blur->coords);
282 TEST_ASSERT_TRUE(!on || lv_area_get_width(&overlap) <= 2 || lv_area_get_height(&overlap) <= 2);
283
284 /* A change behind the transformed area, near its top-left corner. */
285 lv_obj_t * behind = create_obj(tr.x1 + 16, tr.y1 + 16, 10, 10);
286 lv_refr_now(NULL);
287
288 /* Probe the center of the transformed area. It is away from the changed
289 * object, the area's edges and the untransformed coordinates, so only a
290 * full expansion of the transformed area can invalidate it. */
291 int32_t cx = (tr.x1 + tr.x2) / 2;
292 int32_t cy = (tr.y1 + tr.y2) / 2;
293 lv_area_t probe = {cx - 5, cy - 5, cx + 5, cy + 5};
294 TEST_ASSERT_FALSE(lv_area_is_on(&behind->coords, &probe));
295 TEST_ASSERT_FALSE(lv_area_is_on(&blur->coords, &probe));
296
297 invalidate_and_refresh(behind);
298
299 TEST_ASSERT_TRUE(area_was_invalidated(&probe));
300}
