88static void show_page(lv_obj_t * page)
89{
90 if(menu_cur_page) lv_obj_set_hidden(menu_cur_page, true);
91 lv_obj_set_hidden(page, false);
92 menu_cur_page = page;
93
94 /*The back button is only useful when we left the root page*/
95 if(page == menu_root_page) lv_obj_set_hidden(menu_back_btn, true);
96 else lv_obj_set_hidden(menu_back_btn, false);
97}

114static lv_obj_t * menu_create(lv_obj_t * parent)
115{
116 lv_obj_t * menu = lv_obj_create(parent);
117 lv_obj_set_flex_flow(menu, LV_FLEX_FLOW_COLUMN);
118 lv_obj_add_style(menu, &style_menu, 0);
119
120 lv_obj_t * header = lv_obj_create(menu);
121 lv_obj_set_width(header, lv_pct(100));
122 lv_obj_set_height(header, LV_SIZE_CONTENT);
123 lv_obj_set_flex_flow(header, LV_FLEX_FLOW_ROW);
124 lv_obj_set_flex_align(header, LV_FLEX_ALIGN_START, LV_FLEX_ALIGN_CENTER, LV_FLEX_ALIGN_CENTER);
125 lv_obj_add_style(header, &style_header, 0);
126
127 menu_back_btn = lv_button_create(header);
128 lv_obj_add_style(menu_back_btn, &style_header_btn, 0);
129 lv_obj_add_style(menu_back_btn, &style_pressed, LV_STATE_PRESSED);
130#if LV_USE_IMAGE == 1
131 lv_obj_t * back_img = lv_image_create(menu_back_btn);
132 lv_image_set_src(back_img, LV_SYMBOL_LEFT);
133 lv_obj_center(back_img);
134#endif
135 lv_obj_add_event_cb(menu_back_btn, back_event, LV_EVENT_CLICKED, NULL);
136
137 lv_obj_t * title = lv_label_create(header);
138 lv_label_set_text(title, "Menu");
139
140 menu_content = lv_obj_create(menu);
141 lv_obj_set_width(menu_content, lv_pct(100));
142 lv_obj_set_flex_grow(menu_content, 1);
143 lv_obj_add_style(menu_content, &style_flat, 0);
144
145 return menu;
146}

193void lv_example_menu_navigation(void)
194{
195 styles_init();
196
197 lv_obj_t * menu = menu_create(lv_screen_active());
198 lv_display_t * display = lv_display_get_default();
199 lv_obj_set_size(menu, lv_display_get_horizontal_resolution(display), lv_display_get_vertical_resolution(display));
200 lv_obj_center(menu);
201
202 /*Create a sub page*/
203 lv_obj_t * sub_page = menu_page_create();
204 lv_obj_t * label = lv_label_create(sub_page);
205 lv_label_set_text(label, "Hello, I am hiding here");
206
207 /*Create the main page*/
208 lv_obj_t * main_page = menu_page_create();
209 menu_add_item(main_page, "Item 1");
210 menu_add_item(main_page, "Item 2");
211 lv_obj_t * item3 = menu_add_item(main_page, "Item 3 (Click me!)");
212 menu_set_load_page_event(item3, sub_page);
213
214 /*Show the main page first*/
215 menu_root_page = main_page;
216 show_page(main_page);
217}
