105void lv_tileview_set_tile_by_index(lv_obj_t * tv, uint32_t col_id, uint32_t row_id, lv_anim_enable_t anim_en)
106{
107 LV_CHECK_OBJ(tv, &lv_tileview_class, return);
108
109 lv_obj_update_layout(tv);
110
111 int32_t w = lv_obj_get_content_width(tv);
112 int32_t h = lv_obj_get_content_height(tv);
113
114 int32_t tx = col_id * w;
115 int32_t ty = row_id * h;
116
117 uint32_t i;
118 for(i = 0; i < lv_obj_get_child_count(tv); i++) {
119 lv_obj_t * tile_obj = lv_obj_get_child(tv, i);
120 int32_t x = lv_obj_get_x(tile_obj);
121 int32_t y = lv_obj_get_y(tile_obj);
122 if(x == tx && y == ty) {
123 lv_tileview_set_tile(tv, tile_obj, anim_en);
124 return;
125 }
126 }
127
128 LV_LOG_WARN("No tile found with at (%d,%d) index", (int)col_id, (int)row_id);
129}

161static void tileview_event_cb(lv_event_t * e)
162{
163 lv_event_code_t code = lv_event_get_code(e);
164 lv_obj_t * obj = lv_event_get_current_target(e);
165 lv_tileview_t * tv = (lv_tileview_t *) obj;
166
167 if(code == LV_EVENT_SCROLL_END) {
168 lv_indev_t * indev = lv_indev_active();
169 if(indev && indev->state == LV_INDEV_STATE_PRESSED) {
170 return;
171 }
172
173 int32_t w = lv_obj_get_content_width(obj);
174 int32_t h = lv_obj_get_content_height(obj);
175
176 lv_point_t scroll_end;
177 lv_obj_get_scroll_end(obj, &scroll_end);
178 int32_t left = scroll_end.x;
179 int32_t top = scroll_end.y;
180
181 int32_t tx = ((left + (w / 2)) / w) * w;
182 int32_t ty = ((top + (h / 2)) / h) * h;
183
184 lv_dir_t dir = LV_DIR_ALL;
185 uint32_t i;
186 for(i = 0; i < lv_obj_get_child_count(obj); i++) {
187 lv_obj_t * tile_obj = lv_obj_get_child(obj, i);
188 int32_t x = lv_obj_get_x(tile_obj);
189 int32_t y = lv_obj_get_y(tile_obj);
190 if(x == tx && y == ty) {
191 lv_tileview_tile_t * tile = (lv_tileview_tile_t *)tile_obj;
192 tv->tile_act = (lv_obj_t *)tile;
193 dir = tile->dir;
194 lv_obj_send_event(obj, LV_EVENT_VALUE_CHANGED, NULL);
195 break;
196 }
197 }
198 lv_obj_set_scroll_dir(obj, dir);
199 }
200}
