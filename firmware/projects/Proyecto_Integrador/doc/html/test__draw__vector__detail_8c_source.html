57void test_draw_blend_modes_comparison(void)
58{
59 lv_draw_vector_dsc_t * ctx = lv_draw_vector_dsc_create(&layer);
60 lv_vector_path_t * path = lv_vector_path_create(LV_VECTOR_PATH_QUALITY_MEDIUM);
61
62 /* Clear background */
63 lv_area_t rect = {0, 0, 640, 480};
64 lv_draw_vector_dsc_set_fill_color(ctx, lv_color_white());
65 lv_draw_vector_dsc_clear_area(ctx, &rect);
66
67 /* Draw 14 blend mode examples in 4x4 grid within {0,0,640,480} area */
68 const int unit_width = 70;
69 const int unit_height = 70;
70 const int cols = 4;
71 const int rows = 4;
72
73 /* Calculate spacing between units */
74 const int h_space = (640 - cols * unit_width * 3 / 2) / (cols + 1);
75 const int v_space = (480 - rows * unit_height * 3 / 2) / (rows + 1);
76
77 /* Blend modes for each unit */
78 const lv_vector_blend_t blend_modes[] = {
79 LV_VECTOR_BLEND_SRC_OVER,
80 LV_VECTOR_BLEND_SRC_IN,
81 LV_VECTOR_BLEND_DST_OVER,
82 LV_VECTOR_BLEND_DST_IN,
83 LV_VECTOR_BLEND_SCREEN,
84 LV_VECTOR_BLEND_MULTIPLY,
85 LV_VECTOR_BLEND_NONE,
86 LV_VECTOR_BLEND_ADDITIVE,
87 LV_VECTOR_BLEND_SUBTRACTIVE,
88 };
89
90 /* Draw blend mode examples */
91 for(size_t i = 0; i < sizeof(blend_modes) / sizeof(blend_modes[0]); i++) {
92 /* Calculate unit position */
93 int col = i % cols;
94 int row = i / rows;
95 int x = h_space + col * (unit_width * 3 / 2 + h_space);
96 int y = v_space + row * (unit_height * 3 / 2 + v_space);
97
98 /* Draw blue rectangle (120x120 with 1/4 overlap) */
99 lv_vector_path_clear(path);
100 lv_area_t blue_rect = {
101 x + unit_width / 2,
102 y + unit_width / 2,
103 x + unit_width * 3 / 2,
104 y + unit_width * 3 / 2
105 };
106 lv_vector_path_append_rect(path, &blue_rect, 0, 0);
107 lv_draw_vector_dsc_set_fill_color(ctx, lv_color_make(0x00, 0x00, 0xff)); /* Blue */
108 lv_draw_vector_dsc_set_fill_opa(ctx, LV_OPA_COVER);
109 lv_draw_vector_dsc_set_blend_mode(ctx, LV_VECTOR_BLEND_SRC_OVER);
110 lv_draw_vector_dsc_add_path(ctx, path);
111
112 /* Draw green rectangle (full size) */
113 lv_vector_path_clear(path);
114 lv_area_t green_rect = { x, y, x + unit_width, y + unit_height };
115 lv_vector_path_append_rect(path, &green_rect, 0, 0);
116 lv_draw_vector_dsc_set_fill_color(ctx, lv_color_make(0x00, 0xff, 0x00)); /* Green */
117 lv_draw_vector_dsc_set_fill_opa(ctx, LV_OPA_COVER);
118 lv_draw_vector_dsc_set_stroke_opa(ctx, LV_OPA_TRANSP);
119 lv_draw_vector_dsc_set_blend_mode(ctx, LV_VECTOR_BLEND_SRC_OVER);
120 lv_draw_vector_dsc_add_path(ctx, path);
121
122 lv_vector_path_clear(path);
123 lv_vector_path_append_rect(path, &blue_rect, 0, 0);
124 lv_draw_vector_dsc_set_fill_color(ctx, lv_color_make(0x00, 0x00, 0xff)); /* Blue */
125 lv_draw_vector_dsc_set_fill_opa(ctx, LV_OPA_COVER);
126 lv_draw_vector_dsc_set_blend_mode(ctx, blend_modes[i]);
127 lv_draw_vector_dsc_add_path(ctx, path);
128 }
129
130 draw_vector(ctx);
131 draw_snapshot(SNAPSHOT_NAME(blend_modes_comparison));
132
133 /* Cleanup */
134 lv_vector_path_delete(path);
135 lv_draw_vector_dsc_delete(ctx);
136}
