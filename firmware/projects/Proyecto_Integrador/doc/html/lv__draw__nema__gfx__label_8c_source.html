121{
122 if(dsc->opa <= LV_OPA_MIN) return;
123

125
126 lv_area_t clip_area;
127 clip_area = t->clip_area;
128 lv_area_move(&clip_area, -layer->buf_area.x1, -layer->buf_area.y1);
129
130 lv_color_format_t dst_cf = layer->draw_buf->header.cf;

141 _draw_label_iterate_characters(t, dsc, coords);

153static void _draw_nema_gfx_outline(lv_draw_task_t * t, lv_draw_glyph_dsc_t * glyph_draw_dsc)
154{
155
156 lv_area_t blend_area;
157 if(!_lv_area_intersect(&blend_area, glyph_draw_dsc->letter_coords, &t->clip_area))
158 return;
159
160 lv_draw_nema_gfx_unit_t * draw_nema_gfx_unit = (lv_draw_nema_gfx_unit_t *)t->draw_unit;
161
162 lv_nema_gfx_path_t * nema_gfx_path = (lv_nema_gfx_path_t *)glyph_draw_dsc->glyph_data;
163
164 lv_point_t pos = {glyph_draw_dsc->letter_coords->x1, glyph_draw_dsc->letter_coords->y1};
165
166 float scale = FT_F26DOT6_TO_PATH_SCALE(lv_freetype_outline_get_scale(glyph_draw_dsc->g->resolved_font));
167
168 /*Calculate Path Matrix*/
169 nema_matrix3x3_t matrix;
170 nema_mat3x3_load_identity(matrix);
171 nema_mat3x3_scale(matrix, scale, -scale);
172 nema_mat3x3_translate(matrix, pos.x - glyph_draw_dsc->g->ofs_x,
173 pos.y + glyph_draw_dsc->g->box_h + glyph_draw_dsc->g->ofs_y);
174
175 nema_vg_path_clear(nema_gfx_path->path);
176 nema_vg_paint_clear(nema_gfx_path->paint);
177
178 nema_vg_set_fill_rule(NEMA_VG_FILL_EVEN_ODD);
179
180 nema_vg_path_set_shape(nema_gfx_path->path, nema_gfx_path->seg_size, nema_gfx_path->seg, nema_gfx_path->data_size,
181 nema_gfx_path->data);
182
183 nema_vg_paint_set_type(nema_gfx_path->paint, NEMA_VG_PAINT_COLOR);
184
185 lv_color32_t dsc_col32 = lv_color_to_32(glyph_draw_dsc->color, glyph_draw_dsc->opa);
186 uint32_t nema_dsc_color = nema_rgba(dsc_col32.red, dsc_col32.green, dsc_col32.blue, dsc_col32.alpha);
187
188 nema_vg_paint_set_paint_color(nema_gfx_path->paint, nema_dsc_color);
189
190 nema_vg_path_set_matrix(nema_gfx_path->path, matrix);
191 nema_vg_draw_path(nema_gfx_path->path, nema_gfx_path->paint);
192
193 return;
194}

196static void freetype_outline_event_cb(lv_event_t * e)
197{
198 LV_PROFILER_DRAW_BEGIN;
199 lv_event_code_t code = lv_event_get_code(e);
200 lv_freetype_outline_event_param_t * param = lv_event_get_param(e);
201
202 switch(code) {
203 case LV_EVENT_CREATE:
204 param->outline = lv_nema_gfx_path_create();
205 lv_nema_outline_event_alloc(param);
206 break;
207 case LV_EVENT_DELETE:
208 lv_nema_gfx_path_destroy(param->outline);
209 break;
210 case LV_EVENT_INSERT:
211 lv_nema_gfx_outline_push(param);
212 break;
213 default:
214 LV_LOG_WARN("unknown event code: %d", code);
215 break;
216 }
217 LV_PROFILER_DRAW_END;
218}

220static void lv_nema_gfx_outline_push(const lv_freetype_outline_event_param_t * param)
221{
222 LV_PROFILER_DRAW_BEGIN;
223 lv_nema_gfx_path_t * outline = param->outline;
224 LV_ASSERT_NULL(outline);
225
226 lv_freetype_outline_type_t type = param->type;
227 switch(type) {
228 case LV_FREETYPE_OUTLINE_END:
229 lv_nema_gfx_path_end(outline);
230 break;
231 case LV_FREETYPE_OUTLINE_MOVE_TO:
232 lv_nema_gfx_path_move_to(outline, param->to.x, param->to.y);
233 break;
234 case LV_FREETYPE_OUTLINE_LINE_TO:
235 lv_nema_gfx_path_line_to(outline, param->to.x, param->to.y);
236 break;
237 case LV_FREETYPE_OUTLINE_CUBIC_TO:
238 lv_nema_gfx_path_cubic_to(outline, param->control1.x, param->control1.y,
239 param->control2.x, param->control2.y,
240 param->to.x, param->to.y);
241 break;
242 case LV_FREETYPE_OUTLINE_CONIC_TO:
243 lv_nema_gfx_path_quad_to(outline, param->control1.x, param->control1.y,
244 param->to.x, param->to.y);
245 break;
246 default:
247 LV_LOG_ERROR("unknown point type: %d", type);
248 LV_ASSERT(false);
249 break;
250 }
251 LV_PROFILER_DRAW_END;
252}

307static void _draw_nema_gfx_letter(lv_draw_task_t * t, lv_draw_glyph_dsc_t * glyph_draw_dsc,
308 lv_draw_fill_dsc_t * fill_draw_dsc, const lv_area_t * fill_area)
309{
310 if(glyph_draw_dsc) {
311 if(glyph_draw_dsc->format == LV_FONT_GLYPH_FORMAT_NONE) {
312#if LV_USE_FONT_PLACEHOLDER
313 /* Draw a placeholder rectangle*/
314 lv_draw_border_dsc_t border_draw_dsc;
315 lv_draw_border_dsc_init(&border_draw_dsc);
316 border_draw_dsc.opa = glyph_draw_dsc->opa;
317 border_draw_dsc.color = glyph_draw_dsc->color;
318 border_draw_dsc.width = 1;
319 lv_draw_nema_gfx_border(t, &border_draw_dsc, glyph_draw_dsc->bg_coords);
320
321 /*lv_draw_nema_gfx_border alters the GPU blend state; restore it so
322 *subsequent glyphs are blitted correctly.*/
323 lv_color32_t restore_col32 = lv_color_to_32(glyph_draw_dsc->color, glyph_draw_dsc->opa);
324 uint32_t restore_nema_color = nema_rgba(restore_col32.red, restore_col32.green,
325 restore_col32.blue, restore_col32.alpha);
326 _set_color_blend(restore_nema_color, restore_col32.alpha);
327#endif
328 }
329 else if(glyph_draw_dsc->format >= LV_FONT_GLYPH_FORMAT_A1 &&
330 glyph_draw_dsc->format <= LV_FONT_GLYPH_FORMAT_A8) {
331 /*Do not draw transparent things*/
332 if(glyph_draw_dsc->opa <= LV_OPA_MIN)
333 return;
334
335 lv_layer_t * layer = t->target_layer;
336
337 lv_area_t blend_area;
338 if(!lv_area_intersect(&blend_area, glyph_draw_dsc->letter_coords, &t->clip_area))
339 return;
340
341 const lv_draw_buf_t * draw_buf = glyph_draw_dsc->glyph_data;
342 const uint8_t * mask_buf;
343 uint32_t src_cf;
344 lv_area_t mask_area = *glyph_draw_dsc->letter_coords;
345
346 lv_area_t rel_coords;
347 rel_coords = blend_area;
348 lv_area_move(&rel_coords, -layer->buf_area.x1, -layer->buf_area.y1);
349
350 int32_t x, y, w, h;
351
352 /*Read the static font*/
353 if(is_raw_bitmap) {
354 mask_buf = glyph_draw_dsc->glyph_data;
355 src_cf = _bpp_nema_gfx_format(glyph_draw_dsc);
356 x = glyph_draw_dsc->letter_coords->x1 - layer->buf_area.x1;
357 y = glyph_draw_dsc->letter_coords->y1 - layer->buf_area.y1;
358 w = glyph_draw_dsc->g->box_w;
359 h = glyph_draw_dsc->g->box_h;
360 }
361 /*Read the draw buffer*/
362 else {
363 mask_buf = draw_buf->data;
364 src_cf = lv_nemagfx_cf_to_nema(draw_buf->header.cf);
365 mask_area.x2 = mask_area.x1 + lv_draw_buf_width_to_stride(lv_area_get_width(&mask_area), LV_COLOR_FORMAT_A8) - 1;
366 mask_buf += draw_buf->header.stride * (blend_area.y1 - mask_area.y1) + (blend_area.x1 - mask_area.x1);
367 x = rel_coords.x1;
368 y = rel_coords.y1;
369 w = lv_area_get_width(&rel_coords);
370 h = lv_area_get_height(&rel_coords);
371 }
372
373 if(is_raw_bitmap && (glyph_draw_dsc->format <= LV_FONT_GLYPH_FORMAT_A4)) {
374 nema_bind_src_tex((uintptr_t)(mask_buf), w * h, 1, src_cf, glyph_draw_dsc->g->stride, NEMA_FILTER_PS);
375 nema_matrix3x3_t m = {
376 {1, w, -x - (y * w) - (0.5f * w)},
377 {0, 1, 0},
378 {0, 0, 1}
379 };
380 nema_set_matrix(m);
381 nema_raster_rect(x, y, w, h);
382 }
383 else {
384 nema_bind_src_tex((uintptr_t)(mask_buf), lv_area_get_width(&mask_area), lv_area_get_height(&mask_area), src_cf,
385 glyph_draw_dsc->g->stride ? glyph_draw_dsc->g->stride : lv_area_get_width(&mask_area),
386 NEMA_FILTER_BL);
387 nema_blit_rect(x, y, w, h);
388 lv_draw_nema_gfx_unit_t * draw_nema_gfx_unit = (lv_draw_nema_gfx_unit_t *)t->draw_unit;
389 nema_cl_submit(&(draw_nema_gfx_unit->cl));
390 nema_cl_wait(&(draw_nema_gfx_unit->cl));
391 }
392 }
393 else if(glyph_draw_dsc->format == LV_FONT_GLYPH_FORMAT_IMAGE) {
394#if LV_USE_IMGFONT
395 lv_draw_img_dsc_t img_dsc;
396 lv_draw_img_dsc_init(&img_dsc);
397 img_dsc.angle = 0;
398 img_dsc.zoom = LV_ZOOM_NONE;
399 img_dsc.opa = glyph_draw_dsc->opa;
400 img_dsc.src = glyph_draw_dsc->glyph_data;
401 lv_draw_nema_gfx_img(t, &img_dsc, glyph_draw_dsc->letter_coords);
402#endif
403 }
404
405#if LV_USE_FREETYPE && LV_USE_NEMA_VG
406 else if(glyph_draw_dsc->format == LV_FONT_GLYPH_FORMAT_VECTOR) {
407 if(lv_freetype_is_outline_font(glyph_draw_dsc->g->resolved_font)) {
408 _draw_nema_gfx_outline(t, glyph_draw_dsc);
409 }
410 }
411#endif
412
413 }
414
415 if(fill_draw_dsc && fill_area) {
416 lv_draw_nema_gfx_fill(t, fill_draw_dsc, fill_area);
417 }

421static void _draw_label_iterate_characters(lv_draw_task_t * t, const lv_draw_label_dsc_t * dsc,
422 const lv_area_t * coords)
423{
424 const lv_font_t * font = dsc->font;
425 lv_text_attributes_t attributes = {0};
426 attributes.letter_space = dsc->letter_space;
427 attributes.line_space = dsc->line_space;
428 attributes.max_width = LV_COORD_MAX;
429 attributes.text_flags = dsc->flag;
430
431 lv_area_t clipped_area;
432 bool clip_ok = lv_area_intersect(&clipped_area, coords, &t->clip_area);
433 if(!clip_ok) return;
434
435 lv_text_align_t align = dsc->align;
436 lv_base_dir_t base_dir = dsc->bidi_dir;
437
438 lv_bidi_calculate_align_internal(&align, &base_dir, dsc->text);
439
440 if((dsc->flag & LV_TEXT_FLAG_EXPAND) == 0) {
441 /*Normally use the label's width as width*/
442 attributes.max_width = lv_area_get_width(coords);
443 }
444 else {
445 /*If EXPAND is enabled then not limit the text's width to the object's width*/
446 lv_point_t p;
447 lv_text_get_size_attributes(&p, dsc->text, dsc->font, &attributes);
448 attributes.max_width = p.x;
449 }
450
451 int32_t line_height_font = lv_font_get_line_height_internal(font);
452 int32_t line_height = line_height_font + dsc->line_space;
453
454 /*Init variables for the first line*/
455 int32_t line_width = 0;
456 lv_point_t pos;
457 lv_point_set(&pos, coords->x1, coords->y1);
458
459 int32_t x_ofs = 0;
460 int32_t y_ofs = 0;
461 x_ofs = dsc->ofs_x;
462 y_ofs = dsc->ofs_y;
463 pos.y += y_ofs;
464
465 uint32_t line_start = 0;
466 int32_t last_line_start = -1;
467
468 /*Check the hint to use the cached info*/
469 if(dsc->hint && y_ofs == 0 && coords->y1 < 0) {
470 /*If the label changed too much recalculate the hint.*/
471 if(LV_ABS(dsc->hint->coord_y - coords->y1) > LV_LABEL_HINT_UPDATE_TH - 2 * line_height) {
472 dsc->hint->line_start = -1;
473 }
474 last_line_start = dsc->hint->line_start;
475 }
476
477 /*Use the hint if it's valid*/
478 if(dsc->hint && last_line_start >= 0) {
479 line_start = last_line_start;
480 pos.y += dsc->hint->y;
481 }
482
483 uint32_t remaining_len = dsc->text_length;
484
485 uint32_t line_end = line_start + lv_text_get_next_line(&dsc->text[line_start], remaining_len, font, NULL, &attributes);
486
487 /*Go the first visible line*/
488 while(pos.y + line_height_font < t->clip_area.y1) {
489 /*Go to next line*/
490 line_start = line_end;
491 line_end += lv_text_get_next_line(&dsc->text[line_start], remaining_len, font, NULL, &attributes);
492 pos.y += line_height;
493
494 /*Save at the threshold coordinate*/
495 if(dsc->hint && pos.y >= -LV_LABEL_HINT_UPDATE_TH && dsc->hint->line_start < 0) {
496 dsc->hint->line_start = line_start;
497 dsc->hint->y = pos.y - coords->y1;
498 dsc->hint->coord_y = coords->y1;
499 }
500
501 if(dsc->text[line_start] == '\0') return;
502 }
503
504 /*Align to middle*/
505 if(align == LV_TEXT_ALIGN_CENTER) {
506 line_width = lv_text_get_line_width(&dsc->text[line_start], line_end - line_start, font, &attributes);
507
508 pos.x += (lv_area_get_width(coords) - line_width) / 2;
509
510 }
511 /*Align to the right*/
512 else if(align == LV_TEXT_ALIGN_RIGHT) {
513 line_width = lv_text_get_line_width(&dsc->text[line_start], line_end - line_start, font, &attributes);
514 pos.x += lv_area_get_width(coords) - line_width;
515 }
516
517 uint32_t sel_start = dsc->sel_start;
518 uint32_t sel_end = dsc->sel_end;
519 if(sel_start > sel_end) {
520 uint32_t tmp = sel_start;
521 sel_start = sel_end;
522 sel_end = tmp;
523 }
524
525 lv_area_t bg_coords;
526 lv_draw_glyph_dsc_t draw_letter_dsc;
527 lv_draw_glyph_dsc_init(&draw_letter_dsc);
528 draw_letter_dsc.opa = dsc->opa;
529 draw_letter_dsc.bg_coords = &bg_coords;
530 draw_letter_dsc.color = dsc->color;
531 draw_letter_dsc.rotation = dsc->rotation;
532
533 lv_draw_fill_dsc_t fill_dsc;
534 lv_draw_fill_dsc_init(&fill_dsc);
535 fill_dsc.opa = dsc->opa;
536 int32_t underline_width = font->underline_thickness ? font->underline_thickness : 1;
537 int32_t line_start_x;
538 int32_t decor_end_x;
539 uint32_t next_char_offset;
540 uint32_t recolor_command_start_index = 0;
541 int32_t letter_w;
542
543 cmd_state_t recolor_cmd_state = RECOLOR_CMD_STATE_WAIT_FOR_PARAMETER;
544 lv_color_t recolor = lv_color_black(); /* Holds the selected color inside the recolor command */
545 uint8_t is_first_space_after_cmd = 0;
546
547 lv_color32_t dsc_col32 = lv_color_to_32(dsc->color, dsc->opa);
548 uint32_t nema_dsc_color = nema_rgba(dsc_col32.red, dsc_col32.green, dsc_col32.blue, dsc_col32.alpha);
549 lv_color32_t dsc_sel_col32 = lv_color_to_32(dsc->sel_color, dsc->opa);
550 uint32_t nema_dsc_sel_color = nema_rgba(dsc_sel_col32.red, dsc_sel_col32.green, dsc_sel_col32.blue,
551 dsc_sel_col32.alpha);
552 uint32_t blend_color;
553 uint8_t blend_alpha = 255;
554
555 _set_color_blend(nema_dsc_color, dsc_col32.alpha);
556
557 uint8_t cur_state = 2;
558 uint8_t prev_state = 2;
559
560 /*Write out all lines*/
561 while(remaining_len && dsc->text[line_start] != '\0') {
562 pos.x += x_ofs;
563 line_start_x = pos.x;
564 decor_end_x = pos.x - 1;
565
566 /*Write all letter of a line*/
567 next_char_offset = 0;
568#if LV_USE_BIDI
569 char * bidi_txt = lv_malloc(line_end - line_start + 1);
570 LV_ASSERT_MALLOC(bidi_txt);
571 lv_bidi_process_paragraph(dsc->text + line_start, bidi_txt, line_end - line_start, base_dir, NULL, 0);
572#else
573 const char * bidi_txt = dsc->text + line_start;
574#endif
575
576 while(next_char_offset < remaining_len && next_char_offset < line_end - line_start) {
577 uint32_t logical_char_pos = 0;
578
579 /* Check if the text selection is enabled */
580 if(sel_start != LV_DRAW_LABEL_NO_TXT_SEL && sel_end != LV_DRAW_LABEL_NO_TXT_SEL) {
581#if LV_USE_BIDI
582 logical_char_pos = lv_text_encoded_get_char_id(dsc->text, line_start);
583 uint32_t c_idx = lv_text_encoded_get_char_id(bidi_txt, next_char_offset);
584 logical_char_pos += lv_bidi_get_logical_pos(bidi_txt, NULL, line_end - line_start, base_dir, c_idx, NULL);
585#else
586 logical_char_pos = lv_text_encoded_get_char_id(dsc->text, line_start + next_char_offset);
587#endif
588 }
589
590 uint32_t letter;
591 uint32_t letter_next;
592 lv_text_encoded_letter_next_2(bidi_txt, &letter, &letter_next, &next_char_offset);
593
594 /* If recolor is enabled */
595 if((dsc->flag & LV_TEXT_FLAG_RECOLOR) != 0) {
596
597 if(letter == (uint32_t)LV_TXT_COLOR_CMD[0]) {
598 /* Handle the recolor command marker depending of the current recolor state */
599
600 if(recolor_cmd_state == RECOLOR_CMD_STATE_WAIT_FOR_PARAMETER) {
601 recolor_command_start_index = next_char_offset;
602 recolor_cmd_state = RECOLOR_CMD_STATE_PARAMETER;
603 continue;
604 }
605 /*Other start char in parameter escaped cmd. char*/
606 else if(recolor_cmd_state == RECOLOR_CMD_STATE_PARAMETER) {
607 recolor_cmd_state = RECOLOR_CMD_STATE_WAIT_FOR_PARAMETER;
608 }
609 /* If letter is LV_TXT_COLOR_CMD and we were in the CMD_STATE_IN then the recolor close marked has been found */
610 else if(recolor_cmd_state == RECOLOR_CMD_STATE_TEXT_INPUT) {
611 recolor_cmd_state = RECOLOR_CMD_STATE_WAIT_FOR_PARAMETER;
612 continue;
613 }
614 }
615
616 /* Find the first space (aka ' ') after the recolor command parameter, we need to skip rendering it */
617 if((recolor_cmd_state == RECOLOR_CMD_STATE_PARAMETER) && (letter == ' ') && (is_first_space_after_cmd == 0)) {
618 is_first_space_after_cmd = 1;
619 }
620 else {
621 is_first_space_after_cmd = 0;
622 }
623
624 /* Skip the color parameter and wait the space after it
625 * Once we have reach the space ' ', then we will extract the color information
626 * and store it into the recolor variable */
627 if(recolor_cmd_state == RECOLOR_CMD_STATE_PARAMETER) {
628 /* Not an space? Continue with the next character */
629 if(letter != ' ') {
630 continue;
631 }
632
633 /*Get the recolor parameter*/
634 if((next_char_offset - recolor_command_start_index) == LABEL_RECOLOR_PAR_LENGTH + 1) {
635 /* Temporary buffer to hold the recolor information */
636 char buf[LABEL_RECOLOR_PAR_LENGTH + 1];
637 lv_memcpy(buf, &bidi_txt[recolor_command_start_index], LABEL_RECOLOR_PAR_LENGTH);
638 buf[LABEL_RECOLOR_PAR_LENGTH] = '\0';
639
640 uint8_t r, g, b;
641 r = (hex_char_to_num(buf[0]) << 4) + hex_char_to_num(buf[1]);
642 g = (hex_char_to_num(buf[2]) << 4) + hex_char_to_num(buf[3]);
643 b = (hex_char_to_num(buf[4]) << 4) + hex_char_to_num(buf[5]);
644
645 recolor = lv_color_make(r, g, b);
646 }
647 else {
648 recolor.red = dsc->color.red;
649 recolor.blue = dsc->color.blue;
650 recolor.green = dsc->color.green;
651 }
652
653 /*After the parameter the text is in the command*/
654 recolor_cmd_state = RECOLOR_CMD_STATE_TEXT_INPUT;
655 }
656
657 /* Don't draw the first space after the recolor command */
658 if(is_first_space_after_cmd) {
659 continue;
660 }
661 }
662
663 /* If we're in the CMD_STATE_IN state then we need to subtract the recolor command length */
664 if(((dsc->flag & LV_TEXT_FLAG_RECOLOR) != 0) && (recolor_cmd_state == RECOLOR_CMD_STATE_TEXT_INPUT)) {
665 logical_char_pos -= (LABEL_RECOLOR_PAR_LENGTH + 1);
666 }
667
668 letter_w = lv_font_get_glyph_width(font, letter, letter_next);
669
670 if(!lv_text_is_hanging_space(letter)) {
671 decor_end_x = pos.x + letter_w - 1;
672 }
673
674 /*Always set the bg_coordinates for placeholder drawing*/
675 bg_coords.x1 = pos.x;
676 bg_coords.y1 = pos.y;
677 bg_coords.x2 = pos.x + letter_w - 1;
678 bg_coords.y2 = pos.y + line_height - 1;
679
680 if(next_char_offset >= line_end - line_start && decor_end_x >= line_start_x) {
681 if(dsc->decor & LV_TEXT_DECOR_UNDERLINE) {
682 lv_area_t fill_area;
683 fill_area.x1 = line_start_x;
684 fill_area.x2 = decor_end_x;
685 fill_area.y1 = pos.y + font->line_height - font->base_line - font->underline_position;
686 fill_area.y2 = fill_area.y1 + underline_width - 1;
687
688 fill_dsc.color = dsc->color;
689 lv_draw_nema_gfx_fill(t, &fill_dsc, &fill_area);
690 }
691 if(dsc->decor & LV_TEXT_DECOR_STRIKETHROUGH) {
692 lv_area_t fill_area;
693 fill_area.x1 = line_start_x;
694 fill_area.x2 = decor_end_x;
695 fill_area.y1 = pos.y + (font->line_height - font->base_line) * 2 / 3 + font->underline_thickness / 2;
696 fill_area.y2 = fill_area.y1 + underline_width - 1;
697
698 fill_dsc.color = dsc->color;
699 lv_draw_nema_gfx_fill(t, &fill_dsc, &fill_area);
700 }
701 }
702
703 /* Handle text selection */
704 if(sel_start != LV_DRAW_LABEL_NO_TXT_SEL && sel_end != LV_DRAW_LABEL_NO_TXT_SEL
705 && logical_char_pos >= sel_start && logical_char_pos < sel_end) {
706 draw_letter_dsc.color = dsc->sel_color;
707 fill_dsc.color = dsc->sel_bg_color;
708 lv_draw_nema_gfx_fill(t, &fill_dsc, &bg_coords);
709 cur_state = 0 ;
710 blend_alpha = dsc_sel_col32.alpha;
711 blend_color = nema_dsc_sel_color;
712 }
713 else if(recolor_cmd_state == RECOLOR_CMD_STATE_TEXT_INPUT) {
714 draw_letter_dsc.color = recolor;
715 cur_state = 1 ;
716 blend_alpha = dsc_col32.alpha;
717 lv_color32_t dsc_recolor_col32 = lv_color_to_32(recolor, dsc->opa);
718 blend_color = nema_rgba(dsc_recolor_col32.red, dsc_recolor_col32.green, dsc_recolor_col32.blue,
719 dsc_recolor_col32.alpha);
720 }
721 else {
722 draw_letter_dsc.color = dsc->color;
723 cur_state = 2;
724 blend_alpha = dsc_col32.alpha;
725 blend_color = nema_dsc_color;
726 }
727
728 if(cur_state != prev_state) {
729 _set_color_blend(blend_color, blend_alpha);
730 prev_state = cur_state;
731 }
732
733 _draw_letter(t, &draw_letter_dsc, &pos, font, letter);
734
735 if(letter_w > 0) {
736 pos.x += letter_w + dsc->letter_space;
737 }
738 }
739
740#if LV_USE_BIDI
741 lv_free(bidi_txt);
742 bidi_txt = NULL;
743#endif
744 /*Go to next line*/
745 remaining_len -= line_end - line_start;
746 line_start = line_end;
747 if(remaining_len) {
748 line_end += lv_text_get_next_line(&dsc->text[line_start], remaining_len, font, NULL, &attributes);
749 }
750
751 pos.x = coords->x1;
752 /*Align to middle*/
753 if(align == LV_TEXT_ALIGN_CENTER) {
754 line_width =
755 lv_text_get_line_width(&dsc->text[line_start], line_end - line_start, font, &attributes);
756
757 pos.x += (lv_area_get_width(coords) - line_width) / 2;
758 }
759 /*Align to the right*/
760 else if(align == LV_TEXT_ALIGN_RIGHT) {
761 line_width =
762 lv_text_get_line_width(&dsc->text[line_start], line_end - line_start, font, &attributes);
763 pos.x += lv_area_get_width(coords) - line_width;
764 }
765
766 /*Go the next line position*/
767 pos.y += line_height;
768
769 if(pos.y > t->clip_area.y2) break;
770 }
771
772 if(draw_letter_dsc._draw_buf) lv_draw_buf_destroy(draw_letter_dsc._draw_buf);
773
774 LV_ASSERT_MEM_INTEGRITY();
775}

777static void _draw_letter(lv_draw_task_t * t, lv_draw_glyph_dsc_t * dsc, const lv_point_t * pos,
778 const lv_font_t * font, uint32_t letter)
779{
780 lv_font_glyph_dsc_t g;
781
782 if(lv_text_is_marker(letter)) /*Markers are valid letters but should not be rendered.*/
783 return;
784
785 LV_PROFILER_DRAW_BEGIN;
786 bool g_ret = lv_font_get_glyph_dsc_internal(font, &g, letter, '\0');
787 if(g_ret == false) {
788 /*Add warning if the dsc is not found*/
789 LV_LOG_WARN("lv_draw_letter: glyph dsc. not found for U+%" LV_PRIX32, letter);
790 }
791
792 /*Don't draw anything if the character is empty. E.g. space*/
793 if((g.box_h == 0) || (g.box_w == 0)) {
794 LV_PROFILER_DRAW_END;
795 return;
796 }
797
798 lv_area_t letter_coords;
799 letter_coords.x1 = pos->x + g.ofs_x;
800 letter_coords.x2 = letter_coords.x1 + g.box_w - 1;
801 letter_coords.y1 = pos->y + (font->line_height - font->base_line) - g.box_h - g.ofs_y;
802 letter_coords.y2 = letter_coords.y1 + g.box_h - 1;
803 lv_area_move(&letter_coords, -dsc->pivot.x, -dsc->pivot.y);
804
805 /*If the letter is completely out of mask don't draw it*/
806 if(lv_area_is_out(&letter_coords, &t->clip_area, 0) &&
807 dsc->bg_coords &&
808 lv_area_is_out(dsc->bg_coords, &t->clip_area, 0)) {
809 LV_PROFILER_DRAW_END;
810 return;
811 }
812
813 if(g.resolved_font) {
814 lv_draw_buf_t * draw_buf = NULL;
815 if(LV_FONT_GLYPH_FORMAT_NONE < g.format && g.format < LV_FONT_GLYPH_FORMAT_IMAGE) {
816 /*Only check draw buf for bitmap glyph*/
817 draw_buf = lv_draw_buf_reshape(dsc->_draw_buf, 0, g.box_w, g.box_h, LV_STRIDE_AUTO);
818 if(draw_buf == NULL) {
819 if(dsc->_draw_buf) lv_draw_buf_destroy(dsc->_draw_buf);
820
821 uint32_t h = LV_ROUND_UP(g.box_h, 32); /*Assume a larger size to avoid many reallocations*/
822 draw_buf = lv_draw_buf_create_ex(font_draw_buf_handlers, g.box_w, h, LV_COLOR_FORMAT_A8, LV_STRIDE_AUTO);
823 LV_ASSERT_MALLOC(draw_buf);
824 draw_buf->header.h = g.box_h;
825 dsc->_draw_buf = draw_buf;
826 }
827 }
828
829 /* Performance Optimization for lv_font_fmt_txt_dsc_t fonts, always request raw bitmaps */
830 /*Exception for w*h >= NEMA_COORD_LIMIT due to HW limitation on data handling*/
831 is_raw_bitmap = false;
832 if(g.box_h * g.box_w <= NEMA_COORD_LIMIT) {
833 g.req_raw_bitmap = 1;
834 if(font->get_glyph_bitmap == lv_font_get_bitmap_fmt_txt) {
835 lv_font_fmt_txt_dsc_t * fdsc = (lv_font_fmt_txt_dsc_t *)font->dsc;
836 if(fdsc->are_glyphs_dynamic_loaded) {
837 g.req_raw_bitmap = 0;
838 }
839 else if(fdsc->bitmap_format == LV_FONT_FMT_TXT_PLAIN) {
840 is_raw_bitmap = true;
841 }
842 }
843 }
844
845 dsc->glyph_data = g.resolved_font->get_glyph_bitmap(&g, draw_buf);
846
847 dsc->format = dsc->glyph_data ? g.format : LV_FONT_GLYPH_FORMAT_NONE;
848 }
849 else {
850 dsc->format = LV_FONT_GLYPH_FORMAT_NONE;
851 }
852
853 dsc->letter_coords = &letter_coords;
854 dsc->g = &g;
855 _draw_nema_gfx_letter(t, dsc, NULL, NULL);
856
857 if(g.resolved_font && font->release_glyph) {
858 lv_draw_nema_gfx_unit_t * draw_nema_gfx_unit = (lv_draw_nema_gfx_unit_t *)t->draw_unit;
859 nema_cl_submit(&(draw_nema_gfx_unit->cl));
860 nema_cl_wait(&(draw_nema_gfx_unit->cl));
861 font->release_glyph(font, &g);
862 }
863
864 LV_PROFILER_DRAW_END;
865}
