327void lv_demo_music_album_next(bool next)
328{
329 uint32_t id = track_id;
330 if(next) {
331 id++;
332 if(id >= ACTIVE_TRACK_CNT) id = 0;
333 }
334 else {
335 if(id == 0) {
336 id = ACTIVE_TRACK_CNT - 1;
337 }
338 else {
339 id--;
340 }
341 }
342
343 if(playing) {
344 lv_demo_music_play(id);
345 }
346 else {
347 track_load(id);
348 }
349}

359{
360 playing = true;

363 lv_anim_init(&a);
364 lv_anim_set_values(&a, spectrum_i, spectrum_len - 1);
365 lv_anim_set_exec_cb(&a, spectrum_anim_cb);

368 lv_anim_set_reverse_duration(&a, 0);
369 lv_anim_set_completed_cb(&a, spectrum_end_cb);
370 lv_anim_start(&a);
371
372 if(sec_counter_timer) lv_timer_resume(sec_counter_timer);
373 lv_slider_set_range(slider_obj, 0, lv_demo_music_get_track_length(track_id));
374
375 lv_obj_add_state(play_obj, LV_STATE_CHECKED);
376
377}

764int32_t get_cos(int32_t deg, int32_t a)
765{
766 int32_t r = (lv_trigo_cos(deg) * a);
767
768 r += LV_TRIGO_SIN_MAX / 2;
769 return r >> LV_TRIGO_SHIFT;
770}

772int32_t get_sin(int32_t deg, int32_t a)
773{
774 int32_t r = lv_trigo_sin(deg) * a;
775
776 return (r + LV_TRIGO_SIN_MAX / 2) >> LV_TRIGO_SHIFT;
777
778}

789static void spectrum_draw_event_cb(lv_event_t * e)
790{
791 lv_event_code_t code = lv_event_get_code(e);
792
793 if(code == LV_EVENT_REFR_EXT_DRAW_SIZE) {
794#if LV_DEMO_MUSIC_LANDSCAPE
795 lv_event_set_ext_draw_size(e, music_width);
796#else
797 lv_event_set_ext_draw_size(e, music_height);
798#endif
799 }
800 else if(code == LV_EVENT_COVER_CHECK) {
801 lv_event_set_cover_res(e, LV_COVER_RES_NOT_COVER);
802 }
803 else if(code == LV_EVENT_DRAW_MAIN_BEGIN) {
804 lv_obj_t * obj = lv_event_get_target(e);
805 lv_layer_t * layer = lv_event_get_layer(e);
806
807 lv_opa_t opa = lv_obj_get_style_opa_recursive(obj, LV_PART_MAIN);
808 if(opa <= LV_OPA_MIN) return;
809
810 lv_point_t center;
811 lv_area_t obj_coords;
812 lv_obj_get_coords(obj, &obj_coords);
813 center.x = obj_coords.x1 + lv_obj_get_width(obj) / 2;
814 center.y = obj_coords.y1 + lv_obj_get_height(obj) / 2;
815
816 lv_draw_triangle_dsc_t draw_dsc;
817 lv_draw_triangle_dsc_init(&draw_dsc);
818 draw_dsc.opa = LV_OPA_COVER;
819
820 uint16_t r[64];
821 uint32_t i;
822
823 for(i = 0; i < BAR_CNT; i++) {
824 r[i] = BAR_REST_RADIUS;
825 }
826
827 uint32_t s;
828 for(s = 0; s < 4; s++) {
829 uint32_t f;
830 uint32_t band_w = 0; /*Real number of bars in this band.*/
831 switch(s) {
832 case 0:
833 band_w = 20;
834 break;
835 case 1:
836 band_w = 8;
837 break;
838 case 2:
839 band_w = 4;
840 break;
841 case 3:
842 band_w = 2;
843 break;
844 }
845
846 /* Add "side bars" with cosine characteristic.*/
847 for(f = 0; f < band_w; f++) {
848 uint32_t ampl_main = spectrum[spectrum_i][s];
849 int32_t ampl_mod = get_cos(f * 360 / band_w + 180, 180) + 180;
850 int32_t t = BAR_PER_BAND_CNT * s - band_w / 2 + f;
851 if(t < 0) t = BAR_CNT + t;
852 if(t >= BAR_CNT) t = t - BAR_CNT;
853 r[t] += (ampl_main * ampl_mod) >> 9;
854 }
855 }
856
857 const int32_t amax = 20;
858 int32_t animv = spectrum_i - 0;
859 if(animv > amax) animv = amax;
860 for(i = 0; i < BAR_CNT; i++) {
861 uint32_t deg_space = 1;
862 uint32_t deg = i * DEG_STEP + 90;
863 uint32_t j = (i + bar_rot + rnd_array[bar_ofs % 10]) % BAR_CNT;
864 uint32_t k = (i + bar_rot + rnd_array[(bar_ofs + 1) % 10]) % BAR_CNT;
865
866 uint32_t v;
867 if(start_anim) {
868 v = BAR_REST_RADIUS + start_anim_values[i];
869 }
870 else {
871 v = (r[k] * animv + r[j] * (amax - animv)) / amax;
872 }
873
874 if(v < BAR_COLOR1_STOP) draw_dsc.color = BAR_COLOR1;
875 else if(v > (uint32_t)bar_color3_stop) draw_dsc.color = BAR_COLOR3;
876 else if(v > BAR_COLOR2_STOP) draw_dsc.color = lv_color_mix(BAR_COLOR3, BAR_COLOR2,
877 ((v - BAR_COLOR2_STOP) * 255) / (bar_color3_stop - BAR_COLOR2_STOP));
878 else draw_dsc.color = lv_color_mix(BAR_COLOR2, BAR_COLOR1,
879 ((v - BAR_COLOR1_STOP) * 255) / (BAR_COLOR2_STOP - BAR_COLOR1_STOP));
880
881 uint32_t di = deg + deg_space;
882
883 int32_t x1_out = get_cos(di, v);
884 draw_dsc.p[0].x = center.x + x1_out;
885 draw_dsc.p[0].y = center.y + get_sin(di, v);
886
887 di += DEG_STEP - deg_space * 2;
888
889 int32_t x2_out = get_cos(di, v);
890 draw_dsc.p[1].x = center.x + x2_out;
891 draw_dsc.p[1].y = center.y + get_sin(di, v);
892
893 int32_t x2_in = get_cos(di, 0);
894 draw_dsc.p[2].x = center.x + x2_in;
895 draw_dsc.p[2].y = center.y + get_sin(di, 0);
896 lv_draw_triangle(layer, &draw_dsc);
897
898 draw_dsc.p[0].x = center.x - x1_out;
899 draw_dsc.p[1].x = center.x - x2_out;
900 draw_dsc.p[2].x = center.x - x2_in;
901 lv_draw_triangle(layer, &draw_dsc);
902 }
903 }
904 else if(code == LV_EVENT_DELETE) {
905 lv_anim_delete(NULL, start_anim_cb);
906 lv_anim_delete(NULL, spectrum_anim_cb);
907 if(start_anim && stop_start_anim_timer) lv_timer_delete(stop_start_anim_timer);
908 }
909}
