44lv_display_t * lv_lovyan_gfx_create(uint32_t hor_res, uint32_t ver_res, void * buf, uint32_t buf_size_bytes, bool touch)
45{
46 LV_CHECK_ARG(buf != NULL, return NULL);
47 lv_lovyan_gfx_t * dsc = (lv_lovyan_gfx_t *)lv_malloc_zeroed(sizeof(lv_lovyan_gfx_t));
48 LV_ASSERT_MALLOC(dsc);
49 if(dsc == NULL) return NULL;
50
51 lv_display_t * disp = lv_display_create(hor_res, ver_res);
52 if(disp == NULL) {
53 lv_free(dsc);
54 return NULL;
55 }
56
57 dsc->tft = new LGFX();
58 dsc->tft->init(); /* TFT init */
59 dsc->tft->initDMA();
60 dsc->tft->setRotation(0);
61 dsc->tft->setBrightness(255);
62 dsc->tft->startWrite();
63 dsc->tft->fillScreen(0x00000);
64
65 lv_display_set_driver_data(disp, (void *)dsc);
66 lv_display_set_flush_cb(disp, flush_cb);
67 lv_display_set_color_format(disp, LV_COLOR_FORMAT_RGB565_SWAPPED);
68 lv_display_add_event_cb(disp, resolution_changed_event_cb, LV_EVENT_RESOLUTION_CHANGED, NULL);
69 lv_display_set_buffers(disp, (void *)buf, NULL, buf_size_bytes, LV_DISPLAY_RENDER_MODE_PARTIAL);
70
71 if(touch) {
72 /* Register an input device when touch is enabled */
73 lv_indev_t * lv_input = lv_indev_create();
74 lv_indev_set_driver_data(lv_input, (void *)dsc);
75 lv_indev_set_type(lv_input, LV_INDEV_TYPE_POINTER);
76 lv_indev_set_read_cb(lv_input, read_touch);
77 }
78
79 return disp;
80}

86static void flush_cb(lv_display_t * disp, const lv_area_t * area, uint8_t * px_map)
87{
88 lv_lovyan_gfx_t * dsc = (lv_lovyan_gfx_t *)lv_display_get_driver_data(disp);
89
90 uint32_t w = (area->x2 - area->x1 + 1);
91 uint32_t h = (area->y2 - area->y1 + 1);
92
93
94 if(dsc->tft->getStartCount() == 0) {
95 dsc->tft->startWrite();
96 }
97 dsc->tft->pushImageDMA(area->x1, area->y1, w, h, (uint16_t *)px_map);
98 dsc->tft->waitDMA();
99
100 lv_display_flush_ready(disp);
101
102}

104static void resolution_changed_event_cb(lv_event_t * e)
105{
106 lv_display_t * disp = (lv_display_t *)lv_event_get_target(e);

129static void read_touch(lv_indev_t * indev, lv_indev_data_t * data)
130{
131 lv_lovyan_gfx_t * dsc = (lv_lovyan_gfx_t *)lv_indev_get_driver_data(indev);
132 uint16_t x, y;
133 bool touched = dsc->tft->getTouch(&x, &y);
134
135 if(!touched) {
136 data->state = LV_INDEV_STATE_RELEASED;
137 }
138 else {
139 data->state = LV_INDEV_STATE_PRESSED;
140
141 /* LovyanGFX returns coordinates relative to the current software rotation.
142 * LVGL will also rotate the input coordinates based on the display rotation.
143 * To avoid "double rotation", we must reverse the LovyanGFX rotation
144 * and pass raw (Physical/Rotation 0) coordinates to LVGL.
145 */
146 uint8_t rotation = dsc->tft->getRotation();
147
148 /* Note: width() and height() return dimensions for the *current* rotation */
149 int32_t w = dsc->tft->width();
150 int32_t h = dsc->tft->height();
151
152 switch(rotation) {
153 case 1: /* Landscape (90 deg CW) */
154 /* Rot 0 (Phys) Top-Left becomes Rot 1 Top-Right.
155 * Phys X = Rot1 Y
156 * Phys Y = Rot1 Width - 1 - Rot1 X */
157 data->point.x = y;
158 data->point.y = w - 1 - x;
159 break;
160
161 case 2: /* Portrait Inverted (180 deg) */
162 /* Phys X = Rot2 Width - 1 - Rot2 X
163 * Phys Y = Rot2 Height - 1 - Rot2 Y */
164 data->point.x = w - 1 - x;
165 data->point.y = h - 1 - y;
166 break;
167
168 case 3: /* Landscape Inverted (270 deg CW) */
169 /* Phys X = Rot3 Height - 1 - Rot3 Y
170 * Phys Y = Rot3 X */
171 data->point.x = h - 1 - y;
172 data->point.y = x;
173 break;
174
175 default: /* Portrait (0 deg) */
176 /* Pass through raw coordinates */
177 data->point.x = x;
178 data->point.y = y;
179 break;
180 }
181 }
182}
