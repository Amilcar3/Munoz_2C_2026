186static inline lv_value_precise_t resolve_point_coord(lv_value_precise_t coord, int32_t max)
187{
188 if(LV_COORD_IS_PCT((int32_t)coord)) {
189 return LV_CLAMP(0, max * LV_COORD_GET_PCT((int32_t)coord) / 100, max);
190 }

196static void lv_line_event(const lv_obj_class_t * class_p, lv_event_t * e)
197{
198 LV_UNUSED(class_p);
199
200 lv_result_t res;
201
202 /*Call the ancestor's event handler*/
203 res = lv_obj_event_base(MY_CLASS, e);
204 if(res != LV_RESULT_OK) return;
205
206 lv_event_code_t code = lv_event_get_code(e);
207 lv_obj_t * obj = lv_event_get_current_target(e);
208
209 if(code == LV_EVENT_REFR_EXT_DRAW_SIZE) {
210 /*The corner of the skew lines is out of the intended area*/
211 int32_t line_width = lv_obj_get_style_line_width_internal(obj, LV_PART_MAIN);
212 int32_t * s = lv_event_get_param(e);
213 if(*s < line_width) *s = line_width;
214 }
215 else if(code == LV_EVENT_GET_SELF_SIZE) {
216 lv_line_t * line = (lv_line_t *)obj;
217
218 if(line->point_num == 0 || line->point_array.constant == NULL) return;
219
220 lv_point_t * p = lv_event_get_param(e);
221 int32_t w = 0;
222 int32_t h = 0;
223
224 uint32_t i;
225 for(i = 0; i < line->point_num; i++) {
226 if(!LV_COORD_IS_PCT((int32_t)line->point_array.constant[i].x)) {
227 w = (int32_t)LV_MAX(line->point_array.constant[i].x, w);
228 }
229
230 if(!LV_COORD_IS_PCT((int32_t)line->point_array.constant[i].y)) {
231 h = (int32_t)LV_MAX(line->point_array.constant[i].y, h);
232 }
233 }
234
235 p->x = w;
236 p->y = h;
237 }
238 else if(code == LV_EVENT_DRAW_MAIN) {
239 lv_line_t * line = (lv_line_t *)obj;
240 lv_layer_t * layer = lv_event_get_layer(e);
241
242 if(line->point_num < 2 || line->point_array.constant == NULL) return;
243
244 lv_area_t area;
245 lv_obj_get_coords(obj, &area);
246 int32_t x_ofs = area.x1 - lv_obj_get_scroll_x(obj);
247 int32_t y_ofs = area.y1 - lv_obj_get_scroll_y(obj);
248
249 lv_draw_line_dsc_t line_dsc;
250 lv_draw_line_dsc_init(&line_dsc);
251 line_dsc.base.layer = layer;
252 lv_obj_init_draw_line_dsc(obj, LV_PART_MAIN, &line_dsc);
253
254 /*Read all points and draw the lines*/
255 uint32_t i;
256 for(i = 0; i < line->point_num - 1; i++) {
257 int32_t w = lv_obj_get_width(obj);
258 int32_t h = lv_obj_get_height(obj);
259
260 line_dsc.p1.x = resolve_point_coord(line->point_array.constant[i].x, w) + x_ofs;
261 line_dsc.p1.y = resolve_point_coord(line->point_array.constant[i].y, h);
262
263 line_dsc.p2.x = resolve_point_coord(line->point_array.constant[i + 1].x, w) + x_ofs;
264 line_dsc.p2.y = resolve_point_coord(line->point_array.constant[i + 1].y, h);
265
266 if(line->y_inv == 0) {
267 line_dsc.p1.y = line_dsc.p1.y + y_ofs;
268 line_dsc.p2.y = line_dsc.p2.y + y_ofs;
269 }
270 else {
271 line_dsc.p1.y = h - line_dsc.p1.y + y_ofs;
272 line_dsc.p2.y = h - line_dsc.p2.y + y_ofs;
273 }
274
275 lv_draw_line(layer, &line_dsc);
276 line_dsc.round_start = 0; /*Draw the rounding only on the end points after the first line*/
277 }
278 }
