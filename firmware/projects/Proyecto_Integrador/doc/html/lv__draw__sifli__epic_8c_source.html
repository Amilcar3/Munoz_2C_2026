99/* Vector glyphs are not handled by the EPIC label path below.
100 * Keep the check lightweight and fall back only when the first glyph is vector. */
101static bool _epic_label_has_vector_glyph(const lv_draw_label_dsc_t * draw_dsc)
102{
103#if LV_USE_FREETYPE && LV_USE_VECTOR_GRAPHIC && LV_USE_THORVG
104 if(draw_dsc == NULL || draw_dsc->font == NULL || draw_dsc->text == NULL || draw_dsc->text[0] == '\0') {
105 return false;
106 }
107
108 lv_font_glyph_dsc_t glyph_dsc;
109 uint32_t txt_ofs = 0;
110 uint32_t letter = lv_text_encoded_next(draw_dsc->text, &txt_ofs);
111
112 if(letter == '\0') {
113 return false;
114 }
115
116 if(lv_font_get_glyph_dsc(draw_dsc->font, &glyph_dsc, letter, '\0') &&
117 glyph_dsc.format == LV_FONT_GLYPH_FORMAT_VECTOR) {
118 return true;
119 }
120#else
121 LV_UNUSED(draw_dsc);
122#endif
123
124 return false;
125}

131void lv_draw_sifli_epic_init(void)
132{
133 if(epic_draw_unit_initialized) {
134 return;
135 }
136
137 /* Initialize EPIC hardware */
138 lv_epic_init();
139 if(!lv_epic_is_initialized()) {
140 return;
141 }
142
143 /* Initialize draw buffer handlers */
144 lv_draw_buf_sifli_epic_init_handlers();
145
146 /* Create EPIC draw unit */
147 lv_draw_sifli_epic_unit_t * draw_epic_unit = lv_draw_create_unit(sizeof(lv_draw_sifli_epic_unit_t));
148 draw_epic_unit->base_unit.evaluate_cb = _epic_evaluate;
149 draw_epic_unit->base_unit.dispatch_cb = _epic_dispatch;
150 draw_epic_unit->base_unit.delete_cb = _epic_delete;
151 draw_epic_unit->base_unit.name = "SiFli_EPIC";
152#if !LV_USE_SIFLI_EPIC_DRAW_THREAD
153 draw_epic_unit->base_unit.wait_for_finish_cb = _epic_wait_for_finish;
154#endif
155
156#if LV_USE_SIFLI_EPIC_DRAW_THREAD
157 lv_thread_init(&draw_epic_unit->thread, "EPIC", LV_DRAW_THREAD_PRIO,
158 _epic_render_thread_cb, LV_DRAW_THREAD_STACK_SIZE, draw_epic_unit);
159#endif
160
161 epic_draw_unit_initialized = true;

178static int32_t _epic_evaluate(lv_draw_unit_t * draw_unit, lv_draw_task_t * task)
179{
180 LV_UNUSED(draw_unit);
181
182 if(task->target_layer == NULL) {
183 return 0;
184 }
185
186 if(!lv_epic_cf_supported(task->target_layer->color_format, 0)) {
187 return 0;
188 }
189
190 switch(task->type) {
191 case LV_DRAW_TASK_TYPE_FILL: {
192 const lv_draw_fill_dsc_t * draw_dsc = (const lv_draw_fill_dsc_t *)task->draw_dsc;
193
194 if(task->target_layer == NULL) {
195 return 0;
196 }
197
198 /* EPIC doesn't support rounded corners */
199 if(draw_dsc->radius != 0) {
200 return 0;
201 }
202
203 /* Check gradient support */
204 if(draw_dsc->grad.dir != LV_GRAD_DIR_NONE) {
205 if(draw_dsc->grad.dir != LV_GRAD_DIR_HOR && draw_dsc->grad.dir != LV_GRAD_DIR_VER) {
206 return 0;
207 }
208
209 /* Only support 2-stop gradients */
210 if(draw_dsc->grad.stops_count != 2) {
211 return 0;
212 }
213
214 /* Check if output format supports gradient */
215 lv_layer_t * target_layer = task->target_layer;
216 if(target_layer) {
217 lv_color_format_t dest_cf = target_layer->color_format;
218 uint32_t epic_cf = lv_img_cf_to_epic_cf(dest_cf);
219 if(!EPIC_SUPPROT_OUT_FORMAT(epic_cf)) {
220 return 0;
221 }
222 }
223 }
224
225 if(task->preference_score > 70) {
226 task->preference_score = 70;
227 task->preferred_draw_unit_id = DRAW_UNIT_ID_SIFLI_EPIC;
228 }
229 return 1;
230 }
231
232 case LV_DRAW_TASK_TYPE_IMAGE: {
233 const lv_draw_image_dsc_t * draw_dsc = (const lv_draw_image_dsc_t *)task->draw_dsc;
234
235 if(draw_dsc->src == NULL) {
236 return 0;
237 }
238
239 if(draw_dsc->clip_radius != 0) {
240 return 0;
241 }
242
243 if(draw_dsc->blend_mode != LV_BLEND_MODE_NORMAL) {
244 return 0;
245 }
246
247 /* Check if recolor is used */
248 bool has_recolor = (draw_dsc->recolor_opa != LV_OPA_TRANSP);
249 if(has_recolor) {
250 return 0;
251 }
252
253 /* Check if transform is used */
254 bool has_transform = (draw_dsc->rotation != 0 ||
255 draw_dsc->scale_x != LV_SCALE_NONE ||
256 draw_dsc->scale_y != LV_SCALE_NONE);
257
258 /* Check color format support */
259 if(!lv_epic_cf_supported(draw_dsc->header.cf, draw_dsc->header.flags)) {
260 return 0;
261 }
262
263 /* RGB565A8 with transform is not supported */
264 if((LV_COLOR_FORMAT_RGB565A8 == draw_dsc->header.cf) && has_transform) {
265 return 0;
266 }
267
268 if(task->preference_score > 80) {
269 task->preference_score = 80;
270 task->preferred_draw_unit_id = DRAW_UNIT_ID_SIFLI_EPIC;
271 }
272 return 1;
273 }
274
275 case LV_DRAW_TASK_TYPE_BORDER: {
276 const lv_draw_border_dsc_t * draw_dsc = (const lv_draw_border_dsc_t *)task->draw_dsc;
277
278 if(draw_dsc->radius != 0) {
279 return 0;
280 }
281
282 if(task->preference_score > 90) {
283 task->preference_score = 90;
284 task->preferred_draw_unit_id = DRAW_UNIT_ID_SIFLI_EPIC;
285 }
286 return 1;
287 }
288
289#ifdef EPIC_SUPPORT_A8
290 case LV_DRAW_TASK_TYPE_LABEL: {
291 const lv_draw_label_dsc_t * draw_dsc = (const lv_draw_label_dsc_t *)task->draw_dsc;
292 lv_layer_t * target_layer = task->target_layer;
293
294 if(target_layer == NULL) {
295 return 0;
296 }
297
298 if(draw_dsc->font == NULL || draw_dsc->text == NULL || draw_dsc->text[0] == '\0') {
299 return 0;
300 }
301
302 if(_epic_label_has_vector_glyph(draw_dsc)) {
303 return 0;
304 }
305
306 if(!lv_epic_cf_supported(target_layer->color_format, 0)) {
307 return 0;
308 }
309
310 if(task->preference_score > 95) {
311 task->preference_score = 95;
312 task->preferred_draw_unit_id = DRAW_UNIT_ID_SIFLI_EPIC;
313 }
314 return 1;
315 }
316#endif
317
318 case LV_DRAW_TASK_TYPE_LAYER: {
319 const lv_draw_image_dsc_t * draw_dsc = (const lv_draw_image_dsc_t *)task->draw_dsc;
320 lv_layer_t * layer_to_draw = (lv_layer_t *)draw_dsc->src;
321
322 if(layer_to_draw == NULL) {
323 return 0;
324 }
325
326 if(layer_to_draw->draw_buf == NULL) {
327 return 0;
328 }
329
330 /* Check if recolor is used */
331 bool has_recolor = (draw_dsc->recolor_opa != LV_OPA_TRANSP);
332 if(has_recolor) {
333 return 0;
334 }
335
336 /* Check if transform is used */
337 bool has_transform = (draw_dsc->rotation != 0 ||
338 draw_dsc->scale_x != LV_SCALE_NONE ||
339 draw_dsc->scale_y != LV_SCALE_NONE);
340
341 /* Check color format support */
342 if(!lv_epic_cf_supported(layer_to_draw->color_format, 0)) {
343 return 0;
344 }
345
346 /* RGB565A8 with transform is not supported */
347 if((LV_COLOR_FORMAT_RGB565A8 == layer_to_draw->color_format) && has_transform) {
348 return 0;
349 }
350
351 if(task->preference_score > 80) {
352 task->preference_score = 80;
353 task->preferred_draw_unit_id = DRAW_UNIT_ID_SIFLI_EPIC;
354 }
355 return 1;
356 }
357
358 default:
359 return 0;
360 }
361
362 return 0;
363}

365static int32_t _epic_dispatch(lv_draw_unit_t * draw_unit, lv_layer_t * layer)
366{
367 lv_draw_sifli_epic_unit_t * draw_epic_unit = (lv_draw_sifli_epic_unit_t *)draw_unit;
368
369 /* Try to get a ready to draw task */
370 lv_draw_task_t * task = lv_draw_get_next_available_task(layer, NULL, DRAW_UNIT_ID_SIFLI_EPIC);
371
372 if(task == NULL) {
373 return LV_DRAW_UNIT_IDLE;
374 }
375
376 /* Return immediately if EPIC is busy */
377 if(draw_epic_unit->task_act) {
378 return 0;
379 }
380
381 /* Let the SW unit draw this task if not preferred for EPIC */
382 if(task->preferred_draw_unit_id != DRAW_UNIT_ID_SIFLI_EPIC) {
383 return LV_DRAW_UNIT_IDLE;
384 }
385
386 /* Allocate buffer for layer */
387 if(lv_draw_layer_alloc_buf(layer) == NULL) {
388 EPIC_ASSERT_MSG(false, "EPIC: Failed to allocate layer draw buffer");
389 return LV_DRAW_UNIT_IDLE;
390 }
391
392 /* Mark task as in progress */
393 task->state = LV_DRAW_TASK_STATE_IN_PROGRESS;
394 task->draw_unit = &draw_epic_unit->base_unit;
395
396#if LV_USE_SIFLI_EPIC_DRAW_THREAD
397 if(epic_dispatch_log_cnt < 8) {
398 LV_LOG_WARN("EPIC dispatch[%u]: type=%d inited=%d",
399 (unsigned int)epic_dispatch_log_cnt,
400 (int)task->type,
401 draw_epic_unit->inited ? 1 : 0);
402 epic_dispatch_log_cnt++;
403 }
404
405 draw_epic_unit->task_act = task;
406 /* Let the render thread work */
407 if(draw_epic_unit->inited) {
408 lv_thread_sync_signal(&draw_epic_unit->sync);
409 }
410#else
411 draw_epic_unit->task_act = task;
412 /* Execute drawing immediately */
413 _epic_execute_drawing(draw_epic_unit);
414
415 draw_epic_unit->task_act->state = LV_DRAW_TASK_STATE_FINISHED;
416 draw_epic_unit->task_act = NULL;
417
418 /* Request new dispatching */
419 lv_draw_dispatch_request();
420#endif
421
422 return 1;
423}

443static void _epic_execute_drawing(lv_draw_sifli_epic_unit_t * u)
444{
445 lv_draw_task_t * task = u->task_act;
446 lv_layer_t * layer = task->target_layer;
447
448 /* Wait for previous operation to complete */
449 lv_epic_wait();
450
451 if(layer && layer->draw_buf) {
452 lv_area_t draw_area;
453
454 if(lv_area_intersect(&draw_area, &task->_real_area, &task->clip_area)) {
455 lv_area_move(&draw_area, -layer->buf_area.x1, -layer->buf_area.y1);
456 lv_draw_buf_flush_cache(layer->draw_buf, &draw_area);
457 }
458 }
459
460 /* Execute based on task type */
461 switch(task->type) {
462 case LV_DRAW_TASK_TYPE_BORDER:
463 lv_draw_sifli_epic_border(task);
464 break;
465
466 case LV_DRAW_TASK_TYPE_LABEL:
467 lv_draw_sifli_epic_label(task);
468 break;
469
470 case LV_DRAW_TASK_TYPE_FILL:
471 lv_draw_sifli_epic_fill(task);
472 break;
473
474 case LV_DRAW_TASK_TYPE_IMAGE:
475 lv_draw_sifli_epic_img(task);
476 break;
477
478 case LV_DRAW_TASK_TYPE_LAYER:
479 lv_draw_sifli_epic_layer(task);
480 break;
481
482 default:
483 break;
484 }
485
486 /* Wait for operation to complete */
487 lv_epic_wait();
488
489 if(layer && layer->draw_buf) {
490 lv_area_t draw_area;
491
492 if(lv_area_intersect(&draw_area, &task->_real_area, &task->clip_area)) {
493 lv_area_move(&draw_area, -layer->buf_area.x1, -layer->buf_area.y1);
494 lv_draw_buf_invalidate_cache(layer->draw_buf, &draw_area);
495 }
496 }
497}

508#if LV_USE_SIFLI_EPIC_DRAW_THREAD
509static void _epic_render_thread_cb(void * ptr)
510{
511 lv_draw_sifli_epic_unit_t * u = ptr;
512
513 lv_thread_sync_init(&u->sync);
514 u->inited = true;
515
516 if(!epic_thread_start_logged) {
517 LV_LOG_WARN("EPIC render thread started");
518 epic_thread_start_logged = true;
519 }
520
521 while(1) {
522 /* Wait for sync if there is no task set */
523 while(u->task_act == NULL) {
524 if(u->exit_status) {
525 break;
526 }
527 lv_thread_sync_wait(&u->sync);
528 }
529
530 if(u->exit_status) {
531 break;
532 }
533
534 if(epic_render_start_log_cnt < 8) {
535 LV_LOG_WARN("EPIC render start[%u]: type=%d",
536 (unsigned int)epic_render_start_log_cnt,
537 (int)u->task_act->type);
538 epic_render_start_log_cnt++;
539 }
540
541 _epic_execute_drawing(u);
542
543 if(epic_render_done_log_cnt < 8) {
544 LV_LOG_WARN("EPIC render done[%u]: type=%d",
545 (unsigned int)epic_render_done_log_cnt,
546 (int)u->task_act->type);
547 epic_render_done_log_cnt++;
548 }
549
550 /* Signal the ready state to dispatcher */
551 u->task_act->state = LV_DRAW_TASK_STATE_FINISHED;
552
553 /* Cleanup */
554 u->task_act = NULL;
555
556 /* Request new dispatching */
557 lv_draw_dispatch_request();
558 }
559
560 u->inited = false;
561 lv_thread_sync_delete(&u->sync);
562}
