114lv_display_t * lv_linux_fbdev_create(void)
115{
116 lv_tick_set_cb(tick_get_cb);
117
118 lv_linux_fb_t * dsc = lv_malloc_zeroed(sizeof(lv_linux_fb_t));
119 LV_ASSERT_MALLOC(dsc);
120 if(dsc == NULL) return NULL;
121
122 lv_display_t * disp = lv_display_create(800, 480);
123 if(disp == NULL) {
124 lv_free(dsc);
125 return NULL;
126 }
127 dsc->fbfd = -1;
128 /* assume supported by default*/
129 dsc->vsync_supported = true;
130 /* sync on first flush*/
131 dsc->wait_vsync_next = true;
132 lv_display_set_driver_data(disp, dsc);
133 lv_display_set_flush_cb(disp, flush_cb);
134 lv_display_add_event_cb(disp, del_event_cb, LV_EVENT_DELETE, NULL);
135
136 return disp;
137}

139lv_result_t lv_linux_fbdev_set_file(lv_display_t * disp, const char * file)
140{
141 LV_CHECK_ARG(disp != NULL, return LV_RESULT_INVALID);
142 LV_CHECK_ARG(file != NULL, return LV_RESULT_INVALID);
143 char * devname = lv_strdup(file);
144 LV_ASSERT_MALLOC(devname);
145 if(devname == NULL) {
146 return LV_RESULT_INVALID;
147 }
148
149 lv_linux_fb_t * dsc = lv_display_get_driver_data(disp);
150 dsc->devname = devname;
151
152 if(dsc->fbfd > 0) close(dsc->fbfd);
153
154 /* Open the file for reading and writing*/
155 dsc->fbfd = open(dsc->devname, O_RDWR);
156 if(dsc->fbfd == -1) {
157 perror("Error: cannot open framebuffer device");
158 return LV_RESULT_INVALID;
159 }
160 LV_LOG_INFO("The framebuffer device was opened successfully");
161
162 if(!dsc->skip_unblank) {
163 if(ioctl(dsc->fbfd, FBIOBLANK, FB_BLANK_UNBLANK) != 0) {
164 perror("ioctl(FBIOBLANK)");
165 }
166 }
167
168#if LV_LINUX_FBDEV_BSD
169 struct fbtype fb;
170 unsigned line_length;
171
172 /*Get fb type*/
173 if(ioctl(dsc->fbfd, FBIOGTYPE, &fb) != 0) {
174 perror("ioctl(FBIOGTYPE)");
175 return LV_RESULT_INVALID;
176 }
177
178 /*Get screen width*/
179 if(ioctl(dsc->fbfd, FBIO_GETLINEWIDTH, &line_length) != 0) {
180 perror("ioctl(FBIO_GETLINEWIDTH)");
181 return LV_RESULT_INVALID;
182 }
183
184 dsc->vinfo.xres = (unsigned) fb.fb_width;
185 dsc->vinfo.yres = (unsigned) fb.fb_height;
186 dsc->vinfo.bits_per_pixel = fb.fb_depth;
187 dsc->vinfo.xoffset = 0;
188 dsc->vinfo.yoffset = 0;
189 dsc->finfo.line_length = line_length;
190 dsc->finfo.smem_len = dsc->finfo.line_length * dsc->vinfo.yres;
191#else /* LV_LINUX_FBDEV_BSD */
192
193 /* Get fixed screen information*/
194 if(ioctl(dsc->fbfd, FBIOGET_FSCREENINFO, &dsc->finfo) == -1) {
195 perror("Error reading fixed information");
196 return LV_RESULT_INVALID;
197 }
198
199 /* Get variable screen information*/
200 if(ioctl(dsc->fbfd, FBIOGET_VSCREENINFO, &dsc->vinfo) == -1) {
201 perror("Error reading variable information");
202 return LV_RESULT_INVALID;
203 }
204#endif /* LV_LINUX_FBDEV_BSD */
205
206 /* AD5M and some other devices report incorrect bits_per_pixel via VSCREENINFO.
207 * The stride (line_length) is always correct, so calculate true bpp from it.
208 * Example: AD5M reports 16bpp but stride=3200 for 800px width = 4 bytes/pixel = 32bpp */
209 uint32_t stride_bpp = 0;
210 /* Use the virtual (padded) width when available; some framebuffers pad each
211 * line to xres_virtual, so xres alone yields an inflated bpp. The BSD
212 * fb_var_info shim has no xres_virtual field, so fall back to xres there. */
213#if LV_LINUX_FBDEV_BSD
214 uint32_t xres_for_stride = dsc->vinfo.xres;
215#else
216 uint32_t xres_for_stride = dsc->vinfo.xres_virtual > 0 ? dsc->vinfo.xres_virtual : dsc->vinfo.xres;
217#endif
218 if(xres_for_stride > 0) {
219 stride_bpp = (dsc->finfo.line_length * 8) / xres_for_stride;
220 }
221
222 /* Only override if stride_bpp is a standard depth (avoids false positives
223 * from alignment-padded strides or virtual-width framebuffers). */
224 if(stride_bpp > 0 && stride_bpp != dsc->vinfo.bits_per_pixel
225 && (stride_bpp == 8 || stride_bpp == 16 || stride_bpp == 24 || stride_bpp == 32)) {
226 LV_LOG_WARN("bits_per_pixel mismatch: vinfo says %d, stride indicates %d. Using stride value.",
227 dsc->vinfo.bits_per_pixel, stride_bpp);
228 dsc->vinfo.bits_per_pixel = stride_bpp;
229 }
230 else if(stride_bpp > 0 && stride_bpp == dsc->vinfo.bits_per_pixel) {
231 LV_LOG_INFO("bits_per_pixel %d matches stride calculation", dsc->vinfo.bits_per_pixel);
232 }
233 else {
234 LV_LOG_INFO("bits_per_pixel %d retained; stride-derived bpp %u ignored (zero or non-standard)",
235 dsc->vinfo.bits_per_pixel, stride_bpp);
236 }
237
238 /* Log the resolution after any bits_per_pixel correction so it reflects the value actually used */
239 LV_LOG_INFO("%dx%d, %dbpp", dsc->vinfo.xres, dsc->vinfo.yres, dsc->vinfo.bits_per_pixel);
240
241 /* Figure out the size of the screen in bytes*/
242 dsc->screensize = dsc->finfo.smem_len;/*finfo.line_length * vinfo.yres;*/
243
244#if LV_LINUX_FBDEV_MMAP
245 /* Map the device to memory*/
246 dsc->fbp = (char *)mmap(0, dsc->screensize, PROT_READ | PROT_WRITE, MAP_SHARED, dsc->fbfd, 0);
247 if((intptr_t)dsc->fbp == -1) {
248 perror("Error: failed to map framebuffer device to memory");
249 return LV_RESULT_INVALID;
250 }
251#endif
252
253 /* Don't initialise the memory to retain what's currently displayed / avoid clearing the screen.
254 * This is important for applications that only draw to a subsection of the full framebuffer.*/
255
256 LV_LOG_INFO("The framebuffer device was mapped to memory successfully");
257
258 switch(dsc->vinfo.bits_per_pixel) {
259 case 8:
260#if !LV_LINUX_FBDEV_BSD
261 if(dsc->vinfo.grayscale != 1) {
262 LV_LOG_WARN("8bpp framebuffer does not report a grayscale format (grayscale = %u); "
263 "LV_COLOR_FORMAT_L8 is not supported", (unsigned)dsc->vinfo.grayscale);
264 return LV_RESULT_INVALID;
265 }
266#endif
267 lv_display_set_color_format(disp, LV_COLOR_FORMAT_L8);
268 break;
269 case 16:
270 lv_display_set_color_format(disp, LV_COLOR_FORMAT_RGB565);
271 break;
272 case 24:
273 lv_display_set_color_format(disp, LV_COLOR_FORMAT_RGB888);
274 break;
275 case 32:
276 lv_display_set_color_format(disp, LV_COLOR_FORMAT_XRGB8888);
277 break;
278 default:
279 LV_LOG_WARN("Not supported color format (%d bits)", dsc->vinfo.bits_per_pixel);
280 return LV_RESULT_INVALID;
281 }
282
283#if !LV_LINUX_FBDEV_BSD
284 /* Auto-detect BGR framebuffer layout from vinfo field offsets.
285 * Standard RGB: red.offset > blue.offset (e.g., red=16, blue=0 for 32bpp)
286 * BGR layout: red.offset < blue.offset (e.g., red=0, blue=16 for 32bpp)
287 * When BGR is detected, R/B channels are swapped during flush. */
288 if(dsc->vinfo.red.length > 0 && dsc->vinfo.blue.length > 0) {
289 if(dsc->vinfo.red.offset < dsc->vinfo.blue.offset) {
290 dsc->swap_rb = true;
291 LV_LOG_INFO("BGR framebuffer detected (red.offset=%d, blue.offset=%d) — enabling R/B swap",
292 dsc->vinfo.red.offset, dsc->vinfo.blue.offset);
293 }
294 else {
295 dsc->swap_rb = false;
296 LV_LOG_INFO("RGB framebuffer layout (red.offset=%d, blue.offset=%d)",
297 dsc->vinfo.red.offset, dsc->vinfo.blue.offset);
298 }
299 }
300#endif
301
302 int32_t hor_res = dsc->vinfo.xres;
303 int32_t ver_res = dsc->vinfo.yres;
304 int32_t width = dsc->vinfo.width;
305 uint32_t draw_buf_size = hor_res * (dsc->vinfo.bits_per_pixel >> 3);
306 if(LV_LINUX_FBDEV_RENDER_MODE == LV_DISPLAY_RENDER_MODE_PARTIAL) {
307 draw_buf_size *= LV_LINUX_FBDEV_BUFFER_SIZE;
308 }
309 else {
310 draw_buf_size *= ver_res;
311 }
312
313 lv_color_format_t cf = lv_display_get_color_format(disp);
314 uint8_t * draw_buf = NULL;
315 uint8_t * draw_buf_2 = NULL;
316
317 dsc->draw_buf_1 = lv_malloc(draw_buf_size + LV_DRAW_BUF_ALIGN - 1);
318 LV_ASSERT_MALLOC(dsc->draw_buf_1);
319 if(dsc->draw_buf_1 == NULL) {
320 LV_LOG_ERROR("Failed to allocate draw buffer 1 (%u bytes)", draw_buf_size + LV_DRAW_BUF_ALIGN - 1);
321 return LV_RESULT_INVALID;
322 }
323 draw_buf = lv_draw_buf_align(dsc->draw_buf_1, cf);
324
325 if(LV_LINUX_FBDEV_BUFFER_COUNT == 2) {
326 dsc->draw_buf_2 = lv_malloc(draw_buf_size + LV_DRAW_BUF_ALIGN - 1);
327 LV_ASSERT_MALLOC(dsc->draw_buf_2);
328 if(dsc->draw_buf_2 == NULL) {
329 LV_LOG_ERROR("Failed to allocate draw buffer 2 (%u bytes)", draw_buf_size + LV_DRAW_BUF_ALIGN - 1);
330 lv_free(dsc->draw_buf_1);
331 dsc->draw_buf_1 = NULL;
332 return LV_RESULT_INVALID;
333 }
334 draw_buf_2 = lv_draw_buf_align(dsc->draw_buf_2, cf);
335 }
336
337 lv_display_set_resolution(disp, hor_res, ver_res);
338 lv_display_set_buffers(disp, draw_buf, draw_buf_2, draw_buf_size, LV_LINUX_FBDEV_RENDER_MODE);
339
340 if(width > 0) {
341 lv_display_set_dpi(disp, DIV_ROUND_UP(hor_res * 254, width * 10));
342 }
343
344 LV_LOG_INFO("Resolution is set to %" LV_PRId32 "x%" LV_PRId32 " at %" LV_PRId32 "dpi",
345 hor_res, ver_res, lv_display_get_dpi(disp));
346
347 return LV_RESULT_OK;
348}

393static void del_event_cb(lv_event_t * e)
394{
395 if(LV_EVENT_DELETE != lv_event_get_code(e))
396 return;
397
398 lv_display_t * disp = lv_event_get_target(e);
399 lv_linux_fb_t * dsc = lv_display_get_driver_data(disp);
400 if(!dsc) return;
401
402#if LV_LINUX_FBDEV_MMAP
403 if(MAP_FAILED != dsc->fbp) {
404 munmap(dsc->fbp, dsc->screensize);
405 dsc->fbp = MAP_FAILED;
406 }
407#endif
408 if(dsc->fbfd >= 0) {
409 close(dsc->fbfd);
410 dsc->fbfd = -1;
411 }
412 if(dsc->rotated_buf) lv_free(dsc->rotated_buf);
413 if(dsc->swap_line_buf) lv_free(dsc->swap_line_buf);
414 if(dsc->draw_buf_1) lv_free(dsc->draw_buf_1);
415 if(dsc->draw_buf_2) lv_free(dsc->draw_buf_2);
416 if(dsc->devname) lv_free((void *)dsc->devname);
417
418 lv_free(dsc);
419 lv_display_set_driver_data(disp, NULL);
420}

422static void flush_cb(lv_display_t * disp, const lv_area_t * area, uint8_t * color_p)
423{
424 lv_linux_fb_t * dsc = lv_display_get_driver_data(disp);
425
426#if LV_LINUX_FBDEV_MMAP
427 if(dsc->fbp == NULL) {
428 lv_display_flush_ready(disp);
429 return;
430 }
431#endif
432
433 /* Wait once per frame before drawing the next frame */
434 if(LV_LINUX_FBDEV_VSYNC && dsc->vsync_supported && dsc->wait_vsync_next) {
435 uint32_t dummy = 0;
436 if(ioctl(dsc->fbfd, FBIO_WAITFORVSYNC, &dummy) == -1) {
437 /* Disable the optional wait if the framebuffer does not support FBIO_WAITFORVSYNC*/
438 LV_LOG_WARN("FBIO_WAITFORVSYNC unsupported (%s); disabling vsync wait", strerror(errno));
439 dsc->vsync_supported = false;
440 }
441 }
442
443
444 const bool wait_for_last_flush = LV_LINUX_FBDEV_RENDER_MODE == LV_DISPLAY_RENDER_MODE_FULL;
445 const bool is_last_flush = lv_display_flush_is_last(disp);
446 const bool skip_flush = wait_for_last_flush && !is_last_flush;
447 /* wait for vsync every first frame*/
448 dsc->wait_vsync_next = is_last_flush;
449
450 if(skip_flush) {
451 lv_display_flush_ready(disp);
452 return;
453 }
454
455 const lv_color_format_t cf = lv_display_get_color_format(disp);
456 const uint32_t px_size = lv_color_format_get_size(cf);
457
458 lv_area_t rotated_area;
459 const lv_display_rotation_t rotation = lv_display_get_rotation(disp);
460
461 /* Not all framebuffer kernel drivers support hardware rotation, so we need to handle it in software here */
462 if(rotation != LV_DISPLAY_ROTATION_0) {
463 int32_t src_w;
464 int32_t src_h;
465 uint32_t src_stride;
466
467 /* Direct render mode rotation only works if we rotate the whole screen at the same time
468 * To do that, we use the display's resolution and as the area
469 * we also grab the current draw buffer so that we can rotate the whole display */
470 if(LV_LINUX_FBDEV_RENDER_MODE == LV_DISPLAY_RENDER_MODE_DIRECT) {
471 if(!is_last_flush) {
472 /* We need to wait for the last flush when using direct render mode with rotation*/
473 lv_display_flush_ready(disp);
474 return;
475 }
476 lv_draw_buf_t * draw_buf = lv_display_get_buf_active(disp);
477 src_w = lv_display_get_horizontal_resolution(disp);
478 src_h = lv_display_get_vertical_resolution(disp);
479 src_stride = lv_draw_buf_width_to_stride(src_w, cf);
480 color_p = draw_buf->data;
481 rotated_area.x1 = rotated_area.y1 = 0;
482 lv_area_set_width(&rotated_area, src_w);
483 lv_area_set_height(&rotated_area, src_h);
484 }
485 else {
486 /* For partial and full render modes, we need to rotate the current area
487 * In Full mode we will rotate the whole display just like with direct render mode
488 * but we don't need to do anything special since the area is already the full area of the display
489 * For Partial mode we will rotate just the part we're currently displaying*/
490 src_w = lv_area_get_width(area);
491 src_h = lv_area_get_height(area);
492 src_stride = lv_draw_buf_width_to_stride(lv_area_get_width(area), cf);
493 rotated_area = *area;
494 }
495
496 lv_display_rotate_area(disp, &rotated_area);
497 const uint32_t dest_stride = lv_draw_buf_width_to_stride(lv_area_get_width(&rotated_area), cf);
498 const size_t buf_size = dest_stride * lv_area_get_height(&rotated_area);
499 if(!dsc->rotated_buf || dsc->rotated_buf_size != buf_size) {
500 dsc->rotated_buf = lv_realloc(dsc->rotated_buf, buf_size);
501 LV_ASSERT_MALLOC(dsc->rotated_buf);
502 dsc->rotated_buf_size = buf_size;
503 }
504 lv_draw_rotate(color_p, dsc->rotated_buf, src_w, src_h, src_stride, dest_stride, rotation, cf);
505 area = &rotated_area;
506 color_p = dsc->rotated_buf;
507 }
508
509 lv_area_t display_area;
510 lv_area_set(&display_area, 0, 0, dsc->vinfo.xres - 1, dsc->vinfo.yres - 1);
511
512 /* Clip the area to the display bounds */
513 lv_area_t clipped_area;
514 if(!lv_area_intersect(&clipped_area, area, &display_area)) {
515 /* No intersection at all, nothing to render */
516 lv_display_flush_ready(disp);
517 return;
518 }
519
520 uint32_t fb_pos =
521 (clipped_area.x1 + dsc->vinfo.xoffset) * px_size +
522 (clipped_area.y1 + dsc->vinfo.yoffset) * dsc->finfo.line_length;
523 const int32_t w = lv_area_get_width(&clipped_area);
524
525 /* Prepare swap line buffer if R/B swap is active */
526 const size_t line_bytes = w * px_size;
527 uint8_t * swap_buf = NULL;
528 if(dsc->swap_rb) {
529 if(!dsc->swap_line_buf || dsc->swap_line_buf_size < line_bytes) {
530 uint8_t * tmp = lv_realloc(dsc->swap_line_buf, line_bytes);
531 LV_ASSERT_MALLOC(tmp);
532 if(tmp == NULL) {
533 LV_LOG_WARN("Failed to (re)allocate swap line buffer; skipping R/B swap this frame");
534 }
535 else {
536 dsc->swap_line_buf = tmp;
537 dsc->swap_line_buf_size = line_bytes;
538 }
539 }
540 swap_buf = dsc->swap_line_buf && dsc->swap_line_buf_size >= line_bytes ? dsc->swap_line_buf : NULL;
541 }
542
543 if(LV_LINUX_FBDEV_RENDER_MODE == LV_DISPLAY_RENDER_MODE_DIRECT && rotation == LV_DISPLAY_ROTATION_0) {
544 uint32_t color_pos =
545 (clipped_area.x1 - disp->offset_x) * px_size +
546 (clipped_area.y1 - disp->offset_y) * disp->hor_res * px_size;
547 for(int32_t y = clipped_area.y1; y <= clipped_area.y2; y++) {
548 const uint8_t * src = &color_p[color_pos];
549 if(swap_buf) {
550 if(px_size == 4) swap_rb_line_32(swap_buf, src, w);
551 else if(px_size == 3) swap_rb_line_24(swap_buf, src, w);
552 else swap_rb_line_16(swap_buf, src, w);
553 src = swap_buf;
554 }
555 write_to_fb(dsc, fb_pos, src, line_bytes);
556 fb_pos += dsc->finfo.line_length;
557 color_pos += disp->hor_res * px_size;
558 }
559 }
560 else {
561 /* Calculate offset into color_p buffer based on original area */
562 const int32_t x_offset = clipped_area.x1 - area->x1;
563 const int32_t y_offset = clipped_area.y1 - area->y1;
564 const int32_t stride = lv_draw_buf_width_to_stride(lv_area_get_width(area), cf);
565
566 color_p += y_offset * stride + x_offset * px_size;
567
568 for(int32_t y = clipped_area.y1; y <= clipped_area.y2; y++) {
569 const uint8_t * src = color_p;
570 if(swap_buf) {

600/* Swap red and blue channels for BGR framebuffers.
601 * LVGL renders as RGB; these convert to BGR during the line copy. */
602
603static void swap_rb_line_16(uint8_t * dst, const uint8_t * src, int32_t pixel_count)
604{
605 const uint16_t * s = (const uint16_t *)src;
606 uint16_t * d = (uint16_t *)dst;
607 for(int32_t i = 0; i < pixel_count; i++) {
608 uint16_t px = s[i];
609 /* RGB565: RRRRRGGG_GGGBBBBB -> BGR565: BBBBBGGG_GGGRRRRR */
610 uint16_t r = (px >> 11) & 0x1F;
611 uint16_t g = (px >> 5) & 0x3F;
612 uint16_t b = px & 0x1F;
613 d[i] = (b << 11) | (g << 5) | r;
614 }
615}
616
617static void swap_rb_line_24(uint8_t * dst, const uint8_t * src, int32_t pixel_count)
618{
619 for(int32_t i = 0; i < pixel_count; i++) {
620 int32_t off = i * 3;
621 dst[off + 0] = src[off + 2];
622 dst[off + 1] = src[off + 1];
623 dst[off + 2] = src[off + 0];
624 }
625}
626
627static void swap_rb_line_32(uint8_t * dst, const uint8_t * src, int32_t pixel_count)
628{
629 const uint32_t * s = (const uint32_t *)src;
630 uint32_t * d = (uint32_t *)dst;
631 for(int32_t i = 0; i < pixel_count; i++) {
632 uint32_t px = s[i];
633 /* xRGB -> xBGR: swap bits [23:16] (R) and bits [7:0] (B), keep G and x */
634 d[i] = (px & 0xFF00FF00u) | ((px >> 16) & 0xFFu) | ((px & 0xFFu) << 16);
635 }
636}
