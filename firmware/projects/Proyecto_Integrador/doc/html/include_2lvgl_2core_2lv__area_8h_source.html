201
202#define LV_COORD_TYPE_MASK (3 << LV_COORD_TYPE_SHIFT)
203#define LV_COORD_TYPE(x) ((x) & LV_COORD_TYPE_MASK) /*Extract type specifiers*/
204#define LV_COORD_PLAIN(x) ((x) & ~LV_COORD_TYPE_MASK) /*Remove type specifiers*/
205
206#define LV_COORD_TYPE_PX (0 << LV_COORD_TYPE_SHIFT)
207#define LV_COORD_TYPE_SPEC (1 << LV_COORD_TYPE_SHIFT)
208#define LV_COORD_TYPE_PX_NEG (3 << LV_COORD_TYPE_SHIFT)
209
210#define LV_COORD_IS_PX(x) (LV_COORD_TYPE(x) == LV_COORD_TYPE_PX || LV_COORD_TYPE(x) == LV_COORD_TYPE_PX_NEG)
211#define LV_COORD_IS_SPEC(x) (LV_COORD_TYPE(x) == LV_COORD_TYPE_SPEC)
212
213#define LV_COORD_SET_SPEC(x) ((x) | LV_COORD_TYPE_SPEC)
214
216#define LV_COORD_MAX ((1 << LV_COORD_TYPE_SHIFT) - 1)
217#define LV_COORD_MIN (-LV_COORD_MAX)
218
219/*Special coordinates*/
220#define LV_SIZE_CONTENT LV_COORD_SET_SPEC(LV_COORD_MAX)
221#define LV_PCT_STORED_MAX (LV_COORD_MAX - 1)
222#if LV_PCT_STORED_MAX % 2 != 0
223#error LV_PCT_STORED_MAX should be an even number
224#endif
225#define LV_PCT_POS_MAX (LV_PCT_STORED_MAX / 2)
226#define LV_PCT(x) (LV_COORD_SET_SPEC(((x) < 0 ? (LV_PCT_POS_MAX - LV_MAX((x), -LV_PCT_POS_MAX)) : LV_MIN((x), LV_PCT_POS_MAX))))
227#define LV_COORD_IS_PCT(x) ((LV_COORD_IS_SPEC(x) && LV_COORD_PLAIN(x) <= LV_PCT_STORED_MAX))
228#define LV_COORD_GET_PCT(x) (LV_COORD_PLAIN(x) > LV_PCT_POS_MAX ? LV_PCT_POS_MAX - LV_COORD_PLAIN(x) : LV_COORD_PLAIN(x))
