59void * lv_malloc(size_t size)
60{
61 LV_TRACE_MEM("allocating %lu bytes", (unsigned long)size);
62 if(size == 0) {
63 LV_TRACE_MEM("using zero_mem");
64 return &zero_mem;
65 }
66
67 void * alloc = lv_malloc_core(size);
68
69 if(alloc == NULL) {
70 LV_LOG_INFO("couldn't allocate memory (%lu bytes)", (unsigned long)size);
71#if LV_LOG_LEVEL <= LV_LOG_LEVEL_INFO
72 lv_mem_monitor_t mon;
73 lv_mem_monitor(&mon);
74 LV_LOG_INFO("used: %zu (%3d %%), frag: %3d %%, biggest free: %zu",
75 mon.total_size - mon.free_size, mon.used_pct, mon.frag_pct,
76 mon.free_biggest_size);
77#endif
78 return NULL;
79 }
80
81#if LV_MEM_ADD_JUNK
82 lv_memset(alloc, 0xaa, size);
83#endif
84
85 LV_TRACE_MEM("allocated at %p", alloc);
86 return alloc;
87}

116void * lv_calloc(size_t num, size_t size)
117{
118 LV_TRACE_MEM("allocating number of %zu each %zu bytes", num, size);
119 if(num != 0 && size > ((size_t) -1) / num) {
120 LV_LOG_WARN("lv_calloc: overflow detected (num=%zu, size=%zu)", num, size);
121 return NULL;
122 }
123 return lv_malloc_zeroed(num * size);
124}

149void * lv_realloc(void * data_p, size_t new_size)
150{
151 LV_TRACE_MEM("reallocating %p with %lu size", data_p, (unsigned long)new_size);
152 if(new_size == 0) {
153 LV_TRACE_MEM("using zero_mem");
154 lv_free(data_p);
155 return &zero_mem;
156 }
157
158 if(data_p == &zero_mem) return lv_malloc(new_size);
159
160 void * new_p = lv_realloc_core(data_p, new_size);
161
162 if(new_p == NULL) {
163 LV_LOG_ERROR("couldn't reallocate memory");
164 return NULL;
165 }
166
167 LV_TRACE_MEM("reallocated at %p", new_p);
168 return new_p;
169}
