104void lv_draw_pxp_rotate(const void * src_buf, void * dest_buf, int32_t src_width, int32_t src_height,
105 int32_t src_stride, int32_t dest_stride, lv_display_rotation_t rotation,
106 lv_color_format_t cf)
107{
108 lv_pxp_reset();
109
110 /* Convert rotation angle
111 * To be in sync with CPU, the received angle is counterclockwise
112 * and the PXP constants are for clockwise rotation
113 *
114 * counterclockwise clockwise
115 * LV_DISPLAY_ROTATION_90 -> kPXP_Rotate270
116 * LV_DISPLAY_ROTATION_270 -> kPXP_Rotate90
117 */
118 pxp_rotate_degree_t pxp_rotation;
119 switch(rotation) {
120 case LV_DISPLAY_ROTATION_0:
121 pxp_rotation = kPXP_Rotate0;
122 break;
123 case LV_DISPLAY_ROTATION_270:
124 pxp_rotation = kPXP_Rotate90;
125 break;
126 case LV_DISPLAY_ROTATION_180:
127 pxp_rotation = kPXP_Rotate180;
128 break;
129 case LV_DISPLAY_ROTATION_90:
130 pxp_rotation = kPXP_Rotate270;
131 break;
132 default:
133 pxp_rotation = kPXP_Rotate0;
134 break;
135 }
136 PXP_SetRotateConfig(PXP_ID, kPXP_RotateOutputBuffer, pxp_rotation, kPXP_FlipDisable);
137
138 /*Simple blit, no effect - Disable PS buffer*/
139 PXP_SetProcessSurfacePosition(PXP_ID, 0xFFFFU, 0xFFFFU, 0U, 0U);
140
141 /*AS buffer - source image*/
142 pxp_as_buffer_config_t asBufferConfig = {
143 .pixelFormat = pxp_get_as_px_format(cf),
144 .bufferAddr = (uint32_t)src_buf,
145 .pitchBytes = src_stride
146 };
147 PXP_SetAlphaSurfaceBufferConfig(PXP_ID, &asBufferConfig);
148 PXP_SetAlphaSurfacePosition(PXP_ID, 0U, 0U, src_width - 1U, src_height - 1U);
149 PXP_EnableAlphaSurfaceOverlayColorKey(PXP_ID, false);
150
151 /*Output buffer.*/
152 pxp_output_buffer_config_t outputBufferConfig = {
153 .pixelFormat = pxp_get_out_px_format(cf),
154 .interlacedMode = kPXP_OutputProgressive,
155 .buffer0Addr = (uint32_t)dest_buf,
156 .buffer1Addr = (uint32_t)0U,
157 .pitchBytes = dest_stride,
158 .width = src_width,
159 .height = src_height
160 };
161 PXP_SetOutputBufferConfig(PXP_ID, &outputBufferConfig);
162
163 lv_pxp_run();
164}

204static bool _pxp_draw_img_supported(const lv_draw_image_dsc_t * draw_dsc)
205{
206 const lv_image_dsc_t * img_dsc = draw_dsc->src;
207
208 bool has_recolor = (draw_dsc->recolor_opa > LV_OPA_MIN);
209 bool has_transform = (draw_dsc->rotation != 0 || draw_dsc->scale_x != LV_SCALE_NONE ||
210 draw_dsc->scale_y != LV_SCALE_NONE);
211
212 /* Recolor and transformation are not supported at the same time. */
213 if(has_recolor && has_transform)
214 return false;
215
216 bool has_opa = (draw_dsc->opa < (lv_opa_t)LV_OPA_MAX);
217 bool src_has_alpha = (img_dsc->header.cf == LV_COLOR_FORMAT_ARGB8888);
218
219 /*
220 * Recolor or transformation for images w/ opa or alpha channel can't
221 * be obtained in a single PXP configuration. Two steps are required.
222 */
223 if((has_recolor || has_transform) && (has_opa || src_has_alpha))
224 return false;
225
226 /* PXP can only rotate at 90x angles. */
227 if(draw_dsc->rotation % 900)
228 return false;
229
230 /*
231 * PXP is set to process 16x16 blocks to optimize the system for memory
232 * bandwidth and image processing time.
233 * The output engine essentially truncates any output pixels after the
234 * desired number of pixels has been written.
235 * When rotating a source image and the output is not divisible by the block
236 * size, the incorrect pixels could be truncated and the final output image
237 * can look shifted.
238 *
239 * No combination of rotate with flip, scaling or decimation is possible
240 * if buffer is unaligned.
241 */
242 if(has_transform && (img_dsc->header.w % 16 || img_dsc->header.h % 16))
243 return false;
244
245 return true;
246}

248static int32_t _pxp_evaluate(lv_draw_unit_t * u, lv_draw_task_t * t)
249{
250 LV_UNUSED(u);
251
252 const lv_draw_dsc_base_t * draw_dsc_base = (lv_draw_dsc_base_t *) t->draw_dsc;
253
254 if(!_pxp_dest_cf_supported(draw_dsc_base->layer->color_format))
255 return 0;
256
257 switch(t->type) {
258 case LV_DRAW_TASK_TYPE_FILL: {
259 const lv_draw_fill_dsc_t * draw_dsc = (lv_draw_fill_dsc_t *) t->draw_dsc;
260
261 /* Most simple case: just a plain rectangle (no radius, no gradient). */
262 if((draw_dsc->radius != 0) || (draw_dsc->grad.dir != (lv_grad_dir_t)LV_GRAD_DIR_NONE))
263 return 0;
264
265 if(t->preference_score > 70) {
266 t->preference_score = 70;
267 t->preferred_draw_unit_id = DRAW_UNIT_ID_PXP;
268 }
269 return 1;
270 }
271
272 case LV_DRAW_TASK_TYPE_LAYER: {
273 const lv_draw_image_dsc_t * draw_dsc = (lv_draw_image_dsc_t *) t->draw_dsc;
274 lv_layer_t * layer_to_draw = (lv_layer_t *)draw_dsc->src;
275
276 if(!_pxp_src_cf_supported(layer_to_draw->color_format))
277 return 0;
278
279 if(!_pxp_draw_img_supported(draw_dsc))
280 return 0;
281
282 if(t->preference_score > 70) {
283 t->preference_score = 70;
284 t->preferred_draw_unit_id = DRAW_UNIT_ID_PXP;
285 }
286 return 1;
287 }
288
289 case LV_DRAW_TASK_TYPE_IMAGE: {
290 lv_draw_image_dsc_t * draw_dsc = (lv_draw_image_dsc_t *) t->draw_dsc;
291 const lv_image_dsc_t * img_dsc = draw_dsc->src;
292
293 if(img_dsc->header.cf >= LV_COLOR_FORMAT_PROPRIETARY_START)
294 return 0;
295
296 if((!_pxp_src_cf_supported(img_dsc->header.cf)) ||
297 (!pxp_buf_aligned(img_dsc->data, img_dsc->header.stride)))
298 return 0;
299
300 if(!_pxp_draw_img_supported(draw_dsc))
301 return 0;
302
303 if(t->preference_score > 70) {
304 t->preference_score = 70;
305 t->preferred_draw_unit_id = DRAW_UNIT_ID_PXP;
306 }
307 return 1;
308 }
309 default:
310 return 0;
311 }
312
313 return 0;
314}

316static int32_t _pxp_dispatch(lv_draw_unit_t * draw_unit, lv_layer_t * layer)
317{
318 lv_draw_pxp_unit_t * draw_pxp_unit = (lv_draw_pxp_unit_t *) draw_unit;
319
320 /* Return immediately if it's busy with draw task. */
321 if(draw_pxp_unit->task_act)
322 return 0;
323
324 /* Try to get an ready to draw. */
325 lv_draw_task_t * t = lv_draw_get_available_task(layer, NULL, DRAW_UNIT_ID_PXP);
326
327 if(t == NULL || t->preferred_draw_unit_id != DRAW_UNIT_ID_PXP)
328 return LV_DRAW_UNIT_IDLE;
329
330 if(lv_draw_layer_alloc_buf(layer) == NULL) {
331 t->state = LV_DRAW_TASK_STATE_FAILED;
332 return LV_DRAW_UNIT_IDLE;
333 }
334
335 t->state = LV_DRAW_TASK_STATE_IN_PROGRESS;
336 draw_pxp_unit->task_act = t;
337
338#if LV_USE_PXP_DRAW_THREAD
339 /* Let the render thread work. */
340 if(draw_pxp_unit->inited)
341 lv_thread_sync_signal(&draw_pxp_unit->sync);
342#else
343 _pxp_execute_drawing(draw_pxp_unit);
344
345 draw_pxp_unit->task_act->state = LV_DRAW_TASK_STATE_FINISHED;
346 draw_pxp_unit->task_act = NULL;
347
348 /* The draw unit is free now. Request a new dispatching as it can get a new task. */
349 lv_draw_dispatch_request();
350#endif
351
352 return 1;
353}

376static void _pxp_execute_drawing(lv_draw_pxp_unit_t * u)
377{
378 lv_draw_task_t * t = u->task_act;
379 lv_layer_t * layer = t->target_layer;
380 lv_draw_buf_t * draw_buf = layer->draw_buf;
381
382 lv_area_t draw_area;
383 if(!lv_area_intersect(&draw_area, &t->area, &t->clip_area))
384 return; /*Fully clipped, nothing to do*/
385
386 /* Make area relative to the buffer */
387 lv_area_move(&draw_area, -layer->buf_area.x1, -layer->buf_area.y1);
388
389 /* Invalidate only the drawing area */
390 lv_draw_buf_invalidate_cache(draw_buf, &draw_area);
391
392#if LV_USE_PARALLEL_DRAW_DEBUG
393 t->draw_unit = &u->base_unit;
394#endif
395
396 switch(t->type) {
397 case LV_DRAW_TASK_TYPE_FILL:
398 lv_draw_pxp_fill(t);
399 break;
400 case LV_DRAW_TASK_TYPE_IMAGE:
401 lv_draw_pxp_img(t);
402 break;
403 case LV_DRAW_TASK_TYPE_LAYER:
404 lv_draw_pxp_layer(t);
405 break;
406 default:
407 break;
408 }
409
410#if LV_USE_PARALLEL_DRAW_DEBUG
411 /*Layers manage it for themselves*/
412 if(t->type != LV_DRAW_TASK_TYPE_LAYER) {
413 lv_area_t draw_area;
414 if(!lv_area_intersect(&draw_area, &t->area, &t->clip_area))
415 return;
416
417 int32_t idx = u->base_unit.idx;
418
419 lv_draw_rect_dsc_t rect_dsc;
420 lv_draw_rect_dsc_init(&rect_dsc);
421 rect_dsc.bg_color = lv_palette_main(idx % LV_PALETTE_LAST);
422 rect_dsc.border_color = rect_dsc.bg_color;
423 rect_dsc.bg_opa = LV_OPA_10;
424 rect_dsc.border_opa = LV_OPA_80;
425 rect_dsc.border_width = 1;
426 lv_draw_sw_fill((lv_draw_unit_t *)u, &rect_dsc, &draw_area);
427
428 lv_point_t txt_size;
429 lv_text_get_size_attributes(&txt_size, "W", LV_FONT_DEFAULT, 0, 0, 100, LV_TEXT_FLAG_NONE);
430
431 lv_area_t txt_area;
432 txt_area.x1 = draw_area.x1;
433 txt_area.y1 = draw_area.y1;
434 txt_area.x2 = draw_area.x1 + txt_size.x - 1;
435 txt_area.y2 = draw_area.y1 + txt_size.y - 1;
436
437 lv_draw_rect_dsc_init(&rect_dsc);
438 rect_dsc.bg_color = lv_color_white();
439 lv_draw_sw_fill((lv_draw_unit_t *)u, &rect_dsc, &txt_area);
440
441 char buf[8];
442 lv_snprintf(buf, sizeof(buf), "%d", idx);
443 lv_draw_label_dsc_t label_dsc;
444 lv_draw_label_dsc_init(&label_dsc);
445 label_dsc.color = lv_color_black();
446 label_dsc.text = buf;
447 lv_draw_sw_label((lv_draw_unit_t *)u, &label_dsc, &txt_area);
448 }
449#endif
450}

453static void _pxp_render_thread_cb(void * ptr)
454{
455 lv_draw_pxp_unit_t * u = ptr;
456
457 lv_thread_sync_init(&u->sync);
458 u->inited = true;
459
460 while(1) {
461 /* Wait for sync if there is no task set. */
462 while(u->task_act == NULL) {
463 if(u->exit_status)
464 break;
465
466 lv_thread_sync_wait(&u->sync);
467 }
468
469 if(u->exit_status) {
470 LV_LOG_INFO("Ready to exit PXP draw thread.");
471 break;
472 }
473
474 _pxp_execute_drawing(u);
475
476 /* Signal the ready state to dispatcher. */
477 u->task_act->state = LV_DRAW_TASK_STATE_FINISHED;
478
479 /* Cleanup. */
480 u->task_act = NULL;
481
482 /* The draw unit is free now. Request a new dispatching as it can get a new task. */
483 lv_draw_dispatch_request();
484 }
485
486 u->inited = false;
487 lv_thread_sync_delete(&u->sync);
488 LV_LOG_INFO("Exit PXP draw thread.");
489}
490#endif /*LV_USE_PXP_DRAW_THREAD*/
491#endif /*LV_USE_DRAW_PXP*/
