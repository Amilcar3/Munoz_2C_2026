91static lv_result_t init_display(lv_display_t * display)
92{
93 lv_sdl_sw_display_data_t * ddata = lv_malloc_zeroed(sizeof(*ddata));
94 if(!ddata) {
95 LV_LOG_WARN("No memory for display data");
96 return LV_RESULT_INVALID;
97 }
98 ddata->renderer = SDL_CreateRenderer(lv_sdl_window_get_window(display), -1,
99 LV_SDL_ACCELERATED ? SDL_RENDERER_ACCELERATED : SDL_RENDERER_SOFTWARE);
100 if(!ddata->renderer) {
101 LV_LOG_ERROR("Failed to create SDL renderer '%s'", SDL_GetError());
102 lv_free(ddata);
103 return LV_RESULT_INVALID;
104 }
105 lv_sdl_backend_set_display_data(display, ddata);
106
107 int32_t hor_res = lv_sdl_window_get_horizontal_resolution(display);
108 int32_t ver_res = lv_sdl_window_get_vertical_resolution(display);
109
110 resize_display(display);
111
112 uint32_t px_size = lv_color_format_get_size(lv_display_get_color_format(display));
113 lv_memset(ddata->fb1, 0xff, hor_res * ver_res * px_size);
114 if(ddata->fb2) lv_memset(ddata->fb2, 0xff, hor_res * ver_res * px_size);
115
116 lv_display_set_flush_cb(display, flush_cb);
117
118 if(LV_SDL_RENDER_MODE == LV_DISPLAY_RENDER_MODE_PARTIAL) {
119 uint32_t palette_size = LV_COLOR_INDEXED_PALETTE_SIZE(lv_display_get_color_format(display)) * 4;
120 uint32_t buffer_size_bytes = 32 * 1024 + palette_size;
121 ddata->buf1 = sdl_draw_buf_realloc_aligned(NULL, buffer_size_bytes);
122 if(LV_SDL_BUF_COUNT == 2) {
123 ddata->buf2 = sdl_draw_buf_realloc_aligned(NULL, buffer_size_bytes);
124 }
125 lv_display_set_buffers(display, ddata->buf1, ddata->buf2, buffer_size_bytes, LV_SDL_RENDER_MODE);
126 }
127 else {
128 /*LV_DISPLAY_RENDER_MODE_DIRECT or FULL */
129 uint32_t stride = lv_draw_buf_width_to_stride(display->hor_res,
130 lv_display_get_color_format(display));
131 lv_display_set_buffers(display, ddata->fb1, ddata->fb2, stride * display->ver_res,
132 LV_SDL_RENDER_MODE);
133 }
134 return LV_RESULT_OK;
135}
136static lv_result_t resize_display(lv_display_t * display)
137{
138 lv_color_format_t cf = lv_display_get_color_format(display);
139 /*In some cases SDL stride might be different than LVGL render stride, like in I1 format.
140 SDL still uses ARGB8888 as the color format, but LVGL renders in I1, thus causing a mismatch
141 This ensures correct stride for SDL buffers in this case.*/
142 if(cf == LV_COLOR_FORMAT_I1) {
143 cf = LV_COLOR_FORMAT_ARGB8888;
144 }
145
146 SDL_PixelFormatEnum px_format;
147 switch(cf) {
148 case LV_COLOR_FORMAT_ARGB8888:
149 px_format = SDL_PIXELFORMAT_ARGB8888;
150 break;
151 case LV_COLOR_FORMAT_XRGB8888:
152 px_format = SDL_PIXELFORMAT_RGB888;
153 break;
154 case LV_COLOR_FORMAT_RGB888:
155 px_format = SDL_PIXELFORMAT_BGR24;
156 break;
157 case LV_COLOR_FORMAT_RGB565:
158 px_format = SDL_PIXELFORMAT_RGB565;
159 break;
160 default:
161 /*The display has an unsupported color format. We can't do anything with it.*/
162 LV_LOG_ERROR("Unsupported color format: %d", cf);
163 LV_ASSERT(0);
164 return LV_RESULT_INVALID;
165 }
166
167
168 uint32_t stride = lv_draw_buf_width_to_stride(display->hor_res, cf);
169 lv_sdl_sw_display_data_t * ddata = lv_sdl_backend_get_display_data(display);
170
171 ddata->fb1 = sdl_draw_buf_realloc_aligned(ddata->fb1, stride * display->ver_res);
172 LV_ASSERT_MALLOC(ddata->fb1);
173 lv_memzero(ddata->fb1, stride * display->ver_res);
174
175 if(sdl_render_mode() == LV_DISPLAY_RENDER_MODE_PARTIAL) {
176 ddata->fb_act = ddata->fb1;
177 }
178 else {
179 if(LV_SDL_BUF_COUNT == 2) {
180 ddata->fb2 = sdl_draw_buf_realloc_aligned(ddata->fb2, stride * display->ver_res);
181 lv_memset(ddata->fb2, 0x00, stride * display->ver_res);
182 }
183 lv_display_set_buffers(display, ddata->fb1, ddata->fb2, stride * display->ver_res, LV_SDL_RENDER_MODE);
184 }
185 if(ddata->texture) SDL_DestroyTexture(ddata->texture);
186
187
188 ddata->texture = SDL_CreateTexture(ddata->renderer, px_format,
189 SDL_TEXTUREACCESS_STATIC, display->hor_res, display->ver_res);
190 SDL_SetTextureBlendMode(ddata->texture, SDL_BLENDMODE_BLEND);
191 return LV_RESULT_OK;
192}

194static void deinit_display(lv_display_t * display)
195{
196
197 lv_sdl_sw_display_data_t * ddata = lv_sdl_backend_get_display_data(display);
198 LV_ASSERT(ddata != NULL);
199 if(ddata->texture) {
200 SDL_DestroyTexture(ddata->texture);
201 ddata->texture = NULL;
202 }
203 if(ddata->renderer) {
204 SDL_DestroyRenderer(ddata->renderer);
205 ddata->renderer = NULL;
206 }
207
208 if(ddata->fb1) {
209 sdl_draw_buf_free(ddata->fb1);
210 ddata->fb1 = NULL;
211 }
212 if(ddata->fb2) {
213 sdl_draw_buf_free(ddata->fb2);
214 ddata->fb2 = NULL;
215 }
216
217 if(ddata->buf1) {
218 sdl_draw_buf_free(ddata->buf1);
219 ddata->buf1 = NULL;
220 }
221 if(ddata->buf2) {
222 sdl_draw_buf_free(ddata->buf2);
223 ddata->buf2 = NULL;
224 }
225
226 lv_free(ddata);
227 lv_sdl_backend_set_display_data(display, NULL);
228}

238static void * sdl_draw_buf_realloc_aligned(void * ptr, size_t new_size)
239{
240 if(ptr) {
241 sdl_draw_buf_free(ptr);
242 }
243
244 /* No need copy for drawing buffer */
245#ifndef _WIN32
246 /* Size must be multiple of align, See: https://en.cppreference.com/w/c/memory/aligned_alloc */
247#define BUF_ALIGN (LV_DRAW_BUF_ALIGN < sizeof(void *) ? sizeof(void *) : LV_DRAW_BUF_ALIGN)
248 return aligned_alloc(BUF_ALIGN, LV_ALIGN_UP(new_size, BUF_ALIGN));
249#else
250 return _aligned_malloc(LV_ALIGN_UP(new_size, LV_DRAW_BUF_ALIGN), LV_DRAW_BUF_ALIGN);
251#endif /* _WIN32 */
252}

263static void flush_cb(lv_display_t * display, const lv_area_t * area, uint8_t * px_map)
264{
265 lv_color_format_t cf = lv_display_get_color_format(display);
266 lv_sdl_sw_display_data_t * ddata = lv_sdl_backend_get_display_data(display);
267 uint32_t * argb_px_map = NULL;
268
269 if(LV_SDL_RENDER_MODE == LV_DISPLAY_RENDER_MODE_PARTIAL) {
270 if(cf == LV_COLOR_FORMAT_RGB565_SWAPPED) {
271 uint32_t width = lv_area_get_width(area);
272 uint32_t height = lv_area_get_height(area);
273 lv_draw_rgb565_swap(px_map, width * height);
274 }
275 /*Update values in a special OLED I1 --> ARGB8888 case
276 We render everything in I1, but display it in ARGB8888*/
277 if(cf == LV_COLOR_FORMAT_I1) {
278 /*I1 uses 1 bit wide pixels, ARGB8888 uses 4 byte wide pixels*/
279 cf = LV_COLOR_FORMAT_ARGB8888;
280 uint32_t width = lv_area_get_width(area);
281 uint32_t height = lv_area_get_height(area);
282 uint32_t argb_px_map_size = width * height * 4;
283 argb_px_map = malloc(argb_px_map_size);
284 if(argb_px_map == NULL) {
285 LV_LOG_ERROR("malloc failed");
286 lv_display_flush_ready(display);
287 return;
288 }
289 /* skip the palette */
290 px_map += LV_COLOR_INDEXED_PALETTE_SIZE(LV_COLOR_FORMAT_I1) * 4;
291 const uint32_t i1_stride = lv_draw_buf_width_to_stride(width, LV_COLOR_FORMAT_I1);
292 const uint32_t argb8888_stride = lv_draw_buf_width_to_stride(width, LV_COLOR_FORMAT_ARGB8888);
293 lv_draw_i1_to_argb8888(px_map, argb_px_map, width, height, i1_stride, argb8888_stride, 0xFF000000u, 0xFFFFFFFFu);
294 px_map = (uint8_t *)argb_px_map;
295 }
296
297 lv_area_t rotated_area = *area;
298 lv_display_rotate_area(display, &rotated_area);
299
300 int32_t px_map_w = lv_area_get_width(area);
301 int32_t px_map_h = lv_area_get_height(area);
302 uint32_t px_map_stride = lv_draw_buf_width_to_stride(lv_area_get_width(area), cf);
303 uint32_t px_size = lv_color_format_get_size(cf);
304
305 int32_t fb_stride = lv_draw_buf_width_to_stride(display->hor_res, cf);
306 uint8_t * fb_start = ddata->fb_act;
307 fb_start += rotated_area.y1 * fb_stride + rotated_area.x1 * px_size;
308 lv_display_rotation_t rotation = lv_display_get_rotation(display);
309
310 if(rotation == LV_DISPLAY_ROTATION_0) {
311 uint32_t px_map_line_bytes = lv_area_get_width(area) * px_size;
312
313 int32_t y;
314 for(y = area->y1; y <= area->y2; y++) {
315 lv_memcpy(fb_start, px_map, px_map_line_bytes);
316 px_map += px_map_stride;
317 fb_start += fb_stride;
318 }
319 }
320 else {
321 lv_draw_rotate(px_map, fb_start, px_map_w, px_map_h, px_map_stride, fb_stride, rotation, cf);
322 }
323 }
324
325 if(lv_display_flush_is_last(display)) {
326 if(sdl_render_mode() != LV_DISPLAY_RENDER_MODE_PARTIAL) {
327 ddata->fb_act = px_map;
328 }
329 window_update(display);
330 }
331 free(argb_px_map);
332 lv_display_flush_ready(display);
333}

335static lv_result_t window_update(lv_display_t * display)
336{
337 lv_color_format_t cf = lv_display_get_color_format(display);
338 lv_sdl_sw_display_data_t * ddata = lv_sdl_backend_get_display_data(display);
339 if(cf == LV_COLOR_FORMAT_I1) {
340 cf = LV_COLOR_FORMAT_ARGB8888;
341 }
342 uint32_t stride = lv_draw_buf_width_to_stride(display->hor_res, cf);
343 SDL_UpdateTexture(ddata->texture, NULL, ddata->fb_act, stride);
344
345 SDL_RenderClear(ddata->renderer);
346
347 /*Update the renderer with the texture containing the rendered image*/
348 SDL_RenderCopy(ddata->renderer, ddata->texture, NULL, NULL);
349 SDL_RenderPresent(ddata->renderer);
350 return LV_RESULT_OK;
351}
