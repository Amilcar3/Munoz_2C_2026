216static void lv_animimg_constructor(const lv_obj_class_t * class_p, lv_obj_t * obj)
217{
218 LV_TRACE_OBJ_CREATE("begin");
219
220 LV_UNUSED(class_p);
221 LV_ASSERT(obj != NULL);
222 lv_animimg_t * animimg = (lv_animimg_t *)obj;
223
224 animimg->dsc = NULL;
225 animimg->pic_count = -1;
226
227 /*initial animation*/
228 lv_anim_init(&animimg->anim);
229 lv_anim_set_var(&animimg->anim, obj);
230 lv_anim_set_duration(&animimg->anim, 30);
231 lv_anim_set_exec_cb(&animimg->anim, (lv_anim_exec_xcb_t)index_change);
232 lv_anim_set_values(&animimg->anim, 0, 1);
233 lv_anim_set_repeat_count(&animimg->anim, LV_ANIM_REPEAT_INFINITE);
234}

236static void index_change(lv_obj_t * obj, int32_t idx)
237{
238 LV_ASSERT(obj != NULL);
239 lv_animimg_t * animimg = (lv_animimg_t *)obj;
240
241 if(animimg->dsc == NULL) {
242 LV_LOG_WARN("dsc is null");
243 return;
244 }
245
246 if(idx >= animimg->pic_count) idx = animimg->pic_count - 1;
247
248 lv_image_set_src(obj, animimg->dsc[idx]);
249}

251static void lv_animimg_set_src_inner(lv_obj_t * obj, const void * dsc[], size_t num, bool reverse)
252{
253 LV_ASSERT(obj != NULL);
254 LV_ASSERT(dsc != NULL);
255 lv_animimg_t * animimg = (lv_animimg_t *)obj;
256 animimg->dsc = dsc;
257 animimg->pic_count = num;
258 if(reverse) {
259 lv_anim_set_values(&animimg->anim, (int32_t)num, 0);
260 }
261 else {
262 lv_anim_set_values(&animimg->anim, 0, (int32_t)num);
263 }
264}
