1#!/usr/bin/env python3
2"""Generate C files for every XML example by driving `lved-cli.js`.
3
4Each XML example under `examples/` is run through the LVGL Pro editor's CLI
5(`lved-cli.js generate <project>`), which only operates on a project tree, so
6this script shuttles each example XML through `examples/xml_project/screens/`,
7copies the generated `_gen.c` back beside its source XML, and finishes by
8running the `cleanup_examples.py` transformations + wrapping the file in an
9`#if LV_USE_<topic> && LV_BUILD_EXAMPLES` build guard.
10
11Once every XML is processed the script rewrites each touched topic's
12`lv_example_<topic>.h` (one prototype per `.c` on disk), runs
13`code-format.py examples` to astyle the output, and wipes every `.c`/`.h`
14file from `examples/xml_project/` so the scaffolding the CLI generates can't
15collide with the host project's source globbing.
16
17USAGE
18-----
19 python scripts/generate_examples.py [--cli /path/to/lved-cli.js] [paths...]
20
21Without paths, every XML under `examples/` is processed except those inside
22`examples/xml_project/` and the top-level `project.xml` / `globals.xml`.
23Paths may be individual XML files or directories to recurse into.
24"""
25

41REPO_ROOT = Path(__file__).resolve().parent.parent
42EXAMPLES_DIR = REPO_ROOT / "examples"
43PROJECT_DIR = EXAMPLES_DIR / "xml_project"
44SCREENS_DIR = PROJECT_DIR / "screens"
45CODE_FORMATTER = REPO_ROOT / "scripts" / "code-format.py"

47# Files at the examples/ root that are project metadata, not examples.
48ROOT_METADATA = {"project.xml", "globals.xml"}
49
50# Topic folders whose name has no matching `LV_USE_<TOPIC>` macro. LVGL
51# always builds scroll handling, the style system, and the base `lv_obj`
52# widget in, so these examples get gated on `LV_BUILD_EXAMPLES` alone —
53# emitting `LV_USE_STYLES` / `LV_USE_SCROLL` / `LV_USE_OBJ` (undefined)
54# would make the `#if` false and silently drop every example in those
55# folders. `get_started` is a mixed-widget intro topic with no single
56# `LV_USE_GET_STARTED` macro, so it gets the same `LV_BUILD_EXAMPLES`-only
57# treatment.
58TOPICS_WITHOUT_LV_USE = {"scroll", "styles", "obj", "get_started"}
59
60# Categories that group several independent topics, each with its own
61# `LV_USE_<TOPIC>` macro and `lv_example_<topic>.h` header. For these the
62# topic folder is `examples/<category>/<topic>`. Every other category is
63# itself a single topic (e.g. `scroll`, `styles`), so its topic folder is
64# `examples/<category>`.
65#
66# Widget examples are flat (`widgets/<topic>/lv_example_*.xml`); `layouts`
67# and `libs` still nest a feature folder
68# (`<category>/<topic>/<feature>/lv_example_*.xml`). Resolving the topic by
69# category rather than by directory depth handles both layouts uniformly.
70MULTI_TOPIC_CATEGORIES = {"widgets", "layouts", "libs"}

112def generate_one(xml_path: Path, cli_path: str) -> bool:
113 """Generate the C file for a single XML. Return True on success."""
114 name = xml_path.stem # e.g. "lv_example_flex_grow"
115 staged_xml = SCREENS_DIR / xml_path.name
116 generated_c = SCREENS_DIR / f"{name}_gen.c"
117 generated_h = SCREENS_DIR / f"{name}_gen.h"
118 target_c = xml_path.with_suffix(".c")
119
120 SCREENS_DIR.mkdir(parents=True, exist_ok=True)
121 shutil.copyfile(xml_path, staged_xml)
122
123 try:
124 result = _run_generator(cli_path)
125 if result.returncode != 0:
126 sys.stderr.write(
127 f" ! lved-cli failed (exit {result.returncode}):\n{result.stderr}"
128 )
129 return False
130
131 if not generated_c.exists():
132 sys.stderr.write(
133 f" ! expected {generated_c.name} after generation, not found\n"
134 )
135 return False
136
137 shutil.copyfile(generated_c, target_c)
138 cleanup_examples.process(target_c)
139 wrap_with_build_guard(xml_path, target_c)
140 return True
141 finally:
142 # Tidy up so the screens folder doesn't accumulate per-file artifacts
143 # between runs. The pre-existing test example (e.g.
144 # `lv_example_dropdown_text.xml`) is untouched because its name
145 # differs from whatever we staged.
146 for f in (staged_xml, generated_c, generated_h):
147 if f.exists():
148 f.unlink()

151def _topic_dir_for(xml: Path) -> Path:
152 """Return the topic folder for an example XML.
153
154 The topic folder owns the shared `lv_example_<topic>.h` header and maps
155 to the `LV_USE_<TOPIC>` build guard. For a multi-topic category
156 (`widgets`/`layouts`/`libs`) it is `examples/<category>/<topic>`;
157 otherwise the category folder itself is the topic. Computed from the
158 category, not the directory depth, so flat widget examples
159 (`widgets/<topic>/lv_example_*.xml`) and still-nested layouts/libs
160 (`<category>/<topic>/<feature>/...`) both resolve correctly.
161 """
162 if EXAMPLES_DIR not in xml.parents:
163 return xml.parent
164 parts = xml.relative_to(EXAMPLES_DIR).parts
165 if parts[0] in MULTI_TOPIC_CATEGORIES and len(parts) >= 3:
166 return EXAMPLES_DIR / parts[0] / parts[1]
167 return EXAMPLES_DIR / parts[0]

170def _guard_expr_for(xml_path: Path) -> str:
171 """Return the `#if` expression for an example, derived from its topic.
172
173 The topic folder name maps to `LV_USE_<TOPIC>` (e.g. `canvas` →
174 `LV_USE_CANVAS`, `flex` → `LV_USE_FLEX`). Topics in
175 `TOPICS_WITHOUT_LV_USE` have no such macro and are gated on
176 `LV_BUILD_EXAMPLES` only.
177 """
178 topic = _topic_dir_for(xml_path).name
179 if topic in TOPICS_WITHOUT_LV_USE:
180 return "LV_BUILD_EXAMPLES"
181 return f"LV_USE_{topic.upper()} && LV_BUILD_EXAMPLES"
182
183
184# First `#include` line in the cleaned file. `cleanup_examples` collapses the
185# include block to a single relative `lvgl.h`, so the first match is it; the
186# `#if` is inserted right after so the guard wraps everything below the
187# include (mirroring the hand-written examples).
188_FIRST_INCLUDE_RE = re.compile(
189 r"^#include[ \t]+[\"<][^\">\n]+[\">][ \t]*\n", re.MULTILINE
190)
191# Idempotency probe: an existing top-level guard already mentioning
192# LV_BUILD_EXAMPLES means this file is already wrapped.
193_EXISTING_GUARD_RE = re.compile(r"^#if\b.*\bLV_BUILD_EXAMPLES\b", re.MULTILINE)

196def wrap_with_build_guard(xml_path: Path, c_path: Path) -> None:
197 """Wrap a cleaned example C file in its `#if … #endif` build guard.
198
199 Inserts `#if <expr>` directly after the include line and appends
200 `#endif` at EOF. Idempotent: a file that already carries a
201 `LV_BUILD_EXAMPLES` guard is left untouched.
202 """
203 text = c_path.read_text()
204 if _EXISTING_GUARD_RE.search(text):
205 return
206
207 m = _FIRST_INCLUDE_RE.search(text)
208 if not m:
209 sys.stderr.write(
210 f" ! no #include found in {c_path.name}, skipping build guard\n"
211 )
212 return
213
214 guard = f"#if {_guard_expr_for(xml_path)}\n"
215 text = text[: m.end()] + guard + text[m.end():]
216 c_path.write_text(text.rstrip() + "\n#endif\n")

219# The per-widget header is rewritten wholesale on every run. Every prototype
220# comes from a `.c` file actually present in the widget's folder — there is
221# no marker / preserved-region machinery.
222#
223# Convention used to map `.c` file → prototype. Both forms map to
224# `void lv_example_<stem>(void)`:
225# * `<topic>/lv_example_<stem>.c` → directly in the topic folder
226# (every widget example, plus C-only examples anywhere).
227# * `<topic>/<feature>/lv_example_<stem>.c` → still-nested layouts/libs.
228HEADER_TEMPLATE = """\
229/**
230 * @file {header_name}
231 *
232 */
233
234#ifndef {guard}
235#define {guard}
236
237#ifdef __cplusplus
238extern "C" {{
239#endif
240
241{prototypes}
242
243#ifdef __cplusplus
244}} /*extern "C"*/
245#endif
246
247#endif /*{guard}*/
248"""

251def _collect_prototypes(topic_dir: Path) -> list[str]:
252 """Return prototype lines for every `.c` example under `topic_dir`.
253
254 Examples sit either directly in the topic folder (every widget example
255 and any C-only example) or in a feature subfolder (still-nested
256 layouts/libs). Both globs are unioned so the header lists them all,
257 each using the bare file stem as the function name (`void <stem>(void)`).
258 """
259 nested_stems = {c.stem for c in topic_dir.glob("*/lv_example_*.c")}
260 direct_stems = {c.stem for c in topic_dir.glob("lv_example_*.c")}
261
262 return [f"void {stem}(void);" for stem in sorted(nested_stems | direct_stems)]

265def write_topic_headers(touched: list[Path]) -> list[Path]:
266 """Rewrite `lv_example_<topic>.h` from the `.c` files on disk.
267
268 No markers, no preserved region. Adding or removing a `.c` file
269 (XML- or hand-written) automatically reshapes the header on the next
270 run.
271 """
272 topic_dirs = {_topic_dir_for(xml) for xml in touched}
273 written: list[Path] = []
274 for topic_dir in sorted(topic_dirs):
275 if not topic_dir.is_dir() or EXAMPLES_DIR not in topic_dir.parents:
276 continue
277 topic = topic_dir.name
278 prototypes = _collect_prototypes(topic_dir)
279 if not prototypes:
280 continue
281 header_path = topic_dir / f"lv_example_{topic}.h"
282 guard = f"LV_EXAMPLE_{topic.upper()}_H"
283 header_path.write_text(
284 HEADER_TEMPLATE.format(
285 header_name=header_path.name,
286 guard=guard,
287 prototypes="\n".join(prototypes),
288 )
289 )
290 written.append(header_path)
291 return written

294def cleanup_project_files() -> None:
295 """Wipe every `.c`/`.h` file under `examples/xml_project/`.
296
297 The CLI generates scaffolding (`xml_project.{c,h}`, `xml_project_gen.{c,h}`,
298 per-screen `_gen.{c,h}`, plus font/image `_data.c` files) that would collide
299 with the host project's source globbing. We don't need it between runs —
300 the CLI re-creates whatever it needs on next invocation.
301 """
302 if not PROJECT_DIR.is_dir():
303 return
304 for pattern in ("*.c", "*.h"):
305 for f in PROJECT_DIR.rglob(pattern):
306 f.unlink()

322def regenerate_project(cli_path: str) -> bool:
323 """Re-run the CLI against the now-empty `screens/` to sync project files.
324
325 `generate_one` deletes each staged screen in its `finally` block, so once
326 every example is processed `screens/` is empty again. The project-level
327 generated artifacts (`xml_project_gen.{c,h}`, `file_list_gen.cmake`) still
328 reference whatever screen was staged last, though. One final generation
329 against the empty tree rewrites them to match real project content instead
330 of a transient staged screen.
331
332 Skipped (returning False) if `screens/` still holds staged XML, so a
333 partial/interrupted run won't bake a half-state into the project files.
334 """
335 staged = sorted(SCREENS_DIR.glob("*.xml"))
336 if staged:
337 sys.stderr.write(
338 " ! screens/ still has XML, skipping final regeneration: "
339 f"{', '.join(p.name for p in staged)}\n"
340 )
341 return False
342 result = _run_generator(cli_path)
343 if result.returncode != 0:
344 sys.stderr.write(
345 f" ! final regeneration failed (exit {result.returncode}):\n"
346 f"{result.stderr}"
347 )
348 return False
349 return True

352def collect_targets(paths: list[str]) -> list[Path]:
353 """Expand user-supplied paths into a flat list of XML files."""
354 out: list[Path] = []
355 for raw in paths:
356 p = Path(raw).resolve()
357 if p.is_dir():
358 out.extend(sorted(p.rglob("*.xml")))
359 elif p.suffix == ".xml" and p.is_file():
360 out.append(p)
361 else:
362 sys.stderr.write(f"skipping (not an XML or directory): {p}\n")
363 return out

365def main(argv: list[str]) -> int:
366 parser = argparse.ArgumentParser(
367 description="Generate C files for XML examples via lved-cli.js."
368 )
369 parser.add_argument(
370 "--cli",
371 help="path to lved-cli.js (defaults to a PATH lookup)",
372 )
373 parser.add_argument(
374 "paths",
375 nargs="*",
376 help="specific XML files or directories to process; defaults to every "
377 "example under examples/",
378 )
379 args = parser.parse_args(argv)
380
381 cli_path = resolve_cli(args.cli)
382 targets = collect_targets(args.paths) if args.paths else find_example_xmls()
383
384 if not targets:
385 print("No XML examples found.")
386 return 0
387
388 # `finally` guarantees the scaffolding cleanup runs even if generation
389 # raises, so the tree never keeps globbable xml_project `.c`/`.h` behind.
390 ok = 0
391 try:
392 for xml in targets:
393 rel = xml.relative_to(REPO_ROOT) if REPO_ROOT in xml.parents else xml
394 print(f"processing: {rel}")
395 if generate_one(xml, cli_path):
396 ok += 1
397
398 headers = write_topic_headers(targets)
399 for h in headers:
400 rel = h.relative_to(REPO_ROOT) if REPO_ROOT in h.parents else h
401 print(f"wrote header: {rel}")
402
403 print("regenerating project against empty screens/ to sync _gen files")
404 regenerate_project(cli_path)
405
406 print("formatting examples/ via scripts/code-format.py")
407 format_examples()
408 finally:
409 cleanup_project_files()
410
411 print(f"\n{ok} of {len(targets)} examples generated.")
412 return 0 if ok == len(targets) else 1
