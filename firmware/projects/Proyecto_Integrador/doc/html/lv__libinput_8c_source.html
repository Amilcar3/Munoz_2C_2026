109lv_libinput_capability lv_libinput_query_capability(struct libinput_device * device)
110{
111 LV_CHECK_ARG(device != NULL, return LV_LIBINPUT_CAPABILITY_NONE);
112
113 lv_libinput_capability capability = LV_LIBINPUT_CAPABILITY_NONE;
114 if(libinput_device_has_capability(device, LIBINPUT_DEVICE_CAP_KEYBOARD)
115 && (libinput_device_keyboard_has_key(device, KEY_ENTER) || libinput_device_keyboard_has_key(device, KEY_KPENTER))) {
116 capability |= LV_LIBINPUT_CAPABILITY_KEYBOARD;
117 }
118 if(libinput_device_has_capability(device, LIBINPUT_DEVICE_CAP_POINTER)) {
119 capability |= LV_LIBINPUT_CAPABILITY_POINTER;
120 }
121 if(libinput_device_has_capability(device, LIBINPUT_DEVICE_CAP_TOUCH)) {
122 capability |= LV_LIBINPUT_CAPABILITY_TOUCH;
123 }
124 return capability;
125}

134size_t lv_libinput_find_devs(lv_libinput_capability capabilities, char ** found, size_t count, bool force_rescan)
135{
136 LV_CHECK_ARG(found != NULL, return 0);
137
138 if((!devices || force_rescan) && !_rescan_devices()) {
139 return 0;
140 }
141
142 size_t num_found = 0;
143
144 for(size_t i = 0; i < num_devices && num_found < count; ++i) {
145 if(devices[i].capabilities & capabilities) {
146 found[num_found] = devices[i].path;
147 num_found++;
148 }
149 }
150
151 return num_found;
152}

159lv_indev_t * lv_libinput_create(lv_indev_type_t indev_type, const char * dev_path)
160{
161 LV_CHECK_ARG(dev_path != NULL, return NULL);
162
163 lv_libinput_t * dsc = lv_malloc_zeroed(sizeof(lv_libinput_t));
164 LV_ASSERT_MALLOC(dsc);
165 if(dsc == NULL) return NULL;
166
167 dsc->libinput_context = libinput_path_create_context(&interface, NULL);
168 if(!dsc->libinput_context) {
169 LV_LOG_ERROR("libinput_path_create_context failed: %s", strerror(errno));
170 _delete(dsc);
171 return NULL;
172 }
173
174 dsc->libinput_device = libinput_path_add_device(dsc->libinput_context, dev_path);
175 if(!dsc->libinput_device) {
176 _delete(dsc);
177 return NULL;
178 }
179
180 dsc->libinput_device = libinput_device_ref(dsc->libinput_device);
181 if(!dsc->libinput_device) {
182 _delete(dsc);
183 return NULL;
184 }
185
186 dsc->fd = libinput_get_fd(dsc->libinput_context);
187
188 /* Prepare poll */
189 dsc->fds[0].fd = dsc->fd;
190 dsc->fds[0].events = POLLIN;
191 dsc->fds[0].revents = 0;
192
193#if LV_LIBINPUT_XKB
194 struct xkb_rule_names names = XKB_RULE_NAMES;
195 lv_xkb_init(&(dsc->xkb), names);
196#endif /* LV_LIBINPUT_XKB */
197
198 /* Create indev */
199 lv_indev_t * indev = lv_indev_create();
200 if(!indev) {
201 _delete(dsc);
202 return NULL;
203 }
204 lv_indev_set_type(indev, indev_type);
205 lv_indev_set_read_cb(indev, _read);
206 lv_indev_set_driver_data(indev, dsc);
207 lv_indev_add_event_cb(indev, _indev_delete, LV_EVENT_DELETE, NULL);
208
209 /* Set up thread & lock */
210 if(pthread_mutex_init(&dsc->event_lock, NULL) != 0) {
211 LV_LOG_ERROR("pthread_mutex_init failed");
212 lv_indev_delete(indev);
213 return NULL;
214 }
215
216 if(pthread_create(&dsc->worker_thread, NULL, _poll_thread, dsc) != 0) {
217 LV_LOG_ERROR("pthread_create failed");
218 pthread_mutex_destroy(&dsc->event_lock);
219 lv_indev_delete(indev);
220 return NULL;
221 }
222
223 dsc->running = 1;
224
225 return indev;
226}

231
232 lv_indev_delete(indev);

243static bool _rescan_devices(void)
244{
245 _reset_scanned_devices();
246
247 DIR * dir;
248 struct dirent * ent;
249 if(!(dir = opendir("/dev/input"))) {
250 perror("unable to open directory /dev/input");
251 return false;
252 }
253
254 struct libinput * context = libinput_path_create_context(&interface, NULL);
255
256 while((ent = readdir(dir))) {
257 if(strncmp(ent->d_name, "event", 5) != 0) {
258 continue;
259 }
260
261 /* 11 characters for /dev/input/ + length of name + 1 NUL terminator */
262 char * path = malloc((11 + strlen(ent->d_name) + 1) * sizeof(char));
263 if(!path) {
264 perror("could not allocate memory for device node path");
265 libinput_unref(context);
266 _reset_scanned_devices();
267 closedir(dir);
268 return false;
269 }
270 strcpy(path, "/dev/input/");
271 strcat(path, ent->d_name);
272
273 struct libinput_device * device = libinput_path_add_device(context, path);
274 if(!device) {
275 perror("unable to add device to libinput context");
276 free(path);
277 continue;
278 }
279
280 /* The device pointer is guaranteed to be valid until the next libinput_dispatch. Since we're not dispatching events
281 * as part of this function, we don't have to increase its reference count to keep it alive.
282 * https://wayland.freedesktop.org/libinput/doc/latest/api/group__base.html#gaa797496f0150b482a4e01376bd33a47b */
283
284 lv_libinput_capability capabilities = lv_libinput_query_capability(device);
285
286 libinput_path_remove_device(device);
287
288 if(capabilities == LV_LIBINPUT_CAPABILITY_NONE) {
289 free(path);
290 continue;
291 }
292
293 if(!_add_scanned_device(path, capabilities)) {
294 free(path);
295 libinput_unref(context);
296 _reset_scanned_devices();
297 closedir(dir);
298 return false;
299 }
300 }
301
302 libinput_unref(context);
303 closedir(dir);
304 return true;
305}

313static bool _add_scanned_device(char * path, lv_libinput_capability capabilities)
314{
315 LV_ASSERT(path != NULL);
316 /* Double array size every 2^n elements */
317 if((num_devices & (num_devices + 1)) == 0) {
318 struct _lv_libinput_device * tmp = realloc(devices, (2 * num_devices + 1) * sizeof(struct _lv_libinput_device));
319 if(!tmp) {
320 perror("could not reallocate memory for devices array");
321 return false;
322 }
323 devices = tmp;
324 }
325
326 devices[num_devices].path = path;
327 devices[num_devices].capabilities = capabilities;
328 num_devices++;
329
330 return true;
331}

336static void _reset_scanned_devices(void)
337{
338 if(!devices) {
339 return;
340 }
341
342 for(size_t i = 0; i < num_devices; ++i) {
343 free(devices[i].path);
344 }
345 free(devices);
346
347 devices = NULL;
348 num_devices = 0;
349}

351static void * _poll_thread(void * data)
352{
353 LV_ASSERT(data != NULL);
354 lv_libinput_t * dsc = (lv_libinput_t *)data;
355 struct libinput_event * event;
356 int rc = 0;
357
358 LV_LOG_INFO("libinput: poll worker started");
359
360 while(!dsc->deinit) {
361 rc = poll(dsc->fds, nfds, timeout);
362 switch(rc) {
363 case -1:
364 perror(NULL);
365 __attribute__((fallthrough));
366 case 0:
367 continue;
368 default:
369 break;
370 }
371 libinput_dispatch(dsc->libinput_context);
372 pthread_mutex_lock(&dsc->event_lock);
373 while((event = libinput_get_event(dsc->libinput_context)) != NULL) {
374 _read_pointer(dsc, event);
375 _read_keypad(dsc, event);
376 libinput_event_destroy(event);
377 }
378 pthread_mutex_unlock(&dsc->event_lock);
379 LV_LOG_INFO("libinput: event read");
380 }
381
382 dsc->deinit = false; /* Signal that we're done */
383
384 return NULL;
385}

387lv_libinput_event_t * _get_event(lv_libinput_t * dsc)
388{
389 LV_ASSERT(dsc != NULL);
390 if(dsc->start == dsc->end) {
391 return NULL;
392 }
393
394 lv_libinput_event_t * evt = &dsc->points[dsc->start];
395
396 if(++dsc->start == LV_LIBINPUT_MAX_EVENTS)
397 dsc->start = 0;
398
399 return evt;
400}

404 LV_ASSERT(dsc != NULL);
405 return dsc->start != dsc->end;

408lv_libinput_event_t * _create_event(lv_libinput_t * dsc)
409{
410 LV_ASSERT(dsc != NULL);
411 lv_libinput_event_t * evt = &dsc->points[dsc->end];
412
413 if(++dsc->end == LV_LIBINPUT_MAX_EVENTS)
414 dsc->end = 0;
415
416 /* We have overflowed the buffer, start overwriting
417 * old events.
418 */
419 if(dsc->end == dsc->start) {
420 LV_LOG_INFO("libinput: overflowed event buffer!");
421 if(++dsc->start == LV_LIBINPUT_MAX_EVENTS)
422 dsc->start = 0;
423 }
424
425 memset(evt, 0, sizeof(lv_libinput_event_t));
426
427 return evt;
428}

430static void _read(lv_indev_t * indev, lv_indev_data_t * data)
431{
432 LV_ASSERT(indev != NULL);
433 LV_ASSERT(data != NULL);
434 lv_libinput_t * dsc = lv_indev_get_driver_data(indev);
435 LV_ASSERT(dsc != NULL);
436
437 pthread_mutex_lock(&dsc->event_lock);
438
439 lv_libinput_event_t * evt = _get_event(dsc);
440
441 if(!evt)
442 evt = &dsc->last_event; /* indev expects us to report the most recent state */
443
444 data->point = evt->point;
445 data->state = evt->pressed;
446 data->key = evt->key_val;
447 data->continue_reading = _event_pending(dsc);
448
449 dsc->last_event = *evt; /* Remember the last event for the next call */
450
451 pthread_mutex_unlock(&dsc->event_lock);
452
453 if(evt)
454 LV_LOG_TRACE("libinput_read: (%04d, %04d): %d continue_reading? %d", data->point.x, data->point.y, data->state,
455 data->continue_reading);
456}

458static void _read_pointer(lv_libinput_t * dsc, struct libinput_event * event)
459{
460 LV_ASSERT(dsc != NULL);
461 LV_ASSERT(event != NULL);
462 struct libinput_event_touch * touch_event = NULL;
463 struct libinput_event_pointer * pointer_event = NULL;
464 lv_libinput_event_t * evt = NULL;
465 enum libinput_event_type type = libinput_event_get_type(event);
466 int slot = 0;
467
468 switch(type) {
469 case LIBINPUT_EVENT_TOUCH_MOTION:
470 case LIBINPUT_EVENT_TOUCH_DOWN:
471 case LIBINPUT_EVENT_TOUCH_UP:
472 touch_event = libinput_event_get_touch_event(event);
473 break;
474 case LIBINPUT_EVENT_POINTER_MOTION:
475 case LIBINPUT_EVENT_POINTER_BUTTON:
476 case LIBINPUT_EVENT_POINTER_MOTION_ABSOLUTE:
477 pointer_event = libinput_event_get_pointer_event(event);
478 break;
479 default:
480 return; /* We don't care about this events */
481 }
482
483 /* We need to read unrotated display dimensions directly from the driver because libinput won't account
484 * for any rotation inside of LVGL */
485 lv_display_t * disp = lv_display_get_default();
486
487 /* ignore more than 2 fingers as it will only confuse LVGL */
488 if(touch_event && (slot = libinput_event_touch_get_slot(touch_event)) > 1)
489 return;
490
491 evt = _create_event(dsc);
492
493 const int32_t hor_res = disp->physical_hor_res > 0 ? disp->physical_hor_res : disp->hor_res;
494 const int32_t ver_res = disp->physical_ver_res > 0 ? disp->physical_ver_res : disp->ver_res;
495
496 switch(type) {
497 case LIBINPUT_EVENT_TOUCH_MOTION:
498 case LIBINPUT_EVENT_TOUCH_DOWN: {
499 lv_point_t point;
500 point.x = (int32_t)LV_CLAMP(INT32_MIN, libinput_event_touch_get_x_transformed(touch_event, hor_res) - disp->offset_x,
501 INT32_MAX);
502 point.y = (int32_t)LV_CLAMP(INT32_MIN, libinput_event_touch_get_y_transformed(touch_event, ver_res) - disp->offset_y,
503 INT32_MAX);
504 if(point.x < 0 || point.x > disp->hor_res || point.y < 0 || point.y > disp->ver_res) {
505 break; /* ignore touches that are out of bounds */
506 }
507 evt->point = point;
508 evt->pressed = LV_INDEV_STATE_PRESSED;
509 dsc->slots[slot].point = evt->point;
510 dsc->slots[slot].pressed = evt->pressed;
511 break;
512 }
513 case LIBINPUT_EVENT_TOUCH_UP:
514 /*
515 * We don't support "multitouch", but libinput does. To make fast typing with two thumbs
516 * on a keyboard feel good, it's necessary to handle two fingers individually. The edge
517 * case here is if you press a key with one finger and then press a second key with another
518 * finger. No matter which finger you release, it will count as the second finger releasing
519 * and ignore the first because LVGL only stores a single (the latest) pressed state.
520 *
521 * To work around this, we detect the case where one finger is released while the other is
522 * still pressed and insert dummy events so that both release events trigger at the correct
523 * position.
524 */
525 if(slot == 0 && dsc->slots[1].pressed == LV_INDEV_STATE_PRESSED) {
526 /* The first finger is released while the second finger is still pressed.
527 * We turn P1 > P2 > R1 > R2 into P1 > P2 > (P1) > R1 > (P2) > R2.
528 */
529
530 /* Inject the dummy press event for the first finger */
531 lv_libinput_event_t * synth_evt = evt;
532 synth_evt->pressed = LV_INDEV_STATE_PRESSED;
533 synth_evt->point = dsc->slots[0].point;
534
535 /* Append the real release event for the first finger */
536 evt = _create_event(dsc);
537 evt->pressed = LV_INDEV_STATE_RELEASED;
538 evt->point = dsc->slots[0].point;
539
540 /* Inject the dummy press event for the second finger */
541 synth_evt = _create_event(dsc);
542 synth_evt->pressed = LV_INDEV_STATE_PRESSED;
543 synth_evt->point = dsc->slots[1].point;
544 }
545 else if(slot == 1 && dsc->slots[0].pressed == LV_INDEV_STATE_PRESSED) {
546 /* The second finger is released while the first finger is still pressed.
547 * We turn P1 > P2 > R2 > R1 into P1 > P2 > R2 > (P1) > R1.
548 */
549
550 /* Append the real release event for the second finger */
551 evt->pressed = LV_INDEV_STATE_RELEASED;
552 evt->point = dsc->slots[1].point;
553
554 /* Inject the dummy press event for the first finger */
555 lv_libinput_event_t * synth_evt = _create_event(dsc);
556 synth_evt->pressed = LV_INDEV_STATE_PRESSED;
557 synth_evt->point = dsc->slots[0].point;
558 }
559 else {
560 evt->pressed = LV_INDEV_STATE_RELEASED;
561 evt->point = dsc->slots[slot].point;
562 }
563
564 dsc->slots[slot].pressed = evt->pressed;
565 break;
566 case LIBINPUT_EVENT_POINTER_MOTION:
567 dsc->pointer_position.x = (int32_t)LV_CLAMP(0, dsc->pointer_position.x + libinput_event_pointer_get_dx(pointer_event),
568 disp->hor_res - 1);
569 dsc->pointer_position.y = (int32_t)LV_CLAMP(0, dsc->pointer_position.y + libinput_event_pointer_get_dy(pointer_event),
570 disp->ver_res - 1);
571 evt->point.x = dsc->pointer_position.x;
572 evt->point.y = dsc->pointer_position.y;
573 evt->pressed = dsc->pointer_button_down;
574 break;
575 case LIBINPUT_EVENT_POINTER_MOTION_ABSOLUTE: {
576 lv_point_t point;
577 point.x = (int32_t)LV_CLAMP(INT32_MIN, libinput_event_pointer_get_absolute_x_transformed(pointer_event,
578 hor_res) - disp->offset_x, INT32_MAX);
579 point.y = (int32_t)LV_CLAMP(INT32_MIN, libinput_event_pointer_get_absolute_y_transformed(pointer_event,
580 ver_res) - disp->offset_y, INT32_MAX);
581 if(point.x < 0 || point.x > disp->hor_res || point.y < 0 || point.y > disp->ver_res) {
582 break; /* ignore pointer events that are out of bounds */
583 }
584 evt->point = point;
585 evt->pressed = dsc->pointer_button_down;
586 break;
587 }
588 case LIBINPUT_EVENT_POINTER_BUTTON: {
589 enum libinput_button_state button_state = libinput_event_pointer_get_button_state(pointer_event);
590 dsc->pointer_button_down = button_state == LIBINPUT_BUTTON_STATE_RELEASED ? LV_INDEV_STATE_RELEASED :
591 LV_INDEV_STATE_PRESSED;
592 evt->point.x = dsc->pointer_position.x;
593 evt->point.y = dsc->pointer_position.y;
594 evt->pressed = dsc->pointer_button_down;
595 }
596 default:
597 break;
598 }
599}

601static void _read_keypad(lv_libinput_t * dsc, struct libinput_event * event)
602{
603 LV_ASSERT(dsc != NULL);
604 LV_ASSERT(event != NULL);
605 struct libinput_event_keyboard * keyboard_event = NULL;
606 enum libinput_event_type type = libinput_event_get_type(event);
607 lv_libinput_event_t * evt = NULL;
608 switch(type) {
609 case LIBINPUT_EVENT_KEYBOARD_KEY:
610 evt = _create_event(dsc);
611 keyboard_event = libinput_event_get_keyboard_event(event);
612 enum libinput_key_state key_state = libinput_event_keyboard_get_key_state(keyboard_event);
613 uint32_t code = libinput_event_keyboard_get_key(keyboard_event);
614#if LV_LIBINPUT_XKB
615 evt->key_val = lv_xkb_process_key(&(dsc->xkb), code, key_state == LIBINPUT_KEY_STATE_PRESSED);
616#else
617 switch(code) {
618 case KEY_BACKSPACE:
619 evt->key_val = LV_KEY_BACKSPACE;
620 break;
621 case KEY_ENTER:
622 evt->key_val = LV_KEY_ENTER;
623 break;
624 case KEY_PREVIOUS:
625 evt->key_val = LV_KEY_PREV;
626 break;
627 case KEY_NEXT:
628 evt->key_val = LV_KEY_NEXT;
629 break;
630 case KEY_UP:
631 evt->key_val = LV_KEY_UP;
632 break;
633 case KEY_LEFT:
634 evt->key_val = LV_KEY_LEFT;
635 break;
636 case KEY_RIGHT:
637 evt->key_val = LV_KEY_RIGHT;
638 break;
639 case KEY_DOWN:
640 evt->key_val = LV_KEY_DOWN;
641 break;
642 case KEY_TAB:
643 evt->key_val = LV_KEY_NEXT;
644 break;
645 case KEY_HOME:
646 evt->key_val = LV_KEY_HOME;
647 break;
648 case KEY_END:
649 evt->key_val = LV_KEY_END;
650 break;
651 case KEY_ESC:
652 evt->key_val = LV_KEY_ESC;
653 break;
654 default:
655 evt->key_val = 0;
656 break;
657 }
658#endif /* LV_LIBINPUT_XKB */
659 if(evt->key_val != 0) {
660 /* Only record button state when actual output is produced to prevent widgets from refreshing */
661 evt->pressed = (key_state == LIBINPUT_KEY_STATE_RELEASED) ? LV_INDEV_STATE_RELEASED : LV_INDEV_STATE_PRESSED;
662
663 // just release the key immediately after it got pressed.
664 // but don't handle special keys where holding a key makes sense
665 if(evt->key_val != LV_KEY_BACKSPACE &&
666 evt->key_val != LV_KEY_UP &&
667 evt->key_val != LV_KEY_LEFT &&
668 evt->key_val != LV_KEY_RIGHT &&
669 evt->key_val != LV_KEY_DOWN &&

696static void _delete(lv_libinput_t * dsc)
697{
698 if(dsc->running) {
699 dsc->deinit = true;
700
701 /* Give worker thread a whole second to quit */
702 for(int i = 0; i < 100; i++) {
703 if(!dsc->deinit)
704 break;
705 usleep(10000);
706 }
707
708 if(dsc->deinit) {
709 LV_LOG_ERROR("libinput worker thread did not quit in time, cancelling it");
710 pthread_cancel(dsc->worker_thread);
711 }
712
713 pthread_join(dsc->worker_thread, NULL);
714 pthread_mutex_destroy(&dsc->event_lock);
715 }
716
717 if(dsc->libinput_device) {
718 libinput_path_remove_device(dsc->libinput_device);
719 libinput_device_unref(dsc->libinput_device);
720 }
721
722 if(dsc->libinput_context) {
723 libinput_unref(dsc->libinput_context);
724 }
725
726#if LV_LIBINPUT_XKB
727 lv_xkb_deinit(&(dsc->xkb));
728#endif /* LV_LIBINPUT_XKB */
729
730 lv_free(dsc);
731}
