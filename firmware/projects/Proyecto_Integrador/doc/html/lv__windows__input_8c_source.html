75lv_indev_t * lv_windows_acquire_pointer_indev(lv_display_t * display)
76{
77 LV_CHECK_ARG(display != NULL, return NULL);
78 HWND window_handle = lv_windows_get_display_window_handle(display);
79 if(!window_handle) {
80 return NULL;
81 }
82
83 lv_windows_window_context_t * context = lv_windows_get_window_context(
84 window_handle);
85 if(!context) {
86 return NULL;
87 }
88
89 if(!context->pointer.indev) {
90 context->pointer.state = LV_INDEV_STATE_RELEASED;
91 context->pointer.point.x = 0;
92 context->pointer.point.y = 0;
93
94 context->pointer.indev = lv_indev_create();
95 if(context->pointer.indev) {
96 lv_indev_set_type(
97 context->pointer.indev,
98 LV_INDEV_TYPE_POINTER);
99 lv_indev_set_read_cb(
100 context->pointer.indev,
101 lv_windows_pointer_driver_read_callback);
102 lv_indev_set_display(
103 context->pointer.indev,
104 context->display_device_object);
105 lv_indev_add_event_cb(
106 context->pointer.indev,
107 lv_windows_release_pointer_device_event_callback,
108 LV_EVENT_DELETE,
109 context->pointer.indev);
110 lv_indev_set_group(
111 context->pointer.indev,
112 lv_group_get_default());
113 }
114 }
115
116 return context->pointer.indev;
117}

292bool lv_windows_pointer_device_window_message_handler(
293 HWND hWnd,
294 UINT uMsg,
295 WPARAM wParam,
296 LPARAM lParam,
297 LRESULT * plResult)
298{
299 switch(uMsg) {
300 case WM_MOUSEMOVE: {
301 lv_windows_window_context_t * context = (lv_windows_window_context_t *)(
302 lv_windows_get_window_context(hWnd));
303 if(context) {
304 int32_t hor_res = lv_display_get_horizontal_resolution(
305 context->display_device_object);
306 int32_t ver_res = lv_display_get_vertical_resolution(
307 context->display_device_object);
308
309 context->pointer.point.x = lv_windows_zoom_to_logical(
310 GET_X_LPARAM(lParam),
311 context->zoom_level);
312 context->pointer.point.y = lv_windows_zoom_to_logical(
313 GET_Y_LPARAM(lParam),
314 context->zoom_level);
315 if(context->simulator_mode) {
316 context->pointer.point.x = lv_windows_dpi_to_logical(
317 context->pointer.point.x,
318 context->window_dpi);
319 context->pointer.point.y = lv_windows_dpi_to_logical(
320 context->pointer.point.y,
321 context->window_dpi);
322 }
323 if(context->pointer.point.x < 0) {
324 context->pointer.point.x = 0;
325 }
326 if(context->pointer.point.x > hor_res - 1) {
327 context->pointer.point.x = hor_res - 1;
328 }
329 if(context->pointer.point.y < 0) {
330 context->pointer.point.y = 0;
331 }
332 if(context->pointer.point.y > ver_res - 1) {
333 context->pointer.point.y = ver_res - 1;
334 }
335 }
336
337 break;
338 }
339 case WM_LBUTTONDOWN:
340 case WM_LBUTTONUP: {
341 lv_windows_window_context_t * context = (lv_windows_window_context_t *)(
342 lv_windows_get_window_context(hWnd));
343 if(context) {
344 context->pointer.state = (
345 uMsg == WM_LBUTTONDOWN
346 ? LV_INDEV_STATE_PRESSED
347 : LV_INDEV_STATE_RELEASED);
348 }
349
350 break;
351 }
352 case WM_TOUCH: {
353 lv_windows_window_context_t * context = (lv_windows_window_context_t *)(
354 lv_windows_get_window_context(hWnd));
355 if(context) {
356 UINT input_count = LOWORD(wParam);
357 HTOUCHINPUT touch_input_handle = (HTOUCHINPUT)(lParam);
358
359 PTOUCHINPUT inputs = malloc(input_count * sizeof(TOUCHINPUT));
360 if(inputs) {
361 if(lv_windows_get_touch_input_info(
362 touch_input_handle,
363 input_count,
364 inputs,
365 sizeof(TOUCHINPUT))) {
366 for(UINT i = 0; i < input_count; ++i) {
367 POINT Point;
368 Point.x = TOUCH_COORD_TO_PIXEL(inputs[i].x);
369 Point.y = TOUCH_COORD_TO_PIXEL(inputs[i].y);
370 if(!ScreenToClient(hWnd, &Point)) {
371 continue;
372 }
373
374 context->pointer.point.x = lv_windows_zoom_to_logical(
375 Point.x,
376 context->zoom_level);
377 context->pointer.point.y = lv_windows_zoom_to_logical(
378 Point.y,
379 context->zoom_level);
380 if(context->simulator_mode) {
381 context->pointer.point.x = lv_windows_dpi_to_logical(
382 context->pointer.point.x,
383 context->window_dpi);
384 context->pointer.point.y = lv_windows_dpi_to_logical(
385 context->pointer.point.y,
386 context->window_dpi);
387 }
388
389 DWORD MousePressedMask =
390 TOUCHEVENTF_MOVE | TOUCHEVENTF_DOWN;
391
392 context->pointer.state = (
393 inputs[i].dwFlags & MousePressedMask
394 ? LV_INDEV_STATE_PRESSED
395 : LV_INDEV_STATE_RELEASED);
396 }
397 }
398
399 free(inputs);
400 }
401
402 lv_windows_close_touch_input_handle(touch_input_handle);
403 }
404
405 break;
406 }
407 default:
408 // Not Handled
409 return false;
410 }
411
412 // Handled
413 *plResult = 0;
414 return true;
415}

418 lv_indev_t * indev,
419 lv_indev_data_t * data)
420{

557bool lv_windows_keypad_device_window_message_handler(
558 HWND hWnd,
559 UINT uMsg,
560 WPARAM wParam,
561 LPARAM lParam,
562 LRESULT * plResult)
563{
564 LV_UNUSED(lParam);
565
566 switch(uMsg) {
567 case WM_KEYDOWN:
568 case WM_KEYUP: {
569 lv_windows_window_context_t * context = (lv_windows_window_context_t *)(
570 lv_windows_get_window_context(hWnd));
571 if(context) {
572 bool skip_translation = false;
573 uint32_t translated_key = 0;
574
575 switch(wParam) {
576 case VK_UP:
577 translated_key = LV_KEY_UP;
578 break;
579 case VK_DOWN:
580 translated_key = LV_KEY_DOWN;
581 break;
582 case VK_LEFT:
583 translated_key = LV_KEY_LEFT;
584 break;
585 case VK_RIGHT:
586 translated_key = LV_KEY_RIGHT;
587 break;
588 case VK_ESCAPE:
589 translated_key = LV_KEY_ESC;
590 break;
591 case VK_DELETE:
592 translated_key = LV_KEY_DEL;
593 break;
594 case VK_BACK:
595 translated_key = LV_KEY_BACKSPACE;
596 break;
597 case VK_RETURN:
598 translated_key = LV_KEY_ENTER;
599 break;
600 case VK_TAB:
601 case VK_NEXT:
602 translated_key = LV_KEY_NEXT;
603 break;
604 case VK_PRIOR:
605 translated_key = LV_KEY_PREV;
606 break;
607 case VK_HOME:
608 translated_key = LV_KEY_HOME;
609 break;
610 case VK_END:
611 translated_key = LV_KEY_END;
612 break;
613 default:
614 skip_translation = true;
615 break;
616 }
617
618 if(!skip_translation) {
619 lv_windows_push_key_to_keyboard_queue(
620 context,
621 translated_key,
622 ((uMsg == WM_KEYUP)
623 ? LV_INDEV_STATE_RELEASED
624 : LV_INDEV_STATE_PRESSED));
625 }
626 }
627
628 break;
629 }
630 case WM_CHAR: {
631 lv_windows_window_context_t * context = (lv_windows_window_context_t *)(
632 lv_windows_get_window_context(hWnd));
633 if(context) {
634 uint16_t raw_code_point = (uint16_t)(wParam);
635
636 if(raw_code_point >= 0x20 && raw_code_point != 0x7F) {
637 if(IS_HIGH_SURROGATE(raw_code_point)) {
638 context->keypad.utf16_high_surrogate = raw_code_point;
639 }
640 else if(IS_LOW_SURROGATE(raw_code_point)) {
641 context->keypad.utf16_low_surrogate = raw_code_point;
642 }
643
644 uint32_t code_point = raw_code_point;
645
646 if(context->keypad.utf16_high_surrogate &&
647 context->keypad.utf16_low_surrogate) {
648 uint16_t high_surrogate =
649 context->keypad.utf16_high_surrogate;
650 uint16_t low_surrogate =
651 context->keypad.utf16_low_surrogate;
652
653 code_point = (low_surrogate & 0x03FF);
654 code_point += (((high_surrogate & 0x03FF) + 0x40) << 10);
655
656 context->keypad.utf16_high_surrogate = 0;
657 context->keypad.utf16_low_surrogate = 0;
658 }
659
660 uint32_t lvgl_code_point =
661 lv_text_unicode_to_encoded(code_point);
662
663 lv_windows_push_key_to_keyboard_queue(
664 context,
665 lvgl_code_point,
666 LV_INDEV_STATE_PRESSED);
667 lv_windows_push_key_to_keyboard_queue(
668 context,
669 lvgl_code_point,
670 LV_INDEV_STATE_RELEASED);
671 }
672 }
673
674 break;
675 }
676 case WM_IME_SETCONTEXT: {
677 if(wParam == TRUE) {
678 HIMC imm_context_handle = lv_windows_imm_get_context(hWnd);
679 if(imm_context_handle) {
680 lv_windows_imm_associate_context(
681 hWnd,
682 imm_context_handle);
683 lv_windows_imm_release_context(
684 hWnd,
685 imm_context_handle);
686 }
687 }
688
689 *plResult = DefWindowProcW(hWnd, uMsg, wParam, wParam);
690 break;
691 }
692 case WM_IME_STARTCOMPOSITION: {
693 HIMC imm_context_handle = lv_windows_imm_get_context(hWnd);
694 if(imm_context_handle) {
695 lv_obj_t * textarea_object = NULL;
696 lv_obj_t * focused_object = lv_group_get_focused(
697 lv_group_get_default());
698 if(focused_object) {
699 const lv_obj_class_t * object_class = lv_obj_get_class(
700 focused_object);
701
702 if(object_class == &lv_textarea_class) {
703 textarea_object = focused_object;
704 }
705 else if(object_class == &lv_keyboard_class) {
706 textarea_object = lv_keyboard_get_textarea(focused_object);
707 }
708 }
709
710 COMPOSITIONFORM composition_form;
711 composition_form.dwStyle = CFS_POINT;
712 composition_form.ptCurrentPos.x = 0;
713 composition_form.ptCurrentPos.y = 0;
714
715 if(textarea_object) {
716 lv_textarea_t * textarea = (lv_textarea_t *)(textarea_object);
717 lv_obj_t * label_object = lv_textarea_get_label(textarea_object);
718
719 composition_form.ptCurrentPos.x =
720 label_object->coords.x1 + textarea->cursor.area.x1;
721 composition_form.ptCurrentPos.y =
722 label_object->coords.y1 + textarea->cursor.area.y1;
723 }
724
725 lv_windows_imm_set_composition_window(
726 imm_context_handle,
727 &composition_form);
728 lv_windows_imm_release_context(
729 hWnd,
730 imm_context_handle);
731 }
732
733 *plResult = DefWindowProcW(hWnd, uMsg, wParam, wParam);
734 break;
735 }
736 default:
737 // Not Handled
738 return false;
739 }
740
741 // Handled
742 *plResult = 0;
743 return true;
744}
