53void lv_draw_eve_label(lv_draw_task_t * t, const lv_draw_label_dsc_t * dsc, const lv_area_t * coords)
54{
55 if(dsc->opa <= LV_OPA_MIN) return;
56
57 lv_eve_scissor(t->clip_area.x1, t->clip_area.y1, t->clip_area.x2, t->clip_area.y2);
58 lv_eve_save_context();
59 lv_eve_primitive(LV_EVE_PRIMITIVE_BITMAPS);
60 lv_draw_label_iterate_characters(t, dsc, coords, lv_draw_eve_letter_cb);
61 lv_eve_restore_context();
62}

64bool lv_draw_eve_label_font_check(const lv_font_t * font)
65{
66 if(font->get_glyph_bitmap != lv_font_get_bitmap_fmt_txt) {
67 LV_LOG_WARN("lv_draw_eve can only render static fonts for now.");
68 return false;
69 }
70
71 const lv_font_fmt_txt_dsc_t * font_dsc = font->dsc;
72
73 /* Only 4 bpp is supported for now. Support for 1 and 8 bpp can be added. (EVE_L1, EVE_L8) */
74 if(font_dsc->bpp != 4) {
75 LV_LOG_WARN("lv_draw_eve can only render static fonts for now.");
76 return false;
77 }
78
79 return true;
80}

82uint32_t lv_draw_eve_label_upload_glyph(bool burst_is_active, const lv_font_fmt_txt_dsc_t * font_dsc,
83 uint32_t gid_index)
84{
85 const lv_font_fmt_txt_glyph_dsc_t * glyph_dsc = &font_dsc->glyph_dsc[gid_index];
86 const uint8_t * glyph_bitmap = &font_dsc->glyph_bitmap[glyph_dsc->bitmap_index];
87
88 uint16_t g_box_w = glyph_dsc->box_w;
89 uint16_t g_box_h = glyph_dsc->box_h;
90
91 uint16_t g_aligned_stride = (g_box_w + 1) / 2;
92
93 uint32_t glyph_ramg_size = g_aligned_stride * g_box_h;
94
95 uint32_t ramg_addr;
96 uintptr_t glyph_ramg_key = (uintptr_t) glyph_bitmap;
97 bool font_is_loaded = lv_draw_eve_ramg_get_addr(&ramg_addr, glyph_ramg_key, glyph_ramg_size, 1);
98
99 /* If the font is not yet loaded in ramG, load it */
100 if(!font_is_loaded && ramg_addr != LV_DRAW_EVE_RAMG_OUT_OF_RAMG) {
101 if(burst_is_active) {
102 EVE_end_cmd_burst();
103 }
104
105 uint8_t glyph_bitmap_stride_align = font_dsc->stride;
106 font_bitmap_to_ramg(ramg_addr, glyph_bitmap, g_box_w, g_box_h, glyph_bitmap_stride_align);
107
108 if(burst_is_active) {
109 EVE_start_cmd_burst();
110 }
111 }
112
113 return ramg_addr;
114}

120static void lv_draw_eve_letter_cb(lv_draw_task_t * t, lv_draw_glyph_dsc_t * glyph_draw_dsc,
121 lv_draw_fill_dsc_t * fill_draw_dsc, const lv_area_t * fill_area)
122{
123
124 if(fill_draw_dsc && fill_area) {
125 /* draw UNDERLINE and STRIKETHROUGH */
126 lv_eve_draw_rect_simple(fill_area->x1, fill_area->y1, fill_area->x2, fill_area->y2, 0);
127 }
128
129 if(glyph_draw_dsc == NULL)
130 return; /* Important */
131
132 const lv_font_t * font = glyph_draw_dsc->g->resolved_font;
133
134 if(!lv_draw_eve_label_font_check(font)) {
135 return;
136 }
137
138 const lv_font_fmt_txt_dsc_t * font_dsc = font->dsc;
139 uint32_t gid_index = glyph_draw_dsc->g->gid.index;
140 const lv_font_fmt_txt_glyph_dsc_t * glyph_dsc = &font_dsc->glyph_dsc[gid_index];
141
142 uint16_t g_box_w = glyph_dsc->box_w;
143 uint16_t g_box_h = glyph_dsc->box_h;
144
145 uint16_t g_aligned_stride = (g_box_w + 1) / 2;
146
147 uint8_t bpp_eve = EVE_L4;
148
149 uint32_t ramg_addr = lv_draw_eve_label_upload_glyph(true, font_dsc, gid_index);
150 if(ramg_addr == LV_DRAW_EVE_RAMG_OUT_OF_RAMG) {
151 LV_LOG_WARN("Could not load glyph because space could not be allocated in RAM_G.");
152 return;
153 }
154
155 lv_eve_color_opa(glyph_draw_dsc->opa);
156 lv_eve_color(glyph_draw_dsc->color);
157
158 lv_eve_bitmap_source(ramg_addr);
159
160 lv_eve_bitmap_size(EVE_NEAREST, EVE_BORDER, EVE_BORDER, g_box_w, g_box_h);
161 lv_eve_bitmap_layout(bpp_eve, g_aligned_stride, g_box_h);
162
163 lv_eve_vertex_2f(glyph_draw_dsc->letter_coords->x1, glyph_draw_dsc->letter_coords->y1);
164}

166static void font_bitmap_to_ramg(uint32_t addr, const uint8_t * src, uint32_t width,
167 uint32_t height, uint8_t src_stride_align)
168{
169 uint32_t stride = (width + 1) / 2;
170
171 if(src_stride_align == 1 || (src_stride_align == 0 && width % 2 == 0)) {
172 uint32_t size = stride * height;
173 EVE_memWrite_flash_buffer(addr, src, size);
174 return;
175 }
176
177 if(src_stride_align > 0) {
178 uint32_t src_stride = LV_ALIGN_UP(stride, src_stride_align);
179 for(uint32_t y = 0; y < height; y++) {
180 EVE_memWrite_sram_buffer(addr, src, stride);
181 addr += stride;
182 src += src_stride;
183 }
184 return;
185 }
186
187 uint8_t * row_buf = lv_malloc(stride);
188 LV_ASSERT_MALLOC(row_buf);
189
190 uint32_t src_i = 0;
191 uint8_t nibble_1;
192 uint8_t nibble_2;
193 uint8_t key = 0;
194
195 /* Iterate through each row of the bitmap*/
196 for(uint32_t y = 0; y < height; y++) {
197 /* Iterate through each byte of the row*/
198 uint32_t row_i;
199 for(row_i = 0; row_i < (width / 2); ++row_i) {
200 /*Get the two nibbles from the current byte*/
201 if(key == 0) {
202 nibble_1 = GET_NIBBLE_1(src[src_i]);
203 nibble_2 = GET_NIBBLE_2(src[src_i]);
204 }
205 else {
206 nibble_1 = GET_NIBBLE_2(src[src_i - 1]);
207 nibble_2 = GET_NIBBLE_1(src[src_i]);
208 }
209
210 /*Combine the nibbles and assign the result to the output byte*/
211 row_buf[row_i] = (nibble_1 << 4) | nibble_2;
212
213 src_i++;
214 }
215
216 /*process the last remaining nibble*/
217 row_buf[row_i] =
218 (key == 0) ?
219 (GET_NIBBLE_1(src[src_i])) << 4 | 0x0 : (GET_NIBBLE_2(src[src_i - 1])) << 4 | 0x0;
220 key = (key == 0) ? 1 : 0;
221 src_i += (key == 1) ? 1 : 0;
222
223 EVE_memWrite_sram_buffer(addr, row_buf, stride);
224 addr += stride;
225 }
226
227 lv_free(row_buf);
228}
