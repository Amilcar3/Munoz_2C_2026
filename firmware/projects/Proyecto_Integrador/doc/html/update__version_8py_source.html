80 def applyVersion(self, version: Version):
81 with open(self.path, 'r', encoding='utf-8') as file:
82 lines = file.readlines()
83
84 occurrences = 0
85 for i, line in enumerate(lines):
86 line_with_version = self.applyVersionToLine(line, version)
87 if line_with_version:
88 lines[i] = line_with_version
89 occurrences += 1
90
91 # not perfect, but will catch obvious pitfalls
92 if occurrences != self.expected_occurrences:
93 raise Exception(f"Replaced {occurrences} line(s) in {self.path_relative}, "
94 f"expected {self.expected_occurrences}")
95
96 with open(self.path, 'w', encoding='utf-8', newline='') as file:
97 file.writelines(lines)
