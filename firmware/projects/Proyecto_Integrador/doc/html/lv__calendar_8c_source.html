156void lv_calendar_set_month_shown(lv_obj_t * obj, uint32_t year, uint32_t month)
157{
158 LV_CHECK_OBJ(obj, MY_CLASS, return);
159 lv_calendar_t * calendar = (lv_calendar_t *)obj;
160
161 /*Don't return if the new value is the same, as this function is also
162 *used the update the calendar e.g. when switching to Chinese mode*/
163
164 calendar->showed_date.year = year;
165 calendar->showed_date.month = month;
166 calendar->showed_date.day = 1;
167
168 lv_calendar_date_t d;
169 d.year = calendar->showed_date.year;
170 d.month = calendar->showed_date.month;
171 d.day = calendar->showed_date.day;
172
173 uint32_t i;
174
175 /*Remove the disabled state but revert it for day names*/
176 lv_buttonmatrix_clear_button_ctrl_all(calendar->btnm, LV_BUTTONMATRIX_CTRL_DISABLED);
177 for(i = 0; i < 7; i++) {
178 lv_buttonmatrix_set_button_ctrl(calendar->btnm, i, LV_BUTTONMATRIX_CTRL_DISABLED);
179 }
180
181 uint8_t act_mo_len = get_month_length(d.year, d.month);
182 uint8_t day_first = get_day_of_week(d.year, d.month, 1);
183 uint8_t c;
184
185#if LV_USE_CALENDAR_CHINESE
186 lv_calendar_date_t gregorian_time;
187 gregorian_time = d;
188#endif
189
190 for(i = day_first, c = 1; i < act_mo_len + day_first; i++, c++) {
191#if LV_USE_CALENDAR_CHINESE
192 if(calendar->use_chinese_calendar) {
193 gregorian_time.day = c;
194 chinese_calendar_set_day_name(obj, i, c, &gregorian_time);
195 }
196 else
197#endif
198 lv_snprintf(calendar->nums[i], sizeof(calendar->nums[0]), "%d", c);
199 }
200
201 uint8_t prev_mo_len = get_month_length(d.year, d.month - 1);
202
203#if LV_USE_CALENDAR_CHINESE
204 gregorian_time = gregorian_get_last_month_time(&d);
205#endif
206
207 for(i = 0, c = prev_mo_len - day_first + 1; i < day_first; i++, c++) {
208#if LV_USE_CALENDAR_CHINESE
209 if(calendar->use_chinese_calendar) {
210 gregorian_time.day = c;
211 chinese_calendar_set_day_name(obj, i, c, &gregorian_time);
212 }
213 else
214#endif
215 lv_snprintf(calendar->nums[i], sizeof(calendar->nums[0]), "%d", c);
216 lv_buttonmatrix_set_button_ctrl(calendar->btnm, i + 7, LV_BUTTONMATRIX_CTRL_DISABLED);
217 }
218
219#if LV_USE_CALENDAR_CHINESE
220 gregorian_time = gregorian_get_next_month_time(&d);
221#endif
222
223 for(i = day_first + act_mo_len, c = 1; i < 6 * 7; i++, c++) {
224#if LV_USE_CALENDAR_CHINESE
225 if(calendar->use_chinese_calendar) {
226 gregorian_time.day = c;
227 chinese_calendar_set_day_name(obj, i, c, &gregorian_time);
228 }
229 else
230#endif
231 lv_snprintf(calendar->nums[i], sizeof(calendar->nums[0]), "%d", c);
232 lv_buttonmatrix_set_button_ctrl(calendar->btnm, i + 7, LV_BUTTONMATRIX_CTRL_DISABLED);
233 }
234
235 highlight_update(obj);
236
237 /*Reset the focused button if the days changes*/
238 if(lv_buttonmatrix_get_selected_button(calendar->btnm) != LV_BUTTONMATRIX_BUTTON_NONE) {
239 lv_buttonmatrix_set_selected_button(calendar->btnm, day_first + 7);
240 }
241
242 lv_obj_invalidate(obj);
243
244 /* The children of the calendar are probably headers.
245 * Notify them to let the headers updated to the new date*/
246 uint32_t child_cnt = lv_obj_get_child_count(obj);
247 for(i = 0; i < child_cnt; i++) {
248 lv_obj_t * child = lv_obj_get_child(obj, i);
249 if(child == calendar->btnm) continue;
250 lv_obj_send_event(child, LV_EVENT_VALUE_CHANGED, obj);
251 }
252}

312lv_result_t lv_calendar_get_pressed_date(const lv_obj_t * obj, lv_calendar_date_t * date)
313{
314 LV_CHECK_OBJ(obj, MY_CLASS, return LV_RESULT_INVALID);
315 LV_CHECK_ARG(date != NULL, return LV_RESULT_INVALID);
316 lv_calendar_t * calendar = (lv_calendar_t *)obj;
317
318 uint32_t d = lv_buttonmatrix_get_selected_button(calendar->btnm);
319 if(d == LV_BUTTONMATRIX_BUTTON_NONE) {
320 date->year = 0;
321 date->month = 0;
322 date->day = 0;
323 return LV_RESULT_INVALID;
324 }
325
326 const char * txt = lv_buttonmatrix_get_button_text(calendar->btnm, lv_buttonmatrix_get_selected_button(calendar->btnm));
327
328 uint32_t day = 0;
329 uint32_t i = 0;
330 while(txt[i] >= '0' && txt[i] <= '9') {
331 day = day * 10 + (txt[i] - '0');
332 i++;
333 }
334 date->day = day;
335
336 date->year = calendar->showed_date.year;
337 date->month = calendar->showed_date.month;
338
339 return LV_RESULT_OK;
340}

346static void lv_calendar_constructor(const lv_obj_class_t * class_p, lv_obj_t * obj)
347{
348 LV_UNUSED(class_p);
349 lv_calendar_t * calendar = (lv_calendar_t *)obj;
350 LV_ASSERT(obj != NULL);
351
352 lv_memzero(calendar->nums, sizeof(calendar->nums));
353 uint8_t i;
354 uint8_t j = 0;
355 for(i = 0; i < 8 * 7; i++) {
356 /*Every 8th string is "\n"*/
357 if(i != 0 && (i + 1) % 8 == 0) {
358 calendar->map[i] = "\n";
359 }
360 else if(i < 7) {
361 calendar->map[i] = day_names_def[i];
362 }
363 else {
364 calendar->nums[j][0] = 'x';
365 calendar->map[i] = calendar->nums[j];
366 j++;
367 }
368 }
369 calendar->map[8 * 7 - 1] = "";
370
371 calendar->btnm = lv_buttonmatrix_create(obj);
372 lv_buttonmatrix_set_map(calendar->btnm, calendar->map);
373 lv_buttonmatrix_set_button_ctrl_all(calendar->btnm, LV_BUTTONMATRIX_CTRL_CLICK_TRIG | LV_BUTTONMATRIX_CTRL_NO_REPEAT);
374 lv_obj_add_event_cb(calendar->btnm, draw_task_added_event_cb, LV_EVENT_DRAW_TASK_ADDED, NULL);
375 lv_obj_set_width(calendar->btnm, lv_pct(100));
376
377 lv_obj_set_event_bubble(calendar->btnm, true);
378 lv_obj_set_send_draw_task_events(calendar->btnm, true);
379
380 lv_obj_set_flex_flow(obj, LV_FLEX_FLOW_COLUMN);
381 lv_obj_set_flex_grow(calendar->btnm, 1);
382
383 lv_obj_set_style_text_align(calendar->btnm, LV_TEXT_ALIGN_CENTER, LV_PART_MAIN);
384
385 lv_calendar_set_month_shown(obj, 2024, 1);
386 lv_calendar_set_today_date(obj, 2024, 1, 1);
387}

441static uint8_t get_month_length(int32_t year, int32_t month)
442{
443 month--;
444 if(month < 0) {
445 year--; /*Already in the previous year (won't be less than -12 to skip a whole year)*/
446 month = 12 + month; /*`month` is negative, the result will be < 12*/
447 }
448 if(month >= 12) {
449 year++;
450 month -= 12;
451 }
452
453 /*month == 1 is february*/
454 return (month == 1) ? (28 + is_leap_year(year)) : 31 - month % 7 % 2;
455}

462static uint8_t is_leap_year(uint32_t year)
463{
464 return (year % 4) || ((year % 100 == 0) && (year % 400)) ? 0 : 1;
465}

474static uint8_t get_day_of_week(uint32_t year, uint32_t month, uint32_t day)
475{
476 uint32_t a = month < 3 ? 1 : 0;
477 uint32_t b = year - a;
478
479#if LV_CALENDAR_WEEK_STARTS_MONDAY
480 uint32_t day_of_week = (day + (31 * (month - 2 + 12 * a) / 12) + b + (b / 4) - (b / 100) + (b / 400) - 1) % 7;
481#else
482 uint32_t day_of_week = (day + (31 * (month - 2 + 12 * a) / 12) + b + (b / 4) - (b / 100) + (b / 400)) % 7;
483#endif
484
485 return day_of_week ;
486}

488static void highlight_update(lv_obj_t * obj)
489{
490 LV_ASSERT(obj != NULL);
491 lv_calendar_t * calendar = (lv_calendar_t *)obj;
492 uint32_t i;
493
494 /*Clear all kind of selection*/
495 lv_buttonmatrix_clear_button_ctrl_all(calendar->btnm, LV_CALENDAR_CTRL_TODAY | LV_CALENDAR_CTRL_HIGHLIGHT);
496
497 uint8_t day_first = get_day_of_week(calendar->showed_date.year, calendar->showed_date.month, 1);
498 if(calendar->highlighted_dates) {
499 for(i = 0; i < calendar->highlighted_dates_num; i++) {
500 if(calendar->highlighted_dates[i].year == calendar->showed_date.year &&
501 calendar->highlighted_dates[i].month == calendar->showed_date.month) {
502 lv_buttonmatrix_set_button_ctrl(calendar->btnm, calendar->highlighted_dates[i].day - 1 + day_first + 7,
503 LV_CALENDAR_CTRL_HIGHLIGHT);
504 }
505 }
506 }
507
508 if(calendar->showed_date.year == calendar->today.year && calendar->showed_date.month == calendar->today.month) {
509 lv_buttonmatrix_set_button_ctrl(calendar->btnm, calendar->today.day - 1 + day_first + 7, LV_CALENDAR_CTRL_TODAY);
510 }
511}
