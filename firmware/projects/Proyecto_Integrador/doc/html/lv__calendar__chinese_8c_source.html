176const char * lv_calendar_get_day_name(lv_calendar_date_t * gregorian)
177{
178 LV_CHECK_ARG(gregorian != NULL, return NULL);
179
180 uint16_t i, len;
181 lv_calendar_chinese_t chinese_calendar;
182 const char * solar_term_name;
183
184 if(gregorian->year > 2099 || gregorian->year < 1901)
185 return NULL;
186
187 lv_calendar_gregorian_to_chinese(gregorian, &chinese_calendar);
188
189 len = sizeof(festivals_base_chinese) / sizeof(lv_calendar_festival_t);
190 for(i = 0; i < len; i++) {
191 if((chinese_calendar.today.month == festivals_base_chinese[i].month) &&
192 (chinese_calendar.today.day == festivals_base_chinese[i].day) &&
193 chinese_calendar.leep_month == false) {
194 if(chinese_calendar.today.month == 12 && chinese_calendar.today.day == 29) {
195 if((calendar_chinese_table[chinese_calendar.today.year - 1901] & 0xf00000) != 0) {
196 if((calendar_chinese_table[chinese_calendar.today.year - 1901] & 0x000080) == 0) {
197 return festivals_base_chinese[i].festival_name;
198 }
199 }
200 else {
201 if((calendar_chinese_table[chinese_calendar.today.year - 1901] & 0x000100) == 0) {
202 return festivals_base_chinese[i].festival_name;
203 }
204 }
205 }
206 else {
207 return festivals_base_chinese[i].festival_name;
208 }
209 }
210 }
211
212 len = sizeof(festivals_base_gregorian) / sizeof(lv_calendar_festival_t);
213 for(i = 0; i < len; i++) {
214 if((gregorian->month == festivals_base_gregorian[i].month) &&
215 (gregorian->day == festivals_base_gregorian[i].day))
216 return festivals_base_gregorian[i].festival_name;
217 }
218
219 solar_term_name = check_solar_term(gregorian->year, gregorian->month, gregorian->day);
220 if(solar_term_name != NULL) {
221 return solar_term_name;
222 }
223
224 if(chinese_calendar.today.day == 1) {
225 if(chinese_calendar.leep_month == false)
226 return chinese_calendar_month_name[chinese_calendar.today.month - 1];
227 else {
228 return chinese_calendar_leep_month_name[chinese_calendar.today.month - 1];
229 }

235void lv_calendar_gregorian_to_chinese(lv_calendar_date_t * gregorian_time, lv_calendar_chinese_t * chinese_time)
236{
237 LV_CHECK_ARG(gregorian_time != NULL, return);
238 LV_CHECK_ARG(chinese_time != NULL, return);
239
240 uint16_t year = gregorian_time->year;
241 uint8_t month = gregorian_time->month;
242 uint8_t day = gregorian_time->day;
243
244 /*Record the number of days between the Spring Festival
245 and the New Year's Day of that year.*/
246 uint16_t by_spring;
247
248 /*Record the number of days from the gregorian calendar
249 to the New Year's Day of that year.*/
250 uint16_t by_gregorian;
251
252 /*Record the number of days in that month*/
253 uint8_t days_per_month;
254
255 /*Record from which month the calculation starts.*/
256 uint8_t index;
257
258 bool leep_month;
259
260 if(year < 1901 || year > 2099) {
261 chinese_time->leep_month = 0;
262 chinese_time->today.year = 2000;
263 chinese_time->today.month = 1;
264 chinese_time->today.day = 1;
265 return;
266 }
267
268 if(((calendar_chinese_table[year - 1901] & 0x0060) >> 5) == 1)
269 by_spring = (calendar_chinese_table[year - 1901] & 0x001F) - 1;
270 else
271 by_spring = (calendar_chinese_table[year - 1901] & 0x001F) - 1 + 31;
272
273 by_gregorian = month_total_day[month - 1] + day - 1;
274
275 if((!(year % 4)) && (month > 2))
276 by_gregorian++;
277
278 if(by_gregorian >= by_spring) {/*Gregorian calendar days after the Spring Festival*/
279 by_gregorian -= by_spring;
280 month = 1;
281 index = 1;
282 leep_month = false;
283
284 if((calendar_chinese_table[year - 1901] & (0x80000 >> (index - 1))) == 0)
285 days_per_month = 29;
286 else
287 days_per_month = 30;
288
289 while(by_gregorian >= days_per_month) {
290 by_gregorian -= days_per_month;
291 index++;
292
293 if(month == ((calendar_chinese_table[year - 1901] & 0xF00000) >> 20)) {
294 leep_month = !leep_month;
295 if(leep_month == false)
296 month++;
297 }
298 else
299 month++;
300
301 if((calendar_chinese_table[year - 1901] & (0x80000 >> (index - 1))) == 0)
302 days_per_month = 29;
303 else
304 days_per_month = 30;
305 }
306 day = by_gregorian + 1;
307 }
308 else {/*Solar day before the Spring Festival*/
309 by_spring -= by_gregorian;
310 year--;
311 month = 12;
312 if(((calendar_chinese_table[year - 1901] & 0xF00000) >> 20) == 0)
313 index = 12;
314 else
315 index = 13;
316 leep_month = false;
317
318 if((calendar_chinese_table[year - 1901] & (0x80000 >> (index - 1))) == 0)
319 days_per_month = 29;
320 else
321 days_per_month = 30;
322
323 while(by_spring > days_per_month) {
324 by_spring -= days_per_month;
325 index--;
326
327 if(leep_month == false)
328 month--;
329
330 if(month == ((calendar_chinese_table[year - 1901] & 0xF00000) >> 20))
331 leep_month = !leep_month;
332
333 if((calendar_chinese_table[year - 1901] & (0x80000 >> (index - 1))) == 0)
334 days_per_month = 29;
335 else
336 days_per_month = 30;
337 }
338
339 day = days_per_month - by_spring + 1;
340 }
341 chinese_time->today.day = day;
342 chinese_time->today.month = month;
343 chinese_time->today.year = year;
344 chinese_time->leep_month = leep_month;
345}

358static const char * check_solar_term(uint16_t year, uint8_t month, uint8_t day)
359{
360 uint8_t i;
361 uint8_t offsets[24];
362 uint8_t term_month, term_day;
363 uint16_t date_value = month * 100 + day;
364 uint8_t char_count_len = 2;
365 uint8_t range_end_num = 24;
366
367 static const uint8_t solar_terms_min_day[24] = {
368 4, 19, 3, 18, 4, 19, 4, 19, 4, 20, 4, 20,
369 6, 22, 6, 22, 6, 22, 7, 22, 6, 21, 6, 21
370 };
371
372 if(year < 1901 || year > 2099) {
373 return NULL;
374 }
375
376 uint64_t data = solar_terms_data[year - 1901];
377
378 /* Uncompress the solar term data for that year */
379 for(i = 0; i < range_end_num; i++) {
380 uint8_t right = char_count_len * (range_end_num - i - 1);
381 uint64_t x = data >> right;
382 offsets[i] = (uint8_t)(x & 0x3u);
383 }
384
385 /* Check whether the date is any solar term. */
386 for(i = 0; i < 24; i++) {
387 term_month = i / 2 + 1;
388 term_day = solar_terms_min_day[i] + offsets[i];
389 uint16_t term_date = (uint16_t)(term_month * 100 + term_day);
390 if(term_date == date_value) {
391 return solar_terms_name[i];
392 }
393 }
394
395 return NULL;
396}
