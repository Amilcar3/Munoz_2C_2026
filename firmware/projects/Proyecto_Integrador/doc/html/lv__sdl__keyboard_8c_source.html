43lv_indev_t * lv_sdl_keyboard_create(void)
44{
45 lv_sdl_keyboard_t * dsc = lv_malloc_zeroed(sizeof(lv_sdl_keyboard_t));
46 LV_ASSERT_MALLOC(dsc);
47 if(dsc == NULL) return NULL;
48
49 lv_indev_t * indev = lv_indev_create();
50 LV_ASSERT_MALLOC(indev);

52 lv_free(dsc);
53 return NULL;
54 }
55
56 lv_indev_set_type(indev, LV_INDEV_TYPE_KEYPAD);
57 lv_indev_set_read_cb(indev, sdl_keyboard_read);
58 lv_indev_set_driver_data(indev, dsc);
59 lv_indev_set_mode(indev, LV_INDEV_MODE_EVENT);
60 lv_indev_add_event_cb(indev, release_indev_cb, LV_EVENT_DELETE, indev);

63}

69static void sdl_keyboard_read(lv_indev_t * indev, lv_indev_data_t * data)
70{
71 lv_sdl_keyboard_t * dev = lv_indev_get_driver_data(indev);
72
73 const size_t len = lv_strlen(dev->buf);
74
75 /*Send a release manually*/
76 if(dev->dummy_read) {
77 dev->dummy_read = false;
78 data->state = LV_INDEV_STATE_RELEASED;
79 }
80 /*Send the pressed character*/
81 else if(len > 0) {
82 dev->dummy_read = true;
83 data->state = LV_INDEV_STATE_PRESSED;
84 data->key = 0;
85 /*Copy the first UTF8 character from the buffer*/
86 uint32_t utf8_len = lv_text_encoded_size(dev->buf);
87 if(utf8_len == 0) utf8_len = 1; /*Make sure that at least 1 character is read*/
88 lv_memcpy(&data->key, dev->buf, utf8_len);
89
90 /*Drop the first character*/
91 lv_memmove(dev->buf, dev->buf + utf8_len, len - utf8_len + 1);
92 }
93}

107void lv_sdl_keyboard_handler(SDL_Event * event)
108{
109 uint32_t win_id = UINT32_MAX;
110 switch(event->type) {
111 case SDL_KEYDOWN:
112 win_id = event->key.windowID;
113 break;
114 case SDL_TEXTINPUT:
115 win_id = event->text.windowID;
116 break;
117 default:
118 return;
119 }
120
121 lv_display_t * disp = lv_sdl_get_disp_from_win_id(win_id);
122
123
124 /*Find a suitable indev*/
125 lv_indev_t * indev = lv_indev_get_next(NULL);
126 while(indev) {
127 if(lv_indev_get_read_cb(indev) == sdl_keyboard_read) {
128 /*If disp is NULL for any reason use the first indev with the correct type*/
129 if(disp == NULL || lv_indev_get_display(indev) == disp) break;
130 }
131 indev = lv_indev_get_next(indev);
132 }
133 if(indev == NULL) return;
134 lv_sdl_keyboard_t * dsc = lv_indev_get_driver_data(indev);
135
136 /* We only care about SDL_KEYDOWN and SDL_TEXTINPUT events */
137 switch(event->type) {
138 case SDL_KEYDOWN: { /*Button press*/
139 const uint32_t ctrl_key = keycode_to_ctrl_key(event->key.keysym.sym);
140 if(ctrl_key == '\0')
141 return;
142 const size_t len = lv_strlen(dsc->buf);
143 if(len < KEYBOARD_BUFFER_SIZE - 1) {
144 dsc->buf[len] = ctrl_key;
145 dsc->buf[len + 1] = '\0';
146 }
147 break;
148 }
149 case SDL_TEXTINPUT: { /*Text input*/
150 const size_t len = lv_strlen(dsc->buf) + lv_strlen(event->text.text);
151 if(len < KEYBOARD_BUFFER_SIZE - 1)
152 lv_strcat(dsc->buf, event->text.text);
153 }
154 break;
155 default:
156 break;
157
158 }
159
160 size_t len = lv_strlen(dsc->buf);
161 while(len) {
162 lv_indev_read(indev);
163
164 /*Call again to handle dummy read in `sdl_keyboard_read`*/
165 lv_indev_read(indev);
166 len--;
167 }
168}

175static uint32_t keycode_to_ctrl_key(SDL_Keycode sdl_key)
176{
177 /*Remap some key to LV_KEY_... to manage groups*/
178 switch(sdl_key) {
179 case SDLK_RIGHT:
180 case SDLK_KP_PLUS:
181 return LV_KEY_RIGHT;
182
183 case SDLK_LEFT:
184 case SDLK_KP_MINUS:
185 return LV_KEY_LEFT;
186
187 case SDLK_UP:
188 return LV_KEY_UP;
189
190 case SDLK_DOWN:
191 return LV_KEY_DOWN;
192
193 case SDLK_ESCAPE:
194 return LV_KEY_ESC;
195
196 case SDLK_BACKSPACE:
197 return LV_KEY_BACKSPACE;
198
199 case SDLK_DELETE:
200 return LV_KEY_DEL;
201
202 case SDLK_KP_ENTER:
203 case '\r':
204 return LV_KEY_ENTER;
205
206 case SDLK_TAB:
207 case SDLK_PAGEDOWN:
208 return LV_KEY_NEXT;
209
210 case SDLK_PAGEUP:
211 return LV_KEY_PREV;
212
213 case SDLK_HOME:
214 return LV_KEY_HOME;
215
216 case SDLK_END:
217 return LV_KEY_END;
218
219 default:
220 return '\0';
221 }
