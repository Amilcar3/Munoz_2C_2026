130static void lv_monkey_read_cb(lv_indev_t * indev, lv_indev_data_t * data)
131{
132 LV_ASSERT(indev != NULL);
133 LV_ASSERT(data != NULL);
134 lv_monkey_t * monkey = lv_indev_get_user_data(indev);
135 LV_ASSERT(monkey != NULL);
136
137 data->btn_id = monkey->indev_data.btn_id;
138 data->point = monkey->indev_data.point;
139 data->enc_diff = monkey->indev_data.enc_diff;
140 data->state = monkey->indev_data.state;
141}

143static int32_t lv_monkey_random(int32_t howsmall, int32_t howbig)
144{
145 if(howsmall >= howbig) {
146 return howsmall;
147 }
148 int32_t diff = howbig - howsmall;
149 return (int32_t)lv_rand(0, diff) + howsmall;
150}

152static void lv_monkey_timer_cb(lv_timer_t * timer)
153{
154 LV_ASSERT(timer != NULL);
155 lv_monkey_t * monkey = lv_timer_get_user_data(timer);
156 LV_ASSERT(monkey != NULL);
157 LV_ASSERT(monkey->indev != NULL);
158 lv_indev_data_t * data = &monkey->indev_data;
159
160 switch(lv_indev_get_type(monkey->indev)) {
161 case LV_INDEV_TYPE_POINTER:
162 data->point.x = (int32_t)lv_monkey_random(0, LV_HOR_RES - 1);
163 data->point.y = (int32_t)lv_monkey_random(0, LV_VER_RES - 1);
164 break;
165 case LV_INDEV_TYPE_ENCODER:
166 data->enc_diff = (int16_t)lv_monkey_random(monkey->config.input_range.min, monkey->config.input_range.max);
167 break;
168 case LV_INDEV_TYPE_BUTTON:
169 data->btn_id = (uint32_t)lv_monkey_random(monkey->config.input_range.min, monkey->config.input_range.max);
170 break;
171 case LV_INDEV_TYPE_KEYPAD: {
172 int32_t index = lv_monkey_random(0, sizeof(lv_key_map) / sizeof(lv_key_map[0]) - 1);
173 data->key = lv_key_map[index];
174 break;
175 }
176 default:
177 break;
178 }
179
180 data->state = lv_monkey_random(0, 100) < 50 ? LV_INDEV_STATE_RELEASED : LV_INDEV_STATE_PRESSED;
181
182 LV_ASSERT(monkey->timer != NULL);
183 lv_timer_set_period(monkey->timer, lv_monkey_random(monkey->config.period_range.min, monkey->config.period_range.max));
184}
