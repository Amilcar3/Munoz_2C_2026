11/* Drag the divider: resize the top pane; the bottom pane grows to fill the rest. */
12static void splitter_pressing_v_cb(lv_event_t * e)
13{
14 lv_obj_t * divider = lv_event_get_target_obj(e);
15 lv_obj_t * cont = lv_obj_get_parent(divider);
16 lv_obj_t * pane_top = lv_obj_get_child(cont, 0);
17
18 lv_indev_t * indev = lv_indev_active();
19 if(indev == NULL) return;
20
21 lv_point_t p;
22 lv_indev_get_point(indev, &p);
23
24 lv_area_t content;
25 lv_obj_get_content_coords(cont, &content);
26 int32_t avail = lv_area_get_height(&content) - SPLIT_DIVIDER_H;
27 if(avail <= 2 * SPLIT_MIN_PANE) return;
28
29 int32_t top_h = (p.y - content.y1) - SPLIT_DIVIDER_H / 2;
30 if(top_h < SPLIT_MIN_PANE) top_h = SPLIT_MIN_PANE;
31 if(top_h > avail - SPLIT_MIN_PANE) top_h = avail - SPLIT_MIN_PANE;
32
33 lv_obj_set_height(pane_top, top_h);
34}
