204static void wl_g2d_global_handler(void * backend_ctx, struct wl_registry * registry, uint32_t name,
205 const char * interface, uint32_t version)
206{
207
208 lv_wl_g2d_ctx_t * ctx = (lv_wl_g2d_ctx_t *)backend_ctx;
209
210 if(lv_streq(interface, zwp_linux_dmabuf_v1_interface.name)) {
211 version = LV_MIN(version, (uint32_t)zwp_linux_dmabuf_v1_interface.version);
212 ctx->handler = wl_registry_bind(registry, name, &zwp_linux_dmabuf_v1_interface, version);
213
214 if(version >= 4) {
215 struct zwp_linux_dmabuf_feedback_v1 * feedback = zwp_linux_dmabuf_v1_get_default_feedback(ctx->handler);
216 zwp_linux_dmabuf_feedback_v1_add_listener(feedback, &dmabuf_listener_v5, ctx);
217 }
218 else if(version < 3) {
219 zwp_linux_dmabuf_v1_add_listener(ctx->handler, &dmabuf_listener, ctx);
220 }
221 else if(version == 3) {
222 zwp_linux_dmabuf_v1_add_listener(ctx->handler, &dmabuf_listener_v3, ctx);
223 }
224 wl_display_roundtrip(lv_wl_ctx.wl_display);
225 }

227static void init_buffer(lv_wl_g2d_ctx_t * ctx, lv_wl_buffer_t * buffer, uint32_t width, uint32_t height,
228 lv_color_format_t cf)
229{
230 uint32_t drm_cf = lv_cf_to_drm_cf(cf);
231 uint32_t stride = lv_draw_buf_width_to_stride(width, cf);
232 buffer->lv_draw_buf = lv_draw_buf_create(width, height, cf, stride);
233 buffer->dmabuf_fd = g2d_get_buf_fd(buffer->lv_draw_buf);
234 buffer->stride = stride;
235 buffer->offset = 0;
236 buffer->busy = false;
237
238 /* Will be set on the dmabuf callback if the creation is successful*/
239 buffer->wl_buffer = NULL;
240
241 struct zwp_linux_buffer_params_v1 * params = zwp_linux_dmabuf_v1_create_params(ctx->handler);
242
243 zwp_linux_buffer_params_v1_add(params,
244 buffer->dmabuf_fd,
245 0,
246 buffer->offset,
247 buffer->stride,
248 0,
249 0);
250
251 zwp_linux_buffer_params_v1_add_listener(params, &params_listener, buffer);
252 zwp_linux_buffer_params_v1_create(params, width, height, drm_cf, 0);
253}

267static lv_wl_g2d_display_data_t * wl_g2d_create_display_data(lv_wl_g2d_ctx_t * ctx, lv_display_t * display,
268 int32_t width, int32_t height)
269{
270 LV_ASSERT(ctx != NULL);
271 if(!ctx->handler) {
272 LV_LOG_WARN("dmabuf registry not bound. Can't initialize a display with the g2d backend");
273 return NULL;
274 }
275 lv_wl_g2d_display_data_t * ddata = lv_zalloc(sizeof(*ddata));
276 LV_ASSERT_MALLOC(ddata);
277 if(!ddata) {
278 return NULL;
279 }
280
281 lv_color_format_t cf = lv_display_get_color_format(display);
282 if(cf == LV_COLOR_FORMAT_RGB565 && !ctx->supports_rgb565) {
283 LV_LOG_WARN("RGB565 is not supported by the wayland compositor. Falling back to XRGB8888");
284 cf = LV_COLOR_FORMAT_XRGB8888;
285 lv_display_set_color_format(display, cf);
286 }
287
288 ddata->drm_cf = lv_cf_to_drm_cf(cf);
289 for(size_t i = 0; i < LV_WL_G2D_BUF_COUNT; i++) {
290 init_buffer(ctx, &ddata->buffers[i], width, height, cf);
291 }
292
293 lv_display_rotation_t rotation = lv_display_get_rotation(display);
294 if(rotation == LV_DISPLAY_ROTATION_90 || rotation == LV_DISPLAY_ROTATION_270) {
295 init_buffer(ctx, &ddata->rotate_buffer, height, width, cf);
296 }
297 else if(rotation == LV_DISPLAY_ROTATION_180) {
298 init_buffer(ctx, &ddata->rotate_buffer, width, height, cf);
299 }
300
301 wl_display_flush(lv_wl_ctx.wl_display);
302 wl_display_roundtrip(lv_wl_ctx.wl_display);
303 for(size_t i = 0; i < LV_WL_G2D_BUF_COUNT; ++i) {
304 if(!ddata->buffers[i].wl_buffer) {
305 wl_g2d_delete_display_data(ddata);
306 LV_LOG_ERROR("DMABUF creation failed");
307 return NULL;
308 }
309 }
310
311 if(rotation == LV_DISPLAY_ROTATION_0) {
312 lv_display_set_draw_buffers(display, ddata->buffers[0].lv_draw_buf, ddata->buffers[1].lv_draw_buf);
313 return ddata;
314 }
315
316 /*rotation != 0 so use a separate buffer for rendering and two other for flushing*/
317 if(!ddata->rotate_buffer.wl_buffer) {
318 wl_g2d_delete_display_data(ddata);
319 LV_LOG_ERROR("DMABUF creation failed");
320 return NULL;
321 }
322 lv_display_set_draw_buffers(display, ddata->rotate_buffer.lv_draw_buf, NULL);
323
324 return ddata;
325}

327static void wl_g2d_delete_display_data(lv_wl_g2d_display_data_t * ddata)
328{
329 for(int i = 0; i < LV_WL_G2D_BUF_COUNT; i++) {
330 delete_buffer(ddata->buffers + i);
331 }
332
333 delete_buffer(&ddata->rotate_buffer);
334
335 lv_free(ddata);
336}

357static uint32_t lv_cf_to_drm_cf(lv_color_format_t cf)
358{
359 if(cf == LV_COLOR_FORMAT_UNKNOWN) {
360 return DRM_FORMAT_ARGB8888; /* Default to ARGB8888 */
361 }
362
363 switch(cf) {
364 case LV_COLOR_FORMAT_XRGB8888:
365 return DRM_FORMAT_XRGB8888;
366 break;
367 case LV_COLOR_FORMAT_ARGB8888:
368 case LV_COLOR_FORMAT_ARGB8888_PREMULTIPLIED:
369 return DRM_FORMAT_ARGB8888;
370 break;
371 case LV_COLOR_FORMAT_RGB565:
372 return DRM_FORMAT_RGB565;
373 break;
374 default:
375 return DRM_FORMAT_ARGB8888;
376 }
377}

447static void dmabuf_format_table(void * data, struct zwp_linux_dmabuf_feedback_v1 * zwp_linux_dmabuf_feedback,
448 int32_t fd, uint32_t size)
449{
450 lv_wl_g2d_ctx_t * ctx = data;
451
452 LV_UNUSED(zwp_linux_dmabuf_feedback);
453
454 if(fd < 0 || size == 0) {
455 LV_LOG_ERROR("Invalid format table fd=%d size=%u", fd, size);
456 return;
457 }
458
459 /* Map the format table file descriptor */
460 void * table = mmap(NULL, size, PROT_READ, MAP_PRIVATE, fd, 0);
461 if(table == MAP_FAILED) {
462 LV_LOG_ERROR("Failed to mmap format table: %s", strerror(errno));
463 close(fd);
464 return;
465 }
466
467 LV_LOG_TRACE("Received format table with fd %d and size %u", fd, size);
468
469 /* Parse the format table - each entry is 16 bytes: 4 bytes format + 4 bytes padding + 8 bytes modifier */
470 size_t num_formats = size / 16;
471 uint32_t * formats = (uint32_t *)table;
472
473 for(size_t i = 0; i < num_formats; i++) {
474 /* Each entry is 4 uint32_t words */
475 uint32_t format = formats[i * 4];
476 if(format == DRM_FORMAT_RGB565) {
477 ctx->supports_rgb565 = true;
478 }
479 }
480
481 /* Clean up */
482 munmap(table, size);
483 close(fd);
484}

589static bool wait_for_buffer_release(lv_wl_buffer_t * buffer)
590{
591 struct pollfd pfd = { .fd = wl_display_get_fd(lv_wl_ctx.wl_display), .events = POLLIN };
592 const uint32_t start = lv_tick_get();
593
594 while(buffer->busy) {
595 if(lv_tick_elaps(start) >= LV_WL_G2D_RELEASE_TIMEOUT_MS) {
596 return false;
597 }
598
599 wl_display_flush(lv_wl_ctx.wl_display);
600 if(wl_display_prepare_read(lv_wl_ctx.wl_display) == 0) {
601 int ret = poll(&pfd, 1, LV_WL_G2D_RELEASE_POLL_MS);
602 if(ret > 0 && (pfd.revents & POLLIN)) {
603 wl_display_read_events(lv_wl_ctx.wl_display);

625{

628 lv_wl_g2d_display_data_t * ddata = lv_wayland_get_backend_display_data(disp);
629 int32_t src_width = lv_area_get_width(area);
630 int32_t src_height = lv_area_get_height(area);
631 lv_display_rotation_t rotation = lv_display_get_rotation(disp);
632
633 struct wl_surface * surface = lv_wayland_get_window_surface(disp);
634 /* Mark surface damage */
635 wl_surface_damage(surface, area->x1, area->y1, src_width, src_height);
636

638 lv_display_flush_ready(disp);

640 }
641
642 lv_wl_buffer_t * buf = &ddata->buffers[ddata->next_buf_idx];
643 if(buf->busy) {
644 LV_LOG_WARN("Rendered into a buffer the compositor still owns");
645 }
646
647 lv_draw_buf_invalidate_cache(buf->lv_draw_buf, NULL);
648
649 /*Rerender the whole surface if we're using rotation*/
650 if(rotation != LV_DISPLAY_ROTATION_0) {
651 wl_surface_damage(surface, 0, 0,
652 lv_display_get_original_horizontal_resolution(disp),
653 lv_display_get_original_vertical_resolution(disp));
654
655 lv_draw_buf_invalidate_cache(ddata->rotate_buffer.lv_draw_buf, NULL);
656 g2d_rotate(ddata->rotate_buffer.lv_draw_buf, buf->lv_draw_buf,
657 lv_display_get_original_horizontal_resolution(disp),
658 lv_display_get_original_vertical_resolution(disp),
659 lv_display_get_rotation(disp),
660 lv_display_get_color_format(disp));
661
662 }
663
664 /* Finally, attach buffer and commit to surface */
665 struct wl_callback * cb = wl_surface_frame(surface);
666 wl_callback_add_listener(cb, &frame_listener, disp);
667
668 wl_surface_attach(surface, buf->wl_buffer, 0, 0);
669 wl_surface_commit(surface);
670
671 buf->busy = true;
672
673 ddata->next_buf_idx = (ddata->next_buf_idx + 1) % LV_WL_G2D_BUF_COUNT;
674 if(!wait_for_buffer_release(&ddata->buffers[ddata->next_buf_idx])) {
675 LV_LOG_WARN("Timed out waiting for the compositor to release a buffer");
676 }
677}
