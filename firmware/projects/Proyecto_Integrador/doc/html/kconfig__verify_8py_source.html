33def check_kconfig_spaces(file_path):
34 """Check for space-only indentation in a Kconfig file.
35
36 Tab+spaces is allowed (conventional for help text).
37 Pure-space indentation will break kernel/Buildroot Kconfig parsers.
38 """
39 try:
40 with open(file_path, "r", encoding="utf-8") as f:
41 lines = f.readlines()
42 violations = []
43 for line_num, line in enumerate(lines, 1):
44 stripped = line.lstrip()
45 if not stripped:
46 continue
47 indent = line[: len(line) - len(stripped)]
48 if indent and not indent.startswith("\t") and " " in indent:
49 violations.append((line_num, indent))
50 if violations:
51 print(f"Space-only indentation found in {file_path}:")
52 for line_num, indent in violations:
53 print(f" Line {line_num}: {indent.count(' ')} leading spaces")
54 except Exception as e:
55 print(f"Error processing {file_path}: {e}")
