161static void lv_led_event(const lv_obj_class_t * class_p, lv_event_t * e)
162{
163 LV_UNUSED(class_p);
164
165 lv_result_t res;
166
167 /* Call the ancestor's event handler */
168 lv_event_code_t code = lv_event_get_code(e);
169 if(code != LV_EVENT_DRAW_MAIN && code != LV_EVENT_DRAW_MAIN_END) {
170 res = lv_obj_event_base(MY_CLASS, e);
171 if(res != LV_RESULT_OK) return;
172 }
173
174 lv_obj_t * obj = lv_event_get_current_target(e);
175 if(code == LV_EVENT_DRAW_MAIN) {
176 /*Make darker colors in a temporary style according to the brightness*/
177 lv_led_t * led = (lv_led_t *)obj;
178 lv_layer_t * layer = lv_event_get_layer(e);
179
180 lv_draw_rect_dsc_t rect_dsc;
181 lv_draw_rect_dsc_init(&rect_dsc);
182 rect_dsc.base.layer = layer;
183 lv_obj_init_draw_rect_dsc(obj, LV_PART_MAIN, &rect_dsc);
184
185 /*Use the original colors brightness to modify color->led*/
186 rect_dsc.bg_color = lv_color_mix(led->color, lv_color_black(), lv_color_brightness(rect_dsc.bg_color));
187 rect_dsc.bg_grad.stops[0].color = lv_color_mix(led->color, lv_color_black(),
188 lv_color_brightness(rect_dsc.bg_grad.stops[0].color));
189 rect_dsc.bg_grad.stops[1].color = lv_color_mix(led->color, lv_color_black(),
190 lv_color_brightness(rect_dsc.bg_grad.stops[1].color));
191 rect_dsc.shadow_color = lv_color_mix(led->color, lv_color_black(), lv_color_brightness(rect_dsc.shadow_color));
192 rect_dsc.border_color = lv_color_mix(led->color, lv_color_black(), lv_color_brightness(rect_dsc.border_color));
193 rect_dsc.outline_color = lv_color_mix(led->color, lv_color_black(), lv_color_brightness(rect_dsc.outline_color));
194
195 /*Mix. the color with black proportionally with brightness*/
196 rect_dsc.bg_color = lv_color_mix(rect_dsc.bg_color, lv_color_black(), led->bright);
197 rect_dsc.bg_grad.stops[0].color = lv_color_mix(rect_dsc.bg_grad.stops[0].color, lv_color_black(), led->bright);
198 rect_dsc.bg_grad.stops[1].color = lv_color_mix(rect_dsc.bg_grad.stops[1].color, lv_color_black(), led->bright);
199 rect_dsc.border_color = lv_color_mix(rect_dsc.border_color, lv_color_black(), led->bright);
200 rect_dsc.shadow_color = lv_color_mix(rect_dsc.shadow_color, lv_color_black(), led->bright);
201 rect_dsc.outline_color = lv_color_mix(rect_dsc.outline_color, lv_color_black(), led->bright);
202
203 /*Set the current shadow width according to brightness proportionally between LV_LED_BRIGHT_OFF
204 * and LV_LED_BRIGHT_ON*/
205 rect_dsc.shadow_width = ((led->bright - LV_LED_BRIGHT_MIN) * rect_dsc.shadow_width) /
206 (LV_LED_BRIGHT_MAX - LV_LED_BRIGHT_MIN);
207 rect_dsc.shadow_spread = ((led->bright - LV_LED_BRIGHT_MIN) * rect_dsc.shadow_spread) /
208 (LV_LED_BRIGHT_MAX - LV_LED_BRIGHT_MIN);
209
210 lv_draw_rect(layer, &rect_dsc, &obj->coords);
211 }
