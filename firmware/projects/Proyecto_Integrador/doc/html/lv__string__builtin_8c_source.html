61void * LV_ATTRIBUTE_FAST_MEM lv_memcpy(void * dst, const void * src, size_t len)
62{
63 LV_ASSERT(dst != NULL);
64 LV_ASSERT(src != NULL);
65
66 /*The destination is volatile on purpose: it stops the compiler recognizing the byte
67 *loops below as memcpy and calling into it. See #7573.*/
68 volatile uint8_t * d8 = dst;
69 const uint8_t * s8 = src;
70
71 /*Simplify for small memories*/
72 if(len < 16) {
73 while(len) {
74 *d8 = *s8;
75 d8++;
76 s8++;
77 len--;
78 }
79 return dst;
80 }
81
82 lv_uintptr_t d_align = (lv_uintptr_t)d8 & ALIGN_MASK;
83 lv_uintptr_t s_align = (lv_uintptr_t)s8 & ALIGN_MASK;
84
85 /*Byte copy for unaligned memories*/
86 if(s_align != d_align) {
87 while(len >= 32) {
88 _REPEAT8(_COPY(d8, s8));
89 _REPEAT8(_COPY(d8, s8));
90 _REPEAT8(_COPY(d8, s8));
91 _REPEAT8(_COPY(d8, s8));
92 len -= 32;
93 }
94 while(len) {
95 _COPY(d8, s8)
96 len--;
97 }
98 return dst;
99 }
100
101 /*Make the memories aligned*/
102 if(d_align) {
103 d_align = ALIGN_MASK + 1 - d_align;
104 while(d_align && len) {
105 _COPY(d8, s8);
106 d_align--;
107 len--;
108 }
109 }
110
111 /*The bulk of the work, in whole words. MEM_UNIT is as wide as the machine, so this is
112 *8 bytes at a time on a 64 bit target instead of 4*/
113 MEM_UNIT * du = (MEM_UNIT *)(uint8_t *)d8;
114 const MEM_UNIT * su = (const MEM_UNIT *)s8;
115 while(len >= 8 * sizeof(MEM_UNIT)) {
116 _REPEAT8(_COPY(du, su))
117 len -= 8 * sizeof(MEM_UNIT);
118 }
119
120 /*Whatever is left of a whole word. Without this every length that is a multiple of the
121 *block size finished byte by byte*/
122 while(len >= sizeof(MEM_UNIT)) {
123 _COPY(du, su)
124 len -= sizeof(MEM_UNIT);
125 }
126
127 d8 = (volatile uint8_t *)du;
128 s8 = (const uint8_t *)su;
129 while(len) {
130 _COPY(d8, s8)
131 len--;
132 }
133
134 return dst;
135}

137void LV_ATTRIBUTE_FAST_MEM lv_memset(void * dst, uint8_t v, size_t len)
138{
139 LV_ASSERT(dst != NULL);
140
141 uint8_t * d8 = (uint8_t *)dst;
142 lv_uintptr_t d_align = (lv_uintptr_t) d8 & ALIGN_MASK;
143
144 /*Make the address aligned*/
145 if(d_align) {
146 d_align = ALIGN_MASK + 1 - d_align;
147 while(d_align && len) {
148 _SET(d8, v);
149 len--;
150 d_align--;
151 }
152 }
153
154 MEM_UNIT vu = (MEM_UNIT)v;
155 vu |= vu << 8;
156 vu |= vu << 16;
157#ifdef LV_ARCH_64
158 vu |= vu << 32;
159#endif
160
161 MEM_UNIT * du = (MEM_UNIT *)d8;
162 while(len >= 8 * sizeof(MEM_UNIT)) {
163 _REPEAT8(_SET(du, vu));
164 len -= 8 * sizeof(MEM_UNIT);
165 }
166 while(len >= sizeof(MEM_UNIT)) {
167 _SET(du, vu);
168 len -= sizeof(MEM_UNIT);
169 }
170
171 d8 = (uint8_t *)du;
172 while(len) {
173 _SET(d8, v);
174 len--;
175 }
176}

178void * LV_ATTRIBUTE_FAST_MEM lv_memmove(void * dst, const void * src, size_t len)
179{
180 LV_ASSERT(dst != NULL);
181 LV_ASSERT(src != NULL);
182
183 if(dst < src || (char *)dst > ((char *)src + len)) {
184 return lv_memcpy(dst, src, len);
185 }
186
187 if(dst > src) {
188 char * tmp = (char *)dst + len - 1;
189 char * s = (char *)src + len - 1;
190
191 while(len--) {
192 *tmp-- = *s--;
193 }
194 }
195 else {
196 char * tmp = (char *)dst;
197 char * s = (char *)src;
198
199 while(len--) {
200 *tmp++ = *s++;
201 }
202 }
203
204 return dst;
205}
