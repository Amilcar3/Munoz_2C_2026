103void lv_arclabel_set_text(lv_obj_t * obj, const char * text)
104{
105 LV_CHECK_OBJ(obj, MY_CLASS, return);
106 lv_arclabel_t * arclabel = (lv_arclabel_t *)obj;
107
108 /*If text is NULL then just refresh with the current text*/
109 if(text == NULL) text = arclabel->text;
110 if(text == NULL) {
111 lv_obj_invalidate(obj);
112 return;
113 }
114
115 const size_t text_len = lv_strlen(text) + 1;
116
117 /*If set its own text then reallocate it (maybe its size changed)*/
118 if(arclabel->text == text && arclabel->static_txt == 0) {
119 arclabel->text = lv_realloc(arclabel->text, text_len);
120 LV_ASSERT_MALLOC(arclabel->text);
121 if(arclabel->text == NULL) return;
122 }
123 else {
124 /*Free the old text*/
125 if(arclabel->text != NULL && arclabel->static_txt == 0) {
126 lv_free(arclabel->text);
127 arclabel->text = NULL;
128 }
129
130 arclabel->text = lv_malloc(text_len);
131 LV_ASSERT_MALLOC(arclabel->text);
132 if(arclabel->text == NULL) return;
133
134 lv_strcpy(arclabel->text, text);
135
136 /*Now the text is dynamically allocated*/
137 arclabel->static_txt = 0;
138 }
139
140 lv_obj_invalidate(obj);
141}

420static void arclabel_draw_main(lv_event_t * e)
421{
422 LV_ASSERT(e != NULL);
423 lv_obj_t * obj = lv_event_get_current_target(e);
424 LV_ASSERT(obj != NULL);
425 lv_arclabel_t * arclabel = (lv_arclabel_t *)obj;
426
427 const char * text = arclabel->text;
428 const char * text_start = text;
429
430 lv_area_t coords;
431 lv_obj_get_content_coords(obj, &coords);
432
433 lv_layer_t * layer = lv_event_get_layer(e);
434
435 const lv_font_t * font = lv_obj_get_style_text_font_internal(obj, LV_PART_MAIN);
436 const lv_color_t color = lv_obj_get_style_text_color_internal(obj, LV_PART_MAIN);
437 const lv_opa_t opa = LV_OPA_MIX2(layer->opa, lv_obj_get_style_text_opa_internal(obj, LV_PART_MAIN));
438 const int32_t letter_space = lv_obj_get_style_text_letter_space_internal(obj, LV_PART_MAIN);
439
440 int32_t arc_r = 0;
441 bool need_ellipsis = false;
442 uint32_t processed_total_word_count = 0;
443 const lv_value_precise_t total_visible_angle = arclabel_calc_arc_text_total_angle(obj, &arc_r, &need_ellipsis,
444 &processed_total_word_count);
445
446 const int32_t offset = arclabel->offset;
447 const lv_value_precise_t angle_offset = rad_to_deg(offset, arc_r);
448
449 lv_value_precise_t angle_start = 0;
450 switch(arclabel->text_align_h) {
451 case LV_ARCLABEL_TEXT_ALIGN_LEADING:
452 angle_start = angle_offset;
453 break;
454 case LV_ARCLABEL_TEXT_ALIGN_CENTER:
455 angle_start = (arclabel->angle_size + angle_offset - total_visible_angle) / 2;
456 break;
457 case LV_ARCLABEL_TEXT_ALIGN_TRAILING:
458 angle_start = arclabel->angle_size - total_visible_angle;
459 break;
460 default:
461 break;
462 }
463
464 bool draw_ellipsis = false;
465 int ellipsis_dot_index = 0;
466 uint32_t processed_word_count = 0;
467 lv_value_precise_t prev_letter_w = 0;
468 lv_value_precise_t total_arc_length = deg_to_rad(arclabel->angle_size, arc_r);
469 lv_value_precise_t curr_total_arc_length = deg_to_rad(angle_start, arc_r);
470 uint32_t letter = 0;
471 uint32_t letter_next = 0;
472 while(text) {
473 uint32_t word_i = 0;
474 uint32_t text_len = LV_TEXT_LEN_MAX;
475 lv_color_t recolor_color = color;
476 if(arclabel->recolor) text = recolor_cmd_get_next(text, LV_TEXT_LEN_MAX, &text_start, &text_len, &recolor_color);
477 else text = NULL;
478
479
480 while(word_i < text_len && (arclabel->overflow == LV_ARCLABEL_OVERFLOW_VISIBLE ||
481 curr_total_arc_length <= total_arc_length)) {
482 if(draw_ellipsis) {
483 if(ellipsis_dot_index >= 3) break;
484 letter = '.';
485 letter_next = ellipsis_dot_index < 2 ? '.' : '\0';
486 ellipsis_dot_index++;
487 }
488 else lv_text_encoded_letter_next_2(text_start, &letter, &letter_next, &word_i);
489
490 const lv_value_precise_t letter_w = lv_font_get_glyph_width(font, letter, letter_next);
491
492 if(processed_word_count > 0) {
493 if(processed_word_count >= processed_total_word_count && arclabel->overflow != LV_ARCLABEL_OVERFLOW_VISIBLE &&
494 !draw_ellipsis) {
495 if(need_ellipsis) {
496 draw_ellipsis = true;
497 continue;
498 }
499 break;
500 }
501 const lv_value_precise_t arc_offset = (prev_letter_w + letter_w + letter_space) / (lv_value_precise_t)2;
502 curr_total_arc_length += arc_offset;
503 }
504
505 const lv_value_precise_t curr_angle = arclabel->angle_start
506 + rad_to_deg(arclabel->dir == LV_ARCLABEL_DIR_CLOCKWISE
507 ? curr_total_arc_length
508 : total_arc_length - curr_total_arc_length, arc_r);
509
510#if LV_USE_FLOAT
511 const lv_value_precise_t x = cosf(deg_to_rad(curr_angle, 1)) * arc_r;
512 const lv_value_precise_t y = sinf(deg_to_rad(curr_angle, 1)) * arc_r;
513#else
514 const lv_value_precise_t x = (lv_value_precise_t)(lv_trigo_cos(curr_angle) * arc_r / 32767);
515 const lv_value_precise_t y = (lv_value_precise_t)(lv_trigo_sin(curr_angle) * arc_r / 32767);
516#endif
517
518 lv_point_t point = {
519 (int32_t)(x + (lv_value_precise_t)(lv_area_get_width(&coords) / 2 + coords.x1 + arclabel->center_offset.x)),
520 (int32_t)(y + (lv_value_precise_t)(lv_area_get_height(&coords) / 2 + coords.y1 + arclabel->center_offset.y)),
521 };
522
523 lv_draw_letter_dsc_t dsc;
524 lv_draw_letter_dsc_init(&dsc);
525 dsc.font = font;
526 dsc.color = arclabel->recolor ? recolor_color : color;
527 dsc.opa = opa;
528 if(arclabel->dir == LV_ARCLABEL_DIR_CLOCKWISE) dsc.rotation = (int32_t)((curr_angle + 90) * 10);
529 else dsc.rotation = (int32_t)((curr_angle - 90) * 10);
530
531 dsc.unicode = letter;
532 if(dsc.unicode == 0) {
533 break;
534 }
535
536 lv_draw_letter(layer, &dsc, &point);
537
538 prev_letter_w = letter_w;
539 processed_word_count++;
540
541#if LV_ARCLABEL_DEBUG
542 lv_draw_line_dsc_t line_dsc;
543 lv_draw_line_dsc_init(&line_dsc);
544 line_dsc.color = lv_color_make(0x11, 0x45, 0x14);
545 line_dsc.opa = LV_OPA_30;
546 line_dsc.width = 2;
547 line_dsc.p1 = (lv_point_precise_t) {
548 .x = point.x,
549 .y = point.y
550 };
551 line_dsc.p2 = (lv_point_precise_t) {
552 .x = lv_area_get_width(&coords) / 2 + coords.x1,
553 .y = lv_area_get_height(&coords) / 2 + coords.y1
554 };
555
556 lv_draw_line(layer, &line_dsc);
557#endif
558 }
559 }
560}

562static lv_value_precise_t arclabel_calc_arc_text_total_angle(lv_obj_t * obj, int32_t * arc_radius, bool * need_ellipsis,
563 uint32_t * letter_count)
564{
565 lv_arclabel_t * arclabel = (lv_arclabel_t *)obj;
566
567 const char * text = arclabel->text;
568 const char * text_start = text;
569
570 lv_area_t coords;
571 lv_obj_get_content_coords(obj, &coords);
572
573 const lv_font_t * font = lv_obj_get_style_text_font_internal(obj, LV_PART_MAIN);
574 const int32_t letter_space = lv_obj_get_style_text_letter_space_internal(obj, LV_PART_MAIN);
575
576 const int32_t line_height = font->line_height;
577 const int32_t base_line = font->base_line;
578 int32_t arc_r_delta = 0;
579 int32_t arc_r = arclabel->radius;
580
581 if(arc_r == LV_SIZE_CONTENT) arc_r = LV_PCT(100);
582 if(LV_COORD_IS_PCT(arc_r)) {
583 const int32_t width = lv_area_get_width(&coords);
584 const int32_t height = lv_area_get_height(&coords);
585 arc_r = lv_pct_to_px(arc_r, LV_MIN(width, height)) / 2;
586 }
587
588 switch(arclabel->text_align_v) {
589 case LV_ARCLABEL_TEXT_ALIGN_LEADING:
590 arc_r_delta = line_height - base_line;
591 break;
592 case LV_ARCLABEL_TEXT_ALIGN_CENTER:
593 arc_r_delta = line_height / 2 - base_line;
594 break;
595 case LV_ARCLABEL_TEXT_ALIGN_TRAILING:
596 arc_r_delta = -base_line;
597 break;
598 default:
599 break;
600 }
601
602 arc_r += arclabel->dir == LV_ARCLABEL_DIR_CLOCKWISE ? -arc_r_delta : arc_r_delta;
603 if(arc_radius != NULL) *arc_radius = arc_r;
604 lv_value_precise_t total_visible_angle = calc_arc_text_total_angle(text_start, font, arc_r,
605 arclabel->angle_size, letter_space, arclabel->recolor,
606 arclabel->overflow, arclabel->end_overlap, need_ellipsis, letter_count);
607 return total_visible_angle;
608}

610static lv_value_precise_t calc_arc_text_total_angle(const char * text, const lv_font_t * font, const uint32_t radius,
611 const lv_value_precise_t angle_size, const int32_t letter_space, const bool recolor,
612 const lv_arclabel_overflow_t overflow, bool end_overlap, bool * need_ellipsis, uint32_t * letter_count)
613{
614 const char * text_start = text;
615 uint32_t processed_letter_count = 0;
616 lv_value_precise_t prev_letter_w = 0;
617 lv_value_precise_t first_letter_w = 0;
618 const lv_value_precise_t angle_size_in_arc_length = deg_to_rad(angle_size, radius);
619 lv_value_precise_t total_arc_length = 0;
620 lv_value_precise_t pre_total_arc_length = 0;
621 uint32_t pre_processed_letter_count = 0;
622 lv_value_precise_t ellipsis_arc_length = 0;
623 bool full = false;
624 uint32_t letter = 0;
625 uint32_t letter_next = 0;
626
627 lv_value_precise_t available_arc_length = angle_size_in_arc_length;
628 if(overflow == LV_ARCLABEL_OVERFLOW_ELLIPSIS) {
629 lv_value_precise_t dot_width = lv_font_get_glyph_width(font, '.', '.');
630 ellipsis_arc_length = 3 * dot_width + 2 * letter_space;
631 if(available_arc_length > ellipsis_arc_length) available_arc_length -= ellipsis_arc_length;
632 else available_arc_length = 0;
633 }
634
635 while(text) {
636 uint32_t word_i = 0;
637 uint32_t text_len = LV_TEXT_LEN_MAX;
638 if(recolor) text = recolor_cmd_get_next(text, LV_TEXT_LEN_MAX, &text_start, &text_len, NULL);
639 else text = NULL;
640
641 while(word_i < text_len) {
642 const lv_value_precise_t end_arc_w = end_overlap ? 0 : (first_letter_w + letter_space + prev_letter_w) /
643 (lv_value_precise_t)2;
644 if(total_arc_length + end_arc_w > available_arc_length) {
645 full = true;
646 break;
647 }
648
649 pre_total_arc_length = total_arc_length;
650 pre_processed_letter_count = processed_letter_count;
651 lv_text_encoded_letter_next_2(text_start, &letter, &letter_next, &word_i);
652 const lv_value_precise_t letter_w = lv_font_get_glyph_width(font, letter, letter_next);
653
654 if(processed_letter_count > 0) {
655 const lv_value_precise_t arc_offset = (prev_letter_w + letter_w + letter_space) / (lv_value_precise_t)2;
656
657 total_arc_length += arc_offset;
658 }
659 else {
660 first_letter_w = letter_w;
661 }
662
663 if(letter == 0) break;
664
665 prev_letter_w = letter_w;
666 processed_letter_count++;
667 }
668 }
669
670 if(full && (overflow == LV_ARCLABEL_OVERFLOW_ELLIPSIS || overflow == LV_ARCLABEL_OVERFLOW_CLIP)) {
671 total_arc_length = pre_total_arc_length + (overflow == LV_ARCLABEL_OVERFLOW_ELLIPSIS
672 ? ellipsis_arc_length
673 : 0);
674 processed_letter_count = pre_processed_letter_count;
675 }
676
677 if(need_ellipsis && full && overflow == LV_ARCLABEL_OVERFLOW_ELLIPSIS) *need_ellipsis = true;
678 if(letter_count) *letter_count = processed_letter_count;
679
680 return rad_to_deg(total_arc_length, radius);
681}

683static const char * recolor_cmd_get_next(const char * text_in, uint32_t len_in,
684 const char ** text_out, uint32_t * len_out,
685 lv_color_t * color_out)
686{
687 if(!text_in || len_in == 0 || *text_in == '\0') return NULL;
688
689 const char * text = text_in;
690 uint32_t proc_len = 0;
691 bool has_cmd = false;
692
693 if(*text == LV_TXT_COLOR_CMD[0]) {
694 if(len_in < 8) {
695 if(text_out) *text_out = text_in;
696 if(len_out) *len_out = len_in;
697 return NULL;
698 }
699 has_cmd = true;
700 text++;
701
702 int32_t index = 0;
703 uint8_t color_buf[6];
704 while(*text && index < 6) {
705 uint8_t ch = text[index];
706 if(!((ch >= '0' && ch <= '9') || (ch >= 'A' && ch <= 'F') || (ch >= 'a' && ch <= 'f'))) break;
707 if(ch >= 'a' && ch <= 'f') ch -= 'a' - 10;
708 else if(ch >= 'A' && ch <= 'F') ch -= 'A' - 10;
709 else ch -= '0';
710 color_buf[index] = ch;
711 index++;
712 }
713
714 const bool has_valid_param = index == 6 && text[index] == ' ';
715
716 text += index;
717 if(has_valid_param && color_out)
718 *color_out = lv_color_make(color_buf[0] << 4 | color_buf[1],
719 color_buf[2] << 4 | color_buf[3],
720 color_buf[4] << 4 | color_buf[5]);
721
722 proc_len = text - text_in;
723 while(proc_len < len_in && *text && *text++ != ' ') {
724 proc_len++;
725 }
726 }
727
728 const char * text_segment_start = text;
729 while(proc_len < len_in && *text && *text != LV_TXT_COLOR_CMD[0]) {
730 text++;
731 proc_len++;
732 };
733 if(text_out) *text_out = text_segment_start;
734 if(len_out) *len_out = text - text_segment_start;
735 if(*text == '\0') return NULL;
736 if(has_cmd && *text == LV_TXT_COLOR_CMD[0]) {
737 text++;
738 proc_len++;
739 };
740
741 return proc_len < len_in && *text ? text : NULL;
742}
