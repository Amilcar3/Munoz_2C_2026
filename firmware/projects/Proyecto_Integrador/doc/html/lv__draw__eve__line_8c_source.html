21void lv_draw_eve_line(lv_draw_task_t * t, const lv_draw_line_dsc_t * dsc)
22{
23
24 if(dsc->width == 0)
25 return;
26 if(dsc->opa <= LV_OPA_MIN)
27 return;
28 if(dsc->p1.x == dsc->p2.x && dsc->p1.y == dsc->p2.y)
29 return;
30
31
32
33 uint32_t line_w = dsc->width * 8;
34 lv_eve_scissor(t->clip_area.x1, t->clip_area.y1, t->clip_area.x2, t->clip_area.y2);
35 lv_eve_save_context();
36 lv_eve_color_opa(dsc->opa);
37 lv_eve_color(dsc->color);
38
39 if(dsc->dash_gap || dsc->dash_width) {
40 LV_LOG_WARN("line dash_gap and dash_width not implemented by EVE yet.");
41 }
42 /* Check if it's a vertical or horizontal line without rounding */
43 bool is_vertical = (dsc->p1.x == dsc->p2.x);
44 bool is_horizontal = (dsc->p1.y == dsc->p2.y);
45 bool no_round = (!dsc->round_end || !dsc->round_start);
46
47 if((is_vertical || is_horizontal) && no_round) {
48 lv_eve_primitive(LV_EVE_PRIMITIVE_RECTS);
49 lv_eve_vertex_2f(dsc->p1.x, dsc->p1.y);
50 lv_eve_vertex_2f(dsc->p2.x, dsc->p2.y);
51 }
52 else {
53 /* Draw inclined line or line with rounding (not possible without rounding)*/
54 lv_eve_primitive(LV_EVE_PRIMITIVE_LINE_STRIP);
55 lv_eve_line_width(line_w);
56 lv_eve_vertex_2f(dsc->p1.x, dsc->p1.y);
57 lv_eve_vertex_2f(dsc->p2.x, dsc->p2.y);
58 }
59
60 lv_eve_restore_context();
61}
