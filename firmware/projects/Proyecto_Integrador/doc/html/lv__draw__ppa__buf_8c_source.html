65static void cache_msync_rows(const lv_draw_buf_t * draw_buf, const lv_area_t * area, uint32_t dir)
66{
67 size_t line = ppa_cache_line_size(draw_buf->data);
68 if(line == 0) return; /* Not cached */
69
70 size_t start = 0;
71 size_t end = draw_buf->data_size;
72
73 /* Whole rows, and the area is relative to the buffer */
74 if(area) {
75 uint32_t stride = draw_buf->header.stride;
76 int32_t y1 = area->y1 < 0 ? 0 : area->y1;
77 int32_t y2 = LV_MIN(area->y2, (int32_t)draw_buf->header.h - 1);
78 if(stride == 0 || y2 < y1) return;
79
80 start = (size_t)y1 * stride;
81 end = (size_t)(y2 + 1) * stride;
82 }
83
84 /* M2C is refused unless the range is whole cache lines */
85 start &= ~(line - 1);
86 end = (end + line - 1) & ~(line - 1);
87 if(end > draw_buf->data_size) end = draw_buf->data_size & ~(line - 1);
88 if(start >= end) return;
89
90 esp_cache_msync(draw_buf->data + start, end - start, dir | ESP_CACHE_MSYNC_FLAG_TYPE_DATA);
91}
