93lv_obj_t * lv_tabview_add_tab(lv_obj_t * obj, const char * name)
94{
95 LV_CHECK_OBJ(obj, MY_CLASS, return NULL);
96 lv_obj_t * cont = lv_tabview_get_content(obj);
97
98 lv_obj_t * page = lv_obj_create(cont);
99 lv_obj_set_size(page, lv_pct(100), lv_pct(100));
100
101 lv_obj_t * tab_bar = lv_tabview_get_tab_bar(obj);
102
103 lv_obj_t * button = lv_button_create(tab_bar);
104 lv_obj_set_flex_grow(button, 1);
105 lv_obj_set_size(button, lv_pct(100), lv_pct(100));
106 lv_obj_add_event_cb(button, button_clicked_event_cb, LV_EVENT_CLICKED, NULL);
107 lv_group_t * g = lv_group_get_default();
108 if(g) lv_group_add_obj(g, button);
109
110 lv_obj_t * label = lv_label_create(button);
111 lv_label_set_text(label, name);
112 lv_obj_center(label);
113
114 uint32_t tab_idx = lv_obj_get_child_count(cont) - 1;
115 lv_tabview_t * tabview = (lv_tabview_t *)obj;
116 if(tab_idx == tabview->tab_cur) {
117 lv_tabview_set_active(obj, tab_idx, LV_ANIM_OFF);
118 }
119
120 return page;
121}

150void lv_tabview_set_active(lv_obj_t * obj, uint32_t idx, lv_anim_enable_t anim_en)
151{
152 LV_CHECK_OBJ(obj, MY_CLASS, return);
153 lv_tabview_t * tabview = (lv_tabview_t *)obj;
154
155 tabview->tab_cur = idx;
156
157 lv_obj_t * cont = lv_tabview_get_content(obj);
158 lv_obj_t * tab_bar = lv_tabview_get_tab_bar(obj);
159
160 uint32_t tab_cnt = lv_tabview_get_tab_count(obj);
161 if(idx >= tab_cnt) return;
162
163 /*To be sure lv_obj_get_content_width will return valid value*/
164 if(cont == NULL) return;
165
166 lv_obj_update_layout(obj);
167
168 if((tabview->tab_pos & LV_DIR_VER) != 0) {
169 int32_t gap = lv_obj_get_style_pad_column_internal(cont, LV_PART_MAIN);
170 int32_t w = lv_obj_get_content_width(cont);
171 if(lv_obj_get_style_base_dir_internal(obj, LV_PART_MAIN) != LV_BASE_DIR_RTL) {
172 lv_obj_scroll_to_x(cont, idx * (gap + w), anim_en);
173 }
174 else {
175 int32_t id_rtl = -(int32_t)idx;
176 lv_obj_scroll_to_x(cont, (gap + w) * id_rtl, anim_en);
177 }
178 }
179 else {
180 int32_t gap = lv_obj_get_style_pad_row_internal(cont, LV_PART_MAIN);
181 int32_t h = lv_obj_get_content_height(cont);
182 lv_obj_scroll_to_y(cont, idx * (gap + h), anim_en);
183 }
184
185 uint32_t i = 0;
186 lv_obj_t * button = lv_obj_get_child_by_type(tab_bar, i, &lv_button_class);
187 while(button) {
188 lv_obj_set_state(button, LV_STATE_CHECKED, i == idx);
189 i++;
190 button = lv_obj_get_child_by_type(tab_bar, (int32_t)i, &lv_button_class);
191 }
192
193}

195void lv_tabview_set_tab_bar_position(lv_obj_t * obj, lv_dir_t dir)
196{
197 LV_CHECK_OBJ(obj, MY_CLASS, return);
198 lv_tabview_t * tabview = (lv_tabview_t *)obj;
199
200 switch(dir) {
201 case LV_DIR_TOP:
202 lv_obj_set_flex_flow(obj, LV_FLEX_FLOW_COLUMN);
203 break;
204 case LV_DIR_BOTTOM:
205 lv_obj_set_flex_flow(obj, LV_FLEX_FLOW_COLUMN_REVERSE);
206 break;
207 case LV_DIR_LEFT:
208 lv_obj_set_flex_flow(obj, LV_FLEX_FLOW_ROW);
209 break;
210 case LV_DIR_RIGHT:
211 lv_obj_set_flex_flow(obj, LV_FLEX_FLOW_ROW_REVERSE);
212 break;
213 case LV_DIR_HOR:
214 case LV_DIR_VER:
215 case LV_DIR_ALL:
216 case LV_DIR_NONE:
217 break;
218 }
219
220 lv_obj_t * tab_bar = lv_tabview_get_tab_bar(obj);
221 lv_obj_t * cont = lv_tabview_get_content(obj);
222
223 switch(dir) {
224 case LV_DIR_TOP:
225 case LV_DIR_BOTTOM:
226 lv_obj_set_width(cont, LV_PCT(100));
227 lv_obj_set_flex_grow(cont, 1);
228 lv_obj_set_flex_flow(tab_bar, LV_FLEX_FLOW_ROW);
229 lv_obj_set_flex_flow(cont, LV_FLEX_FLOW_ROW);
230 lv_obj_set_scroll_snap_x(cont, LV_SCROLL_SNAP_CENTER);
231 lv_obj_set_scroll_snap_y(cont, LV_SCROLL_SNAP_NONE);
232 break;
233 case LV_DIR_LEFT:
234 case LV_DIR_RIGHT:
235 lv_obj_set_height(cont, LV_PCT(100));
236 lv_obj_set_flex_grow(cont, 1);
237 lv_obj_set_flex_flow(tab_bar, LV_FLEX_FLOW_COLUMN);
238 lv_obj_set_flex_flow(cont, LV_FLEX_FLOW_COLUMN);
239 lv_obj_set_scroll_snap_x(cont, LV_SCROLL_SNAP_NONE);
240 lv_obj_set_scroll_snap_y(cont, LV_SCROLL_SNAP_CENTER);
241 break;
242 case LV_DIR_HOR:
243 case LV_DIR_VER:
244 case LV_DIR_ALL:
245 case LV_DIR_NONE:
246 break;
247 }
248
249 bool was_ver = tabview->tab_pos & LV_DIR_VER;
250 bool now_ver = dir & LV_DIR_VER;
251
252 if(was_ver != now_ver) {
253 int32_t dpi = lv_display_get_dpi(lv_obj_get_display(obj));
254 if(now_ver) {
255 lv_obj_set_size(tab_bar, lv_pct(100), dpi / 2);
256 }
257 else {
258 lv_obj_set_size(tab_bar, dpi, lv_pct(100));
259 }
260 }
261 tabview->tab_pos = dir;
262 /* Update the tab bar size after the position is changed*/
263 lv_tabview_set_tab_bar_size(obj, tabview->tab_bar_size);
264}

326static void lv_tabview_constructor(const lv_obj_class_t * class_p, lv_obj_t * obj)
327{
328 LV_UNUSED(class_p);
329 lv_tabview_t * tabview = (lv_tabview_t *)obj;
330 tabview->tab_pos = LV_DIR_NONE; /*Invalid value to apply the default TOP direction correctly*/
331
332 lv_obj_set_size(obj, LV_PCT(100), LV_PCT(100));
333
334 lv_obj_t * cont;
335
336 lv_obj_create(obj);
337 cont = lv_obj_create(obj);
338 lv_obj_set_flex_flow(cont, LV_FLEX_FLOW_ROW);
339
340 lv_obj_add_event_cb(cont, cont_scroll_end_event_cb, LV_EVENT_LAYOUT_CHANGED, NULL);
341 lv_obj_add_event_cb(cont, cont_scroll_end_event_cb, LV_EVENT_SCROLL_END, NULL);
342 lv_obj_set_scrollbar_mode(cont, LV_SCROLLBAR_MODE_OFF);
343
344 const lv_dir_t default_direction = LV_DIR_TOP;
345 const int32_t dpi = lv_display_get_dpi(lv_obj_get_display(obj));
346 if(default_direction & LV_DIR_VER) {
347 tabview->tab_bar_size = dpi / 2;
348 }
349 else {
350 tabview->tab_bar_size = dpi;
351 }
352 lv_tabview_set_tab_bar_position(obj, default_direction);
353
354 lv_obj_set_scroll_one(cont, true);
355 lv_obj_set_scroll_on_focus(cont, false);
356}

358static void lv_tabview_event(const lv_obj_class_t * class_p, lv_event_t * e)
359{
360 LV_UNUSED(class_p);
361 lv_result_t res = lv_obj_event_base(&lv_tabview_class, e);
362 if(res != LV_RESULT_OK) return;
363
364 lv_event_code_t code = lv_event_get_code(e);
365 lv_obj_t * target = lv_event_get_current_target(e);
366
367 if(code == LV_EVENT_SIZE_CHANGED) {
368 lv_tabview_set_active(target, lv_tabview_get_tab_active(target), LV_ANIM_OFF);
369 }
370}

372static void button_clicked_event_cb(lv_event_t * e)
373{
374 lv_obj_t * button = lv_event_get_current_target(e);
375
376 lv_obj_t * tv = lv_obj_get_parent(lv_obj_get_parent(button));
377
378 if(tv == NULL) return;
379
380 /* Remember currently active tab before the click */
381 uint32_t prev_idx = lv_tabview_get_tab_active(tv);
382
383 /* Index of the button that was clicked */
384 uint32_t idx = lv_obj_get_index_by_type(button, &lv_button_class);
385
386 /* Switch to the requested tab */
387 lv_tabview_set_active(tv, idx, LV_ANIM_OFF);
388
389 /* If the tab really changed, notify listeners just like the
390 * swipe/scroll handler does. */
391 if(prev_idx != idx) {
392 lv_obj_send_event(tv, LV_EVENT_VALUE_CHANGED, NULL);
393 }
394}

396static void cont_scroll_end_event_cb(lv_event_t * e)
397{
398 lv_obj_t * cont = lv_event_get_current_target(e);
399 lv_event_code_t code = lv_event_get_code(e);
400
401 lv_obj_t * tv = lv_obj_get_parent(cont);
402 lv_tabview_t * tv_obj = (lv_tabview_t *)tv;
403 if(code == LV_EVENT_LAYOUT_CHANGED) {
404 lv_tabview_set_active(tv, lv_tabview_get_tab_active(tv), LV_ANIM_OFF);
405 }
406 else if(code == LV_EVENT_SCROLL_END) {
407 lv_indev_t * indev = lv_indev_active();
408 if(indev && indev->state == LV_INDEV_STATE_PRESSED) {
409 return;
410 }
411
412 lv_point_t p;
413 lv_obj_get_scroll_end(cont, &p);
414
415 int32_t t;
416 if((tv_obj->tab_pos & LV_DIR_VER) != 0) {
417 int32_t w = lv_obj_get_content_width(cont);
418 if(lv_obj_get_style_base_dir_internal(tv, LV_PART_MAIN) == LV_BASE_DIR_RTL) t = -(p.x - w / 2) / w;
419 else t = (p.x + w / 2) / w;
420 }
421 else {
422 int32_t h = lv_obj_get_content_height(cont);
423 t = (p.y + h / 2) / h;
424 }
425
426 if(t < 0) t = 0;
427 bool new_tab = false;
428 if(t != (int32_t)lv_tabview_get_tab_active(tv)) new_tab = true;
429
430 /*If not scrolled by an indev set the tab immediately*/
431 if(lv_indev_active()) {
432 lv_tabview_set_active(tv, t, LV_ANIM_ON);
433 }
434 else {
435 lv_tabview_set_active(tv, t, LV_ANIM_OFF);
436 }
437
438 if(new_tab) lv_obj_send_event(tv, LV_EVENT_VALUE_CHANGED, NULL);
439 }
