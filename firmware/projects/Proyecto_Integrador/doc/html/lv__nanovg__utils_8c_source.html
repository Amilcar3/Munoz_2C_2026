128void lv_nanovg_path_append_arc_right_angle(NVGcontext * ctx,
129 float start_x, float start_y,
130 float center_x, float center_y,
131 float end_x, float end_y)
132{
133 LV_PROFILER_DRAW_BEGIN;
134 float dx1 = center_x - start_x;
135 float dy1 = center_y - start_y;
136 float dx2 = end_x - center_x;
137 float dy2 = end_y - center_y;
138
139 float c = SIGN(dx1 * dy2 - dx2 * dy1) * PATH_ARC_MAGIC;
140
141 nvgBezierTo(ctx,
142 start_x - c * dy1, start_y + c * dx1,
143 end_x - c * dy2, end_y + c * dx2,
144 end_x, end_y);
145 LV_PROFILER_DRAW_END;
146}

148void lv_nanovg_path_append_arc(NVGcontext * ctx,
149 float cx, float cy,
150 float radius,
151 float start_angle,
152 float sweep,
153 bool pie)
154{
155 LV_PROFILER_DRAW_BEGIN;
156
157 if(radius <= 0) {
158 LV_PROFILER_DRAW_END;
159 return;
160 }
161
162 /* just circle */
163 if(sweep >= 360.0f || sweep <= -360.0f) {
164 nvgCircle(ctx, cx, cy, radius);
165 LV_PROFILER_DRAW_END;
166 return;
167 }
168
169 start_angle = NVG_MATH_RADIANS(start_angle);
170 sweep = NVG_MATH_RADIANS(sweep);
171
172 int n_curves = (int)ceil(NVG_MATH_FABSF(sweep / NVG_MATH_HALF_PI));
173 float sweep_sign = sweep < 0 ? -1.f : 1.f;
174 float fract = fmodf(sweep, NVG_MATH_HALF_PI);
175 fract = (nvg_math_is_zero(fract)) ? NVG_MATH_HALF_PI * sweep_sign : fract;
176
177 /* Start from here */
178 float start_x = radius * NVG_MATH_COSF(start_angle);
179 float start_y = radius * NVG_MATH_SINF(start_angle);
180
181 if(pie) {
182 nvgMoveTo(ctx, cx, cy);
183 nvgLineTo(ctx, start_x + cx, start_y + cy);
184 }
185
186 for(int i = 0; i < n_curves; ++i) {
187 float end_angle = start_angle + ((i != n_curves - 1) ? NVG_MATH_HALF_PI * sweep_sign : fract);
188 float end_x = radius * NVG_MATH_COSF(end_angle);
189 float end_y = radius * NVG_MATH_SINF(end_angle);
190
191 /* variables needed to calculate bezier control points */
192
196 float ax = start_x;
197 float ay = start_y;
198 float bx = end_x;
199 float by = end_y;
200 float q1 = ax * ax + ay * ay;
201 float q2 = ax * bx + ay * by + q1;
202 float k2 = (4.0f / 3.0f) * ((NVG_MATH_SQRTF(2 * q1 * q2) - q2) / (ax * by - ay * bx));
203
204 /* Next start point is the current end point */
205 start_x = end_x;
206 start_y = end_y;
207
208 end_x += cx;
209 end_y += cy;
210
211 float ctrl1_x = ax - k2 * ay + cx;
212 float ctrl1_y = ay + k2 * ax + cy;
213 float ctrl2_x = bx + k2 * by + cx;
214 float ctrl2_y = by - k2 * bx + cy;
215
216 nvgBezierTo(ctx, ctrl1_x, ctrl1_y, ctrl2_x, ctrl2_y, end_x, end_y);
217 start_angle = end_angle;
218 }
219
220 if(pie) {
221 nvgClosePath(ctx);
222 }
223
224 LV_PROFILER_DRAW_END;
225}

271lv_draw_buf_t * lv_nanovg_reshape_global_image(struct _lv_draw_nanovg_unit_t * u,
272 lv_color_format_t cf,
273 uint32_t w,
274 uint32_t h)
275{
276 LV_ASSERT_NULL(u);
277
278 uint32_t stride = (w * lv_color_format_get_bpp(cf) + 7) >> 3;
279 lv_draw_buf_t * tmp_buf = lv_draw_buf_reshape(u->image_buf, cf, w, h, stride);
280
281 if(!tmp_buf) {
282 if(u->image_buf) {
283 lv_draw_buf_destroy(u->image_buf);
284 u->image_buf = NULL;
285 }
286
287 tmp_buf = lv_draw_buf_create(w, h, cf, stride);
288 if(!tmp_buf) {
289 return NULL;
290 }
291 }
292
293 u->image_buf = tmp_buf;
294
295 return u->image_buf;
296}
