50void lv_draw_pxp_layer(lv_draw_task_t * t)
51{
52 lv_draw_image_dsc_t * draw_dsc = t->draw_dsc;
53
54 lv_layer_t * layer_to_draw = (lv_layer_t *)draw_dsc->src;
55 const lv_draw_buf_t * draw_buf = layer_to_draw->draw_buf;
56
57 /* It can happen that nothing was draw on a layer and therefore its buffer is not allocated.
58 * In this case just return.
59 */
60 if(draw_buf == NULL)
61 return;
62
63 const lv_area_t area_to_draw = {
64 .x1 = 0,
65 .y1 = 0,
66 .x2 = draw_buf->header.w - 1,
67 .y2 = draw_buf->header.h - 1
68 };
69 lv_draw_buf_invalidate_cache(draw_buf, &area_to_draw);
70
71 lv_draw_image_dsc_t new_draw_dsc = *draw_dsc;
72 new_draw_dsc.src = draw_buf;
73 t->draw_dsc = &new_draw_dsc;
74 lv_draw_pxp_img(t);
75 t->draw_dsc = draw_dsc;
76
77#if LV_USE_LAYER_DEBUG || LV_USE_PARALLEL_DRAW_DEBUG
78 const lv_area_t * coords = &t->area;
79 lv_area_t area_rot;
80 area_rot = *coords;
81 if(draw_dsc->rotation || draw_dsc->scale_x != LV_SCALE_NONE || draw_dsc->scale_y != LV_SCALE_NONE) {
82 int32_t w = lv_area_get_width(coords);
83 int32_t h = lv_area_get_height(coords);
84
85 lv_image_buf_get_transformed_area(&area_rot, w, h, draw_dsc->rotation, draw_dsc->scale_x, draw_dsc->scale_y,
86 &draw_dsc->pivot);
87
88 area_rot.x1 += coords->x1;
89 area_rot.y1 += coords->y1;
90 area_rot.x2 += coords->x1;
91 area_rot.y2 += coords->y1;
92 }
93 lv_area_t draw_area;
94 if(!lv_area_intersect(&draw_area, &area_rot, &t->clip_area)) return;
95#endif
96
97#if LV_USE_LAYER_DEBUG
98 lv_draw_fill_dsc_t fill_dsc;
99 lv_draw_fill_dsc_init(&fill_dsc);
100 fill_dsc.color = lv_color_hex(layer_to_draw->color_format == LV_COLOR_FORMAT_ARGB8888 ? 0xff0000 : 0x00ff00);
101 fill_dsc.opa = LV_OPA_20;
102 lv_draw_sw_fill(t, &fill_dsc, &area_rot);
103
104 lv_draw_border_dsc_t border_dsc;
105 lv_draw_border_dsc_init(&border_dsc);
106 border_dsc.color = fill_dsc.color;
107 border_dsc.opa = LV_OPA_60;
108 border_dsc.width = 2;
109 lv_draw_sw_border(t, &border_dsc, &area_rot);
110
111#endif
112
113#if LV_USE_PARALLEL_DRAW_DEBUG
114 int32_t idx = t->draw_unit->idx;
115
116 lv_draw_fill_dsc_t fill_dsc;
117 lv_draw_rect_dsc_init(&fill_dsc);
118 fill_dsc.color = lv_palette_main(idx % LV_PALETTE_LAST);
119 fill_dsc.opa = LV_OPA_10;
120 lv_draw_sw_fill(t, &fill_dsc, &area_rot);
121
122 lv_draw_border_dsc_t border_dsc;
123 lv_draw_border_dsc_init(&border_dsc);
124 border_dsc.color = lv_palette_main(idx % LV_PALETTE_LAST);
125 border_dsc.opa = LV_OPA_100;
126 border_dsc.width = 2;
127 lv_draw_sw_border(t, &border_dsc, &area_rot);
128
129 lv_point_t txt_size;
130 lv_text_get_size_attributes(&txt_size, "W", LV_FONT_DEFAULT, 0, 0, 100, LV_TEXT_FLAG_NONE);
131
132 lv_area_t txt_area;
133 txt_area.x1 = draw_area.x1;
134 txt_area.x2 = draw_area.x1 + txt_size.x - 1;
135 txt_area.y2 = draw_area.y2;
136 txt_area.y1 = draw_area.y2 - txt_size.y + 1;
137
138 lv_draw_fill_dsc_init(&fill_dsc);
139 fill_dsc.color = lv_color_black();
140 lv_draw_sw_fill(t, &fill_dsc, &txt_area);
141
142 char buf[8];
143 lv_snprintf(buf, sizeof(buf), "%d", idx);
144 lv_draw_label_dsc_t label_dsc;
145 lv_draw_label_dsc_init(&label_dsc);
146 label_dsc.color = lv_color_white();
147 label_dsc.text = buf;
148 lv_draw_sw_label(t, &label_dsc, &txt_area);
149#endif
150}
