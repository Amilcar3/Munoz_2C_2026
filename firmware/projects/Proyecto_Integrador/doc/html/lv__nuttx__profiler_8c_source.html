61void lv_nuttx_profiler_init(void)
62{
63 cpu_freq = (uint32_t)up_perf_getfreq() / 1000000;
64 if(cpu_freq == 0) {
65 LV_LOG_ERROR("Failed to get CPU frequency");
66 return;
67 }
68 LV_LOG_USER("CPU frequency: %" LV_PRIu32 " MHz", cpu_freq);
69
70#if LV_USE_NUTTX_TRACE_FILE
71 trace_fd = -1;
72#endif
73
74 lv_profiler_builtin_config_t config;
75 lv_profiler_builtin_config_init(&config);
76 config.tick_per_sec = 1000000000; /* 1 sec = 1000000000 nsec */
77 config.tick_get_cb = tick_get_cb;
78 config.flush_cb = flush_cb;
79 lv_profiler_builtin_init(&config);
80}

111static uint64_t tick_get_cb(void)
112{
113 static uint32_t prev_tick = 0;
114 static uint64_t cur_tick_ns = 0;
115 uint32_t act_time = up_perf_gettime();
116 uint64_t elaps;
117
118 /*If there is no overflow in sys_time simple subtract*/
119 if(act_time >= prev_tick) {
120 elaps = act_time - prev_tick;
121 }
122 else {
123 elaps = UINT32_MAX - prev_tick + 1;
124 elaps += act_time;
125 }
126
127 cur_tick_ns += TICK_TO_NSEC(elaps);
128 prev_tick = act_time;
129 return cur_tick_ns;
130}
