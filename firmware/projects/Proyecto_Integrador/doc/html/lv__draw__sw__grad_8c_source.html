90static lv_draw_sw_grad_calc_t * allocate_item(const lv_grad_dsc_t * g, int32_t w, int32_t h)
91{
92 int32_t size;
93 switch(g->dir) {
94 case LV_GRAD_DIR_HOR:
95 case LV_GRAD_DIR_LINEAR:
96 case LV_GRAD_DIR_RADIAL:
97 case LV_GRAD_DIR_CONICAL:
98 size = w;
99 break;
100 case LV_GRAD_DIR_VER:
101 size = h;
102 break;
103 default:
104 size = 64;
105 }
106
107 size_t req_size = ALIGN(sizeof(lv_draw_sw_grad_calc_t)) + ALIGN(size * sizeof(lv_color_t)) + ALIGN(size * sizeof(
108 lv_opa_t));
109 lv_draw_sw_grad_calc_t * item = lv_malloc(req_size);
110 LV_ASSERT_MALLOC(item);
111 if(item == NULL) return NULL;
112
113 uint8_t * p = (uint8_t *)item;
114 item->color_map = (lv_color_t *)(p + ALIGN(sizeof(*item)));
115 item->opa_map = (lv_opa_t *)(p + ALIGN(sizeof(*item)) + ALIGN(size * sizeof(lv_color_t)));
116 item->size = size;
117 return item;
118}

122static inline int32_t extend_w(int32_t w, lv_grad_extend_t extend)
123{
124 if(extend == LV_GRAD_EXTEND_PAD) {
125 return w < 0 ? 0 : LV_MIN(w, 255);
126 }
127 if(extend == LV_GRAD_EXTEND_REPEAT) {
128 return w & 255;
129 }
130 /*LV_GRAD_EXTEND_REFLECT*/
131 w &= 511;
132 if(w > 255)
133 w ^= 511; /* 511 - w */
134 return w;
135}

143lv_draw_sw_grad_calc_t * lv_draw_sw_grad_get(const lv_grad_dsc_t * g, int32_t w, int32_t h)
144{
145 /* No gradient, no cache */
146 if(g->dir == LV_GRAD_DIR_NONE) return NULL;
147
148 /* Step 1: Search cache for the given key */
149 lv_draw_sw_grad_calc_t * item = allocate_item(g, w, h);
150 if(item == NULL) {
151 LV_LOG_WARN("Failed to allocate item for the gradient");
152 return item;
153 }
154
155 /* Step 3: Fill it with the gradient, as expected */
156 uint32_t i;
157 for(i = 0; i < item->size; i++) {
158 lv_draw_sw_grad_color_calculate(g, item->size, i, &item->color_map[i], &item->opa_map[i]);
159 }
160 return item;
161}

163void LV_ATTRIBUTE_FAST_MEM lv_draw_sw_grad_color_calculate(const lv_grad_dsc_t * dsc, int32_t range,
164 int32_t frac, lv_color_t * color_out, lv_opa_t * opa_out)
165{
166 lv_color_t tmp;
167 /*Clip out-of-bounds first*/
168 int32_t min = (dsc->stops[0].frac * range) >> 8;
169 if(frac <= min) {
170 GRAD_CONV(tmp, dsc->stops[0].color);
171 *color_out = tmp;
172 *opa_out = dsc->stops[0].opa;
173 return;
174 }
175
176 int32_t max = (dsc->stops[dsc->stops_count - 1].frac * range) >> 8;
177 if(frac >= max) {
178 GRAD_CONV(tmp, dsc->stops[dsc->stops_count - 1].color);
179 *color_out = tmp;
180 *opa_out = dsc->stops[dsc->stops_count - 1].opa;
181 return;
182 }
183
184 /*Find the 2 closest stop now*/
185 int32_t d = 0;
186 int32_t found_i = 0;
187 for(uint8_t i = 1; i < dsc->stops_count; i++) {
188 int32_t cur = (dsc->stops[i].frac * range) >> 8;
189 if(frac <= cur) {
190 found_i = i;
191 break;
192 }
193 }
194
195 LV_ASSERT(found_i != 0);
196
197 lv_color_t one, two;
198 one = dsc->stops[found_i - 1].color;
199 two = dsc->stops[found_i].color;
200 min = (dsc->stops[found_i - 1].frac * range) >> 8;
201 max = (dsc->stops[found_i].frac * range) >> 8;
202 d = max - min;
203
204 /*Then interpolate*/
205 frac -= min;
206 lv_opa_t mix = (frac * 255) / d;
207 lv_opa_t imix = 255 - mix;
208
209 *color_out = GRAD_CM(LV_UDIV255(two.red * mix + one.red * imix),
210 LV_UDIV255(two.green * mix + one.green * imix),
211 LV_UDIV255(two.blue * mix + one.blue * imix));
212
213 *opa_out = LV_UDIV255(dsc->stops[found_i].opa * mix + dsc->stops[found_i - 1].opa * imix);
214}

224/*
225 Calculate radial gradient based on the following equation:
226
227 | P - (C1 - C0)w - C0 | = (r1 - r0)w + r0, where
228
229 P: {xp, yp} is the point of interest
230 C0: {x0, y0} is the center of the start circle
231 C1: {x1, y1} is the center of the end circle
232 r0 is the radius of the start circle
233 r1 is the radius of the end circle
234 w is the unknown variable
235 || is the length of the vector
236
237 The above equation can be rewritten as:
238
239 ((r1-r0)^2 - (x1-x0)^2 - (y1-y0)^2) * w^2 + 2*((xp-x0)*(x1-x0) + (yp-y0)*(y1-y0)) * w + (-(xp-x0)^2 - (yp-y0)^) = 0
240
241 The roots of the quadratical equation can be obtained using the well-known formula (-b +- sqrt(b^2 - 4ac)) / 2a
242 We only need the more positive root.
243
244 Let's denote
245 dx = x1 - x0
246 dy = y1 - y0
247 dr = r1 - r0
248
249 Thus:
250
251 w = (-b(xp, yp) + sqrt(sqr(b(xp, yp)) - 4 * a * c(xp, yp))) / (2 * a), where
252
253 b(xp, yp) = 2dx * xp + 2dy * yp + 2(r0 * dr - x0 * dx - y0 * dy)
254 c(xp, yp) = r0^2 - (xp - x0)^2 - (yp - y0)^2
255
256 Rewrite b(xp, yp) as:
257
258 b(xp, yp) = xp * bpx + yp * bpy + bc, where
259
260 bpx = 2dx
261 bpy = 2dy
262 bc = 2(r0 * dr - x0 * dx - y0 * dy)
263
264 We can pre-calculate the constants, because they do not depend on the pixel coordinates.
265
266*/
267
268void lv_draw_sw_grad_radial_setup(lv_grad_dsc_t * dsc, const lv_area_t * coords)
269{
270 lv_point_t start = dsc->params.radial.focal;
271 lv_point_t end = dsc->params.radial.end;
272 lv_point_t start_extent = dsc->params.radial.focal_extent;
273 lv_point_t end_extent = dsc->params.radial.end_extent;
274 lv_grad_radial_state_t * state = lv_malloc(sizeof(lv_grad_radial_state_t));
275 dsc->state = state;
276
277 /* Convert from percentage coordinates */
278 int32_t wdt = lv_area_get_width(coords);
279 int32_t hgt = lv_area_get_height(coords);
280
281 start.x = lv_pct_to_px(start.x, wdt);
282 end.x = lv_pct_to_px(end.x, wdt);
283 start_extent.x = lv_pct_to_px(start_extent.x, wdt);
284 end_extent.x = lv_pct_to_px(end_extent.x, wdt);
285 start.y = lv_pct_to_px(start.y, hgt);
286 end.y = lv_pct_to_px(end.y, hgt);
287 start_extent.y = lv_pct_to_px(start_extent.y, hgt);
288 end_extent.y = lv_pct_to_px(end_extent.y, hgt);
289
290 /* Calculate radii */
291 int16_t r_start = lv_sqrt32(lv_sqr(start_extent.x - start.x) + lv_sqr(start_extent.y - start.y));
292 int16_t r_end = lv_sqrt32(lv_sqr(end_extent.x - end.x) + lv_sqr(end_extent.y - end.y));
293 LV_ASSERT(r_end != 0);
294
295 /* Create gradient color map */
296 state->cgrad = lv_draw_sw_grad_get(dsc, 256, 0);
297
298 state->x0 = start.x;
299 state->y0 = start.y;
300 state->r0 = r_start;
301 int32_t dr = r_end - r_start;
302 if(end.x == start.x && end.y == start.y) {
303 LV_ASSERT(dr != 0);
304 state->a4 = lv_sqr(dr) << 2;
305 state->bpx = 0;
306 state->bpy = 0;
307 state->bc = (state->r0 * dr) << 1;
308 state->dx = 0;
309 state->inv_dr = (1 << (8 + 16)) / dr;
310 }
311 else {
312 int32_t dx = end.x - start.x;
313 int32_t dy = end.y - start.y;
314 state->dx = dx; /* needed for incremental calculation */
315 state->a4 = (lv_sqr(dr) - lv_sqr(dx) - lv_sqr(dy)) << 2;
316 /* b(xp, yp) = xp * bpx + yp * bpy + bc */
317 state->bpx = dx << 1;
318 state->bpy = dy << 1;
319 state->bc = (state->r0 * dr - state->x0 * dx - state->y0 * dy) << 1;
320 }
321 state->inv_a4 = state->a4 != 0 ? (1 << (13 + 16)) / state->a4 : 0;
322 /* check for possible clipping */
323 if(dsc->extend == LV_GRAD_EXTEND_PAD &&
324 /* if extend mode is 'pad', then we can clip to the end circle's bounding box, if the start circle is entirely within the end circle */
325 (lv_sqr(start.x - end.x) + lv_sqr(start.y - end.y) < lv_sqr(r_end - r_start))) {
326 if(r_end > r_start) {
327 lv_area_set(&state->clip_area, end.x - r_end, end.y - r_end, end.x + r_end, end.y + r_end);
328 }
329 else {
330 lv_area_set(&state->clip_area, start.x - r_start, start.y - r_start, start.x + r_start, start.y + r_start);
331 }
332 }
333 else {
334 state->clip_area.x1 = -0x7fffffff;
335 }
336}

348void LV_ATTRIBUTE_FAST_MEM lv_draw_sw_grad_radial_get_line(lv_grad_dsc_t * dsc, int32_t xp, int32_t yp,
349 int32_t width, lv_draw_sw_grad_calc_t * result)
350{
351 lv_grad_radial_state_t * state = (lv_grad_radial_state_t *)dsc->state;
352 lv_color_t * buf = result->color_map;
353 lv_opa_t * opa = result->opa_map;
354 lv_draw_sw_grad_calc_t * grad = state->cgrad;
355
356 int32_t w; /* the result: this is an offset into the 256 element gradient color table */
357 int32_t b, db, c, dc;
358
359 /* check for possible clipping */
360 if(state->clip_area.x1 != -0x7fffffff) {
361 /* fill line with end color for pixels outside the clipped region */
362 lv_color_t * _buf = buf;
363 lv_opa_t * _opa = opa;
364 lv_color_t _c = grad->color_map[255];
365 lv_opa_t _o = grad->opa_map[255];
366 int32_t _w = width;
367 for(; _w > 0; _w--) {
368 *_buf++ = _c;
369 *_opa++ = _o;
370 }
371 /* is this line fully outside the clip area? */
372 if(yp < state->clip_area.y1 ||
373 yp >= state->clip_area.y2 ||
374 xp >= state->clip_area.x2 ||
375 xp + width < state->clip_area.x1) {
376 return;
377 }
378 else { /* not fully outside: clip line to the bounding box */
379 int32_t _x1 = LV_MAX(xp, state->clip_area.x1);
380 int32_t _x2 = LV_MIN(xp + width, state->clip_area.x2);
381 buf += _x1 - xp;
382 opa += _x1 - xp;
383 xp = _x1;
384 width = _x2 - _x1;
385 }
386 }
387
388 b = xp * state->bpx + yp * state->bpy + state->bc;
389 c = lv_sqr(state->r0) - lv_sqr(xp - state->x0) - lv_sqr(yp - state->y0);
390 /* We can save some calculations by using the previous values of b and c */
391 db = state->dx << 1;
392 dc = ((xp - state->x0) << 1) + 1;
393
394 if(state->a4 == 0) { /* not a quadratic equation: solve linear equation: w = -c/b */
395 for(; width > 0; width--) {
396 w = extend_w(b == 0 ? 0 : -(c << 8) / b, dsc->extend);
397 *buf++ = grad->color_map[w];
398 *opa++ = grad->opa_map[w];
399 b += db;
400 c -= dc;
401 dc += 2;
402 }
403 }
404 else { /* solve quadratical equation */
405 if(state->bpx ||
406 state->bpy) { /* general case (circles are not concentric): w = (-b + sqrt(b^2 - 4ac))/2a (we only need the more positive root)*/
407 int32_t a4 = state->a4 >> 4;
408 for(; width > 0; width--) {
409 int32_t det = lv_sqr(b >> 4) - (a4 * (c >> 4)); /* b^2 shifted down by 2*4=8, 4ac shifted down by 8 */
410 /* check determinant: if negative, then there is no solution: use starting color */
411 w = det < 0 ? 0 : extend_w(((lv_sqrt32(det) - (b >> 4)) * state->inv_a4) >> 16,
412 dsc->extend); /* square root shifted down by 4 (includes *256 to set output range) */
413 *buf++ = grad->color_map[w];
414 *opa++ = grad->opa_map[w];
415 b += db;
416 c -= dc;
417 dc += 2;
418 }
419 }
420 else { /* special case: concentric circles: w = (sqrt((xp-x0)^2 + (yx-y0)^2)-r0)/(r1-r0) */
421 c = lv_sqr(xp - state->x0) + lv_sqr(yp - state->y0);
422 for(; width > 0; width--) {
423 w = extend_w((((lv_sqrt32(c) - state->r0)) * state->inv_dr) >> 16, dsc->extend);
424 *buf++ = grad->color_map[w];
425 *opa++ = grad->opa_map[w];
426 c += dc;
427 dc += 2;
428 }
429 }
430 }
431}

433/*
434 Calculate linear gradient based on the following equation:
435
436 w = ((P - C0) x (C1 - C0)) / | C1 - C0 |^2, where
437
438 P: {xp, yp} is the point of interest
439 C0: {x0, y0} is the start point of the gradient vector
440 C1: {x1, y1} is the end point of the gradient vector
441 w is the unknown variable
442
443 || is the length of the vector
444 x is a dot product
445
446 The above equation can be rewritten as:
447
448 w = xp * (dx / (dx^2 + dy^2)) + yp * (dy / (dx^2 + dy^2)) - (x0 * dx + y0 * dy) / (dx^2 + dy^2), where
449
450 dx = x1 - x0
451 dy = y1 - y0
452
453 We can pre-calculate the constants, because they do not depend on the pixel coordinates.
454
455*/
456
457void lv_draw_sw_grad_linear_setup(lv_grad_dsc_t * dsc, const lv_area_t * coords)
458{
459 lv_point_t start = dsc->params.linear.start;
460 lv_point_t end = dsc->params.linear.end;
461 lv_grad_linear_state_t * state = lv_malloc(sizeof(lv_grad_linear_state_t));
462 dsc->state = state;
463
464 /* Create gradient color map */
465 state->cgrad = lv_draw_sw_grad_get(dsc, 256, 0);
466
467 /* Convert from percentage coordinates */
468 int32_t wdt = lv_area_get_width(coords);
469 int32_t hgt = lv_area_get_height(coords);
470
471 start.x = lv_pct_to_px(start.x, wdt);
472 end.x = lv_pct_to_px(end.x, wdt);
473 start.y = lv_pct_to_px(start.y, hgt);
474 end.y = lv_pct_to_px(end.y, hgt);
475
476 /* Precalculate constants */
477 int32_t dx = end.x - start.x;
478 int32_t dy = end.y - start.y;
479
480 int32_t l2 = lv_sqr(dx) + lv_sqr(dy);
481 if(l2 == 0) l2 = 1;
482 state->a = (dx << 16) / l2;
483 state->b = (dy << 16) / l2;
484 state->c = ((start.x * dx + start.y * dy) << 16) / l2;
485}

497void LV_ATTRIBUTE_FAST_MEM lv_draw_sw_grad_linear_get_line(lv_grad_dsc_t * dsc, int32_t xp, int32_t yp,
498 int32_t width, lv_draw_sw_grad_calc_t * result)
499{
500 lv_grad_linear_state_t * state = (lv_grad_linear_state_t *)dsc->state;
501 lv_color_t * buf = result->color_map;
502 lv_opa_t * opa = result->opa_map;
503 lv_draw_sw_grad_calc_t * grad = state->cgrad;
504
505 int32_t w; /* the result: this is an offset into the 256 element gradient color table */
506 int32_t x, d;
507
508 x = xp * state->a + yp * state->b - state->c;
509 d = state->a;
510
511 for(; width > 0; width--) {
512 w = extend_w(x >> 8, dsc->extend);
513 *buf++ = grad->color_map[w];
514 *opa++ = grad->opa_map[w];
515 x += d;
516 }
517}

519/*
520 Calculate conical gradient based on the following equation:
521
522 w = (atan((yp - y0)/(xp - x0)) - alpha) / (beta - alpha), where
523
524 P: {xp, yp} is the point of interest
525 C0: {x0, y0} is the center of the gradient
526 alpha is the start angle
527 beta is the end angle
528 w is the unknown variable
529*/
530
531void lv_draw_sw_grad_conical_setup(lv_grad_dsc_t * dsc, const lv_area_t * coords)
532{
533 lv_point_t c0 = dsc->params.conical.center;
534 int32_t alpha = dsc->params.conical.start_angle % 360;
535 int32_t beta = dsc->params.conical.end_angle % 360;
536 lv_grad_conical_state_t * state = lv_malloc(sizeof(lv_grad_conical_state_t));
537 dsc->state = state;
538
539 /* Create gradient color map */
540 state->cgrad = lv_draw_sw_grad_get(dsc, 256, 0);
541
542 /* Convert from percentage coordinates */
543 int32_t wdt = lv_area_get_width(coords);
544 int32_t hgt = lv_area_get_height(coords);
545
546 c0.x = lv_pct_to_px(c0.x, wdt);
547 c0.y = lv_pct_to_px(c0.y, hgt);
548
549 /* Precalculate constants */
550 if(beta <= alpha)
551 beta += 360;
552 state->x0 = c0.x;
553 state->y0 = c0.y;
554 state->a = alpha;
555 state->da = beta - alpha;
556 state->inv_da = (1 << 16) / (beta - alpha);
557}

569void LV_ATTRIBUTE_FAST_MEM lv_draw_sw_grad_conical_get_line(lv_grad_dsc_t * dsc, int32_t xp, int32_t yp,
570 int32_t width, lv_draw_sw_grad_calc_t * result)
571{
572 lv_grad_conical_state_t * state = (lv_grad_conical_state_t *)dsc->state;
