200static void grid_update(lv_obj_t * cont, void * user_data)
201{
202 LV_ASSERT(cont != NULL);
203 LV_LOG_INFO("update %p container", (void *)cont);
204 LV_UNUSED(user_data);
205
206 lv_grid_calc_t c;
207 lv_result_t res = calc(cont, &c);
208 if(res != LV_RESULT_OK) {
209 calc_free(&c);
210 return;
211 }
212
213 item_repos_hint_t hint;
214 lv_memzero(&hint, sizeof(hint));
215
216 /*Calculate the grids absolute x and y coordinates.
217 *It will be used as helper during item repositioning to avoid calculating this value for every children*/
218 int32_t pad_left = lv_obj_get_style_space_left_internal(cont, LV_PART_MAIN);
219 int32_t pad_top = lv_obj_get_style_space_top_internal(cont, LV_PART_MAIN);
220 hint.grid_abs.x = pad_left + cont->coords.x1 - lv_obj_get_scroll_x(cont);
221 hint.grid_abs.y = pad_top + cont->coords.y1 - lv_obj_get_scroll_y(cont);
222
223 uint32_t i;
224 for(i = 0; i < cont->spec_attr->child_cnt; i++) {
225 lv_obj_t * item = cont->spec_attr->children[i];
226 item_repos(item, &c, &hint);
227 }
228 calc_free(&c);
229
230 int32_t w_set = lv_obj_get_style_width_internal(cont, LV_PART_MAIN);
231 int32_t h_set = lv_obj_get_style_height_internal(cont, LV_PART_MAIN);
232 if(w_set == LV_SIZE_CONTENT || h_set == LV_SIZE_CONTENT) {
233 lv_obj_refr_size(cont);
234 }
235
236 lv_obj_send_event(cont, LV_EVENT_LAYOUT_CHANGED, NULL);
237
238 LV_TRACE_LAYOUT("finished");
239}

247static lv_result_t calc(lv_obj_t * cont, lv_grid_calc_t * calc_out)
248{
249 LV_ASSERT(cont != NULL);
250 LV_ASSERT(calc_out != NULL);
251 lv_memzero(calc_out, sizeof(lv_grid_calc_t));
252 if(lv_obj_get_child(cont, 0) == NULL) {
253 return LV_RESULT_INVALID;
254 }
255
256 if(calc_rows(cont, calc_out) == LV_RESULT_INVALID) {
257 /* Warning is already logged inside `calc_rows` */
258 return LV_RESULT_INVALID;
259 }
260 if(calc_cols(cont, calc_out) == LV_RESULT_INVALID) {
261 /* Warning is already logged inside `calc_cols` */
262 return LV_RESULT_INVALID;
263 }
264
265 int32_t col_gap = lv_obj_get_style_pad_column_internal(cont, LV_PART_MAIN);
266 int32_t row_gap = lv_obj_get_style_pad_row_internal(cont, LV_PART_MAIN);
267
268 bool rev = lv_obj_get_style_base_dir_internal(cont, LV_PART_MAIN) == LV_BASE_DIR_RTL;
269
270 int32_t w_set = lv_obj_get_style_width_internal(cont, LV_PART_MAIN);
271 int32_t h_set = lv_obj_get_style_height_internal(cont, LV_PART_MAIN);
272 bool auto_w = w_set == LV_SIZE_CONTENT && !cont->w_layout;
273 int32_t cont_w = lv_obj_get_content_width(cont);
274 calc_out->grid_w = grid_align(cont_w, auto_w, get_grid_col_align(cont), col_gap, calc_out->col_num, calc_out->w,
275 calc_out->x, rev);
276
277 bool auto_h = h_set == LV_SIZE_CONTENT && !cont->h_layout;
278 int32_t cont_h = lv_obj_get_content_height(cont);
279 calc_out->grid_h = grid_align(cont_h, auto_h, get_grid_row_align(cont), row_gap, calc_out->row_num, calc_out->h,
280 calc_out->y, false);
281
282 LV_ASSERT_MEM_INTEGRITY();
283 return LV_RESULT_OK;
284}

299static lv_result_t calc_cols(lv_obj_t * cont, lv_grid_calc_t * c)
300{
301 LV_ASSERT(cont != NULL);
302 LV_ASSERT(c != NULL);
303 const int32_t * col_templ;
304 col_templ = get_col_dsc(cont);
305
306 /*If there is no descriptor check if it's a subgrid*/
307 bool subgrid = false;
308 if(col_templ == NULL) {
309 lv_obj_t * parent = lv_obj_get_parent(cont);
310 if(parent == NULL) {
311 LV_LOG_WARN("No column descriptor, and there is no parent for a screen to process subgrid");
312 return LV_RESULT_INVALID;
313 }
314
315 col_templ = get_col_dsc(parent);
316 if(col_templ == NULL) {
317 LV_LOG_WARN("No column descriptor found even on the parent");
318 return LV_RESULT_INVALID;
319 }
320
321 int32_t pos = get_col_pos(cont);
322 int32_t span = get_col_span(cont);
323
324 int32_t * col_templ_sub = lv_malloc(sizeof(int32_t) * (span + 1));
325 lv_memcpy(col_templ_sub, &col_templ[pos], sizeof(int32_t) * span);
326 col_templ_sub[span] = LV_GRID_TEMPLATE_LAST;
327 col_templ = col_templ_sub;
328 subgrid = true;
329 }
330
331 int32_t cont_w = lv_obj_get_content_width(cont);
332
333 c->col_num = count_tracks(col_templ);
334 c->x = lv_malloc(sizeof(int32_t) * c->col_num);
335 c->w = lv_malloc(sizeof(int32_t) * c->col_num);
336
337 /*Set sizes for CONTENT cells*/
338 uint32_t i;
339 for(i = 0; i < c->col_num; i++) {
340 int32_t size = LV_COORD_MIN;
341 if(IS_CONTENT(col_templ[i])) {
342 /*Check the size of children of this cell*/
343 uint32_t ci;
344 for(ci = 0; ci < lv_obj_get_child_count(cont); ci++) {
345 lv_obj_t * item = lv_obj_get_child(cont, ci);
346 if((lv_obj_is_ignore_layout(item) || lv_obj_is_hidden(item) || lv_obj_is_floating(item))) continue;
347 uint32_t col_span = get_col_span(item);
348 if(col_span != 1) continue;
349
350 uint32_t col_pos = get_col_pos(item);
351 if(col_pos != i) continue;
352
353 size = LV_MAX(size, lv_obj_get_width(item));
354 }
355 if(size >= 0) c->w[i] = size;
356 else c->w[i] = 0;
357 }
358 }
359
360 uint32_t col_fr_cnt = 0;
361 int32_t grid_w = 0;
362
363 for(i = 0; i < c->col_num; i++) {
364 int32_t x = col_templ[i];
365 if(IS_FR(x)) {
366 col_fr_cnt += GET_FR(x);
367 }
368 else if(IS_CONTENT(x)) {
369 grid_w += c->w[i];
370 }
371 else {
372 c->w[i] = x;
373 grid_w += x;
374 }
375 }
376
377 int32_t col_gap = lv_obj_get_style_pad_column_internal(cont, LV_PART_MAIN);
378 cont_w -= col_gap * (c->col_num - 1);
379 int32_t free_w = cont_w - grid_w;
380 if(free_w < 0) free_w = 0;
381
382 for(i = 0; i < c->col_num && col_fr_cnt; i++) {
383 int32_t x = col_templ[i];
384 if(IS_FR(x)) {
385 int32_t f = GET_FR(x);
386 c->w[i] = lv_div_round_closest(free_w * f, col_fr_cnt);
387 /*By updating remaining fr and width, we ensure f == col_fr_cnt
388 *in the last loop iteration. That means the last iteration will
389 *not have rounding errors and use all remaining space.*/
390 col_fr_cnt -= f;
391 free_w -= c->w[i];
392 }
393 }
394
395 if(subgrid) {
396 lv_free((void *)col_templ);
397 }
398 return LV_RESULT_OK;
399}

401static lv_result_t calc_rows(lv_obj_t * cont, lv_grid_calc_t * c)
402{
403 LV_ASSERT(cont != NULL);
404 LV_ASSERT(c != NULL);
405 const int32_t * row_templ;
406 row_templ = get_row_dsc(cont);
407
408 /*If there is no descriptor check if it's a subgrid*/
409 bool subgrid = false;
410 if(row_templ == NULL) {
411 lv_obj_t * parent = lv_obj_get_parent(cont);
412 if(parent == NULL) {
413 LV_LOG_WARN("No row descriptor, and there is no parent for a screen to process subgrid");
414 return LV_RESULT_INVALID;
415 }
416
417 row_templ = get_row_dsc(parent);
418 if(row_templ == NULL) {
419 LV_LOG_WARN("No row descriptor found even on the parent");
420 return LV_RESULT_INVALID;
421 }
422
423 int32_t pos = get_row_pos(cont);
424 int32_t span = get_row_span(cont);
425
426 int32_t * row_templ_sub = lv_malloc(sizeof(int32_t) * (span + 1));
427 lv_memcpy(row_templ_sub, &row_templ[pos], sizeof(int32_t) * span);
428 row_templ_sub[span] = LV_GRID_TEMPLATE_LAST;
429 row_templ = row_templ_sub;
430 subgrid = true;
431 }
432
433 c->row_num = count_tracks(row_templ);
434 c->y = lv_malloc(sizeof(int32_t) * c->row_num);
435 c->h = lv_malloc(sizeof(int32_t) * c->row_num);
436 /*Set sizes for CONTENT cells*/
437 uint32_t i;
438 for(i = 0; i < c->row_num; i++) {
439 int32_t size = LV_COORD_MIN;
440 if(IS_CONTENT(row_templ[i])) {
441 /*Check the size of children of this cell*/
442 uint32_t ci;
443 for(ci = 0; ci < lv_obj_get_child_count(cont); ci++) {
444 lv_obj_t * item = lv_obj_get_child(cont, ci);
445 if((lv_obj_is_ignore_layout(item) || lv_obj_is_hidden(item) || lv_obj_is_floating(item))) continue;
446 uint32_t row_span = get_row_span(item);
447 if(row_span != 1) continue;
448
449 uint32_t row_pos = get_row_pos(item);
450 if(row_pos != i) continue;
451
452 size = LV_MAX(size, lv_obj_get_height(item));
453 }
454 if(size >= 0) c->h[i] = size;
455 else c->h[i] = 0;
456 }
457 }
458
459 uint32_t row_fr_cnt = 0;
460 int32_t grid_h = 0;
461
462 for(i = 0; i < c->row_num; i++) {
463 int32_t x = row_templ[i];
464 if(IS_FR(x)) {
465 row_fr_cnt += GET_FR(x);
466 }
467 else if(IS_CONTENT(x)) {
468 grid_h += c->h[i];
469 }
470 else {
471 c->h[i] = x;
472 grid_h += x;
473 }
474 }
475
476 int32_t row_gap = lv_obj_get_style_pad_row_internal(cont, LV_PART_MAIN);
477 int32_t cont_h = lv_obj_get_content_height(cont) - row_gap * (c->row_num - 1);
478 int32_t free_h = cont_h - grid_h;
479 if(free_h < 0) free_h = 0;
480
481 for(i = 0; i < c->row_num && row_fr_cnt; i++) {
482 int32_t x = row_templ[i];
483 if(IS_FR(x)) {
484 int32_t f = GET_FR(x);
485 c->h[i] = lv_div_round_closest(free_h * f, row_fr_cnt);
486 /*By updating remaining fr and height, we ensure f == row_fr_cnt
487 *in the last loop iteration. That means the last iteration will
488 *not have rounding errors and use all remaining space.*/
489 row_fr_cnt -= f;
490 free_h -= c->h[i];
491 }
492 }
493
494 if(subgrid) {
495 lv_free((void *)row_templ);
496 }
497 return LV_RESULT_OK;
498}

506static void item_repos(lv_obj_t * item, lv_grid_calc_t * c, item_repos_hint_t * hint)
507{
508 LV_ASSERT(c != NULL);
509 LV_ASSERT(item != NULL);
510 LV_ASSERT(hint != NULL);
511
512 if(lv_obj_is_ignore_layout(item) || lv_obj_is_hidden(item) || lv_obj_is_floating(item)) return;
513
514 int32_t col_span = get_col_span(item);
515 if(col_span <= 0) {
516 LV_LOG_WARN("Column span was %" LV_PRId32 ", setting it to 1", col_span);
517 col_span = 1;
518 }
519
520 int32_t row_span = get_row_span(item);
521 if(row_span <= 0) {
522 LV_LOG_WARN("Row span was %" LV_PRId32 ", setting it to 1", row_span);
523 row_span = 1;
524 }
525
526 bool rev = lv_obj_get_style_base_dir_internal(lv_obj_get_parent(item), LV_PART_MAIN) == LV_BASE_DIR_RTL;
527
528 int32_t col_pos = get_col_pos(item);
529 if(col_pos < 0) {
530 LV_LOG_WARN("Column position was %" LV_PRId32 ", setting it to 0", col_pos);
531 col_pos = 0;
532 }
533
534 if(col_pos >= (int32_t)c->col_num) {
535 LV_LOG_WARN("Column position was %" LV_PRId32 ", setting to %" LV_PRId32, col_pos, c->col_num - 1);
536 col_pos = c->col_num - 1;
537 }
538
539 int32_t row_pos = get_row_pos(item);
540 if(row_pos < 0) {
541 LV_LOG_WARN("Row position was %" LV_PRId32 ", setting it to 0", row_pos);
542 row_pos = 0;
543 }
544
545 if(row_pos >= (int32_t)c->row_num) {
546 LV_LOG_WARN("Row position was %" LV_PRId32 ", setting to %" LV_PRId32, row_pos, c->row_num - 1);
547 row_pos = c->row_num - 1;
548 }
549
550 if(col_pos + col_span > (int32_t)c->col_num) {
551 col_span = c->col_num - col_pos;
552 LV_LOG_WARN("Column span is too large, limiting it to %" LV_PRId32, col_span);
553 }
554
555 if(row_pos + row_span > (int32_t)c->row_num) {
556 row_span = c->row_num - row_pos;
557 LV_LOG_WARN("Row span is too large, limiting it to %" LV_PRId32, row_span);
558 }
559
560 lv_grid_align_t col_align = get_cell_col_align(item);
561 lv_grid_align_t row_align = get_cell_row_align(item);
562
563 int32_t col_x1 = 0;
564 int32_t col_x2 = 0;
565 int32_t col_w = 0;
566
567 if(rev && col_span > 1) {
568 col_x1 = c->x[col_pos + col_span - 1];
569 col_x2 = c->x[col_pos] + c->w[col_pos];
570 col_w = col_x2 - col_x1;
571 }
572 else {
573 col_x1 = c->x[col_pos];
574 col_x2 = c->x[col_pos + col_span - 1] + c->w[col_pos + col_span - 1];
575 col_w = col_x2 - col_x1;
576 }
577
578 int32_t row_y1 = c->y[row_pos];
579 int32_t row_y2 = c->y[row_pos + row_span - 1] + c->h[row_pos + row_span - 1];
580 int32_t row_h = row_y2 - row_y1;
581
582 /*If the item has RTL base dir switch start and end*/
583 if(lv_obj_get_style_base_dir_internal(item, LV_PART_MAIN) == LV_BASE_DIR_RTL) {
584 if(col_align == LV_GRID_ALIGN_START) col_align = LV_GRID_ALIGN_END;
585 else if(col_align == LV_GRID_ALIGN_END) col_align = LV_GRID_ALIGN_START;
586 }
587
588 int32_t x;
589 int32_t y;
590 int32_t item_w = lv_area_get_width(&item->coords);
591 int32_t item_h = lv_area_get_height(&item->coords);
592
593 col_pos = rev && col_span > 1 ? col_pos + col_span - 1 : col_pos;
594
595 switch(col_align) {
596 default:
597 case LV_GRID_ALIGN_START:
598 x = c->x[col_pos] + lv_obj_get_style_margin_left_internal(item, LV_PART_MAIN);
599 item->w_layout = 0;
600 break;
601 case LV_GRID_ALIGN_STRETCH:
602 x = c->x[col_pos] + lv_obj_get_style_margin_left_internal(item, LV_PART_MAIN);
603 item_w = col_w - get_margin_hor(item);
604 item->w_layout = 1;
605 break;
606 case LV_GRID_ALIGN_CENTER:
607 x = c->x[col_pos] + (col_w - item_w) / 2 + (lv_obj_get_style_margin_left_internal(item, LV_PART_MAIN) -
608 lv_obj_get_style_margin_right_internal(item, LV_PART_MAIN)) / 2;
609 item->w_layout = 0;
610 break;
611 case LV_GRID_ALIGN_END:
612 x = c->x[col_pos] + col_w - lv_obj_get_width(item) - lv_obj_get_style_margin_right_internal(item, LV_PART_MAIN);
613 item->w_layout = 0;
614 break;
615 }
616
617 switch(row_align) {
618 default:
619 case LV_GRID_ALIGN_START:
620 y = c->y[row_pos] + lv_obj_get_style_margin_top_internal(item, LV_PART_MAIN);
621 item->h_layout = 0;
622 break;
623 case LV_GRID_ALIGN_STRETCH:
624 y = c->y[row_pos] + lv_obj_get_style_margin_top_internal(item, LV_PART_MAIN);
625 item_h = row_h - get_margin_ver(item);
626 item->h_layout = 1;
627 break;
628 case LV_GRID_ALIGN_CENTER:
629 y = c->y[row_pos] + (row_h - item_h) / 2 + (lv_obj_get_style_margin_top_internal(item, LV_PART_MAIN) -
630 lv_obj_get_style_margin_bottom_internal(item, LV_PART_MAIN)) / 2;
631 item->h_layout = 0;
632 break;
633 case LV_GRID_ALIGN_END:
634 y = c->y[row_pos] + row_h - lv_obj_get_height(item) - lv_obj_get_style_margin_bottom_internal(item, LV_PART_MAIN);
635 item->h_layout = 0;
636 break;
637 }
638
639 /*Set a new size if required*/
640 if(lv_obj_get_width(item) != item_w || lv_obj_get_height(item) != item_h) {
641 lv_area_t old_coords;
642 old_coords = item->coords;
643 lv_obj_invalidate(item);
644 lv_area_set_width(&item->coords, item_w);
645 lv_area_set_height(&item->coords, item_h);
646 lv_obj_invalidate(item);
647 lv_obj_send_event(item, LV_EVENT_SIZE_CHANGED, &old_coords);
648 lv_obj_send_event(lv_obj_get_parent(item), LV_EVENT_CHILD_CHANGED, item);
649
650 }
651
652 /*Handle percentage value of translate*/
653 int32_t tr_x = lv_obj_get_style_translate_x_internal(item, LV_PART_MAIN);
654 int32_t tr_y = lv_obj_get_style_translate_y_internal(item, LV_PART_MAIN);
655 int32_t w = lv_obj_get_width(item);
656 int32_t h = lv_obj_get_height(item);
657 if(LV_COORD_IS_PCT(tr_x)) tr_x = (w * LV_COORD_GET_PCT(tr_x)) / 100;
658 if(LV_COORD_IS_PCT(tr_y)) tr_y = (h * LV_COORD_GET_PCT(tr_y)) / 100;
659
660 x += tr_x;
661 y += tr_y;
662
663 int32_t diff_x = hint->grid_abs.x + x - item->coords.x1;
664 int32_t diff_y = hint->grid_abs.y + y - item->coords.y1;
665 if(diff_x || diff_y) {
666 lv_obj_invalidate(item);
667 item->coords.x1 += diff_x;
668 item->coords.x2 += diff_x;
669 item->coords.y1 += diff_y;
670 item->coords.y2 += diff_y;
671 lv_obj_invalidate(item);
672 lv_obj_move_children_by(item, diff_x, diff_y, false);
673 }
674}

688static int32_t grid_align(int32_t cont_size, bool auto_size, lv_grid_align_t align, int32_t gap,
689 uint32_t track_num,
690 int32_t * size_array, int32_t * pos_array, bool reverse)
691{
692 LV_ASSERT(size_array != NULL);
693 LV_ASSERT(pos_array != NULL);
694 int32_t grid_size = 0;
695 uint32_t i;
696
697 if(auto_size) {
698 pos_array[0] = 0;
699 }
700 else {
701 /*With spaced alignment gap will be calculated from the remaining space*/
702 if(align == LV_GRID_ALIGN_SPACE_AROUND || align == LV_GRID_ALIGN_SPACE_BETWEEN || align == LV_GRID_ALIGN_SPACE_EVENLY) {
703 gap = 0;
704 if(track_num == 1) align = LV_GRID_ALIGN_CENTER;
705 }
706
707 /*Get the full grid size with gap*/
708 for(i = 0; i < track_num; i++) {
709 grid_size += size_array[i] + gap;
710 }
711 grid_size -= gap;
712
713 /*Calculate the position of the first item and set gap is necessary*/
714 switch(align) {
715 case LV_GRID_ALIGN_STRETCH: /*Not supported for tracks, fall back to START*/
716 case LV_GRID_ALIGN_START:
717 pos_array[0] = 0;
718 break;
719 case LV_GRID_ALIGN_CENTER:
720 pos_array[0] = (cont_size - grid_size) / 2;
721 break;
722 case LV_GRID_ALIGN_END:
723 pos_array[0] = cont_size - grid_size;
724 break;
725 case LV_GRID_ALIGN_SPACE_BETWEEN:
726 pos_array[0] = 0;
727 gap = (int32_t)(cont_size - grid_size) / (int32_t)(track_num - 1);
728 break;
729 case LV_GRID_ALIGN_SPACE_AROUND:
730 gap = (int32_t)(cont_size - grid_size) / (int32_t)(track_num);
731 pos_array[0] = gap / 2;
732 break;
733 case LV_GRID_ALIGN_SPACE_EVENLY:
734 gap = (int32_t)(cont_size - grid_size) / (int32_t)(track_num + 1);
735 pos_array[0] = gap;
736 break;
737 }
738 }
739
740 /*Set the position of all tracks from the start position, gaps and track sizes*/
741 for(i = 0; i < track_num - 1; i++) {
742 pos_array[i + 1] = pos_array[i] + size_array[i] + gap;
743 }
744
745 int32_t total_gird_size = pos_array[track_num - 1] + size_array[track_num - 1] - pos_array[0];
746
747 if(reverse) {
748 for(i = 0; i < track_num; i++) {
749 pos_array[i] = cont_size - pos_array[i] - size_array[i];
750 }
751
752 }
753
754 /*Return the full size of the grid*/
755 return total_gird_size;
756}
