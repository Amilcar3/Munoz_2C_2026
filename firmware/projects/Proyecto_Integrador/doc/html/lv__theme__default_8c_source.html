626lv_theme_t * lv_theme_default_init(lv_display_t * disp, lv_color_t color_primary, lv_color_t color_secondary, bool dark,
627 const lv_font_t * font)
628{
629 LV_CHECK_ARG(font != NULL, return NULL);
630
631 if(!disp) {
632 LOG_NULL_DISPLAY_DEPRECATED_MESSAGE();
633 disp = lv_display_get_default();
634 }
635 LV_CHECK_ARG(disp != NULL, return NULL);
636
637 /*This trick is required only to avoid the garbage collection of
638 *styles' data if LVGL is used in a binding (e.g. MicroPython)
639 *In a general case styles could be in a simple `static lv_style_t my_style...` variables*/
640
641 if(!lv_theme_default_is_inited()) {
642 theme_def = lv_malloc_zeroed(sizeof(my_theme_t));
643 LV_ASSERT_MALLOC(theme_def);
644 }
645
646 my_theme_t * theme = theme_def;
647
648 lv_display_t * new_disp = disp;
649 int32_t new_dpi = lv_display_get_dpi(new_disp);
650 int32_t hor_res = lv_display_get_horizontal_resolution(new_disp);
651 int32_t ver_res = lv_display_get_vertical_resolution(new_disp);
652 int32_t greater_res = LV_MAX(hor_res, ver_res);
653 disp_size_t new_size;
654
655 if(greater_res <= 320) new_size = DISP_SMALL;
656 else if(greater_res < 720) new_size = DISP_MEDIUM;
657 else new_size = DISP_LARGE;
658
659 /* check theme information whether will change or not*/
660 if(theme->inited && theme->disp_dpi == new_dpi &&
661 theme->disp_size == new_size &&
662 lv_color_eq(theme->base.color_primary, color_primary) &&
663 lv_color_eq(theme->base.color_secondary, color_secondary) &&
664 (theme->base.flags == (dark ? MODE_DARK : 0)) &&
665 theme->base.font_small == font) {
666 return (lv_theme_t *) theme;
667 }
668
669 theme->disp_size = new_size;
670 theme->disp_dpi = new_dpi;
671 theme->base.disp = new_disp;
672 theme->base.color_primary = color_primary;
673 theme->base.color_secondary = color_secondary;
674 theme->base.font_small = font;
675 theme->base.font_normal = font;
676 theme->base.font_large = font;
677 theme->base.apply_cb = theme_apply;
678 theme->base.flags = dark ? MODE_DARK : 0;
679#if LV_USE_EXT_DATA
680 theme->base.ext_data.free_cb = NULL;
681 theme->base.ext_data.data = NULL;
682#endif
683
684 /*Remove the callback before triggering style refresh to prevent
685 *resolution_change_event_cb from re-entering lv_theme_default_init
686 *during lv_obj_report_style_change. Re-added below.*/
687 lv_display_remove_event_cb_with_user_data(new_disp, resolution_change_event_cb, theme);
688
689 style_init(theme);
690
691 theme->inited = true;
692
693 if(lv_display_get_theme(disp) == (lv_theme_t *)theme) {
694 lv_obj_report_style_change(NULL);
695 }
696
697 lv_display_add_event_cb(new_disp, resolution_change_event_cb, LV_EVENT_RESOLUTION_CHANGED, theme);
698
699 return (lv_theme_t *) theme;
700}

704 my_theme_t * theme = theme_def;
705 if(theme == NULL) return false;
706 return theme->inited;

709lv_theme_t * lv_theme_default_get(void)

712 return NULL;

718void lv_theme_default_deinit(void)
719{
720 my_theme_t * theme = theme_def;
721 if(theme) {
722 if(theme->inited) {
723 lv_style_t * theme_styles = (lv_style_t *)(&(theme->styles));
724 uint32_t i;
725 for(i = 0; i < sizeof(my_theme_styles_t) / sizeof(lv_style_t); i++) {
726 lv_style_reset(theme_styles + i);
727 }
728 }
729#if LV_USE_EXT_DATA
730 if(theme->base.ext_data.free_cb) {
731 theme->base.ext_data.free_cb(theme->base.ext_data.data);
732 theme->base.ext_data.data = NULL;
733 }
734#endif
735 lv_free(theme_def);
736 theme_def = NULL;
737 }
