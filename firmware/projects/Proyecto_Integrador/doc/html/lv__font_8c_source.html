124bool lv_font_get_glyph_dsc_internal(const lv_font_t * font, lv_font_glyph_dsc_t * dsc_out, uint32_t letter,
125 uint32_t letter_next)
126{
127
128 LV_ASSERT(dsc_out != NULL);
129 LV_ASSERT(font != NULL);
130
131#if LV_USE_FONT_PLACEHOLDER
132 const lv_font_t * placeholder_font = NULL;
133#endif
134 const lv_font_t * f = font;
135 const bool has_kerning = f->kerning != LV_FONT_KERNING_NONE;
136
137 lv_memzero(dsc_out, sizeof(lv_font_glyph_dsc_t));
138
139 while(f) {
140 bool found = f->get_glyph_dsc(f, dsc_out, letter,
141 has_kerning ? letter_next : 0);
142 if(found) {
143 if(!dsc_out->is_placeholder) {
144 dsc_out->resolved_font = f;
145 return true;
146 }
147#if LV_USE_FONT_PLACEHOLDER
148 else if(placeholder_font == NULL) {
149 placeholder_font = f;
150 }
151#endif
152 }
153 f = f->fallback;
154 }
155
156#if LV_USE_FONT_PLACEHOLDER
157 if(placeholder_font != NULL) {
158 placeholder_font->get_glyph_dsc(placeholder_font, dsc_out, letter,
159 has_kerning ? letter_next : 0);
160 dsc_out->resolved_font = placeholder_font;
161 return true;
162 }
163#endif
164
165#if LV_USE_FONT_PLACEHOLDER
166 dsc_out->box_w = font->line_height / 2;
167 dsc_out->adv_w = dsc_out->box_w + 2;
168#else
169 dsc_out->box_w = 0;
170 dsc_out->adv_w = 0;
171#endif
172
173 dsc_out->stride = 0;
174 dsc_out->resolved_font = NULL;
175 dsc_out->box_h = font->line_height;
176 dsc_out->ofs_x = 0;
177 dsc_out->ofs_y = 0;
178 dsc_out->format = LV_FONT_GLYPH_FORMAT_A1;
179 dsc_out->is_placeholder = true;
180
181 return false;
182}

243int32_t lv_font_glyph_dsc_compare(const lv_font_glyph_dsc_t * lhs, const lv_font_glyph_dsc_t * rhs)
244{
245 LV_ASSERT(lhs != NULL);
246 LV_ASSERT(rhs != NULL);
247 if(lhs->resolved_font != rhs->resolved_font) {
248 return (lv_uintptr_t)lhs->resolved_font > (lv_uintptr_t)rhs->resolved_font ? 1 : -1;
249 }
250
251 int32_t ret = lhs->format - rhs->format;
252 if(ret != 0) {
253 return ret;
254 }
255
256 if(lhs->format == LV_FONT_GLYPH_FORMAT_IMAGE) {
257 if(lhs->gid.src == rhs->gid.src) {
258 return 0;
259 }
260 return (lv_uintptr_t)lhs->gid.src > (lv_uintptr_t)rhs->gid.src ? 1 : -1;
261 }
262
263 if(lhs->gid.index == rhs->gid.index) {
264 return 0;
265 }
266 return lhs->gid.index > rhs->gid.index ? 1 : -1;
267}
