84void test_refr_inv_area_clipped_to_screen(void)
85{
86 refr_disp_create(100, 100, LV_COLOR_FORMAT_XRGB8888, LV_DISPLAY_RENDER_MODE_DIRECT, 1, 100);
87 refr_ctx.disp->inv_p = 0;
88
89 /*The areas that are completely out of the screen are dropped*/
90 lv_area_t right = {100, 10, 120, 20};
91 lv_area_t below = {10, 100, 20, 120};
92 lv_area_t left = {-50, 10, -1, 20};
93 lv_area_t above = {10, -50, 20, -1};
94
95 TEST_ASSERT_EQUAL(LV_RESULT_INVALID, lv_inv_area(refr_ctx.disp, &right));
96 TEST_ASSERT_EQUAL(LV_RESULT_INVALID, lv_inv_area(refr_ctx.disp, &below));
97 TEST_ASSERT_EQUAL(LV_RESULT_INVALID, lv_inv_area(refr_ctx.disp, &left));
98 TEST_ASSERT_EQUAL(LV_RESULT_INVALID, lv_inv_area(refr_ctx.disp, &above));
99 TEST_ASSERT_EQUAL_UINT32(0, refr_ctx.disp->inv_p);
100
101 /*The area that hangs over the edges is stored clipped*/
102 lv_area_t area = { -20, -30, 120, 130};
103 lv_area_t expected = {0, 0, 99, 99};
104 TEST_ASSERT_EQUAL(LV_RESULT_OK, lv_inv_area(refr_ctx.disp, &area));
105 TEST_ASSERT_EQUAL_UINT32(1, refr_ctx.disp->inv_p);
106 TEST_ASSERT_TRUE(lv_area_is_equal(&expected, &refr_ctx.disp->inv_areas[0]));
107
108 /*The caller's area is not modified*/
109 lv_area_t untouched = { -20, -30, 120, 130};
110 TEST_ASSERT_TRUE(lv_area_is_equal(&untouched, &area));
111}

170void test_refr_inv_area_buffer_overflow_falls_back_to_full_screen(void)
171{
172 refr_disp_create(400, 400, LV_COLOR_FORMAT_XRGB8888, LV_DISPLAY_RENDER_MODE_DIRECT, 1, 400);
173 refr_ctx.disp->inv_p = 0;
174
175 /*Fill the invalid area buffer with distinct, non-overlapping areas*/
176 uint32_t i;
177 for(i = 0; i < LV_INV_BUF_SIZE; i++) {
178 lv_area_t a;
179 a.x1 = (int32_t)(i % 20) * 20;
180 a.y1 = (int32_t)(i / 20) * 20;
181 a.x2 = a.x1 + 9;
182 a.y2 = a.y1 + 9;
183 TEST_ASSERT_EQUAL(LV_RESULT_OK, lv_inv_area(refr_ctx.disp, &a));
184 }
185 TEST_ASSERT_EQUAL_UINT32(LV_INV_BUF_SIZE, refr_ctx.disp->inv_p);
186
187 /*One more area does not fit, so the whole screen is invalidated instead*/
188 lv_area_t extra = {300, 300, 309, 309};
189 lv_area_t full = {0, 0, 399, 399};
190 TEST_ASSERT_EQUAL(LV_RESULT_OK, lv_inv_area(refr_ctx.disp, &extra));
191 TEST_ASSERT_EQUAL_UINT32(1, refr_ctx.disp->inv_p);
192 TEST_ASSERT_TRUE(lv_area_is_equal(&full, &refr_ctx.disp->inv_areas[0]));
193}
