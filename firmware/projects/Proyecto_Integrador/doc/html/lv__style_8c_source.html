185{
186 LV_CHECK_ARG(style != NULL, return);
187#if LV_USE_ASSERT_STYLE

225void lv_style_merge(lv_style_t * dst, const lv_style_t * src)
226{
227 LV_CHECK_ARG(dst != NULL, return);
228 LV_CHECK_ARG(src != NULL, return);
229 if(lv_style_is_const_internal(dst)) {
230 LV_LOG_WARN("The destination can not be a constant style");
231 return;
232 }
233
234 /*Source is empty*/
235 if(src->values_and_props == NULL) {
236 LV_LOG_TRACE("Source style is empty");
237 return;
238 }
239 if(src->prop_cnt == 0) {
240 LV_LOG_TRACE("Source style has no properties");
241 return;
242 }
243
244 /* Merge the styles */
245 int32_t i;
246 if(lv_style_is_const_internal(src)) {
247 lv_style_const_prop_t * props_and_values = (lv_style_const_prop_t *)src->values_and_props;
248 for(i = 0; props_and_values[i].prop != LV_STYLE_PROP_INV; i++) {
249 lv_style_set_prop(dst, props_and_values[i].prop, props_and_values[i].value);
250 }
251 }
252 else {
253 lv_style_prop_t * props = (lv_style_prop_t *)src->values_and_props + src->prop_cnt * sizeof(lv_style_value_t);
254 lv_style_value_t * values = (lv_style_value_t *)src->values_and_props;
255 for(i = 0; i < src->prop_cnt; i++) {
256 lv_style_set_prop(dst, props[i], values[i]);
257 }
258 }
259}

261lv_style_prop_t lv_style_register_prop(uint8_t flag)
262{
263 if(lv_style_custom_prop_flag_lookup_table == NULL) {
264 lv_style_custom_prop_flag_lookup_table_size = 0;
265 last_custom_prop_id = (uint16_t)LV_STYLE_LAST_BUILT_IN_PROP;
266 }
267
268 /*
269 * Allocate the lookup table if it's not yet available.
270 */
271 size_t required_size = (last_custom_prop_id + 1 - LV_STYLE_LAST_BUILT_IN_PROP);
272 if(lv_style_custom_prop_flag_lookup_table_size < required_size) {
273 /* Round required_size up to the nearest 32-byte value */
274 required_size = (required_size + 31) & ~31;
275 LV_ASSERT_MSG(required_size > 0, "required size has become 0?");
276 uint8_t * old_p = lv_style_custom_prop_flag_lookup_table;
277 uint8_t * new_p = lv_realloc(old_p, required_size * sizeof(uint8_t));
278 if(new_p == NULL) {
279 LV_LOG_ERROR("Unable to allocate space for custom property lookup table");
280 return LV_STYLE_PROP_INV;
281 }
282 lv_style_custom_prop_flag_lookup_table = new_p;
283 lv_style_custom_prop_flag_lookup_table_size = required_size;
284 }
285 last_custom_prop_id++;
286 /* This should never happen - we should bail out above */
287 LV_ASSERT_NULL(lv_style_custom_prop_flag_lookup_table);
288 lv_style_custom_prop_flag_lookup_table[last_custom_prop_id - LV_STYLE_NUM_BUILT_IN_PROPS] = flag;
289 return last_custom_prop_id;
290}

292lv_style_prop_t lv_style_get_num_custom_props(void)

294 return last_custom_prop_id - LV_STYLE_LAST_BUILT_IN_PROP;

297bool lv_style_remove_prop(lv_style_t * style, lv_style_prop_t prop)
298{
299 LV_CHECK_ARG(style != NULL && LV_STYLE_SENTINEL_OK(style), return false);
300
301 if(lv_style_is_const_internal(style)) {
302 LV_LOG_ERROR("Cannot remove prop from const style");
303 return false;
304 }
305
306 if(style->prop_cnt == 0) return false;
307
308 LV_PROFILER_STYLE_BEGIN;
309
310 uint8_t * tmp = (lv_style_prop_t *)style->values_and_props + style->prop_cnt * sizeof(lv_style_value_t);
311 uint8_t * old_props = (uint8_t *)tmp;
312 uint32_t i;
313 for(i = 0; i < style->prop_cnt; i++) {
314 if(old_props[i] == prop) {
315 lv_style_value_t * old_values = (lv_style_value_t *)style->values_and_props;
316
317 size_t size = (style->prop_cnt - 1) * (sizeof(lv_style_value_t) + sizeof(lv_style_prop_t));
318 uint8_t * new_values_and_props = lv_malloc(size);
319 if(new_values_and_props == NULL) {
320 LV_PROFILER_STYLE_END;
321 return false;
322 }
323
324 style->values_and_props = new_values_and_props;
325 style->prop_cnt--;
326
327 tmp = new_values_and_props + style->prop_cnt * sizeof(lv_style_value_t);
328 uint8_t * new_props = (uint8_t *)tmp;
329 lv_style_value_t * new_values = (lv_style_value_t *)new_values_and_props;
330
331 uint32_t j;
332 for(i = j = 0; j <= style->prop_cnt;
333 j++) { /*<=: because prop_cnt already reduced but all the old props. needs to be checked.*/
334 if(old_props[j] != prop) {
335 new_values[i] = old_values[j];
336 new_props[i++] = old_props[j];
337 }
338 }
339
340 lv_free(old_values);
341 LV_PROFILER_STYLE_END;
342 return true;
343 }
344 }
345
346 LV_PROFILER_STYLE_END;
347 return false;
348}

350void lv_style_set_prop(lv_style_t * style, lv_style_prop_t prop, lv_style_value_t value)
351{
352 LV_CHECK_ARG(style != NULL && LV_STYLE_SENTINEL_OK(style), return);
353
354 if(lv_style_is_const_internal(style)) {
355 LV_LOG_ERROR("Cannot set property of constant style");
356 return;
357 }
358
359 LV_CHECK_ARG(prop != LV_STYLE_PROP_INV, return);
360 LV_PROFILER_STYLE_BEGIN;
361 lv_style_prop_t * props;
362 int32_t i;
363
364 if(style->values_and_props) {
365 props = (lv_style_prop_t *)style->values_and_props + style->prop_cnt * sizeof(lv_style_value_t);
366 for(i = style->prop_cnt - 1; i >= 0; i--) {
367 if(props[i] == prop) {
368 lv_style_value_t * values = (lv_style_value_t *)style->values_and_props;
369 values[i] = value;
370 LV_PROFILER_STYLE_END;
371 return;
372 }
373 }
374 }
375
376 size_t size = (style->prop_cnt + 1) * (sizeof(lv_style_value_t) + sizeof(lv_style_prop_t));
377 uint8_t * values_and_props = lv_realloc(style->values_and_props, size);
378 if(values_and_props == NULL) {
379 LV_PROFILER_STYLE_END;
380 return;
381 }
382
383 style->values_and_props = values_and_props;
384
385 props = values_and_props + style->prop_cnt * sizeof(lv_style_value_t);
386 /*Shift all props to make place for the value before them*/
387 for(i = style->prop_cnt - 1; i >= 0; i--) {
388 props[i + sizeof(lv_style_value_t) / sizeof(lv_style_prop_t)] = props[i];
389 }
390 style->prop_cnt++;
391
392 /*Go to the new position with the props*/
393 props = values_and_props + style->prop_cnt * sizeof(lv_style_value_t);
394 lv_style_value_t * values = (lv_style_value_t *)values_and_props;
395
396 /*Set the new property and value*/
397 props[style->prop_cnt - 1] = prop;
398 values[style->prop_cnt - 1] = value;
399
400 uint32_t group = lv_style_get_prop_group(prop);
401 style->has_group |= (uint32_t)1 << group;
402 LV_PROFILER_STYLE_END;
403}
