26void test_theme(void)
27{
28 lv_display_t * display = lv_display_get_default();
29 lv_theme_t * theme = lv_display_get_theme(display);
30 TEST_ASSERT_NOT_NULL(theme);
31
32 lv_theme_t new_theme;
33 lv_memzero(&new_theme, sizeof(lv_theme_t));
34 new_theme.font_small = &lv_font_montserrat_16;
35 new_theme.font_normal = &lv_font_montserrat_18;
36 new_theme.font_large = &lv_font_montserrat_20;
37 new_theme.color_primary = lv_color_hex(0xFF0000);
38 new_theme.color_secondary = lv_color_hex(0x00FF00);
39
40 lv_theme_set_parent(&new_theme, theme);
41 TEST_ASSERT_EQUAL_PTR(theme, new_theme.parent);
42
43 lv_theme_set_apply_cb(&new_theme, theme_apply);
44 TEST_ASSERT_EQUAL_PTR(theme_apply, new_theme.apply_cb);
45
46 lv_theme_set_apply_cb(&new_theme, NULL);
47 TEST_ASSERT_NULL(new_theme.apply_cb);
48 lv_display_set_theme(display, &new_theme);
49
50 lv_obj_t * obj = lv_obj_create(lv_screen_active());
51 TEST_ASSERT_EQUAL_PTR(&lv_font_montserrat_16, lv_theme_get_font_small(obj));
52 TEST_ASSERT_EQUAL_PTR(&lv_font_montserrat_18, lv_theme_get_font_normal(obj));
53 TEST_ASSERT_EQUAL_PTR(&lv_font_montserrat_20, lv_theme_get_font_large(obj));
54 TEST_ASSERT_EQUAL_COLOR(lv_color_hex(0xFF0000), lv_theme_get_color_primary(NULL));
55 TEST_ASSERT_EQUAL_COLOR(lv_color_hex(0x00FF00), lv_theme_get_color_secondary(NULL));
56
57 /* Verify default values */
58 lv_display_set_theme(display, NULL);
59 TEST_ASSERT_EQUAL_PTR(LV_FONT_DEFAULT, lv_theme_get_font_small(obj));
60 TEST_ASSERT_EQUAL_PTR(LV_FONT_DEFAULT, lv_theme_get_font_normal(obj));
61 TEST_ASSERT_EQUAL_PTR(LV_FONT_DEFAULT, lv_theme_get_font_large(obj));
62 TEST_ASSERT_EQUAL_COLOR(lv_palette_main(LV_PALETTE_BLUE_GREY), lv_theme_get_color_primary(NULL));
63 TEST_ASSERT_EQUAL_COLOR(lv_palette_main(LV_PALETTE_BLUE), lv_theme_get_color_secondary(NULL));
64}
