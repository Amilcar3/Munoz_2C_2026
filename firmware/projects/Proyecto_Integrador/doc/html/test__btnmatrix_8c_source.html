523void test_button_matrix_draw_task_culling(void)
524{
525 /*Buttons farther outside the clip area than this margin must not create
526 *draw tasks. The margin must be larger than the widget's internal clip
527 *margin (row/column gap with a dpi/10 floor) so buttons whose shadow or
528 *outline could reach the clip area are exempt from the check.*/
529 const int32_t margin = 40;
530
531 /*Create a small container that clips its content*/
532 lv_obj_t * cont = lv_obj_create(active_screen);
533 lv_obj_set_size(cont, 300, 150);
534 lv_obj_center(cont);
535 lv_obj_set_style_pad_all(cont, 0, 0);
536
537 /*Create a buttonmatrix much taller than the container so entire rows
538 *fall far outside the clip area on both sides*/
539 static const char * map[] = {"A", "B", "C", "\n",
540 "D", "E", "F", "\n",
541 "G", "H", "I", "\n",
542 "J", "K", "L", ""
543 };
544 lv_obj_t * btnm_clip = lv_buttonmatrix_create(cont);
545 lv_buttonmatrix_set_map(btnm_clip, map);
546 lv_obj_set_size(btnm_clip, 280, 600);
547 lv_obj_set_send_draw_task_events(btnm_clip, true);
548 lv_obj_add_event_cb(btnm_clip, partial_clip_draw_task_cb, LV_EVENT_DRAW_TASK_ADDED, NULL);
549
550 /*Scroll so the middle rows are visible, top and bottom rows clipped*/
551 lv_obj_update_layout(active_screen);
552 lv_obj_scroll_to_y(cont, 225, LV_ANIM_OFF);
553
554 drawn_buttons_mask = 0;
555 lv_obj_invalidate(active_screen);
556 lv_refr_now(NULL);
557
558 lv_area_t clip_area;
559 lv_obj_get_coords(cont, &clip_area);
560
561 lv_area_t obj_coords;
562 lv_obj_get_coords(btnm_clip, &obj_coords);
563
564 lv_buttonmatrix_t * btnm_priv = (lv_buttonmatrix_t *)btnm_clip;
565 bool culled_above = false;
566 bool culled_below = false;
567 uint32_t i;
568 for(i = 0; i < btnm_priv->btn_cnt; i++) {
569 lv_area_t btn_area = btnm_priv->button_areas[i];
570 lv_area_move(&btn_area, obj_coords.x1, obj_coords.y1);
571 bool drawn = (drawn_buttons_mask >> i) & 1;
572
573 if(lv_area_is_on(&btn_area, &clip_area)) {
574 TEST_ASSERT_TRUE_MESSAGE(drawn, "Button intersecting the clip area was not drawn");
575 }
576 else {
577 lv_area_t enlarged = btn_area;
578 lv_area_increase(&enlarged, margin, margin);
579 if(!lv_area_is_on(&enlarged, &clip_area)) {
580 TEST_ASSERT_FALSE_MESSAGE(drawn, "Button far outside the clip area created draw tasks");
581 if(btn_area.y2 < clip_area.y1) culled_above = true;
582 if(btn_area.y1 > clip_area.y2) culled_below = true;
583 }
584 }
585 }
586
587 /*Self-check: the geometry must exercise culling on both sides of the
588 *clip area, otherwise the assertions above are vacuous*/
589 TEST_ASSERT_TRUE_MESSAGE(culled_above, "No button far above the clip area, adjust the test geometry");
590 TEST_ASSERT_TRUE_MESSAGE(culled_below, "No button far below the clip area, adjust the test geometry");
591}
