109lv_result_t lv_bin_decoder_info(lv_image_decoder_t * decoder, lv_image_decoder_dsc_t * dsc, lv_image_header_t * header)
110{
111 LV_CHECK_ARG(dsc != NULL, return LV_RESULT_INVALID);
112 LV_CHECK_ARG(header != NULL, return LV_RESULT_INVALID);
113
114 /*check it even if unused as it might be required in the future*/
115 LV_CHECK_ARG(decoder != NULL, return LV_RESULT_INVALID);
116 LV_UNUSED(decoder);
117
118 const void * src = dsc->src;
119 lv_image_src_t src_type = dsc->src_type;
120
121 if(src_type == LV_IMAGE_SRC_VARIABLE) {
122 lv_image_dsc_t * image = (lv_image_dsc_t *)src;
123 lv_memcpy(header, &image->header, sizeof(lv_image_header_t));
124 }
125 else if(src_type == LV_IMAGE_SRC_FILE) {
126 /*Support only "*.bin" files*/
127 if(lv_strcmp(lv_fs_get_ext(src), "bin")) return LV_RESULT_INVALID;
128
129 lv_fs_res_t res;
130 uint32_t rn;
131 res = lv_fs_read(&dsc->file, header, sizeof(lv_image_header_t), &rn);
132
133 if(res != LV_FS_RES_OK || rn != sizeof(lv_image_header_t)) {
134 LV_LOG_WARN("Read file header failed: %d with len: %" LV_PRIu32 ", expected: %zu", res, rn, sizeof(lv_image_header_t));
135 return LV_RESULT_INVALID;
136 }
137
143 if(header->magic != LV_IMAGE_HEADER_MAGIC) {
144 LV_LOG_WARN("Legacy bin image detected: %s", (char *)src);
145 header->cf = header->magic;
146 header->magic = LV_IMAGE_HEADER_MAGIC;
147 }
148
149 /*File is always read to buf, thus data can be modified.*/
150 header->flags |= LV_IMAGE_FLAGS_MODIFIABLE;
151 }
152 else if(src_type == LV_IMAGE_SRC_SYMBOL) {
153 /*The size depend on the font but it is unknown here. It should be handled outside of the
154 *function*/
155 header->w = 1;
156 header->h = 1;
157 /*Symbols always have transparent parts. Important because of cover check in the draw
158 *function. The actual value doesn't matter because lv_draw_label will draw it*/
159 header->cf = LV_COLOR_FORMAT_A8;
160 }
161 else {
162 LV_LOG_WARN("Image get info found unknown src type");
163 return LV_RESULT_INVALID;
164 }
165
166 if(header->cf == LV_COLOR_FORMAT_UNKNOWN) {
167 LV_LOG_WARN("Image color format is unknown");
168 return LV_RESULT_INVALID;
169 }
170
171 /*For backward compatibility, all images are not premultiplied for now.*/
172 if(header->magic != LV_IMAGE_HEADER_MAGIC) {
173 header->flags &= ~LV_IMAGE_FLAGS_PREMULTIPLIED;
174 }
175
176 return LV_RESULT_OK;
177}

179lv_result_t lv_bin_decoder_open(lv_image_decoder_t * decoder, lv_image_decoder_dsc_t * dsc)
180{
181 LV_UNUSED(decoder);
182 LV_CHECK_ARG(dsc != NULL, return LV_RESULT_INVALID);
183
184 lv_result_t res = LV_RESULT_INVALID;
185 lv_fs_res_t fs_res = LV_FS_RES_UNKNOWN;
186 bool use_directly = false; /*If the image is already decoded and can be used directly*/
187
188 /*Open the file if it's a file*/
189 if(dsc->src_type == LV_IMAGE_SRC_FILE) {
190 /*Support only "*.bin" files*/
191 if(lv_strcmp(lv_fs_get_ext(dsc->src), "bin")) return LV_RESULT_INVALID;
192
193 /*If the file was open successfully save the file descriptor*/
194 decoder_data_t * decoder_data = get_decoder_data(dsc);
195 if(decoder_data == NULL) {
196 return LV_RESULT_INVALID;
197 }
198
199 dsc->user_data = decoder_data;
200 lv_fs_file_t * f = lv_malloc(sizeof(*f));
201 if(f == NULL) {
202 free_decoder_data(dsc);
203 return LV_RESULT_INVALID;
204 }
205
206 fs_res = lv_fs_open(f, dsc->src, LV_FS_MODE_RD);
207 if(fs_res != LV_FS_RES_OK) {
208 LV_LOG_WARN("Open file failed: %d", fs_res);
209 lv_free(f);
210 free_decoder_data(dsc);
211 return LV_RESULT_INVALID;
212 }
213
214 decoder_data->f = f; /*Now free_decoder_data will take care of the file*/
215
216 lv_color_format_t cf = dsc->header.cf;
217
218 if(dsc->header.flags & LV_IMAGE_FLAGS_COMPRESSED) {
219 res = decode_compressed(decoder, dsc);
220 }
221 else if(LV_COLOR_FORMAT_IS_INDEXED(cf)) {
222 if(dsc->args.use_indexed) {
223 /*Palette for indexed image and whole image of A8 image are always loaded to RAM for simplicity*/
224 res = load_indexed(decoder, dsc);
225 }
226 else {
227 res = decode_indexed(decoder, dsc);
228 }
229 }
230 else if(LV_COLOR_FORMAT_IS_ALPHA_ONLY(cf)) {
231 res = decode_alpha_only(decoder, dsc);
232 }
233#if LV_BIN_DECODER_RAM_LOAD
234 else if(cf == LV_COLOR_FORMAT_ARGB8888 \
235 || cf == LV_COLOR_FORMAT_XRGB8888 \
236 || cf == LV_COLOR_FORMAT_RGB888 \
237 || cf == LV_COLOR_FORMAT_RGB565 \
238 || cf == LV_COLOR_FORMAT_RGB565_SWAPPED \
239 || cf == LV_COLOR_FORMAT_RGB565A8 \
240 || cf == LV_COLOR_FORMAT_ARGB8565) {
241 res = decode_rgb(decoder, dsc);
242 }
243#else
244 else {
245 /* decode them in get_area_cb */
246 res = LV_RESULT_OK;
247 }
248#endif
249 }
250
251 else if(dsc->src_type == LV_IMAGE_SRC_VARIABLE) {
252 /*The variables should have valid data*/
253 lv_image_dsc_t * image = (lv_image_dsc_t *)dsc->src;
254 if(image->data == NULL) {
255 return LV_RESULT_INVALID;
256 }
257
258 lv_color_format_t cf = image->header.cf;
259 if(dsc->header.flags & LV_IMAGE_FLAGS_COMPRESSED) {
260 res = decode_compressed(decoder, dsc);
261 }
262 else if(LV_COLOR_FORMAT_IS_INDEXED(cf)) {
263 /*Need decoder data to store converted image*/
264 decoder_data_t * decoder_data = get_decoder_data(dsc);
265 if(decoder_data == NULL) {
266 return LV_RESULT_INVALID;
267 }
268
269 if(dsc->args.use_indexed) {
270 /*Palette for indexed image and whole image of A8 image are always loaded to RAM for simplicity*/
271 res = load_indexed(decoder, dsc);
272 use_directly = true; /*If draw unit supports indexed image, it can be used directly.*/
273 }
274 else {
275 res = decode_indexed(decoder, dsc);
276 }
277 }
278 else if(LV_COLOR_FORMAT_IS_ALPHA_ONLY(cf) && cf != LV_COLOR_FORMAT_A8) {
279 /*Alpha only image will need decoder data to store pointer to decoded image, to free it when decoder closes*/
280 decoder_data_t * decoder_data = get_decoder_data(dsc);
281 if(decoder_data == NULL) {
282 return LV_RESULT_INVALID;
283 }
284
285 res = decode_alpha_only(decoder, dsc);
286 }
287 else {
288 /*In case of uncompressed formats the image stored in the ROM/RAM.
289 *So simply give its pointer*/
290
291 decoder_data_t * decoder_data = get_decoder_data(dsc);
292 lv_draw_buf_t * decoded;
293 if(image->header.flags & LV_IMAGE_FLAGS_ALLOCATED) {
294 decoded = (lv_draw_buf_t *)image;
295 res = LV_RESULT_OK;
296 }
297 else {
298 decoded = &decoder_data->c_array;
299 if(image->header.stride == 0) {
300 /*If image doesn't have stride, treat it as lvgl v8 legacy image format*/
301 lv_image_dsc_t tmp = *image;
302 tmp.header.stride = (tmp.header.w * lv_color_format_get_bpp(cf) + 7) >> 3;
303 res = lv_draw_buf_from_image(decoded, &tmp);
304 }
305 else
306 res = lv_draw_buf_from_image(decoded, image);
307 }
308
309 if(res == LV_RESULT_OK) {
310 dsc->decoded = decoded;
311
312 if(decoded->header.stride == 0) {
313 /*Use the auto calculated value from decoder_info callback*/
314 decoded->header.stride = dsc->header.stride;
315 }
316
317 use_directly = true; /*A variable image that can be used directly.*/
318 }
319 }
320 }
321
322 if(res != LV_RESULT_OK) {
323 free_decoder_data(dsc);
324 return res;
325 }
326
327 if(dsc->decoded == NULL) return LV_RESULT_OK; /*Need to read via get_area_cb*/
328
329 lv_draw_buf_t * decoded = (lv_draw_buf_t *)dsc->decoded;
330 if(dsc->header.flags & LV_IMAGE_FLAGS_PREMULTIPLIED) {
331 lv_draw_buf_set_flag(decoded, LV_IMAGE_FLAGS_PREMULTIPLIED);
332 }
333
334 lv_draw_buf_t * adjusted = lv_image_decoder_post_process(dsc, decoded);
335 if(adjusted == NULL) {
336 free_decoder_data(dsc);
337 return LV_RESULT_INVALID;
338 }
339
340 /*The adjusted draw buffer is newly allocated.*/
341 if(adjusted != decoded) {
342 use_directly = false; /*Cannot use original image directly*/
343 free_decoder_data(dsc);
344 decoder_data_t * decoder_data = get_decoder_data(dsc);
345 decoder_data->decoded = adjusted; /*Now this new buffer need to be free'd on decoder close*/
346 }
347 dsc->decoded = adjusted;
348
349 /* Copy user flags to the decoded image */
350 if(dsc->header.flags & LV_IMAGE_FLAGS_USER_MASK) {
351 lv_draw_buf_set_flag((lv_draw_buf_t *)dsc->decoded, dsc->header.flags & LV_IMAGE_FLAGS_USER_MASK);
352 }
353
354 /*Do not put image to cache if it can be used directly.*/
355 if(use_directly || dsc->args.no_cache) {
356 if(dsc->args.flush_cache && use_directly) {
357 dsc->args.flush_cache = false;
358 }
359 return LV_RESULT_OK;
360 }
361
362 /*If the image cache is disabled, just return the decoded image*/
363 if(!lv_image_cache_is_enabled()) return LV_RESULT_OK;
364
365 /*Add it to cache*/
366 lv_image_cache_data_t search_key;
367 search_key.src_type = dsc->src_type;
368 search_key.src = dsc->src;
369 search_key.slot.size = dsc->decoded->data_size;
370
371 lv_cache_entry_t * cache_entry = lv_image_decoder_add_to_cache(decoder, &search_key, dsc->decoded, dsc->user_data);
372 if(cache_entry == NULL) {
373 free_decoder_data(dsc);
374 return LV_RESULT_INVALID;
375 }
376 dsc->cache_entry = cache_entry;
377 decoder_data_t * decoder_data = get_decoder_data(dsc);
378 decoder_data->decoded = NULL; /*Cache will take care of it*/
379
380 return LV_RESULT_OK;
381}

397lv_result_t lv_bin_decoder_get_area(lv_image_decoder_t * decoder, lv_image_decoder_dsc_t * dsc,
398 const lv_area_t * full_area, lv_area_t * decoded_area)
399{
400 LV_CHECK_ARG(dsc != NULL, return LV_RESULT_INVALID);
401 LV_CHECK_ARG(full_area != NULL, return LV_RESULT_INVALID);
402 LV_CHECK_ARG(decoded_area != NULL, return LV_RESULT_INVALID);
403
404 LV_UNUSED(decoder);
405 lv_color_format_t cf = dsc->header.cf;
406 LV_CHECK_ARG_FORMAT_MSG(
407 LV_COLOR_FORMAT_IS_INDEXED(cf)
408 || cf == LV_COLOR_FORMAT_ARGB8888
409 || cf == LV_COLOR_FORMAT_XRGB8888
410 || cf == LV_COLOR_FORMAT_RGB888
411 || cf == LV_COLOR_FORMAT_RGB565
412 || cf == LV_COLOR_FORMAT_RGB565_SWAPPED
413 || cf == LV_COLOR_FORMAT_ARGB8565
414 || cf == LV_COLOR_FORMAT_RGB565A8,
415 return LV_RESULT_INVALID, "Unsupported color format 0x%02x", cf);
416 LV_CHECK_ARG_MSG(full_area->x1 >= 0 && full_area->x2 < (int32_t)dsc->header.w && full_area->y1 >= 0 &&
417 full_area->y2 < (int32_t)dsc->header.h, return LV_RESULT_INVALID, "Area outside image bounds");
418 LV_CHECK_ARG_MSG(dsc->user_data, return LV_RESULT_INVALID, "decoder data unavailable");
419
420 lv_fs_res_t res = LV_FS_RES_UNKNOWN;
421 decoder_data_t * decoder_data = dsc->user_data;
422 lv_fs_file_t * f = decoder_data->f;
423 uint32_t bpp = lv_color_format_get_bpp(cf);
424 int32_t w_px = lv_area_get_width(full_area);
425 uint8_t * img_data = NULL;
426 lv_draw_buf_t * decoded = NULL;
427 uint32_t offset = dsc->src_type == LV_IMAGE_SRC_FILE ? sizeof(lv_image_header_t) : 0; /*Skip the image header*/
428
429 /*We only support read line by line for now*/
430 if(decoded_area->y1 == LV_COORD_MIN) {
431 /*Indexed image is converted to ARGB888*/
432 lv_color_format_t cf_decoded = LV_COLOR_FORMAT_IS_INDEXED(cf) ? LV_COLOR_FORMAT_ARGB8888 : cf;
433
434 decoded = lv_draw_buf_reshape(decoder_data->decoded_partial, cf_decoded, w_px, 1, LV_STRIDE_AUTO);
435 if(decoded == NULL) {
436 if(decoder_data->decoded_partial != NULL) {
437 lv_draw_buf_destroy(decoder_data->decoded_partial);
438 decoder_data->decoded_partial = NULL;
439 }
440 decoded = lv_draw_buf_create_ex(image_cache_draw_buf_handlers, w_px, 1, cf_decoded, LV_STRIDE_AUTO);
441 if(decoded == NULL) return LV_RESULT_INVALID;
442 decoder_data->decoded_partial = decoded; /*Free on decoder close*/
443 }
444 *decoded_area = *full_area;
445 decoded_area->y2 = decoded_area->y1;
446 }
447 else {
448 decoded_area->y1++;
449 decoded_area->y2++;
450 decoded = decoder_data->decoded_partial; /*Already allocated*/
451 }
452
453 img_data = decoded->data; /*Get the buffer to operate on*/
454
455 if(decoded_area->y1 > full_area->y2) {
456 return LV_RESULT_INVALID;
457 }
458
459 if(LV_COLOR_FORMAT_IS_INDEXED(cf)) {
460 int32_t x_fraction = decoded_area->x1 % (8 / bpp);
461 uint32_t len = (w_px * bpp + 7) / 8 + 1; /*10px for 1bpp may across 3bytes*/
462 uint8_t * buf = NULL;
463
464 offset += dsc->palette_size * 4; /*Skip palette*/
465 offset += decoded_area->y1 * dsc->header.stride;
466 offset += decoded_area->x1 * bpp / 8; /*Move to x1*/
467 if(dsc->src_type == LV_IMAGE_SRC_FILE) {
468 buf = lv_malloc(len);
469 LV_ASSERT_NULL(buf);
470 if(buf == NULL)
471 return LV_RESULT_INVALID;
472
473 res = fs_read_file_at(f, offset, buf, len, NULL);
474 if(res != LV_FS_RES_OK) {
475 lv_free(buf);
476 return LV_RESULT_INVALID;
477 }
478 }
479 else {
480 const lv_image_dsc_t * image = dsc->src;
481 buf = (void *)(image->data + offset);
482 }
483
484 decode_indexed_line(cf, dsc->palette, x_fraction, w_px, buf, (lv_color32_t *)img_data);
485
486 if(dsc->src_type == LV_IMAGE_SRC_FILE) lv_free((void *)buf);
487
488 dsc->decoded = decoded; /*Return decoded image*/
489 return LV_RESULT_OK;
490 }
491
492 if(cf == LV_COLOR_FORMAT_ARGB8888 || cf == LV_COLOR_FORMAT_XRGB8888 || cf == LV_COLOR_FORMAT_RGB888
493 || cf == LV_COLOR_FORMAT_RGB565 || cf == LV_COLOR_FORMAT_RGB565_SWAPPED || cf == LV_COLOR_FORMAT_ARGB8565) {
494 uint32_t len = (w_px * bpp) / 8;
495 offset += decoded_area->y1 * dsc->header.stride;
496 offset += decoded_area->x1 * bpp / 8; /*Move to x1*/
497 res = fs_read_file_at(f, offset, img_data, len, NULL);
498 if(res != LV_FS_RES_OK) {
499 return LV_RESULT_INVALID;
500 }
501
502 dsc->decoded = decoded; /*Return decoded image*/
503 return LV_RESULT_OK;
504 }
505
506 if(cf == LV_COLOR_FORMAT_RGB565A8) {
507 bpp = 16; /* RGB565 + A8 mask*/
508 uint32_t len = decoded->header.stride;
509 offset += decoded_area->y1 * dsc->header.stride; /*Move to y1*/
510 offset += decoded_area->x1 * bpp / 8; /*Move to x1*/
511 res = fs_read_file_at(f, offset, img_data, len, NULL);
512 if(res != LV_FS_RES_OK) {
513 return LV_RESULT_INVALID;
514 }
515
516 /*Now the A8 mask*/
517 offset = sizeof(lv_image_header_t);
518 offset += dsc->header.h * dsc->header.stride; /*Move to A8 map*/
519 offset += decoded_area->y1 * (dsc->header.stride / 2); /*Move to y1*/
520 offset += decoded_area->x1 * 1; /*Move to x1*/
521 res = fs_read_file_at(f, offset, img_data + len, w_px * 1, NULL);
522 if(res != LV_FS_RES_OK) {
523 return LV_RESULT_INVALID;
524 }
525
526 dsc->decoded = decoded; /*Return decoded image*/
527 return LV_RESULT_OK;
528 }
529
530 return LV_RESULT_INVALID;
531}

573static lv_result_t decode_indexed(lv_image_decoder_t * decoder, lv_image_decoder_dsc_t * dsc)
574{
575 LV_UNUSED(decoder); /*Unused*/
576 LV_ASSERT(dsc != NULL);
577 lv_fs_res_t res;
578 uint32_t rn;
579 decoder_data_t * decoder_data = dsc->user_data;
580 lv_fs_file_t * f = decoder_data->f;
581 lv_color_format_t cf = dsc->header.cf;
582 uint32_t palette_len = sizeof(lv_color32_t) * LV_COLOR_INDEXED_PALETTE_SIZE(cf);
583 const lv_color32_t * palette;
584 const uint8_t * indexed_data = NULL;
585 lv_draw_buf_t * draw_buf_indexed = NULL;
586 uint32_t stride = dsc->header.stride;
587
588 bool is_compressed = dsc->header.flags & LV_IMAGE_FLAGS_COMPRESSED;
589 if(is_compressed) {
590 uint8_t * data = decoder_data->decompressed->data;
591 palette = (lv_color32_t *)data;
592 indexed_data = data + palette_len;
593 }
594 else if(dsc->src_type == LV_IMAGE_SRC_FILE) {
595 /*read palette for indexed image*/
596 palette = lv_malloc(palette_len);
597 LV_ASSERT_MALLOC(palette);
598 if(palette == NULL) {
599 LV_LOG_ERROR("Out of memory");
600 return LV_RESULT_INVALID;
601 }
602

610 decoder_data->palette = (void *)palette; /*Need to free when decoder closes*/

612#if LV_BIN_DECODER_RAM_LOAD
613 draw_buf_indexed = lv_draw_buf_create_ex(image_cache_draw_buf_handlers, dsc->header.w, dsc->header.h, cf,
614 dsc->header.stride);

633 LV_LOG_WARN("Read indexed image failed: %d, with len: %" LV_PRIu32 ", expected: %" LV_PRIu32, res, rn, data_len);
634 goto exit_with_buf;
635 }
636#endif
637 }
638 else if(dsc->src_type == LV_IMAGE_SRC_VARIABLE) {
639 lv_image_dsc_t * image = (lv_image_dsc_t *)dsc->src;
640 palette = (lv_color32_t *)image->data;
641 indexed_data = image->data + palette_len;
642 }
643 else {
644 return LV_RESULT_INVALID;
645 }
646
647 dsc->palette = palette;
648 dsc->palette_size = LV_COLOR_INDEXED_PALETTE_SIZE(cf);
649
650#if LV_BIN_DECODER_RAM_LOAD
651 /*Convert to ARGB8888, since sw renderer cannot render it directly even it's in RAM*/
652 lv_draw_buf_t * decoded = lv_draw_buf_create_ex(image_cache_draw_buf_handlers, dsc->header.w, dsc->header.h,
653 LV_COLOR_FORMAT_ARGB8888,
654 0);
655 if(decoded == NULL) {
656 LV_LOG_ERROR("No memory for indexed image");
657 goto exit_with_buf;
658 }
659
660 stride = decoded->header.stride;
661 uint8_t * img_data = decoded->data;
662
663 const uint8_t * in = indexed_data;
664 uint8_t * out = img_data;
665 for(uint32_t y = 0; y < dsc->header.h; y++) {
666 decode_indexed_line(cf, dsc->palette, 0, dsc->header.w, in, (lv_color32_t *)out);
667 in += dsc->header.stride;
668 out += stride;
669 }
670
671 dsc->decoded = decoded;
672 decoder_data->decoded = decoded; /*Free when decoder closes*/
673 if(dsc->src_type == LV_IMAGE_SRC_FILE && !is_compressed) {
674 decoder_data->palette = (void *)palette; /*Free decoder data on close*/
675 lv_draw_buf_destroy(draw_buf_indexed);
676 }
677
678 return LV_RESULT_OK;
679exit_with_buf:
680 if(dsc->src_type == LV_IMAGE_SRC_FILE && !is_compressed) {
681 lv_free((void *)palette);
682 decoder_data->palette = NULL;
683 }
684
685 if(draw_buf_indexed) lv_draw_buf_destroy(draw_buf_indexed);
686 return LV_RESULT_INVALID;
687#else
688 LV_UNUSED(stride);
689 LV_UNUSED(indexed_data);
690 LV_UNUSED(draw_buf_indexed);
691 /*It needs to be read by get_area_cb later*/
692 return LV_RESULT_OK;
693#endif

834static inline uint8_t bit_extend(uint8_t value, uint8_t bpp)
835{
836 if(value == 0) return 0;
837
838 uint8_t res = value;
839 uint8_t bpp_now = bpp;
840 while(bpp_now < 8) {
841 res |= value << (8 - bpp_now);
842 bpp_now += bpp;
843 };
844
845 return res;
846}

848static lv_result_t decode_alpha_only(lv_image_decoder_t * decoder, lv_image_decoder_dsc_t * dsc)
849{
850 LV_UNUSED(decoder);
851 LV_ASSERT(dsc != NULL);
852 lv_fs_res_t res;
853 uint32_t rn;
854 decoder_data_t * decoder_data = dsc->user_data;
855 uint8_t bpp = lv_color_format_get_bpp(dsc->header.cf);
856
857 if(bpp == 0) {
858 LV_LOG_ERROR("Error color format: %d", dsc->header.cf);
859 return LV_RESULT_INVALID;
860 }
861
862 uint32_t w = (dsc->header.stride * 8) / bpp;
863 uint32_t buf_stride = (w * 8 + 7) >> 3; /*stride for img_data*/
864 uint32_t buf_len = w * dsc->header.h; /*always decode to A8 format*/
865 lv_draw_buf_t * decoded;
866 uint32_t file_len = (uint32_t)dsc->header.stride * dsc->header.h;
867
868 decoded = lv_draw_buf_create_ex(image_cache_draw_buf_handlers, dsc->header.w, dsc->header.h, LV_COLOR_FORMAT_A8,
869 buf_stride);
870 if(decoded == NULL) {
871 LV_LOG_ERROR("Out of memory");
872 return LV_RESULT_INVALID;
873 }
874
875 uint8_t * img_data = decoded->data;
876
877 if(dsc->header.flags & LV_IMAGE_FLAGS_COMPRESSED) {
878 /*Copy from image data*/
879 lv_memcpy(img_data, decoder_data->decompressed->data, file_len);
880 }
881 else if(dsc->src_type == LV_IMAGE_SRC_FILE) {
882 res = fs_read_file_at(decoder_data->f, sizeof(lv_image_header_t), img_data, file_len, &rn);
883 if(res != LV_FS_RES_OK || rn != file_len) {
884 LV_LOG_WARN("Read header failed: %d, with len: %" LV_PRIu32 ", expected: %" LV_PRIu32, res, rn, file_len);
885 lv_draw_buf_destroy(decoded);
886 return LV_RESULT_INVALID;
887 }
888 }
889 else if(dsc->src_type == LV_IMAGE_SRC_VARIABLE) {
890 /*Copy from image data*/
891 lv_memcpy(img_data, ((lv_image_dsc_t *)dsc->src)->data, file_len);
892 }
893
894 if(dsc->header.cf != LV_COLOR_FORMAT_A8) {
895 /*Convert A1/2/4 to A8 from last pixel to first pixel*/
896 uint8_t * in = img_data + file_len - 1;
897 uint8_t * out = img_data + buf_len - 1;
898 uint8_t mask = (1 << bpp) - 1;
899 uint8_t shift = 0;
900 for(uint32_t i = 0; i < buf_len; i++) {
910 uint8_t value = ((*in >> shift) & mask);
911 *out = bit_extend(value, bpp);
912 shift += bpp;
913 if(shift >= 8) {
914 shift = 0;
915 in--;
916 }
917 out--;
918 }
919 }
920
921 decoder_data->decoded = decoded;
922 dsc->decoded = decoded;
923 return LV_RESULT_OK;
924}

926static lv_result_t decode_compressed(lv_image_decoder_t * decoder, lv_image_decoder_dsc_t * dsc)
927{
928#if LV_BIN_DECODER_RAM_LOAD
929 uint32_t rn;
930 uint32_t len;
931 uint32_t compressed_len;
932 LV_ASSERT(dsc != NULL);
933 decoder_data_t * decoder_data = get_decoder_data(dsc);
934 lv_result_t res;
935 lv_fs_res_t fs_res;
936 uint8_t * file_buf = NULL;
937 lv_image_compressed_t * compressed = &decoder_data->compressed;
938
939 lv_memzero(compressed, sizeof(lv_image_compressed_t));
940
941 if(dsc->src_type == LV_IMAGE_SRC_FILE) {
942 lv_fs_file_t * f = decoder_data->f;
943
944 if(lv_fs_seek(f, 0, LV_FS_SEEK_END) != LV_FS_RES_OK ||
945 lv_fs_tell(f, &compressed_len) != LV_FS_RES_OK) {
946 LV_LOG_WARN("Failed to get compressed file len");
947 return LV_RESULT_INVALID;
948 }
949
950 compressed_len -= sizeof(lv_image_header_t);
951 compressed_len -= 12;
952
953 /*Read compress header*/
954 len = 12;
955 fs_res = fs_read_file_at(f, sizeof(lv_image_header_t), compressed, len, &rn);
956 if(fs_res != LV_FS_RES_OK || rn != len) {
957 LV_LOG_WARN("Read compressed header failed: %d, with len: %" LV_PRIu32 ", expected: %" LV_PRIu32, fs_res, rn, len);
958 return LV_RESULT_INVALID;
959 }
960
961 if(compressed->compressed_size != compressed_len) {
962 LV_LOG_WARN("Compressed size mismatch: %" LV_PRIu32" != %" LV_PRIu32, compressed->compressed_size, compressed_len);
963 return LV_RESULT_INVALID;
964 }
965
966 file_buf = lv_malloc(compressed_len);
967 if(file_buf == NULL) {
968 LV_LOG_WARN("No memory for compressed file");
969 return LV_RESULT_INVALID;
970
971 }
972
973 /*Continue to read the compressed data following compression header*/
974 fs_res = lv_fs_read(f, file_buf, compressed_len, &rn);
975 if(fs_res != LV_FS_RES_OK || rn != compressed_len) {
976 LV_LOG_WARN("Read compressed file failed: %d, with len: %" LV_PRIu32 ", expected: %" LV_PRIu32, fs_res, rn,
977 compressed_len);
978 lv_free(file_buf);
979 return LV_RESULT_INVALID;
980 }
981
982 /*Decompress the image*/
983 compressed->data = file_buf;
984 }
985 else if(dsc->src_type == LV_IMAGE_SRC_VARIABLE) {
986 lv_image_dsc_t * image = (lv_image_dsc_t *)dsc->src;
987 compressed_len = image->data_size;
988
989 /*Read compress header*/
990 len = 12;
991 compressed_len -= len;
992 lv_memcpy(compressed, image->data, len);
993 compressed->data = image->data + len;
994 if(compressed->compressed_size != compressed_len) {
995 LV_LOG_WARN("Compressed size mismatch: %" LV_PRIu32" != %" LV_PRIu32, compressed->compressed_size, compressed_len);
996 return LV_RESULT_INVALID;
997 }
998 }
999 else {
1000 LV_LOG_WARN("Compressed image only support file or variable");
1001 return LV_RESULT_INVALID;
1002 }
1003
1004 res = decompress_image(dsc, compressed);
1005 compressed->data = NULL; /*No need to store the data any more*/
1006 lv_free(file_buf);
1007 if(res != LV_RESULT_OK) {
1008 LV_LOG_WARN("Decompress failed");
1009 return LV_RESULT_INVALID;
1010 }
1011
1012 /*Depends on the cf, need to further decode image like an C-array image*/
1013 lv_image_dsc_t * image = (lv_image_dsc_t *)dsc->src;
1014 if(dsc->src_type == LV_IMAGE_SRC_VARIABLE && image->data == NULL) {
1015 return LV_RESULT_INVALID;
1016 }
1017
1018 lv_color_format_t cf = dsc->header.cf;
1019 if(LV_COLOR_FORMAT_IS_INDEXED(cf)) {
1020 if(dsc->args.use_indexed) res = load_indexed(decoder, dsc);
1021 else res = decode_indexed(decoder, dsc);
1022 }
1023 else if(LV_COLOR_FORMAT_IS_ALPHA_ONLY(cf)) {
1024 res = decode_alpha_only(decoder, dsc);
1025 }
1026 else {
1027 /*The decompressed data is the original image data.*/
1028 dsc->decoded = decoder_data->decompressed;
1029
1030 /*Transfer ownership of decompressed to `decoded` since it can be used directly*/
1031 decoder_data->decoded = decoder_data->decompressed;
1032 decoder_data->decompressed = NULL;
1033 res = LV_RESULT_OK;
1034 }
1035
1036 return res;
1037#else
1038 LV_UNUSED(decompress_image);
1039 LV_UNUSED(decoder);
1040 LV_UNUSED(dsc);
1041 LV_LOG_ERROR("Need LV_BIN_DECODER_RAM_LOAD to be enabled");
1042 return LV_RESULT_INVALID;
1043#endif
1044}

1046static lv_result_t decode_indexed_line_i1(const lv_color32_t * palette, int32_t x,
1047 int32_t w_px, const uint8_t * in, lv_color32_t * out)
1048{
1049 /*8 pixels per byte, MSB first (bit 7 is the left-most pixel)*/
1050 in += x >> 3;
1051 int32_t shift = 7 - (x & 0x7);
1052
1053 int32_t i = 0;
1054
1055 /*Head: consume the rest of the first byte so the main loop starts byte-aligned.
1056 *Skipped entirely when x is already byte-aligned (shift == 7).*/
1057 if(shift != 7) {
1058 uint8_t byte = *in++;
1059 while(shift >= 0 && i < w_px) {
1060 out[i++] = palette[(byte >> shift) & 0x1];
1061 shift--;
1062 }
1063 }
1064
1065 /*Body: one full byte -> 8 pixels per iteration, fully unrolled.
1066 *No per-pixel branch or shift bookkeeping, so the compiler can pipeline it.*/
1067 for(; i + 8 <= w_px; i += 8) {
1068 uint8_t byte = *in++;
1069 out[i + 0] = palette[(byte >> 7) & 0x1];
1070 out[i + 1] = palette[(byte >> 6) & 0x1];
1071 out[i + 2] = palette[(byte >> 5) & 0x1];
1072 out[i + 3] = palette[(byte >> 4) & 0x1];
1073 out[i + 4] = palette[(byte >> 3) & 0x1];
1074 out[i + 5] = palette[(byte >> 2) & 0x1];
1075 out[i + 6] = palette[(byte >> 1) & 0x1];
1076 out[i + 7] = palette[(byte >> 0) & 0x1];
1077 }
1078
1079 /*Tail: the remaining 1..7 pixels of the last partial byte.*/
1080 if(i < w_px) {
1081 uint8_t byte = *in;
1082 shift = 7;
1083 for(; i < w_px; i++) {
1084 out[i] = palette[(byte >> shift) & 0x1];
1085 shift--;
1086 }
1087 }
1088
1089 return LV_RESULT_OK;
1090}

1093static lv_result_t decode_indexed_line_i2(const lv_color32_t * palette, int32_t x,
1094 int32_t w_px, const uint8_t * in, lv_color32_t * out)
1095{
1096 /*4 pixels per byte, MSB first: bits 6-7, 4-5, 2-3, 0-1.*/
1097 in += x >> 2;
1098 int32_t shift = 6 - 2 * (x & 0x3);
1099
1100 int32_t i = 0;
1101
1102 /*Head: consume the rest of the first byte so the main loop starts byte-aligned.
1103 *Skipped entirely when x is already byte-aligned (shift == 6).*/
1104 if(shift != 6) {
1105 uint8_t byte = *in++;
1106 while(shift >= 0 && i < w_px) {
1107 out[i++] = palette[(byte >> shift) & 0x3];
1108 shift -= 2;
1109 }
1110 }
1111
1112 /*Body: one full byte -> 4 pixels per iteration, fully unrolled.*/
1113 for(; i + 4 <= w_px; i += 4) {
1114 uint8_t byte = *in++;
1115 out[i + 0] = palette[(byte >> 6) & 0x3];
1116 out[i + 1] = palette[(byte >> 4) & 0x3];
1117 out[i + 2] = palette[(byte >> 2) & 0x3];
1118 out[i + 3] = palette[(byte >> 0) & 0x3];
1119 }
1120
1121 /*Tail: the remaining 1..3 pixels of the last partial byte.*/
1122 if(i < w_px) {
1123 uint8_t byte = *in;
1124 shift = 6;
1125 for(; i < w_px; i++) {
1126 out[i] = palette[(byte >> shift) & 0x3];
1127 shift -= 2;
1128 }
1129 }
1130
1131 return LV_RESULT_OK;
1132}

1134static lv_result_t decode_indexed_line_i4(const lv_color32_t * palette, int32_t x,
1135 int32_t w_px, const uint8_t * in, lv_color32_t * out)
1136{
1137 /*2 pixels per byte: high nibble (bits 4-7) is the left pixel, low nibble (bits 0-3) the right.*/
1138 in += x >> 1;
1139
1140 int32_t i = 0;
1141
1142 /*Head: if x is odd we start mid-byte at the low nibble; consume it to byte-align.*/
1143 if((x & 0x1) && i < w_px) {
1144 out[i++] = palette[*in++ & 0x0F];
1145 }
1146
1147 /*Body: one full byte -> 2 pixels per iteration, unrolled.*/
1148 for(; i + 2 <= w_px; i += 2) {
1149 uint8_t byte = *in++;
1150 out[i + 0] = palette[byte >> 4];
1151 out[i + 1] = palette[byte & 0x0F];
1152 }
1153
1154 /*Tail: one pixel left -> high nibble of the last byte.*/
1155 if(i < w_px) {
1156 out[i] = palette[*in >> 4];
1157 }
1158
1159 return LV_RESULT_OK;
1160}

1177static lv_result_t decode_indexed_line(lv_color_format_t color_format, const lv_color32_t * palette, int32_t x,
1178 int32_t w_px, const uint8_t * in, lv_color32_t * out)
1179{
1180 switch(color_format) {
1181 case LV_COLOR_FORMAT_I1:
1182 return decode_indexed_line_i1(palette, x, w_px, in, out);
1183 case LV_COLOR_FORMAT_I2:
1184 return decode_indexed_line_i2(palette, x, w_px, in, out);
1185 case LV_COLOR_FORMAT_I4:
1186 return decode_indexed_line_i4(palette, x, w_px, in, out);
1187 case LV_COLOR_FORMAT_I8:
