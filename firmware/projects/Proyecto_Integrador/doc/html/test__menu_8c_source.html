122

125 lv_obj_t * page3 = lv_menu_page_create(menu, "Page 3");

135
136 /* After clear history, back button click should not navigate back */

139 lv_area_t coords;
140 lv_obj_get_coords(back_btn, &coords);
141 lv_test_mouse_click_at((coords.x1 + coords.x2) / 2, (coords.y1 + coords.y2) / 2);
142
143 /* Should still be on page3 since history was cleared */
144 TEST_ASSERT_EQUAL_PTR(page3, lv_menu_get_cur_main_page(menu));
