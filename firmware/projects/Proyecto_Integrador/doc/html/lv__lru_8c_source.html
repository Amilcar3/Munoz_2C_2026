69lv_lru_t * lv_lru_create(size_t cache_size, size_t average_length, lv_lru_free_cb_t value_free,
70 lv_lru_free_cb_t key_free)
71{
72 // create the cache
73 lv_lru_t * cache = lv_malloc_zeroed(sizeof(lv_lru_t));
74 if(!cache) {
75 LV_LOG_WARN("LRU Cache unable to create cache object");
76 return NULL;
77 }
78 cache->hash_table_size = cache_size / average_length;
79 cache->average_item_length = average_length;
80 cache->free_memory = cache_size;
81 cache->total_memory = cache_size;
82 cache->seed = lv_rand(1, UINT32_MAX);
83 cache->value_free = value_free ? value_free : lv_free;
84 cache->key_free = key_free ? key_free : lv_free;
85
86 // size the hash table to a guesstimate of the number of slots required (assuming a perfect hash)
87 cache->items = lv_malloc_zeroed(sizeof(lv_lru_item_t *) * cache->hash_table_size);
88 if(!cache->items) {
89 LV_LOG_WARN("LRU Cache unable to create cache hash table");
90 lv_free(cache);
91 return NULL;
92 }
93 return cache;
94}

96void lv_lru_delete(lv_lru_t * cache)
97{
98 LV_ASSERT_NULL(cache);
99
100 // free each of the cached items, and the hash table
101 lv_lru_item_t * item = NULL, * next = NULL;
102 uint32_t i = 0;
103 if(cache->items) {
104 for(; i < cache->hash_table_size; i++) {
105 item = cache->items[i];
106 while(item) {
107 next = (lv_lru_item_t *) item->next;
108 cache->value_free(item->value);
109 cache->key_free(item->key);
110 cache->free_memory += item->value_length;
111 lv_free(item);
112 item = next;
113 }
114 }
115 lv_free(cache->items);
116 }
117
118 if(cache->free_items) {
119 item = cache->free_items;
120 while(item) {
121 next = (lv_lru_item_t *) item->next;
122 lv_free(item);
123 item = next;
124 }
125 }
126
127 // free the cache
128 lv_free(cache);
129}

131lv_lru_res_t lv_lru_set(lv_lru_t * cache, const void * key, size_t key_length, void * value, size_t value_length)
132{
133 test_for_missing_cache();
134 test_for_missing_key();
135 test_for_missing_value();
136 test_for_value_too_large();
137
138 // see if the key already exists
139 uint32_t hash_index = lv_lru_hash(cache, key, key_length);
140 int required = 0;
141 lv_lru_item_t * item = NULL, * prev = NULL;
142 item = cache->items[hash_index];
143
144 while(item && lv_lru_cmp_keys(item, key, key_length)) {
145 prev = item;
146 item = (lv_lru_item_t *) item->next;
147 }
148
149 if(item) {
150 // update the value and value_lengths
151 required = (int)(value_length - item->value_length);
152 cache->value_free(item->value);
153 item->value = value;
154 item->value_length = value_length;
155
156 }
157 else {
158 // insert a new item
159 item = lv_lru_pop_or_create_item(cache);
160 item->value = value;
161 item->key = lv_malloc(key_length);
162 lv_memcpy(item->key, key, key_length);
163 item->value_length = value_length;
164 item->key_length = key_length;
165 required = (int) value_length;
166
167 if(prev)
168 prev->next = item;
169 else
170 cache->items[hash_index] = item;
171 }
172 item->access_count = ++cache->access_count;
173
174 // remove as many items as necessary to free enough space
175 if(required > 0 && (size_t) required > cache->free_memory) {
176 while(cache->free_memory < (size_t) required)
177 lv_lru_remove_lru_item(cache);
178 }
179 cache->free_memory -= required;
180 return LV_LRU_OK;
181}

199 else {
200 *value = NULL;

228void lv_lru_remove_lru_item(lv_lru_t * cache)
229{
230 lv_lru_item_t * min_item = NULL, * min_prev = NULL;
231 lv_lru_item_t * item = NULL, * prev = NULL;
232 uint32_t i = 0, min_index = -1;
233 uint64_t min_access_count = -1;
234
235 for(; i < cache->hash_table_size; i++) {
236 item = cache->items[i];
237 prev = NULL;
238
239 while(item) {
240 if(item->access_count < min_access_count || (int64_t) min_access_count == -1) {
241 min_access_count = item->access_count;
242 min_item = item;
243 min_prev = prev;
244 min_index = i;
245 }
246 prev = item;
247 item = item->next;
248 }
249 }
250
251 if(min_item) {
252 lv_lru_remove_item(cache, min_prev, min_item, min_index);
253 }
254}

260static uint32_t lv_lru_hash(lv_lru_t * cache, const void * key, uint32_t key_length)
261{
262 uint32_t m = 0x5bd1e995;
263 uint32_t r = 24;
264 uint32_t h = cache->seed ^ key_length;
265 char * data = (char *) key;
266
267 while(key_length >= 4) {
268 uint32_t k = *(uint32_t *) data;
269 k *= m;
270 k ^= k >> r;
271 k *= m;
272 h *= m;
273 h ^= k;
274 data += 4;
275 key_length -= 4;
276 }
277
278 if(key_length >= 3) {
279 h ^= data[2] << 16;
280 }
281 if(key_length >= 2) {
282 h ^= data[1] << 8;
283 }
284 if(key_length >= 1) {
285 h ^= data[0];
286 h *= m;
287 }
288
289 h ^= h >> 13;
290 h *= m;
291 h ^= h >> 15;
292 return h % cache->hash_table_size;
293}

305static void lv_lru_remove_item(lv_lru_t * cache, lv_lru_item_t * prev, lv_lru_item_t * item, uint32_t hash_index)
306{
307 if(prev) {
308 prev->next = item->next;
309 }
310 else {
311 cache->items[hash_index] = (lv_lru_item_t *) item->next;
312 }
313
314 // free memory and update the free memory counter
315 cache->free_memory += item->value_length;
316 cache->value_free(item->value);
317 cache->key_free(item->key);
318
319 // push the item to the free items queue
320 lv_memzero(item, sizeof(lv_lru_item_t));
321 item->next = cache->free_items;
322 cache->free_items = item;
323}

325static lv_lru_item_t * lv_lru_pop_or_create_item(lv_lru_t * cache)
326{
327 lv_lru_item_t * item = NULL;
328

330 item = cache->free_items;

332 lv_memzero(item, sizeof(lv_lru_item_t));
333 }
334 else {
335 item = lv_malloc_zeroed(sizeof(lv_lru_item_t));
336 }
337
338 return item;
