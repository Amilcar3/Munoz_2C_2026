6def resolve_source_name(addr, prefix="lv_"):
7 """Resolve a code address to a short name derived from its source file.
8
9 Tries block_for_pc first, then find_pc_line as fallback.
10 Strips common prefixes (e.g. "lv_fs_stdio.c" -> "stdio").
11
12 Args:
13 addr: Integer address of a function/callback.
14 prefix: Prefix to strip from filename (default "lv_").
15
16 Returns:
17 Short name string, or None if resolution fails.
18 """
19 if not addr:
20 return None
21 fname = None
22 try:
23 block = gdb.block_for_pc(addr)
24 if block and block.function and block.function.symtab:
25 fname = block.function.symtab.filename
26 except (gdb.error, RuntimeError):
27 pass
28 if not fname:
29 try:
30 sal = gdb.find_pc_line(addr)
31 if sal and sal.symtab:
32 fname = sal.symtab.filename
33 except gdb.error:
34 pass
35 if not fname:
36 return None
37 base = os.path.basename(fname).replace(".c", "")
38 if prefix and base.startswith(prefix):
39 base = base[len(prefix) :]
40 return base

43def build_global_field_map(field_type_name):
44 """Build address -> field name map for fields of a given type in lv_global_t.
45
46 Scans lv_global_t struct fields, finds those matching the given type,
47 and returns a dict mapping their runtime addresses to field names.
48
49 Args:
50 field_type_name: C type name string (e.g. "lv_fs_drv_t").
51
52 Returns:
53 Dict mapping int address -> str field name.
54 """
55 try:
56 lv_global_val = gdb.parse_and_eval("lv_global")
57 lv_global_addr = int(lv_global_val.address)
58 lv_global_type = gdb.lookup_type("lv_global_t")
59 target_type = gdb.lookup_type(field_type_name)
60 result = {}
61 for field in lv_global_type.fields():
62 if field.type.strip_typedefs() == target_type:
63 addr = lv_global_addr + field.bitpos // 8
64 result[addr] = field.name
65 return result
66 except gdb.error:
67 return {}

71_COORD_TYPE_SHIFT = 29
72_COORD_TYPE_MASK = 3 << _COORD_TYPE_SHIFT
73_COORD_TYPE_PX = 0 << _COORD_TYPE_SHIFT

81def format_coord(val):
82 """Format an lv_coord_t value into a human-readable string.
83
84 Decodes special LVGL coordinate encodings:
85 - LV_SIZE_CONTENT -> "CONTENT"
86 - LV_PCT(x) -> "x%"
87 - LV_COORD_TYPE_PX_NEG -> negative pixel
88 - plain pixel -> "123"
89 """
90 val = int(val)
91 if val == _SIZE_CONTENT:
92 return "CONTENT"
93 coord_type = val & _COORD_TYPE_MASK
94 if coord_type == _COORD_TYPE_SPEC:
95 plain = val & ~_COORD_TYPE_MASK
96 if plain <= _PCT_POS_MAX:
97 return f"{plain}%"
98 return f"{_PCT_POS_MAX - plain}%"
99 # LV_COORD_TYPE_PX_NEG needs no branch of its own. There is no
100 # LV_COORD_SET_PX_NEG: the tag exists so LV_COORD_IS_PX() accepts negatives,
101 # and in two's complement every negative int32 already has those top bits,
102 # so such a value is an ordinary negative pixel count. Taking
103 # LV_COORD_PLAIN() of it and negating turned -5 into -536870907.
104 return str(val)

123def format_enum(type_name, value, prefix="", bitmask=False):
124 """Name a value of a C enum, reading the names from the target's debug info.
125
126 The fields holding these values are plain unsigned bitfields, so GDB prints
127 them as bare numbers and the enum has to be looked up by name. Reading it
128 from the target rather than from a generated table means the names always
129 match the binary in front of you - and when the enum was optimised out,
130 the raw number is still printed.
131 """
132 if value is None:
133 return None
134 value = int(value)
135 members = _enum_members(type_name)
136 if not members:
137 return str(value)
138
139 def _short(name):
140 return name[len(prefix):] if prefix and name.startswith(prefix) else name
141
142 if value in members:
143 return _short(members[value])
144 if not bitmask:
145 return str(value)
146
147 parts, covered = [], 0
148 for bit in sorted(b for b in members if b and not b & (b - 1)):
149 if value & bit:
150 parts.append(_short(members[bit]))
151 covered |= bit
152 return "|".join(parts) if parts and covered == value else str(value)
