87#define FB_FMT_AYUV 24 /* BPP=32 Combined YUV and alpha */

99#define FB_FMT_HDYC 31 /* BPP=16 UYVY except uses the BT709

109#define FB_FMT_VYUY 37 /* BPP=? ATI Packed YUV Data */

113#define FB_FMT_YVYU 39 /* BPP=16 YUV 4:2:2 */

117#define FB_FMT_Y41T 43 /* BPP=12 Y41P LSB for transparency */
118#define FB_FMT_Y42T 44 /* BPP=16 UYVY LSB for transparency */

123/* Packed Planar YUV Formats ************************************************/
124
125#define FB_FMT_YVU9 46 /* BPP=9 8-bit Y followed by 8-bit
126 * 4x4 VU */
127#define FB_FMT_YUV9 47 /* BPP=9? */
128#define FB_FMT_IF09 48 /* BPP=9.5 YVU9 + 4x4 plane of delta
129 * relative to tframe. */
130#define FB_FMT_YV16 49 /* BPP=16 8-bit Y followed by 8-bit
131 * 2x1 VU */
132#define FB_FMT_YV12 50 /* BPP=12 8-bit Y followed by 8-bit
133 * 2x2 VU */
134#define FB_FMT_I420 51 /* BPP=12 8-bit Y followed by 8-bit
135 * 2x2 UV */
136#define FB_FMT_IYUV FB_FMT_I420 /* BPP=12 */
137#define FB_FMT_NV12 52 /* BPP=12 8-bit Y followed by an
138 * interleaved 2x2 UV */
139#define FB_FMT_NV21 53 /* BPP=12 NV12 with UV reversed */
140#define FB_FMT_IMC1 54 /* BPP=12 YV12 except UV planes same
141 * stride as Y */
142#define FB_FMT_IMC2 55 /* BPP=12 IMC1 except UV lines
143 * interleaved at half stride
144 * boundaries */
145#define FB_FMT_IMC3 56 /* BPP=12 As IMC1 except that UV
146 * swapped */
147#define FB_FMT_IMC4 57 /* BPP=12 As IMC2 except that UV
148 * swapped */
149#define FB_FMT_CLPL 58 /* BPP=12 YV12 but including a level
150 * of indirection. */
151#define FB_FMT_Y41B 59 /* BPP=12? 4:1:1 planar. */
152#define FB_FMT_Y42B 60 /* BPP=16? YUV 4:2:2 planar. */
153#define FB_FMT_CXY1 61 /* BPP=12 */
