125lv_obj_t * lv_demo_gltf(const char * path)
126{
127 lv_obj_t * viewer = lv_gltf_create(lv_screen_active());
128 lv_obj_set_size(viewer, LV_PCT(100), LV_PCT(100));
129 lv_obj_set_scrollable(viewer, false);
130 lv_gltf_set_background_mode(viewer, LV_GLTF_BG_MODE_ENVIRONMENT);
131 lv_gltf_model_t * model = lv_gltf_load_model_from_file(viewer, path);
132 if(!model) {
133 lv_obj_delete(viewer);
134 return NULL;
135 }
136
137 init_subjects(viewer);
138 create_control_panel(viewer);
139
140 mouse_event_data_t * mouse_state = lv_zalloc(sizeof(*mouse_state));
141 LV_ASSERT_MALLOC(mouse_state);
142 mouse_state->sensitivity = 0.3;
143
144 lv_obj_add_event_cb(viewer, on_mouse_event, LV_EVENT_PRESSED, mouse_state);
145 lv_obj_add_event_cb(viewer, on_mouse_event, LV_EVENT_PRESSING, mouse_state);
146 lv_obj_add_event_cb(viewer, on_mouse_event, LV_EVENT_RELEASED, mouse_state);
147 lv_obj_add_event_cb(viewer, on_mouse_event, LV_EVENT_PRESS_LOST, mouse_state);
148 lv_obj_add_event_cb(viewer, on_mouse_event, LV_EVENT_DELETE, mouse_state);
149
150 lv_gltf_set_camera(viewer, LV_DEMO_GLTF_CAMERA_NUM);
151 return viewer;
152}

375static void on_mouse_event(lv_event_t * e)
376{
377 lv_event_code_t event_code = lv_event_get_code(e);
378 lv_obj_t * viewer = lv_event_get_target_obj(e);
379 lv_indev_t * indev = lv_indev_active();
380 mouse_event_data_t * mouse_state = (mouse_event_data_t *)lv_event_get_user_data(e);
381
382 if(!mouse_state) {
383 return;
384 }
385
386 lv_point_t current_pos;
387 lv_indev_get_point(indev, &current_pos);
388
389 switch(event_code) {
390 case LV_EVENT_PRESSED:
391 mouse_state->is_dragging = true;
392 mouse_state->last_pos = current_pos;
393 break;
394 case LV_EVENT_PRESSING:
395 if(mouse_state->is_dragging && lv_gltf_get_camera(viewer) == 0) {
396 int32_t delta_x = current_pos.x - mouse_state->last_pos.x;
397 int32_t delta_y = current_pos.y - mouse_state->last_pos.y;
398
399 float current_yaw = lv_gltf_get_yaw(viewer);
400 float current_pitch = lv_gltf_get_pitch(viewer);
401
402 float new_yaw = current_yaw + (delta_x * -mouse_state->sensitivity);
403 float new_pitch = current_pitch + (delta_y * -mouse_state->sensitivity);
404
405 if(new_pitch > 89.0f)
406 new_pitch = 89.0f;
407 if(new_pitch < -89.0f)
408 new_pitch = -89.0f;
409
410 lv_subject_set_float(yaw_subject, new_yaw);
411 lv_subject_set_float(pitch_subject, new_pitch);
412 }
413 mouse_state->last_pos = current_pos;
414 break;
415
416 case LV_EVENT_RELEASED:
417 case LV_EVENT_PRESS_LOST:
418 mouse_state->is_dragging = false;
419 break;
420 case LV_EVENT_DELETE:
421 lv_free(mouse_state);
422 default:
423 break;
424 }
425}

427static void on_animation_play_pause_event(lv_event_t * e)
428{
429 lv_event_code_t code = lv_event_get_code(e);
430 lv_obj_t * btn = lv_event_get_target_obj(e);
431 play_pause_event_data_t * event_data = (play_pause_event_data_t *)lv_event_get_user_data(e);
432
433 if(code == LV_EVENT_CLICKED) {
434 lv_gltf_model_t * model = lv_gltf_get_primary_model(event_data->viewer);
435
436 if(lv_gltf_model_is_animation_paused(model)) {
437 lv_gltf_model_play_animation(model, lv_gltf_model_get_animation(model));
438 lv_obj_set_style_bg_color(btn, PAUSE_BTN_COLOR, LV_PART_MAIN);
439 lv_label_set_text_static(event_data->label, LV_SYMBOL_PAUSE);
440 }
441 else {
442 lv_obj_set_style_bg_color(btn, PLAY_BTN_COLOR, LV_PART_MAIN);
443 lv_label_set_text_static(event_data->label, LV_SYMBOL_PLAY);
444 lv_gltf_model_pause_animation(model);
445 }
446 }
447 else if(code == LV_EVENT_DELETE) {
448 lv_free(event_data);
449 }
450 else {
451 LV_ASSERT_FORMAT_MSG(0, "Unhandled event: %d", code);
452 }
453}
