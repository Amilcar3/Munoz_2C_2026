50lv_gltf_model_loader_t * lv_gltf_model_loader_create(void)
51{
52 lv_gltf_model_loader_t * loader = lv_zalloc(sizeof(*loader));
53 if(!loader) {
54 return NULL;
55 }
56
57 lv_rb_init(&loader->textures_map,
58 (lv_rb_compare_t)opengl_texture_compare_cb,
59 sizeof(lv_opengl_texture_t));
60 return loader;

63void lv_gltf_model_loader_delete(lv_gltf_model_loader_t * loader)
64{
65 if(!loader) {
66 return;
67 }
68
69 lv_rb_node_t * node;
70 while((node = loader->textures_map.root)) {
71 lv_opengl_texture_t * texture = node->data;
72 GL_CALL(glDeleteTextures(1, &texture->id));
73 lv_rb_drop_node(&loader->textures_map, node);
74 }
75
76 lv_rb_destroy(&loader->textures_map);
77 lv_free(loader);
78}

80void lv_gltf_model_loader_store_texture(lv_gltf_model_loader_t * loader, uint32_t texture_hash, uint32_t texture_id)
81{
82 LV_CHECK_ARG(loader != NULL, return);
83 lv_opengl_texture_t key = { .id = texture_id, .hash = texture_hash };
84 lv_rb_node_t * node = lv_rb_insert(&loader->textures_map, &key);
85 if(!node) {
86 LV_LOG_WARN("Failed to cache texture hash: %d id: %d",
87 texture_hash, texture_id);
88 return;
89 }
90 lv_memcpy(node->data, &key, sizeof(key));
91}

94uint32_t lv_gltf_model_loader_get_texture(lv_gltf_model_loader_t * loader, uint32_t texture_hash)
95{
96 LV_CHECK_ARG(loader != NULL, return 0);
97 lv_opengl_texture_t key = { .hash = texture_hash };
98 lv_rb_node_t * node = lv_rb_find(&loader->textures_map, &key);
99 if(!node) {
100 LV_LOG_INFO("Couldn't find texture with hash %" LV_PRIu32 " in cache",
101 texture_hash);
102 return GL_NONE;
103 }
104 return ((lv_opengl_texture_t *)node->data)->id;
105}
