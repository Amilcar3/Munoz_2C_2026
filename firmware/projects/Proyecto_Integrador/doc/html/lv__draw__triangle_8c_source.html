59void lv_draw_triangle(lv_layer_t * layer, const lv_draw_triangle_dsc_t * draw_dsc)
60{
61 LV_CHECK_ARG(layer != NULL, return);
62 LV_CHECK_ARG(draw_dsc != NULL, return);
63
64 if(draw_dsc->opa <= LV_OPA_MIN) return;
65
66 LV_PROFILER_DRAW_BEGIN;
67
68 lv_area_t a;
69 a.x1 = (int32_t)LV_MIN3(draw_dsc->p[0].x, draw_dsc->p[1].x, draw_dsc->p[2].x);
70 a.y1 = (int32_t)LV_MIN3(draw_dsc->p[0].y, draw_dsc->p[1].y, draw_dsc->p[2].y);
71 a.x2 = (int32_t)LV_MAX3(draw_dsc->p[0].x, draw_dsc->p[1].x, draw_dsc->p[2].x);
72 a.y2 = (int32_t)LV_MAX3(draw_dsc->p[0].y, draw_dsc->p[1].y, draw_dsc->p[2].y);
73
74 if(draw_dsc->base.drop_shadow_opa) {
75 lv_layer_t * ds_layer = lv_draw_layer_create_drop_shadow(layer, &draw_dsc->base, &a);
76 LV_ASSERT_NULL(ds_layer);
77 lv_draw_triangle_dsc_t ds_dsc = *draw_dsc;
78 ds_dsc.base.drop_shadow_opa = 0; /*Disable drop shadow so rendering below will render plain triangle*/
79 lv_draw_triangle(ds_layer, &ds_dsc);
80 lv_draw_layer_finish_drop_shadow(ds_layer, &draw_dsc->base);
81 }
82
83
84 lv_draw_task_t * t = lv_draw_add_task(layer, &a, LV_DRAW_TASK_TYPE_TRIANGLE);
85
86 lv_memcpy(t->draw_dsc, draw_dsc, sizeof(*draw_dsc));
87
88 lv_draw_finalize_task_creation(layer, t);
89 LV_PROFILER_DRAW_END;
90}
