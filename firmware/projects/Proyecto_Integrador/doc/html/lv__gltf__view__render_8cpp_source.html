136void lv_gltf_view_render_deinit(lv_gltf_t * viewer)
137{
138 /* Releasing the render targets unbinds the framebuffer, and this runs while the rest
139 * of the screen is being drawn, so the binding has to be put back */
140 GLint prev_framebuffer = 0;
141 GL_CALL(glGetIntegerv(GL_FRAMEBUFFER_BINDING, &prev_framebuffer));
142
143 setup_cleanup_opengl_output(&viewer->state.render_state);
144 setup_cleanup_opengl_output(&viewer->state.opaque_render_state);
145 viewer->state.render_state_ready = false;
146
147 GL_CALL(glBindFramebuffer(GL_FRAMEBUFFER, prev_framebuffer));
148}

150GLuint lv_gltf_view_render(lv_gltf_t * viewer)
151{
152 const size_t n = lv_array_size(&viewer->models);
153
154 if(n == 0) {
155 return GL_NONE;
156 }
157 lv_gltf_model_data_t * model_data = (lv_gltf_model_data_t *)lv_array_at(&viewer->models, 0);
158
159 bool dirty = lv_memcmp(&viewer->last_desc, &viewer->desc, sizeof(viewer->desc)) != 0;
160
161 for(size_t i = 0; i < n; ++i) {
162 lv_gltf_model_data_t * model_data = (lv_gltf_model_data_t *)lv_array_at(&viewer->models, i);
163 dirty |= model_data->model->is_animation_enabled || model_data->model->write_ops_pending;
164 }
165
166 GLuint texture_id = GL_NONE;
167 texture_id = lv_gltf_view_render_model(viewer, model_data, true, dirty);
168 for(size_t i = 1; i < n; ++i) {
169 lv_gltf_model_data_t * model_data = (lv_gltf_model_data_t *)lv_array_at(&viewer->models, i);
170 lv_gltf_view_render_model(viewer, model_data, false, dirty);
171 }
172
173 lv_memcpy(&(viewer->last_desc), &viewer->desc, sizeof(viewer->desc));
174
175 return texture_id;
176}

182static void lv_gltf_view_push_opengl_state(lv_opengl_state_t * state)
183{
184 /* Blend state */
185 GL_CALL(glGetBooleanv(GL_BLEND, &state->blend_enabled));
186 GL_CALL(glGetIntegerv(GL_BLEND_SRC_ALPHA, &state->blend_src));
187 GL_CALL(glGetIntegerv(GL_BLEND_DST_ALPHA, &state->blend_dst));
188 GL_CALL(glGetIntegerv(GL_BLEND_EQUATION, &state->blend_equation));
189
190 /* Depth state */
191 GL_CALL(glGetBooleanv(GL_DEPTH_TEST, &state->depth_test_enabled));
192 GL_CALL(glGetBooleanv(GL_DEPTH_WRITEMASK, &state->depth_mask));
193 GL_CALL(glGetIntegerv(GL_DEPTH_FUNC, &state->depth_func));
194
195 /* Face culling state */
196 GL_CALL(glGetBooleanv(GL_CULL_FACE, &state->cull_face_enabled));
197 GL_CALL(glGetIntegerv(GL_CULL_FACE_MODE, &state->cull_face_mode));
198 GL_CALL(glGetIntegerv(GL_FRONT_FACE, &state->front_face));
199
200 /* Stencil state */
201 GL_CALL(glGetBooleanv(GL_STENCIL_TEST, &state->stencil_test_enabled));
202 GL_CALL(glGetIntegerv(GL_STENCIL_WRITEMASK, (GLint *)&state->stencil_mask));
203 GL_CALL(glGetIntegerv(GL_STENCIL_FUNC, &state->stencil_func));
204 GL_CALL(glGetIntegerv(GL_STENCIL_REF, &state->stencil_ref));
205 GL_CALL(glGetIntegerv(GL_STENCIL_VALUE_MASK, (GLint *)&state->stencil_value_mask));
206
207 /* Buffer bindings */
208#ifndef GL_VERTEX_ARRAY_BINDING
209#ifdef GL_VERTEX_ARRAY_BINDING_OES
210#define GL_VERTEX_ARRAY_BINDING GL_VERTEX_ARRAY_BINDING_OES
211#else
212#define GL_VERTEX_ARRAY_BINDING 0x85B5
213#endif
214#endif
215 GL_CALL(glGetIntegerv(GL_VERTEX_ARRAY_BINDING, (GLint *)&state->current_vao));
216 GL_CALL(glGetIntegerv(GL_ARRAY_BUFFER_BINDING, (GLint *)&state->current_vbo));
217 GL_CALL(glGetIntegerv(GL_ELEMENT_ARRAY_BUFFER_BINDING, (GLint *)&state->current_ibo));
218 GL_CALL(glGetIntegerv(GL_CURRENT_PROGRAM, (GLint *)&state->current_program));
219 GL_CALL(glGetIntegerv(GL_FRAMEBUFFER_BINDING, (GLint *)&state->current_framebuffer));
220
221 /* Texture state */
222 GL_CALL(glGetIntegerv(GL_ACTIVE_TEXTURE, &state->active_texture));
223 GL_CALL(glGetIntegerv(GL_TEXTURE_BINDING_2D, (GLint *)&state->bound_texture_2d));
224
225 /* Viewport and scissor */
226 GL_CALL(glGetIntegerv(GL_VIEWPORT, state->viewport));
227 GL_CALL(glGetBooleanv(GL_SCISSOR_TEST, &state->scissor_test_enabled));
228 GL_CALL(glGetIntegerv(GL_SCISSOR_BOX, state->scissor_box));
229
230 /* Clear values */
231 GL_CALL(glGetFloatv(GL_COLOR_CLEAR_VALUE, state->clear_color));
232 GL_CALL(glGetFloatv(GL_DEPTH_CLEAR_VALUE, &state->clear_depth));
233}

235static void lv_gltf_view_pop_opengl_state(const lv_opengl_state_t * state)
236{
237 /* Restore blend state */
238 if(state->blend_enabled) {
239 GL_CALL(glEnable(GL_BLEND));
240 }
241 else {
242 GL_CALL(glDisable(GL_BLEND));
243 }
244 GL_CALL(glBlendFunc(state->blend_src, state->blend_dst));
245 GL_CALL(glBlendEquation(state->blend_equation));
246
247 /* Restore depth state */
248 if(state->depth_test_enabled) {
249 GL_CALL(glEnable(GL_DEPTH_TEST));
250 }
251 else {
252 GL_CALL(glDisable(GL_DEPTH_TEST));
253 }
254 GL_CALL(glDepthMask(state->depth_mask));
255 GL_CALL(glDepthFunc(state->depth_func));
256
257 /* Restore face culling state */
258 if(state->cull_face_enabled) {
259 GL_CALL(glEnable(GL_CULL_FACE));
260 }
261 else {
262 GL_CALL(glDisable(GL_CULL_FACE));
263 }
264 GL_CALL(glCullFace(state->cull_face_mode));
265 GL_CALL(glFrontFace(state->front_face));
266
267 /* Restore stencil state */
268 if(state->stencil_test_enabled) {
269 GL_CALL(glEnable(GL_STENCIL_TEST));
270 }
271 else {
272 GL_CALL(glDisable(GL_STENCIL_TEST));
273 }
274 GL_CALL(glStencilMask(state->stencil_mask));
275 GL_CALL(glStencilFunc(state->stencil_func, state->stencil_ref, state->stencil_value_mask));
276
277 /* Restore buffer bindings */
278 GL_CALL(glBindFramebuffer(GL_FRAMEBUFFER, state->current_framebuffer));
279 GL_CALL(glBindVertexArray(state->current_vao));
280 GL_CALL(glBindBuffer(GL_ARRAY_BUFFER, state->current_vbo));
281 GL_CALL(glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, state->current_ibo));
282 GL_CALL(glUseProgram(state->current_program));
283
284 /* Restore texture state */
285 GL_CALL(glActiveTexture(state->active_texture));
286 GL_CALL(glBindTexture(GL_TEXTURE_2D, state->bound_texture_2d));
287
288 /* Restore viewport and scissor */
289 GL_CALL(glViewport(state->viewport[0], state->viewport[1], state->viewport[2], state->viewport[3]));
290 if(state->scissor_test_enabled) {
291 GL_CALL(glEnable(GL_SCISSOR_TEST));
292 }
293 else {
294 GL_CALL(glDisable(GL_SCISSOR_TEST));
295 }
296 GL_CALL(glScissor(state->scissor_box[0], state->scissor_box[1], state->scissor_box[2], state->scissor_box[3]));
297
298 /* Restore clear values */
299 GL_CALL(glClearColor(state->clear_color[0], state->clear_color[1], state->clear_color[2], state->clear_color[3]));
300 GL_CALL(glClearDepthf(state->clear_depth));
301}

303static GLuint lv_gltf_view_render_model(lv_gltf_t * viewer, lv_gltf_model_data_t * modeld, bool is_first_model,
304 bool dirty)
305{
306 lv_gltf_view_state_t * vstate = &viewer->state;
307 lv_gltf_view_desc_t * view_desc = &viewer->desc;
308 bool opt_draw_bg = is_first_model && (view_desc->bg_mode == LV_GLTF_BG_MODE_ENVIRONMENT);
309 bool opt_aa_this_frame = (view_desc->aa_mode == LV_GLTF_AA_MODE_ON) ||
310 (view_desc->aa_mode == LV_GLTF_AA_MODE_DYNAMIC && modeld->last_frame_no_motion == true);
311 if(!is_first_model) {
312 /* If this data object is a secondary render pass, inherit the anti-alias setting for this frame from the first gltf_data drawn*/
313 opt_aa_this_frame = view_desc->frame_was_antialiased;
314 }
315
316 lv_opengl_state_t opengl_state;
317 lv_gltf_view_push_opengl_state(&opengl_state);
318
319 int32_t last_render_w = view_desc->render_width;
320 int32_t last_render_h = view_desc->render_height;
321 view_desc->render_width = lv_obj_get_width((lv_obj_t *)viewer) * (opt_aa_this_frame ? 2 : 1);
322 view_desc->render_height = lv_obj_get_height((lv_obj_t *)viewer) * (opt_aa_this_frame ? 2 : 1);
323
324 bool new_size = last_render_h != view_desc->render_height || last_render_w != view_desc->render_width;
325
326 if(opt_aa_this_frame != modeld->last_frame_was_antialiased) {
327 /* Antialiasing state has changed since the last render, so the render target has to be
328 * rebuilt with the mipmap setting the new mode needs */
329 if(is_first_model) {
330 new_size = true;
331 }
332 modeld->last_frame_was_antialiased = opt_aa_this_frame;
333 }
334
335 view_desc->frame_was_antialiased = opt_aa_this_frame;
336
337 if(new_size || !vstate->render_state_ready) {
338 /* The previous target is replaced, not kept, so it has to be released first */
339 setup_cleanup_opengl_output(&vstate->render_state);
340 vstate->render_state_ready = true;
341 vstate->render_state =
342 setup_primary_output(view_desc->render_width, view_desc->render_height, opt_aa_this_frame);
343 setup_finish_frame();
344 }
345 if(vstate->render_opaque_buffer && vstate->opaque_render_state.framebuffer == 0) {
346 vstate->opaque_render_state =
347 setup_opaque_output(vstate->opaque_frame_buffer_width, vstate->opaque_frame_buffer_height);
348 setup_finish_frame();
349 }
350
351 bool last_frame_no_motion = modeld->_last_frame_no_motion;
352 modeld->_last_frame_no_motion = modeld->last_frame_no_motion;
353 modeld->last_frame_no_motion = true;
354
355
356 if(dirty || lv_gltf_model_needs_transforms(modeld->model) ||
357 (modeld->model->camera != modeld->last_camera_index)) {
358 modeld->last_frame_no_motion = false;
359 lv_gltf_view_recache_all_transforms(modeld);
360 }
361 else if(modeld->last_frame_no_motion && modeld->_last_frame_no_motion && last_frame_no_motion) {
362 /* Nothing changed at all, return the previous output frame */
363 setup_finish_frame();
364 lv_gltf_view_pop_opengl_state(&opengl_state);
365 return vstate->render_state.texture;
366 }
367
368 render_skins(viewer, modeld);
369
370 NodeDistanceVector distance_sort_nodes;
371
372 for(const auto & kv : modeld->model->blended_nodes_by_material_index) {
373 for(const auto & pair : kv.second) {
374 auto node = pair.first;
375 auto new_node = NodeIndexDistancePair(
376 fastgltf::math::length(
377 modeld->model->view_pos -
378 lv_gltf_data_get_centerpoint(modeld->model, lv_gltf_data_get_node_transform(modeld->model, node),
379 node->meshIndex.value(), pair.second)),
380 NodeIndexPair(node, pair.second));
381 distance_sort_nodes.push_back(new_node);
382 }
383 }
384 std::sort(distance_sort_nodes.begin(), distance_sort_nodes.end(),
385 [](const NodeIndexDistancePair & a, const NodeIndexDistancePair & b) {
386 return a.first > b.first;
387 });
388
389 /* Reset the last material index to an unused value once per frame at the start*/
390 modeld->last_material_index = 99999;
391
392 if(vstate->render_opaque_buffer) {
393 std::optional<lv_gltf_matrices_saver_t> saver;
394 if(!is_first_model) {
395 /* Cache the current matrices */
396 saver.emplace(viewer);
397 }
398
399 if(modeld->model->camera > 0) {
400 setup_view_proj_matrix_from_camera(viewer, modeld->model->camera - 1, view_desc, modeld->model, true);
401 }
402 else {
403 setup_view_proj_matrix(viewer, view_desc, modeld->model, true);
404 }
405 lv_result_t result = setup_restore_opaque_output(viewer, &vstate->opaque_render_state,
406 vstate->opaque_frame_buffer_width,
407 vstate->opaque_frame_buffer_height, is_first_model);
408 LV_ASSERT_MSG(result == LV_RESULT_OK, "Failed to setup opaque output which should never happen");
409 if(result != LV_RESULT_OK) {
410 lv_gltf_view_pop_opengl_state(&opengl_state);
411 return vstate->render_state.texture;
412 }
413
414 if(opt_draw_bg) {
415 setup_draw_environment_background(&viewer->shader_manager, viewer, view_desc->blur_bg);
416 }
417
418 render_materials(viewer, modeld, modeld->model->opaque_nodes_by_material_index, true);
419
420 for(const auto & node_distance_pair : distance_sort_nodes) {
421 const auto & node_element = node_distance_pair.second;
422 const auto & node = node_element.first;
423 draw_primitive(node_element.second, viewer, modeld, *node, node->meshIndex.value(),
424 lv_gltf_data_get_node_transform(modeld->model, node), viewer->environment, true);
425 }
426
427 GL_CALL(glBindTexture(GL_TEXTURE_2D, vstate->opaque_render_state.texture));
428 GL_CALL(glGenerateMipmap(GL_TEXTURE_2D));
429 GL_CALL(glBindTexture(GL_TEXTURE_2D, GL_NONE));
430 setup_finish_frame();
431 }
432
433 if(is_first_model) {
434 if(modeld->model->camera > 0) {
435 setup_view_proj_matrix_from_camera(viewer, modeld->model->camera - 1, view_desc, modeld->model, false);
436 }
437 else {
438 setup_view_proj_matrix(viewer, view_desc, modeld->model, false);
439 }
440 }
441
442 lv_result_t result = render_primary_output(viewer, &vstate->render_state, view_desc->render_width,
443 view_desc->render_height, is_first_model);
444
445 LV_ASSERT_MSG(result == LV_RESULT_OK, "Failed to restore primary output which should never happen");
446 if(result != LV_RESULT_OK) {
447 lv_gltf_view_pop_opengl_state(&opengl_state);
448 return vstate->render_state.texture;
449 }
450 if(opt_draw_bg) {
451 setup_draw_environment_background(&viewer->shader_manager, viewer, view_desc->blur_bg);
452 }
453
454 render_materials(viewer, modeld, modeld->model->opaque_nodes_by_material_index, false);
455
456 for(const auto & node_distance_pair : distance_sort_nodes) {
457 const auto & node_element = node_distance_pair.second;
458 const auto & node = node_element.first;
459 draw_primitive(node_element.second, viewer, modeld, *node, node->meshIndex.value(),
460 lv_gltf_data_get_node_transform(modeld->model, node), viewer->environment, false);
461 }
462 if(opt_aa_this_frame) {
463 GL_CALL(glBindTexture(GL_TEXTURE_2D, vstate->render_state.texture));
464 GL_CALL(glGenerateMipmap(GL_TEXTURE_2D));
465 GL_CALL(glBindTexture(GL_TEXTURE_2D, GL_NONE));
466 }
467 setup_finish_frame();
468 lv_gltf_view_pop_opengl_state(&opengl_state);
469 return vstate->render_state.texture;
470}

493static void render_skins(lv_gltf_t * viewer, lv_gltf_model_data_t * modeld)
494{
495 LV_UNUSED(viewer);
496 size_t skin_count = lv_gltf_data_get_skins_size(modeld->model);
497 if(skin_count == 0) {
498 return;
499 }
500 const size_t skin_texture_count = lv_array_size(&modeld->skin_textures);
501 for(size_t i = 0; i < skin_texture_count; ++i) {
502 GLuint * texture_id = (GLuint *)lv_array_at(&modeld->skin_textures, i);
503 GL_CALL(glDeleteTextures(1, texture_id));
504 }
505 lv_array_clear(&modeld->skin_textures);
506
507 for(size_t i = 0; i < skin_count; ++i) {
508 const auto & skin_index = lv_gltf_data_get_skin(modeld->model, i);
509 const auto & skin = modeld->model->asset.skins[skin_index];
510 auto & ibm = modeld->model->ibm_by_skin_then_node[skin_index];
511
512 size_t num_joints = skin.joints.size();
513 size_t tex_width = std::ceil(std::sqrt((float)num_joints * 8.0f));
514
515 GLuint rtex;
516 GL_CALL(glGenTextures(1, &rtex));
517 lv_array_push_back(&modeld->skin_textures, &rtex);
518
519 GL_CALL(glBindTexture(GL_TEXTURE_2D, rtex));
520 GL_CALL(glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_S, GL_CLAMP_TO_EDGE));
521 GL_CALL(glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_T, GL_CLAMP_TO_EDGE));
522 GL_CALL(glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_NEAREST));
523 GL_CALL(glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_NEAREST));
524 float * texture_data = (float *)lv_malloc(tex_width * tex_width * 4 * sizeof(*texture_data));
525 LV_ASSERT_MALLOC(texture_data);
526 size_t texture_data_index = 0;
527
528 for(uint64_t j = 0; j < num_joints; j++) {
529 auto & joint_node = modeld->model->asset.nodes[skin.joints[j]];
530 fastgltf::math::fmat4x4 final_joint_matrix =
531 lv_gltf_data_get_node_transform(modeld->model, &joint_node) * ibm[&joint_node];
532
533 lv_memcpy(&texture_data[texture_data_index], final_joint_matrix.data(), sizeof(float) * 16);
534 lv_memcpy(&texture_data[texture_data_index + 16],
535 fastgltf::math::transpose(fastgltf::math::inverse(final_joint_matrix)).data(),
536 sizeof(float) * 16);
537
538 texture_data_index += 32;
539 }
540 GL_CALL(glTexImage2D(GL_TEXTURE_2D, 0, GL_RGBA32F, tex_width, tex_width, 0, GL_RGBA, GL_FLOAT, texture_data));
541 GL_CALL(glPixelStorei(GL_UNPACK_ALIGNMENT, 1));
542 GL_CALL(glBindTexture(GL_TEXTURE_2D, GL_NONE));
543 lv_free(texture_data);
544 }

567static bool setup_primitive(int32_t prim_num, lv_gltf_t * viewer, lv_gltf_model_data_t * modeld, fastgltf::Node & node,
568 std::size_t mesh_index, const fastgltf::math::fmat4x4 & matrix,

588 if((modeld->last_material_index == materialIndex) && (modeld->last_pass_was_transmission == is_transmission_pass)) {

635 GL_CALL(glActiveTexture(GL_TEXTURE0 + tex_num));

639 }
640 if(env_tex->diffuse != GL_NONE) {
641 GL_CALL(glActiveTexture(GL_TEXTURE0 + tex_num));
642 GL_CALL(glBindTexture(GL_TEXTURE_CUBE_MAP, env_tex->diffuse));
643 GL_CALL(glUniform1i(uniforms->env_diffuse_sampler, tex_num++));

668 GL_CALL(glUniform1i(uniforms->env_sheen_e_lut_sampler, tex_num++));

871static void draw_lights(lv_gltf_model_data_t * modeld, GLuint program)
872{
873
874 lv_gltf_model_t * model = modeld->model;
875
876 if(model->node_by_light_index.empty()) {
877 return;
878 }
879 size_t max_light_nodes = model->node_by_light_index.size();
880 size_t max_scene_lights = model->asset.lights.size();
881 if(max_scene_lights != max_light_nodes) {
882 LV_LOG_ERROR("Scene light count (%zu) != scene light node count (%zu)\n", max_scene_lights, max_light_nodes);
883 return;
884 }
885
886 char tag[100];
887 char prefix[20];
888 for(size_t i = 0; i < max_scene_lights; i++) {
889 // Update each field of the light struct
890 lv_snprintf(prefix, sizeof(prefix), "u_Lights[%zu]", i + 1);
891 auto & lightNode = model->node_by_light_index[i];
892 const fastgltf::math::fmat4x4 & light_matrix = lv_gltf_data_get_node_transform(modeld->model, lightNode);
893
894 lv_snprintf(tag, sizeof(tag), "%s.position", prefix);
895 glUniform3fv(glGetUniformLocation(program, tag), 1, &light_matrix[3][0]);
896
897 lv_snprintf(tag, sizeof(tag), "%s.direction", prefix);
898 float tlight_dir[3] = { -light_matrix[2][0], -light_matrix[2][1], -light_matrix[2][2] };
899
900 glUniform3fv(glGetUniformLocation(program, tag), 1, &tlight_dir[0]);
901
902 lv_snprintf(tag, sizeof(tag), "%s.range", prefix);
903 const auto & m = light_matrix.data();
904
905 if(model->asset.lights[i].range.has_value()) {
906 float light_scale = fastgltf::math::length(fastgltf::math::fvec3(m[0], m[4], m[8]));
907 glUniform1f(glGetUniformLocation(program, tag), model->asset.lights[i].range.value() * light_scale);
908 }
909 else {
910 glUniform1f(glGetUniformLocation(program, tag), 9999.f);
911 }
912
913 lv_snprintf(tag, sizeof(tag), "%s.color", prefix);
914 glUniform3fv(glGetUniformLocation(program, tag), 1, &(model->asset.lights[i].color.data()[0]));
915
916 lv_snprintf(tag, sizeof(tag), "%s.intensity", prefix);
917 glUniform1f(glGetUniformLocation(program, tag), model->asset.lights[i].intensity);
918
919 lv_snprintf(tag, sizeof(tag), "%s.innerConeCos", prefix);
920 if(model->asset.lights[i].innerConeAngle.has_value()) {
921 glUniform1f(glGetUniformLocation(program, tag),
922 std::cos(model->asset.lights[i].innerConeAngle.value()));
923
924 }
925 else {
926 glUniform1f(glGetUniformLocation(program, tag), -1.0f);
927 }
928
929 lv_snprintf(tag, sizeof(tag), "%s.outerConeCos", prefix);
930
931 if(model->asset.lights[i].outerConeAngle.has_value()) {
932 glUniform1f(glGetUniformLocation(program, tag),
933 std::cos(model->asset.lights[i].outerConeAngle.value()));
934
935 }
936 else {
937 glUniform1f(glGetUniformLocation(program, tag), -1.0f);
938 }
939 lv_snprintf(tag, sizeof(tag), "%s.type", prefix);
940 glUniform1i(glGetUniformLocation(program, tag), (GLint)model->asset.lights[i].type);
941 }
942}

945 int32_t texture_h, bool prepare_bg)

968static fastgltf::math::nvec4 color_convert_to_srgb(fastgltf::math::nvec4 color)
969{
970 const float SRGB_GAMMA = 2.4f;
971 const float INV_SRGB_GAMMA = 1.0f / SRGB_GAMMA;
972 // Apply the sRGB conversion formula:
973 // sRGB = 12.92 * C, if C <= 0.0031308
974 // sRGB = 1.055 * C^(1/2.4) - 0.055, if C > 0.0031308
975
976 fastgltf::math::nvec4 srgbColor;
977 for(int i = 0; i < 3; i++) { // Loop through R, G, B channels
978 float c = color[i];
979 if(c <= 0.0031308f) {
980 srgbColor[i] = 12.92f * c;
981 }
982 else {
983 srgbColor[i] = 1.055f * pow(c, INV_SRGB_GAMMA) - 0.055f;
984 }
985 }
986
987 srgbColor[3] = color[3];
988 return srgbColor;
989}

1022static fastgltf::math::fmat3x3 create_texture_transform_matrix(std::unique_ptr<fastgltf::TextureTransform> & transform)
1023{
1024 fastgltf::math::fmat3x3 rotation = fastgltf::math::fmat3x3(0.f);
1025 fastgltf::math::fmat3x3 scale = fastgltf::math::fmat3x3(0.f);
1026 fastgltf::math::fmat3x3 translation = fastgltf::math::fmat3x3(0.f);
1027 fastgltf::math::fmat3x3 result = fastgltf::math::fmat3x3(0.f);
1028
1029 float s = std::sin(transform->rotation);
1030 float c = std::cos(transform->rotation);
1031 rotation[0][0] = c;
1032 rotation[1][1] = c;
1033 rotation[0][1] = s;
1034 rotation[1][0] = -s;
1035 rotation[2][2] = 1.0f;
1036
1037 scale[0][0] = transform->uvScale[0];
1038 scale[1][1] = transform->uvScale[1];
1039 scale[2][2] = 1.0f;
1040
1041 translation[0][0] = 1.0f;
1042 translation[1][1] = 1.0f;
1043 translation[0][2] = transform->uvOffset[0];
1044 translation[1][2] = transform->uvOffset[1];
1045 translation[2][2] = 1.0f;
1046
1047 result = translation * rotation;
1048 result = result * scale;
1049 return result;
1050}

1159static void setup_view_proj_matrix_from_camera(lv_gltf_t * viewer, uint32_t camera, lv_gltf_view_desc_t * view_desc,
1160 lv_gltf_model_t * model, bool transmission_pass)
1161{
1162 /* The following matrix math is for the projection matrices as defined by the glTF spec:*/
1163 /* https://registry.khronos.org/glTF/specs/2.0/glTF-2.0.html#projection-matrices*/
1164
1165 fastgltf::math::fmat4x4 projection;
1166 const auto & asset = lv_gltf_data_get_asset(model);
1167
1168 auto width = view_desc->render_width;
1169 auto height = view_desc->render_height;
1170 /* It's possible the transmission pass should simply use the regular passes aspect despite having different metrics itself. */
1171 /* TODO: test both ways to see which has less distortion*/
1172
1173 float aspect = (float)width / (float)height;
1174 if(transmission_pass) {
1175 width = LV_GLTF_TRANSMISSION_PASS_SIZE;
1176 height = LV_GLTF_TRANSMISSION_PASS_SIZE;
1177 }
1178
1179 std::visit(fastgltf::visitor{
1180 [&](fastgltf::Camera::Perspective & perspective)
1181 {
1182 projection = fastgltf::math::fmat4x4(0.0f);
1183 projection[0][0] = 1.f / (aspect * tan(0.5f * perspective.yfov));
1184 projection[1][1] = 1.f / (tan(0.5f * perspective.yfov));
1185 projection[2][3] = -1;
1186
1187 if(perspective.zfar.has_value()) {
1188 // Finite projection matrix
1189 projection[2][2] = (*perspective.zfar + perspective.znear) /
1190 (perspective.znear - *perspective.zfar);
1191 projection[3][2] = (2 * *perspective.zfar * perspective.znear) /
1192 (perspective.znear - *perspective.zfar);
1193 }
1194 else {
1195 // Infinite projection matrix
1196 projection[2][2] = -1;
1197 projection[3][2] = -2 * perspective.znear;
1198 }
1199 },
1200 [&](fastgltf::Camera::Orthographic & orthographic)
1201 {
1202 projection = fastgltf::math::fmat4x4(1.0f);
1203 projection[0][0] = (1.f / orthographic.xmag) * aspect;
1204 projection[1][1] = 1.f / orthographic.ymag;
1205 projection[2][2] = 2.f / (orthographic.znear - orthographic.zfar);
1206 projection[3][2] =
1207 (orthographic.zfar + orthographic.znear) / (orthographic.znear - orthographic.zfar);
1208 },
1209 },
1210 asset->cameras[camera].camera);
1211
1212 viewer->view_matrix = model->view_mat;
1213 viewer->camera_pos = model->view_pos;
1214 viewer->projection_matrix = projection;
1215 viewer->view_projection_matrix = projection * viewer->view_matrix;
1216}

1218static void setup_view_proj_matrix(lv_gltf_t * viewer, lv_gltf_view_desc_t * view_desc, lv_gltf_model_t * model,
1219 bool transmission_pass)
1220{
1221 LV_UNUSED(model);
1222 const lv_gltf_model_data_t * main_model = (lv_gltf_model_data_t *)lv_array_at(&viewer->models, 0);
1223 auto b_radius = lv_gltf_data_get_radius(main_model->model);
1224
1225 float radius = b_radius * LV_GLTF_DISTANCE_SCALE_FACTOR;
1226 radius *= view_desc->distance;
1227
1228 fastgltf::math::fvec3 rcam_dir = fastgltf::math::fvec3(0.0f, 0.0f, 1.0f);
1229
1230 /* Pitch turns around X, yaw turns around Y */
1231 fastgltf::math::fmat3x3 rotation1 =
1232 fastgltf::math::asMatrix(lv_gltf_math_euler_to_quaternion(fastgltf::math::radians(view_desc->pitch), 0.f, 0.f));
1233 fastgltf::math::fmat3x3 rotation2 =
1234 fastgltf::math::asMatrix(lv_gltf_math_euler_to_quaternion(0.f, fastgltf::math::radians(view_desc->yaw), 0.f));
1235
1236 rcam_dir = rotation1 * rcam_dir;
1237 rcam_dir = rotation2 * rcam_dir;
1238
1239 fastgltf::math::fvec3 ncam_dir = fastgltf::math::normalize(rcam_dir);
1240 fastgltf::math::fvec3 cam_target = fastgltf::math::fvec3(view_desc->focal_x, view_desc->focal_y, view_desc->focal_z);
1241 fastgltf::math::fvec3 cam_position = fastgltf::math::fvec3(cam_target[0] + (ncam_dir[0] * radius),
1242 cam_target[1] + (ncam_dir[1] * radius),
1243 cam_target[2] + (ncam_dir[2] * radius));
1244
1245 fastgltf::math::fmat4x4 view_mat =
1246 lv_gltf_math_look_at_rh(cam_position, cam_target, fastgltf::math::fvec3(0.0f, 1.0f, 0.0f));
1247
1248 // Create Projection Matrix
1249 fastgltf::math::fmat4x4 projection;
1250 float fov = view_desc->fov;
1251
1252 float znear = b_radius * 0.05f;
1253 float zfar = b_radius * std::max(4.0, 8.0 * view_desc->distance);
1254 auto width = view_desc->render_width;
1255 auto height = view_desc->render_height;
1256 // It's possible the transmission pass should simply use the regular passes aspect despite having different metrics itself. Testing both ways to see which has less distortion
1257 float aspect = (float)width / (float)height;
1258 if(transmission_pass) {
1259 width = 256;
1260 height = 256;
1261 }
1262
1263 if(fov <= 0.0f) {
1264 // Isometric view: create an orthographic projection
1265 float orthoSize = view_desc->distance * b_radius; // Adjust as needed
1266
1267 projection = fastgltf::math::fmat4x4(1.0f);
1268 projection[0][0] = -(orthoSize * aspect);
1269 projection[1][1] = (orthoSize);
1270 projection[2][2] = 2.f / (znear - zfar);
1271 projection[3][2] = (zfar + znear) / (znear - zfar);
1272
1273 }
1274 else {
1275 // Perspective view
1276 projection = fastgltf::math::fmat4x4(0.0f);
1277 LV_ASSERT(width != 0 && height != 0);
1278 projection[0][0] = 1.f / (aspect * tan(0.5f * fastgltf::math::radians(fov)));
1279 projection[1][1] = 1.f / (tan(0.5f * fastgltf::math::radians(fov)));
1280 projection[2][3] = -1;
1281
1282 // Finite projection matrix
1283 projection[2][2] = (zfar + znear) / (znear - zfar);
1284 projection[3][2] = (2.f * zfar * znear) / (znear - zfar);
1285 }
1286
1287 viewer->view_matrix = view_mat;
1288 viewer->projection_matrix = projection;
1289 viewer->view_projection_matrix = projection * view_mat;
1290 viewer->camera_pos = cam_position;
1291}
