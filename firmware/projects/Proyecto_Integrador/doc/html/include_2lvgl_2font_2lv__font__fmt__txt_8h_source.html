72 /*
73 According the specification there are 4 formats:
74 https://github.com/lvgl/lv_font_conv/blob/master/doc/font_spec.md
75
76 For simplicity introduce "relative code point":
77 rcp = codepoint - range_start
78
79 and a search function:
80 search a "value" in an "array" and returns the index of "value".
81
82 Format 0 tiny
83 unicode_list == NULL && glyph_id_ofs_list == NULL
84 glyph_id = glyph_id_start + rcp
85
86 Format 0 full
87 unicode_list == NULL && glyph_id_ofs_list != NULL
88 glyph_id = glyph_id_start + glyph_id_ofs_list[rcp]
89
90 Sparse tiny
91 unicode_list != NULL && glyph_id_ofs_list == NULL
92 glyph_id = glyph_id_start + search(unicode_list, rcp)
93
94 Sparse full
95 unicode_list != NULL && glyph_id_ofs_list != NULL
96 glyph_id = glyph_id_start + glyph_id_ofs_list[search(unicode_list, rcp)]
97 */
