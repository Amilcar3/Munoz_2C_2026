46def write_set_cmd(fout, expr, is_parent_scope):
47
48 fout.write(f'set({expr})\n')
49
50 # This makes the variable usable from the top level directory
51 if is_parent_scope == True:

54# Boolean options outside of the LV_USE_*/LV_BUILD_*
55# namespaces to be export to CMake
56EXTRA_SYMBOLS = [
57 "LV_CHECK_ARG_ASSERT_ON_FAIL",
58]

60def symbol_pattern(prefix: str):
61 """Match the `#define`s to export, `prefix` being "" or "CONFIG_".
62
63 Only the LV_USE_*/LV_BUILD_* namespaces and the symbols explicitly listed
64 in EXTRA_SYMBOLS are exported."""
65 assert prefix == "" or prefix == "CONFIG_"
66
67 namespaces = ["LV_USE", "LV_BUILD", "LV_[0-9A-Z_]+_USE"]
68 extras = ["%s\\b" % symbol for symbol in EXTRA_SYMBOLS]
69 alternatives = "|".join(prefix + name for name in namespaces + extras)
70
71 return "^#define +(%s)" % alternatives

73def generate_cmake_variables(path_input: str, path_output: str, kconfig: bool, debug: bool, is_parent_scope: bool):
74 fin = open(path_input)
75 fout = open(path_output, "w", newline='')
76
77 BARE_PATTERN = symbol_pattern("")
78
79 if kconfig:
80 # If we use Kconfig, we must check for CONFIG_LV_USE_* and
81 # CONFIG_LV_BUILD_* defines
82 CONFIG_PATTERN = symbol_pattern("CONFIG_")
83 CONFIG_PREFIX = ""
84 else:
85 # Otherwise check the LV_USE_* and LV_BUILD_* defines
86 CONFIG_PATTERN = BARE_PATTERN
87 CONFIG_PREFIX = "CONFIG_"
88
89
90 # Using the expanded lv_conf_internal, we don't have to deal with regexp,
91 # as all the #define will be aligned on the left with a single space before the value
92 for line in fin.read().splitlines():
93
94 # For the rest of the configs, simply add CONFIG_ and write the name of the define
95 # all LV_USE_* or LV_BUILD_* configs where the value is 0 or 1,
96 # as these are the ones that are needed in cmake
97 # To detect the configuration of LVGL to perform conditional compilation/linking
98 if re.search(f'{CONFIG_PATTERN}.* +[01] *$', line):
99
100 parts = line.split()
101 if len(parts) < 3:
102 continue
103
104 name = parts[1]
105 value = parts[2].strip()
106
107 write_set_cmd(fout, f'{CONFIG_PREFIX}{name} {value}', is_parent_scope)
108
109 # When using kconfig: catch bare LV_* defines with a literal 0/1 value
110 # that aren't backed by Kconfig (value is not a CONFIG_* reference)
111 elif kconfig and re.search(f'{BARE_PATTERN}.* +[01] *$', line):
112 parts = line.split()
113 if len(parts) < 3:
114 continue
115
116 name = parts[1]
117 value = parts[2].strip()
118 # Export with CONFIG_ prefix so it's consistent with kconfig-backed vars
119 write_set_cmd(fout, f'CONFIG_{name} {value}', is_parent_scope)
120
121 else:
122 # Useful for debugging expressions that are unhandled,
123 # if the script fails in 'unexpected ways'
124 if debug == True:
125 print(f"DBG: Skipping expression: '{line} - not handled'")
126
127
128if __name__ == '__main__':
129 args = get_args()
