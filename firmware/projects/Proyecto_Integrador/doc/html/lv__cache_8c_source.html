43lv_cache_t * lv_cache_create(const lv_cache_class_t * cache_class,
44 size_t node_size, size_t max_size,
45 lv_cache_ops_t ops)
46{
47 lv_cache_t * cache = cache_class->alloc_cb();
48 LV_ASSERT_MALLOC(cache);
49
50 cache->clz = cache_class;
51 cache->node_size = node_size;
52 cache->max_size = max_size;
53 cache->size = 0;
54 cache->ops = ops;
55
56 if(cache->clz->init_cb(cache) == false) {
57 LV_LOG_ERROR("Cache init failed");
58 lv_free(cache);
59 return NULL;
60 }
61
62 lv_mutex_init(&cache->lock);
63
64 return cache;
65}

95 if(entry != NULL) {

103void lv_cache_release(lv_cache_t * cache, lv_cache_entry_t * entry, void * user_data)
104{
105 LV_ASSERT_NULL(entry);
106
107 LV_PROFILER_CACHE_BEGIN;
108
109 lv_mutex_lock(&cache->lock);
110 lv_cache_entry_release_data(entry, user_data);
111
112 if(lv_cache_entry_get_ref(entry) == 0 && lv_cache_entry_is_invalid(entry)) {
113 cache->ops.free_cb(lv_cache_entry_get_data(entry), user_data);

115 }
116 lv_mutex_unlock(&cache->lock);
117
118 LV_PROFILER_CACHE_END;

144lv_cache_entry_t * lv_cache_acquire_or_create(lv_cache_t * cache, const void * key, void * user_data)
145{
146 LV_ASSERT_NULL(cache);
147 LV_ASSERT_NULL(key);
148
149 LV_PROFILER_CACHE_BEGIN;
150
151 lv_mutex_lock(&cache->lock);
152 lv_cache_entry_t * entry = NULL;
153
154 if(cache->size != 0) {
155 entry = cache->clz->get_cb(cache, key, user_data);
156 if(entry != NULL) {
157 lv_cache_entry_acquire_data(entry);
158 lv_mutex_unlock(&cache->lock);
159
160 LV_PROFILER_CACHE_END;
161 return entry;
162 }
163 }
164
165 if(cache->max_size == 0) {
166 lv_mutex_unlock(&cache->lock);
167
168 LV_PROFILER_CACHE_END;
169 return NULL;
170 }
171
172 entry = cache_add_internal_no_lock(cache, key, user_data);
173 if(entry == NULL) {
174 lv_mutex_unlock(&cache->lock);
175
176 LV_PROFILER_CACHE_END;
177 return NULL;
178 }
179 bool create_res = cache->ops.create_cb(lv_cache_entry_get_data(entry), user_data);
180 if(create_res == false) {
181 cache->clz->remove_cb(cache, entry, user_data);
182 cache->ops.free_cb(lv_cache_entry_get_data(entry), user_data);
183 lv_cache_entry_delete(entry);
184 entry = NULL;
185 }
186 else {
187 lv_cache_entry_acquire_data(entry);
188 }
189 lv_mutex_unlock(&cache->lock);
190
191 LV_PROFILER_CACHE_END;
192 return entry;
193}

306static void cache_drop_internal_no_lock(lv_cache_t * cache, const void * key, void * user_data)
307{
308 lv_cache_entry_t * entry = cache->clz->get_cb(cache, key, user_data);
309 if(entry == NULL) {
310 return;
311 }
312
313 if(lv_cache_entry_get_ref(entry) == 0) {
314 cache->clz->remove_cb(cache, entry, user_data);
315 cache->ops.free_cb(lv_cache_entry_get_data(entry), user_data);
316 lv_cache_entry_delete(entry);
317 }
318 else {
319 lv_cache_entry_set_flag(entry, LV_CACHE_ENTRY_FLAG_INVALID);
320 cache->clz->remove_cb(cache, entry, user_data);
321 }
322}

324static bool cache_evict_one_internal_no_lock(lv_cache_t * cache, void * user_data)
325{
326 lv_cache_entry_t * victim = cache->clz->get_victim_cb(cache, user_data);
327
328 if(victim == NULL) {
329 LV_LOG_ERROR("No victim found");
330 return false;
331 }
332
333 cache->clz->remove_cb(cache, victim, user_data);
334 cache->ops.free_cb(lv_cache_entry_get_data(victim), user_data);
335 lv_cache_entry_delete(victim);
336 return true;
337}

339static lv_cache_entry_t * cache_add_internal_no_lock(lv_cache_t * cache, const void * key, void * user_data)
340{
341 lv_cache_reserve_cond_res_t reserve_cond_res = cache->clz->reserve_cond_cb(cache, key, 0, user_data);
342 if(reserve_cond_res == LV_CACHE_RESERVE_COND_TOO_LARGE) {
343 LV_LOG_ERROR("data %p is too large that exceeds max size (%" LV_PRIu32 ")", key, cache->max_size);
344 return NULL;
345 }
346
347 for(; reserve_cond_res == LV_CACHE_RESERVE_COND_NEED_VICTIM;
348 reserve_cond_res = cache->clz->reserve_cond_cb(cache, key, 0, user_data))
349 if(cache_evict_one_internal_no_lock(cache, user_data) == false)
350 return NULL;
351
352 lv_cache_entry_t * entry = cache->clz->add_cb(cache, key, user_data);
353
354 return entry;
