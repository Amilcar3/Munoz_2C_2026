315void test_gltf_viewer_camera_selection(void)
316{
317 lv_obj_t * gltf = create_view();
318 TEST_ASSERT_NOT_NULL(lv_gltf_load_model_from_file(gltf, ASSET("cameras.gltf")));
319
320 /* The asset holds a perspective and an orthographic camera */
321 TEST_ASSERT_EQUAL(2, lv_gltf_get_camera_count(gltf));
322
323 /* Index 0 is the viewer's own camera, 1 and 2 are the ones from the file */
324 TEST_ASSERT_EQUAL(LV_GLTF_DEFAULT_CAMERA, lv_gltf_get_camera(gltf));
325
326 for(uint32_t i = 0; i <= 2; i++) {
327 lv_gltf_set_camera(gltf, i);
328 TEST_ASSERT_EQUAL(i, lv_gltf_get_camera(gltf));
329 }
330
331 /* An index beyond the last camera is clamped to the last one */
332 lv_gltf_set_camera(gltf, 3);
333 TEST_ASSERT_EQUAL(2, lv_gltf_get_camera(gltf));
334 lv_gltf_set_camera(gltf, 1000);
335 TEST_ASSERT_EQUAL(2, lv_gltf_get_camera(gltf));
336
337 lv_gltf_set_camera(gltf, LV_GLTF_DEFAULT_CAMERA);
338 TEST_ASSERT_EQUAL(LV_GLTF_DEFAULT_CAMERA, lv_gltf_get_camera(gltf));
339}
