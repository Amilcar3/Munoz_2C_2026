116void lv_text_get_size_attributes(lv_point_t * size_res, const char * text, const lv_font_t * font,
117 lv_text_attributes_t * attributes)
118{
119 uint32_t line_start = 0;
120 uint32_t new_line_start = 0;
121 uint16_t letter_height = 0;
122 size_res->x = 0;
123 size_res->y = 0;
124
125 LV_ASSERT_NULL(attributes);
126 LV_ASSERT_NULL(font);
127 LV_ASSERT_NULL(text);
128
129 letter_height = lv_font_get_line_height_internal(font);
130
131 if(attributes->text_flags & LV_TEXT_FLAG_EXPAND) {
132 attributes->max_width = LV_COORD_MAX;
133 }
134
135 /*Calc. the height and longest line*/
136 while(text[line_start] != '\0') {
137 new_line_start += lv_text_get_next_line(
138 &text[line_start], LV_TEXT_LEN_MAX, font, NULL, attributes);
139
140 if((unsigned long)size_res->y +
141 (unsigned long)letter_height + (unsigned long)attributes->line_space > LV_MAX_OF(int32_t)) {
142 LV_LOG_WARN("integer overflow while calculating text height");
143 return;
144 }
145 else {
146 size_res->y += letter_height;
147 size_res->y += attributes->line_space;
148 }
149
150 /*Calculate the longest line*/
151 int32_t act_line_length = lv_text_get_line_width(
152 &text[line_start], new_line_start - line_start, font, attributes);
153
154 size_res->x = LV_MAX(act_line_length, size_res->x);
155 line_start = new_line_start;
156 }
157
158 /*Make the text one line taller if the last character is '\n' or '\r'*/
159 if((line_start != 0) && (text[line_start - 1] == '\n' || text[line_start - 1] == '\r')) {
160 size_res->y += letter_height + attributes->line_space;
161 }
162
163 /*Correction with the last line space or set the height manually if the text is empty*/
164 if(size_res->y == 0)
165 size_res->y = letter_height;
166 else
167 size_res->y -= attributes->line_space;
168}

170bool lv_text_is_cmd(lv_text_cmd_state_t * state, uint32_t c)
171{
172 bool ret = false;
173
174 if(c == (uint32_t)LV_TXT_COLOR_CMD[0]) {
175 if(*state == LV_TEXT_CMD_STATE_WAIT) { /*Start char*/
176 *state = LV_TEXT_CMD_STATE_PAR;
177 ret = true;
178 }
179 /*Other start char in parameter is escaped cmd. char*/
180 else if(*state == LV_TEXT_CMD_STATE_WAIT) {
181 *state = LV_TEXT_CMD_STATE_WAIT;
182 }
183 /*Command end*/
184 else if(*state == LV_TEXT_CMD_STATE_IN) {
185 *state = LV_TEXT_CMD_STATE_WAIT;
186 ret = true;
187 }
188 }
189
190 /*Skip the color parameter and wait the space after it*/
191 if(*state == LV_TEXT_CMD_STATE_PAR) {
192 if(c == ' ') {
193 *state = LV_TEXT_CMD_STATE_IN; /*After the parameter the text is in the command*/
194 }
195 ret = true;
196 }
197
198 return ret;
199}

232static uint32_t lv_text_get_next_word(const char * txt, const lv_font_t * font,
233 int32_t letter_space, int32_t max_width,
234 lv_text_flag_t flag, uint32_t * word_w_ptr,
235 lv_text_cmd_state_t * cmd_state)
236{
237 if(txt == NULL || txt[0] == '\0') return 0;
238 if(font == NULL) return 0;
239
240 if(flag & LV_TEXT_FLAG_EXPAND) max_width = LV_COORD_MAX;
241
242 uint32_t i = 0, i_next = 0, i_next_next = 0; /*Iterating index into txt*/
243 uint32_t letter = 0; /*Letter at i*/
244 uint32_t letter_next = 0; /*Letter at i_next*/
245 int32_t letter_w;
246 int32_t cur_w = 0; /*Pixel Width of traversed string*/
247 uint32_t word_len = 0; /*Number of characters in the traversed word*/
248 uint32_t break_index = NO_BREAK_FOUND; /*only used for "long" words*/
249 uint32_t break_letter_count = 0; /*Number of characters up to the long word break point*/
250
251 letter = lv_text_encoded_next(txt, &i_next);
252 i_next_next = i_next;
253
254 /*Obtain the full word, regardless if it fits or not in max_width*/
255 while(txt[i] != '\0') {
256 letter_next = lv_text_encoded_next(txt, &i_next_next);
257 word_len++;
258
259 /*Handle the recolor command*/
260 if((flag & LV_TEXT_FLAG_RECOLOR) != 0) {
261 if(lv_text_is_cmd(cmd_state, letter)) {
262 i = i_next;
263 i_next = i_next_next;
264 letter = letter_next;
265 continue; /*Skip the letter if it is part of a command*/
266 }
267 }
268
269 letter_w = lv_font_get_glyph_width(font, letter, letter_next);
270 cur_w += letter_w;
271
272 if(letter_w > 0) {
273 cur_w += letter_space;
274 }
275
276 const bool is_break = (letter == '\n' || letter == '\r' || lv_text_is_break_char(letter));
277
278 /*Test if this character fits within max_width*/
279 if(!lv_text_is_hanging_space(letter) && break_index == NO_BREAK_FOUND &&
280 (cur_w - letter_space) > max_width) {
281 break_index = i;
282 break_letter_count = word_len - 1;
283 if(flag & LV_TEXT_FLAG_BREAK_ALL) {
284 break;
285 }
286 /*break_index is now pointing at the character that doesn't fit*/
287 }
288
289 /*Check for new line chars and breakchars*/
290 if(is_break) {
291 /*Update the output width on the first character if it fits.
292 *Must do this here in case first letter is a break character.*/
293 if(i == 0 && break_index == NO_BREAK_FOUND && word_w_ptr != NULL) *word_w_ptr = cur_w;
294 word_len--;
295 break;
296 }
297 else if(lv_text_is_a_word(letter_next) || lv_text_is_a_word(letter)) {
298 /*Found a word for single letter, usually true for CJK*/
299 *word_w_ptr = cur_w;
300 i = i_next;
301 break;
302 }
303
304 /*Update the output width*/
305 if(word_w_ptr != NULL && break_index == NO_BREAK_FOUND) *word_w_ptr = cur_w;
306
307 i = i_next;
308 i_next = i_next_next;
309 letter = letter_next;
310 }
311
312 /*Entire Word fits in the provided space*/
313 if(break_index == NO_BREAK_FOUND) {
314 if(word_len == 0 || (letter == '\r' && letter_next == '\n')) i = i_next;
315 return i;
316 }
317
318#if LV_TXT_LINE_BREAK_LONG_LEN > 0
319 /*Word doesn't fit in provided space, but isn't "long"*/
320 if(word_len < LV_TXT_LINE_BREAK_LONG_LEN) {
321 if(flag & LV_TEXT_FLAG_BREAK_ALL) return break_index;
322 if(word_w_ptr != NULL) *word_w_ptr = 0; /*Return no word*/
323 return 0;
324 }
325
326 /*Word is "long," but insufficient amounts can fit in provided space*/
327 if(break_letter_count < LV_TXT_LINE_BREAK_LONG_PRE_MIN_LEN) {
328 if(flag & LV_TEXT_FLAG_BREAK_ALL) return break_index;
329 if(word_w_ptr != NULL) *word_w_ptr = 0;
330 return 0;
331 }
332
333 /*Word is a "long", but letters may need to be better distributed*/
334 {
335 i = break_index;
336 int32_t n_move = LV_TXT_LINE_BREAK_LONG_POST_MIN_LEN - (word_len - break_letter_count);
337 /*Move pointer "i" backwards*/
338 for(; n_move > 0; n_move--) {
339 lv_text_encoded_prev(txt, &i);
344 }
345 }
346 return i;
347#else
348 if(flag & LV_TEXT_FLAG_BREAK_ALL) return break_index;
349 if(word_w_ptr != NULL) *word_w_ptr = 0; /*Return no word*/
350 (void) break_letter_count;
351 return 0;
352#endif
353}

359static bool is_space_run(const char * txt, uint32_t len)
360{
361 LV_ASSERT(txt);
362
363 if(len == 0) return false;
364
365 for(uint32_t i = 0; i < len; i++) {
366 if(txt[i] != ' ') return false;
367 }
368
369 return true;
370}

373 const lv_text_attributes_t * attributes)
374{
375 LV_ASSERT(txt);
376
377 /*length may reach past the end of the string, so find the real end first*/
378 uint32_t len = 0;
379 while(len < length && txt[len] != '\0') len++;
380
381 while(len > 0 && lv_text_is_hanging_space((uint8_t)txt[len - 1])) len--;
382
383 return lv_text_get_width(txt, len, font, attributes);
384}

386uint32_t lv_text_get_next_line(const char * txt, uint32_t len,
387 const lv_font_t * font, int32_t * used_width, lv_text_attributes_t * attributes)
388{
389
390 if(used_width) *used_width = 0;
391
392 if(txt == NULL) return 0;
393 if(txt[0] == '\0') return 0;
394 if(font == NULL) return 0;
395
396 int32_t line_w = 0;
397
398 /*If max_width doesn't matter simply find the new line character
399 *without thinking about word wrapping*/
400 if((attributes->text_flags & LV_TEXT_FLAG_EXPAND) ||
401 (attributes->text_flags & LV_TEXT_FLAG_FIT)) {
402
403 uint32_t i;
404 for(i = 0; i < len && txt[i] != '\n' && txt[i] != '\r' && txt[i] != '\0'; i++) {
405 /*Just find the new line chars or string ends by incrementing `i`*/
406 }
407 if(i < len && txt[i] != '\0') i++; /*To go beyond `\n`*/
408 if(used_width) *used_width = -1;
409 return i;
410 }
411
412 if(attributes->text_flags & LV_TEXT_FLAG_EXPAND) {
413 attributes->max_width = LV_COORD_MAX;
414 }
415 lv_text_cmd_state_t cmd_state = LV_TEXT_CMD_STATE_WAIT;
416
417 uint32_t i = 0; /*Iterating index into txt*/
418 uint32_t max_width = attributes->max_width;
419 bool explicit_new_line = false;
420 uint32_t hanging_w = 0; /*Width of the white space which may hang out of the line*/
421
422 while(i < len && txt[i] != '\0' && max_width > hanging_w) {
423 lv_text_flag_t word_flag = attributes->text_flags;
424
425 if(i == 0) word_flag |= LV_TEXT_FLAG_BREAK_ALL;
426
427 uint32_t word_w = 0;
428 uint32_t advance = lv_text_get_next_word(&txt[i], font, attributes->letter_space,
429 max_width - hanging_w, word_flag, &word_w, &cmd_state);
430
431 if(advance == 0) {
432 break;
433 }
434
435 /*A space is only charged to the line if a word follows it on the same line*/
436 if(is_space_run(&txt[i], advance)) {
437 hanging_w += word_w;
438 }
439 else {
440 max_width -= word_w + hanging_w;
441 line_w += word_w + hanging_w;
442 hanging_w = 0;
443 }
444
445 i += advance;
446
447 if(txt[0] == '\n' || txt[0] == '\r') {
448 explicit_new_line = true;
449 break;
450 }
451
452 if(txt[i] == '\n' || txt[i] == '\r') {
453 i++; /*Include the following newline in the current line*/
454 explicit_new_line = true;
455 break;
456 }
457 }
458
459 /*The text ran out, not the line: lv_spangroup continues it with the next span*/
460 if(i >= len || txt[i] == '\0') {
461 line_w += hanging_w;
462 }
463
464 /*Always step at least one to avoid infinite loops*/
465 if(i == 0) {
466 uint32_t letter = lv_text_encoded_next(txt, &i);
467 if(used_width != NULL) {
468 line_w = lv_font_get_glyph_width(font, letter, '\0');
469 }
470 }
471
472 if(used_width != NULL) {
473 *used_width = line_w;
474 }
475
476 /*Skip leading spaces of the next line only for automatic word wrapping*/
477 if(!explicit_new_line) {
478 while(i < len && txt[i] == ' ') {
479 i++;
480 }
481 }
482
483 return i;
484}

486int32_t lv_text_get_width(const char * txt, uint32_t length, const lv_font_t * font,
487 const lv_text_attributes_t * attributes)
488{
489 if(txt == NULL) return 0;
490 if(font == NULL) return 0;
491 if(txt[0] == '\0') return 0;
492
493 uint32_t i = 0;
494 int32_t width = 0;
495 lv_text_cmd_state_t cmd_state = LV_TEXT_CMD_STATE_WAIT;
496
497 if(length != 0) {
498 while(txt[i] != '\0' && i < length) {
499
500 uint32_t letter;
501 uint32_t letter_next;
502
503 lv_text_encoded_letter_next_2(txt, &letter, &letter_next, &i);
504
505 if((attributes->text_flags & LV_TEXT_FLAG_RECOLOR) != 0) {
506 if(lv_text_is_cmd(&cmd_state, letter) != false) {
507 continue;
508 }
509 }
510
511 int32_t char_width = lv_font_get_glyph_width(font, letter, letter_next);
512 if(char_width > 0) {
513 width += char_width;
514 width += attributes->letter_space;
515 }
516 }
517
518 if(width > 0) {
519 width -= attributes->letter_space; /*Trim the last letter space. Important if the text is center
520 aligned*/
521 }
522 }
523
524 return width;
525}

527void lv_text_ins(char * txt_buf, uint32_t pos, const char * ins_txt)
528{
529 if(txt_buf == NULL || ins_txt == NULL) return;
530
531 size_t old_len = lv_strlen(txt_buf);
532 size_t ins_len = lv_strlen(ins_txt);
533 if(ins_len == 0) return;
534
535 size_t new_len = ins_len + old_len;
536 pos = lv_text_encoded_get_byte_id(txt_buf, pos); /*Convert to byte index instead of letter index*/
537
538 /*Copy the second part into the end to make place to text to insert*/
539 size_t i;
540 for(i = new_len; i >= pos + ins_len; i--) {
541 txt_buf[i] = txt_buf[i - ins_len];
542 }
543
544 /*Copy the text into the new space*/
545 lv_memcpy(txt_buf + pos, ins_txt, ins_len);
546}

548void lv_text_cut(char * txt, uint32_t pos, uint32_t len)
549{
550 if(txt == NULL) return;
551
552 size_t old_len = lv_strlen(txt);
553
554 pos = lv_text_encoded_get_byte_id(txt, pos); /*Convert to byte index instead of letter index*/
555 if(pos >= old_len) return;
556
557 len = lv_text_encoded_get_byte_id(&txt[pos], len);
558 if(len > old_len - pos) len = old_len - pos; /*Don't cut more than what's left*/
559
560 /*Copy the second part into the end to make place to text to insert*/
561 uint32_t i;
562 for(i = pos; i <= old_len - len; i++) {
563 txt[i] = txt[i + len];
564 }
565}

567char * lv_text_set_text_vfmt(const char * fmt, va_list ap)
568{
569 /*Allocate space for the new text by using trick from C99 standard section 7.19.6.12*/
570 va_list ap_copy;
571 va_copy(ap_copy, ap);
572 uint32_t len = lv_vsnprintf(NULL, 0, fmt, ap_copy);
573 va_end(ap_copy);
574
575 char * text = 0;
576#if LV_USE_ARABIC_PERSIAN_CHARS
577 /*Put together the text according to the format string*/
578 char * raw_txt = lv_malloc(len + 1);
579 LV_ASSERT_MALLOC(raw_txt);
580 if(raw_txt == NULL) {
581 return NULL;
582 }
583
584 lv_vsnprintf(raw_txt, len + 1, fmt, ap);
585
586 /*Get the size of the Arabic text and process it*/
587 size_t len_ap = lv_text_ap_strlen(raw_txt);
588 text = lv_malloc(len_ap + 1);
589 LV_ASSERT_MALLOC(text);
590 if(text == NULL) {
591 return NULL;
592 }
593 lv_text_ap_proc(raw_txt, text);
594
595 lv_free(raw_txt);
596#else
597 text = lv_malloc(len + 1);
598 LV_ASSERT_MALLOC(text);
599 if(text == NULL) {
600 return NULL;
601 }
602
603 lv_vsnprintf(text, len + 1, fmt, ap);
604#endif
605
606 return text;
607}

655static uint32_t lv_text_unicode_to_utf8(uint32_t letter_uni)
656{
657 if(letter_uni < 128) return letter_uni;
658 uint8_t bytes[4];
659
660 if(letter_uni < 0x0800) {
661 bytes[0] = ((letter_uni >> 6) & 0x1F) | 0xC0;
662 bytes[1] = ((letter_uni >> 0) & 0x3F) | 0x80;
663 bytes[2] = 0;
664 bytes[3] = 0;
665 }
666 else if(letter_uni < 0x010000) {
667 bytes[0] = ((letter_uni >> 12) & 0x0F) | 0xE0;
668 bytes[1] = ((letter_uni >> 6) & 0x3F) | 0x80;
669 bytes[2] = ((letter_uni >> 0) & 0x3F) | 0x80;
670 bytes[3] = 0;
671 }
672 else if(letter_uni < 0x110000) {
673 bytes[0] = ((letter_uni >> 18) & 0x07) | 0xF0;
674 bytes[1] = ((letter_uni >> 12) & 0x3F) | 0x80;
675 bytes[2] = ((letter_uni >> 6) & 0x3F) | 0x80;
676 bytes[3] = ((letter_uni >> 0) & 0x3F) | 0x80;
677 }
678 else {
679 return 0;
680 }
681
682 uint32_t * res_p = (uint32_t *)bytes;

691static uint32_t lv_text_utf8_conv_wc(uint32_t c)
692{
693#if LV_BIG_ENDIAN_SYSTEM == 0
694 /*Swap the bytes (UTF-8 is big endian, but the MCUs are little endian)*/
695 if((c & 0x80) != 0) {
696 uint32_t swapped;
697 uint8_t c8[4];
698 lv_memcpy(c8, &c, 4);
699 swapped = (c8[0] << 24) + (c8[1] << 16) + (c8[2] << 8) + (c8[3]);
700 uint8_t i;
701 for(i = 0; i < 4; i++) {
702 if((swapped & 0xFF) == 0)
703 swapped = (swapped >> 8); /*Ignore leading zeros (they were in the end originally)*/
704 }
705 c = swapped;
706 }
707#endif
708 return c;

719static uint32_t lv_text_utf8_next(const char * txt, uint32_t * i)
720{
728
729 uint32_t result = 0;
730
731 /*Dummy 'i' pointer is required*/
732 uint32_t i_tmp = 0;
733 if(i == NULL) i = &i_tmp;
734
735 /* Ensure the string is not null */
736 if(txt == NULL || txt[*i] == '\0') {
737 return result;
738 }
739
740 /*Normal ASCII*/
741 if(LV_IS_ASCII(txt[*i])) {
742 result = txt[*i];
743 (*i)++;
744 }
745 /*Real UTF-8 decode*/
746 else {
747 /*2 bytes UTF-8 code*/
748 if(LV_IS_2BYTES_UTF8_CODE(txt[*i])) {
749 result = (uint32_t)(txt[*i] & 0x1F) << 6;
750 (*i)++;
751 if(LV_IS_INVALID_UTF8_CODE(txt[*i])) return 0;
752 result += (txt[*i] & 0x3F);
753 (*i)++;
754 }
755 /*3 bytes UTF-8 code*/
756 else if(LV_IS_3BYTES_UTF8_CODE(txt[*i])) {
757 result = (uint32_t)(txt[*i] & 0x0F) << 12;
758 (*i)++;
759
760 if(LV_IS_INVALID_UTF8_CODE(txt[*i])) return 0;
761 result += (uint32_t)(txt[*i] & 0x3F) << 6;
762 (*i)++;
763
764 if(LV_IS_INVALID_UTF8_CODE(txt[*i])) return 0;
765 result += (txt[*i] & 0x3F);
766 (*i)++;
767 }
768 /*4 bytes UTF-8 code*/
769 else if(LV_IS_4BYTES_UTF8_CODE(txt[*i])) {
770 result = (uint32_t)(txt[*i] & 0x07) << 18;
771 (*i)++;
772
773 if(LV_IS_INVALID_UTF8_CODE(txt[*i])) return 0;
774 result += (uint32_t)(txt[*i] & 0x3F) << 12;
775 (*i)++;
776
777 if(LV_IS_INVALID_UTF8_CODE(txt[*i])) return 0;
778 result += (uint32_t)(txt[*i] & 0x3F) << 6;
779 (*i)++;
780
781 if(LV_IS_INVALID_UTF8_CODE(txt[*i])) return 0;
782 result += txt[*i] & 0x3F;
783 (*i)++;
784 }
785 else {
786 (*i)++; /*Not UTF-8 char. Go the next.*/
787 }
788 }
789 return result;
790}

799static uint32_t lv_text_utf8_prev(const char * txt, uint32_t * i)
800{
801 uint8_t c_size;
802 uint8_t cnt = 0;
803
804 /*Try to find a !0 long UTF-8 char by stepping one character back*/
805 (*i)--;
806 do {
807 if(cnt >= 4) return 0; /*No UTF-8 char found before the initial*/
808
809 c_size = lv_text_encoded_size(&txt[*i]);
810 if(c_size == 0) {
811 if(*i != 0)
812 (*i)--;
813 else
814 return 0;
815 }
816 cnt++;
817 } while(c_size == 0);
818
819 uint32_t i_tmp = *i;
820 uint32_t letter = lv_text_encoded_next(txt, &i_tmp); /*Character found, get it*/
821
822 return letter;
823}

837 uint8_t c_size = lv_text_encoded_size(&txt[byte_cnt]);
