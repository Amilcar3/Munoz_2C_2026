7static const char * heading_to_cardinal(int32_t heading)
8{
9 /* Normalize heading to range [0, 360) */
10 while(heading < 0) heading += 360;
11 while(heading >= 360) heading -= 360;
12
13 if(heading < 23) return "N";
14 else if(heading < 68) return "NE";
15 else if(heading < 113) return "E";
16 else if(heading < 158) return "SE";
17 else if(heading < 203) return "S";
18 else if(heading < 248) return "SW";
19 else if(heading < 293) return "W";
20 else if(heading < 338) return "NW";
21
22 return "N";
23}
