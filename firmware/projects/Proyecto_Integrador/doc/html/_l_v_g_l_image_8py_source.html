207def bit_extend(value, bpp):
208 """
209 Extend value from bpp to 8 bit with interpolation to reduce rounding error.
210 """
211
212 if value == 0:
213 return 0
214
215 res = value
216 bpp_now = bpp
217 while bpp_now < 8:
218 res |= value << (8 - bpp_now)
219 bpp_now += bpp
220
221 return res

224def unpack_colors(data: bytes, cf: ColorFormat, w) -> List:
225 """
226 Unpack lvgl 1/2/4/8/16/32 bpp color to png color: alpha map, grey scale,
227 or R,G,B,(A) map
228 """
229 ret = []
230 bpp = cf.bpp
231 if bpp == 8:
232 ret = data
233 elif bpp == 4:
234 if cf == ColorFormat.A4:
235 values = [x * 17 for x in range(16)]
236 else:
237 values = [x for x in range(16)]
238
239 for p in data:
240 for i in range(2):
241 ret.append(values[(p >> (4 - i * 4)) & 0x0f])
242 if len(ret) % w == 0:
243 break
244
245 elif bpp == 2:
246 if cf == ColorFormat.A2:
247 values = [x * 85 for x in range(4)]
248 else: # must be ColorFormat.I2
249 values = [x for x in range(4)]
250 for p in data:
251 for i in range(4):
252 ret.append(values[(p >> (6 - i * 2)) & 0x03])
253 if len(ret) % w == 0:
254 break
255 elif bpp == 1:
256 if cf == ColorFormat.A1:
257 values = [0, 255]
258 else:
259 values = [0, 1]
260 for p in data:
261 for i in range(8):
262 ret.append(values[(p >> (7 - i)) & 0x01])
263 if len(ret) % w == 0:
264 break
265 elif bpp == 16:
266 if cf == ColorFormat.RGB565:
267 # This is RGB565
268 pixels = [(data[2 * i + 1] << 8) | data[2 * i]
269 for i in range(len(data) // 2)]
270 for p in pixels:
271 ret.append(bit_extend((p >> 11) & 0x1f, 5)) # R
272 ret.append(bit_extend((p >> 5) & 0x3f, 6)) # G
273 ret.append(bit_extend((p >> 0) & 0x1f, 5)) # B
274 elif cf == ColorFormat.RGB565_SWAPPED:
275 # This is RGB565_SWAPPED
276 pixels = [(data[2 * i] << 8) | data[2 * i + 1]
277 for i in range(len(data) // 2)]
278 for p in pixels:
279 ret.append(bit_extend((p >> 11) & 0x1f, 5)) # R
280 ret.append(bit_extend((p >> 5) & 0x3f, 6)) # G
281 ret.append(bit_extend((p >> 0) & 0x1f, 5)) # B
282 elif cf == ColorFormat.AL88:
283 # AL88: low 8bit = Luminance, high 8bit = Alpha
284 L = data[0::2] # low byte: luminance
285 A = data[1::2] # high byte: alpha
286 for luma, alpha in zip(L, A):
287 ret.append(luma) # L
288 ret.append(alpha) # A
289
290 elif bpp == 24:
291 if cf == ColorFormat.RGB888:
292 B = data[0::3]
293 G = data[1::3]
294 R = data[2::3]
295 for r, g, b in zip(R, G, B):
296 ret += [r, g, b]
297 elif cf == ColorFormat.RGB565A8:
298 alpha_size = len(data) // 3
299 pixel_alpha = data[-alpha_size:]
300 pixel_data = data[:-alpha_size]
301 pixels = [(pixel_data[2 * i + 1] << 8) | pixel_data[2 * i]
302 for i in range(len(pixel_data) // 2)]
303
304 for a, p in zip(pixel_alpha, pixels):
305 ret.append(bit_extend((p >> 11) & 0x1f, 5)) # R
306 ret.append(bit_extend((p >> 5) & 0x3f, 6)) # G
307 ret.append(bit_extend((p >> 0) & 0x1f, 5)) # B
308 ret.append(a)
309 elif cf == ColorFormat.ARGB8565:
310 L = data[0::3]
311 H = data[1::3]
312 A = data[2::3]
313
314 for h, l, a in zip(H, L, A):
315 p = (h << 8) | (l)
316 ret.append(bit_extend((p >> 11) & 0x1f, 5)) # R
317 ret.append(bit_extend((p >> 5) & 0x3f, 6)) # G
318 ret.append(bit_extend((p >> 0) & 0x1f, 5)) # B
319 ret.append(a) # A
320
321 elif bpp == 32:
322 B = data[0::4]
323 G = data[1::4]
324 R = data[2::4]
325 A = data[3::4]
326 if cf == ColorFormat.ARGB8888_PREMULTIPLIED:
327 for r, g, b, a in zip(R, G, B, A):
328 r = (r * a // 255)
329 g = (g * a // 255)
330 b = (b * a // 255)
331 ret += [r, g, b, a]
332 else:
333 for r, g, b, a in zip(R, G, B, A):
334 ret += [r, g, b, a]
335 else:
336 assert 0
337
338 return ret

502
503 def __init__(self,
504 cf: ColorFormat,
505 method: CompressMethod,
506 raw_data: bytes = b''):
507 self.blk_size = (cf.bpp + 7) // 8
508 self.compress = method
509 self.raw_data = raw_data
510 self.raw_data_len = len(raw_data)
511 self.compressed = self._compress(raw_data)
512
513 def _compress(self, raw_data: bytes) -> bytearray:
514 if self.compress == CompressMethod.NONE:
515 return raw_data
516
517 if self.compress == CompressMethod.RLE:
518 # RLE compression performs on pixel unit, pad data to pixel unit
519 pad = b'\x00' * 0
520 if self.raw_data_len % self.blk_size:
521 pad = b'\x00' * (self.blk_size - self.raw_data_len % self.blk_size)
522 compressed = RLEImage().rle_compress(raw_data + pad, self.blk_size)
523 elif self.compress == CompressMethod.LZ4:
524 compressed = lz4.block.compress(raw_data, store_size=False)
525 else:
526 raise ParameterError(f"Invalid compress method: {self.compress}")
527
528 self.compressed_len = len(compressed)
529
530 bin = bytearray()
531 bin += uint32_t(self.compress.value)
532 bin += uint32_t(self.compressed_len)
533 bin += uint32_t(self.raw_data_len)
534 bin += compressed
535 return bin

624 def premultiply(self):
625 """
626 Pre-multiply image RGB data with alpha, set corresponding image header flags
627 """
628 if self.premultiplied:
629 raise ParameterError("Image already pre-multiplied")
630
631 if not self.cf.has_alpha:
632 raise ParameterError(f"Image has no alpha channel: {self.cf.name}")
633
634 if self.cf.is_indexed:
635
636 def multiply(b, g, r, a):
637 # The precision is reduced, the correct way would be to divide by 255,
638 # but this is consistent with the premultiply function in the code.
639 r, g, b = (r * a) >> 8, (g * a) >> 8, (b * a) >> 8
640 return uint8_t(b) + uint8_t(g) + uint8_t(r) + uint8_t(a)
641
642 # process the palette only.
643 palette_size = self.cf.ncolors * 4
644 palette = self.data[:palette_size]
645 palette = [
646 multiply(palette[i], palette[i + 1], palette[i + 2],
647 palette[i + 3]) for i in range(0, len(palette), 4)
648 ]
649 palette = b''.join(palette)
650 self.data = palette + self.data[palette_size:]
651 elif self.cf is ColorFormat.ARGB8888:
652
653 def multiply(b, g, r, a):
654 r, g, b = (r * a) >> 8, (g * a) >> 8, (b * a) >> 8
655 return uint32_t((a << 24) | (r << 16) | (g << 8) | (b << 0))
656
657 line_width = self.w * 4
658 for h in range(self.h):
659 offset = h * self.stride
660 map = self.data[offset:offset + self.stride]
661
662 processed = b''.join([
663 multiply(map[i], map[i + 1], map[i + 2], map[i + 3])
664 for i in range(0, line_width, 4)
665 ])
666 self.data[offset:offset + line_width] = processed
667 elif self.cf is ColorFormat.RGB565A8:
668
669 def multiply(data, a):
670 r = (data >> 11) & 0x1f
671 g = (data >> 5) & 0x3f
672 b = (data >> 0) & 0x1f
673
674 r, g, b = (r * a) // 255, (g * a) // 255, (b * a) // 255
675 return uint16_t((r << 11) | (g << 5) | (b << 0))
676
677 line_width = self.w * 2
678 for h in range(self.h):
679 # alpha map offset for this line
680 offset = self.h * self.stride + h * (self.stride // 2)
681 a = self.data[offset:offset + self.stride // 2]
682
683 # RGB map offset
684 offset = h * self.stride
685 rgb = self.data[offset:offset + self.stride]
686
687 processed = b''.join([
688 multiply((rgb[i + 1] << 8) | rgb[i], a[i // 2])
689 for i in range(0, line_width, 2)
690 ])
691 self.data[offset:offset + line_width] = processed
692 elif self.cf is ColorFormat.ARGB8565:

696 g = (data >> 5) & 0x3f
697 b = (data >> 0) & 0x1f
698
699 r, g, b = (r * a) // 255, (g * a) // 255, (b * a) // 255
700 return uint24_t((a << 16) | (r << 11) | (g << 5) | (b << 0))

703 for h in range(self.h):
704 offset = h * self.stride
705 map = self.data[offset:offset + self.stride]
706
707 processed = b''.join([
708 multiply((map[i + 1] << 8) | map[i], map[i + 2])
709 for i in range(0, line_width, 3)
710 ])

938 def _png_to_indexed(self, cf: ColorFormat, filename: str):
939 # convert to palette mode
940 auto_cf = cf is None
941
942 # read the image data to get the metadata
943 reader = png.Reader(filename=filename)
944 w, h, rows, metadata = reader.read()
945
946 # to preserve original palette data only convert the image if needed. For this
947 # check if image has a palette and the requested palette size equals the existing one
948 if not 'palette' in metadata or not auto_cf and len(metadata['palette']) != 2 ** cf.bpp:
949 # reread and convert file
950 reader = png.Reader(
951 bytes=PngQuant(256 if auto_cf else cf.ncolors).convert(filename))
952 w, h, rows, _ = reader.read()
953
954 palette = reader.palette(alpha="force") # always return alpha
955
956 palette_len = len(palette)
957 if auto_cf:
958 if palette_len <= 2:
959 cf = ColorFormat.I1
960 elif palette_len <= 4:
961 cf = ColorFormat.I2
962 elif palette_len <= 16:
963 cf = ColorFormat.I4
964 else:
965 cf = ColorFormat.I8
966
967 if palette_len != cf.ncolors:
968 if not auto_cf:

986 rawdata += e

988 for e in png.pack_rows(rows, cf.bpp):
989 rawdata += e

1024 def _png_to_al88(self, cf: ColorFormat, filename: str):
1025 reader = png.Reader(str(filename))
1026 w, h, rows, info = reader.asRGBA8()
1027 if not info['alpha']:
1028 raise FormatError(f"{filename} has no alpha channel")

1032 R = row[0::4]
1033 G = row[1::4]
1034 B = row[2::4]
1035 A = row[3::4]
1036 for r, g, b, a in zip(R, G, B, A):
1037 # Calculate luminance using ITU-R BT.709 coefficients
1038 r_linear = self.sRGB_to_linear(r / 255.0)
1039 g_linear = self.sRGB_to_linear(g / 255.0)
1040 b_linear = self.sRGB_to_linear(b / 255.0)

1042 luma_byte = int(self.linear_to_sRGB(luma) * 255)
1043 # AL88: low byte = luminance, high byte = alpha
1044 rawdata += uint8_t(luma_byte) # L
1045 rawdata += uint8_t(a) # A

1047 self.set_data(ColorFormat.AL88, w, h, rawdata)

1237 def rle_compress(self, data: bytearray, blksize: int, threshold=16):
1238 index = 0
1239 data_len = len(data)
1240 compressed_data = []
1241 memview = memoryview(data)
1242 while index < data_len:
1243 repeat_cnt = self.get_repeat_count(memview[index:], blksize)
1244 if repeat_cnt == 0:
1245 # done
1246 break
1247 elif repeat_cnt < threshold:
1248 nonrepeat_cnt = self.get_nonrepeat_count(
1249 memview[index:], blksize, threshold)
1250 ctrl_byte = uint8_t(nonrepeat_cnt | 0x80)
1251 compressed_data.append(ctrl_byte)
1252 compressed_data.append(memview[index:index +
1253 nonrepeat_cnt * blksize])
1254 index += nonrepeat_cnt * blksize
1255 else:
1256 ctrl_byte = uint8_t(repeat_cnt)
1257 compressed_data.append(ctrl_byte)
1258 compressed_data.append(memview[index:index + blksize])
1259 index += repeat_cnt * blksize
1260
1261 return b"".join(compressed_data)

1263 def get_repeat_count(self, data: bytearray, blksize: int):
1264 if len(data) < blksize:
1265 return 0
1266
1267 start = data[:blksize]
1268 index = 0
1269 repeat_cnt = 0
1270 value = 0
1271
1272 while index < len(data):
1273 value = data[index:index + blksize]
1274
1275 if value == start:
1276 repeat_cnt += 1
1277 if repeat_cnt == 127: # limit max repeat count to max value of signed char.
1278 break
1279 else:
1280 break
1281 index += blksize
1282
1283 return repeat_cnt
1284
1285 def get_nonrepeat_count(self, data: bytearray, blksize: int, threshold):
1286 if len(data) < blksize:
1287 return 0
1288
1289 pre_value = data[:blksize]
1290
1291 index = 0
1292 nonrepeat_count = 0
1293
1294 repeat_cnt = 0
1295 while True:
1296 value = data[index:index + blksize]
1297 if value == pre_value:
1298 repeat_cnt += 1
1299 if repeat_cnt > threshold:
1300 # repeat found.
1301 break
1302 else:
1303 pre_value = value
1304 nonrepeat_count += 1 + repeat_cnt
1305 repeat_cnt = 0
1306 if nonrepeat_count >= 127: # limit max repeat count to max value of signed char.
1307 nonrepeat_count = 127
1308 break
1309
1310 index += blksize # move to next position
1311 if index >= len(data): # data end
1312 nonrepeat_count += repeat_cnt
1313 break
1314
1315 return nonrepeat_count
