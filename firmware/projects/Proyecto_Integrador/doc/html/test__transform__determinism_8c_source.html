48static void invalidate_middle_and_compare(lv_obj_t * img)
49{
50 lv_refr_now(NULL);
51
52 lv_display_t * disp = lv_display_get_default();
53 lv_draw_buf_t * front = lv_display_get_buf_active(disp);
54 uint32_t size = front->header.stride * front->header.h;
55 uint8_t * full_render = lv_malloc(size);
56 TEST_ASSERT_NOT_NULL(full_render);
57 lv_memcpy(full_render, front->data, size);
58
59 /*Invalidate only the middle part of the image and render again*/
60 lv_area_t area;
61 lv_obj_get_coords(img, &area);
62 int32_t w = lv_area_get_width(&area);
63 int32_t h = lv_area_get_height(&area);
64 area.x1 += w / 4;
65 area.x2 -= w / 4;
66 area.y1 += h / 4;
67 area.y2 -= h / 4;
68 lv_inv_area(disp, &area);
69 lv_refr_now(NULL);
70
71 /*Compare and release the buffer before asserting, else a failing assert
72 *aborts the test and leaks `full_render`*/
73 uint32_t diff_at = 0;
74 while(diff_at < size && full_render[diff_at] == ((const uint8_t *)front->data)[diff_at]) diff_at++;
75 lv_free(full_render);
76
77 if(diff_at < size) {
78 char msg[128];
79 lv_snprintf(msg, sizeof(msg), "The partial and the full render differ at byte %" LV_PRIu32 " of %" LV_PRIu32,
80 diff_at, size);
81 TEST_FAIL_MESSAGE(msg);
82 }
83}
