94lv_result_t lv_image_decoder_get_info(const void * src, lv_image_header_t * header)
95{
96 LV_CHECK_ARG(src != NULL, return LV_RESULT_INVALID);
97 LV_CHECK_ARG(header != NULL, return LV_RESULT_INVALID);
98
99 LV_PROFILER_DECODER_BEGIN;
100 lv_image_decoder_dsc_t dsc;
101 lv_memzero(&dsc, sizeof(lv_image_decoder_dsc_t));
102 dsc.src = src;
103 dsc.src_type = lv_image_src_get_type(src);

106 lv_image_decoder_t * decoder = image_decoder_get_info(&dsc, header);

108
109 LV_PROFILER_DECODER_END;
110 return decoder ? LV_RESULT_OK : LV_RESULT_INVALID;

113lv_result_t lv_image_decoder_open(lv_image_decoder_dsc_t * dsc, const void * src, const lv_image_decoder_args_t * args)
114{
115 LV_CHECK_ARG(dsc != NULL, return LV_RESULT_INVALID);
116
117 LV_PROFILER_DECODER_BEGIN;
118 lv_memzero(dsc, sizeof(lv_image_decoder_dsc_t));
119
120 if(src == NULL) {
121 LV_PROFILER_DECODER_END;
122 return LV_RESULT_INVALID;
123 }
124 dsc->src = src;
125 dsc->src_type = lv_image_src_get_type(src);
126
127 lv_mutex_lock(img_decoder_open_lock_p);
128
129 if(lv_image_cache_is_enabled()) {
130 dsc->cache = img_cache_p;
131 /*Try cache first, unless we are told to ignore cache.*/
132 if(!(args && args->no_cache)) {
133 /*
134 * Check the cache first
135 * If the image is found in the cache, just return it.*/
136 if(try_cache(dsc) == LV_RESULT_OK) {
137 lv_mutex_unlock(img_decoder_open_lock_p);
138 LV_PROFILER_DECODER_END;
139 return LV_RESULT_OK;
140 }
141 }
142 }
143
144 /*Find the decoder that can open the image source, and get the header info in the same time.*/
145 dsc->decoder = image_decoder_get_info(dsc, &dsc->header);
146 if(dsc->decoder == NULL) {
147 lv_mutex_unlock(img_decoder_open_lock_p);
148 LV_PROFILER_DECODER_END;
149 return LV_RESULT_INVALID;
150 }
151
152 /*Make a copy of args*/
153 dsc->args = args ? *args : (lv_image_decoder_args_t) {
154 .stride_align = LV_DRAW_BUF_STRIDE_ALIGN != 1,
155 .premultiply = false,
156 .no_cache = false,
157 .use_indexed = false,
158 .flush_cache = false,
159 };
160
161 /*
162 * We assume that if a decoder can get the info, it can open the image.
163 * If decoder open failed, free the source and return error.
164 * If decoder open succeed, add the image to cache if enabled.
165 * */
166 LV_ASSERT(dsc->decoder->open_cb != NULL);
167 LV_PROFILER_DECODER_BEGIN_TAG(dsc->decoder->name);
168 lv_result_t res = dsc->decoder->open_cb(dsc->decoder, dsc);
169 LV_PROFILER_DECODER_END_TAG(dsc->decoder->name);
170
171 if(res == LV_RESULT_OK && dsc->decoded != NULL) {
172 LV_ASSERT_MSG(dsc->decoded->unaligned_data && dsc->decoded->handlers, "Invalid draw buffer");
173
174 /* Flush the D-Cache if enabled and the image was successfully opened */
175 if(dsc->args.flush_cache) {
176 lv_draw_buf_flush_cache(dsc->decoded, NULL);
177 LV_LOG_INFO("Flushed D-cache: src %p (%s) (W%d x H%d, data: %p cf: %d)",
178 src,
179 dsc->src_type == LV_IMAGE_SRC_FILE ? (const char *)src : "c-array",
180 dsc->decoded->header.w,
181 dsc->decoded->header.h,
182 (void *)dsc->decoded->data,
183 dsc->decoded->header.cf);
184 }
185 }
186
187 lv_mutex_unlock(img_decoder_open_lock_p);
188 LV_PROFILER_DECODER_END;
189 return res;
190}

211void lv_image_decoder_close(lv_image_decoder_dsc_t * dsc)
212{
213 if(dsc == NULL) return;
214
215 LV_PROFILER_DECODER_BEGIN;
216 if(!dsc->decoder) {
217 LV_PROFILER_DECODER_END;
218 return;
219 }
220
221 lv_mutex_lock(img_decoder_open_lock_p);
222
223 if(dsc->decoder->close_cb) {
224 LV_PROFILER_DECODER_BEGIN_TAG(dsc->decoder->name);
225 dsc->decoder->close_cb(dsc->decoder, dsc);
226 LV_PROFILER_DECODER_END_TAG(dsc->decoder->name);
227 }
228
229 if(lv_image_cache_is_enabled() && dsc->cache && dsc->cache_entry) {
230 /*Decoded data is in cache, release it from cache's callback*/
231 lv_cache_release(dsc->cache, dsc->cache_entry, NULL);
232 }
233
234 lv_mutex_unlock(img_decoder_open_lock_p);
235 LV_PROFILER_DECODER_END;
236}

298lv_cache_entry_t * lv_image_decoder_add_to_cache(lv_image_decoder_t * decoder,
299 lv_image_cache_data_t * search_key,
300 const lv_draw_buf_t * decoded, void * user_data)
301{
302 LV_PROFILER_DECODER_BEGIN;
303 lv_cache_entry_t * cache_entry = lv_cache_add(img_cache_p, search_key, NULL);
304 if(cache_entry == NULL) {
305 LV_PROFILER_DECODER_END;
306 return NULL;
307 }
308
309 lv_image_cache_data_t * cached_data;
310 cached_data = lv_cache_entry_get_data(cache_entry);
311
312 /*Set the cache entry to decoder data*/
313 cached_data->decoded = decoded;
314 if(cached_data->src_type == LV_IMAGE_SRC_FILE) {
315 cached_data->src = lv_strdup(cached_data->src);
316 }
317 cached_data->user_data = user_data; /*Need to free data on cache invalidate instead of decoder_close*/
318 cached_data->decoder = decoder;
319
320 LV_PROFILER_DECODER_END;
321 return cache_entry;
322}

324lv_draw_buf_t * lv_image_decoder_post_process(lv_image_decoder_dsc_t * dsc, lv_draw_buf_t * decoded)
325{
326 LV_CHECK_ARG(dsc != NULL, return decoded);
327
328 LV_PROFILER_DECODER_BEGIN;
329 if(decoded == NULL) {
330 LV_PROFILER_DECODER_END;
331 return NULL; /*No need to adjust*/
332 }
333
334 lv_image_decoder_args_t * args = &dsc->args;
335 if(args->stride_align && decoded->header.cf != LV_COLOR_FORMAT_RGB565A8) {
336 uint32_t stride_expect = lv_draw_buf_width_to_stride(decoded->header.w, decoded->header.cf);
337 if(decoded->header.stride != stride_expect) {
338 LV_LOG_TRACE("Stride mismatch");
339 lv_result_t res = lv_draw_buf_adjust_stride(decoded, stride_expect);
340 if(res != LV_RESULT_OK) {
341 lv_draw_buf_t * aligned = lv_draw_buf_create_ex(image_cache_draw_buf_handlers, decoded->header.w, decoded->header.h,
342 decoded->header.cf, stride_expect);
343 if(aligned == NULL) {
344 LV_LOG_ERROR("No memory for Stride adjust.");
345 LV_PROFILER_DECODER_END;
346 return NULL;
347 }
348
349 lv_draw_buf_copy(aligned, NULL, decoded, NULL);
350 decoded = aligned;
351 }
352 }
353 }
354
355 /*Premultiply alpha channel*/
356 if(args->premultiply
357 && !LV_COLOR_FORMAT_IS_ALPHA_ONLY(decoded->header.cf)
358 && lv_color_format_has_alpha(decoded->header.cf)
359 && !lv_draw_buf_has_flag(decoded, LV_IMAGE_FLAGS_PREMULTIPLIED)
360 && decoded->header.cf != LV_COLOR_FORMAT_ARGB8888_PREMULTIPLIED /*Hasn't done yet*/
361 ) {
362 LV_LOG_TRACE("Alpha premultiply.");
363 if(lv_draw_buf_has_flag(decoded, LV_IMAGE_FLAGS_MODIFIABLE)) {
364 /*Do it directly*/
365 lv_draw_buf_premultiply(decoded);
366 }
367 else {
368 decoded = lv_draw_buf_dup_ex(image_cache_draw_buf_handlers, decoded);
369 if(decoded == NULL) {
370 LV_LOG_ERROR("No memory for premultiplying.");
371 LV_PROFILER_DECODER_END;
372 return NULL;
373 }
374
375 lv_draw_buf_premultiply(decoded);
376 }
377 }
378
379 LV_PROFILER_DECODER_END;
380 return decoded;
381}

387static lv_image_decoder_t * image_decoder_get_info(lv_image_decoder_dsc_t * dsc, lv_image_header_t * header)
388{
389 LV_ASSERT(dsc != NULL);
390 LV_ASSERT(header != NULL);
391 LV_PROFILER_DECODER_BEGIN;
392 lv_memzero(header, sizeof(lv_image_header_t));
393
394 const void * src = dsc->src;
395 lv_image_src_t src_type = dsc->src_type;
396
397 if(src_type == LV_IMAGE_SRC_VARIABLE) {
398 const lv_image_dsc_t * img_dsc = src;
399 if(img_dsc->data == NULL) {
400 LV_PROFILER_DECODER_END;
401 return NULL;
402 }
403 }
404
405 if(src_type == LV_IMAGE_SRC_FILE) LV_LOG_TRACE("Try to find decoder for %s", (const char *)src);
406 else LV_LOG_TRACE("Try to find decoder for %p", src);
407
408 lv_image_decoder_t * decoder;
409 bool is_header_cache_enabled = lv_image_header_cache_is_enabled();
410
411 if(is_header_cache_enabled && src_type == LV_IMAGE_SRC_FILE) {
412 lv_image_header_cache_data_t search_key;
413 search_key.src_type = src_type;
414 search_key.src = src;
415
416 lv_cache_entry_t * entry = lv_cache_acquire(img_header_cache_p, &search_key, NULL);
417
418 if(entry) {
419 lv_image_header_cache_data_t * cached_data = lv_cache_entry_get_data(entry);
420 *header = cached_data->header;
421 decoder = cached_data->decoder;
422 lv_cache_release(img_header_cache_p, entry, NULL);
423
424 LV_LOG_TRACE("Found decoder %s in header cache", decoder->name);
425 LV_PROFILER_DECODER_END;
426 return decoder;
427 }
428 }
429
430 if(src_type == LV_IMAGE_SRC_FILE) {
431 lv_fs_res_t fs_res = lv_fs_open(&dsc->file, src, LV_FS_MODE_RD);
432 if(fs_res != LV_FS_RES_OK) {
433 LV_LOG_ERROR("File open failed: %" LV_PRIu32, (uint32_t)fs_res);
434 LV_PROFILER_DECODER_END;
435 return NULL;
436 }
437 }
438
439 /*Search the decoders*/
440 LV_LL_READ(img_decoder_ll_p, decoder) {
441 /*Info and Open callbacks are required*/
442 if(decoder->info_cb && decoder->open_cb) {
443 lv_fs_seek(&dsc->file, 0, LV_FS_SEEK_SET);
444 LV_PROFILER_DECODER_BEGIN_TAG(decoder->name);
445 lv_result_t res = decoder->info_cb(decoder, dsc, header);
446 LV_PROFILER_DECODER_END_TAG(decoder->name);
447
448 if(res == LV_RESULT_OK) {
449 if(header->stride == 0) {
450 LV_LOG_INFO("Image decoder didn't set stride. Calculate it from width.");
451 header->stride = img_width_to_stride(header);
452 }
453 break;
454 }
455 else {
456 LV_LOG_TRACE("Can't open image with decoder %s. Trying next decoder.", decoder->name);
457 }
458 }
459 }
460
461 if(decoder == NULL) LV_LOG_TRACE("No decoder found");
462 else LV_LOG_TRACE("Found decoder %s", decoder->name);
463
464 if(src_type == LV_IMAGE_SRC_FILE) {
465 lv_fs_close(&dsc->file);
466 }
467
468 if(is_header_cache_enabled && src_type == LV_IMAGE_SRC_FILE && decoder) {
469 lv_cache_entry_t * entry;
470 lv_image_header_cache_data_t search_key;
471 search_key.src_type = src_type;
472 search_key.src = lv_strdup(src);
473 search_key.decoder = decoder;
474 search_key.header = *header;
475 entry = lv_cache_add(img_header_cache_p, &search_key, NULL);
476
477 if(entry == NULL) {
478 if(src_type == LV_IMAGE_SRC_FILE) lv_free((void *)search_key.src);
479 LV_PROFILER_DECODER_END;
480 return NULL;
481 }
482
483 lv_cache_release(img_header_cache_p, entry, NULL);
484 }
485
486 LV_PROFILER_DECODER_END;
487 return decoder;
488}

490static uint32_t img_width_to_stride(lv_image_header_t * header)
491{
492 LV_ASSERT(header != NULL);
493 if(header->cf == LV_COLOR_FORMAT_RGB565A8) {
494 return header->w * 2;
495 }
496 else {
497 return ((uint32_t)header->w * lv_color_format_get_bpp(header->cf) + 7) >> 3;
498 }
499}

501static lv_result_t try_cache(lv_image_decoder_dsc_t * dsc)
502{
503 LV_ASSERT(dsc != NULL);
504 LV_PROFILER_DECODER_BEGIN;
505 lv_cache_t * cache = dsc->cache;
506
507 lv_image_cache_data_t search_key;
508 search_key.src_type = dsc->src_type;
509 search_key.src = dsc->src;
510
511 lv_cache_entry_t * entry = lv_cache_acquire(cache, &search_key, NULL);
512
513 if(entry) {
514 lv_image_cache_data_t * cached_data = lv_cache_entry_get_data(entry);
515 dsc->decoded = cached_data->decoded;
516 dsc->decoder = (lv_image_decoder_t *)cached_data->decoder;
517 dsc->cache_entry = entry; /*Save the cache to release it in decoder_close*/
518 LV_PROFILER_DECODER_END;
519 return LV_RESULT_OK;
520 }
521
522 LV_PROFILER_DECODER_END;
523 return LV_RESULT_INVALID;
524}
