110static void * fs_open(lv_fs_drv_t * drv, const char * path, lv_fs_mode_t mode)
111{
112 LV_ASSERT(path != NULL);
113 LV_UNUSED(drv);
114
115 int flags = 0;
116 if(mode == LV_FS_MODE_WR) flags = O_WRONLY | O_CREAT;
117 else if(mode == LV_FS_MODE_RD) flags = O_RDONLY;
118 else if(mode == (LV_FS_MODE_WR | LV_FS_MODE_RD)) flags = O_RDWR | O_CREAT;
119
120 /*Make the path relative to the current directory (the projects root folder)*/
121 char buf[LV_FS_MAX_PATH_LEN];
122 lv_snprintf(buf, sizeof(buf), LV_FS_POSIX_PATH "%s", path);
123
124 int fd = open(buf, flags, 0666);
125 if(fd < 0) {
126 LV_LOG_WARN("Could not open file: %s, flags: 0x%x, errno: %d", buf, flags, errno);
127 return NULL;
128 }
129
130 return FD2FILEP(fd);
131}

218static lv_fs_res_t fs_seek(lv_fs_drv_t * drv, void * file_p, uint32_t pos, lv_fs_whence_t whence)
219{
220 LV_ASSERT(file_p != NULL);
221 LV_UNUSED(drv);
222 int w;
223 switch(whence) {
224 case LV_FS_SEEK_SET:
225 w = SEEK_SET;
226 break;
227 case LV_FS_SEEK_CUR:
228 w = SEEK_CUR;
229 break;
230 case LV_FS_SEEK_END:
231 w = SEEK_END;
232 break;
233 default:
234 return LV_FS_RES_INV_PARAM;
235 }
236
237 int fd = FILEP2FD(file_p);
238 off_t offset = lseek(fd, pos, w);
239 if(offset < 0) {
240 LV_LOG_WARN("Could not seek file: %d, errno: %d", fd, errno);
241 return fs_errno_to_res(errno);
242 }
243
244 return LV_FS_RES_OK;
245}

305static lv_fs_res_t fs_dir_read(lv_fs_drv_t * drv, void * dir_p, char * fn, uint32_t fn_len)
306{
307 LV_ASSERT(dir_p != NULL);
308 LV_ASSERT(fn != NULL);
309 LV_UNUSED(drv);
310 LV_ASSERT(fn_len > 0);
311
312 struct dirent * entry;
313 do {
314 entry = readdir(dir_p);
315 if(entry) {
316 if(entry->d_type == DT_DIR) lv_snprintf(fn, fn_len, "/%s", entry->d_name);
317 else lv_strlcpy(fn, entry->d_name, fn_len);
318 }
319 else {
320 lv_strlcpy(fn, "", fn_len);
321 }
322 } while(lv_strcmp(fn, "/.") == 0 || lv_strcmp(fn, "/..") == 0);
323
324 return LV_FS_RES_OK;
325}

352static lv_fs_res_t fs_errno_to_res(int errno_val)
353{
354 switch(errno_val) {

356 return LV_FS_RES_OK;
357
358 case EIO: /* I/O error */
359 return LV_FS_RES_HW_ERR;
360
361 case EFAULT: /* Bad address */
362 return LV_FS_RES_FS_ERR;
363
364 case ENOENT: /* No such file or directory */
365 return LV_FS_RES_NOT_EX;
366
367 case ENOSPC: /* No space left on device */
368 return LV_FS_RES_FULL;
369
370 case EALREADY: /* Operation already in progress */
371 return LV_FS_RES_LOCKED;
372
373 case EACCES: /* Permission denied */
374 return LV_FS_RES_DENIED;
375
376 case EBUSY: /* Device or resource busy */
377 return LV_FS_RES_BUSY;
378
379 case ETIMEDOUT: /* Connection timed out */
380 return LV_FS_RES_TOUT;
381
382 case ENOSYS: /* Invalid system call number */
383 return LV_FS_RES_NOT_IMP;
384
385 case ENOMEM: /* Out of memory */
386 return LV_FS_RES_OUT_OF_MEM;
387
388 case EINVAL: /* "Invalid argument" */
389 return LV_FS_RES_INV_PARAM;
390
391 default:
392 break;
393 }
394
395 return LV_FS_RES_UNKNOWN;
