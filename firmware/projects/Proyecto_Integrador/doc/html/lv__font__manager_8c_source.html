88{
89 LV_CHECK_ARG_MSG(recycle_cache_size > 0, return NULL, "recycle_cache_size must be greater than 0");

94 return NULL;

99 lv_ll_init(&manager->src_ll, sizeof(lv_font_src_t));
100
101 manager->recycle_manager = lv_font_manager_recycle_create(recycle_cache_size);
102
103 LV_LOG_INFO("success");
104 return manager;

107bool lv_font_manager_delete(lv_font_manager_t * manager)
108{
109 if(manager == NULL) return false;
110
111 /* Resource leak check */
112 if(lv_font_manager_check_resource(manager)) {
113 LV_LOG_ERROR("unfreed resource detected, delete failed!");
114 return false;
115 }
116
117 /* clean recycle_manager */
118 lv_font_manager_recycle_delete(manager->recycle_manager);
119
120 /* clean path map */
121 lv_font_src_t * font_src;
122 LV_LL_READ(&manager->src_ll, font_src) {
123 LV_LOG_INFO("remove src: %s", font_src->name);
124 if(!font_src->is_static) {
125 lv_free(font_src->name);
126 font_src->class_p->free_src_cb(font_src->src);
127 }
128
129 font_src->name = NULL;
130 font_src->src = NULL;
131 }
132 lv_ll_clear(&manager->src_ll);
133
134 lv_free(manager);
135
136 LV_LOG_INFO("success");
137 return true;
138}

166bool lv_font_manager_remove_src(lv_font_manager_t * manager, const char * name)

170

172 if(!font_src) {
173 LV_LOG_WARN("src: %s not found", name);

176

183
184 lv_ll_remove(&manager->src_ll, font_src);

187 lv_free(font_src->name);
188 font_src->class_p->free_src_cb(font_src->src);
189 }
190
191 font_src->name = NULL;
192 font_src->src = NULL;
193
194 lv_free(font_src);
195
196 LV_LOG_WARN("src: %s remove success", name);
197 return true;

200lv_font_t * lv_font_manager_create_font(lv_font_manager_t * manager,
201 const char * font_family,
202 uint32_t render_mode,
203 uint32_t size,
204 uint32_t style,
205 lv_font_kerning_t kerning)
206{
207
208 LV_CHECK_ARG(manager != NULL, return NULL);
209 LV_CHECK_ARG(font_family != NULL, return NULL);
210
211 lv_font_info_t ft_info;
212 lv_memzero(&ft_info, sizeof(ft_info));
213 ft_info.name = font_family;
214 ft_info.render_mode = render_mode;
215 ft_info.size = size;
216 ft_info.style = style;
217 ft_info.kerning = kerning;
218
219 lv_font_t * ret_font;
220
221 if(IS_FONT_FAMILY_NAME(ft_info.name)) {
222 ret_font = lv_font_manager_create_font_family(manager, &ft_info);
223 }
224 else {
225 ret_font = lv_font_manager_create_font_single(manager, &ft_info);
226 }
227
228 /* Append fallback font to make LV_SYMBOL displayable */
229 lv_font_t * cur_font = ret_font;
230 while(cur_font) {
231 if(cur_font->fallback == NULL) {
232 cur_font->fallback = LV_FONT_DEFAULT;
233 break;
234 }
235 cur_font = (lv_font_t *)cur_font->fallback;
236 }
237
238 return ret_font;
239}

284{
285 LV_ASSERT(manager != NULL);
286 LV_ASSERT(font != NULL);
287
288 if(font == LV_FONT_DEFAULT) {

313static const char * strncpy_until(char * dest, const char * src, size_t n, char c)
314{
315 LV_ASSERT(dest != NULL);
316 LV_ASSERT(src != NULL);
317
318 size_t i = 0;
319 while(i < n && *src != '\0' && *src != c) {
320 *dest++ = *src++;
321 i++;
322 }
323
324 if(i < n) {
325 *dest = '\0';
326 }
327
328 return src;
329}

331static lv_font_t * lv_font_manager_create_font_family(lv_font_manager_t * manager, const lv_font_info_t * ft_info)
332{
333 LV_ASSERT(manager != NULL);
334 LV_ASSERT(ft_info != NULL);
335
336 lv_font_t * first_font = NULL;
337 lv_font_t * pre_font = NULL;
338
339 const char * family_str = ft_info->name;
340 LV_LOG_INFO("font-family: %s", family_str);
341
342 char tmp_name[LV_FONT_MANAGER_NAME_MAX_LEN] = { 0 };
343 lv_font_info_t tmp_ft_info = *ft_info;
344 tmp_ft_info.name = tmp_name;
345
346 while(1) {
347 family_str = strncpy_until(tmp_name, family_str, sizeof(tmp_name) - 1, ',');
348
349 lv_font_t * cur_font = lv_font_manager_create_font_single(manager, &tmp_ft_info);
350
351 if(cur_font) {
352 /* save first font pointer */
353 if(!first_font) {
354 first_font = cur_font;
355 }
356
357 /* append font fallback */
358 if(pre_font) {
359 pre_font->fallback = cur_font;
360 }
361
362 pre_font = cur_font;
363 }
364
365 /* stop */
366 if(*family_str == '\0') {
367 break;
368 }
369
370 if(*family_str != ',') {
371 LV_LOG_ERROR("font name buffer is too small, please increase LV_FONT_MANAGER_NAME_MAX_LEN");
372 break;
373 }
374
375 /* skip ',' */
376 family_str++;
377 }
378
379 return first_font;
380}

445static bool lv_font_manager_check_resource(lv_font_manager_t * manager)
446{
447 LV_ASSERT(manager != NULL);
448
449 /* Check the recorded font */
450 lv_ll_t * rec_ll = &manager->rec_ll;
451 uint32_t rec_ll_len = lv_ll_get_len(rec_ll);
452 if(rec_ll_len) {
453 LV_LOG_WARN("lvgl font resource[%" LV_PRIu32 "]:", rec_ll_len);
454
455 lv_font_rec_node_t * node;
456 LV_LL_READ(rec_ll, node) {
457 LV_LOG_WARN("font: %p(%d) -> ref: %s(%d)",
458 (void *)node,
459 (int)node->font.line_height,
460 node->refer_node_p->ft_info.name,
461 node->refer_node_p->ft_info.size);
462 }
463 }
464
465 /* Check the recorded font resources created by font creator */
466 lv_ll_t * refer_ll = &manager->refer_ll;
467 uint32_t refer_ll_len = lv_ll_get_len(refer_ll);
468 if(refer_ll_len) {
469 LV_LOG_WARN("font resource[%" LV_PRIu32 "]:", refer_ll_len);
470
471 lv_font_refer_node_t * node;
472 LV_LL_READ(refer_ll, node) {
473 LV_LOG_WARN("font: %s(%d), ref_cnt = %d",
474 node->ft_info.name,
475 node->ft_info.size,
476 node->ref_cnt);
477 }
478 }
479
480 /* Check resource leak */
481 bool has_resource = (rec_ll_len || refer_ll_len);
482
483 return has_resource;
484}

583static const lv_font_refer_node_t * lv_font_manager_get_font(lv_font_manager_t * manager,
584 const lv_font_info_t * ft_info)
585{
586 LV_ASSERT(manager != NULL);
587 LV_ASSERT(ft_info != NULL);
588
589 /* check refer_node is existed */
590 lv_font_refer_node_t * refer_node = lv_font_manager_search_refer_node(manager, ft_info);
591 if(refer_node) {
592 refer_node->ref_cnt++;
593 LV_LOG_INFO("refer_node existed, ref_cnt++ = %d", refer_node->ref_cnt);
594 return refer_node;
595 }
596
597 /* not found refer_node, start to create font */
598
599 lv_font_t * font = lv_font_manager_create_font_wrapper(manager, ft_info);
600
601 if(!font) {
602 return NULL;
603 }
604
605 /* add refer_node to refer_ll */
606 refer_node = lv_ll_ins_head(&manager->refer_ll);
607 LV_ASSERT_MALLOC(refer_node);
608 lv_memzero(refer_node, sizeof(lv_font_refer_node_t));
609
610 lv_strncpy(refer_node->name, ft_info->name, sizeof(refer_node->name) - 1);
611
612 const lv_font_src_t * font_src = lv_font_manager_get_src(manager, ft_info->name);
613 LV_ASSERT(font_src != NULL);
614
615 /* copy font data */
616 refer_node->font_p = font;
617 refer_node->ft_info = *ft_info;
618 refer_node->ft_info.name = refer_node->name;
619 refer_node->ft_info.class_p = font_src->class_p;
620 refer_node->ref_cnt = 1;
621
622 LV_LOG_INFO("success");
623 return refer_node;
624}

626static bool lv_font_manager_drop_font(lv_font_manager_t * manager, const lv_font_refer_node_t * node)
627{
628 LV_ASSERT(manager != NULL);
629 LV_ASSERT(node != NULL);
630
631 /* Check refer_node is existed */
632 lv_font_refer_node_t * refer_node = lv_font_manager_search_refer_node(manager, &node->ft_info);
633 if(!refer_node) {
634 LV_LOG_WARN("NO record found for font: %s(%d),"
635 " it was not created by font manager",
636 node->ft_info.name, node->ft_info.size);
637 return false;
638 }
639
640 refer_node->ref_cnt--;
641
642 /* If ref_cnt is > 0, no need to delete font */
643 if(refer_node->ref_cnt > 0) {
644 LV_LOG_INFO("refer_node existed, ref_cnt-- = %d", refer_node->ref_cnt);
645 return true;
646 }
647
648 /* if ref_cnt is about to be 0, free font resource */
649 lv_font_manager_delete_font_wrapper(manager, refer_node);
650
651 /* free refer_node */
652 lv_ll_remove(&manager->refer_ll, refer_node);
653 lv_free(refer_node);
654
655 LV_LOG_INFO("success");
656 return true;
657}
