47uint32_t lv_img_cf_to_epic_cf(lv_color_format_t cf)
48{
49 uint32_t color_mode;
50
51 switch(cf) {
52 /* 2 byte (+alpha) formats */
53 case LV_COLOR_FORMAT_RGB565:
54 color_mode = EPIC_INPUT_RGB565;
55 break;
56
57#ifdef EPIC_SUPPORT_MASK
58 case LV_COLOR_FORMAT_RGB565A8:
59 color_mode = EPIC_INPUT_RGB565;
60 break;
61#endif
62
63 /* 3 byte (+alpha) formats */
64 case LV_COLOR_FORMAT_RGB888:
65 color_mode = EPIC_INPUT_RGB888;
66 break;
67
68 case LV_COLOR_FORMAT_ARGB8888:
69 case LV_COLOR_FORMAT_XRGB8888:
70 color_mode = EPIC_INPUT_ARGB8888;
71 break;
72
73 case LV_COLOR_FORMAT_ARGB8565:
74 color_mode = EPIC_INPUT_ARGB8565;
75 break;
76
77#ifdef EPIC_SUPPORT_A8
78 case LV_COLOR_FORMAT_A8:
79 color_mode = EPIC_INPUT_A8;
80 break;
81#endif
82
83#ifdef EPIC_SUPPORT_A4
84 case LV_COLOR_FORMAT_A4:
85 color_mode = EPIC_INPUT_A4;
86 break;
87#endif
88
89#ifdef EPIC_SUPPORT_A2
90 case LV_COLOR_FORMAT_A2:
91 color_mode = EPIC_INPUT_A2;
92 break;
93#endif
94
95#ifdef EPIC_SUPPORT_L8
96 case LV_COLOR_FORMAT_I8:
97 case LV_COLOR_FORMAT_L8:
98 color_mode = EPIC_INPUT_L8;
99 break;
100#endif
101
102#ifdef EPIC_SUPPORT_YUV
103 case LV_COLOR_FORMAT_I420:
104 color_mode = EPIC_INPUT_YUV420_PLANAR;
105 break;
106
107 case LV_COLOR_FORMAT_YUY2:
108 color_mode = EPIC_INPUT_YUV422_PACKED_YUYV;
109 break;
110
111 case LV_COLOR_FORMAT_UYVY:
112 color_mode = EPIC_INPUT_YUV422_PACKED_UYVY;
113 break;
114#endif
115
116 default:
117 LV_ASSERT(false);
118 color_mode = EPIC_INPUT_RGB565; /* Fallback */
119 break;
120 }
121
122 return color_mode;
123}

135uint32_t lv_epic_setup_layers(EPIC_LayerConfigTypeDef * epic_bg_layer,
136 EPIC_LayerConfigTypeDef * epic_output_layer,
137 lv_draw_task_t * draw_task,
138 const lv_area_t * coords)
139{
140 lv_layer_t * layer = draw_task->target_layer;
141 uint16_t dest_total_width;
142
143 lv_area_t blend_area;
144 if(!lv_area_intersect(&blend_area, coords, &draw_task->clip_area)) {
145 return 1; /* Fully clipped, nothing to do */
146 }
147
148 if(!lv_area_intersect(&blend_area, &blend_area, &layer->buf_area)) {
149 return 1; /* Fully clipped, nothing to do */
150 }
151
152 lv_color_format_t dest_cf = layer->color_format;
153 dest_total_width = layer->draw_buf ? lv_epic_stride_to_width(layer->draw_buf->header.stride, dest_cf) :
154 lv_area_get_width(&layer->buf_area);
155
156 /* Setup background layer */
157 HAL_EPIC_LayerConfigInit(epic_bg_layer);
158 epic_bg_layer->color_en = false;
159 epic_bg_layer->data = (uint8_t *)lv_draw_layer_go_to_xy(layer, 0, 0);
160 epic_bg_layer->color_mode = lv_img_cf_to_epic_cf(dest_cf);
161 epic_bg_layer->width = lv_area_get_width(&layer->buf_area);
162 epic_bg_layer->total_width = dest_total_width;
163 epic_bg_layer->height = lv_area_get_height(&layer->buf_area);
164 epic_bg_layer->x_offset = layer->buf_area.x1;
165 epic_bg_layer->y_offset = layer->buf_area.y1;
166
167 /* Setup output layer */
168 memcpy(epic_output_layer, epic_bg_layer, sizeof(*epic_output_layer));
169 epic_output_layer->width = lv_area_get_width(&blend_area);
170 epic_output_layer->height = lv_area_get_height(&blend_area);
171 epic_output_layer->x_offset = blend_area.x1;
172 epic_output_layer->y_offset = blend_area.y1;
173 epic_output_layer->data = (uint8_t *)lv_draw_layer_go_to_xy(layer,
174 blend_area.x1 - layer->buf_area.x1,
175 blend_area.y1 - layer->buf_area.y1);
176
177 LV_ASSERT((epic_output_layer->height * epic_output_layer->width) <= EPIC_MAX_PIXELS);
178
179 return 0; /* Success */
180}

182bool lv_epic_cf_supported(lv_color_format_t cf, uint32_t flags)
183{
184 /* EZIP images */
185 if((LV_COLOR_FORMAT_RAW == cf || LV_COLOR_FORMAT_RAW_ALPHA == cf) &&
186 (0 != (flags & LV_IMAGE_FLAGS_EZIP))) {
187 return true;
188 }
189
190#if defined(EPIC_INPUT_JPEG)
191 if((LV_COLOR_FORMAT_RAW == cf || LV_COLOR_FORMAT_RAW_ALPHA == cf) &&
192 (0 != (flags & LV_IMAGE_FLAGS_JPEG))) {
193 return true;
194 }
195#endif
196
197 /* Normal images */
198 if(LV_COLOR_FORMAT_RGB565 == cf ||
199 LV_COLOR_FORMAT_ARGB8565 == cf ||
200 LV_COLOR_FORMAT_RGB888 == cf ||
201 LV_COLOR_FORMAT_ARGB8888 == cf ||
202 LV_COLOR_FORMAT_XRGB8888 == cf) {
203 return true;
204 }
205
206#ifdef EPIC_SUPPORT_A8
207 if(LV_COLOR_FORMAT_A8 == cf) return true;
208#endif
209
210#ifdef EPIC_SUPPORT_MASK
211 if(LV_COLOR_FORMAT_RGB565A8 == cf) return true;
212#endif
213
214#ifdef EPIC_SUPPORT_A4
215 if(LV_COLOR_FORMAT_A4 == cf) return true;
216#endif
217
218#ifdef EPIC_SUPPORT_A2
219 if(LV_COLOR_FORMAT_A2 == cf) return true;
220#endif
221
222#ifdef EPIC_SUPPORT_L8
223 if((LV_COLOR_FORMAT_I8 == cf) || (LV_COLOR_FORMAT_L8 == cf)) return true;
224#endif
225
226#ifdef EPIC_SUPPORT_YUV
227 if(LV_COLOR_FORMAT_I420 == cf ||
228 LV_COLOR_FORMAT_YUY2 == cf ||
229 LV_COLOR_FORMAT_UYVY == cf) {
230 return true;
231 }
232#endif
233
234 return false;
235}

237uint16_t lv_epic_stride_to_width(uint32_t stride, lv_color_format_t cf)
238{
239 uint8_t bpp = lv_color_format_get_bpp(cf);
240
241 if(bpp == 0) {
242 return 0;
243 }
244
245 return (uint16_t)((stride * 8U) / bpp);
246}
