1#!/usr/bin/env python3
2"""Check markdown links in docs/src/.
3
4Built around two concerns with very different CI tolerances:
5
6 ERRORS (always fail CI)
7 * Local links — root-relative (/foo) or relative (./foo, ../foo) — that
8 do not resolve to a real .mdx file under docs/src/. The local file
9 tree is the source of truth; we do not consult the deployed sitemap
10 so a new page added in this PR is treated as valid.
11 * Raw markdown links into /api or /api-private — these routes are
12 auto-generated from C headers (Doxygen -> JSON -> MDX). API
13 references must be written with the <ApiLink name="..." /> JSX
14 component, not a raw markdown link, per docs/src/contributing/
15 writing_docs.mdx.
16 * Legacy domain links — anything pointing at docs.lvgl.io/master|latest
17 or older versioned mirrors. These are conceptually internal links
18 that need to be migrated to current internal routes.
19
20 WARNINGS (never fail CI)
21 * Links into /examples — also auto-generated at build time (from the
22 examples/ tree at the repo root) so we cannot verify them locally.
23 Listed for visibility; treat as "check after deploy".
24 * External links (http(s)://...) — probed only when --check-external
25 is passed. MANUAL / PERIODIC USE ONLY: ~800 outbound requests per
26 run are too slow, flaky, and unfriendly to targets for per-PR CI.
27 Schedule weekly or run ad-hoc; never as a required PR gate.
28
29Routing rules (mirrors Astro file-based routing in docs/src/):
30 foo.mdx -> /foo
31 foo/index.mdx -> /foo
32 foo/bar.mdx -> /foo/bar
33 docs/src/index.mdx -> /
34
35Convention reminder: this repo writes internal links as /foo/bar (root-
36relative). The script accepts ./foo and ../foo too, but they are not the
37house style — see docs/src/contributing/writing_docs.mdx.
38"""
39from __future__ import annotations

41import argparse

49DOCS_ROOT = Path(__file__).resolve().parents[1] / "docs" / "src"

51# [text](url) — non-greedy text, url stops at whitespace, ) or ".
52# Skips images (![...]) by requiring no leading '!' (negative lookbehind).
53MD_LINK = re.compile(r"(?<!\!)\‍[([^\‍]]*?)\‍]\‍(\s*([^\s)]+)(?:\s+\"[^\"]*\")?\s*\‍)")
54
55# Matches the legacy Sphinx-site domain in any of its forms.
56LEGACY_HOST = re.compile(
57 r"^https?://(?:docs\.lvgl\.io|lvgl\.io/docs)/(?:master|latest|v\d[^/]*)/",
58 re.IGNORECASE,
59)
60
61EXTERNAL_OK_STATUSES = {200, 301, 302, 307, 308}
62EXTERNAL_TIMEOUT = 10
63EXTERNAL_WORKERS = 20
64USER_AGENT = "lvgl-doc-link-checker/1.0"

66# Auto-generated route prefixes — these exist on the deployed site but have
67# no source files under docs/src/, so we can't validate them by walking the
68# local tree. Each tuple is (prefix-without-trailing-slash, classification).
69# * /api and /api-private are built from C headers (Doxygen). Authors must
70# use <ApiLink name="..." /> instead of a raw markdown link — error.
71# * /examples is built from the repo's top-level examples/ tree at docs
72# build time, so we cannot verify locally — warning only.
73AUTOGEN_PREFIXES: tuple[tuple[str, str], ...] = (
74 ("/api", "api"),
75 ("/api-private", "api"),
76 ("/examples", "example"),
77)

80def autogen_kind(target: str) -> str | None:
81 """Return 'api' / 'example' if `target` is under an auto-generated route, else None."""
82 for prefix, kind in AUTOGEN_PREFIXES:
83 if target == prefix or target.startswith(prefix + "/"):
84 return kind
85 return None

88def build_local_routes() -> set[str]:
89 """Walk docs/src/ and return the set of valid site routes."""
90 routes: set[str] = set()
91 for path in DOCS_ROOT.rglob("*.mdx"):
92 # Skip anything under _-prefixed directories (Astro convention for non-route assets).
93 if any(part.startswith("_") for part in path.relative_to(DOCS_ROOT).parts):
94 continue
95 rel = path.relative_to(DOCS_ROOT).with_suffix("").as_posix()
96 route = "/" + rel
97 if route.endswith("/index"):
98 route = route[: -len("/index")] or "/"
99 routes.add(route)
100 return routes

103def resolve_link(link: str, source_file: Path) -> tuple[str, str] | None:
104 """Classify and normalize a link.
105
106 Returns (kind, value) where kind is one of:
107 "local" — value is the normalized site route (e.g. "/integration/rtos/zephyr")
108 "api" — value is the normalized site route under /api or /api-private
109 "example" — value is the normalized site route under /examples
110 "legacy" — value is the original URL (legacy docs.lvgl.io mirror)
111 "external" — value is the original URL
112 Returns None for things we don't check (anchors, mailto, empty, etc.).
113 """
114 # Strip anchor/query for resolution purposes; we don't verify anchors.
115 bare = link.split("#", 1)[0].split("?", 1)[0]
116 if not bare:
117 return None # pure #anchor or empty
118 if bare.startswith("//"):
119 return ("external", link)
120 if LEGACY_HOST.match(link):
121 return ("legacy", link)
122 if re.match(r"^[a-zA-Z][a-zA-Z0-9+.-]*:", link):
123 # Any scheme — http, https, mailto, ftp, etc.
124 if link.lower().startswith(("http://", "https://")):
125 return ("external", link)
126 return None # mailto:, tel:, etc. — not link-checkable
127
128 # Local link.
129 if bare.startswith("/"):
130 target = bare
131 else:
132 rel_dir = source_file.parent.relative_to(DOCS_ROOT)
133 joined = (rel_dir / bare).as_posix()
134 parts: list[str] = []
135 for p in joined.split("/"):
136 if p in ("", "."):
137 continue
138 if p == "..":
139 if parts:
140 parts.pop()
141 continue
142 parts.append(p)
143 target = "/" + "/".join(parts)
144
145 target = target.rstrip("/")
146 target = re.sub(r"\.(mdx?|html)$", "", target)
147 if not target:
148 target = "/"
149 kind = autogen_kind(target)
150 if kind is not None:
151 return (kind, target)
152 return ("local", target)

155def check_external(url: str) -> tuple[str, int | None, str | None]:
156 """HEAD then GET fallback. Return (url, status_code, error_message)."""
157 for method in ("HEAD", "GET"):
158 try:
159 req = urllib.request.Request(url, method=method, headers={"User-Agent": USER_AGENT})
160 with urllib.request.urlopen(req, timeout=EXTERNAL_TIMEOUT) as resp:
161 return (url, resp.status, None)
162 except urllib.error.HTTPError as e:
163 # 4xx/5xx — only retry HEAD->GET on 405 Method Not Allowed.
164 if method == "HEAD" and e.code == 405:
165 continue
166 return (url, e.code, None)
167 except Exception as e:
168 if method == "HEAD":
169 continue
170 return (url, None, f"{type(e).__name__}: {e}")
171

173def main() -> int:
174 parser = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
175 parser.add_argument(
176 "--check-external",
177 action="store_true",
178 help=(
179 "MANUAL / PERIODIC USE ONLY. Also probe external http(s) URLs and warn on "
180 "non-2xx/3xx responses. Warnings never affect exit code. Do not enable in "
181 "per-PR CI — ~800 outbound requests are slow, flaky, and unfriendly to targets."
182 ),
183 )
184 args = parser.parse_args()
185
186 routes = build_local_routes()
187 print(f"Indexed {len(routes)} local routes from {DOCS_ROOT.relative_to(DOCS_ROOT.parents[1])}", file=sys.stderr)
188
189 # (path, lineno, link-text, raw url) for each finding.
190 broken_local: dict[str, list[tuple[Path, int, str, str]]] = defaultdict(list)
191 api_links: list[tuple[Path, int, str, str, str]] = [] # +resolved target
192 legacy: list[tuple[Path, int, str, str]] = []
193 example_links: list[tuple[Path, int, str, str, str]] = [] # +resolved target
194 externals: list[tuple[Path, int, str, str]] = []
195
196 checked_local = 0
197 for path in sorted(DOCS_ROOT.rglob("*.mdx")):
198 text = path.read_text(encoding="utf-8", errors="replace")
199 for lineno, line in enumerate(text.splitlines(), 1):
200 for m in MD_LINK.finditer(line):
201 text_part, url = m.group(1), m.group(2)
202 resolved = resolve_link(url, path)
203 if resolved is None:
204 continue
205 kind, value = resolved
206 if kind == "api":
207 api_links.append((path, lineno, text_part, url, value))
208 elif kind == "example":
209 example_links.append((path, lineno, text_part, url, value))
210 elif kind == "legacy":
211 legacy.append((path, lineno, text_part, url))
212 elif kind == "external":
213 externals.append((path, lineno, text_part, url))
214 elif kind == "local":
215 checked_local += 1
216 if value not in routes:
217 broken_local[value].append((path, lineno, text_part, url))
218
219 exit_code = 0
220
221 # --- Section 1: local links (ERRORS) ---
222 print("\n=== Local links (errors) ===")
223 if not broken_local:
224 print(f"OK: checked {checked_local} local links — all resolve to docs/src/ files.")
225 else:
226 total = sum(len(v) for v in broken_local.values())
227 print(f"ERROR: {total} broken local link(s) across {len(broken_local)} unique targets:\n")
228 for target in sorted(broken_local):
229 for path, lineno, text_part, url in broken_local[target]:
230 rel = path.relative_to(DOCS_ROOT.parents[1])
231 print(f" {rel}:{lineno}")
232 print(f" [{text_part}]({url}) -> {target} (no matching file)")
233 exit_code = 1
234
235 # --- Section 2: API references via raw markdown link (ERRORS) ---
236 print("\n=== API references written as raw markdown links (errors) ===")
237 if not api_links:
238 print("OK: no markdown links into /api or /api-private.")
239 else:
240 print(
241 f"ERROR: {len(api_links)} markdown link(s) point at /api or /api-private. "
242 f"These routes are auto-generated; use the <ApiLink name=\"...\" /> "
243 f"component instead (see docs/src/contributing/writing_docs.mdx).\n"
244 )
245 for path, lineno, text_part, url, target in api_links:
246 rel = path.relative_to(DOCS_ROOT.parents[1])
247 print(f" {rel}:{lineno}")
248 print(f" [{text_part}]({url}) -> {target}")
249 exit_code = 1
250
251 # --- Section 3: legacy domain (ERRORS) ---
252 print("\n=== Legacy docs.lvgl.io domain links (errors) ===")
253 if not legacy:
254 print("OK: no legacy-domain links.")
255 else:
256 print(f"ERROR: {len(legacy)} link(s) still point at docs.lvgl.io/master|latest|...:\n")
257 for path, lineno, text_part, url in legacy:
258 rel = path.relative_to(DOCS_ROOT.parents[1])
259 print(f" {rel}:{lineno}")
260 print(f" [{text_part}]({url})")
261 exit_code = 1
262
263 # --- Section 4: /examples links (WARNINGS) ---
264 print("\n=== /examples links — auto-generated routes, cannot verify locally (warnings) ===")
265 if not example_links:
266 print("OK: no links into /examples.")
267 else:
268 print(
269 f"WARNING: {len(example_links)} link(s) into /examples. These routes are "
270 f"built post-CI from the examples/ tree at the repo root, so this script "
271 f"cannot verify them. Confirm after deploy:\n"
272 )
273 for path, lineno, text_part, url, target in example_links:
274 rel = path.relative_to(DOCS_ROOT.parents[1])
275 print(f" {rel}:{lineno}")
276 print(f" [{text_part}]({url}) -> {target}")
277 # Warnings do NOT change exit_code.
278
279 # --- Section 5: external links (WARNINGS, opt-in) ---
280 print("\n=== External links — MANUAL / PERIODIC CHECK ONLY (warnings, never fails CI) ===")
281 print("NOTE: This section is NOT intended for per-PR CI. It issues ~800 outbound")
282 print(" HTTP requests; running on every PR is slow, flaky (transient host")
283 print(" failures, rate limits, captive portals), and unfriendly to the targets.")
284 print(" Run on a schedule (e.g. weekly) or on-demand instead.")
285 if not args.check_external:
286 print(f"\nSKIPPED: {len(externals)} external link occurrence(s). Pass --check-external to verify.")
287 elif not externals:
288 print("\nOK: no external links found.")
289 else:
290 unique = sorted({url for _, _, _, url in externals})
291 print(f"\nChecking {len(unique)} unique external URL(s) (HEAD/GET, timeout {EXTERNAL_TIMEOUT}s, {EXTERNAL_WORKERS} workers)...")
292 results: dict[str, tuple[int | None, str | None]] = {}
293 with ThreadPoolExecutor(max_workers=EXTERNAL_WORKERS) as ex:
294 futures = {ex.submit(check_external, u): u for u in unique}
295 for fut in as_completed(futures):
296 u, status, err = fut.result()
297 results[u] = (status, err)
298 bad = []
299 for path, lineno, text_part, url in externals:
300 status, err = results[url]
301 if status is not None and status in EXTERNAL_OK_STATUSES:
302 continue
303 bad.append((path, lineno, text_part, url, status, err))
304 if not bad:
305 print(f"OK: all {len(unique)} external URL(s) responded with 200/3xx.")
306 else:
307 print(f"WARNING: {len(bad)} external link occurrence(s) did not return 200/3xx (not failing CI):\n")
308 for path, lineno, text_part, url, status, err in bad:
309 rel = path.relative_to(DOCS_ROOT.parents[1])
310 detail = f"HTTP {status}" if status is not None else err
311 print(f" {rel}:{lineno} [{detail}]")
312 print(f" [{text_part}]({url})")
313 # Warnings do NOT change exit_code.
314
315 return exit_code
