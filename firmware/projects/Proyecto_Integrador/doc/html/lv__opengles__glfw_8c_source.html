101lv_opengles_window_t * lv_opengles_glfw_window_create_ex(int32_t hor_res, int32_t ver_res, bool use_mouse_indev,
102 bool h_flip, bool v_flip, const char * title)
103{
104 LV_CHECK_ARG(title != NULL, return NULL);
105
106 if(lv_glfw_init() != 0) {
107 LV_LOG_ERROR("Failed to init glfw");
108 return NULL;
109 }
110
111 lv_opengles_window_t * window = lv_ll_ins_tail(&glfw_window_ll);
112 LV_ASSERT_MALLOC(window);
113 if(window == NULL) return NULL;
114 if(window == NULL) {
115 LV_LOG_ERROR("Failed to create glfw window");
116 return NULL;
117 }
118 lv_memzero(window, sizeof(*window));
119
120 /* Create window with graphics context */
121 lv_opengles_window_t * existing_window = lv_ll_get_head(&glfw_window_ll);
122 window->window = glfwCreateWindow(hor_res, ver_res, title, NULL,
123 existing_window ? existing_window->window : NULL);
124 if(window->window == NULL) {
125 LV_LOG_ERROR("glfwCreateWindow fail");
126 lv_ll_remove(&glfw_window_ll, window);
127 lv_free(window);
128 return NULL;
129 }
130
131 glfwSetWindowTitle(window->window, title);
132
133 window->h_flip = h_flip;
134 window->v_flip = v_flip;
135 window->hor_res = hor_res;
136 window->ver_res = ver_res;
137
138 lv_ll_init(&window->textures, sizeof(lv_opengles_window_texture_t));
139 window->use_indev = use_mouse_indev;
140#if LV_USE_DRAW_OPENGLES
141 window->direct_render_invalidated = 1;
142#endif
143
144 glfwSetWindowUserPointer(window->window, window);
145 lv_glfw_timer_init();
146 lv_glfw_window_config(window->window, use_mouse_indev);
147
148 lv_result_t res = lv_glad_init();
149 if(res != LV_RESULT_OK) {
150 LV_LOG_ERROR("Failed to init glad");
151 glfwDestroyWindow(window->window);
152 lv_ll_remove(&glfw_window_ll, window);
153 lv_free(window);
154 return NULL;
155 }
156
157 glfwMakeContextCurrent(window->window);
158 lv_opengles_init();
159
160 return window;
161}

176void lv_opengles_window_delete(lv_opengles_window_t * window)
177{
178 if(window == NULL) return;
179
180 glfwDestroyWindow(window->window);
181
182 lv_opengles_window_texture_t * texture;
183 while((texture = lv_ll_get_head(&window->textures))) {
184 if(texture->texture_id) {
185 lv_opengles_window_texture_remove(texture);
186 }
187 else {
188 lv_display_delete(texture->disp);
189 }
190 }
191
192 lv_ll_remove(&glfw_window_ll, window);
193 lv_free(window);
194
195 if(lv_ll_is_empty(&glfw_window_ll)) {
196 lv_glfw_window_quit();
197#if !LV_USE_DRAW_OPENGLES
198 if(window_display_texture) {
199 GL_CALL(glDeleteTextures(1, &window_display_texture));
200 window_display_texture = 0;
201 }
202#endif
203 }

221lv_opengles_window_texture_t * lv_opengles_window_add_texture(lv_opengles_window_t * window, unsigned int texture_id,
222 int32_t w, int32_t h)
223{
224 LV_CHECK_ARG(window != NULL, return NULL);
225
226 lv_opengles_window_texture_t * texture = lv_ll_ins_tail(&window->textures);
227 LV_ASSERT_MALLOC(texture);
228 if(texture == NULL) return NULL;
229 lv_memzero(texture, sizeof(*texture));
230 texture->window = window;
231 texture->texture_id = texture_id;
232 texture->disp = lv_opengles_texture_get_from_texture_id(texture_id);
233 lv_area_set(&texture->area, 0, 0, w - 1, h - 1);
234 texture->opa = LV_OPA_COVER;
235
236 if(window->use_indev && texture->disp) {
237 lv_indev_t * indev = lv_indev_create();
238 if(indev == NULL) {
239 lv_ll_remove(&window->textures, texture);
240 lv_free(texture);
241 return NULL;
242 }
243 texture->indev = indev;
244 lv_indev_set_type(indev, LV_INDEV_TYPE_POINTER);
245 lv_indev_set_read_cb(indev, indev_read_cb);
246 lv_indev_set_driver_data(indev, texture);
247 lv_indev_set_mode(indev, LV_INDEV_MODE_EVENT);
248 lv_indev_set_display(indev, texture->disp);
249 }
250
251#if LV_USE_DRAW_OPENGLES
252 window->direct_render_invalidated = 1;
253#endif
254
255 return texture;
256}

278lv_display_t * lv_opengles_window_display_create(lv_opengles_window_t * window, int32_t w, int32_t h)
279{
280 LV_CHECK_ARG(window != NULL, return NULL);
281
282 lv_display_t * disp = lv_display_create(w, h);
283 if(disp == NULL) {
284 return NULL;
285 }
286
287 lv_opengles_window_texture_t * dsc = lv_ll_ins_tail(&window->textures);
288 LV_ASSERT_MALLOC(dsc);
289 if(dsc == NULL) {
290 lv_display_delete(disp);
291 return NULL;
292 }
293 lv_memzero(dsc, sizeof(*dsc));
294 dsc->window = window;
295 dsc->disp = disp;
296 lv_area_set(&dsc->area, 0, 0, w - 1, h - 1);
297 dsc->opa = LV_OPA_COVER;
298
299#if LV_USE_DRAW_OPENGLES
300 static size_t LV_ATTRIBUTE_MEM_ALIGN dummy_buf;
301 lv_display_set_buffers(disp, &dummy_buf, NULL, h * lv_draw_buf_width_to_stride(w, LV_COLOR_FORMAT_ARGB8888),
302 LV_DISPLAY_RENDER_MODE_FULL);
303 lv_display_set_draw_buf_handlers(disp, lv_draw_opengles_get_draw_buf_handlers());
304#else
305 uint32_t stride = lv_draw_buf_width_to_stride(w, lv_display_get_color_format(disp));
306 uint32_t buf_size = stride * h;
307 dsc->fb = malloc(buf_size);
308 LV_ASSERT_MALLOC(dsc->fb);
309 if(dsc->fb == NULL) {
310 lv_display_delete(disp);
311 lv_ll_remove(&window->textures, dsc);
312 lv_free(dsc);
313 return NULL;
314 }
315 lv_display_set_buffers(disp, dsc->fb, NULL, buf_size, LV_DISPLAY_RENDER_MODE_DIRECT);
316#endif
317
318 if(window->use_indev) {
319 lv_indev_t * indev = lv_indev_create();
320 if(indev == NULL) {
321 lv_display_delete(disp);
322 lv_ll_remove(&window->textures, dsc);
323 lv_free(dsc);
324 return NULL;
325 }
326 dsc->indev = indev;
327 lv_indev_set_type(indev, LV_INDEV_TYPE_POINTER);
328 lv_indev_set_read_cb(indev, indev_read_cb);
329 lv_indev_set_driver_data(indev, dsc);
330 lv_indev_set_mode(indev, LV_INDEV_MODE_EVENT);
331 lv_indev_set_display(indev, disp);
332 }
333
334 lv_display_set_driver_data(disp, dsc);
335 lv_display_delete_refr_timer(disp);
336 lv_display_set_flush_cb(disp, window_display_flush_cb);
337 lv_display_add_event_cb(disp, window_display_delete_cb, LV_EVENT_DELETE, disp);
338
339#if LV_USE_DRAW_OPENGLES
340 window->direct_render_invalidated = 1;
341#endif
342
343 return disp;
344}

411

417{
418 if(glad_inited) {

421
422
423 if(!gladLoadGL((GLADloadfunc)glfwGetProcAddress)) {
424 LV_LOG_ERROR("Failed to load OpenGL functions");

427
428 LV_LOG_INFO("GL version: %s", glGetString(GL_VERSION));
429 LV_LOG_INFO("GLSL version: %s", glGetString(GL_SHADING_LANGUAGE_VERSION));
430
431 glad_inited = true;
432
433 return LV_RESULT_OK;

475static void window_update_handler(lv_timer_t * t)
476{
477 LV_UNUSED(t);
478
479 lv_opengles_window_t * window;
480
481 glfwPollEvents();
482
483 /* delete windows that are ready to close */
484 window = lv_ll_get_head(&glfw_window_ll);
485 while(window) {
486 lv_opengles_window_t * window_to_delete = window->closing ? window : NULL;
487 window = lv_ll_get_next(&glfw_window_ll, window);
488 if(window_to_delete) {
489 glfwSetWindowShouldClose(window_to_delete->window, GLFW_TRUE);
490 lv_opengles_window_delete(window_to_delete);
491 }
492 }
493
494 /* render each window */
495 LV_LL_READ(&glfw_window_ll, window) {
496 glfwMakeContextCurrent(window->window);
497 lv_opengles_viewport(0, 0, window->hor_res, window->ver_res);
498
499#if LV_USE_DRAW_OPENGLES
500 lv_opengles_window_texture_t * textures_head;
501 bool window_display_direct_render =
502 !window->direct_render_invalidated
503 && (textures_head = lv_ll_get_head(&window->textures))
504 && textures_head->texture_id == 0 /* it's a window display */
505 && lv_ll_get_next(&window->textures, textures_head) == NULL /* it's the only one */
506 && textures_head->opa == LV_OPA_COVER
507 && textures_head->area.x1 == 0
508 && textures_head->area.y1 == 0
509 && textures_head->area.x2 == window->hor_res - 1
510 && textures_head->area.y2 == window->ver_res - 1
511 ;
512 window->direct_render_invalidated = 0;
513 if(!window_display_direct_render) {
514 lv_opengles_render_clear();
515 }
516#else
517 lv_opengles_render_clear();
518#endif
519
520 /* render each texture in the window */
521 lv_opengles_window_texture_t * texture;
522 LV_LL_READ(&window->textures, texture) {
523 if(texture->texture_id == 0) { /* it's a window display */
524#if LV_USE_DRAW_OPENGLES
525 lv_display_set_render_mode(texture->disp,
526 window_display_direct_render ? LV_DISPLAY_RENDER_MODE_DIRECT : LV_DISPLAY_RENDER_MODE_FULL);
527#endif
528
529 lv_display_t * default_save = lv_display_get_default();
530 lv_display_set_default(texture->disp);
531 lv_display_refr_timer(NULL);
532 lv_display_set_default(default_save);
533
534#if !LV_USE_DRAW_OPENGLES
535 ensure_init_window_display_texture();
536
537 /* set the dimensions and format to complete the texture */
538 const lv_color_format_t cf = lv_display_get_color_format(texture->disp);
539 const int32_t tex_w = lv_area_get_width(&texture->area);
540 const int32_t tex_h = lv_area_get_height(&texture->area);
541 if(lv_opengles_texture_upload_buf(window_display_texture, texture->fb, tex_w, tex_h,
542 lv_draw_buf_width_to_stride(tex_w, cf), cf) != LV_RESULT_OK) {
543 continue;
544 }
545
546 GL_CALL(glGenerateMipmap(GL_TEXTURE_2D));
547
548 GL_CALL(glBindTexture(GL_TEXTURE_2D, 0));
549
550 lv_opengles_render_texture_cf(window_display_texture, &texture->area, texture->opa, window->hor_res,
551 window->ver_res,
552 &texture->area, window->h_flip, window->v_flip, cf);
553#endif
554 }
555 else {
556 /* if the added texture is an LVGL opengles texture display, refresh it before rendering it */
557 if(texture->disp != NULL) {
558#if LV_USE_DRAW_OPENGLES
559 lv_display_t * default_save = lv_display_get_default();
560 lv_display_set_default(texture->disp);
561 lv_display_refr_timer(NULL);
562 lv_display_set_default(default_save);
563#else
564 lv_refr_now(texture->disp);
565#endif
566 }
567
568#if LV_USE_DRAW_OPENGLES
569 /*The draw unit rendered into the texture, in the OpenGL channel order*/
570 lv_opengles_render_texture(texture->texture_id, &texture->area, texture->opa, window->hor_res, window->ver_res,
571 &texture->area, window->h_flip, texture->disp == NULL ? window->v_flip : !window->v_flip);
572#else
573 if(texture->disp != NULL) {
574 /*The texture display uploaded its buffer, so its color format tells the channel order*/
575 lv_opengles_render_texture_cf(texture->texture_id, &texture->area, texture->opa, window->hor_res,
576 window->ver_res, &texture->area, window->h_flip, window->v_flip,
577 lv_display_get_color_format(texture->disp));
578 }
579 else {
580 /*A texture of the application, assume the OpenGL channel order*/
581 lv_opengles_render_texture(texture->texture_id, &texture->area, texture->opa, window->hor_res,
582 window->ver_res, &texture->area, window->h_flip, window->v_flip);
583 }
584#endif
585 }
586 }
587
588 /* Swap front and back buffers */
589 glfwSwapBuffers(window->window);
590 }
591}

637static void proc_mouse(lv_opengles_window_t * window)
638{
639 /* mouse activity will affect the topmost LVGL display texture */
640 lv_opengles_window_texture_t * texture;
641 LV_LL_READ_BACK(&window->textures, texture) {
