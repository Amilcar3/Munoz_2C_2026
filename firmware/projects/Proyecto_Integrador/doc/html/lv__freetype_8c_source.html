94lv_result_t lv_freetype_init(uint32_t max_glyph_cnt)
95{
96 if(ft_ctx) {
97 LV_LOG_WARN("freetype already initialized");
98 return LV_RESULT_INVALID;
99 }
100
101 ft_ctx = lv_malloc_zeroed(sizeof(lv_freetype_context_t));
102 LV_ASSERT_MALLOC(ft_ctx);
103 if(!ft_ctx) {
104 LV_LOG_ERROR("malloc failed for lv_freetype_context_t");
105 return LV_RESULT_INVALID;
106 }
107
108 lv_freetype_context_t * ctx = lv_freetype_get_context();
109
110 ctx->max_glyph_cnt = max_glyph_cnt;
111
112 FT_Error error;
113
114 error = FT_Init_FreeType(&ctx->library);
115 if(error) {
116 FT_ERROR_MSG("FT_Init_FreeType", error);
117 return LV_RESULT_INVALID;
118 }
119
120 lv_ll_init(&ctx->face_id_ll, sizeof(face_id_node_t));
121
122 lv_cache_ops_t ops = {
123 .compare_cb = (lv_cache_compare_cb_t)cache_node_cache_compare_cb,
124 .create_cb = (lv_cache_create_cb_t)cache_node_cache_create_cb,
125 .free_cb = (lv_cache_free_cb_t)cache_node_cache_free_cb,
126 };
127 ctx->cache_node_cache = lv_cache_create(&lv_cache_class_lru_rb_count, sizeof(lv_freetype_cache_node_t), INT32_MAX, ops);
128 lv_cache_set_name(ctx->cache_node_cache, "FREETYPE_CACHE_NODE");
129
130 return LV_RESULT_OK;
131}

157lv_font_t * lv_freetype_font_create_with_info(const lv_font_info_t * font_info)
158{
159 LV_CHECK_ARG(font_info != NULL, return NULL);
160 if(font_info->size == 0) {
161 LV_LOG_ERROR("font size can't be zero");
162 return NULL;
163 }
164
165 const char * pathname = font_info->name;
166
167 size_t pathname_len = pathname ? lv_strlen(pathname) : 0;
168 if(pathname_len == 0) {
169 LV_LOG_ERROR("font pathname can't be empty");
170 return NULL;
171 }
172
173 lv_freetype_context_t * ctx = lv_freetype_get_context();
174
175 lv_freetype_cache_node_t search_key = {
176 .pathname = lv_freetype_req_face_id(ctx, pathname),
177 .style = font_info->style,
178 .render_mode = font_info->render_mode,
179 /* Normalize weight: DEFAULT(0) resolves to BOLD(700) or NORMAL(400) to avoid duplicate cache nodes */
180 .weight = font_info->weight > LV_FREETYPE_VAR_WEIGHT_DEFAULT ? font_info->weight
181 : ((font_info->style & LV_FREETYPE_FONT_STYLE_BOLD) ? LV_FREETYPE_VAR_WEIGHT_BOLD : LV_FREETYPE_VAR_WEIGHT_NORMAL),
182 };
183
184 bool cache_hitting = true;
185 lv_cache_entry_t * cache_node_entry = lv_cache_acquire(ctx->cache_node_cache, &search_key, NULL);
186 if(cache_node_entry == NULL) {
187 cache_hitting = false;
188 cache_node_entry = lv_cache_acquire_or_create(ctx->cache_node_cache, &search_key, NULL);
189 if(cache_node_entry == NULL) {
190 lv_freetype_drop_face_id(ctx, (FTC_FaceID)search_key.pathname);
191 LV_LOG_ERROR("cache node creating failed");
192 return NULL;
193 }
194 }
195
196 lv_freetype_font_dsc_t * dsc = lv_malloc_zeroed(sizeof(lv_freetype_font_dsc_t));
197 LV_ASSERT_MALLOC(dsc);
198
199 dsc->face_id = (FTC_FaceID)search_key.pathname;
200 dsc->render_mode = font_info->render_mode;
201 dsc->context = ctx;
202 dsc->size = font_info->size;
203 dsc->style = font_info->style;
204 dsc->kerning = font_info->kerning;
205 dsc->magic_num = LV_FREETYPE_FONT_DSC_MAGIC_NUM;
206 dsc->cache_node = lv_cache_entry_get_data(cache_node_entry);
207 dsc->cache_node_entry = cache_node_entry;
208
209 if(cache_hitting == false && freetype_on_font_create(dsc, ctx->max_glyph_cnt) == false) {
210 lv_cache_release(ctx->cache_node_cache, dsc->cache_node_entry, NULL);
211 lv_freetype_drop_face_id(ctx, dsc->face_id);
212 lv_free(dsc);
213 return NULL;
214 }
215 freetype_on_font_set_cbs(dsc);
216
217 FT_Face face = dsc->cache_node->face;
218 FT_Error error;
219 if(FT_IS_SCALABLE(face)) {
220 error = FT_Set_Pixel_Sizes(face, 0, font_info->size);
221 }
222 else {
223 LV_LOG_WARN("font is not scalable, selecting available size");
224 error = FT_Select_Size(face, 0);
225 }
226 if(error) {
227 FT_ERROR_MSG("FT_Set_Pixel_Sizes", error);
228 return NULL;
229 }
230
231 if(dsc->kerning != LV_FONT_KERNING_NONE && !dsc->cache_node->face_has_kerning) {
232 LV_LOG_WARN("font: '%s' doesn't have kerning info", pathname);
233 }
234
235 lv_font_t * font = &dsc->font;
236 font->dsc = dsc;
237 font->subpx = LV_FONT_SUBPX_NONE;
238 font->line_height = FT_F26DOT6_TO_INT(face->size->metrics.height);
239 font->base_line = -FT_F26DOT6_TO_INT(face->size->metrics.descender);
240
241 FT_Fixed scale = face->size->metrics.y_scale;
242 int8_t thickness = FT_F26DOT6_TO_INT(FT_MulFix(scale, face->underline_thickness));
243 font->underline_position = FT_F26DOT6_TO_INT(FT_MulFix(scale, face->underline_position));
244 font->underline_thickness = thickness < 1 ? 1 : thickness;
245
246 return font;
247}

263void lv_freetype_font_delete(lv_font_t * font)
264{
265 if(font == NULL) return;
266 lv_freetype_context_t * ctx = lv_freetype_get_context();
267 if(!ctx) {
268 /* Freetype already torn down (e.g. static destruction order). Nothing to release. */
269 return;
270 }
271 lv_freetype_font_dsc_t * dsc = (lv_freetype_font_dsc_t *)(font->dsc);
272 LV_ASSERT_FREETYPE_FONT_DSC(dsc);
273
274 lv_cache_release(ctx->cache_node_cache, dsc->cache_node_entry, NULL);
275 if(lv_cache_entry_get_ref(dsc->cache_node_entry) == 0) {
276 lv_cache_drop(ctx->cache_node_cache, dsc->cache_node, NULL);
277 }
278
279 lv_freetype_drop_face_id(dsc->context, dsc->face_id);
280
281 /* invalidate magic number */
282 lv_memzero(dsc, sizeof(lv_freetype_font_dsc_t));
283 lv_free(dsc);
284}

291void lv_freetype_italic_transform(FT_Face face)
292{
293 LV_ASSERT(face != NULL);
294 FT_Matrix matrix;
295 matrix.xx = FT_INT_TO_F16DOT16(1);
296 matrix.xy = LV_FREETYPE_OBLIQUE_SLANT_DEF;

311static bool freetype_on_font_create(lv_freetype_font_dsc_t * dsc, uint32_t max_glyph_cnt)
312{
313 LV_ASSERT(dsc != NULL);
314 /*
315 * Glyph info uses a small amount of memory, and uses glyph info more frequently,
316 * so it plans to use twice the maximum number of caches here to
317 * get a better info acquisition performance.*/
318 lv_cache_t * glyph_cache = lv_freetype_create_glyph_cache(max_glyph_cnt * 2);
319 if(glyph_cache == NULL) {
320 LV_LOG_ERROR("glyph cache creating failed");
321 return false;
322 }
323 dsc->cache_node->glyph_cache = glyph_cache;
324
325 lv_cache_t * draw_data_cache = NULL;
326 if(dsc->render_mode == LV_FREETYPE_FONT_RENDER_MODE_BITMAP) {
327 draw_data_cache = lv_freetype_create_draw_data_image(max_glyph_cnt);
328 }
329 else if(dsc->render_mode == LV_FREETYPE_FONT_RENDER_MODE_OUTLINE) {
330 draw_data_cache = lv_freetype_create_draw_data_outline(max_glyph_cnt);
331 }
332 else {
333 LV_LOG_ERROR("unknown render mode");
334 return false;
335 }
336
337 if(draw_data_cache == NULL) {
338 LV_LOG_ERROR("draw data cache creating failed");
339 return false;
340 }
341
342 dsc->cache_node->draw_data_cache = draw_data_cache;
343
344 return true;
345}

373static FTC_FaceID lv_freetype_req_face_id(lv_freetype_context_t * ctx, const char * pathname)
374{
375 LV_ASSERT(pathname && pathname[0] != '\0');
376
377 lv_ll_t * ll_p = &ctx->face_id_ll;
378 face_id_node_t * node;
379
380 /* search cache */
381 LV_LL_READ(ll_p, node) {
382 if(strcmp(node->pathname, pathname) == 0) {
383 node->ref_cnt++;
384 LV_LOG_INFO("reuse face_id: %s, ref_cnt = %d", node->pathname, node->ref_cnt);
385 return node->pathname;
386 }
387 }
388
389 /* insert new cache */
390 node = lv_ll_ins_tail(ll_p);
391 LV_ASSERT_MALLOC(node);
392
393#if LV_USE_FS_MEMFS
394 if(pathname[0] == LV_FS_MEMFS_LETTER) {
395#if !LV_FREETYPE_USE_LVGL_PORT
396 LV_LOG_WARN("LV_FREETYPE_USE_LVGL_PORT is not enabled");
397#endif
398 node->pathname = lv_malloc(sizeof(lv_fs_path_ex_t));
399 LV_ASSERT_MALLOC(node->pathname);
400 lv_memcpy(node->pathname, pathname, sizeof(lv_fs_path_ex_t));
401 }
402 else
403#endif
404 {
405 node->pathname = lv_strdup(pathname);
406 LV_ASSERT(node->pathname != NULL);
407 }
408
409 LV_LOG_INFO("add face_id: %s", node->pathname);
410
411 node->ref_cnt = 1;
412 return node->pathname;
413}

415static void lv_freetype_drop_face_id(lv_freetype_context_t * ctx, FTC_FaceID face_id)
416{
417 lv_ll_t * ll_p = &ctx->face_id_ll;
418 face_id_node_t * node;
419 LV_LL_READ(ll_p, node) {
420 if(face_id == node->pathname) {
421 LV_LOG_INFO("found face_id: %s, ref_cnt = %d", node->pathname, node->ref_cnt);
422 node->ref_cnt--;
423 if(node->ref_cnt == 0) {
424 LV_LOG_INFO("drop face_id: %s", node->pathname);
425 lv_ll_remove(ll_p, node);
426 lv_free(node->pathname);
427 lv_free(node);
428 }
429 return;
430 }
431 }
432
433 LV_ASSERT_MSG(false, "face_id not found");
434}

440static bool cache_node_cache_create_cb(lv_freetype_cache_node_t * node, void * user_data)
441{
442 LV_UNUSED(user_data);
443 lv_freetype_context_t * ctx = lv_freetype_get_context();
444
445 /* Cache miss, load face */
446 FT_Face face;
447 FT_Error error = FT_New_Face(ctx->library, node->pathname, 0, &face);
448 if(error) {
449 FT_ERROR_MSG("FT_New_Face", error);
450 return false;
451 }
452
453 node->ref_size = LV_FREETYPE_OUTLINE_REF_SIZE_DEF;
454
455 if(node->style & LV_FREETYPE_FONT_STYLE_ITALIC) {
456 lv_freetype_italic_transform(face);
457 }
458
459 node->face = face;
460 node->face_has_kerning = FT_HAS_KERNING(face);
461 lv_mutex_init(&node->face_lock);
462
463#if LV_FREETYPE_CACHE_FT_GLYPH_L1
464 lv_freetype_glyph_l1_init(node);
465#endif
466
467 /* Apply variable font weight */
468 if(!lv_freetype_set_weight_if_variable(face, node->weight)) {
469 /* Only log when an explicit weight was requested, to avoid spamming logs
470 * for default weights on non-variable fonts. */
471 if(node->weight != LV_FREETYPE_VAR_WEIGHT_NORMAL && node->weight != LV_FREETYPE_VAR_WEIGHT_BOLD) {
472 LV_LOG_INFO("font '%s' does not support variable weight, requested weight %d ignored", node->pathname, node->weight);
473 }
474 }
475
476 return true;
477}

490/* If the font is variable, set the 'wght' axis to the requested value. Returns true if applied. */
491static bool lv_freetype_set_weight_if_variable(FT_Face face, int weight)
492{
493 if(!face)
494 return false;
495 if(!FT_HAS_MULTIPLE_MASTERS(face))
496 return false;
497
498 FT_MM_Var * mm_var = NULL;
499 FT_Error mm_err = FT_Get_MM_Var(face, &mm_var);
500 if(mm_err != 0 || !mm_var || mm_var->num_axis == 0) {
501 if(mm_err != 0) {
502 FT_ERROR_MSG("FT_Get_MM_Var", mm_err);
503 }
504 else if(mm_var) {
505 /* FT_Get_MM_Var succeeded but no usable axes; free to avoid leak */
506 lv_freetype_done_mm_var(mm_var);
507 }
508
509 return false;
510 }
511
512 FT_UInt axis_count = mm_var->num_axis;
513 FT_Fixed coords_stack[LV_FREETYPE_MAX_STACK_AXES];
514 FT_Fixed * coords = coords_stack;
515 const bool use_heap = axis_count > LV_FREETYPE_MAX_STACK_AXES;
516 bool applied = false;
517 FT_ULong wght_tag;
518 int wght_index;
519 FT_UInt i;
520
521 if(use_heap) {
522 coords = (FT_Fixed *)lv_malloc(axis_count * sizeof(FT_Fixed));
523 LV_ASSERT_MALLOC(coords);
524 if(!coords) {
525 LV_LOG_ERROR("failed to allocate memory for %u font axes", axis_count);
526 lv_freetype_done_mm_var(mm_var);
527 return false;
528 }
529 }
530
531 FT_Error coord_err = FT_Get_Var_Design_Coordinates(face, axis_count, coords);
532 if(coord_err != 0) {
533 FT_ERROR_MSG("FT_Get_Var_Design_Coordinates", coord_err);
534 goto cleanup;
535 }
536
537 wght_tag = FT_MAKE_TAG('w', 'g', 'h', 't');
538 wght_index = -1;
539 for(i = 0; i < axis_count; i++) {
540 if(mm_var->axis[i].tag == wght_tag) {
541 wght_index = (int)i;
542 break;
543 }
544 }
545
546 if(wght_index >= 0) {
547 FT_Fixed min_v = mm_var->axis[wght_index].minimum;
548 FT_Fixed max_v = mm_var->axis[wght_index].maximum;
549 FT_Fixed target;
550 FT_Error set_err;
551
552 weight = LV_CLAMP(LV_FREETYPE_VAR_WEIGHT_MIN, weight, LV_FREETYPE_VAR_WEIGHT_MAX);
553 target = LV_CLAMP(min_v, FT_INT_TO_F16DOT16(weight), max_v);
554 coords[wght_index] = target;
555
556 set_err = FT_Set_Var_Design_Coordinates(face, axis_count, coords);
557 if(set_err != 0) {
558 FT_ERROR_MSG("FT_Set_Var_Design_Coordinates", set_err);
559 }
560 else {
561 applied = true;
562 }
563 }
564
565cleanup:
566 if(use_heap) {
567 lv_free(coords);
568 }
569
570 lv_freetype_done_mm_var(mm_var);
571 return applied;
572}

574static void cache_node_cache_free_cb(lv_freetype_cache_node_t * node, void * user_data)
575{

592static lv_cache_compare_res_t cache_node_cache_compare_cb(const lv_freetype_cache_node_t * lhs,
593 const lv_freetype_cache_node_t * rhs)
594{
595 if(lhs->render_mode != rhs->render_mode) {
596 return lhs->render_mode > rhs->render_mode ? 1 : -1;
597 }
598 if(lhs->style != rhs->style) {
599 return lhs->style > rhs->style ? 1 : -1;
600 }
601 if(lhs->weight != rhs->weight) {
602 return lhs->weight > rhs->weight ? 1 : -1;
603 }
604
605 int32_t cmp_res = lv_strcmp(lhs->pathname, rhs->pathname);
606 if(cmp_res != 0) {
607 return cmp_res > 0 ? 1 : -1;
608 }
609
610 return 0;
