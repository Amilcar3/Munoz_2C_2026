144static void time_observer_cb(lv_observer_t * observer, lv_subject_t * subject)
145{
146 int32_t hour = lv_subject_get_int(lv_subject_get_group_element(subject, 0));
147 int32_t minute = lv_subject_get_int(lv_subject_get_group_element(subject, 1));
148 int32_t format = lv_subject_get_int(lv_subject_get_group_element(subject, 2));
149 int32_t am_pm = lv_subject_get_int(lv_subject_get_group_element(subject, 3));
150
151 lv_obj_t * label = (lv_obj_t *) lv_observer_get_target(observer);
152
153 if(format == TIME_FORMAT_24) {
154 lv_label_set_text_fmt(label, "%" LV_PRId32 ":%02" LV_PRId32, hour, minute);
155 }
156 else {
157 lv_label_set_text_fmt(label, "%" LV_PRId32":%02" LV_PRId32" %s", hour + 1, minute, am_pm == TIME_AM ? "am" : "pm");
158 }
159}

161/*Change the hour options on format change*/
162static void hour_roller_options_update(lv_observer_t * observer, lv_subject_t * subject)
163{
164 lv_obj_t * roller = (lv_obj_t *) lv_observer_get_target(observer);
165 int32_t prev_selected = lv_roller_get_selected(roller);
166 int32_t v = lv_subject_get_int(subject);
167 if(v == TIME_FORMAT_12) {
168 prev_selected--;
169 if(prev_selected > 12) prev_selected -= 12;
170 lv_roller_set_options(roller, hour12_options, LV_ROLLER_MODE_NORMAL);
171 }
172 else {
173 prev_selected++;
174 lv_roller_set_options(roller, hour24_options, LV_ROLLER_MODE_NORMAL);
175 }
176
177 lv_roller_set_selected(roller, prev_selected, LV_ANIM_OFF);
178 lv_obj_send_event(roller, LV_EVENT_VALUE_CHANGED, NULL);
179}
