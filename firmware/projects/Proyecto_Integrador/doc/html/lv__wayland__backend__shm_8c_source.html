114static void buffer_release(void * data, struct wl_buffer * wl_buffer)
115{
116 lv_wl_shm_display_data_t * ddata = data;
117 for(size_t i = 0; i < LV_WL_SHM_BUF_COUNT; ++i) {
118 if(wl_buffer == ddata->buffers[i].wl_buffer) {
119 ddata->buffers[i].busy = false;
120 }
121 }
122
123 if(ddata->delete_on_release) {
124 shm_delete_display_data(ddata);
125 }
126}

170static lv_wl_shm_display_data_t * shm_create_display_data(lv_wl_shm_ctx_t * ctx,
171 lv_display_t * display,
172 int32_t width,
173 int32_t height)
174{
175 lv_wl_shm_display_data_t * ddata = lv_zalloc(sizeof(*ddata));
176 if(!ddata) {
177 LV_LOG_ERROR("Failed to allocate data for display");
178 return NULL;
179 }
180
181 const lv_display_rotation_t rotation = lv_display_get_rotation(display);
182 lv_color_format_t cf = lv_display_get_color_format(display);
183 int32_t shm_cf = lv_cf_to_shm_cf(cf);
184
185 if(shm_cf < 0) {
186 LV_LOG_WARN("Unsupported color format %d. Falling back to XRGB8888", cf);
187 cf = LV_COLOR_FORMAT_XRGB8888;
188 lv_display_set_color_format(display, cf);
189 shm_cf = WL_SHM_FORMAT_XRGB8888;
190 }
191 /* safe cast as per check above*/
192 ddata->shm_cf = (uint32_t)shm_cf;
193
194 const bool needs_rotation = rotation != LV_DISPLAY_ROTATION_0;
195 const int32_t phy_width = lv_display_get_original_horizontal_resolution(display);
196 const int32_t phy_height = lv_display_get_original_vertical_resolution(display);
197 const uint32_t phy_stride = lv_draw_buf_width_to_stride(phy_width, cf);
198 const size_t phy_buf_size = phy_stride * phy_height;
199
200 ddata->mmap_size = phy_buf_size * LV_WL_SHM_BUF_COUNT;
201
202 ddata->fd = create_shm_file(ddata->mmap_size);
203 if(ddata->fd < 0) {
204 LV_LOG_ERROR("Failed to create shm file");
205 goto shm_file_err;
206 }
207
208 ddata->mmap_ptr = mmap(NULL, ddata->mmap_size, PROT_READ | PROT_WRITE, MAP_SHARED, ddata->fd, 0);
209 if(ddata->mmap_ptr == MAP_FAILED) {
210 LV_LOG_ERROR("Failed to map shm file: %s", strerror(errno));
211 goto mmap_err;
212 }
213
214 ddata->pool = wl_shm_create_pool(ctx->shm, ddata->fd, ddata->mmap_size);
215 if(!ddata->pool) {
216 LV_LOG_ERROR("Failed to create wl_shm_pool");
217 goto shm_pool_err;
218 }
219
220 for(size_t i = 0; i < LV_WL_SHM_BUF_COUNT; ++i) {
221 size_t offset = i * phy_buf_size;
222 ddata->buffers[i].wl_buffer =
223 wl_shm_pool_create_buffer(ddata->pool, offset, phy_width, phy_height, phy_stride, ddata->shm_cf);
224
225 if(!ddata->buffers[i].wl_buffer) {
226 LV_LOG_ERROR("Failed to create wl_buffer %zu", i);
227 goto pool_buffer_err;
228 }
229 wl_buffer_add_listener(ddata->buffers[i].wl_buffer, &buffer_listener, ddata);
230 ddata->buffers[i].busy = false;
231 }
232
233 if(needs_rotation) {
234 const uint32_t stride = lv_draw_buf_width_to_stride(width, cf);
235 const size_t buf_size = stride * height;
236
237 ddata->rotated_buf = lv_malloc(buf_size);
238 LV_ASSERT_MALLOC(ddata->rotated_buf);
239 if(!ddata->rotated_buf) {
240 LV_LOG_ERROR("Failed to allocate LVGL render buffer");
241 goto rotated_buf_err;
242 }
243
244 lv_display_set_buffers(display, ddata->rotated_buf, NULL,
245 buf_size, LV_DISPLAY_RENDER_MODE_DIRECT);
246 }
247 else {
248 lv_display_set_buffers(display, ddata->mmap_ptr,
249 (uint8_t *)ddata->mmap_ptr + phy_buf_size,
250 phy_buf_size, LV_DISPLAY_RENDER_MODE_DIRECT);
251 }
252
253 return ddata;
254
255 lv_free(ddata->rotated_buf);
256rotated_buf_err:
257pool_buffer_err:
258 wl_shm_pool_destroy(ddata->pool);
259shm_pool_err:
260 munmap(ddata->mmap_ptr, ddata->mmap_size);
261mmap_err:
262 close(ddata->fd);
263shm_file_err:
264 lv_free(ddata);
265 return NULL;
266}

268static void shm_delete_display_data(lv_wl_shm_display_data_t * ddata)
269{
270 for(size_t i = 0; i < LV_WL_SHM_BUF_COUNT; ++i) {
271 lv_wl_buffer_t * buffer = &ddata->buffers[i];
272 if(!buffer->wl_buffer) {
273 continue;
274 }
275 if(buffer->busy) {
276 /* Defer the deletion of this display data until the buffers are released */
277 LV_LOG_INFO("Buffer is still busy, deferring deletion to when its released");
278 ddata->delete_on_release = true;
279 return;
280 }
281 wl_buffer_destroy(ddata->buffers[i].wl_buffer);
282 ddata->buffers[i].wl_buffer = NULL;
283 }
284
285 if(ddata->pool) {
286 wl_shm_pool_destroy(ddata->pool);
287 ddata->pool = NULL;
288 }
289
290
291 if(ddata->mmap_ptr != MAP_FAILED) {
292 munmap(ddata->mmap_ptr, ddata->mmap_size);
293 ddata->mmap_ptr = MAP_FAILED;
294 }
295
296 if(ddata->fd >= 0) {
297 close(ddata->fd);
298 ddata->fd = -1;
299 }
300 if(ddata->rotated_buf) {
301 lv_free(ddata->rotated_buf);
302 ddata->rotated_buf = NULL;
303 }
304
305 LV_LOG_INFO("Deleted buffers and display data");
306 lv_free(ddata);
307}

366static int create_shm_file(size_t size)
367{
368 int fd = -1;
369 char name[255];
370 snprintf(name, sizeof(name), "/lvgl-wayland-%d-%ld", getpid(), (long)lv_tick_get());
371
372 fd = shm_open(name, O_RDWR | O_CREAT | O_EXCL, 0600);
373 if(fd < 0) {
374 LV_LOG_ERROR("shm_open failed: %s", strerror(errno));
375 return -1;
376 }
377
378 shm_unlink(name);
379
380 if(ftruncate(fd, size) < 0) {
381 LV_LOG_ERROR("ftruncate failed: %s", strerror(errno));
382 close(fd);
383 return -1;
384 }
385 return fd;
386}

398static void shm_flush_cb(lv_display_t * disp, const lv_area_t * area, uint8_t * px_map)
399{
400 lv_wl_shm_display_data_t * ddata = lv_wayland_get_backend_display_data(disp);
401 struct wl_surface * surface = lv_wayland_get_window_surface(disp);
402 if(!surface) {
403 lv_display_flush_ready(disp);
404 return;
405 }
406
407 const lv_display_rotation_t rotation = lv_display_get_rotation(disp);
408 const lv_color_format_t cf = lv_display_get_color_format(disp);
409
410 /* When using ARGB8888, the compositor expects premultiplied ARGB8888 so premultiply it here*/
411 if(ddata->shm_cf == WL_SHM_FORMAT_ARGB8888 && cf != LV_COLOR_FORMAT_ARGB8888_PREMULTIPLIED) {
412 const int32_t w = lv_area_get_width(area);
413 const int32_t h = lv_area_get_height(area);
414 const int32_t buf_w = lv_display_get_horizontal_resolution(disp);
415 lv_color32_t * start_pos = (lv_color32_t *)px_map + (area->y1 * buf_w) + area->x1;
416
417 for(int32_t y = 0; y < h; ++y) {
418 size_t index = y * buf_w;
419 for(int32_t x = 0; x < w; ++x) {
420 lv_color_premultiply(start_pos + (index++));
421 }
422 }
423 }
424
425 /* If we have rotation, copy from rotated_buf to Wayland buffer */
426 if(rotation != LV_DISPLAY_ROTATION_0) {
427 const int32_t hor_res = lv_display_get_horizontal_resolution(disp);
428 const int32_t ver_res = lv_display_get_vertical_resolution(disp);
429 const uint32_t src_stride = lv_draw_buf_width_to_stride(hor_res, cf);
430
431 const int32_t phy_width = lv_display_get_original_horizontal_resolution(disp);
432 const int32_t phy_height = lv_display_get_original_vertical_resolution(disp);
433 const uint32_t dest_stride = lv_draw_buf_width_to_stride(phy_width, cf);
434
435 size_t buf_size = dest_stride * phy_height;
436 uint8_t * wl_buf = (uint8_t *)ddata->mmap_ptr + (ddata->curr_wl_buffer_idx * buf_size);
437
438 lv_draw_rotate(ddata->rotated_buf, wl_buf, hor_res, ver_res,
439 src_stride, dest_stride, rotation, cf);
440
441 wl_surface_damage(surface, 0, 0, phy_width, phy_height);
442 }
443 else {
444 const int32_t w = lv_area_get_width(area);
445 const int32_t h = lv_area_get_height(area);
446 wl_surface_damage(surface, area->x1, area->y1, w, h);
447 }
448
449 if(!lv_display_flush_is_last(disp)) {
450 lv_display_flush_ready(disp);
451 return;
452 }
453
454 lv_wl_buffer_t * buffer = &ddata->buffers[ddata->curr_wl_buffer_idx];
455 if(buffer->busy) {
456 LV_LOG_WARN("Failed to acquire a non-busy buffer");
457 }
458
459 struct wl_callback * callback = wl_surface_frame(surface);
460 wl_callback_add_listener(callback, &frame_listener, disp);
461
462 wl_surface_attach(surface, buffer->wl_buffer, 0, 0);
463 wl_surface_commit(surface);
464
465 buffer->busy = true;
466 ddata->curr_wl_buffer_idx = (ddata->curr_wl_buffer_idx + 1) % LV_WL_SHM_BUF_COUNT;
467}
