59lv_display_t * lv_sdl_window_create(int32_t hor_res, int32_t ver_res)
60{
61 if(!inited) {
62#if LV_SDL_USE_EGL && defined(SDL_VIDEO_DRIVER_X11)
63 /* We only support x11 with EGL for now*/
64 SDL_SetHintWithPriority("SDL_VIDEODRIVER", "x11", SDL_HINT_OVERRIDE);
65 SDL_SetHint(SDL_HINT_VIDEO_X11_FORCE_EGL, "1");
66#endif
67 SDL_Init(SDL_INIT_VIDEO);
68 SDL_StartTextInput();
69 event_handler_timer = lv_timer_create(sdl_event_handler, 5, NULL);
70 lv_tick_set_cb(SDL_GetTicks);
71 lv_delay_set_cb(SDL_Delay);
72
73 inited = true;
74 }
75
76 lv_sdl_window_t * dsc = lv_malloc_zeroed(sizeof(lv_sdl_window_t));
77 LV_ASSERT_MALLOC(dsc);
78 if(dsc == NULL) return NULL;
79
80 lv_display_t * disp = lv_display_create(hor_res, ver_res);
81 if(disp == NULL) {
82 lv_free(dsc);
83 return NULL;
84 }
85 lv_display_set_driver_data(disp, dsc);
86 lv_result_t res = window_create(disp);
87 if(res != LV_RESULT_OK) {
88 LV_LOG_ERROR("Failed to initialize window");
89 lv_free(dsc);
90 lv_display_delete(disp);
91 return NULL;
92 }
93
94 lv_display_add_event_cb(disp, release_disp_cb, LV_EVENT_DELETE, disp);
95 lv_display_add_event_cb(disp, res_chg_event_cb, LV_EVENT_RESOLUTION_CHANGED, NULL);
96 lv_display_add_event_cb(disp, res_chg_event_cb, LV_EVENT_COLOR_FORMAT_CHANGED, NULL);
97
98 /*Process the initial events*/
99 sdl_event_handler(NULL);
100
101 return disp;
102}

138lv_display_t * lv_sdl_get_disp_from_win_id(uint32_t win_id)
139{
140 lv_display_t * disp = lv_display_get_next(NULL);
141 if(win_id == UINT32_MAX) return disp;
142
143 while(disp) {
144 lv_sdl_window_t * dsc = lv_display_get_driver_data(disp);
145 if(dsc != NULL && SDL_GetWindowID(dsc->window) == win_id) {
146 return disp;
147 }
148 disp = lv_display_get_next(disp);
149 }
150 return NULL;
151}

241static void sdl_event_handler(lv_timer_t * t)
242{
243 LV_UNUSED(t);
244
245 /*Refresh handling*/
246 SDL_Event event;
247 while(SDL_PollEvent(&event)) {
248 lv_sdl_mouse_handler(&event);
249#if LV_SDL_MOUSEWHEEL_MODE == LV_SDL_MOUSEWHEEL_MODE_ENCODER
250 lv_sdl_mousewheel_handler(&event);
251#endif
252 lv_sdl_keyboard_handler(&event);
253
254 if(event.type == SDL_WINDOWEVENT) {
255 lv_display_t * disp = lv_sdl_get_disp_from_win_id(event.window.windowID);
256 if(disp == NULL) continue;
257 lv_sdl_window_t * dsc = lv_display_get_driver_data(disp);
258 switch(event.window.event) {
259#if SDL_VERSION_ATLEAST(2, 0, 5)
260 case SDL_WINDOWEVENT_TAKE_FOCUS:
261#endif
262 case SDL_WINDOWEVENT_EXPOSED:
263 lv_sdl_backend_ops.redraw(disp);
264 break;
265 case SDL_WINDOWEVENT_RESIZED:
266 dsc->ignore_size_chg = 1;
267 int32_t hres = (int32_t)((float)(event.window.data1) / dsc->zoom);
268 int32_t vres = (int32_t)((float)(event.window.data2) / dsc->zoom);
269 lv_display_set_resolution(disp, hres, vres);
270 dsc->ignore_size_chg = 0;
271 lv_refr_now(disp);
272 break;
273 case SDL_WINDOWEVENT_CLOSE:
274 lv_display_delete(disp);
275 break;
276 default:
277 break;
278 }
279 }
280 if(event.type == SDL_QUIT) {
281 SDL_Quit();
282 lv_deinit();
283 inited = false;
284#if LV_SDL_DIRECT_EXIT
285 exit(0);
286#endif
287 }
288 }
289}

291static lv_result_t window_create(lv_display_t * disp)
292{
293 LV_ASSERT(disp != NULL);
294 lv_sdl_window_t * dsc = lv_display_get_driver_data(disp);
295 LV_ASSERT(dsc != NULL);
296 dsc->zoom = 1.0;
297
298 int flag = 0;
299#if LV_SDL_USE_EGL
300 flag |= SDL_WINDOW_OPENGL;
301#endif
302
303#if LV_SDL_FULLSCREEN
304 flag |= SDL_WINDOW_FULLSCREEN;
305#endif
306
307 int32_t hor_res = (int32_t)((float)(disp->hor_res) * dsc->zoom);
308 int32_t ver_res = (int32_t)((float)(disp->ver_res) * dsc->zoom);
309 dsc->window = SDL_CreateWindow("LVGL Simulator",
310 SDL_WINDOWPOS_UNDEFINED, SDL_WINDOWPOS_UNDEFINED,
311 hor_res, ver_res, flag); /*last param. SDL_WINDOW_BORDERLESS to hide borders*/
312 if(!dsc->window) {
313 LV_LOG_ERROR("Failed to create SDL window: '%s'", SDL_GetError());
314 return LV_RESULT_INVALID;
315 }
316 if(lv_sdl_backend_ops.init_display(disp) != LV_RESULT_OK) {
317 LV_LOG_ERROR("Failed to initialize SDL backend");
318 SDL_DestroyWindow(dsc->window);
319 return LV_RESULT_INVALID;
320 }
321
322 /*Some platforms (e.g. Emscripten) seem to require setting the size again */
323 SDL_SetWindowSize(dsc->window, hor_res, ver_res);
324 return LV_RESULT_OK;
325}

327static void res_chg_event_cb(lv_event_t * e)
328{
329 LV_ASSERT(e != NULL);
330 lv_display_t * disp = lv_event_get_current_target(e);
331 LV_ASSERT(disp != NULL);
332 if(lv_sdl_backend_ops.resize_display(disp) != LV_RESULT_OK) {
333 LV_LOG_WARN("Failed to resize display");
334 return;
335 }
336
337 lv_sdl_window_t * dsc = lv_display_get_driver_data(disp);
338 if(!dsc->ignore_size_chg) {
339 SDL_SetWindowSize(dsc->window,
340 lv_sdl_window_get_horizontal_resolution(disp),
341 lv_sdl_window_get_vertical_resolution(disp));
342 }
343}

346static void release_disp_cb(lv_event_t * e)
347{
348 if(lv_deinit_in_progress) {
349 lv_sdl_quit();
350 }
351 LV_ASSERT(e != NULL);
352 lv_display_t * disp = (lv_display_t *) lv_event_get_user_data(e);
353 LV_ASSERT(disp != NULL);
354 lv_sdl_window_t * dsc = lv_display_get_driver_data(disp);
355 LV_ASSERT(dsc != NULL);
356
357 lv_sdl_backend_ops.deinit_display(disp);
358 SDL_DestroyWindow(dsc->window);
359 lv_free(dsc);
360 lv_display_set_driver_data(disp, NULL);
361}
