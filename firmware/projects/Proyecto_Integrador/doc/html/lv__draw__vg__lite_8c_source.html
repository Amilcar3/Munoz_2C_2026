116static void draw_execute(lv_draw_vg_lite_unit_t * u)
117{
118 lv_draw_task_t * t = u->task_act;
119 lv_layer_t * layer = t->target_layer;
120
121 /* remember draw unit for access to unit's context */
122 t->draw_unit = (lv_draw_unit_t *)u;
123
124 lv_vg_lite_buffer_from_draw_buf(&u->target_buffer, layer->draw_buf);
125
126 /* VG-Lite will output premultiplied image, set the flag correspondingly. */
127 lv_draw_buf_set_flag(layer->draw_buf, LV_IMAGE_FLAGS_PREMULTIPLIED);
128
129 vg_lite_identity(&u->global_matrix);
130 if(layer->buf_area.x1 || layer->buf_area.y1) {
131 vg_lite_translate(-layer->buf_area.x1, -layer->buf_area.y1, &u->global_matrix);
132 }
133
134#if LV_DRAW_TRANSFORM_USE_MATRIX
135 vg_lite_matrix_t layer_matrix;
136 lv_vg_lite_matrix(&layer_matrix, &t->matrix);
137 lv_vg_lite_matrix_multiply(&u->global_matrix, &layer_matrix);
138#endif
139
140 if(vg_lite_query_feature(gcFEATURE_BIT_VG_SCISSOR)) {
141 /* Crop out extra pixels drawn due to scaling accuracy issues */
142 lv_area_t scissor_area = layer->phy_clip_area;
143 lv_area_move(&scissor_area, -layer->buf_area.x1, -layer->buf_area.y1);
144 lv_vg_lite_set_scissor_area(u, &scissor_area);
145 }
146
147 switch(t->type) {
148 case LV_DRAW_TASK_TYPE_LETTER:
149 lv_draw_vg_lite_letter(t, t->draw_dsc, &t->area);
150 break;
151 case LV_DRAW_TASK_TYPE_LABEL:
152 lv_draw_vg_lite_label(t, t->draw_dsc, &t->area);
153 break;
154 case LV_DRAW_TASK_TYPE_FILL:
155 lv_draw_vg_lite_fill(t, t->draw_dsc, &t->area);
156 break;
157 case LV_DRAW_TASK_TYPE_BORDER:
158 lv_draw_vg_lite_border(t, t->draw_dsc, &t->area);
159 break;
160 case LV_DRAW_TASK_TYPE_BOX_SHADOW:
161 lv_draw_vg_lite_box_shadow(t, t->draw_dsc, &t->area);
162 break;
163 case LV_DRAW_TASK_TYPE_IMAGE:
164 lv_draw_vg_lite_img(t, t->draw_dsc, &t->area, false);
165 break;
166 case LV_DRAW_TASK_TYPE_ARC:
167 lv_draw_vg_lite_arc(t, t->draw_dsc, &t->area);
168 break;
169 case LV_DRAW_TASK_TYPE_LINE:
170 lv_draw_line_iterate(t, t->draw_dsc, lv_draw_vg_lite_line);
171 break;
172 case LV_DRAW_TASK_TYPE_LAYER:
173 lv_draw_vg_lite_layer(t, t->draw_dsc, &t->area);
174 break;
175 case LV_DRAW_TASK_TYPE_TRIANGLE:
176 lv_draw_vg_lite_triangle(t, t->draw_dsc);
177 break;
178 case LV_DRAW_TASK_TYPE_MASK_RECTANGLE:
179 lv_draw_vg_lite_mask_rect(t, t->draw_dsc, &t->area);
180 break;
181#if LV_USE_VECTOR_GRAPHIC
182 case LV_DRAW_TASK_TYPE_VECTOR:
183 lv_draw_vg_lite_vector(t, t->draw_dsc);
184 break;
185#endif
186 default:
187 break;
188 }
189
190 lv_vg_lite_flush(u);
191}

193static int32_t draw_dispatch(lv_draw_unit_t * draw_unit, lv_layer_t * layer)
194{
195 lv_draw_vg_lite_unit_t * u = (lv_draw_vg_lite_unit_t *)draw_unit;
196
197 /* Return immediately if it's busy with draw task. */
198 if(u->task_act) {
199 return 0;
200 }
201
202 /* Try to get an ready to draw. */
203 lv_draw_task_t * t = lv_draw_get_available_task(layer, NULL, VG_LITE_DRAW_UNIT_ID);
204
205 /* Return 0 is no selection, some tasks can be supported by other units. */
206 if(!t || t->preferred_draw_unit_id != VG_LITE_DRAW_UNIT_ID) {
207 lv_vg_lite_finish(u);
208 return LV_DRAW_UNIT_IDLE;
209 }
210
211 /* Return if target buffer format is not supported. */
212 if(!lv_vg_lite_is_dest_cf_supported(layer->color_format)) {
213 return LV_DRAW_UNIT_IDLE;
214 }
215
216 void * buf = lv_draw_layer_alloc_buf(layer);
217 if(!buf) {
218 t->state = LV_DRAW_TASK_STATE_FAILED;
219 return LV_DRAW_UNIT_IDLE;
220 }
221
222 t->state = LV_DRAW_TASK_STATE_IN_PROGRESS;
223 u->task_act = t;
224
225 draw_execute(u);
226
227 u->task_act->state = LV_DRAW_TASK_STATE_FINISHED;
228 u->task_act = NULL;
229
230 /*The draw unit is free now. Request a new dispatching as it can get a new task*/
231 lv_draw_dispatch_request();
232
233 return 1;
234}

236static int32_t draw_evaluate(lv_draw_unit_t * draw_unit, lv_draw_task_t * task)
237{
238 LV_UNUSED(draw_unit);
239
240 /* Return if target buffer format is not supported. */
241 const lv_draw_dsc_base_t * base_dsc = task->draw_dsc;
242 if(!lv_vg_lite_is_dest_cf_supported(base_dsc->layer->color_format)) {
243 return -1;
244 }
245
246 switch(task->type) {
247 case LV_DRAW_TASK_TYPE_LETTER:
248 case LV_DRAW_TASK_TYPE_LABEL:
249 case LV_DRAW_TASK_TYPE_FILL:
250 case LV_DRAW_TASK_TYPE_BORDER:
251#if LV_VG_LITE_USE_BOX_SHADOW
252 case LV_DRAW_TASK_TYPE_BOX_SHADOW:
253#endif
254 case LV_DRAW_TASK_TYPE_LAYER:
255 case LV_DRAW_TASK_TYPE_LINE:
256 case LV_DRAW_TASK_TYPE_TRIANGLE:
257 case LV_DRAW_TASK_TYPE_MASK_RECTANGLE:
258
259#if LV_USE_VECTOR_GRAPHIC
260 case LV_DRAW_TASK_TYPE_VECTOR:
261#endif
262 break;
263
264 case LV_DRAW_TASK_TYPE_ARC: {
265 if(!check_arc_is_supported(task->draw_dsc)) {
266 return 0;
267 }
268 }
269 break;
270
271 case LV_DRAW_TASK_TYPE_IMAGE: {
272 if(!check_image_is_supported(task->draw_dsc)) {
273 return 0;
274 }
275 }
276 break;
277
278 default:
279 /*The draw unit is not able to draw this task. */
280 return 0;
281 }
282
283 if(task->preference_score > 80) {
284 /* The draw unit is able to draw this task. */
285 task->preference_score = 80;
286 task->preferred_draw_unit_id = VG_LITE_DRAW_UNIT_ID;
287 }
288
289 return 1;
290}

307static void draw_event_cb(lv_event_t * e)
308{
309 lv_event_code_t code = lv_event_get_code(e);
310
311 switch(code) {
312 case LV_EVENT_CANCEL: {
313#if LV_USE_VECTOR_GRAPHIC
319 lv_draw_vg_lite_unit_t * unit = lv_event_get_current_target(e);
320 lv_cache_drop_all(lv_vg_lite_grad_ctx_get_cache(unit->grad_ctx), NULL);
321 lv_cache_drop_all(unit->stroke_cache, NULL);
322 LV_LOG_INFO("dropt all cache");
323#endif
324 }
325 break;
326 case LV_EVENT_FOCUSED:
327 lv_vg_lite_set_dump_param_enable(true);
328 break;
329 case LV_EVENT_DEFOCUSED:
330 lv_vg_lite_set_dump_param_enable(false);
331 break;
332 case LV_EVENT_HIT_TEST:
333 lv_vg_lite_dump_info();
334 break;
335 default:
336 break;
337 }
338}
