65lv_indev_t * lv_uefi_simple_pointer_indev_create(lv_point_t * display_res)
66{
67 lv_indev_t * indev = NULL;
68 lv_uefi_simple_pointer_context_t * indev_ctx = NULL;
69
70 indev_ctx = lv_calloc(1, sizeof(lv_uefi_simple_pointer_context_t));
71 LV_ASSERT_MALLOC(indev_ctx);
72
73 indev_ctx->signature = SIMPLE_POINTER_INDEV_SIGNATURE;
74
75 if(display_res != NULL) {
76 indev_ctx->display_res.x = display_res->x;
77 indev_ctx->display_res.y = display_res->y;
78 }
79 else {
80 indev_ctx->display_res.x = lv_display_get_horizontal_resolution(lv_display_get_default());
81 indev_ctx->display_res.y = lv_display_get_vertical_resolution(lv_display_get_default());
82 }
83
84 lv_ll_init(&indev_ctx->handles, sizeof(lv_uefi_simple_pointer_handle_context_t));
85
86 indev = lv_indev_create();
87 lv_indev_set_type(indev, LV_INDEV_TYPE_POINTER);
88 lv_indev_set_user_data(indev, indev_ctx);
89 lv_indev_add_event_cb(indev, _simple_pointer_indev_event_cb, LV_EVENT_DELETE, indev);
90 lv_indev_set_read_cb(indev, _simple_pointer_read_cb);
91
92 return indev;
93}

101bool lv_uefi_simple_pointer_indev_add_handle(lv_indev_t * indev, EFI_HANDLE handle)
102{
103 LV_CHECK_ARG(indev != NULL, return false);
104 EFI_SIMPLE_POINTER_PROTOCOL * interface = NULL;
105 lv_uefi_simple_pointer_handle_context_t * handle_ctx = NULL;
106
107 lv_uefi_simple_pointer_context_t * indev_ctx = (lv_uefi_simple_pointer_context_t *)lv_indev_get_user_data(indev);
108 LV_CHECK_ARG(indev_ctx != NULL, return false);
109
110 if(indev_ctx->signature != SIMPLE_POINTER_INDEV_SIGNATURE) return false;
111
112 interface = (EFI_SIMPLE_POINTER_PROTOCOL *)lv_uefi_protocol_open(handle, &_uefi_guid_simple_pointer);
113 if(!_simple_pointer_interface_is_valid(interface)) {
114 lv_uefi_protocol_close(handle, &_uefi_guid_simple_pointer);
115 LV_LOG_WARN("[lv_uefi] The SIMPLE_POINTER interface is not valid.");
116 return false;
117 }
118
119 handle_ctx = (lv_uefi_simple_pointer_handle_context_t *) lv_ll_ins_head(&indev_ctx->handles);
120 LV_ASSERT_MALLOC(handle_ctx);
121
122 handle_ctx->handle = handle;
123 handle_ctx->interface = interface;
124 handle_ctx->pixel_per_step_8.x = (((indev_ctx->display_res.x) << 8) / 50) /
125 interface->Mode->ResolutionX;
126 handle_ctx->pixel_per_step_8.y = (((indev_ctx->display_res.y) << 8) / 50) /
127 interface->Mode->ResolutionY;
128
129 return true;
130}

136void lv_uefi_simple_pointer_indev_add_all(lv_indev_t * indev)
137{
138 LV_CHECK_ARG(indev != NULL, return);
139 EFI_STATUS status;
140 EFI_HANDLE * handles = NULL;
141 UINTN no_handles;
142 UINTN index;
143
144 lv_uefi_simple_pointer_context_t * indev_ctx = (lv_uefi_simple_pointer_context_t *)lv_indev_get_user_data(indev);
145 LV_CHECK_ARG(indev_ctx != NULL, return);
146
147 if(indev_ctx->signature != SIMPLE_POINTER_INDEV_SIGNATURE) return;
148
149 status = gLvEfiBS->LocateHandleBuffer(ByProtocol, &_uefi_guid_simple_pointer, NULL, &no_handles, &handles);
150 if(status != EFI_SUCCESS) {
151 LV_LOG_ERROR("[lv_uefi] LocateHandleBuffer(SIMPLE_POINTER) failed with error code %llx.", status);
152 return;
153 }
154
155 for(index = 0; index < no_handles; index++) {
156 lv_uefi_simple_pointer_indev_add_handle(indev, handles[index]);
157 }
158
159 if(handles != NULL) gLvEfiBS->FreePool(handles);
160}

182static void _simple_pointer_read_cb(lv_indev_t * indev, lv_indev_data_t * data)
183{
184 void * node = NULL;
185
186 lv_uefi_simple_pointer_context_t * indev_ctx = (lv_uefi_simple_pointer_context_t *)lv_indev_get_user_data(indev);
187 LV_ASSERT(indev_ctx != NULL);
188
189 /* Read from all registered devices */
190 for(node = lv_ll_get_head(&indev_ctx->handles); node != NULL; node = lv_ll_get_next(&indev_ctx->handles, node)) {
191 lv_uefi_simple_pointer_handle_context_t * handle_ctx = (lv_uefi_simple_pointer_handle_context_t *) node;
192 bool was_pressed = false;
193
194 _simple_pointer_read(indev_ctx, handle_ctx, &was_pressed);
195
196 data->state |= was_pressed ? LV_INDEV_STATE_PRESSED : LV_INDEV_STATE_RELEASED;
197 }
198
199 /* Sanitize the events position */
200 if(indev_ctx->position.x < 0) {
201 indev_ctx->position.x = 0;
202 }
203 else if(indev_ctx->position.x > indev_ctx->display_res.x - 1) {
204 indev_ctx->position.x = indev_ctx->display_res.x - 1;
205 }
206
207 if(indev_ctx->position.y < 0) {
208 indev_ctx->position.y = 0;
209 }
210 else if(indev_ctx->position.y > indev_ctx->display_res.y - 1) {
211 indev_ctx->position.y = indev_ctx->display_res.y - 1;
212 }
213
214 data->point.x = indev_ctx->position.x;
215 data->point.y = indev_ctx->position.y;
216
217 data->continue_reading = FALSE;
218}

227static bool _simple_pointer_interface_is_valid(const EFI_SIMPLE_POINTER_PROTOCOL * interface)
228{
229 if(interface == NULL) return FALSE;
230 if(interface->Mode == NULL) return FALSE;
231 if(interface->Mode->ResolutionX == 0) return FALSE;
232 if(interface->Mode->ResolutionX >= 256) return FALSE;
233 if(interface->Mode->ResolutionY == 0) return FALSE;
234 if(interface->Mode->ResolutionY >= 256) return FALSE;
235 return TRUE;
236}

247static void _simple_pointer_read(lv_uefi_simple_pointer_context_t * indev_ctx,
248 lv_uefi_simple_pointer_handle_context_t * handle_ctx, bool * was_pressed)
249{
250 EFI_STATUS status;
251 EFI_SIMPLE_POINTER_STATE state;
252 lv_point_t pointer_mov;
253
254 LV_ASSERT(indev_ctx != NULL);
255 LV_ASSERT(handle_ctx != NULL);
256 LV_ASSERT(was_pressed != NULL);
257
258 status = handle_ctx->interface->GetState(
259 handle_ctx->interface,
260 &state);
261 if(status == EFI_NOT_READY) return;
262 if(status != EFI_SUCCESS) {
263 LV_LOG_ERROR("[lv_uefi] GetState failed.");
264 return;
265 }
266
267 pointer_mov.x = (state.RelativeMovementX * handle_ctx->pixel_per_step_8.x) >> 8;
268 pointer_mov.y = (state.RelativeMovementY * handle_ctx->pixel_per_step_8.y) >> 8;
269
270 indev_ctx->position.x += pointer_mov.x;
271 indev_ctx->position.y += pointer_mov.y;
272
273 /* Set the state to pressed if one of the interfaces reports a press */
274 *was_pressed = state.LeftButton;
275}
