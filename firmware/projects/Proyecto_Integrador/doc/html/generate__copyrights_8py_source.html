1#!/usr/bin/env python3
2"""Regenerate COPYRIGHTS.md from the single source of truth.
3
4Component data lives in sbom/third_party.json, which is also used to
5generate the SPDX SBOM (see generate_sbom.py). Edit that JSON, then run this
6script to refresh COPYRIGHTS.md.
7
8Usage:
9 python3 scripts/generate_copyrights.py [--output COPYRIGHTS.md] [--check]
10
11 --check exit non-zero if COPYRIGHTS.md is out of date (for CI).
12"""
13
14import argparse
15import json
16import os
17import sys
18
19REPO_ROOT = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
20DATA_FILE = os.path.join(REPO_ROOT, "sbom", "third_party.json")
21
22HEADER = (
23 "LVGL uses the following third-party libraries.\n"
24 "Each entry lists its SPDX license expression below. The full license text "
25 "is, for most libraries, found in the corresponding library folder (see the "
26 "paths); some libraries only reference their license rather than shipping "
27 "the full text. A machine-readable inventory is available in `sbom/`.\n"
28)
29
30
31def render(data):
32 lines = [HEADER]
33 for comp in data["components"]:
34 lines.append("**%s**" % comp["name"])
35
36 paths = comp.get("paths", [])
37 if len(paths) == 1:
38 lines.append("- Path: %s" % paths[0])
39 elif paths:
40 lines.append("- Path:")
41 for p in paths:
42 lines.append(" - %s" % p)
43
44 sources = comp.get("sources", [])
45 if len(sources) == 1 and not sources[0].get("note"):
46 lines.append("- Source: %s" % sources[0]["url"])
47 elif sources:
48 lines.append("- Source:")
49 for s in sources:
50 suffix = " (*%s*)" % s["note"] if s.get("note") else ""
51 lines.append(" - %s%s" % (s["url"], suffix))
52
53 if comp.get("license"):
54 lines.append("- License: %s" % comp["license"])
55
56 if comp.get("note"):
57 lines.append("- Note: %s" % comp["note"])
58
59 lines.append("") # blank line between entries
60
61 return "\n".join(lines).rstrip() + "\n"
62
63
64def main():
65 parser = argparse.ArgumentParser(description="Regenerate COPYRIGHTS.md.")
66 parser.add_argument("--output", default=os.path.join(REPO_ROOT, "COPYRIGHTS.md"),
67 help="Output file (default: COPYRIGHTS.md).")
68 parser.add_argument("--check", action="store_true",
69 help="Exit non-zero if the output is out of date.")
70 args = parser.parse_args()
71
72 with open(DATA_FILE, encoding="utf-8") as fh:
73 data = json.load(fh)
74 content = render(data)
75
76 if args.check:
77 current = ""
78 if os.path.isfile(args.output):
79 with open(args.output, encoding="utf-8") as fh:
80 current = fh.read()
81 if current != content:
82 print("%s is out of date; run scripts/generate_copyrights.py" % args.output)
83 return 1
84 print("%s is up to date" % args.output)
85 return 0
86
87 with open(args.output, "w", encoding="utf-8") as fh:
88 fh.write(content)
89 print("Wrote %s (%d components)" % (args.output, len(data["components"])))
90 return 0
