93void lv_draw_nema_gfx_init(void)
94{
95 lv_draw_nema_gfx_unit_t * draw_nema_gfx_unit = lv_draw_create_unit(sizeof(lv_draw_nema_gfx_unit_t));
96 /*Initialize NemaGFX*/
97 nema_init();
98
99 lv_draw_buf_nema_gfx_init_handlers();
100
101 draw_nema_gfx_unit->base_unit.dispatch_cb = nema_gfx_dispatch;
102 draw_nema_gfx_unit->base_unit.evaluate_cb = nema_gfx_evaluate;
103 draw_nema_gfx_unit->base_unit.delete_cb = nema_gfx_delete;
104 draw_nema_gfx_unit->base_unit.wait_for_finish_cb = nema_gfx_wait_for_finish;
105 draw_nema_gfx_unit->base_unit.name = "NEMA_GFX";
106
107#if LV_USE_NEMA_VG
108 /*Initialize NemaVG */
109 nema_vg_init(LV_NEMA_GFX_MAX_RESX, LV_NEMA_GFX_MAX_RESY);
110 /* Allocate VG Buffers*/
111 draw_nema_gfx_unit->paint = nema_vg_paint_create();
112 draw_nema_gfx_unit->gradient = nema_vg_grad_create();
113 draw_nema_gfx_unit->path = nema_vg_path_create();
114 /*Initialize Freetype Support*/
115 lv_draw_nema_gfx_label_init(&(draw_nema_gfx_unit->base_unit));
116#endif
117 /*Create GPU Command List*/
118 draw_nema_gfx_unit->cl = nema_cl_create();
119 /*Bind Command List*/
120 nema_cl_bind_circular(&(draw_nema_gfx_unit->cl));
121
122
123#if LV_USE_OS
124 lv_thread_init(&draw_nema_gfx_unit->thread, "nemagfx", LV_DRAW_THREAD_PRIO, nema_gfx_render_thread_cb, 2 * 1024,
125 draw_nema_gfx_unit);
126#endif
127}

160static int32_t nema_gfx_evaluate(lv_draw_unit_t * draw_unit, lv_draw_task_t * task)
161{
162 lv_draw_nema_gfx_unit_t * draw_nema_gfx_unit = (lv_draw_nema_gfx_unit_t *)draw_unit;
163
164 switch(task->type) {
165 case LV_DRAW_TASK_TYPE_LAYER: {
166 if(task->preference_score > 80) {
167 task->preference_score = 80;
168 task->preferred_draw_unit_id = DRAW_UNIT_ID_NEMA_GFX;
169 }
170 return 1;
171 }
172#if LV_USE_NEMA_VG
173 case LV_DRAW_TASK_TYPE_TRIANGLE:
174 case LV_DRAW_TASK_TYPE_ARC:
175 case LV_DRAW_TASK_TYPE_FILL: {
176 if(task->preference_score > 80) {
177 task->preference_score = 80;
178 task->preferred_draw_unit_id = DRAW_UNIT_ID_NEMA_GFX;
179 }
180 return 1;
181 }
182#else
183 case LV_DRAW_TASK_TYPE_FILL: {
184 lv_draw_fill_dsc_t * draw_fill_dsc = (lv_draw_fill_dsc_t *) task->draw_dsc;
185 if((draw_fill_dsc->grad.dir == (lv_grad_dir_t)LV_GRAD_DIR_NONE)) {
186 if(task->preference_score > 80) {
187 task->preference_score = 80;
188 task->preferred_draw_unit_id = DRAW_UNIT_ID_NEMA_GFX;
189 }
190 return 1;
191 }
192 break;
193 }
194 case LV_DRAW_TASK_TYPE_TRIANGLE: {
195 lv_draw_triangle_dsc_t * draw_triangle_dsc = (lv_draw_triangle_dsc_t *) task->draw_dsc;
196 if((draw_triangle_dsc->grad.dir == (lv_grad_dir_t)LV_GRAD_DIR_NONE)) {
197 if(task->preference_score > 80) {
198 task->preference_score = 80;
199 task->preferred_draw_unit_id = DRAW_UNIT_ID_NEMA_GFX;
200 }
201 return 1;
202 }
203 break;
204 }
205#endif
206 case LV_DRAW_TASK_TYPE_IMAGE: {
207 lv_draw_image_dsc_t * draw_image_dsc = (lv_draw_image_dsc_t *) task->draw_dsc;
208 /*Guard for previous NemaGFX Version*/
209#ifndef NEMA_BLOP_RECOLOR
210 if(draw_image_dsc->recolor_opa > LV_OPA_MIN)
211 break;
212#endif
213 const lv_image_dsc_t * img_dsc = draw_image_dsc->src;
214 if(!lv_nemagfx_is_cf_supported(img_dsc->header.cf))
215 break;
216
217 if(draw_image_dsc->blend_mode != LV_BLEND_MODE_SUBTRACTIVE) {
218 if(task->preference_score > 80) {
219 task->preference_score = 80;
220 task->preferred_draw_unit_id = DRAW_UNIT_ID_NEMA_GFX;
221 }
222 return 1;
223 }
224 break;
225 }
226 case LV_DRAW_TASK_TYPE_LABEL: {
227 lv_draw_label_dsc_t * draw_label_dsc = (lv_draw_label_dsc_t *) task->draw_dsc;
228 lv_font_fmt_txt_dsc_t * fdsc = (lv_font_fmt_txt_dsc_t *)(draw_label_dsc->font->dsc);
229 if(fdsc->bitmap_format != LV_FONT_FMT_TXT_COMPRESSED) {
230 if(task->preference_score > 80) {
231 task->preference_score = 80;
232 task->preferred_draw_unit_id = DRAW_UNIT_ID_NEMA_GFX;
233 }
234 return 1;
235 }
236 break;
237 }
238 case LV_DRAW_TASK_TYPE_LINE: {
239 lv_draw_line_dsc_t * draw_line_dsc = (lv_draw_line_dsc_t *) task->draw_dsc;
240 bool is_dashed = (draw_line_dsc->dash_width && draw_line_dsc->dash_gap);
241 if(!is_dashed && !(draw_line_dsc->round_end || draw_line_dsc->round_start)) {
242 if(task->preference_score > 80) {
243 task->preference_score = 80;
244 task->preferred_draw_unit_id = DRAW_UNIT_ID_NEMA_GFX;
245 }
246 return 1;
247 }
248 break;
249 }
250 case LV_DRAW_TASK_TYPE_BORDER: {
251 const lv_draw_border_dsc_t * draw_dsc = (lv_draw_border_dsc_t *) task->draw_dsc;
252 if((!(draw_dsc->side != (lv_border_side_t)LV_BORDER_SIDE_FULL && draw_dsc->radius > 0)) &&
253 (draw_dsc->radius > draw_dsc->width)) {
254 if(task->preference_score > 80) {
255 task->preference_score = 80;
256 task->preferred_draw_unit_id = DRAW_UNIT_ID_NEMA_GFX;
257 }
258 return 1;
259 }
260 break;
261 }
262#if LV_USE_VECTOR_GRAPHIC && LV_USE_NEMA_VG
263 case LV_DRAW_TASK_TYPE_VECTOR: {
264 if(task->preference_score > 80) {
265 task->preference_score = 80;
266 task->preferred_draw_unit_id = DRAW_UNIT_ID_NEMA_GFX;
267 }
268 return 1;
269 }
270#endif
271 case LV_DRAW_TASK_TYPE_BOX_SHADOW:
272 case LV_DRAW_TASK_TYPE_MASK_RECTANGLE:
273 case LV_DRAW_TASK_TYPE_MASK_BITMAP:
274 default:
275 break;
276 }
277 nema_cl_wait(&(draw_nema_gfx_unit->cl));
278 return 0;
279}

281static int32_t nema_gfx_dispatch(lv_draw_unit_t * draw_unit, lv_layer_t * layer)
282{
283 lv_draw_nema_gfx_unit_t * draw_nema_gfx_unit = (lv_draw_nema_gfx_unit_t *) draw_unit;
284
285 /* Return immediately if it's busy with draw task. */
286 if(draw_nema_gfx_unit->task_act)
287 return 0;
288
289 /* Try to get an ready to draw. */
290 lv_draw_task_t * t = lv_draw_get_available_task(layer, NULL, DRAW_UNIT_ID_NEMA_GFX);
291
292 /* Return 0 is no selection, some tasks can be supported by other units. */
293 if(t == NULL || t->preferred_draw_unit_id != DRAW_UNIT_ID_NEMA_GFX)
294 return LV_DRAW_UNIT_IDLE;
295
296 void * buf = lv_draw_layer_alloc_buf(layer);
297 if(buf == NULL) {
298 t->state = LV_DRAW_TASK_STATE_FAILED;
299 return LV_DRAW_UNIT_IDLE;
300 }
301
302 t->state = LV_DRAW_TASK_STATE_IN_PROGRESS;
303 draw_nema_gfx_unit->task_act = t;
304
305#if LV_USE_OS
306 /* Let the render thread work. */
307 if(draw_nema_gfx_unit->inited)
308 lv_thread_sync_signal(&draw_nema_gfx_unit->sync);
309#else
310 nema_gfx_execute_drawing(draw_nema_gfx_unit);
311
312 draw_nema_gfx_unit->task_act->state = LV_DRAW_TASK_STATE_FINISHED;
313 draw_nema_gfx_unit->task_act = NULL;
314
315 /* The draw unit is free now. Request a new dispatching as it can get a new task. */
316 lv_draw_dispatch_request();
317#endif
318
319 return 1;
320}

322static void nema_gfx_execute_drawing(lv_draw_nema_gfx_unit_t * u)
323{
324 lv_draw_task_t * t = u->task_act;
325 /* remember draw unit for access to unit's context */
326 t->draw_unit = (lv_draw_unit_t *)u;
327
328 lv_layer_t * layer = t->target_layer;
329 lv_area_t clipped_area;
330 int32_t x;
331 int32_t y;
332
333 if(!lv_area_intersect(&clipped_area, &t->_real_area, &t->clip_area)) {
334 /* Nothing to do */
335 return;
336 }
337
338 x = 0 - t->target_layer->buf_area.x1;
339 y = 0 - t->target_layer->buf_area.y1;
340
341 lv_area_move(&clipped_area, x, y);
342
343 /* Clean cache */
344 lv_draw_buf_flush_cache(layer->draw_buf, &clipped_area);
345
346 switch(t->type) {
347 case LV_DRAW_TASK_TYPE_FILL:
348 lv_draw_nema_gfx_fill(t, t->draw_dsc, &t->area);
349 break;
350 case LV_DRAW_TASK_TYPE_IMAGE:
351 lv_draw_nema_gfx_img(t, t->draw_dsc, &t->area);
352 break;
353 case LV_DRAW_TASK_TYPE_TRIANGLE:
354 lv_draw_nema_gfx_triangle(t, t->draw_dsc);
355 break;
356 case LV_DRAW_TASK_TYPE_LABEL:
357 lv_draw_nema_gfx_label(t, t->draw_dsc, &t->area);
358 break;
359 case LV_DRAW_TASK_TYPE_LAYER:
360 lv_draw_nema_gfx_layer(t, t->draw_dsc, &t->area);
361 break;
362 case LV_DRAW_TASK_TYPE_LINE:
363 lv_draw_line_iterate(t, t->draw_dsc, lv_draw_nema_gfx_line);
364 break;
365#if LV_USE_NEMA_VG
366 case LV_DRAW_TASK_TYPE_ARC:
367 lv_draw_nema_gfx_arc(t, t->draw_dsc, &t->area);
368 break;
369#endif
370 case LV_DRAW_TASK_TYPE_BORDER:
371 lv_draw_nema_gfx_border(t, t->draw_dsc, &t->area);
372 break;
373#if LV_USE_VECTOR_GRAPHIC && LV_USE_NEMA_VG
374 case LV_DRAW_TASK_TYPE_VECTOR:
375 lv_draw_nema_gfx_vector(t, t->draw_dsc, &t->area);
376 break;
377#endif
378 default:
379 break;
380 }
381
382#if !LV_USE_OS
383 nema_cl_wait(&(u->cl));
384#endif
385
386 /* Invalidate cache */
387 lv_draw_buf_invalidate_cache(layer->draw_buf, &clipped_area);
388}

391{

402 LV_LOG_INFO("Cancel NemaGFX draw thread.");
403 _draw_nema_gfx_unit->exit_status = true;
404
405 if(_draw_nema_gfx_unit->inited)
406 lv_thread_sync_signal(&_draw_nema_gfx_unit->sync);

409
410 return res;
411#endif
412
413#if LV_USE_NEMA_VG == 0 && LV_USE_OS == LV_OS_NONE
414 LV_UNUSED(draw_unit);
415#endif
416 return 0;

420static void nema_gfx_render_thread_cb(void * ptr)
421{
422 lv_draw_nema_gfx_unit_t * u = ptr;
423
424 lv_thread_sync_init(&u->sync);
425 u->inited = true;
426
427 while(1) {
428 /* Wait for sync if there is no task set. */
429 while(u->task_act == NULL) {
430 if(u->exit_status)
431 break;
432
433 lv_thread_sync_wait(&u->sync);
434 }
435
436 if(u->exit_status) {
437 LV_LOG_INFO("Ready to exit NemaGFX draw thread.");
438 break;
439 }
440
441 if(u->task_act) {
442 nema_gfx_execute_drawing(u);
443 }
444 /* Signal the ready state to dispatcher. */
445 u->task_act->state = LV_DRAW_TASK_STATE_FINISHED;
446 /* Cleanup. */
447 u->task_act = NULL;
448
449 /* The draw unit is free now. Request a new dispatching as it can get a new task. */
450 lv_draw_dispatch_request();
451 }
452
453 u->inited = false;
454 lv_thread_sync_delete(&u->sync);
455 LV_LOG_INFO("Exit NemaGFX draw thread.");
456}
