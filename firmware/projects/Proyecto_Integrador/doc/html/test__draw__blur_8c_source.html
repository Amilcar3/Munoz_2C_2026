37static void small_canvas_render(const char * name_sub, lv_color_format_t cf, void * canvas_buf, int32_t blur_radius,
38 uint32_t corner_radius)

40 lv_obj_t * canvas = lv_canvas_create(lv_screen_active());
41 lv_canvas_set_buffer(canvas, canvas_buf, CANVAS_WIDTH, CANVAS_HEIGHT, cf);
42 lv_canvas_fill_bg(canvas, lv_color_hex3(0xccc), LV_OPA_COVER);
43
44 lv_layer_t layer;
45 lv_canvas_init_layer(canvas, &layer);
46
47 /*A label in the background*/
48 lv_draw_label_dsc_t label_dsc;
49 lv_draw_label_dsc_init(&label_dsc);
50 label_dsc.color = lv_palette_main(LV_PALETTE_RED);
51 label_dsc.font = &lv_font_montserrat_14;
52 label_dsc.decor = LV_TEXT_DECOR_UNDERLINE;

55 "Curabitur sed velit sed neque tristique sagittis vel et sapien.";
56
57 lv_area_t label1_coords = {10, 10, CANVAS_WIDTH - 10, CANVAS_HEIGHT - 10};
58 lv_draw_label(&layer, &label_dsc, &label1_coords);
59
60 /*Blur the middle of the canvas*/
61 lv_draw_blur_dsc_t blur_dsc;
62 lv_draw_blur_dsc_init(&blur_dsc);
63 blur_dsc.corner_radius = corner_radius;
64 blur_dsc.blur_radius = blur_radius;
65
66 lv_area_t fill_coords = {25, 20, CANVAS_WIDTH - 25, CANVAS_HEIGHT - 20};
67 lv_draw_blur(&layer, &blur_dsc, &fill_coords);

70 lv_draw_fill_dsc_t fill_dsc;
71 lv_draw_fill_dsc_init(&fill_dsc);
72 fill_dsc.color = lv_palette_lighten(LV_PALETTE_BLUE, 1);
73 fill_dsc.radius = corner_radius;
74 fill_dsc.opa = LV_OPA_30;
75
76 lv_draw_fill(&layer, &fill_dsc, &fill_coords);
77
78 /*Add label on the blurred area*/
79 char buf[128];
80 lv_snprintf(buf, sizeof(buf), "%s\nblur:%d", name_sub, blur_radius);
81
82 lv_draw_label_dsc_init(&label_dsc);
83 label_dsc.color = lv_color_black();
84 label_dsc.font = &lv_font_montserrat_14;
85 label_dsc.align = LV_TEXT_ALIGN_CENTER;
86 label_dsc.text = buf;
87
88 lv_area_t label2_coords = {30, 35, CANVAS_WIDTH - 30, CANVAS_HEIGHT - 35};
89 lv_draw_label(&layer, &label_dsc, &label2_coords);
90
91 lv_canvas_finish_layer(canvas, &layer);
92}
