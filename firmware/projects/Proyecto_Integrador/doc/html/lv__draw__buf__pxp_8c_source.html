66static void _invalidate_cache(const lv_draw_buf_t * draw_buf, const lv_area_t * area)
67{
68 const lv_image_header_t * header = &draw_buf->header;
69 uint32_t stride = header->stride;
70 lv_color_format_t cf = header->cf;
71
72 if(area->y1 == 0) {
73 uint32_t size = stride * lv_area_get_height(area);
74
75 /* Invalidate full buffer. */
76 DEMO_CleanInvalidateCacheByAddr((void *)draw_buf->data, size);
77 return;
78 }
79
80 const uint8_t * buf_u8 = draw_buf->data;
81 /*Cache management requires us to know the cache line size for proper alignment */
82 uint8_t align_bytes = __SCB_DCACHE_LINE_SIZE;
83 uint8_t bits_per_pixel = lv_color_format_get_bpp(cf);
84
85 uint16_t align_pixels = align_bytes * 8 / bits_per_pixel;
86 uint16_t offset_x = 0;
87
88 if(area->x1 >= (int32_t)(area->x1 % align_pixels)) {
89 uint16_t shift_x = area->x1 - (area->x1 % align_pixels);
90
91 offset_x = area->x1 - shift_x;
92 buf_u8 += (shift_x * bits_per_pixel) / 8;
93 }
94
95 if(area->y1) {
96 uint16_t shift_y = area->y1;
97
98 buf_u8 += shift_y * stride;
99 }
100
101 /* Area to clear can start from a different offset in buffer.
102 * Invalidate the area line by line.
103 */
104 uint16_t line_pixels = offset_x + lv_area_get_width(area);
105 uint16_t line_size = (line_pixels * bits_per_pixel) / 8;
106 uint16_t area_height = lv_area_get_height(area);
107
108 for(uint16_t y = 0; y < area_height; y++) {
109 const void * line_addr = buf_u8 + y * stride;
110
111 DEMO_CleanInvalidateCacheByAddr((void *)line_addr, line_size);
112 }
113}
