259void test_calendar_get_pressed_date_parses_single_and_double_digit_days(void)
260{
261 lv_calendar_set_month_shown(g_calendar, 2022, 9);
262
263 lv_obj_t * btnm = lv_calendar_get_btnmatrix(g_calendar);
264 TEST_ASSERT_NOT_NULL(btnm);
265
266 const char * const * map = lv_buttonmatrix_get_map(btnm);
267 TEST_ASSERT_NOT_NULL(map);
268
269 uint32_t btn_id = 0;
270 int32_t single_digit_id = -1;
271 int32_t double_digit_id = -1;
272
273 for(uint32_t i = 0; map[i] != NULL; i++) {
274 /* Skip separators (newline entries) */
275 if(map[i][0] == '\0') break; /* empty string terminates the map */
276 if(map[i][0] == '\n') continue; /* row separator */
277
278 size_t len = strlen(map[i]);
279
280 if(single_digit_id < 0 && len == 1 && map[i][0] == '5') {
281 single_digit_id = (int32_t)btn_id;
282 }
283 if(double_digit_id < 0 && len >= 2 && map[i][0] == '1' && map[i][1] == '5') {
284 double_digit_id = (int32_t)btn_id;
285 }
286
287 btn_id++;
288 }
289
290 TEST_ASSERT_TRUE(single_digit_id >= 0);
291 TEST_ASSERT_TRUE(double_digit_id >= 0);
292
293 lv_calendar_date_t pressed_date;
294
295 lv_buttonmatrix_set_selected_button(btnm, (uint32_t)single_digit_id);
296 lv_result_t res = lv_calendar_get_pressed_date(g_calendar, &pressed_date);
297 TEST_ASSERT_EQUAL(LV_RESULT_OK, res);
298 TEST_ASSERT_EQUAL_UINT8(5, pressed_date.day);
299
300 lv_buttonmatrix_set_selected_button(btnm, (uint32_t)double_digit_id);
301 res = lv_calendar_get_pressed_date(g_calendar, &pressed_date);
302 TEST_ASSERT_EQUAL(LV_RESULT_OK, res);
303 TEST_ASSERT_EQUAL_UINT8(15, pressed_date.day);
304}
