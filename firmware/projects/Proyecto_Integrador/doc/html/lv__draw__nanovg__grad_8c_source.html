42bool lv_nanovg_grad_to_paint(NVGcontext * ctx, const lv_vector_gradient_t * grad, NVGpaint * paint)
43{
44 LV_PROFILER_DRAW_BEGIN;
45
46 LV_ASSERT_NULL(grad);
47 LV_ASSERT_NULL(paint);
48
49 if(grad->stops_count < 2) {
50 LV_LOG_WARN("stops_count(%d) should be 2 for gradient", grad->stops_count);
51 LV_PROFILER_DRAW_END;
52 return false;
53 }
54
55 const NVGcolor icol = lv_nanovg_color_convert(grad->stops[0].color, grad->stops[0].opa);
56 const NVGcolor ocol = lv_nanovg_color_convert(grad->stops[1].color, grad->stops[1].opa);
57
58 switch(grad->style) {
59 case LV_VECTOR_GRADIENT_STYLE_LINEAR:
60 *paint = nvgLinearGradient(ctx, grad->x1, grad->y1, grad->x2, grad->y2, icol, ocol);
61 break;
62
63 case LV_VECTOR_GRADIENT_STYLE_RADIAL: {
64 const float inr = grad->cr * grad->stops[0].frac / 255;
65 const float outr = grad->cr * grad->stops[1].frac / 255;
66 *paint = nvgRadialGradient(ctx, grad->cx, grad->cy, inr, outr, icol, ocol);
67 }
68 break;
69
70 default:
71 LV_LOG_WARN("Unsupported gradient style: %d", grad->style);
72 LV_PROFILER_DRAW_END;
73 return false;
74 }
75
76 LV_PROFILER_DRAW_END;
77 return true;
78}

80void lv_nanovg_draw_grad(
81 NVGcontext * ctx,
82 const lv_vector_gradient_t * grad,
83 enum NVGwinding winding,
84 enum NVGcompositeOperation composite_operation)
85{
86 LV_PROFILER_DRAW_BEGIN;
87
88 NVGpaint paint;
89 if(!lv_nanovg_grad_to_paint(ctx, grad, &paint)) {
90 LV_PROFILER_DRAW_END;
91 return;
92 }
93
94 nvgPathWinding(ctx, winding);
95 nvgGlobalCompositeOperation(ctx, composite_operation);
96 nvgFillPaint(ctx, paint);
97 nvgFill(ctx);
98
99 LV_PROFILER_DRAW_END;
100}

102void lv_nanovg_draw_grad_helper(
103 NVGcontext * ctx,
104 const lv_area_t * area,
105 const lv_grad_dsc_t * grad_dsc,
106 enum NVGwinding winding,
107 enum NVGcompositeOperation composite_operation)
108{
109 LV_ASSERT_NULL(ctx);
110 LV_ASSERT_NULL(area);
111 LV_ASSERT_NULL(grad_dsc);
112
113 lv_vector_gradient_t grad;
114 lv_memzero(&grad, sizeof(grad));
115
116 grad.style = LV_VECTOR_GRADIENT_STYLE_LINEAR;
117 grad.stops_count = grad_dsc->stops_count;
118 lv_memcpy(grad.stops, grad_dsc->stops, sizeof(lv_grad_stop_t) * grad_dsc->stops_count);
119
120 /*convert to spread mode*/
121 switch(grad_dsc->extend) {
122 case LV_GRAD_EXTEND_PAD:
123 grad.spread = LV_VECTOR_GRADIENT_SPREAD_PAD;
124 break;
125 case LV_GRAD_EXTEND_REPEAT:
126 grad.spread = LV_VECTOR_GRADIENT_SPREAD_REPEAT;
127 break;
128 case LV_GRAD_EXTEND_REFLECT:
129 grad.spread = LV_VECTOR_GRADIENT_SPREAD_REFLECT;
130 break;
131 default:
132 LV_LOG_WARN("Unsupported gradient extend mode: %d", grad_dsc->extend);
133 grad.spread = LV_VECTOR_GRADIENT_SPREAD_PAD;
134 break;
135 }
136
137 switch(grad_dsc->dir) {
138 case LV_GRAD_DIR_VER:
139 grad.x1 = area->x1;
140 grad.y1 = area->y1;
141 grad.x2 = area->x1;
142 grad.y2 = area->y2 + 1;
143 break;
144
145 case LV_GRAD_DIR_HOR:
146 grad.x1 = area->x1;
147 grad.y1 = area->y1;
148 grad.x2 = area->x2 + 1;
149 grad.y2 = area->y1;
150 break;
151
152 case LV_GRAD_DIR_LINEAR: {
153 int32_t w = lv_area_get_width(area);
154 int32_t h = lv_area_get_height(area);
155
156 grad.x1 = lv_pct_to_px(grad_dsc->params.linear.start.x, w) + area->x1;
157 grad.y1 = lv_pct_to_px(grad_dsc->params.linear.start.y, h) + area->y1;
158 grad.x2 = lv_pct_to_px(grad_dsc->params.linear.end.x, w) + area->x1;
159 grad.y2 = lv_pct_to_px(grad_dsc->params.linear.end.y, h) + area->y1;
160 }
161 break;
162
163 case LV_GRAD_DIR_RADIAL: {
164 grad.style = LV_VECTOR_GRADIENT_STYLE_RADIAL;
165 int32_t w = lv_area_get_width(area);
166 int32_t h = lv_area_get_height(area);
167
168 grad.cx = lv_pct_to_px(grad_dsc->params.radial.focal.x, w) + area->x1;
169 grad.cy = lv_pct_to_px(grad_dsc->params.radial.focal.y, h) + area->y1;
170 int32_t end_extent_x = lv_pct_to_px(grad_dsc->params.radial.end_extent.x, w) + area->x1;
171 int32_t end_extent_y = lv_pct_to_px(grad_dsc->params.radial.end_extent.y, h) + area->y1;
172 grad.cr = LV_MAX(end_extent_x - grad.cx, end_extent_y - grad.cy);
173 }
174 break;
175
176 default:
177 LV_LOG_WARN("Unsupported gradient direction: %d", grad_dsc->dir);
178 return;
179 }
180
181 lv_nanovg_draw_grad(ctx, &grad, winding, composite_operation);
182}
