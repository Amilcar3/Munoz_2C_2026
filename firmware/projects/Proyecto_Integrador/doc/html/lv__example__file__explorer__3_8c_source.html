27static void sort_by_file_kind(lv_obj_t * tb, int16_t lo, int16_t hi)
28{
29 if(lo >= hi) return;
30
31 int16_t lt = lo;
32 int16_t i = lo + 1;
33 int16_t gt = hi;
34 const char * v = lv_table_get_cell_value(tb, lo, 1);
35 while(i <= gt) {
36 if(strcmp(lv_table_get_cell_value(tb, i, 1), v) < 0)
37 exch_table_item(tb, lt++, i++);
38 else if(strcmp(lv_table_get_cell_value(tb, i, 1), v) > 0)
39 exch_table_item(tb, i, gt--);
40 else
41 i++;
42 }
43
44 sort_by_file_kind(tb, lo, lt - 1);
45 sort_by_file_kind(tb, gt + 1, hi);
46}
