106 def _read_image(self, strict: bool = False) -> Optional[Image.Image]:
107 """Read buffer data and convert to PIL Image.
108
109 Args:
110 strict: If True, raise on any size mismatch instead of ignoring.
111
112 Returns:
113 PIL.Image or None on failure.
114 """
115 _require_imaging()
116 header = self.super_value("header")
117 stride = int(header["stride"])
118 height = int(header["h"])
119 cf_info = self.color_format_info()
120 data_ptr = self.super_value("data")
121 data_size = int(self.super_value("data_size"))
122 # The width is in the header. Deriving it from the stride instead turns
123 # any row padding into extra pixels, which shears the whole image.
124 width = int(header["w"])
125 bpp = cf_info["bpp"]
126 row_bytes = (width * bpp + 7) // 8 if bpp else 0
127 if not stride:
128 stride = row_bytes
129 expected_data_size = stride * height
130
131 if not data_ptr or width <= 0 or height <= 0:
132 if strict:
133 raise ValueError(f"Invalid buffer: ptr={data_ptr}, {width}x{height}")
134 return None
135 if data_size < expected_data_size:
136 if strict:
137 raise ValueError(
138 f"Data too small: expected at least {expected_data_size},"
139 f" got {data_size}"
140 )
141 return None
142 if data_size > expected_data_size and strict:
143 gdb.write(
144 f"Warning: data_size {data_size} exceeds expected"
145 f" {expected_data_size}, extra bytes will be ignored\n"
146 )
147
148 pixel_data = (
149 gdb.selected_inferior()
150 .read_memory(int(data_ptr), expected_data_size)
151 .tobytes()
152 )
153 if stride > row_bytes:
154 # _convert_to_image expects rows packed back to back, so drop the
155 # padding at the end of each row.
156 pixel_data = b"".join(
157 pixel_data[y * stride:y * stride + row_bytes]
158 for y in range(height)
159 )
160 return self._convert_to_image(pixel_data, width, height, cf_info["value"])

162 def data_dump(self, filename: str, format: str = None) -> bool:

215 def _convert_to_image(
216 self, pixel_data: bytes, width: int, height: int, color_format: int
217 ) -> Optional[Image.Image]:
218 """
219 Convert raw pixel data to PIL Image based on color format.
220
221 Args:
222 pixel_data: Raw pixel bytes
223 width: Image width
224 height: Image height
225 color_format: LVGL color format value
226
227 Returns:
228 PIL.Image or None if conversion fails
229 """
230 _require_imaging()
231 try:
232 if color_format == self._color_formats["RGB565"]:
233 # Convert RGB565 to RGB888
234 arr = np.frombuffer(pixel_data, dtype=np.uint8)
235 arr = arr.reshape((height, width, 2))
236 rgb565 = np.frombuffer(pixel_data, dtype=np.uint16).reshape(
237 (height, width)
238 )
239 r = (((rgb565 & 0xF800) >> 11) << 3).astype(np.uint8)
240 g = ((rgb565 & 0x07E0) >> 3).astype(np.uint8)
241 b = ((rgb565 & 0x001F) << 3).astype(np.uint8)
242 return Image.fromarray(np.dstack((r, g, b)), "RGB")
243
244 elif color_format == self._color_formats["RGB888"]:
245 arr = np.frombuffer(pixel_data, dtype=np.uint8).reshape(
246 height, width, 3
247 )
248 rgb_arr = arr[:, :, [2, 1, 0]] # BGR -> RGB
249 return Image.fromarray(rgb_arr, "RGB")
250
251 elif color_format == self._color_formats["ARGB8888"]:
252 # lv_color32_t memory layout: [blue, green, red, alpha]
253 arr = np.frombuffer(pixel_data, dtype=np.uint8).reshape(-1, 4)
254 rgba = arr[:, [2, 1, 0, 3]] # BGRA -> RGBA
255 return Image.frombytes("RGBA", (width, height), rgba.tobytes())
256
257 elif color_format == self._color_formats["XRGB8888"]:
258 arr = np.frombuffer(pixel_data, dtype=np.uint8).reshape(-1, 4)
259 rgb = arr[:, [2, 1, 0]] # BGR -> RGB
260 return Image.fromarray(rgb.reshape(height, width, 3), "RGB")
261
262 raise ValueError(f"Unsupported color format: 0x{color_format:x}")
263
264 except Exception as e:
265 print(f"Conversion error: {str(e)}")
266 return None
