48void lv_draw_nanovg_border(lv_draw_task_t * t, const lv_draw_border_dsc_t * dsc, const lv_area_t * coords)
49{
50 LV_PROFILER_DRAW_BEGIN;
51
52 lv_draw_nanovg_unit_t * u = (lv_draw_nanovg_unit_t *)t->draw_unit;
53
54 lv_area_t clip_area;
55 if(!lv_area_intersect(&clip_area, coords, &t->clip_area)) {
56 LV_PROFILER_DRAW_END;
57 return;
58 }
59
60 int32_t w = lv_area_get_width(coords);
61 int32_t h = lv_area_get_height(coords);
62 float r_out = dsc->radius;
63 if(dsc->radius) {
64 float r_short = LV_MIN(w, h) / 2.0f;
65 r_out = LV_MIN(r_out, r_short);
66 }
67
68 nvgBeginPath(u->vg);
69
70 /* outer rect */
71 lv_nanovg_path_append_rect(u->vg,
72 coords->x1, coords->y1,
73 w, h,
74 r_out);
75
76 /* inner rect */
77 enum NVGwinding winding = path_append_inner_rect(u->vg, dsc, coords->x1, coords->y1, w, h, r_out);
78
79 lv_nanovg_fill(
80 u->vg,
81 winding,
82 NVG_SOURCE_OVER,
83 lv_nanovg_color_convert(dsc->color, dsc->opa));
84
85 LV_PROFILER_DRAW_END;
86}

92static enum NVGwinding path_append_inner_rect(NVGcontext * ctx,
93 const lv_draw_border_dsc_t * dsc,
94 int32_t x, int32_t y, int32_t w, int32_t h,
95 float r)
96{
97 LV_PROFILER_DRAW_BEGIN;
98
99 const float half_w = w / 2.0f;
100 const float half_h = h / 2.0f;
101 const int32_t border_w = dsc->width;
102 const float border_w_max = LV_MIN(half_w, half_h);
103
104 /* normal fill, no inner rect */
105 if(border_w >= border_w_max) {
106 LV_PROFILER_DRAW_END;
107 return NVG_CCW;
108 }
109
110 const float r_in = r - border_w;
111
112 /* full border, simple rect */
113 if(dsc->side == LV_BORDER_SIDE_FULL) {
114 lv_nanovg_path_append_rect(ctx,
115 x + border_w, y + border_w,
116 w - border_w * 2, h - border_w * 2,
117 r_in < 0 ? 0 : r_in);
118 LV_PROFILER_DRAW_END;
119 return NVG_CW;
120 }
121
122 /* reset outer rect path */
123 nvgBeginPath(ctx);
124
125 /* no-radius case */
126 if(dsc->radius <= 0) {
127 if(dsc->side & LV_BORDER_SIDE_TOP) {
128 lv_nanovg_path_append_rect(ctx,
129 x,
130 y,
131 w,
132 border_w,
133 0);
134 }
135 if(dsc->side & LV_BORDER_SIDE_LEFT) {
136 lv_nanovg_path_append_rect(ctx,
137 x,
138 y,
139 border_w,
140 h,
141 0);
142 }
143 if(dsc->side & LV_BORDER_SIDE_BOTTOM) {
144 lv_nanovg_path_append_rect(ctx,
145 x,
146 y + h - border_w,
147 w,
148 border_w,
149 0);
150 }
151 if(dsc->side & LV_BORDER_SIDE_RIGHT) {
152 lv_nanovg_path_append_rect(ctx,
153 x + w - border_w,
154 y,
155 border_w,
156 h,
157 0);
158 }
159
160 LV_PROFILER_DRAW_END;
161 return NVG_CCW;
162 }
163
164 /* coordinate reference map: https://github.com/lvgl/lvgl/pull/6796 */
165 const float c1_x = x + r;
166 const float c1_y = y + r;
167 const float c2_x = x + w - r;
168 const float c2_y = c1_y;
169 const float c3_x = c2_x;
170 const float c3_y = y + h - r;
171 const float c4_x = c1_x;
172 const float c4_y = c3_y;
173
174 /* When border_w > r, No need to calculate the intersection of the arc and the line */
175 if(r_in <= 0) {
176 const float p1_x = x;
177 const float p1_y = y + border_w;
178 const float p2_x = x;
179 const float p2_y = y + r;
180 const float p3_x = x + r;
181 const float p3_y = y;
182 const float p4_x = x + border_w;
183 const float p4_y = y;
184
185 const float p5_x = x + w - border_w;
186 const float p5_y = y;
187 const float p6_x = x + w - r;
188 const float p6_y = y;
189 const float p7_x = x + w;
190 const float p7_y = y + r;
191 const float p8_x = x + w;
192 const float p8_y = y + border_w;
193
194 const float p9_x = x + w;
195 const float p9_y = y + h - border_w;
196 const float p10_x = x + w;
197 const float p10_y = y + h - r;
198 const float p11_x = x + w - r;
199 const float p11_y = y + h;
200 const float p12_x = x + w - border_w;
201 const float p12_y = y + h;
202
203 const float p13_x = x + border_w;
204 const float p13_y = y + h;
205 const float p14_x = x + r;
206 const float p14_y = y + h;
207 const float p15_x = x;
208 const float p15_y = y + h - r;
209 const float p16_x = x;
210 const float p16_y = y + h - border_w;
211
212 if(dsc->side & LV_BORDER_SIDE_BOTTOM) {
213 nvgMoveTo(ctx, p16_x, p16_y);
214 nvgLineTo(ctx, p9_x, p9_y);
215 nvgLineTo(ctx, p10_x, p10_y);
216 lv_nanovg_path_append_arc_right_angle(ctx, p10_x, p10_y, c3_x, c3_y, p11_x, p11_y);
217 nvgLineTo(ctx, p14_x, p14_y);
218 lv_nanovg_path_append_arc_right_angle(ctx, p14_x, p14_y, c4_x, c4_y, p15_x, p15_y);
219 nvgClosePath(ctx);
220 }
221
222 if(dsc->side & LV_BORDER_SIDE_TOP) {
223 nvgMoveTo(ctx, p1_x, p1_y);
224 nvgLineTo(ctx, p2_x, p2_y);
225 lv_nanovg_path_append_arc_right_angle(ctx, p2_x, p2_y, c1_x, c1_y, p3_x, p3_y);
226 nvgLineTo(ctx, p6_x, p6_y);
227 lv_nanovg_path_append_arc_right_angle(ctx, p6_x, p6_y, c2_x, c2_y, p7_x, p7_y);
228 nvgLineTo(ctx, p8_x, p8_y);
229 nvgClosePath(ctx);
230 }
231
232 if(dsc->side & LV_BORDER_SIDE_LEFT) {
233 nvgMoveTo(ctx, p4_x, p4_y);
234 nvgLineTo(ctx, p13_x, p13_y);
235 nvgLineTo(ctx, p14_x, p14_y);
236 lv_nanovg_path_append_arc_right_angle(ctx, p14_x, p14_y, c4_x, c4_y, p15_x, p15_y);
237 nvgLineTo(ctx, p2_x, p2_y);

246 nvgLineTo(ctx, p10_x, p10_y);
247 lv_nanovg_path_append_arc_right_angle(ctx, p10_x, p10_y, c3_x, c3_y, p11_x, p11_y);
248 nvgLineTo(ctx, p12_x, p12_y);
249 nvgClosePath(ctx);
250 }

252 LV_PROFILER_DRAW_END;

256 /* When border_w < r, Calculate the intersection of an arc and a line */
257
258 /* r^2 - r_in^2 = offset^2 */
259 const float offset = NVG_MATH_SQRTF((2 * r - border_w) * border_w);
260 const float sweep_alpha = NVG_MATH_DEGREES(NVG_MATH_ACOSF(r_in / r));
261 const float sweep_beta = 90 - sweep_alpha;

312 nvgMoveTo(ctx, p21_x, p21_y);

323 nvgLineTo(ctx, p8_x, p8_y);
324 lv_nanovg_path_append_arc(ctx, c2_x, c2_y, r, 270, sweep_alpha, false);
