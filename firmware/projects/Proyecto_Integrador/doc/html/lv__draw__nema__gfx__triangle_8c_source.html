42void lv_draw_nema_gfx_triangle(lv_draw_task_t * t, const lv_draw_triangle_dsc_t * dsc)
43{
44 if(dsc->opa <= LV_OPA_MIN) return;

46 lv_draw_nema_gfx_unit_t * draw_nema_gfx_unit = (lv_draw_nema_gfx_unit_t *)t->draw_unit;
47
48 lv_layer_t * layer = t->target_layer;
49
50 lv_area_t rel_clip_area;
51 rel_clip_area = t->clip_area;
52 lv_area_move(&rel_clip_area, -layer->buf_area.x1, -layer->buf_area.y1);
53
54 lv_area_t coords;
55 coords.x1 = (int32_t)LV_MIN3(dsc->p[0].x, dsc->p[1].x, dsc->p[2].x);
56 coords.y1 = (int32_t)LV_MIN3(dsc->p[0].y, dsc->p[1].y, dsc->p[2].y);
57 coords.x2 = (int32_t)LV_MAX3(dsc->p[0].x, dsc->p[1].x, dsc->p[2].x);
58 coords.y2 = (int32_t)LV_MAX3(dsc->p[0].y, dsc->p[1].y, dsc->p[2].y);
59
60 lv_area_move(&coords, -layer->buf_area.x1, -layer->buf_area.y1);
61
62 lv_area_t clipped_coords;
63 if(!lv_area_intersect(&clipped_coords, &coords, &rel_clip_area))
64 return; /* Fully clipped, nothing to do */
65
66 nema_set_clip(rel_clip_area.x1, rel_clip_area.y1, lv_area_get_width(&rel_clip_area),
67 lv_area_get_height(&rel_clip_area));
68
69 lv_color_format_t dst_cf = layer->draw_buf->header.cf;
70 uint32_t dst_nema_cf = lv_nemagfx_cf_to_nema(dst_cf);
71
72 /* the stride should be computed internally for NEMA_TSC images and images missing a stride value */
73 int32_t stride = (dst_cf >= LV_COLOR_FORMAT_NEMA_TSC_START && dst_cf <= LV_COLOR_FORMAT_NEMA_TSC_END) ?
74 -1 : lv_area_get_width(&(layer->buf_area)) * lv_color_format_get_size(dst_cf);
75
76 nema_bind_dst_tex((uintptr_t)NEMA_VIRT2PHYS(layer->draw_buf->data), lv_area_get_width(&(layer->buf_area)),
77 lv_area_get_height(&(layer->buf_area)), dst_nema_cf, stride);
78
79 if(dsc->grad.dir == (lv_grad_dir_t)LV_GRAD_DIR_NONE) {
80
81 lv_color32_t col32 = lv_color_to_32(dsc->color, dsc->opa);
82
83 if(col32.alpha < 255U) {
84 nema_set_blend_fill(NEMA_BL_SIMPLE);
85 }
86 else {
87 nema_set_blend_fill(NEMA_BL_SRC);
88 }
89
90 uint32_t bg_color = nema_rgba(col32.red, col32.green, col32.blue, col32.alpha);
91
92 nema_enable_aa(1, 1, 1, 0);
93 nema_fill_triangle(dsc->p[0].x - layer->buf_area.x1, dsc->p[0].y - layer->buf_area.y1,
94 dsc->p[1].x - layer->buf_area.x1, dsc->p[1].y - layer->buf_area.y1,
95 dsc->p[2].x - layer->buf_area.x1, dsc->p[2].y - layer->buf_area.y1, bg_color);
96 }
97#if LV_USE_NEMA_VG
98 else {
99
100 nema_vg_path_clear(draw_nema_gfx_unit->path);
101 nema_vg_paint_clear(draw_nema_gfx_unit->paint);
102
103 nema_vg_paint_set_type(draw_nema_gfx_unit->paint, NEMA_VG_PAINT_GRAD_LINEAR);
104 nema_vg_set_blend(NEMA_BL_SRC_OVER | NEMA_BLOP_SRC_PREMULT);
105 nema_vg_set_fill_rule(NEMA_VG_FILL_EVEN_ODD);
106
107
108 float stops[LV_GRADIENT_MAX_STOPS];
109 color_var_t colors[LV_GRADIENT_MAX_STOPS];
110
111 uint32_t cnt = LV_MAX(dsc->grad.stops_count, LV_GRADIENT_MAX_STOPS);
112
113 for(uint8_t i = 0; i < cnt; i++) {
114 stops[i] = (float)(dsc->grad.stops[i].frac) / 255.f;
115 colors[i].a = dsc->grad.stops[i].opa;
116 colors[i].r = dsc->grad.stops[i].color.red;
117 colors[i].g = dsc->grad.stops[i].color.green;
118 colors[i].b = dsc->grad.stops[i].color.blue;
119 }
120
121 nema_vg_grad_set(draw_nema_gfx_unit->gradient, cnt, stops, colors);
122
123 //Calculate Bounding Box
124 float min_x = LV_MIN(dsc->p[0].x, dsc->p[1].x);
125 min_x = LV_MIN(dsc->p[2].x, min_x);
126
127 float min_y = LV_MIN(dsc->p[0].y, dsc->p[1].y);
128 min_y = LV_MIN(dsc->p[2].y, min_y);
129
130 float max_x = LV_MAX(dsc->p[0].x, dsc->p[1].x);
131 max_x = LV_MAX(dsc->p[2].x, max_x);
132
133 float max_y = LV_MAX(dsc->p[0].y, dsc->p[1].y);
134 max_y = LV_MAX(dsc->p[2].y, max_y);
135
136 float x0, x1, y0, y1;
137
138 if(dsc->grad.dir == LV_GRAD_DIR_HOR) {
139 x0 = min_x;
140 x1 = max_x;
141 y0 = min_y;
142 y1 = min_y;
143 }
144 else {
145 x0 = min_x;
146 x1 = min_x;
147 y0 = min_y;
148 y1 = max_y;
149 }
150
151 y0 -= (float) layer->buf_area.y1;
152 y1 -= (float) layer->buf_area.y1;
153 x0 -= (float) layer->buf_area.x1;
154 x1 -= (float) layer->buf_area.x1;
155
156 uint8_t cmds_polygon[] = {NEMA_VG_PRIM_MOVE, NEMA_VG_PRIM_POLYGON};
157 float triangle_coords[] = {dsc->p[0].x - layer->buf_area.x1, dsc->p[0].y - layer->buf_area.y1,
158 4.0f,
159 dsc->p[1].x - layer->buf_area.x1, dsc->p[1].y - layer->buf_area.y1,
160 dsc->p[2].x - layer->buf_area.x1, dsc->p[2].y - layer->buf_area.y1
161 };
162 nema_enable_aa(0, 0, 0, 0);
163 nema_vg_paint_set_grad_linear(draw_nema_gfx_unit->paint, draw_nema_gfx_unit->gradient, x0, y0, x1, y1,
164 NEMA_TEX_CLAMP | NEMA_FILTER_BL);
165 nema_vg_path_set_shape(draw_nema_gfx_unit->path, 2, cmds_polygon, 7, triangle_coords);
166 nema_vg_draw_path(draw_nema_gfx_unit->path, draw_nema_gfx_unit->paint);
167 }
168#endif
169
170 nema_cl_submit(&(draw_nema_gfx_unit->cl));
171
172}
