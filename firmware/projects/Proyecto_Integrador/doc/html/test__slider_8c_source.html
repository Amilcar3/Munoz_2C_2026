683void test_slider_knob_follows_drag_while_parent_is_scrolled(void)
684{
685 /* A slider dragged to its end doesn't change value, so it keeps its vertical scroll
686 * chain and a vertical move can make the parent the indev's scroll object.
687 * The knob must keep following the pointer anyway because the drag is already ongoing. */
688 lv_obj_t * cont = lv_obj_create(active_screen);
689 lv_obj_set_size(cont, 300, 160);
690 lv_obj_center(cont);
691 lv_obj_set_scroll_dir(cont, LV_DIR_VER);
692 lv_obj_set_style_pad_all(cont, 0, 0);
693 /* No momentum so the scroll ends with the release */
694 lv_obj_set_scroll_momentum(cont, false);
695
696 lv_obj_t * slider_in_cont = lv_slider_create(cont);
697 lv_obj_set_size(slider_in_cont, 200, 20);
698 lv_obj_set_pos(slider_in_cont, 0, 60);
699 lv_slider_set_value(slider_in_cont, 100, LV_ANIM_OFF);
700
701 value_changed_cnt = 0;
702 lv_obj_add_event_cb(slider_in_cont, value_changed_event_cb, LV_EVENT_VALUE_CHANGED, NULL);
703
704 /* Make the container really scrollable downwards */
705 lv_obj_t * spacer = lv_obj_create(cont);
706 lv_obj_set_size(spacer, 10, 100);
707 lv_obj_set_pos(spacer, 0, 200);
708
709 lv_obj_update_layout(cont);
710 lv_refr_now(NULL);
711 TEST_ASSERT_GREATER_THAN_INT32(0, lv_obj_get_scroll_bottom(cont));
712
713 lv_area_t coords;
714 lv_obj_get_coords(slider_in_cont, &coords);
715 const int32_t press_y = (coords.y1 + coords.y2) / 2;
716 const int32_t target_x = (coords.x1 + coords.x2) / 2;
717
718 /* Press at the right end and drag further right: dragging starts but the value
719 * stays clamped at the maximum */
720 lv_test_mouse_move_to(coords.x2 - 2, press_y);
721 lv_test_mouse_press();
722 lv_test_wait(50);
723 lv_test_mouse_move_by(20, 0);
724 lv_test_wait(50);
725
726 TEST_ASSERT_TRUE(lv_slider_is_dragged(slider_in_cont));
727 TEST_ASSERT_EQUAL_INT32(100, lv_slider_get_value(slider_in_cont));
728 TEST_ASSERT_EQUAL_UINT32(0, value_changed_cnt);
729
730 /* Move up so the indev starts scrolling the container */
731 lv_test_mouse_move_by(0, -40);
732 lv_test_wait(50);
733
734 lv_indev_t * mouse = lv_test_indev_get_indev(LV_INDEV_TYPE_POINTER);
735 TEST_ASSERT_EQUAL_PTR(cont, lv_indev_get_scroll_obj(mouse));
736 TEST_ASSERT_TRUE(lv_slider_is_dragged(slider_in_cont));
737
738 /* Drag back to the middle. The knob must follow even though a scroll object is active */
739 lv_test_mouse_move_by(target_x - (coords.x2 - 2 + 20), 0);
740 lv_test_wait(50);
741
742 TEST_ASSERT_INT_WITHIN(5, 50, lv_slider_get_value(slider_in_cont));
743 TEST_ASSERT_GREATER_THAN_UINT32(0, value_changed_cnt);
744
745 lv_test_mouse_release();
746 lv_test_wait(200);
747}
