43lv_result_t lv_thread_init(lv_thread_t * thread, const char * const name, lv_thread_prio_t prio,
44 void (*callback)(void *), size_t stack_size,
45 void * user_data)
46{
47 LV_UNUSED(name);
48 static const osPriority_t prio_map[] = {
49 [LV_THREAD_PRIO_LOWEST] = osPriorityLow,
50 [LV_THREAD_PRIO_LOW] = osPriorityBelowNormal,
51 [LV_THREAD_PRIO_MID] = osPriorityNormal,
52 [LV_THREAD_PRIO_HIGH] = osPriorityHigh,
53 [LV_THREAD_PRIO_HIGHEST] = osPriorityRealtime7,
54 };
55
56 osThreadAttr_t c_tThreadAttribute = {
57 .stack_size = stack_size,
58 .priority = prio_map[prio],
59 };
60
61 *thread = osThreadNew(callback, user_data, &c_tThreadAttribute);
62
63 if(NULL == *thread) {
64 LV_LOG_WARN("Error: Failed to create a cmsis-rtos2 thread.");
65 return LV_RESULT_INVALID;
66 }
67
68 return LV_RESULT_OK;
69
70}

153lv_result_t lv_thread_sync_wait(lv_thread_sync_t * sync)
154{
155 uint32_t ret = osEventFlagsWait(*sync, 0x01, osFlagsWaitAny, osWaitForever);
156 if(ret & (1 << 31)) {
157 LV_LOG_WARN("Error: failed to wait a cmsis-rtos2 EventFlag %d", ret);
158 return LV_RESULT_INVALID;
159 }
160
161 return LV_RESULT_OK;
162}
