45void test_custom_prop_ids(void)
46{
47 uint8_t fake_flag = 0;
48 uint32_t initial_custom_props = lv_style_get_num_custom_props();
49 uint32_t max_props_to_register = 64;
50 for(uint32_t i = 0; i < max_props_to_register; i++) {
51 lv_style_prop_t prop = lv_style_register_prop(fake_flag);
52 /* Should have a higher index than the last built-in prop */
53 TEST_ASSERT_GREATER_THAN(LV_STYLE_LAST_BUILT_IN_PROP, prop);
54 if(i == 0) {
55 /* Should be equal to the first expected index of a custom prop */
56 TEST_ASSERT_EQUAL(LV_STYLE_NUM_BUILT_IN_PROPS + initial_custom_props, prop);
57 }
58 /*We should find our flags*/
59 TEST_ASSERT_EQUAL(fake_flag, lv_style_prop_lookup_flags(prop));
60 if(fake_flag == 0xff)
61 fake_flag = 0;
62 else
63 fake_flag++;
64 }
65 TEST_ASSERT_EQUAL(initial_custom_props + max_props_to_register, lv_style_get_num_custom_props());
66 /*
67 * Check that the resizing algorithm works correctly, given that 64 props
68 * were registered + whatever's built-in. A failure here may just indicate
69 * that LVGL registers more built-in properties now and this needs adjustment.
70 */
71 TEST_ASSERT_EQUAL(LV_GLOBAL_DEFAULT()->style_custom_table_size, 64);
72}

123void test_style_copy(void)
124{
125 lv_style_t style1;
126 lv_style_t style2;
127 lv_style_t copied_style;
128
129 const lv_color_t copied_bg_color = lv_color_hex(0x0000ff);
130 const lv_color_t copied_border_color = lv_color_hex(0x00ff00);
131 const lv_color_t copied_outline_color = lv_color_hex(0xffff00);
132
133 lv_style_init(&style1);
134 lv_style_set_bg_color(&style1, lv_color_hex(0xff0000)); // this should get overwritten
135 lv_style_set_outline_color(&style1, copied_outline_color);
136
137 lv_style_init(&style2);
138 lv_style_set_bg_color(&style2, copied_bg_color);
139 lv_style_set_border_color(&style2, copied_border_color);
140
141 lv_style_init(&copied_style);
142 lv_style_copy(&copied_style, &style1);
143 lv_style_copy(&copied_style, &style2); // This should reset `copied_style` then duplicate the properties of `style2`
144
145 lv_style_value_t value;
146
147 TEST_ASSERT_TRUE(lv_style_get_prop(&copied_style, LV_STYLE_BG_COLOR, &value) == LV_STYLE_RES_FOUND);
148 TEST_ASSERT_EQUAL_COLOR(copied_bg_color, value.color);
149 TEST_ASSERT_TRUE(lv_style_get_prop(&copied_style, LV_STYLE_BORDER_COLOR, &value) == LV_STYLE_RES_FOUND);
150 TEST_ASSERT_EQUAL_COLOR(copied_border_color, value.color);
151 TEST_ASSERT_TRUE(lv_style_get_prop(&copied_style, LV_STYLE_OUTLINE_COLOR, &value) == LV_STYLE_RES_NOT_FOUND);
152
153 /* Changing the original style should not impact the style that copied it */
154 lv_style_set_bg_color(&style2, lv_color_hex(0x00ff00));
155 TEST_ASSERT_TRUE(lv_style_get_prop(&copied_style, LV_STYLE_BG_COLOR, &value) == LV_STYLE_RES_FOUND);
156 TEST_ASSERT_EQUAL_COLOR(copied_bg_color, value.color);
157
158 lv_style_reset(&style1);
159 lv_style_reset(&style2);
160 lv_style_reset(&copied_style);
161}
