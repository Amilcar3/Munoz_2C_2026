170void lv_roller_set_selected(lv_obj_t * obj, uint32_t sel_opt, lv_anim_enable_t anim)
171{
172 LV_CHECK_OBJ(obj, MY_CLASS, return);
173
174 /*Set the value even if it's the same as the current value because
175 *if moving to the next option with an animation which was just deleted in the PRESS Call the ancestor's event handler
176 *nothing will continue the animation.*/
177
178 lv_roller_t * roller = (lv_roller_t *)obj;
179
180 /*In infinite mode interpret the new ID relative to the currently visible "page"*/
181 if(roller->mode == LV_ROLLER_MODE_INFINITE) {
182 uint32_t real_option_cnt = roller->option_cnt / roller->inf_page_cnt;
183 uint32_t current_page = roller->sel_opt_id / real_option_cnt;
184 /*Set by the user to e.g. 0, 1, 2, 3...
185 *Upscale the value to the current page*/
186 if(sel_opt < real_option_cnt) {
187 uint32_t act_opt = roller->sel_opt_id - current_page * real_option_cnt;
188 int32_t sel_opt_signed = sel_opt;
189 /*Huge jump? Probably from last to first or first to last option.*/
190 if((uint32_t)LV_ABS((int16_t)(act_opt - sel_opt)) > real_option_cnt / 2) {
191 if(act_opt > sel_opt) sel_opt_signed += real_option_cnt;
192 else sel_opt_signed -= real_option_cnt;
193 }
194 sel_opt = sel_opt_signed + real_option_cnt * current_page;
195 }
196 }
197
198 roller->sel_opt_id = sel_opt < roller->option_cnt ? sel_opt : roller->option_cnt - 1;
199 roller->sel_opt_id_ori = roller->sel_opt_id;
200
201 refr_position(obj, anim);
202}

204bool lv_roller_set_selected_str(lv_obj_t * obj, const char * sel_opt, lv_anim_enable_t anim)
205{
206 LV_CHECK_OBJ(obj, MY_CLASS, return false);
207 LV_CHECK_ARG(sel_opt != NULL, return false);
208
209 const char * options = lv_roller_get_options(obj);
210 size_t options_len = lv_strlen(options);
211
212 bool option_found = false;
213
214 uint32_t current_option = 0;
215 size_t line_start = 0;
216
217 for(size_t i = 0; i < options_len; i++) {
218 if(options[i] == '\n') {
219 /* See if this is the correct option */
220 if(lv_strncmp(&options[line_start], sel_opt, i - line_start) == 0) {
221 lv_roller_set_selected(obj, current_option, anim);
222 option_found = true;
223 break;
224 }
225
226 current_option++;
227 line_start = i + 1;
228 }
229 }
230
231 return option_found;
232}

314lv_result_t lv_roller_get_option_str(const lv_obj_t * obj, uint32_t option, char * buf, uint32_t buf_size)
315{
316 LV_CHECK_OBJ(obj, MY_CLASS, return LV_RESULT_INVALID);
317 LV_CHECK_ARG(buf != NULL, return LV_RESULT_INVALID);
318
319 lv_obj_t * label = get_label(obj);
320 uint32_t i;
321 uint32_t line = 0;
322 const char * opt_txt = lv_label_get_text(label);
323 size_t txt_len = lv_strlen(opt_txt);
324
325 for(i = 0; i < txt_len && line != option; i++) {
326 if(opt_txt[i] == '\n') line++;
327 }
328
329 if(line != option) {
330 return LV_RESULT_INVALID;
331 }
332
333 uint32_t c;
334 for(c = 0; i < txt_len && opt_txt[i] != '\n'; c++, i++) {
335 if(buf_size && c >= buf_size - 1) {
336 LV_LOG_WARN("the buffer was too small");
337 break;
338 }
339 buf[c] = opt_txt[i];
340 }
341
342 buf[c] = '\0';
343 return LV_RESULT_OK;
344}

396static void lv_roller_event(const lv_obj_class_t * class_p, lv_event_t * e)
397{
398 LV_UNUSED(class_p);
399 LV_ASSERT(e != NULL);
400 lv_result_t res;
401
402 /*Call the ancestor's event handler*/
403 res = lv_obj_event_base(MY_CLASS, e);
404 if(res != LV_RESULT_OK) return;
405
406 const lv_event_code_t code = lv_event_get_code(e);
407 lv_obj_t * obj = lv_event_get_current_target(e);
408 LV_ASSERT(obj != NULL);
409 lv_roller_t * roller = (lv_roller_t *)obj;
410
411 if(code == LV_EVENT_GET_SELF_SIZE) {
412 lv_point_t * p = lv_event_get_param(e);
413 p->x = get_selected_label_width(obj);
414 }
415 else if(code == LV_EVENT_STYLE_CHANGED) {
416 lv_obj_t * label = get_label(obj);
417 /*Be sure the label's style is updated before processing the roller*/
418 if(label) lv_obj_send_event(label, LV_EVENT_STYLE_CHANGED, NULL);
419 lv_obj_refresh_self_size(obj);
420 refr_position(obj, LV_ANIM_OFF);
421 }
422 else if(code == LV_EVENT_SIZE_CHANGED) {
423 refr_position(obj, LV_ANIM_OFF);
424 }
425 else if(code == LV_EVENT_PRESSED) {
426 if(roller->option_cnt <= 1) return;
427
428 roller->moved = 0;
429 lv_anim_delete(get_label(obj), set_y_anim);
430 }
431 else if(code == LV_EVENT_CLICKED) {
432 if(roller->moved == 1) {
433 lv_event_stop_processing(e);
434 }
435 }
436 else if(code == LV_EVENT_PRESSING) {
437 if(roller->option_cnt <= 1) return;
438
439 lv_indev_t * indev = lv_event_get_indev(e);
440 /*Without an input device nothing moved, so the vector stays zero*/
441 lv_point_t p = { 0, 0 };
442 if(indev != NULL) lv_indev_get_vect(indev, &p);
443 transform_vect_recursive(obj, &p);
444 if(p.y) {
445 lv_obj_t * label = get_label(obj);
446 lv_obj_set_y(label, lv_obj_get_y_aligned(label) + p.y);
447 roller->moved = 1;
448 }
449 }
450 else if(code == LV_EVENT_RELEASED || code == LV_EVENT_PRESS_LOST) {
451 release_handler(obj);
452 }
453 else if(code == LV_EVENT_FOCUSED) {
454 lv_group_t * g = lv_obj_get_group(obj);
455 lv_indev_t * indev = lv_indev_active();
456 lv_indev_type_t indev_type = indev != NULL ? lv_indev_get_type(indev) : LV_INDEV_TYPE_NONE;
457
458 /*Encoders need special handling*/
459 if(indev_type == LV_INDEV_TYPE_ENCODER) {
460 const bool editing = g != NULL && lv_group_get_editing(g);
461
462 /*Save the current state when entered to edit mode*/
463 if(editing) {
464 roller->sel_opt_id_ori = roller->sel_opt_id;
465 }
466 else { /*In navigate mode revert the original value*/
467 if(roller->sel_opt_id != roller->sel_opt_id_ori) {
468 roller->sel_opt_id = roller->sel_opt_id_ori;
469 refr_position(obj, LV_ANIM_ON);
470 }
471 }
472 }
473 else {
474 /*Save the current value. Used to revert this
475 *state if ENTER won't be pressed*/
476 roller->sel_opt_id_ori = roller->sel_opt_id;
477 }
478 }
479 else if(code == LV_EVENT_DEFOCUSED) {
480 /*Revert the original state*/
481 if(roller->sel_opt_id != roller->sel_opt_id_ori) {
482 roller->sel_opt_id = roller->sel_opt_id_ori;
483 refr_position(obj, LV_ANIM_ON);
484 }
485 }
486 else if(code == LV_EVENT_KEY) {
487 if(roller->option_cnt <= 1) return;
488
489 uint32_t c = lv_event_get_key(e);
490 if(c == LV_KEY_RIGHT || c == LV_KEY_DOWN) {
491 if(roller->sel_opt_id + 1 < roller->option_cnt) {
492 uint32_t ori_id = roller->sel_opt_id_ori; /*lv_roller_set_selected will overwrite this*/
493 lv_roller_set_selected(obj, roller->sel_opt_id + 1, LV_ANIM_ON);
494 roller->sel_opt_id_ori = ori_id;
495 }
496 }
497 else if(c == LV_KEY_LEFT || c == LV_KEY_UP) {
498 if(roller->sel_opt_id > 0) {
499 uint32_t ori_id = roller->sel_opt_id_ori; /*lv_roller_set_selected will overwrite this*/
500 lv_roller_set_selected(obj, roller->sel_opt_id - 1, LV_ANIM_ON);
501 roller->sel_opt_id_ori = ori_id;
502 }
503 }
504 }
505 else if(code == LV_EVENT_ROTARY) {
506 if(roller->option_cnt <= 1) return;
507
508 int32_t r = lv_event_get_rotary_diff(e);
509 int32_t new_id = roller->sel_opt_id + r;
510 new_id = LV_CLAMP(0, new_id, (int32_t)roller->option_cnt - 1);
511 if((int32_t)roller->sel_opt_id != new_id) {
512 uint32_t ori_id = roller->sel_opt_id_ori; /*lv_roller_set_selected will overwrite this*/
513 lv_roller_set_selected(obj, new_id, LV_ANIM_ON);
514 roller->sel_opt_id_ori = ori_id;
515 }
516 }
517 else if(code == LV_EVENT_REFR_EXT_DRAW_SIZE) {
518 lv_obj_t * label = get_label(obj);
519 lv_obj_refresh_ext_draw_size(label);
520 }
521 else if(code == LV_EVENT_DRAW_MAIN || code == LV_EVENT_DRAW_POST) {
522 draw_main(e);
523 }
524#if LV_USE_TRANSLATION
525 else if(code == LV_EVENT_TRANSLATION_LANGUAGE_CHANGED) {
526 if(roller->options_translation_tag) {
527
528 uint32_t prev_sel = roller->sel_opt_id;
529
530 lv_result_t options_res = update_options(obj, lv_tr(roller->options_translation_tag), roller->mode);
531 if(options_res == LV_RESULT_INVALID) {
532 LV_LOG_WARN("Couldn't set the new translated options");
533 }
534 else {
535 lv_roller_set_selected(obj, prev_sel, LV_ANIM_OFF);
536 }
537 }
538 }
539#endif /*LV_USE_TRANSLATION*/
540}

542static void lv_roller_label_event(const lv_obj_class_t * class_p, lv_event_t * e)
543{
544 LV_UNUSED(class_p);
545
546 LV_ASSERT(e != NULL);
547 lv_result_t res;
548
549 lv_event_code_t code = lv_event_get_code(e);
550 /*LV_EVENT_DRAW_MAIN will be called in the draw function*/
551 if(code != LV_EVENT_DRAW_MAIN) {
552 /* Call the ancestor's event handler */
553 res = lv_obj_event_base(MY_CLASS_LABEL, e);
554 if(res != LV_RESULT_OK) return;
555 }
556
557 lv_obj_t * label = lv_event_get_current_target(e);
558 if(code == LV_EVENT_REFR_EXT_DRAW_SIZE) {
559 /*If the selected text has a larger font it needs some extra space to draw it*/
560 int32_t * s = lv_event_get_param(e);
561 lv_obj_t * obj = lv_obj_get_parent(label);
562 int32_t sel_w = get_selected_label_width(obj);
563 int32_t label_w = lv_obj_get_width(label);
564 *s = LV_MAX(*s, sel_w - label_w);
565 }
566 else if(code == LV_EVENT_SIZE_CHANGED) {
567 refr_position(lv_obj_get_parent(label), LV_ANIM_OFF);
568 }
569 else if(code == LV_EVENT_DRAW_MAIN) {
570 draw_label(e);
571 }

574static void draw_main(lv_event_t * e)
575{
576 LV_ASSERT(e != NULL);
577 lv_event_code_t code = lv_event_get_code(e);
578 lv_obj_t * obj = lv_event_get_current_target(e);
579 LV_ASSERT(obj != NULL);
580 if(code == LV_EVENT_DRAW_MAIN) {
581 /*Draw the selected rectangle*/
582 lv_layer_t * layer = lv_event_get_layer(e);
583 lv_area_t sel_area;
584 get_sel_area(obj, &sel_area);
585 lv_draw_rect_dsc_t sel_dsc;
586 lv_draw_rect_dsc_init(&sel_dsc);
587 sel_dsc.base.layer = layer;
588 lv_obj_init_draw_rect_dsc(obj, LV_PART_SELECTED, &sel_dsc);
589 lv_draw_rect(layer, &sel_dsc, &sel_area);
590 }
591 /*Post draw when the children are drawn*/
592 else if(code == LV_EVENT_DRAW_POST) {
593 lv_layer_t * layer = lv_event_get_layer(e);
594
595 lv_draw_label_dsc_t label_dsc;
596 lv_draw_label_dsc_init(&label_dsc);
597 label_dsc.base.layer = layer;
598 lv_obj_init_draw_label_dsc(obj, LV_PART_SELECTED, &label_dsc);
599
600 lv_text_attributes_t attributes = {0};
601 attributes.letter_space = label_dsc.letter_space;
602 attributes.line_space = label_dsc.line_space;
603 attributes.max_width = lv_obj_get_width(obj);
604 attributes.text_flags = LV_TEXT_FLAG_EXPAND;
605

607 lv_area_t sel_area;
608 get_sel_area(obj, &sel_area);

612 if(area_ok) {

614 if(lv_label_get_recolor(label)) label_dsc.flag |= LV_TEXT_FLAG_RECOLOR;
615

617 lv_point_t label_sel_size;
618 lv_text_get_size_attributes(&label_sel_size, lv_label_get_text(label), label_dsc.font, &attributes);

620 /*Move the selected label proportionally with the background label*/
621 int32_t roller_h = lv_obj_get_height(obj);
622 const lv_font_t * normal_label_font = lv_obj_get_style_text_font_internal(obj, LV_PART_MAIN);
623 /*label offset from the middle line of the roller*/
624 int32_t label_y_prop = (label->coords.y1 + normal_label_font->line_height / 2) - (roller_h / 2 + obj->coords.y1);

626 /*Proportional position from the middle line.
627 *Will be 0 for the first option, and 1 for the last option (upscaled by << 14)*/
628 int32_t remain_h = lv_obj_get_height(label) - normal_label_font->line_height;
629 if(remain_h > 0) {
630 label_y_prop = (label_y_prop << 14) / remain_h;
631 }

634 *a larger font won't centered on selected area.*/
635 int32_t corr = label_dsc.font->line_height;

637 /*Apply the proportional position to the selected text*/
638 int32_t label_sel_y = roller_h / 2 + obj->coords.y1;
639 label_sel_y += ((label_sel_size.y - corr) * label_y_prop) >> 14;
640 label_sel_y -= corr / 2;
641
642 int32_t bwidth = lv_obj_get_style_border_width_internal(obj, LV_PART_MAIN);
643 int32_t pleft = lv_obj_get_style_pad_left_internal(obj, LV_PART_MAIN);
644 int32_t pright = lv_obj_get_style_pad_right_internal(obj, LV_PART_MAIN);

648 label_sel_area.x1 = obj->coords.x1 + pleft + bwidth;
649 label_sel_area.y1 = label_sel_y;
650 label_sel_area.x2 = obj->coords.x2 - pright - bwidth;
651 label_sel_area.y2 = label_sel_area.y1 + label_sel_size.y;
652
653 label_dsc.flag |= LV_TEXT_FLAG_EXPAND;
654 const lv_area_t clip_area_ori = layer->_clip_area;
655 layer->_clip_area = mask_sel;
656 label_dsc.text = lv_label_get_text(label);
657 lv_draw_label(layer, &label_dsc, &label_sel_area);
658 layer->_clip_area = clip_area_ori;
659 }

741static void refr_position(lv_obj_t * obj, lv_anim_enable_t anim_en)
742{
743 LV_ASSERT(obj != NULL);
744 lv_obj_t * label = get_label(obj);
745 if(label == NULL) return;

751 case LV_TEXT_ALIGN_CENTER:
752 x = (lv_obj_get_content_width(obj) - lv_obj_get_width(label)) / 2;
753 break;
754 case LV_TEXT_ALIGN_RIGHT:
755 x = lv_obj_get_content_width(obj) - lv_obj_get_width(label);

767 const int32_t line_space = lv_obj_get_style_text_line_space_internal(obj, LV_PART_MAIN);
768 const int32_t font_h = lv_font_get_line_height_internal(font);

775 inf_normalize(obj);
776 }
777
778 /* Calculate animation configuration */
779 lv_roller_t * roller = (lv_roller_t *)obj;
780 int32_t id = roller->sel_opt_id;
781 const int32_t sel_y1 = id * (font_h + line_space);
782 const int32_t mid_y1 = h / 2 - font_h / 2;
783 const int32_t new_y = mid_y1 - sel_y1;

785 if(anim_en == LV_ANIM_OFF || anim_time == 0) {

790 lv_anim_t a;
791 lv_anim_init(&a);
792 lv_anim_set_var(&a, label);
793 lv_anim_set_exec_cb(&a, set_y_anim);
794 lv_anim_set_values(&a, lv_obj_get_y(label), new_y);
795 lv_anim_set_duration(&a, anim_time);
796 lv_anim_set_completed_cb(&a, scroll_anim_completed_cb);
797 lv_anim_set_path_cb(&a, lv_anim_path_ease_out);
798 lv_anim_start(&a);

802static lv_result_t release_handler(lv_obj_t * obj)
803{
804 LV_ASSERT(obj != NULL);
805 lv_obj_t * label = get_label(obj);
806 if(label == NULL) return LV_RESULT_OK;
807
808 lv_indev_t * indev = lv_indev_active();
809 lv_roller_t * roller = (lv_roller_t *)obj;
810
811 /*Leave edit mode once a new option is selected*/
812 lv_indev_type_t indev_type = indev != NULL ? lv_indev_get_type(indev) : LV_INDEV_TYPE_NONE;
813 if(indev_type == LV_INDEV_TYPE_ENCODER || indev_type == LV_INDEV_TYPE_KEYPAD) {
814 roller->sel_opt_id_ori = roller->sel_opt_id;
815
816 if(indev_type == LV_INDEV_TYPE_ENCODER) {
817 lv_group_t * g = lv_obj_get_group(obj);
818 if(g != NULL && lv_group_get_editing(g)) {
819 lv_group_set_editing(g, false);
820 }
821 }
822 }
823
824 if(indev_type == LV_INDEV_TYPE_POINTER || indev_type == LV_INDEV_TYPE_BUTTON) {
825 /*Search the clicked option (For KEYPAD and ENCODER the new value should be already set)*/
826 int16_t new_opt = -1;
827 if(roller->moved == 0) {
828 new_opt = 0;
829 lv_point_t p;
830 lv_indev_get_point(indev, &p);
831 lv_obj_transform_point(obj, &p, LV_OBJ_POINT_TRANSFORM_FLAG_INVERSE_RECURSIVE);
832 p.y -= label->coords.y1;
833 p.x -= label->coords.x1;
834 uint32_t letter_i;
835 letter_i = lv_label_get_letter_on(label, &p, true);
836
837 const char * txt = lv_label_get_text(label);
838 uint32_t i = 0;
839 uint32_t i_prev = 0;
840
841 uint32_t letter_cnt = 0;
842 for(letter_cnt = 0; letter_cnt < letter_i; letter_cnt++) {
843 uint32_t letter = lv_text_encoded_next(txt, &i);
844 /*Count he lines to reach the clicked letter. But ignore the last '\n' because it
845 * still belongs to the clicked line*/
846 if(letter == '\n' && i_prev != letter_i) new_opt++;
847 i_prev = i;
848 }
849 }
850 else {
851 /*If dragged then align the list to have an element in the middle*/
852 const lv_font_t * font = lv_obj_get_style_text_font_internal(obj, LV_PART_MAIN);
853 int32_t line_space = lv_obj_get_style_text_line_space_internal(obj, LV_PART_MAIN);
854 int32_t font_h = lv_font_get_line_height_internal(font);
855
856 int32_t label_unit = font_h + line_space;
857 int32_t mid = obj->coords.y1 + (obj->coords.y2 - obj->coords.y1) / 2;
858
859 lv_point_t p = indev->pointer.scroll_throw_vect_ori;
860 transform_vect_recursive(obj, &p);
861
862 int32_t scroll_throw = indev->scroll_throw;
863 int32_t sum = 0;
864 int32_t v = p.y;
865 while(v) {
866 sum += v;
867 v = v * (100 - scroll_throw) / 100;
868 }
869
870 int32_t label_y1 = label->coords.y1 + sum;
871 int32_t id = (mid - label_y1) / label_unit;
872
873 if(id < 0) id = 0;
874 if(id >= (int32_t)roller->option_cnt) id = roller->option_cnt - 1;
875
876 new_opt = id;
877 }
878
879 if(new_opt >= 0) {
880 lv_roller_set_selected(obj, new_opt, LV_ANIM_ON);
881 }
882 }
883
884 uint32_t id = roller->sel_opt_id; /*Just to use uint32_t in event data*/
885 lv_result_t res = lv_obj_send_event(obj, LV_EVENT_VALUE_CHANGED, &id);
886 return res;
887}

959static void transform_vect_recursive(lv_obj_t * roller, lv_point_t * vect)
960{
961 LV_ASSERT(roller != NULL);
962 LV_ASSERT(vect != NULL);
963 int16_t angle = 0;
964 int32_t scale_x = 256;
965 int32_t scale_y = 256;
966 lv_obj_t * parent = roller;
967 while(parent) {
968 angle += lv_obj_get_style_transform_rotation_internal(parent, LV_PART_MAIN);
969 int32_t zoom_act_x = lv_obj_get_style_transform_scale_x_safe_internal(parent, LV_PART_MAIN);
970 int32_t zoom_act_y = lv_obj_get_style_transform_scale_y_safe_internal(parent, LV_PART_MAIN);
971 scale_x = (scale_x * zoom_act_x) >> 8;
972 scale_y = (scale_y * zoom_act_y) >> 8;
973 parent = lv_obj_get_parent(parent);
974 }
975 lv_point_t pivot = { 0, 0 };
976
977 if(scale_x == 0) {
978 scale_x = 1;
979 }
980
981 if(scale_y == 0) {
982 scale_y = 1;
983 }
984
985 scale_x = 256 * 256 / scale_x;
986 scale_y = 256 * 256 / scale_y;
987 lv_point_transform(vect, -angle, scale_x, scale_y, &pivot, false);
988}

990static lv_result_t update_options(lv_obj_t * obj, const char * options, lv_roller_mode_t mode)
991{
992 LV_ASSERT(obj);
993 LV_ASSERT(options);
994
995 lv_roller_t * roller = (lv_roller_t *)obj;
996 lv_obj_t * label = get_label(obj);
997
998 roller->sel_opt_id = 0;
999 roller->sel_opt_id_ori = 0;
1000
1001 /*Count the '\n'-s to determine the number of options*/
1002 roller->option_cnt = 0;
1003 uint32_t cnt;
1004 for(cnt = 0; options[cnt] != '\0'; cnt++) {
1005 if(options[cnt] == '\n') roller->option_cnt++;
1006 }
1007 roller->option_cnt++; /*Last option has no `\n`*/
1008
1009 if(mode == LV_ROLLER_MODE_NORMAL) {
1010 roller->mode = LV_ROLLER_MODE_NORMAL;
1011 lv_label_set_text(label, options);
1012 }
1013 else {
1014
1015 const lv_font_t * font = lv_obj_get_style_text_font(obj, LV_PART_MAIN);
1016
1017 int32_t normal_h = roller->option_cnt * (lv_font_get_line_height_internal(font) +
1018 lv_obj_get_style_text_letter_space_internal(obj, LV_PART_MAIN));
1019 uint32_t inf_page_cnt = LV_CLAMP(3, EXTRA_INF_SIZE / normal_h, 15);
1020 if(!(inf_page_cnt & 1)) inf_page_cnt++; /*Make it odd*/
1021 LV_LOG_INFO("Using %" LV_PRIu32 " pages to make the roller look infinite", inf_page_cnt);
1022
1023 size_t opt_len = lv_strlen(options) + 1; /*+1 to add '\n' after option lists*/
1024 size_t opt_extra_len = opt_len * inf_page_cnt;
1025 if(opt_extra_len == 0) {
1026 /*Prevent write overflow*/
1027 opt_extra_len = 1;
1028 }
1029
1030 /*Have a the options multiple times so that it looks infinite */
1031 char * opt_extra = lv_malloc(opt_extra_len);
1032 LV_ASSERT_MALLOC(opt_extra);
1033 if(opt_extra == NULL) {
1034 LV_LOG_WARN("Couldn't allocate the buffer for the options");
1035 return LV_RESULT_INVALID;
1036 }
1037
1038 roller->inf_page_cnt = inf_page_cnt;
1039 roller->mode = LV_ROLLER_MODE_INFINITE;
1040 uint32_t i;
1041 for(i = 0; i < roller->inf_page_cnt; i++) {
1042 lv_strcpy(&opt_extra[opt_len * i], options);
1043 opt_extra[opt_len * (i + 1) - 1] = '\n';
1044 }
1045 opt_extra[opt_extra_len - 1] = '\0';
1046 lv_label_set_text(label, opt_extra);
1047 lv_free(opt_extra);
1048
1049 roller->sel_opt_id = ((roller->inf_page_cnt / 2) + 0) * roller->option_cnt;
1050
1051 roller->option_cnt = roller->option_cnt * roller->inf_page_cnt;
1052 inf_normalize(obj);
1053 }
1054
1055
1056 roller->sel_opt_id_ori = roller->sel_opt_id;
1057
1058 /*If the selected text has larger font the label needs some extra draw padding to draw it.*/
1059 lv_obj_refresh_ext_draw_size(label);
1060
1061 return LV_RESULT_OK;
1062}
