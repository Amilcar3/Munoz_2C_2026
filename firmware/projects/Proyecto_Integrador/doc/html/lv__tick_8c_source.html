45uint32_t lv_tick_get(void)
46{
47 lv_tick_state_t * state_p = &state;
48
49 if(state_p->tick_get_cb)
50 return state_p->tick_get_cb();
51
52 /*If `lv_tick_inc` is called from an interrupt while `sys_time` is read
53 *the result might be corrupted.
54 *This loop detects if `lv_tick_inc` was called while reading `sys_time`.
55 *If `tick_irq_flag` was cleared in `lv_tick_inc` try to read again
56 *until `tick_irq_flag` remains `1`.*/
57 uint32_t result;
58 do {
59 state_p->sys_irq_flag = 1;
60 result = state_p->sys_time;
61 } while(!state_p->sys_irq_flag); /*Continue until see a non interrupted cycle*/
62
63 return result;
64}

77void lv_delay_ms(uint32_t ms)
78{
79 if(state.delay_cb) {
80 state.delay_cb(ms);
81 }
82 else {
83 uint32_t t = lv_tick_get();
84 while(lv_tick_elaps(t) < ms) {
85 /*Do something to no call `lv_tick_elaps` too often as it might interfere with interrupts*/
86 volatile uint32_t i;
87 volatile uint32_t x = ms;
88 for(i = 0; i < 100; i++) {
89 x = x * 3;
90 }
91 }
92 }
