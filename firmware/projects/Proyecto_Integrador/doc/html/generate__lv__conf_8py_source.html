60def parse_defaults(path: str):
61 defaults = {}
62
63 with open(path, 'r', encoding='utf-8') as file:
64 for line in file.readlines():
65 if len(line.strip()) == 0 or line.startswith('#'):
66 continue
67 search_result = re.search(r'([A-Z0-9_]+)\s+(.+)', line)
68 if not search_result:

70 continue
71 groups = search_result.groups()
