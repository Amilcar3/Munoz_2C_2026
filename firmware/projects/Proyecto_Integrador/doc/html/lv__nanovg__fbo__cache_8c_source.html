64void lv_nanovg_fbo_cache_init(lv_draw_nanovg_unit_t * u)
65{
66 LV_ASSERT_NULL(u);
67 LV_ASSERT(u->fbo_cache == NULL);
68
69 const lv_cache_ops_t ops = {
70 .compare_cb = (lv_cache_compare_cb_t)fbo_compare_cb,
71 .create_cb = (lv_cache_create_cb_t)fbo_create_cb,
72 .free_cb = (lv_cache_free_cb_t)fbo_free_cb,
73 };
74
75 u->fbo_cache = lv_cache_create(&lv_cache_class_lru_ll_count, sizeof(fbo_item_t), LV_NANOVG_FBO_CACHE_CNT, ops);
76 lv_cache_set_name(u->fbo_cache, "NVG_FBO");

157static lv_cache_compare_res_t fbo_compare_cb(const fbo_item_t * lhs, const fbo_item_t * rhs)
158{
159 if(lhs->width != rhs->width) {
160 return lhs->width > rhs->width ? 1 : -1;
161 }
162
163 if(lhs->height != rhs->height) {
164 return lhs->height > rhs->height ? 1 : -1;
165 }
166
167 if(lhs->flags != rhs->flags) {
168 return lhs->flags > rhs->flags ? 1 : -1;
169 }
170
171 if(lhs->format != rhs->format) {
172 return lhs->format > rhs->format ? 1 : -1;
173 }
174
175 return 0;
176}
