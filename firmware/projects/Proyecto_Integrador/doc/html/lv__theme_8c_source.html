148static void apply_theme(lv_theme_t * th, lv_obj_t * obj)
149{
150 if(th->parent) apply_theme(th->parent, obj);
151 if(th->apply_cb) {
152 th->apply_cb(th, obj);
153 for(uint32_t i = 0; i < obj->style_cnt; i++) {
154 obj->styles[i].is_theme = 1;
155 }
156 }
157}
158
159static void apply_theme_recursion(lv_theme_t * th, lv_obj_t * obj)
160{
161 const lv_obj_class_t * original_class_p = obj->class_p;
162
163 if(obj->class_p->base_class && obj->class_p->theme_inheritable == LV_OBJ_CLASS_THEME_INHERITABLE_TRUE) {
164 /*Apply the base class theme in obj*/
165 obj->class_p = obj->class_p->base_class;
166
167 /*apply the base first*/
168 apply_theme_recursion(th, obj);
169 }
170
171 /*Restore the original class*/
172 obj->class_p = original_class_p;
173
174 apply_theme(th, obj);
175}
