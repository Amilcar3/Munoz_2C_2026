44lv_display_t * lv_test_display_egl_create(int32_t hor_res, int32_t ver_res)
45{
46 lv_display_t * disp = lv_opengles_pbuffer_create(hor_res, ver_res);
47 if(!disp) {
48 LV_LOG_ERROR("Failed to create the EGL pbuffer display for NanoVG headless testing");
49 return NULL;
50 }
51 lv_test_display_egl_data_t * ddata = lv_malloc(sizeof(*ddata));
52 if(!ddata) {
53 LV_LOG_ERROR("failed to allocate display data");
54 lv_opengles_pbuffer_t * pbuffer = lv_opengles_pbuffer_get_handle(disp);
55 lv_display_delete(disp);
56 lv_opengles_pbuffer_destroy(pbuffer);
57 return NULL;
58 }
59
60 ddata->fixed_hor_res = hor_res;
61 ddata->fixed_ver_res = ver_res;
62
63 lv_display_set_user_data(disp, ddata);
64
65 /* We don't support resolution nor color format changes but bind the events anyway
66 * to catch and fail in case that happens
67 */
68 lv_display_add_event_cb(disp, egl_resolution_changed_cb, LV_EVENT_RESOLUTION_CHANGED, NULL);
69 lv_display_add_event_cb(disp, egl_color_format_changed_cb, LV_EVENT_COLOR_FORMAT_CHANGED, NULL);
70 lv_display_add_event_cb(disp, delete_event_cb, LV_EVENT_DELETE, NULL);
71
72 if(lv_opengles_pbuffer_get_gles_version(disp) < 3) {
73 LV_LOG_WARN("Only an OpenGL ES 2 context was available; glTF tests will not run");
74 }
75
76 LV_LOG_INFO("EGL headless test display created (%" LV_PRId32 "x%" LV_PRId32 ")", hor_res, ver_res);
77 return disp;
78}

96static void egl_resolution_changed_cb(lv_event_t * e)
97{
98 lv_display_t * disp = lv_event_get_target(e);
99 int32_t hor_res = lv_display_get_original_horizontal_resolution(disp);
100 int32_t ver_res = lv_display_get_original_vertical_resolution(disp);
101 lv_test_display_egl_data_t * ddata = lv_display_get_user_data(disp);
102 LV_ASSERT(ddata != NULL);
103
104 /* Resizing would require recreating the FBO and the draw buffer, which this
105 * headless test display does not support. Fail loudly so the offending test is
106 * caught rather than silently producing corrupt screenshots. */
107 LV_ASSERT_FORMAT_MSG(hor_res == ddata->fixed_hor_res && ver_res == ddata->fixed_ver_res,
108 "NanoVG headless test display does not support changing resolution "
109 "(%" LV_PRId32 "x%" LV_PRId32 " -> %" LV_PRId32 "x%" LV_PRId32 "); "
110 "skip or adjust this test for NanoVG",
111 ddata->fixed_hor_res, ddata->fixed_ver_res, hor_res, ver_res);
112}

114static void egl_color_format_changed_cb(lv_event_t * e)
115{
116 lv_display_t * disp = lv_event_get_target(e);
117 lv_color_format_t cf = lv_display_get_color_format(disp);
118
119 /* The readback path always reads the FBO back as straight-alpha 32bpp BGRA
120 * (XRGB8888/ARGB8888) using a fixed draw buffer, ignoring disp->color_format.
121 * Other formats (incl. premultiplied) can't be represented, so reject them. */
122 LV_ASSERT_FORMAT_MSG(cf == LV_COLOR_FORMAT_XRGB8888 || cf == LV_COLOR_FORMAT_ARGB8888,
123 "NanoVG headless test display only supports XRGB8888/ARGB8888 "
124 "(got color format %d); skip or adjust this test for NanoVG",
125 (int)cf);
126}
