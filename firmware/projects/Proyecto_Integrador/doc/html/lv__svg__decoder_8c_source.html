89static lv_result_t svg_decoder_info(lv_image_decoder_t * decoder, lv_image_decoder_dsc_t * src,
90 lv_image_header_t * header)
91{
92 lv_image_src_t src_type = src->src_type;
93
94 int width = 0;
95 int height = 0;
96
97 if(src_type == LV_IMAGE_SRC_FILE || src_type == LV_IMAGE_SRC_VARIABLE) {
98 const void * src_data = src->src;
99 uint8_t * buf = NULL;
100
101 if(src_type == LV_IMAGE_SRC_FILE) {
102 /*Support only "*.svg" files*/
103 if(lv_strcmp(lv_fs_get_ext(src_data), "svg")) {
104 return LV_RESULT_INVALID;
105 }
106
107 uint32_t rn;
108 lv_fs_res_t res;
109 uint32_t file_size = 0;
110 res = lv_fs_seek(&src->file, 0, LV_FS_SEEK_END);
111 if(res == LV_FS_RES_OK) {
112 lv_fs_tell(&src->file, &file_size);
113 lv_fs_seek(&src->file, 0, LV_FS_SEEK_SET); /* Reset position to start */
114 }
115#ifdef LV_USE_SVG_DEBUG
116 LV_LOG_INFO("LVGL file_size = %d.", file_size);
117
118#endif
119 if(file_size > 512)
120 file_size = 512;
121
122 buf = (uint8_t *)lv_zalloc(file_size);
123 LV_ASSERT_NULL(buf);
124 /* read some bytes for searching svg header */
125 res = lv_fs_read(&src->file, buf, file_size, &rn);
126 if(res != LV_FS_RES_OK) {
127 LV_LOG_WARN("can't open %s", (char *)src_data);
128 lv_free(buf);
129 return LV_RESULT_INVALID;
130 }
131
132 if(!valid_svg_data(buf, rn)) {
133 lv_free(buf);
134 return LV_RESULT_INVALID;
135 }
136
137 width = LV_DPI_DEF;
138 height = LV_DPI_DEF;
139
140 uint8_t * svg_start = NULL;
141 uint8_t * svg_end = NULL;
142 uint8_t * ptr = buf;
143 uint8_t * ptr_end = buf + file_size - 1;
144 while(ptr < ptr_end) {
145 if(*ptr == '<') {
146 if(lv_strncmp((char *)(ptr + 1), "svg", 3) == 0) {
147 svg_start = ptr;
148 }
149 }
150 if(svg_start && (*ptr == '>')) {
151 svg_end = ptr;
152 break;
153 }
154 ptr++;
155 }
156
157 if(svg_start && svg_end) {
158 lv_svg_node_t * svg_doc = lv_svg_load_data((char *)svg_start, svg_end - svg_start);
159 lv_svg_render_obj_t * svg_header = lv_svg_render_create(svg_doc);
160 if(svg_header->tag == LV_SVG_TAG_SVG) {
161 lv_area_t bounds;
162 svg_header->clz->get_bounds(svg_header, &bounds);
163 width = lv_area_get_width(&bounds) - 1;
164 height = lv_area_get_height(&bounds) - 1;
165 }
166 lv_svg_render_delete(svg_header);
167 lv_svg_node_delete(svg_doc);
168 }
169 else
170 LV_LOG_WARN("can't find svg viewport tag end");
171
172 lv_free(buf);
173 }
174 else {
175 const lv_image_dsc_t * img_dsc = src_data;
176 uint32_t data_size = img_dsc->data_size;
177 width = img_dsc->header.w;
178 height = img_dsc->header.h;
179
180 if(!valid_svg_data(img_dsc->data, data_size)) {
181 return LV_RESULT_INVALID;
182 }
183 }
184
185 header->cf = LV_COLOR_FORMAT_ARGB8888;
186 header->w = width;
187 header->h = height;
188 header->stride = width * 4;
189 header->flags |= LV_IMAGE_FLAGS_CUSTOM_DRAW;
190
191 decoder->custom_draw_cb = svg_draw;
192
193 return LV_RESULT_OK;
194 }
195
196 return LV_RESULT_INVALID;
197}

199static lv_result_t svg_decoder_open(lv_image_decoder_t * decoder, lv_image_decoder_dsc_t * dsc)
200{
201 LV_UNUSED(decoder);
202 LV_PROFILER_DECODER_BEGIN_TAG("lv_svg_decoder_open");
203
204 uint8_t * svg_data = NULL;
205 uint32_t svg_data_size = 0;
206
207 if(dsc->src_type == LV_IMAGE_SRC_FILE) {
208 const char * fn = dsc->src;
209 if(lv_strcmp(lv_fs_get_ext(fn), "svg") == 0) { /*Check the extension*/
210
211 svg_data = alloc_file(fn, &svg_data_size);
212 if(svg_data == NULL) {
213 LV_LOG_WARN("can't load file: %s", (const char *)dsc->src);
214 LV_PROFILER_DECODER_END_TAG("lv_svg_decoder_open");
215 return LV_RESULT_INVALID;
216 }
217
218 }
219 else {
220 LV_PROFILER_DECODER_END_TAG("lv_svg_decoder_open");
221 return LV_RESULT_INVALID;
222 }
223 }
224 else if(dsc->src_type == LV_IMAGE_SRC_VARIABLE) {
225 const lv_image_dsc_t * img_dsc = dsc->src;
226 svg_data = (uint8_t *)img_dsc->data;
227 svg_data_size = (uint32_t)img_dsc->data_size;
228 }
229 else {
230 LV_PROFILER_DECODER_END_TAG("lv_svg_decoder_open");
231 return LV_RESULT_INVALID;
232 }
233
234 lv_svg_node_t * svg_doc = lv_svg_load_data((char *)svg_data, svg_data_size);
235 lv_svg_render_obj_t * draw_list = lv_svg_render_create(svg_doc);
236
237 if(dsc->src_type == LV_IMAGE_SRC_FILE) {
238 lv_free(svg_data);
239 }
240 lv_svg_node_delete(svg_doc);
241
242 /* create a fake draw_buf object */
243 lv_draw_buf_t * draw_buf = lv_zalloc(sizeof(lv_draw_buf_t));
244
245 draw_buf->header.w = 1;
246 draw_buf->header.h = 1;
247 draw_buf->header.cf = LV_COLOR_FORMAT_ARGB8888;
248 draw_buf->header.flags = LV_IMAGE_FLAGS_ALLOCATED | LV_IMAGE_FLAGS_CUSTOM_DRAW;
249 draw_buf->header.stride = 4;
250 draw_buf->header.magic = LV_IMAGE_HEADER_MAGIC;
251 draw_buf->data = NULL;
252 draw_buf->unaligned_data = (void *)draw_list;
253 draw_buf->data_size = lv_svg_render_get_size(draw_list);
254 draw_buf->handlers = &_svg_draw_buf_handler;
255
256 dsc->decoded = draw_buf;
257
258 if(!dsc->args.no_cache && lv_image_cache_is_enabled()) {
259
260 lv_image_cache_data_t search_key;
261 search_key.src_type = dsc->src_type;
262 search_key.src = dsc->src;
263 search_key.slot.size = dsc->decoded->data_size;
264
265 lv_cache_entry_t * entry = lv_image_decoder_add_to_cache(decoder, &search_key, draw_buf, NULL);
266
267 if(entry == NULL) {
268 lv_draw_buf_destroy(draw_buf);
269 LV_PROFILER_DECODER_END_TAG("lv_svg_decoder_open");
270 return LV_RESULT_INVALID;
271 }
272 dsc->cache_entry = entry;
273 }
274
275 LV_PROFILER_DECODER_END_TAG("lv_svg_decoder_open");
276 return LV_RESULT_OK;
277}

287static uint8_t * alloc_file(const char * filename, uint32_t * size)
288{
289 uint8_t * data = NULL;
290 lv_fs_file_t f;
291 uint32_t data_size;
292 uint32_t rn;
293 lv_fs_res_t res;
294
295 *size = 0;
296
297 res = lv_fs_open(&f, filename, LV_FS_MODE_RD);
298 if(res != LV_FS_RES_OK) {
299 LV_LOG_WARN("can't open %s", filename);
300 return NULL;
301 }
302
303 res = lv_fs_seek(&f, 0, LV_FS_SEEK_END);
304 if(res != LV_FS_RES_OK) {
305 goto failed;
306 }
307
308 res = lv_fs_tell(&f, &data_size);
309 if(res != LV_FS_RES_OK) {
310 goto failed;
311 }
312
313 res = lv_fs_seek(&f, 0, LV_FS_SEEK_SET);
314 if(res != LV_FS_RES_OK) {
315 goto failed;
316 }
317
318 /*Read file to buffer*/
319 data = lv_malloc(data_size);
320 if(data == NULL) {
321 LV_LOG_WARN("malloc failed for data size %u", data_size);
322 goto failed;
323 }
324
325 res = lv_fs_read(&f, data, data_size, &rn);
326
327 if(res == LV_FS_RES_OK && rn == data_size) {
328 *size = rn;
329 }
330 else {
331 LV_LOG_WARN("read file failed");
332 lv_free(data);
333 data = NULL;
334 }
335
336failed:
337 lv_fs_close(&f);
338 return data;
339}

347static void svg_draw(lv_layer_t * layer, const lv_image_decoder_dsc_t * decoder_dsc, const lv_area_t * coords,
348 const lv_draw_image_dsc_t * image_dsc, const lv_area_t * clip_area)
349{
350 const lv_draw_buf_t * draw_buf = decoder_dsc->decoded;
351 const lv_svg_render_obj_t * list = draw_buf->unaligned_data;
352
353 LV_PROFILER_DRAW_BEGIN;
354
355 lv_draw_vector_dsc_t * dsc = lv_draw_vector_dsc_create(layer);
356
357 /*Save the widget so that `LV_EVENT_DRAW_TASK_ADDED` can be sent to it in `lv_draw_vector`*/
358 dsc->base.obj = image_dsc->base.obj;
359
360 lv_matrix_t matrix;
361 lv_matrix_identity(&matrix);
362 lv_matrix_translate(&matrix, coords->x1, coords->y1);
363 dsc->ctx->scissor_area = *clip_area;
364 if(image_dsc) {
365 int32_t off_x = (lv_area_get_width(coords) - (int32_t)image_dsc->header.w - 1) / 2;
366 int32_t off_y = (lv_area_get_height(coords) - (int32_t)image_dsc->header.h - 1) / 2;
367
368 if(image_dsc->pivot.x != 0 || image_dsc->pivot.y != 0) {
369 lv_matrix_translate(&matrix, off_x, off_y);
370 }
371 lv_matrix_translate(&matrix, image_dsc->pivot.x, image_dsc->pivot.y);
372 lv_matrix_rotate(&matrix, image_dsc->rotation / 10.0f);
373 lv_matrix_scale(&matrix, image_dsc->scale_x / 256.0f, image_dsc->scale_y / 256.0f);
374 lv_matrix_translate(&matrix, -image_dsc->pivot.x, -image_dsc->pivot.y);
375 }
376 lv_draw_vector_dsc_set_transform(dsc, &matrix);
377 lv_draw_svg_render(dsc, list);
378 lv_draw_vector(dsc);
379 lv_draw_vector_dsc_delete(dsc);
380
381 LV_PROFILER_DRAW_END;
382}
