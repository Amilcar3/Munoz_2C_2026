1"""Small helpers for reading kconfiglib options and turning them into C text.
2
3Kconfig stores each option as an object carrying dependencies, defaults, and
4expressions. These helpers pull out the useful bits (an option's default
5value, what it depends on, what it pulls in) and render them as the text that
6goes into the generated config headers.
7
8No imports from our own model, so both the parse and emit layers can use them
9without import cycles.
10"""

32def dep_terms(expr) -> list:
33 """Flat list of the individual terms in an ``A && B && C`` dependency
34 (here ``[A, B, C]``). ``[]`` for no expression."""
35 if expr is None:
36 return []
37 return [t for t in split_expr(expr, AND) if t is not None]

46def bool_default(sym) -> str:
47 """An on/off option's own default, as ``"1"`` or ``"0"``.
48
49 Kconfig secretly glues an option's ``depends on`` onto every one of its
50 defaults, so an option under ``if LV_USE_X`` would read as 0 whenever
51 ``LV_USE_X`` is off. We strip that glued-on dependency back off to report
52 the value the option takes when its feature is on. Other conditions on a
53 default (``default y if C``) are still checked against the current config.
54 """
55 assert sym.type in (BOOL, TRISTATE)
56 dd_keys = {term_key(x) for x in dep_terms(sym.direct_dep)}
57 for value, cond in sym.defaults:
58 rest = [x for x in dep_terms(cond) if term_key(x) not in dd_keys]
59 if all(expr_value(x) > 0 for x in rest):
60 return "1" if expr_value(value) > 0 else "0"
61 return "0"

64def default_value_text(sym, value) -> str:
65 """One ``default`` clause's value, as the C text *sym* would take from it.
66
67 A default pointing at another int/hex option is followed through to that
68 option's value; a bare number is kept as-is, gaining an ``0x`` prefix when
69 *sym* is hex."""
70 if isinstance(value, Symbol) and value.nodes and value.type in (INT, HEX):
71 return resolve_int_value(value)
72 s = expr_str(value)
73 if sym.type == HEX and not s.startswith("0x"):
74 s = "0x" + s
75 return s

78def resolve_int_value(sym) -> str:
79 """An int/hex option's value as C text (e.g. ``"32"`` or ``"0x40"``).
80
81 Checks each ``default N if <cond>`` in order (ignoring the option's own
82 ``depends on``) and returns the first that applies. When nothing applies -
83 e.g. a count gated off because its driver is disabled - returns ``0``
84 (``0x0`` for hex).
85
86 Not usable for an int derived from a "pick one" group (``default 128 if
87 LV_PPA_BURST_LENGTH_128``): every member of a gated-off group evaluates to
88 n, so no default applies and the value collapses to a meaningless ``0``.
89 Use :func:`choice_int_value` for those."""
90 dd_keys = {term_key(x) for x in dep_terms(sym.direct_dep)}
91 for value, cond in sym.defaults:
92 rest = [x for x in dep_terms(cond) if term_key(x) not in dd_keys]
93 if all(expr_value(x) > 0 for x in rest):
94 return default_value_text(sym, value)
95 return "0x0" if sym.type == HEX else "0"

98def choice_int_value(sym, choice, pairs) -> str:
99 """The value of an int derived from a "pick one" group, as C text.
100
101 *pairs* is ``[(member, value_expr), ...]``, one per ``default`` of *sym*.
102 The value is taken from whichever member the group selects - via
103 :func:`choice_default`, which reports the member the group *would* pick even
104 when it is gated off - instead of evaluating each default's condition
105 against the current config the way :func:`resolve_int_value` does."""
106 sel = choice_default(choice)
107 value = next((v for member, v in pairs if member is sel), pairs[0][1])
108 return default_value_text(sym, value)

111def scalar_default(sym) -> str:
112 """An int/hex/string option's own default, as a C literal.
113
114 Numbers go through :func:`resolve_int_value`. Strings keep their quotes
115 (``"foo"``); as with the other defaults, the option's own ``depends on`` is
116 stripped so a gated option reports the value it takes when its feature is
117 on.
118 """
119 if sym.type in (INT, HEX):
120 return resolve_int_value(sym)
121 dd_keys = {term_key(x) for x in dep_terms(sym.direct_dep)}
122 for value, cond in sym.defaults:
123 rest = [x for x in dep_terms(cond) if term_key(x) not in dd_keys]
124 if all(expr_value(x) > 0 for x in rest):
125 return expr_str(value)
126 return '""'

139def is_int_const(sym) -> bool:
140 """True if *sym* is a named token standing for a fixed number: no prompt and
141 a single literal default (e.g. ``LV_STDLIB_BUILTIN`` = ``0``). These are
142 emitted as bare ``#define``s in the internal "Config options" block; when an
143 enum picks one we emit its *name*, not the number.
144 """
145 if sym.type not in (INT, HEX):
146 return False
147 if any(node.prompt for node in sym.nodes):
148 return False
149 if len(sym.defaults) != 1:
150 return False
151
152 value, _cond = sym.defaults[0]
153 # A numeric literal default is modelled as an undefined symbol whose name is
154 # the literal text, not as an ``is_constant`` symbol.
155 if not (isinstance(value, Symbol) and not value.nodes):
156 return False
157 try:
158 int(value.name, 0)
159 except (ValueError, TypeError):
160 return False
161 return True

170 """The text of a numeric-literal default value, or None if it isn't one.

174 if not (isinstance(value, Symbol) and not value.nodes):

177 int(value.name, 0)
178 except (ValueError, TypeError):

190def derived_int_const_table(sym):
191 """Similar to :func:`is_int_const` but this one supports multiple default values"""
192
193 if sym.type not in (INT, HEX):
194 return None
195 if any(node.prompt for node in sym.nodes):
196 return None
197 if len(sym.defaults) < 2:
198 return None
199
200 dd_keys = {term_key(x) for x in dep_terms(sym.direct_dep)}
201 choice = None
202 table: dict[str, str] = {}
203 fallback = None
204
205 for value, cond in sym.defaults:
206 literal = _literal_value(value)
207 if literal is None:
208 return None
209 rest = [x for x in dep_terms(cond) if term_key(x) not in dd_keys]
210 if not rest: # an unconditional default covers every remaining member
211 if fallback is None:
212 fallback = literal
213 continue
214 if len(rest) != 1:
215 return None
216 members = or_terms(rest[0])
217 for m in members:
218 # An undefined symbol in a condition is always a typo: Kconfig reads
219 # it as permanently false, so the clause would just never fire.
220 if (
221 isinstance(m, Symbol)
222 and not m.nodes
223 and not m.is_constant
224 and _literal_value(m) is None
225 ):
226 raise ValueError(
227 f"{sym.name}: `default {literal} if {m.name}` references "
228 f"{m.name}, which no Kconfig file defines."
229 )
230 if not all(isinstance(m, Symbol) and m.choice is not None for m in members):
231 return None
232 for m in members:
233 if choice is None:
234 choice = m.choice
235 elif m.choice is not choice:
236 return None
237 table.setdefault(m.name, literal)
238
239 if choice is None or int_owns_choice(sym, choice):
240 return None
241
242 pairs, missing = [], []
243 for member in choice.syms:
244 literal = table.get(member.name, fallback)
245 if literal is None:
246 missing.append(member.name)
247 else:
248 pairs.append((member, literal))
249 if missing:
250 raise ValueError(
251 f"{sym.name} is derived from choice {choice.name} but no default "
252 f"covers: {', '.join(missing)}.\n"
253 f"Add a `default <n> if <member>` clause for each, or an "
254 f"unconditional final default."
255 )
256 return choice, pairs

265def choice_default(choice):
266 """The selected member of a "pick one" group. If the group is gated off,
267 the member it would default to - never nothing."""
268 if choice.selection:
269 return choice.selection
270 dd_keys = {term_key(x) for x in dep_terms(getattr(choice, "direct_dep", None))}
271 for sym, cond in getattr(choice, "defaults", []):
272 rest = [x for x in dep_terms(cond) if term_key(x) not in dd_keys]
273 if all(expr_value(x) > 0 for x in rest):
274 return sym
275 return choice.syms[0] if choice.syms else None

293def select_targets(item) -> list[str]:
294 """The user-settable options *item* auto-enables via ``select`` (by name).

307def select_lines(item) -> str:
308 """A comment listing the options *item* auto-enables (via ``select``).
309
310 Returns ``"\\n\\nEnable: <targets>"`` naming the user-settable options this
311 one turns on, or ``""`` if none. Shown in ``lv_conf_template.h`` so a
312 hand-written ``lv_conf.h`` knows to enable them too. Dependencies
313 (``depends on``) are left out - the template already nests each option in
314 its matching ``#if``, so they're visible from the structure."""
315 targets = select_targets(item)
316 return "\n\nEnable: " + ", ".join(targets) if targets else ""

319def member_requires(member, choice) -> str | None:
320 """A choice *member*'s own ``depends on`` as a C condition, or ``None``.
321
322 The choice's ambient dependencies (its enclosing ``if`` / ``depends on``,
323 e.g. ``LV_USE_SDL``) and the implicit "this choice is active" term are
324 stripped, leaving just what the member itself requires (e.g. the SDL EGL
325 backend's ``LV_USE_DRAW_OPENGLES || LV_USE_DRAW_NANOVG``).
326 """
327 ambient = {term_key(x) for x in dep_terms(getattr(choice, "direct_dep", None))}
328 rest = [
329 x
330 for x in dep_terms(getattr(member, "direct_dep", None))
331 if term_key(x) not in ambient and not isinstance(x, Choice)
332 ]
333 if not rest:
334 return None
335 expr = rest[0]
336 for term in rest[1:]:
337 expr = (AND, expr, term)
338 return rev_dep_c_expr(expr)

350def _render_bool_expr(expr, guard):
351 """Turn a Kconfig bool expression into C text.
352
353 Returns ``(c_text, precedence)``, or ``None`` if it renders to nothing.
354 Precedence (atom ``3``, ``&&`` ``2``, ``||`` ``1``) tells the caller where
355 parentheses are needed to keep the C meaning correct.
356
357 * ``&&`` / ``||`` / ``!`` map straight to C; comparisons (``==``, ``<``, ...)
358 already render as valid C;
359 * a "pick one" member listed in *guard* (member -> ``(macro, token)``)
360 becomes ``<macro> == <token>``, which works on both config paths;
361 * a bare group term (a member's implicit "this group is active" dependency)
362 renders to nothing and drops out of its enclosing ``&&`` / ``||``.
363 """
364 if isinstance(expr, Choice):
365 return None
366 if isinstance(expr, Symbol):
367 kconf = expr.kconfig
368 if expr is kconf.y:
369 return ("1", 3)
370 if expr is kconf.n:
371 return ("0", 3)
372 if expr.name in guard:
373 macro, token = guard[expr.name]
374 return (f"{macro} == {token}", 3)
375 if expr.choice is not None and expr.choice.name:
376 return (f"{expr.choice.name} == {expr.name}", 3)
377 return (expr.name, 3)
378 op = expr[0]
379 if op == NOT:
380 inner = _render_bool_expr(expr[1], guard)
381 if inner is None:
382 return None
383 text, prec = inner
384 if prec < 3 or " " in text:
385 text = f"({text})"
386 return (f"!{text}", 3)
387 if op in (AND, OR):
388 left = _render_bool_expr(expr[1], guard)
389 right = _render_bool_expr(expr[2], guard)
390 # A dropped Choice operand leaves the other side standing alone.
391 if left is None:
392 return right
393 if right is None:
394 return left
395 myprec = 2 if op == AND else 1
396 sep = "&&" if op == AND else "||"
397
398 def wrap(piece):
399 text, prec = piece
400 # parens for correctness, plus &&-groups inside || for clarity
401 if prec < myprec or (op == OR and prec == 2):
402 return f"({text})"
403 return text
404
405 return (f"{wrap(left)} {sep} {wrap(right)}", myprec)
406 # comparisons (==, <, ...): kconfiglib already renders C-compatible text.
407 return (expr_str(expr), 3)

410def rev_dep_c_expr(expr, guard=None) -> str | None:
411 """Render any bool expression as a C condition (the workhorse wrapper).
412
413 When the expression mentions a "pick one" group it drops the bare group
414 term (e.g. the anonymous Wayland backend choice) and keeps the rest, rather
415 than giving up. A named group member can be rewritten into
416 ``<macro> == <token>`` via *guard* - but only when that macro is
417 integer-valued (not the font-default choice; see
418 :func:`config_headers.emit.select_guards`). ``None`` only if the whole
419 thing renders to nothing."""
420 rendered = _render_bool_expr(expr, guard or {})
421 return rendered[0] if rendered else None

424def c_comment(text: str) -> list[str]:
425 """Wrap *text* (one or more lines) in a Doxygen ``/** ... */`` block."""
426 lines = [ln.rstrip() for ln in text.strip("\n").split("\n")]
427 if not lines or lines == [""]:
428 return []
429 if len(lines) == 1:
430 return [f"/** {lines[0]} */"]
431 out = [f"/** {lines[0]}"]
432 out += [f" * {ln}" if ln else " *" for ln in lines[1:]]
433 out += [" */"]
434 return out
