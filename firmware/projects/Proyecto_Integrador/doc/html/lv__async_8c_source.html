69lv_result_t lv_async_call_cancel(lv_async_cb_t async_xcb, void * user_data)
70{
71 LV_CHECK_ARG(async_xcb != NULL, return LV_RESULT_INVALID);
72
73 lv_timer_t * timer = lv_timer_get_next(NULL);
74 lv_result_t res = LV_RESULT_INVALID;
75
76 while(timer != NULL) {
77 /*Find the next timer node*/
78 lv_timer_t * timer_next = lv_timer_get_next(timer);
79
80 /*Find async timer callback*/
81 if(timer->timer_cb == lv_async_timer_cb) {
82 lv_async_info_t * info = (lv_async_info_t *)timer->user_data;
83
84 /*Match user function callback and user data*/
85 if(info->cb == async_xcb && info->user_data == user_data) {
86 lv_timer_delete(timer);
87 lv_free(info);
88 res = LV_RESULT_OK;
89 }
90 }
91
92 timer = timer_next;
93 }
94
95 return res;
