2"""Generate a Software Bill of Materials (SBOM) for LVGL in SPDX 3.0 (JSON-LD).
3
4The component data comes from a single source of truth:
5 sbom/third_party.json
6which is also used to regenerate COPYRIGHTS.md (see generate_copyrights.py).
7
8The LVGL version is read at run time from include/lvgl/lv_version.h so the
9document always matches the checked-out tree.
10
11Output: SPDX 3.0.1 JSON-LD written to the SBOM folder (default: ./sbom).
12
13Usage:
14 python3 scripts/generate_sbom.py [--output-dir sbom] [--date 2026-07-15]
15
16SPDX 3.0.1 spec: https://spdx.github.io/spdx-spec/v3.0.1/
17"""

34def read_version():
35 """Return the LVGL version string, e.g. '9.6.0-dev'."""
36 path = os.path.join(REPO_ROOT, "include", "lvgl", "lv_version.h")
37 with open(path, encoding="utf-8") as fh:
38 text = fh.read()
39
40 def grab(macro):
41 m = re.search(r"#define\s+%s\s+\"?([^\"\n]+)\"?" % macro, text)
42 return m.group(1).strip() if m else None
43
44 version = "%s.%s.%s" % (grab("LVGL_VERSION_MAJOR"),
45 grab("LVGL_VERSION_MINOR"),
46 grab("LVGL_VERSION_PATCH"))
47 info = grab("LVGL_VERSION_INFO")
48 if info:
49 version += "-" + info
50 return version

62def build_document(data, version, spec_version, created):
63 """Build the SPDX 3.0 JSON-LD document as a Python dict."""
64 project = data["project"]
65 # Third-party components only (LVGL's own code is part of the root package).
66 components = [c for c in data["components"] if c.get("third_party", True)]
67
68 ns = "%s/lvgl-%s" % (NS_BASE, version)
69
70 def sid(fragment):
71 return "%s#%s" % (ns, fragment)
72
73 creation_info = "_:creationinfo"
74 tool_id = sid("Tool-generate_sbom.py")
75 doc_id = sid("SPDXDocument")
76 sbom_id = sid("SBOM")
77 root_id = sid("Package-%s" % project["name"])
78
79 graph = []
80
81 # --- Agent registry ----------------------------------------------------
82 # Suppliers/creators (Person/Organization) are deduplicated by name and
83 # emitted once at the end of the graph.
84 agents = {} # name -> {"id": ..., "type": ...}
85
86 def agent_ref(supplier):
87 """Register a supplier {name, type} and return its SPDX element id."""
88 name = supplier["name"]
89 if name not in agents:
90 slug = re.sub(r"[^A-Za-z0-9]+", "-", name).strip("-")
91 kind = "Organization" if supplier.get("type") == "organization" else "Person"
92 agents[name] = {"id": sid("Agent-" + slug), "type": kind}
93 return agents[name]["id"]
94
95 creator_id = agent_ref({"name": "LVGL Kft", "type": "organization"})
96
97 # --- CreationInfo (blank node) -----------------------------------------
98 graph.append({
99 "type": "CreationInfo",
100 "@id": creation_info,
101 "specVersion": spec_version,
102 "created": created,
103 "createdBy": [creator_id],
104 "createdUsing": [tool_id],
105 })
106 graph.append({
107 "type": "Tool",
108 "spdxId": tool_id,
109 "creationInfo": creation_info,
110 "name": "scripts/generate_sbom.py",
111 })
112
113 # --- Licensing elements ------------------------------------------------
114 # Custom (non-SPDX-list) licenses referenced by a LicenseRef-* need an
115 # explicit CustomLicense element so the reference resolves in-document.
116 # Build these first so their URIs can be wired into the LicenseExpression
117 # elements below via simplelicensing_customIdToUri.
118 custom_license_uris = {} # "LicenseRef-*" -> CustomLicense element id
119 for comp in components:
120 ref = comp["license"]
121 if not ref.startswith("LicenseRef-"):
122 continue
123 if ref in custom_license_uris:
124 continue
125 license_file = comp.get("license_file")
126 if not license_file:
127 raise SystemExit(
128 "Custom license %r (component %r) needs a 'license_file' in "
129 "third_party.json so its text can be embedded." % (ref, comp["key"]))
130 text = read_file(license_file)
131 if text is None:
132 raise SystemExit(
133 "License file %r for custom license %r (component %r) is "
134 "missing; cannot embed its text." % (license_file, ref, comp["key"]))
135 cid = sid("CustomLicense-" + re.sub(r"[^A-Za-z0-9]+", "_", ref))
136 custom_license_uris[ref] = cid
137 graph.append({
138 "type": "expandedlicensing_CustomLicense",
139 "spdxId": cid,
140 "creationInfo": creation_info,
141 "name": ref,
142 # Schema requires the license text under simplelicensing_licenseText.
143 "simplelicensing_licenseText": text,
144 "expandedlicensing_isOsiApproved": False,
145 })
146
147 # One LicenseExpression element per distinct expression; reused by relationships.
148 all_licenses = sorted({c["license"] for c in components} | {project["license"]})
149 license_expr_ids = {}
150 for expr in all_licenses:
151 frag = "LicenseExpression-" + re.sub(r"[^A-Za-z0-9.]+", "_", expr)
152 lid = sid(frag)
153 license_expr_ids[expr] = lid
154 element = {
155 "type": "simplelicensing_LicenseExpression",
156 "spdxId": lid,
157 "creationInfo": creation_info,
158 "simplelicensing_licenseExpression": expr,
159 }
160 # Map any LicenseRef-* tokens in this expression to their CustomLicense
161 # element so consumers can resolve the custom license text.
162 id_to_uri = [
163 {"type": "DictionaryEntry", "key": ref, "value": uri}
164 for ref, uri in sorted(custom_license_uris.items())
165 if re.search(r"\b%s\b" % re.escape(ref), expr)
166 ]
167 if id_to_uri:
168 element["simplelicensing_customIdToUri"] = id_to_uri
169 graph.append(element)
170
171 # --- LVGL root package -------------------------------------------------
172 graph.append({
173 "type": "software_Package",
174 "spdxId": root_id,
175 "creationInfo": creation_info,
176 "name": project["name"],
177 "summary": project.get("summary", ""),
178 "software_packageVersion": version,
179 "software_copyrightText": project.get("copyright", "NOASSERTION"),
180 "software_downloadLocation": project["download_location"],
181 "software_homePage": project.get("homepage", project["download_location"]),
182 "software_primaryPurpose": "library",
183 "suppliedBy": agent_ref(project["supplier"]),
184 "externalIdentifier": [{
185 "type": "ExternalIdentifier",
186 "externalIdentifierType": "packageUrl",
187 "identifier": "pkg:github/lvgl/lvgl@%s" % version,
188 }],
189 })
190
191 # --- Third-party component packages ------------------------------------
192 component_ids = []
193 relationships = []
194 for comp in components:
195 cid = sid("Package-" + comp["key"])
196 component_ids.append(cid)
197
198 sources = comp.get("sources", [])
199 source_urls = [s["url"] for s in sources]
200 comment = "In-tree path: %s" % ", ".join(comp["paths"])
201 if comp.get("note"):
202 comment += " | " + comp["note"]
203 # The first source is the SPDX download location (below); preserve any
204 # additional upstream sources in the comment so they are not dropped.
205 for extra in sources[1:]:
206 suffix = " (%s)" % extra["note"] if extra.get("note") else ""
207 comment += " | Additional source: %s%s" % (extra["url"], suffix)
208
209 pkg = {
210 "type": "software_Package",
211 "spdxId": cid,
212 "creationInfo": creation_info,
213 "name": comp["name"],
214 "software_packageVersion": comp.get("version", "NOASSERTION"),
215 "software_copyrightText": comp.get("copyright", "NOASSERTION"),
216 "software_primaryPurpose": comp.get("purpose", "library"),
217 "comment": comment,
218 }
219 if comp.get("supplier"):
220 pkg["suppliedBy"] = agent_ref(comp["supplier"])
221 if source_urls:
222 pkg["software_downloadLocation"] = source_urls[0]
223 pkg["software_homePage"] = source_urls[0]
224 if comp.get("purl"):
225 purl = comp["purl"]
226 # Note: use a distinct name here — do not reuse `version`, which
227 # holds the LVGL document version used further below.
228 comp_version = comp.get("version", "NOASSERTION")
229 # Qualify the purl with the vendored version when one is known and
230 # the purl does not already carry a version/qualifier of its own.
231 if comp_version and comp_version != "NOASSERTION" and "@" not in purl and "?" not in purl:
232 purl = "%s@%s" % (purl, comp_version)
233 pkg["externalIdentifier"] = [{
234 "type": "ExternalIdentifier",
235 "externalIdentifierType": "packageUrl",
236 "identifier": purl,
237 }]
238 graph.append(pkg)
239
240 # lvgl vendors (contains) the third-party copy. We deliberately do not
241 # emit a blanket `dependsOn` here: many of these libraries are optional
242 # (gated behind LV_USE_* toggles that are off by default), so claiming
243 # lvgl depends on all of them would overstate the dependency graph.
244 relationships.append({
245 "type": "Relationship",
246 "spdxId": sid("Relationship-contains-%s" % comp["key"]),
247 "creationInfo": creation_info,
248 "from": root_id,
249 "relationshipType": "contains",
250 "to": [cid],
251 })
252 # component license (declared == concluded for a vendored copy)
253 for rel_type in ("hasDeclaredLicense", "hasConcludedLicense"):
254 relationships.append({
255 "type": "Relationship",
256 "spdxId": sid("Relationship-%s-%s" % (rel_type, comp["key"])),
257 "creationInfo": creation_info,
258 "from": cid,
259 "relationshipType": rel_type,
260 "to": [license_expr_ids[comp["license"]]],
261 })
262
263 # lvgl's own license
264 for rel_type in ("hasDeclaredLicense", "hasConcludedLicense"):
265 relationships.append({
266 "type": "Relationship",
267 "spdxId": sid("Relationship-%s-%s" % (rel_type, project["name"])),
268 "creationInfo": creation_info,
269 "from": root_id,
270 "relationshipType": rel_type,
271 "to": [license_expr_ids[project["license"]]],
272 })
273
274 graph.extend(relationships)
275
276 # --- Agent elements (creator + package suppliers) ----------------------
277 for name, info in agents.items():
278 graph.append({
279 "type": info["type"],
280 "spdxId": info["id"],
281 "creationInfo": creation_info,
282 "name": name,
283 })
284
285 # --- SBOM element ------------------------------------------------------
286 # Every element built so far (packages, licenses, relationships, agents,
287 # tool) is part of the SBOM's contents, so list them all here. Otherwise a
288 # consumer starting from the SBOM root cannot reach the relationships.
289 sbom_elements = [e["spdxId"] for e in graph if e.get("spdxId")]
290 graph.append({
291 "type": "software_Sbom",
292 "spdxId": sbom_id,
293 "creationInfo": creation_info,
294 "software_sbomType": ["source"],
295 "rootElement": [root_id],
296 "element": sbom_elements,
297 })
298
299 # --- SpdxDocument (wraps everything) -----------------------------------
300 doc_elements = [e.get("spdxId") for e in graph if e.get("spdxId")]
301 graph.append({
302 "type": "SpdxDocument",
303 "spdxId": doc_id,
304 "creationInfo": creation_info,
305 "name": "LVGL %s SBOM" % version,
306 "profileConformance": ["core", "software", "simpleLicensing", "expandedLicensing"],
307 "rootElement": [sbom_id],
308 "element": doc_elements,
309 })
310
311 document = {
312 "@context": "https://spdx.org/rdf/%s/spdx-context.jsonld" % spec_version,
313 "@graph": graph,
314 }
315 return document

322def _normalized(document):
323 """Copy with the volatile creation timestamp blanked, for drift comparison."""
324 doc = json.loads(json.dumps(document))
325 for element in doc.get("@graph", []):
326 if element.get("type") == "CreationInfo":
327 element["created"] = "<normalized>"
328 return _serialize(doc)

331def main():
332 parser = argparse.ArgumentParser(description="Generate an SPDX 3.0 SBOM for LVGL.")
333 parser.add_argument("--output-dir", default=os.path.join(REPO_ROOT, "sbom"),
334 help="Directory to write the SBOM into (default: ./sbom).")
335 parser.add_argument("--date", default=None,
336 help="Creation date (YYYY-MM-DD). Defaults to today (UTC).")
337 parser.add_argument("--spec-version", default="3.0.1",
338 help="SPDX specification version (default: 3.0.1).")
339 parser.add_argument("--check", action="store_true",
340 help="Exit non-zero if the committed SBOM is out of date "
341 "(ignores the creation timestamp).")
342 args = parser.parse_args()
343
344 with open(DATA_FILE, encoding="utf-8") as fh:
345 data = json.load(fh)
346
347 version = read_version()
348 # The file name carries no version: the SBOM is bundled inside a given LVGL
349 # version anyway, and the version is recorded inside the document.
350 out_path = os.path.join(args.output_dir, "lvgl.spdx.json")
351
352 if args.date:
353 # strptime alone accepts non-zero-padded input like "2026-7-1", which
354 # would produce an invalid SPDX timestamp, so enforce the exact shape.
355 if not re.fullmatch(r"\d{4}-\d{2}-\d{2}", args.date):
356 parser.error("--date must be in YYYY-MM-DD format (zero-padded)")
357 try:
358 datetime.datetime.strptime(args.date, "%Y-%m-%d")
359 except ValueError:
360 parser.error("--date must be a valid date in YYYY-MM-DD format")
361 created = "%sT00:00:00Z" % args.date
362 else:
363 created = datetime.datetime.now(datetime.timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ")
364
365 document = build_document(data, version, args.spec_version, created)
366 components = [c for c in data["components"] if c.get("third_party", True)]
367
368 if args.check:
369 if not os.path.isfile(out_path):
370 print("%s is missing; run scripts/generate_sbom.py" % out_path)
371 return 1
372 with open(out_path, encoding="utf-8") as fh:
373 current = fh.read()
374 if _normalized(json.loads(current)) != _normalized(document):
375 print("%s is out of date; run scripts/generate_sbom.py" % out_path)
376 return 1
377 print("%s is up to date" % out_path)
378 return 0
379
380 os.makedirs(args.output_dir, exist_ok=True)
381 with open(out_path, "w", encoding="utf-8") as fh:
382 fh.write(_serialize(document))
383
384 print("Wrote SPDX %s SBOM for LVGL %s" % (args.spec_version, version))
385 print(" %s" % out_path)
386 print(" %d third-party components, %d SPDX elements"
387 % (len(components), len(document["@graph"])))
388 return 0
