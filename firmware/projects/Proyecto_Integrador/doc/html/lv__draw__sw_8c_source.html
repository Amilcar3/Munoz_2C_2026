74void lv_draw_sw_init(void)
75{
76
77#if LV_DRAW_SW_COMPLEX == 1
78 lv_draw_sw_mask_init();
79#endif
80
81 lv_draw_sw_unit_t * draw_sw_unit = lv_draw_create_unit(sizeof(lv_draw_sw_unit_t));
82 draw_sw_unit->base_unit.dispatch_cb = dispatch;
83 draw_sw_unit->base_unit.evaluate_cb = evaluate;
84#if LV_USE_OS != LV_OS_NONE
85 draw_sw_unit->base_unit.wait_for_finish_cb = wait_for_finish;
86#endif /*LV_USE_OS*/
87 draw_sw_unit->base_unit.delete_cb = LV_USE_OS ? lv_draw_sw_delete : NULL;
88
89#if LV_USE_DRAW_SW_ASM == LV_DRAW_SW_ASM_NEON
90 draw_sw_unit->base_unit.name = "SW_NEON";
91#elif LV_USE_DRAW_SW_ASM == LV_DRAW_SW_ASM_HELIUM && LV_USE_DRAW_ARM2D_SYNC
92 draw_sw_unit->base_unit.name = "SW_ARM2D";
93#elif LV_USE_DRAW_SW_ASM == LV_DRAW_SW_ASM_HELIUM
94 draw_sw_unit->base_unit.name = "SW_HELIUM";
95#elif LV_USE_DRAW_SW_ASM == LV_DRAW_SW_ASM_RISCV_V
96 draw_sw_unit->base_unit.name = "SW_RISCV_V";
97#elif LV_USE_DRAW_SW_ASM == LV_DRAW_SW_ASM_SVE2
98 draw_sw_unit->base_unit.name = "SW_SVE2";
99#elif LV_USE_DRAW_SW_ASM == LV_DRAW_SW_ASM_CUSTOM
100 draw_sw_unit->base_unit.name = "SW_CUSTOM";
101#else
102 draw_sw_unit->base_unit.name = "SW";
103#endif
104
105#if LV_USE_OS
106 uint32_t i;
107 for(i = 0; i < LV_DRAW_SW_DRAW_UNIT_CNT; i++) {
108 lv_draw_sw_thread_dsc_t * thread_dsc = &draw_sw_unit->thread_dscs[i];

112 lv_thread_sync_init(&thread_dsc->task_done_sync);
113 lv_thread_init(&thread_dsc->thread, "swdraw", LV_DRAW_THREAD_PRIO, render_thread_cb,
114 LV_DRAW_THREAD_STACK_SIZE, thread_dsc);
115 }
116#endif
117
118#if LV_USE_VECTOR_GRAPHIC && LV_USE_THORVG
119 if(LV_DRAW_SW_DRAW_UNIT_CNT > 1) {
120 tvg_engine_init(TVG_ENGINE_SW, LV_DRAW_SW_DRAW_UNIT_CNT);
121 }
122 else {
123 tvg_engine_init(TVG_ENGINE_SW, 0);
124 }
125#endif
126
127 lv_ll_init(&LV_GLOBAL_DEFAULT()->draw_sw_blend_handler_ll, sizeof(lv_draw_sw_custom_blend_handler_t));

166bool lv_draw_sw_register_blend_handler(lv_draw_sw_custom_blend_handler_t * handler)
167{
168 lv_draw_sw_custom_blend_handler_t * existing_handler = NULL;
169 lv_draw_sw_custom_blend_handler_t * new_handler = NULL;
170
171 // Check if a handler is already registered for the color format
172 LV_LL_READ(&LV_GLOBAL_DEFAULT()->draw_sw_blend_handler_ll, existing_handler) {
173 if(existing_handler->dest_cf == handler->dest_cf) {
174 new_handler = existing_handler;
175 break;
176 }
177 }
178
179 if(new_handler == NULL) {
180 new_handler = lv_ll_ins_head(&LV_GLOBAL_DEFAULT()->draw_sw_blend_handler_ll);
181 if(new_handler == NULL) {
182 LV_ASSERT_MALLOC(new_handler);
183 return false;
184 }
185 }
186
187 lv_memcpy(new_handler, handler, sizeof(lv_draw_sw_custom_blend_handler_t));
188 return true;
189}

223static int32_t evaluate(lv_draw_unit_t * draw_unit, lv_draw_task_t * task)
224{
225 LV_UNUSED(draw_unit);
226
227 switch(task->type) {
228 case LV_DRAW_TASK_TYPE_IMAGE:
229 case LV_DRAW_TASK_TYPE_LAYER: {
230 lv_draw_image_dsc_t * draw_dsc = task->draw_dsc;
231
232 /* not support skew */
233 if(draw_dsc->skew_x != 0 || draw_dsc->skew_y != 0) {
234 return 0;
235 }
236
237 bool masked = draw_dsc->bitmap_mask_src != NULL;
238
239 lv_color_format_t cf = draw_dsc->header.cf;
240 if(masked && (cf == LV_COLOR_FORMAT_A8 || cf == LV_COLOR_FORMAT_RGB565A8)) {
241 return 0;
242 }
243
244 if(cf >= LV_COLOR_FORMAT_PROPRIETARY_START) {
245 return 0;
246 }
247 }
248 break;
249#if LV_USE_3DTEXTURE
250 case LV_DRAW_TASK_TYPE_3D:
251 return 0;
252#endif
253 default:
254 break;
255 }
256
257 if(task->preference_score >= 100) {
258 task->preference_score = 100;
259 task->preferred_draw_unit_id = DRAW_UNIT_ID_SW;
260 }
261
262 return 0;
263}

265static int32_t dispatch(lv_draw_unit_t * draw_unit, lv_layer_t * layer)
266{
267 LV_PROFILER_DRAW_BEGIN;
268 lv_draw_sw_unit_t * draw_sw_unit = (lv_draw_sw_unit_t *) draw_unit;
269
270#if LV_USE_OS
271 uint32_t i;
272 uint32_t taken_cnt = 0;
273 /* All idle (couldn't take any tasks): return LV_DRAW_UNIT_IDLE;
274 * All busy: return 0; as 0 tasks were taken
275 * Otherwise return taken_cnt;
276 */
277
278 /*If at least one is busy, it's not all idle*/
279 bool all_idle = true;
280 for(i = 0; i < LV_DRAW_SW_DRAW_UNIT_CNT; i++) {
281 if(draw_sw_unit->thread_dscs[i].task_act) {
282 all_idle = false;
283 break;
284 }
285 }
286
287 lv_draw_task_t * t = NULL;
288 for(i = 0; i < LV_DRAW_SW_DRAW_UNIT_CNT; i++) {
289 lv_draw_sw_thread_dsc_t * thread_dsc = &draw_sw_unit->thread_dscs[i];
290
291 /*Do nothing if busy*/
292 if(thread_dsc->task_act) continue;
293
294 /*Find an available task. Start from the previously taken task.*/
295 t = lv_draw_get_next_available_task(layer, t, DRAW_UNIT_ID_SW);
296
297 /*If there is not available task don't try other threads as there won't be available
298 *tasks for then either*/
299 if(t == NULL) {
300 LV_PROFILER_DRAW_END;
301 if(all_idle) return LV_DRAW_UNIT_IDLE; /*Couldn't start rendering*/
302 else return taken_cnt;
303 }
304
305 /*Allocate a buffer if not done yet.*/
306 void * buf = lv_draw_layer_alloc_buf(layer);
307 /*Do not return is failed. The other thread might already have a buffer can do something. */
308 if(buf == NULL) {
309 t->state = LV_DRAW_TASK_STATE_FAILED;
310 continue;
311 }
312
313 /*Take the task*/
314 all_idle = false;
315 taken_cnt++;
316 t->state = LV_DRAW_TASK_STATE_IN_PROGRESS;
317 thread_dsc->task_act = t;
318
319 /*Let the render thread work*/
320 lv_thread_sync_signal(&thread_dsc->new_task_sync);
321 }
322
323 LV_PROFILER_DRAW_END;
324 if(all_idle) return LV_DRAW_UNIT_IDLE; /*Couldn't start rendering*/
325 else return taken_cnt;
326
327#else
328 /*Return immediately if it's busy with draw task*/
329 if(draw_sw_unit->task_act) {
330 LV_PROFILER_DRAW_END;
331 return 0;
332 }
333
334 lv_draw_task_t * t = NULL;
335 t = lv_draw_get_available_task(layer, NULL, DRAW_UNIT_ID_SW);
336 if(t == NULL) {
337 LV_PROFILER_DRAW_END;
338 return LV_DRAW_UNIT_IDLE; /*Couldn't start rendering*/
339 }
340
341 void * buf = lv_draw_layer_alloc_buf(layer);
342 if(buf == NULL) {
343 t->state = LV_DRAW_TASK_STATE_FAILED;
344 LV_PROFILER_DRAW_END;
345 return LV_DRAW_UNIT_IDLE; /*Couldn't start rendering*/
346 }
347
348 t->state = LV_DRAW_TASK_STATE_IN_PROGRESS;
349 draw_sw_unit->task_act = t;
350
351 execute_drawing(t);
352 draw_sw_unit->task_act->state = LV_DRAW_TASK_STATE_FINISHED;
353 draw_sw_unit->task_act = NULL;
354
355 /*The draw unit is free now. Request a new dispatching as it can get a new task*/
356 lv_draw_dispatch_request();
357
358 LV_PROFILER_DRAW_END;
359 return 1;
360#endif
361
362}

364#if LV_USE_OS
365static void render_thread_cb(void * ptr)
366{
367 lv_draw_sw_thread_dsc_t * thread_dsc = ptr;
368
369 while(1) {
370 while(thread_dsc->task_act == NULL) {
371 if(thread_dsc->exit_status) {
372 break;
373 }
374 lv_thread_sync_wait(&thread_dsc->new_task_sync);
375 }
376
377 if(thread_dsc->exit_status) {
378 LV_LOG_INFO("ready to exit software rendering thread");
379 break;
380 }
381
382 execute_drawing((lv_draw_task_t *)thread_dsc->task_act);
383#if LV_USE_PARALLEL_DRAW_DEBUG
384 parallel_debug_draw(thread_dsc->task_act, thread_dsc->idx);
385#endif
386 thread_dsc->task_act->state = LV_DRAW_TASK_STATE_FINISHED;
387 thread_dsc->task_act = NULL;
388
389 lv_thread_sync_signal(&thread_dsc->task_done_sync);
390 /*The draw unit is free now. Request a new dispatching as it can get a new task*/
391 lv_draw_dispatch_request();
392
393 }
394
395 LV_LOG_INFO("exit software rendering thread");
396}

411static void execute_drawing(lv_draw_task_t * t)
412{
413 LV_PROFILER_DRAW_BEGIN;
414 /*Render the draw task*/
415 switch(t->type) {
416 case LV_DRAW_TASK_TYPE_FILL:
417 lv_draw_sw_fill(t, t->draw_dsc, &t->area);
418 break;
419 case LV_DRAW_TASK_TYPE_BORDER:
420 lv_draw_sw_border(t, t->draw_dsc, &t->area);
421 break;
422 case LV_DRAW_TASK_TYPE_BOX_SHADOW:
423 lv_draw_sw_box_shadow(t, t->draw_dsc, &t->area);
424 break;
425 case LV_DRAW_TASK_TYPE_LETTER:
426 lv_draw_sw_letter(t, t->draw_dsc, &t->area);
427 break;
428 case LV_DRAW_TASK_TYPE_LABEL:
429 lv_draw_sw_label(t, t->draw_dsc, &t->area);
430 break;
431 case LV_DRAW_TASK_TYPE_IMAGE:
432 lv_draw_sw_image(t, t->draw_dsc, &t->area);
433 break;
434 case LV_DRAW_TASK_TYPE_ARC:
435 lv_draw_sw_arc(t, t->draw_dsc, &t->area);
436 break;
437 case LV_DRAW_TASK_TYPE_LINE:
438 lv_draw_line_iterate(t, t->draw_dsc, lv_draw_sw_line);
439 break;
440 case LV_DRAW_TASK_TYPE_BLUR:
441 lv_draw_sw_blur(t, t->draw_dsc, &t->area);
442 break;
443 case LV_DRAW_TASK_TYPE_TRIANGLE:
444 lv_draw_sw_triangle(t, t->draw_dsc);
445 break;
446 case LV_DRAW_TASK_TYPE_LAYER:
447 lv_draw_sw_layer(t, t->draw_dsc, &t->area);
448 break;
449 case LV_DRAW_TASK_TYPE_MASK_RECTANGLE:
450 lv_draw_sw_mask_rect(t, t->draw_dsc);
451 break;
452#if LV_USE_VECTOR_GRAPHIC && LV_USE_THORVG
453 case LV_DRAW_TASK_TYPE_VECTOR:
454 lv_draw_sw_vector(t, t->draw_dsc);
455 break;
456#endif
457 default:
458 break;
459 }
460
461
462 LV_PROFILER_DRAW_END;
463}
