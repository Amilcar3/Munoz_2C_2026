68static void * fs_open(lv_fs_drv_t * drv, const char * path, lv_fs_mode_t mode)
69{
70 LV_ASSERT(path != NULL);
71 LV_UNUSED(drv);
72
73 const char * flags;
74 if(mode == LV_FS_MODE_WR)
75 flags = FILE_WRITE;
76 else if(mode == LV_FS_MODE_RD)
77 flags = FILE_READ;
78 else if(mode == (LV_FS_MODE_WR | LV_FS_MODE_RD))
79 flags = FILE_WRITE;
80 else {
81 LV_LOG_WARN("Unsupported mode: 0x%x", (unsigned)mode);
82 return NULL;
83 }
84
85 char buf[LV_FS_MAX_PATH_LEN];
86 lv_snprintf(buf, sizeof(buf), LV_FS_ARDUINO_ESP_LITTLEFS_PATH "%s", path);
87
88 File file = LittleFS.open(buf, flags);
89 if(!file) {
90 return NULL;
91 }
92
93 ArduinoEspLittleFile * lf = new ArduinoEspLittleFile{file};
94
95 return (void *)lf;
96}
