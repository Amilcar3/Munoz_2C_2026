1#!/usr/bin/env python3
2"""Clean up generated LVGL example C files.
3
4Generator output (`screenN_gen.c`) is verbose — section-banner comments, an
5empty `style_inited` block, `LV_TRACE_OBJ_CREATE` traces, redundant
6`lv_obj_set_name` calls, etc. — and this script post-processes those files
7into the hand-written-looking examples we ship in `examples/`.
8
9Each `.c` file that has a sibling `.xml` of the same basename is run through
10`TRANSFORMATIONS` in order; each transform is `(source, path) -> str` and
11pure, so the runner only writes back when the final text differs. Every
12transform is idempotent — safe to re-run on already-processed output without
13duplicating content.
14"""
15

39# =============================================================================
40# Generator boilerplate removal
41# =============================================================================
42
43# Matches the empty style-init scaffold the generator always emits:
44#
45# static bool style_inited = false;
46#
47# if (!style_inited) {
48# <only whitespace>
49# style_inited = true;
50# }
51#
52# When a real style-init block has body content we leave it alone; the inner
53# `(?:[ \t]*\n)*` only consumes blank lines between brace and `style_inited =
54# true;`. The trailing `(?:[ \t]*\n)?` eats one blank line below so we don't
55# leave a yawning gap after removal.
56EMPTY_STYLE_INITED_RE = re.compile(
57 r"""
58 [ \t]*static\ bool\ style_inited\ =\ false;[ \t]*\n
59 (?:[ \t]*\n)*
60 [ \t]*if\ \‍(!style_inited\‍)\ \{[ \t]*\n
61 (?:[ \t]*\n)*
62 [ \t]*style_inited\ =\ true;[ \t]*\n
63 [ \t]*\}[ \t]*\n
64 (?:[ \t]*\n)?
65 """,
66 re.VERBOSE,
67)
68
69
70def remove_empty_style_inited(source: str) -> str:
71 return EMPTY_STYLE_INITED_RE.sub("", source)

129# Replace the generator's `lv_obj_create(NULL)` root with `lv_screen_active()`
130# and rename all references to that root (default name `lv_obj_0`) to `screen`.
131ROOT_OBJ_CREATE_RE = re.compile(
132 r"lv_obj_t\s*\*\s*lv_obj_0\s*=\s*lv_obj_create\s*\‍(\s*NULL\s*\‍)\s*;"

137def root_obj_to_screen(source: str) -> str:
138 source = ROOT_OBJ_CREATE_RE.sub("lv_obj_t * screen = lv_screen_active();", source)
139 return LV_OBJ_0_REF_RE.sub("screen", source)

142# Argument-content pattern that tolerates one level of nested parens, so we
143# can match values like `lv_pct(100)` without truncating at the inner `)`.
144# Without this, `[^)]+` would stop at the first `)`.
145_ARG_PAT = r"(?:[^()]|\‍([^)]*\‍))*"
146
147# Adjacent width/height pair on the same object → `lv_obj_set_size(obj, W, H)`.
148# Group 1 = indentation, group 2 = object name (must match on both lines via
149# the backreference `\2`), groups 3/4 = the width/height values.
150WIDTH_HEIGHT_PAIR_RE = re.compile(
151 rf"([ \t]*)lv_obj_set_width\‍(\s*([A-Za-z_]\w*)\s*,\s*({_ARG_PAT})\‍)\s*;[ \t]*\n"
152 rf"\1lv_obj_set_height\‍(\s*\2\s*,\s*({_ARG_PAT})\‍)\s*;"
153)
154
155# Same shape, for x/y → `lv_obj_set_pos`. Note: we don't handle the reverse
156# order (height-then-width or y-then-x) because the generator's output is
157# consistent. Add a sibling regex if that assumption breaks.
158X_Y_PAIR_RE = re.compile(
159 rf"([ \t]*)lv_obj_set_x\‍(\s*([A-Za-z_]\w*)\s*,\s*({_ARG_PAT})\‍)\s*;[ \t]*\n"
160 rf"\1lv_obj_set_y\‍(\s*\2\s*,\s*({_ARG_PAT})\‍)\s*;"
161)
162
163
164def combine_size_and_pos(source: str) -> str:
165 source = WIDTH_HEIGHT_PAIR_RE.sub(r"\1lv_obj_set_size(\2, \3, \4);", source)
166 source = X_Y_PAIR_RE.sub(r"\1lv_obj_set_pos(\2, \3, \4);", source)
167 return source

184# `lv_obj_t * <name>_create(void)` → `void <file_stem>(void)`.
185# The generator names the entry function either by screen index
186# (`screenN_create`, older builds) or by the screen XML's filename
187# (`<file_stem>_create`, current builds). The broader `\w+_create` match
188# handles both. We drop the `_create` suffix entirely so example functions
189# read as `void lv_example_<feature>(void)` — matching the legacy C-only
190# examples. The new return type is `void` because `remove_return_screen`
191# already dropped the return statement — leaving `lv_obj_t *` would emit a
192# function declared to return a pointer with no return statement.
193CREATE_FUNC_DECL_RE = re.compile(
194 r"lv_obj_t\s*\*\s*\w+_create\s*\‍(\s*void\s*\‍)"

227# =============================================================================
228# XML doc mapping
229# =============================================================================
230#
231# The sibling `.xml` carries semantic comments that the generator drops:
232#
233# * A top-level `<!-- ... -->` block (typically with `@title`/`@brief`)
234# before `<screen>` — becomes a doxygen `/** ... */` block above the
235# create function.
236# * Inline `<!-- ... -->` before each element under `<view>` — becomes a
237# `/* ... */` comment above the matching `lv_<type>_create(...)` line in
238# the C file.
239
240# Match a `lv_obj_t * <name> = lv_<type>_create(...)` line so we can map
241# XML annotations to the right C creation by tag/order. Group 1 = indent,
242# group 2 = the `lv_<type>` token (e.g. `lv_arc`, `lv_label`).
243CREATE_LINE_RE = re.compile(
244 r"([ \t]*)lv_obj_t\s*\*\s*\w+\s*=\s*(lv_\w+)_create\s*\‍("
245)

248def _xml_elements_with_comments(xml_path: Path) -> list[tuple[str, str | None]]:
249 """Return (tag, preceding_comment) for each XML element in document order.
250
251 A comment binds to the next element that follows it, regardless of
252 nesting depth (a comment inside `<view>` followed by `<lv_arc>` annotates
253 that arc; a comment inside `<lv_arc>` followed by `<lv_label>` annotates
254 that label).
255
256 Top-level comments outside the root (e.g. before `<screen>`) are NOT
257 returned here — those are handled separately by
258 `add_top_level_doc_comment` via a regex on the raw text, because
259 `ET.parse` discards everything outside the root.
260 """
261 parser = ET.XMLParser(target=ET.TreeBuilder(insert_comments=True))
262 tree = ET.parse(xml_path, parser=parser)
263 root = tree.getroot()
264
265 annotations: list[tuple[str, str | None]] = []
266 pending: str | None = None
267
268 def walk(elem):
269 nonlocal pending
270 for child in elem:
271 if child.tag is ET.Comment:
272 # Comments get stored, awaiting the next non-comment element.
273 pending = (child.text or "").strip()
274 else:
275 annotations.append((child.tag, pending))
276 pending = None
277 # Recurse so nested elements are emitted in document order.
278 walk(child)
279
280 walk(root)
281 return annotations

284# A comment block at the very start of the XML file (before any element).
285# Used as the source for the per-function doxygen block.
286TOP_LEVEL_XML_COMMENT_RE = re.compile(r"\A\s*<!--(.*?)-->", re.DOTALL)
287
288# Locates the `void <name>_create(void)` line so we can insert the doxygen
289# block immediately above it.
290FUNCTION_DECL_RE = re.compile(
291 r"^void\s+lv_example_\w+\s*\‍(\s*void\s*\‍)", re.MULTILINE
292)
293
294# Matches a single `/** ... */` block at the very end of a substring (we slice
295# `source[:fn_start]` and search this against it). Important nuance: the inner
296# pattern `(?:[^*]|\*(?!/))*` refuses to cross a closing `*/`. Without that
297# guard, the more obvious `[\s\S]*?\*/` combined with the trailing `\Z`
298# anchor will silently expand across multiple adjacent doc blocks (because
299# non-greedy backtracks until the `\Z` anchor is satisfied) — which used to
300# eat the `@file` header along with the `@title` block above the function.
301EXISTING_DOC_BLOCK_RE = re.compile(
302 r"/\*\*(?:[^*]|\*(?!/))*\*/[ \t]*\n\s*\Z"
303)
304
305
306def _format_doxygen_block(body: str) -> str:
307 """Wrap a multi-line plain-text body into `/** * ... */` form.
308
309 Blank lines become bare ` *` (so doxygen renders them as paragraph
310 breaks); non-blank lines are reindented under ` * `.
311 """
312 out = ["/**"]
313 for line in body.splitlines():
314 stripped = line.strip()
315 out.append(f" * {stripped}" if stripped else " *")
316 out.append(" */")
317 return "\n".join(out)

320def add_top_level_doc_comment(source: str, path: Path) -> str:
321 """Lift the XML's top-level comment into a doxygen block above the function.
322
323 Idempotent: if a `@title`/`@brief`-bearing block already sits right above
324 the function, it gets replaced rather than stacked. The (separate) `@file`
325 header at the top of the file is preserved because `EXISTING_DOC_BLOCK_RE`
326 can't span across multiple doc blocks (see the note on that regex).
327 """
328 xml_path = path.with_suffix(".xml")
329 if not xml_path.exists():
330 return source
331
332 m = TOP_LEVEL_XML_COMMENT_RE.match(xml_path.read_text())
333 if not m or not m.group(1).strip():
334 return source
335
336 comment_block = _format_doxygen_block(m.group(1).strip()) + "\n"
337
338 fn_match = FUNCTION_DECL_RE.search(source)
339 if not fn_match:
340 return source
341 fn_start = fn_match.start()
342
343 # If a pre-existing @title/@brief doc block sits right above the function,
344 # strip it before inserting the fresh one (so re-runs replace, not stack).
345 # The "@title"/"@brief" content check ensures we only touch *our* block —
346 # not, say, a legit doxygen comment the user wrote about the function.
347 before_fn = source[:fn_start]
348 existing = EXISTING_DOC_BLOCK_RE.search(before_fn)
349 if existing and ("@title" in existing.group(0) or "@brief" in existing.group(0)):
350 source = source[: existing.start()] + source[fn_start:]
351 fn_start = existing.start()
352
353 return source[:fn_start] + comment_block + source[fn_start:]

356def _format_inline_comment(comment: str, indent: str) -> list[str]:
357 """Format an XML comment as 1+ properly indented C block-comment lines.
358
359 The XML source preserves whatever whitespace sat between `<!--` and `-->`
360 (typically tabs from nested XML indentation). Emitting `/* {comment} */`
361 verbatim leaves continuation lines stuck with XML indentation, which
362 looks broken in C source. We strip each line and re-indent continuations
363 under the start of the first comment word (3 spaces past `indent` to
364 clear the opening `/* `).
365 """
366 lines = [ln.strip() for ln in comment.splitlines()]
367 # Drop blank leading/trailing lines so re-indent doesn't create stray
368 # empty comment rows.
369 while lines and not lines[0]:
370 lines.pop(0)
371 while lines and not lines[-1]:
372 lines.pop()
373 if not lines:
374 return []
375 if len(lines) == 1:
376 return [f"{indent}/* {lines[0]} */\n"]
377 cont = indent + " "
378 out = [f"{indent}/* {lines[0]}\n"]
379 for ln in lines[1:-1]:
380 out.append(f"{cont}{ln}\n")
381 out.append(f"{cont}{lines[-1]} */\n")
382 return out

385def _strip_preceding_comment(out_lines: list[str]) -> None:
386 """Pop any single-line `//` or `/* */` block comment at the tail of out_lines.
387
388 Used by `map_xml_comments` for idempotency: re-running the cleanup
389 should replace a stale comment block (e.g. an earlier misindented copy,
390 or the legacy `// ...` form), not stack a new one on top of it.
391 """
392 if not out_lines:
393 return
394 last = out_lines[-1].rstrip()
395 if last.lstrip().startswith("//"):
396 out_lines.pop()
397 return
398 if last.endswith("*/"):
399 # Walk back to the line that opens the block; both single-line and
400 # multi-line `/* */` shapes collapse to this.
401 i = len(out_lines) - 1
402 while i >= 0 and "/*" not in out_lines[i]:
403 i -= 1
404 if i >= 0:
405 del out_lines[i:]

408def map_xml_comments(source: str, path: Path) -> str:
409 """Map per-element XML comments to `/* */` comments above C creations.
410
411 Pairing is by sequential order: the Nth `lv_<type>_create(...)` line in
412 the C file is associated with the Nth `lv_*` element in the XML's
413 document order (after filtering out wrapper tags like `<view>`).
414
415 The tag is verified to match (e.g. an `lv_arc` create line only consumes
416 an `lv_arc` annotation) — a mismatch likely means the C file diverged
417 from the XML, in which case we'd want to bail rather than misattribute.
418 Currently we silently skip mismatches; tighten this if the script starts
419 inserting comments in wrong places.
420
421 Idempotent across comment styles and shapes: any existing `//` or
422 `/* */` block immediately above the create line is stripped and replaced
423 so re-runs converge instead of accumulating. (Generated files don't have
424 hand-written comments above create calls — every such comment is
425 something this function previously inserted.)
426 """
427 xml_path = path.with_suffix(".xml")
428 if not xml_path.exists():
429 return source
430
431 # Drop wrapper tags (e.g. `<view>`) that have no C counterpart — without
432 # this filter the pairing-by-index gets shifted by one.
433 annotations = [
434 (tag, comment)
435 for tag, comment in _xml_elements_with_comments(xml_path)
436 if tag.startswith("lv_")
437 ]
438
439 out_lines: list[str] = []
440 idx = 0
441 for line in source.splitlines(keepends=True):
442 m = CREATE_LINE_RE.match(line)
443 if m and idx < len(annotations):
444 tag, comment = annotations[idx]
445 if tag == m.group(2) and comment:
446 indent = m.group(1)
447 _strip_preceding_comment(out_lines)
448 out_lines.extend(_format_inline_comment(comment, indent))
449 # Always advance the annotation pointer for any create line so
450 # one missing comment doesn't desync the whole mapping.
451 idx += 1
452 out_lines.append(line)
453
454 return "".join(out_lines)

472def _load_globals_subjects() -> dict[str, dict]:
473 """Return a mapping `name -> {type, value, min_value, max_value}` from
474 `globals.xml`.
475
476 Cached on first call. Subjects keep their declaration order (Python 3.7+
477 dict preserves insertion order), which we lean on when emitting init
478 blocks so the example files stay diff-stable run-to-run.
479 """
480 cache = getattr(_load_globals_subjects, "_cache", None)
481 if cache is not None:
482 return cache
483
484 subjects: dict[str, dict] = {}
485 if GLOBALS_XML_PATH.exists():
486 tree = ET.parse(GLOBALS_XML_PATH)
487 subjects_elem = tree.getroot().find("subjects")
488 if subjects_elem is not None:
489 for child in subjects_elem:
490 # ElementTree gives us comment nodes too when comments are
491 # captured; filter those out by checking the tag is a real
492 # string.
493 if not isinstance(child.tag, str):
494 continue
495 name = child.get("name")
496 if not name:
497 continue
498 subjects[name] = {
499 "type": child.tag, # "int", "string", "float"
500 "value": child.get("value", "0"),
501 "min_value": child.get("min_value"),
502 "max_value": child.get("max_value"),
503 }
504
505 _load_globals_subjects._cache = subjects # type: ignore[attr-defined]
506 return subjects

509def _subject_decl_lines(name: str, meta: dict) -> list[str]:
510 """Top-of-function static declarations for a subject (4-space indent).
511
512 `string` subjects need two backing buffers — value and previous-value —
513 plus the subject itself.
514 """
515 out = [f" static lv_subject_t {name};"]
516 if meta["type"] == "string":
517 out.append(f" static char {name}_buf[{SUBJECT_STRING_BUF_SIZE}];")
518 out.append(f" static char {name}_prev_buf[{SUBJECT_STRING_BUF_SIZE}];")
519 return out

522def _subject_init_lines(name: str, meta: dict) -> list[str]:
523 """Init-block calls for a subject (8-space indent, inside `if (!inited)`)."""
524 typ = meta["type"]
525 value = meta["value"]
526 out: list[str] = []
527 if typ == "int":
528 out.append(f" lv_subject_init_int(&{name}, {value});")
529 if meta.get("min_value") is not None:
530 out.append(
531 f" lv_subject_set_min_value_int(&{name}, {meta['min_value']});"
532 )
533 if meta.get("max_value") is not None:
534 out.append(
535 f" lv_subject_set_max_value_int(&{name}, {meta['max_value']});"
536 )
537 elif typ == "float":
538 out.append(f" lv_subject_init_float(&{name}, {value});")
539 elif typ == "string":
540 # Multi-line call to match the canonical formatting of generated
541 # project code; the args are too long to fit comfortably on one line.
542 out.extend([
543 f" lv_subject_init_string(&{name},",
544 f" {name}_buf,",
545 f" {name}_prev_buf,",
546 f" {SUBJECT_STRING_BUF_SIZE},",
547 f' "{value}");',
548 ])
549 return out

581def init_subjects(source: str, path: Path) -> str:
582 """Add `static lv_subject_t` declarations + init calls for any subject
583 the example references, so the file is self-contained.
584
585 Idempotent: a subject that already has a `static lv_subject_t <name>`
586 declaration isn't re-declared, and one that already has an
587 `lv_subject_init_*(&<name>` call isn't re-initialised.
588 """
589 meta = _load_globals_subjects()
590 if not meta:
591 return source
592
593 used = _used_subjects(source)
594 if not used:
595 return source
596
597 new_decls: list[str] = []
598 new_inits: list[str] = []
599 for name in used:
600 if not re.search(rf"\bstatic\s+lv_subject_t\s+{re.escape(name)}\b", source):
601 new_decls.extend(_subject_decl_lines(name, meta[name]))
602 if not re.search(
603 rf"lv_subject_init_(?:int|float|string)\s*\‍(\s*&{re.escape(name)}\b",
604 source,
605 ):
606 new_inits.extend(_subject_init_lines(name, meta[name]))
607
608 if not new_decls and not new_inits:
609 return source
610
611 block_match = INITED_BLOCK_RE.search(source)
612 if block_match:
613 # The example already has a `style_inited` block (typically because
614 # it has styles). Splice the subject decls in just before the block
615 # and the init lines in just before `style_inited = true;`.
616 block_start = block_match.start()
617 head = block_match.group("head")
618 body = block_match.group("body")
619 tail = block_match.group("tail")
620
621 if new_inits:
622 extra = "\n".join(new_inits) + "\n"
623 body = body + extra
624
625 new_block = head + body + tail
626
627 before_block = source[:block_start]
628 after_block = source[block_match.end():]
629
630 if new_decls:
631 # Insert decls just before the block, separated by a blank line
632 # from any earlier static declarations.
633 decl_text = "\n".join(new_decls) + "\n\n"
634 before_block = before_block + decl_text
635
636 return before_block + new_block + after_block
637
638 # No existing block — synthesise one right after the function's `{`.
639 fn_match = FUNCTION_OPEN_BRACE_RE.search(source)
640 if not fn_match:
641 return source
642
643 insert_pos = fn_match.end()
644 lines: list[str] = []
645 if new_decls:
646 lines.extend(new_decls)
647 lines.append("")
648 lines.append(" static bool style_inited = false;")
649 lines.append("")
650 lines.append(" if (!style_inited) {")
651 lines.extend(new_inits)
652 lines.append(" style_inited = true;")
653 lines.append(" }")
654 lines.append("")
655 block_text = "\n".join(lines) + "\n"
656 return source[:insert_pos] + block_text + source[insert_pos:]

659# =============================================================================
660# Image declarations
661# =============================================================================
662#
663# The generator emits `lv_image_set_src(img, my_image)` for registered image
664# names. To compile against a C-array image the call needs to pass `&my_image`
665# (an `lv_image_dsc_t *`) and the example must declare the symbol via
666# `LV_IMAGE_DECLARE(my_image)` so the linker can resolve the extern.
667#
668# Image names that look like LVGL constants (start with `LV_`) — e.g.
669# `LV_SYMBOL_OK` — are skipped because those are string macros, not C-array
670# image descriptors.
671
672# Captures the `lv_image_set_src(obj, [&]name)` call. Groups:
673# 1 = call prefix up to and including the `,`/whitespace before the arg
674# 2 = optional `&` already present (idempotency)
675# 3 = the image identifier
676LV_IMAGE_SET_SRC_RE = re.compile(
677 r"(lv_image_set_src\s*\‍(\s*[A-Za-z_]\w*\s*,\s*)(&?)([A-Za-z_]\w*)"
678)
679
680# A registered image can also reach a style by name through the image-src
681# style setters — `lv_style_set_bg_image_src(&style, name)`,
682# `lv_style_set_arc_image_src(...)`, `lv_obj_set_style_bg_image_src(obj,
683# name, sel)`, … — which need the same `&name` + `LV_IMAGE_DECLARE` fix. The
684# first argument is a `&style`/`obj`, so it is matched loosely as `[^,]+`.
685LV_STYLE_IMAGE_SRC_RE = re.compile(
686 r"(lv_(?:style_set|obj_set_style)_\w*image_src\s*\‍(\s*[^,]+,\s*)(&?)([A-Za-z_]\w*)"
687)
688
689
690def declare_and_ref_images(source: str, path: Path) -> str:
691 images: list[str] = [] # preserve first-seen order for stable output
692
693 def repl(m: re.Match) -> str:
694 prefix, amp, name = m.group(1), m.group(2), m.group(3)
695 if name.startswith("LV_"):
696 # `LV_SYMBOL_*` etc. — symbol macros, not C-array images.
697 return m.group(0)
698 if name not in images:
699 images.append(name)
700 return f"{prefix}&{name}" if not amp else m.group(0)
701
702 new_source = LV_IMAGE_SET_SRC_RE.sub(repl, source)
703 new_source = LV_STYLE_IMAGE_SRC_RE.sub(repl, new_source)
704 if not images:
705 return new_source
706
707 # Add `LV_IMAGE_DECLARE(name);` once per image at the top of the function
708 # body. Idempotent: skip names that already have a declare line.
709 fn_match = FUNCTION_OPEN_BRACE_RE.search(new_source)
710 if not fn_match:
711 return new_source
712
713 insert_pos = fn_match.end()
714 decls: list[str] = []
715 for img in images:
716 if not re.search(rf"LV_IMAGE_DECLARE\s*\‍(\s*{re.escape(img)}\s*\‍)", new_source):
717 decls.append(f" LV_IMAGE_DECLARE({img});")
718 if not decls:
719 return new_source
720
721 block = "\n".join(decls) + "\n\n"
722 return new_source[:insert_pos] + block + new_source[insert_pos:]

895# =============================================================================
896# Coalesce the declaration block
897# =============================================================================
898#
899# `declare_global_images`, `declare_and_ref_images` and `declare_and_ref_fonts`
900# each splice their own `\n\n`-terminated block in at the function's opening
901# brace, so a file using both ends up with the font block, a blank line, then
902# the image block — two stray groups in insertion order. Gather every
903# `LV_*_DECLARE` line that leads the function body and re-emit them as a single
904# contiguous block, images before fonts, followed by exactly one blank line —
905# the hand-written convention the example tree already uses.
906
907DECLARE_LINE_RE = re.compile(r"[ \t]*LV_(IMAGE|FONT)_DECLARE\‍([^)]*\‍);[ \t]*\Z")
908
909
910def coalesce_declares(source: str, path: Path) -> str:
911 fn_match = FUNCTION_OPEN_BRACE_RE.search(source)
912 if not fn_match:
913 return source
914
915 start = fn_match.end()
916 lines = source[start:].split("\n")
917
918 images: list[str] = []
919 fonts: list[str] = []
920 i = 0
921 while i < len(lines):
922 line = lines[i]
923 if line.strip() == "":
924 i += 1
925 continue
926 m = DECLARE_LINE_RE.fullmatch(line)
927 if not m:
928 break
929 (images if m.group(1) == "IMAGE" else fonts).append(line.strip())
930 i += 1
931
932 if not images and not fonts:
933 return source
934
935 block = "\n".join(f" {d}" for d in images + fonts) + "\n\n"
936 return source[:start] + block + "\n".join(lines[i:])

981# =============================================================================
982# Drop unused pointer variables
983# =============================================================================
984#
985# The generator assigns every created/added object to a local even when the
986# example never touches it again, and not always an `lv_obj_t *` — e.g.
987#
988# lv_obj_t * lv_calendar_header_arrow_0 = lv_calendar_add_header_arrow(cal);
989# lv_subject_increment_dsc_t * subject_increment_event_0 =
990# lv_obj_add_subject_increment_event(btn, &s, LV_EVENT_CLICKED, -1);
991#
992# The call has the side effect we want but the variable is dead. Hand-written
993# examples just write the bare call. Strip the `<type> * <name> = ` prefix
994# for any single-token-typed pointer local when `<name>` occurs exactly once
995# in the whole file (i.e. only at its own declaration); a variable referenced
996# anywhere else is kept as-is. The optional-`const` + single type token
997# matches the generator's actual output without reaching for arbitrary
998# multi-word C types. `static` decls can't match (the `*` must follow the
999# type token directly), so subject/style scaffolding is untouched.
1000# Idempotent: the rewritten bare call no longer matches the pattern.
1001
1002PTR_DECL_ASSIGN_RE = re.compile(
1003 r"^([ \t]*)(?:const\s+)?[A-Za-z_]\w*\s*\*\s*([A-Za-z_]\w*)\s*=\s*([^\n]+;)[ \t]*$",
1004 re.MULTILINE,
1005)
1006
1007
1008def drop_unused_ptr_vars(source: str) -> str:
1009 def repl(m: re.Match) -> str:
1010 indent, name, rhs = m.group(1), m.group(2), m.group(3)
1011 # A used variable appears at least twice (declaration + >=1 use).
1012 # `\b` stops `lv_calendar_0` matching inside
1013 # `lv_calendar_header_arrow_0`, and still counts `&name` uses.
1014 if len(re.findall(rf"\b{re.escape(name)}\b", source)) > 1:
1015 return m.group(0)
1016 return f"{indent}{rhs}"
1017
1018 return PTR_DECL_ASSIGN_RE.sub(repl, source)

1021# =============================================================================
1022# Cast OR-ed enum flag expressions
1023# =============================================================================
1024#
1025# The generator emits OR-ed enum constants both in array initializers and as
1026# call arguments, e.g.
1027#
1028# static const lv_buttonmatrix_ctrl_t m[] = {A, B | C, D};
1029# lv_chart_add_series(c, col, LV_CHART_AXIS_PRIMARY_Y | LV_CHART_AXIS_PRIMARY_X);
1030#
1031# In C `B | C` is an `int`, which is fine; in C++ an `int` won't implicitly
1032# convert to the expected `enum` type, so g++ fails with "invalid conversion
1033# from 'int' to 'lv_..._t'". Wrap each `|`-run of constants from a *single*
1034# known enum family in an explicit `(lv_..._t)(...)` cast so the same source
1035# compiles as both C and C++.
1036#
1037# Only true `enum` types are listed. Deliberately excluded:
1038# * LV_PART_* | LV_STATE_* — combine into `lv_style_selector_t`, which is
1039# `typedef uint32_t` (no enum, `int` converts cleanly). Mixed-family runs
1040# are skipped anyway.
1041# * LV_SCALE_LABEL_ROTATE_* — plain `#define` int macros, no enum type.
1042ENUM_FLAG_TYPES = {
1043 "LV_BUTTONMATRIX_CTRL_": "lv_buttonmatrix_ctrl_t",
1044 "LV_CHART_AXIS_": "lv_chart_axis_t",
1045 "LV_DIR_": "lv_dir_t",
1046 "LV_FS_MODE_": "lv_fs_mode_t",
1047 "LV_GRIDNAV_CTRL_": "lv_gridnav_ctrl_t",
1048}
1049
1050ORED_CONSTS_RE = re.compile(r"LV_[A-Z0-9_]+(?:\s*\|\s*LV_[A-Z0-9_]+)+")
1051
1052
1053def _enum_flag_type(token: str) -> str | None:
1054 for pfx, typ in ENUM_FLAG_TYPES.items():
1055 if token.startswith(pfx):
1056 return typ
1057 return None
1058
1059
1060def cast_ored_enum_flags(source: str) -> str:
1061 def repl(m: re.Match) -> str:
1062 run = m.group(0)
1063 types = {_enum_flag_type(t.strip()) for t in run.split("|")}
1064 # Cast only when every operand is from the same known enum family;
1065 # a mixed run (e.g. LV_PART_*|LV_STATE_*) or any unknown operand
1066 # leaves `types` with >1 entry or a `None`.
1067 if len(types) != 1 or None in types:
1068 return run
1069 typ = types.pop()
1070 # Idempotent: don't re-wrap a run already inside its cast.
1071 if source[: m.start()].endswith(f"({typ})("):
1072 return run
1073 return f"({typ})({run})"
1074
1075 return ORED_CONSTS_RE.sub(repl, source)

1078# =============================================================================
1079# Pipeline
1080# =============================================================================
1081#
1082# Every transformation has signature `(source: str, path: Path) -> str`.
1083# Functions that don't need `path` are wrapped in a `lambda` so the runner
1084# can call them uniformly. The order matters in a few places:
1085#
1086# * `remove_return_screen` and `rename_create_function` must both run
1087# before any reader assumes the function is `void` — together they make
1088# it so.
1089# * `replace_includes_with_lv_examples` must run before `fix_file_directive`
1090# would be meaningless either way, but conceptually the include block
1091# belongs to the "file identity" group.
1092# * `add_top_level_doc_comment` reads from the XML and writes above the
1093# function, so it must run after `rename_create_function` (so the
1094# function decl is already in its final shape).
1095# * `remove_blank_after_open_brace` runs last so it can clean up blanks
1096# left by everything else.
1097TRANSFORMATIONS = [
1098 # Subject decls/inits — must run before the empty-block removal so the
1099 # block isn't yet "empty" when we want to fill it.
1100 init_subjects,
1101 # Generator boilerplate.
1102 lambda s, p: remove_empty_style_inited(s),
1103 lambda s, p: remove_lv_trace_obj_create(s),
1104 lambda s, p: remove_lv_obj_set_name(s),
1105 # Code modernization.
1106 lambda s, p: root_obj_to_screen(s),
1107 lambda s, p: combine_size_and_pos(s),
1108 remove_return_screen,
1109 # Identity / naming.
1110 rename_create_function,
1111 # Image/font declarations + `&` prefix. These insert `LV_*_DECLARE`
1112 # lines at the function's opening brace, which `FUNCTION_OPEN_BRACE_RE`
1113 # only locates once the signature is the final `void <name>_create(void)`
1114 # — so they must run *after* `remove_return_screen` +
1115 # `rename_create_function`, not before.
1116 declare_global_images,
1117 declare_and_ref_images,
1118 declare_and_ref_fonts,
1119 # Merge the blocks the three transforms above each spliced in separately
1120 # into one contiguous, images-then-fonts declare group.
1121 coalesce_declares,
1122 lambda s, p: remove_section_banners(s),
1123 lambda s, p: remove_brief_line(s),
1124 replace_includes_with_lv_examples,
1125 fix_file_directive,
1126 # XML doc mapping.
1127 add_top_level_doc_comment,
1128 map_xml_comments,
1129 # Rename the guard variable now that the block has potentially gained
1130 # subject inits — `inited` reads better than the legacy `style_inited`.
1131 # Must follow `remove_empty_style_inited` which matches the literal name.
1132 lambda s, p: rename_style_inited(s),
1133 # Strip dead `<type> * x = ...;` declarations (x referenced only at
1134 # its own declaration). Must run after `map_xml_comments`: removing a
1135 # `... = lv_<type>_create(...)` LHS would otherwise desync the
1136 # comment-to-create pairing, which keys off `CREATE_LINE_RE`.
1137 lambda s, p: drop_unused_ptr_vars(s),
1138 # Cast OR-ed enum flag runs so examples compile under g++ too.
1139 lambda s, p: cast_ored_enum_flags(s),
1140 # Whitespace cleanup (always last so it can mop up).
1141 lambda s, p: remove_blank_after_open_brace(s),
1142 lambda s, p: strip_trailing_whitespace(s),
1143]
