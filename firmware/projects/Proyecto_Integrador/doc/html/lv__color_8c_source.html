147lv_color_t lv_color_hsv_to_rgb(uint16_t h, uint8_t s, uint8_t v)
148{
149 h = (uint32_t)((uint32_t)h * 255) / 360;
150 s = (uint16_t)((uint16_t)s * 255) / 100;
151 v = (uint16_t)((uint16_t)v * 255) / 100;
152
153 uint8_t r, g, b;
154
155 uint8_t region, remainder, p, q, t;
156
157 if(s == 0) {
158 return lv_color_make(v, v, v);
159 }
160
161 region = h / 43;
162 remainder = (h - (region * 43)) * 6;
163
164 p = (v * (255 - s)) >> 8;
165 q = (v * (255 - ((s * remainder) >> 8))) >> 8;
166 t = (v * (255 - ((s * (255 - remainder)) >> 8))) >> 8;
167
168 switch(region) {
169 case 0:
170 r = v;
171 g = t;
172 b = p;
173 break;
174 case 1:
175 r = q;
176 g = v;
177 b = p;
178 break;
179 case 2:
180 r = p;
181 g = v;
182 b = t;
183 break;
184 case 3:
185 r = p;
186 g = q;
187 b = v;
188 break;
189 case 4:
190 r = t;
191 g = p;
192 b = v;
193 break;
194 default:
195 r = v;
196 g = p;
197 b = q;
198 break;
199 }
200
201 lv_color_t result = lv_color_make(r, g, b);
202 return result;
203}

205lv_color_hsv_t lv_color_rgb_to_hsv(uint8_t r8, uint8_t g8, uint8_t b8)
206{
207 uint16_t r = ((uint32_t)r8 << 10) / 255;
208 uint16_t g = ((uint32_t)g8 << 10) / 255;
209 uint16_t b = ((uint32_t)b8 << 10) / 255;
210
211 uint16_t rgbMin = r < g ? (r < b ? r : b) : (g < b ? g : b);
212 uint16_t rgbMax = r > g ? (r > b ? r : b) : (g > b ? g : b);
213
214 lv_color_hsv_t hsv;
215
216 // https://en.wikipedia.org/wiki/HSL_and_HSV#Lightness
217 hsv.v = (100 * rgbMax) >> 10;
218
219 int32_t delta = rgbMax - rgbMin;
220 if(delta < 3) {
221 hsv.h = 0;
222 hsv.s = 0;
223 return hsv;
224 }
225
226 // https://en.wikipedia.org/wiki/HSL_and_HSV#Saturation
227 hsv.s = 100 * delta / rgbMax;
228 if(hsv.s < 3) {
229 hsv.h = 0;
230 return hsv;
231 }
232
233 // https://en.wikipedia.org/wiki/HSL_and_HSV#Hue_and_chroma
234 int32_t h;
235 if(rgbMax == r)
236 h = (((g - b) << 10) / delta) + (g < b ? (6 << 10) : 0); // between yellow & magenta
237 else if(rgbMax == g)
238 h = (((b - r) << 10) / delta) + (2 << 10); // between cyan & yellow
239 else if(rgbMax == b)
240 h = (((r - g) << 10) / delta) + (4 << 10); // between magenta & cyan
241 else
242 h = 0;
243 h *= 60;
244 h >>= 10;
245 if(h < 0) h += 360;
246
247 hsv.h = h;
248 return hsv;
249}

316uint16_t LV_ATTRIBUTE_FAST_MEM lv_color_16_16_mix(uint16_t c1, uint16_t c2, uint8_t mix)
317{
318 if(mix == 255) return c1;
319 if(mix == 0) return c2;
320 if(c1 == c2) return c1;
321
322 uint16_t ret;
323
324 /* Source: https://stackoverflow.com/a/50012418/1999969*/
325 mix = (uint32_t)((uint32_t)mix + 4) >> 3;
326
327 /*0x7E0F81F = 0b00000111111000001111100000011111*/
328 uint32_t bg = (uint32_t)(c2 | ((uint32_t)c2 << 16)) & 0x7E0F81F;
329 uint32_t fg = (uint32_t)(c1 | ((uint32_t)c1 << 16)) & 0x7E0F81F;
330 uint32_t result = ((((fg - bg) * mix) >> 5) + bg) & 0x7E0F81F;
331 ret = (uint16_t)(result >> 16) | result;
332
333 return ret;
334}
