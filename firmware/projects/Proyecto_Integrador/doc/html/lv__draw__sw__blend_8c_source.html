72void lv_draw_sw_blend(lv_draw_task_t * t, const lv_draw_sw_blend_dsc_t * dsc)
73{
74 /*Do not draw transparent things*/
75 if(dsc->opa <= LV_OPA_MIN) return;
76 if(dsc->mask_buf && dsc->mask_res == LV_DRAW_SW_MASK_RES_TRANSP) return;
77
78 lv_area_t blend_area;
79 if(!lv_area_intersect(&blend_area, dsc->blend_area, &t->clip_area)) return;
80
81 LV_PROFILER_DRAW_BEGIN;
82 lv_layer_t * layer = t->target_layer;
83 uint32_t layer_stride_byte = layer->draw_buf->header.stride;
84
85 lv_draw_sw_blend_handler_t handler = lv_draw_sw_get_blend_handler(layer->color_format);
86 if(handler) {
87 handler(t, dsc);
88 LV_PROFILER_DRAW_END;
89 return;
90 }
91
92 if(dsc->src_buf == NULL) {
93 lv_draw_sw_blend_fill_dsc_t fill_dsc;
94 fill_dsc.dest_w = lv_area_get_width(&blend_area);
95 fill_dsc.dest_h = lv_area_get_height(&blend_area);
96 fill_dsc.dest_stride = layer_stride_byte;
97 fill_dsc.opa = dsc->opa;
98 fill_dsc.color = dsc->color;
99 fill_dsc.mask_stride = 0;
100
101 if(dsc->mask_buf == NULL) fill_dsc.mask_buf = NULL;
102 else if(dsc->mask_res == LV_DRAW_SW_MASK_RES_FULL_COVER) fill_dsc.mask_buf = NULL;
103 else fill_dsc.mask_buf = dsc->mask_buf;
104
105
106 fill_dsc.relative_area = blend_area;
107 lv_area_move(&fill_dsc.relative_area, -layer->buf_area.x1, -layer->buf_area.y1);
108 fill_dsc.dest_buf = lv_draw_layer_go_to_xy(layer, blend_area.x1 - layer->buf_area.x1,
109 blend_area.y1 - layer->buf_area.y1);
110 if(fill_dsc.mask_buf) {
111 fill_dsc.mask_stride = dsc->mask_stride == 0 ? lv_area_get_width(dsc->mask_area) : dsc->mask_stride;
112 fill_dsc.mask_buf += fill_dsc.mask_stride * (blend_area.y1 - dsc->mask_area->y1) +
113 (blend_area.x1 - dsc->mask_area->x1);
114 }
115
116 lv_draw_sw_blend_color(layer->color_format, &fill_dsc);
117 }
118 else {
119 if(!lv_area_intersect(&blend_area, &blend_area, dsc->src_area)) {
120 LV_PROFILER_DRAW_END;
121 return;
122 }
123
124 if(dsc->mask_area && !lv_area_intersect(&blend_area, &blend_area, dsc->mask_area)) {
125 LV_PROFILER_DRAW_END;
126 return;
127 }
128
129 lv_draw_sw_blend_image_dsc_t image_dsc;
130 image_dsc.dest_w = lv_area_get_width(&blend_area);
131 image_dsc.dest_h = lv_area_get_height(&blend_area);
132 image_dsc.dest_stride = layer_stride_byte;
133
134 image_dsc.opa = dsc->opa;
135 image_dsc.blend_mode = dsc->blend_mode;
136 image_dsc.src_stride = dsc->src_stride;
137 image_dsc.src_color_format = dsc->src_color_format;
138
139 const uint8_t * src_buf = dsc->src_buf;
140 uint32_t src_px_size = lv_color_format_get_bpp(dsc->src_color_format);
141 src_buf += image_dsc.src_stride * (blend_area.y1 - dsc->src_area->y1);
142 src_buf += ((blend_area.x1 - dsc->src_area->x1) * src_px_size) >> 3;
143 image_dsc.src_buf = src_buf;
144 image_dsc.mask_stride = 0;
145
146 if(dsc->mask_buf == NULL) image_dsc.mask_buf = NULL;
147 else if(dsc->mask_res == LV_DRAW_SW_MASK_RES_FULL_COVER) image_dsc.mask_buf = NULL;
148 else image_dsc.mask_buf = dsc->mask_buf;
149
150 if(image_dsc.mask_buf) {
151 LV_ASSERT_NULL(dsc->mask_area);
152 image_dsc.mask_buf = dsc->mask_buf;
153 image_dsc.mask_stride = dsc->mask_stride ? dsc->mask_stride : lv_area_get_width(dsc->mask_area);
154 image_dsc.mask_buf += image_dsc.mask_stride * (blend_area.y1 - dsc->mask_area->y1) +
155 (blend_area.x1 - dsc->mask_area->x1);
156 }
157
158 image_dsc.relative_area = blend_area;
159 lv_area_move(&image_dsc.relative_area, -layer->buf_area.x1, -layer->buf_area.y1);
160
161 image_dsc.src_area = *dsc->src_area;
162 lv_area_move(&image_dsc.src_area, -layer->buf_area.x1, -layer->buf_area.y1);
163
164 image_dsc.dest_buf = lv_draw_layer_go_to_xy(layer, blend_area.x1 - layer->buf_area.x1,
165 blend_area.y1 - layer->buf_area.y1);
166
167 lv_draw_sw_blend_image(layer->color_format, &image_dsc);
168 }
169 LV_PROFILER_DRAW_END;
170}
