218def run_benchmark(config_name: str, pull: bool) -> None:
219 """Run the benchmark using docker"""

221 def volume(src, dst):
222 return ["-v", f"{src}:{dst}"]

224 # The Infrabase images nest the repo at /so3, so the user space lives at
225 # /so3/so3/usr (kernel at /so3/so3/so3). All mount targets use that path.
226 def so3_usr_lib(path):
227 return f"/so3/so3/usr/lib/{path}"
228
229 so3_usr_build = "/so3/so3/usr/build"
230 persistence_dir = "/persistence"
231 container_name = get_container_name(config_name)
232 build_dir = get_build_dir(config_name)
233
234 for v in get_docker_volumes(config_name):
235 subprocess.check_call(["docker", "volume", "create", v])
236
237 lvgl_src_path = os.path.join(lvgl_root_dir, "src")
238 lvgl_h_path = os.path.join(lvgl_root_dir, "lvgl.h")
239 lvgl_demos_path = os.path.join(lvgl_root_dir, "demos")
240 lv_conf_path = os.path.join(build_dir, "lv_conf.h")
241 if not os.path.exists(lv_conf_path):
242 generate_config(config_name)
243 assert os.path.exists(lv_conf_path)
244
245 docker_image_name = benchmark_configs[config_name]["image_name"]
246
247 volumes = [
248 # This is necessary to create a loop device
249 volume("/dev", "/dev"),
250 # Replace container's lvgl source and lv_conf
251 volume(lvgl_h_path, so3_usr_lib("lvgl/lvgl.h")),
252 volume(lvgl_src_path, so3_usr_lib("lvgl/src")),
253 volume(lvgl_demos_path, so3_usr_lib("lvgl/demos")),
254 volume(lv_conf_path, so3_usr_lib("lv_conf.h")),
255 # Cache build and disk folders so we don't regenerate everything in consecutive runs
256 volume(get_build_cache_volume(config_name), so3_usr_build),
257 volume(get_disk_cache_volume(config_name), persistence_dir),
258 ]

262 "docker",
263 "run",

266 interactive,
267 "--name",
268 container_name,
269 ]
270
271 if pull:
272 command.append("--pull=always")
273
