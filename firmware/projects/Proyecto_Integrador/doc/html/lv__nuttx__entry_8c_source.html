209void lv_nuttx_run(lv_nuttx_result_t * result)
210{
211#if LV_USE_NUTTX_LIBUV
212 LV_CHECK_ARG(result != NULL, return);
213 lv_nuttx_uv_loop(result);
214#else
215 LV_UNUSED(result);
216 while(1) {
217 uint32_t idle;
218 idle = lv_timer_handler();
219
220 /* Minimum sleep of 1ms */
221 idle = idle ? idle : 1;
222 /* Handle LV_DEF_REFR_PERIOD */
223 idle = idle != LV_NO_TIMER_READY ? idle : LV_DEF_REFR_PERIOD;
224 usleep(idle * 1000);
225 }
226#endif
227}

231uint32_t lv_nuttx_get_idle(void)
232{
233 struct cpuload_s cpuload;
234 int ret = clock_cpuload(0, &cpuload);
235 if(ret < 0) {
236 LV_LOG_WARN("clock_cpuload failed: %d", ret);
237 return 0;
238 }
239
240 uint32_t idle = cpuload.active * 100 / cpuload.total;
241 LV_LOG_TRACE("active = %" LV_PRIu32 ", total = %" LV_PRIu32,
242 cpuload.active, cpuload.total);
243
244 return idle;
245}

289{
290 struct timespec ts;
291
292 clock_gettime(CLOCK_MONOTONIC, &ts);
293 uint32_t tick = ts.tv_sec * 1000 + ts.tv_nsec / 1000000;

295 return tick;
296}

332static void check_stack_size(void)
333{
334 pthread_t tid = pthread_self();
335 ssize_t stack_size = pthread_get_stacksize_np(tid);
336 LV_LOG_USER("tid: %d, Stack size : %zd", (int)tid, stack_size);
337
338 if(stack_size < LV_NUTTX_MIN_STACK_SIZE) {
339 LV_LOG_ERROR("Stack size is too small. Please increase it to %d bytes or more.",
340 LV_NUTTX_MIN_STACK_SIZE);
341 }
342}
