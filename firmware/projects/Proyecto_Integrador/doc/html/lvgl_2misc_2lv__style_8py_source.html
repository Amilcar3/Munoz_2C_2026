37def decode_selector(selector: int) -> str:
38 """Decode selector into part + state string."""
39 part_val = (selector >> 16) & 0xFF
40 state_val = selector & 0xFFFF
41
42 part_str = PART_NAMES.get(part_val, f"PART({part_val:#x})")
43
44 if state_val == 0:
45 state_str = "DEFAULT"
46 elif state_val == 0xFFFF:
47 state_str = "ANY"
48 else:
49 flags = [name for bit, name in STATE_FLAGS.items() if state_val & bit]
50 state_str = "|".join(flags) if flags else f"STATE({state_val:#x})"
51
52 return f"{part_str}|{state_str}"

55def decode_coord(raw: int) -> str:
56 """Render an int32 coordinate the way the source wrote it.
57
58 LV_PCT() and LV_SIZE_CONTENT survive in the stored value as a type tag in
59 the top bits, so a percentage never has to be inferred from geometry.
60 """
61 bits = raw & 0xFFFFFFFF
62 # Only LV_COORD_TYPE_SPEC carries an encoding. LV_COORD_TYPE_PX_NEG is not a
63 # second one: there is no LV_COORD_SET_PX_NEG, and in two's complement every
64 # negative int32 already has those top bits, so a "PX_NEG" value is just a
65 # negative number and prints as itself.
66 if bits & _COORD_TYPE_MASK != _COORD_TYPE_SPEC:
67 return str(raw)
68 plain = bits & ~_COORD_TYPE_MASK
69 if plain == _COORD_MAX:
70 return "content"
71 # Not unreachable: _COORD_TYPE_MASK covers bits 29-30 only, so a negative
72 # raw keeps bit 31 in `plain`. LVGL's LV_COORD_IS_PCT() guards the same way.
73 if plain > _PCT_STORED_MAX:
74 return str(raw)
75 pct = plain if plain <= _PCT_POS_MAX else _PCT_POS_MAX - plain
76 return f"{pct}%"

79def decode_enum(prop_id: int, raw: int) -> str:
80 """Name an enum value, decomposing bitmasks such as BORDER_SIDE.
81
82 Only single-bit members are combined, and only when they account for every
83 set bit. Otherwise a sequential enum like FLEX_FLOW would render an
84 out-of-range value as a meaningless list of its low members.
85 """
86 names = ENUM_PROP_VALUES[prop_id]
87 if raw in names:
88 return names[raw]
89 parts, covered = [], 0
90 for bit in sorted(b for b in names if b and not b & (b - 1)):
91 if raw & bit:
92 parts.append(names[bit])
93 covered |= bit
94 return "|".join(parts) if parts and covered == raw else str(raw)

97def read_symbol(addr: int) -> "str | None":
98 """Resolve an address to its C symbol name, or None."""
99 if addr not in _symbol_cache:
100 try:
101 out = gdb.execute(f"info symbol {addr:#x}", to_string=True).strip()
102 except gdb.error:
103 out = ""
104 name = out.split(" in section ")[0] if " in section " in out else None
105 _symbol_cache[addr] = name
106 return _symbol_cache[addr]

121def _read_image_src(addr: int) -> "str | None":
122 """Read an image source pointer the way lv_image_src_get_type() does."""
123 try:
124 # bytes(): GDB returns a memoryview whose element type differs
125 # between versions - b'\xf3' on some, 243 on others.
126 first = bytes(gdb.selected_inferior().read_memory(addr, 1))[0]
127 if first == _IMAGE_HEADER_MAGIC or first == _IMAGE_HEADER_LEGACY:
128 return None # an lv_image_dsc_t, so the symbol name is the useful part
129 return gdb.Value(addr).cast(gdb.lookup_type("char").pointer()).string()
130 except (gdb.error, gdb.MemoryError, UnicodeDecodeError):
131 return None

148def _style_value_data(prop_id: int, value: Value) -> dict:
149 """Extract style value as pure data dict (no ANSI codes).
150
151 Returns dict with 'value_str', optional 'color_rgb' and, for pointers,
152 'ptr' so the raw address is never lost behind a resolved name.
153 """
154 try:
155 if prop_id in COLOR_PROPS:
156 color = value.color
157 r = int(color.red) & 0xFF
158 g = int(color.green) & 0xFF
159 b = int(color.blue) & 0xFF
160 return {
161 "value_str": f"#{r:02x}{g:02x}{b:02x}",
162 "color_rgb": {"r": r, "g": g, "b": b},
163 }
164 elif prop_id in POINTER_PROPS:
165 ptr = int(value.ptr)
166 data = {"value_str": _format_pointer(prop_id, ptr)}
167 if ptr:
168 data["ptr"] = f"{ptr:#x}"
169 return data
170 elif prop_id in ENUM_PROP_VALUES:
171 return {"value_str": decode_enum(prop_id, int(value.num))}
172 elif prop_id in BOOL_PROPS:
173 return {"value_str": "true" if int(value.num) else "false"}
174 elif prop_id in COORD_PROPS:
175 return {"value_str": decode_coord(int(value.num))}
176 else:
177 return {"value_str": str(int(value.num))}
178 except CorruptedError:
179 return {"value_str": str(value)}

211 def __init__(self, style: ValueInput):

214 def __iter__(self) -> Iterator[StyleEntry]:
215 prop_cnt = int(self.prop_cnt)
216 if prop_cnt == 0xFF:
217 # Constant style: lv_style_const_prop_t array terminated by prop==0
218 const_props = self.values_and_props.cast("lv_style_const_prop_t", ptr=True)
219 j = 0
220 while True:
221 prop_id = int(const_props[j].prop)
222 if prop_id == 0 or prop_id == 0xFF:
223 break
224 yield StyleEntry(prop_id, const_props[j].value)
225 j += 1
226 elif prop_cnt > 0:
227 # Normal style: values[prop_cnt] then props[prop_cnt]
228 # C code: (lv_style_prop_t*)vp + prop_cnt * sizeof(lv_style_value_t)
229 # The pointer arithmetic uses lv_style_prop_t element size as stride.
230 base = self.values_and_props
231 value_t = gdb.lookup_type("lv_style_value_t")
232 prop_t = gdb.lookup_type("lv_style_prop_t")
233 values_ptr = base.cast(value_t, ptr=True)
234 props_offset = prop_cnt * value_t.sizeof * prop_t.sizeof
235 props_ptr = Value(int(base) + props_offset).cast(prop_t, ptr=True)
236
237 for j in range(prop_cnt):
238 prop_id = int(props_ptr[j])
239 if prop_id == 0:
240 continue
241 yield StyleEntry(prop_id, values_ptr[j])
