248def get_release_branches(working_dir):
249
250 out = subprocess.check_output(("git", "-C", working_dir, "branch", "--format", "%(refname)", "--all"))
251 branches = out.decode().strip().splitlines()
252
253 release_versions = []
254 for branch_name in branches:
255 release_branch = re.fullmatch(r"refs/remotes/origin/release/v([0-9]+)\.([0-9]+)", branch_name)
256 if release_branch is None:
257 continue
258 release_versions.append((int(release_branch[1]), int(release_branch[2])))
259
260 release_versions.sort()
261
262 default_branch = None
263 if "refs/remotes/origin/master" in branches:
264 default_branch = "master"
265 elif "refs/remotes/origin/main" in branches:
266 default_branch = "main"
267
268 return release_versions, default_branch
