96#define TRANS_32_VALUE(big_endian, data) big_endian ? \
97 ((*(data) << 24) | (*((data) + 1) << 16) | (*((data) + 2) << 8) | *((data) + 3)) : \
98 (*(data) | (*((data) + 1) << 8) | (*((data) + 2) << 16) | (*((data) + 3) << 24))
99#define TRANS_16_VALUE(big_endian, data) big_endian ? \
100 ((*(data) << 8) | *((data) + 1)) : (*(data) | (*((data) + 1) << 8))

139static lv_result_t decoder_info(lv_image_decoder_t * decoder, lv_image_decoder_dsc_t * dsc, lv_image_header_t * header)
140{
141 LV_UNUSED(decoder); /*Unused*/
142 lv_image_src_t src_type = dsc->src_type; /*Get the source type*/
143
144 /*If it's a JPEG file...*/
145 if(src_type == LV_IMAGE_SRC_FILE) {
146 const char * src = dsc->src;
147 uint32_t jpg_signature = 0;
148 uint32_t rn;
149 lv_fs_read(&dsc->file, &jpg_signature, sizeof(jpg_signature), &rn);
150
151 if(rn != sizeof(jpg_signature)) {
152 LV_LOG_WARN("file: %s signature len = %" LV_PRIu32 " error", src, rn);
153 return LV_RESULT_INVALID;
154 }
155
156 const char * ext = lv_fs_get_ext(src);
157 bool is_jpeg_ext = (lv_strcmp(ext, "jpg") == 0)
158 || (lv_strcmp(ext, "jpeg") == 0);
159
160 if(!IS_JPEG_SIGNATURE(jpg_signature)) {
161 if(is_jpeg_ext) {
162 LV_LOG_WARN("file: %s signature = 0X%" LV_PRIX32 " error", src, jpg_signature);
163 }
164 return LV_RESULT_INVALID;
165 }
166
167 uint32_t width;
168 uint32_t height;
169
170 if(!get_jpeg_head_info(src, &width, &height)) {
171 return LV_RESULT_INVALID;
172 }
173
174 /*Save the data in the header*/
175 header->cf = LV_COLOR_FORMAT_RGB888;
176 header->w = width;
177 header->h = height;
178
179 return LV_RESULT_OK;
180 }
181
182 return LV_RESULT_INVALID; /*If didn't succeeded earlier then it's an error*/
183}

191static lv_result_t decoder_open(lv_image_decoder_t * decoder, lv_image_decoder_dsc_t * dsc)
192{
193 LV_UNUSED(decoder); /*Unused*/
194
195 /*If it's a JPEG file...*/
196 if(dsc->src_type == LV_IMAGE_SRC_FILE) {
197 const char * fn = dsc->src;
198 lv_draw_buf_t * decoded = decode_jpeg_file(fn);
199 if(decoded == NULL) {
200 LV_LOG_WARN("decode jpeg file failed");
201 return LV_RESULT_INVALID;
202 }
203
204 dsc->decoded = decoded;
205
206 if(dsc->args.no_cache) return LV_RESULT_OK;
207
208 /*If the image cache is disabled, just return the decoded image*/
209 if(!lv_image_cache_is_enabled()) return LV_RESULT_OK;
210
211 /*Add the decoded image to the cache*/
212 lv_image_cache_data_t search_key;
213 search_key.src_type = dsc->src_type;
214 search_key.src = dsc->src;
215 search_key.slot.size = decoded->data_size;
216
217 lv_cache_entry_t * entry = lv_image_decoder_add_to_cache(decoder, &search_key, decoded, NULL);
218
219 if(entry == NULL) {
220 lv_draw_buf_destroy(decoded);
221 return LV_RESULT_INVALID;
222 }
223 dsc->cache_entry = entry;
224 return LV_RESULT_OK; /*If not returned earlier then it failed*/
225 }
226
227 return LV_RESULT_INVALID; /*If not returned earlier then it failed*/
228}

254static lv_draw_buf_t * decode_jpeg_file(const char * filename)
255{
256 /* This struct contains the JPEG decompression parameters and pointers to
257 * working space (which is allocated as needed by the JPEG library).
258 */
259 struct jpeg_decompress_struct cinfo;
260 /* We use our private extension JPEG error handler.
261 * Note that this struct must live as long as the main JPEG parameter
262 * struct, to avoid dangling-pointer problems.
263 */
264 error_mgr_t jerr;
265
266 /* More stuff */
267 JSAMPARRAY buffer; /* Output row buffer */
268
269 uint32_t row_stride; /* physical row width in output buffer */
270
271 lv_draw_buf_t * decoded = NULL;
272
273 /* In this example we want to open the input file before doing anything else,
274 * so that the setjmp() error recovery below can assume the file is open.
275 * VERY IMPORTANT: use "b" option to fopen() if you are on a machine that
276 * requires it in order to read binary files.
277 */
278
279 uint32_t data_size;
280 uint8_t * data = lv_fs_load_with_alloc(filename, &data_size);
281 if(data == NULL) {
282 LV_LOG_WARN("can't load file %s", filename);
283 return NULL;
284 }
285
286 /* read jpeg exif orientation */
287 image_orientation_t image_orientation = get_jpeg_direction(data, data_size);
288
289 /* allocate and initialize JPEG decompression object */
290
291 /* We set up the normal JPEG error routines, then override error_exit. */
292 cinfo.err = jpeg_std_error(&jerr.pub);
293 jerr.pub.error_exit = error_exit;
294 /* Establish the setjmp return context for my_error_exit to use. */
295 if(setjmp(jerr.jb)) {
296 LV_LOG_WARN("decoding error");
297 if(decoded) {
298 lv_draw_buf_destroy(decoded);
299 }
300
301 /* If we get here, the JPEG code has signaled an error.
302 * We need to clean up the JPEG object, close the input file, and return.
303 */
304 jpeg_destroy_decompress(&cinfo);
305 lv_free(data);
306 return NULL;
307 }
308
309 /* Now we can initialize the JPEG decompression object. */
310 jpeg_create_decompress(&cinfo);
311
312 /* specify data source (eg, a file or buffer) */
313 jpeg_mem_src(&cinfo, data, data_size);
314
315 /* read file parameters with jpeg_read_header() */
316 jpeg_read_header(&cinfo, TRUE);
317
318 /* We can ignore the return value from jpeg_read_header since
319 * (a) suspension is not possible with the stdio data source, and
320 * (b) we passed TRUE to reject a tables-only JPEG file as an error.
321 * See libjpeg.doc for more info.
322 */
323
324 /* set parameters for decompression */
325 if(cinfo.jpeg_color_space == JCS_CMYK || cinfo.jpeg_color_space == JCS_YCCK) {
326 cinfo.out_color_space = JCS_CMYK;
327 }
328 else {
329 cinfo.out_color_space = JCS_EXT_BGR;
330 }
331
332 /* In this example, we don't need to change any of the defaults set by
333 * jpeg_read_header(), so we do nothing here.
334 */
335
336 /* Start decompressor */
337 jpeg_start_decompress(&cinfo);
338
339 /* We can ignore the return value since suspension is not possible
340 * with the stdio data source.
341 */
342
343 /* We may need to do some setup of our own at this point before reading
344 * the data. After jpeg_start_decompress() we have the correct scaled
345 * output image dimensions available, as well as the output colormap
346 * if we asked for color quantization.
347 * In this example, we need to make an output work buffer of the right size.
348 */
349 /* JSAMPLEs per row in output buffer */
350 row_stride = cinfo.output_width * cinfo.output_components;
351 /* Make a one-row-high sample array that will go away when done with image */
352 buffer = (*cinfo.mem->alloc_sarray)
353 ((j_common_ptr) &cinfo, JPOOL_IMAGE, row_stride, 1);
354
355 uint32_t width = cinfo.output_width;
356 uint32_t height = cinfo.output_height;
357 convert_size_with_orientation(image_orientation, &width, &height);
358
359 lv_image_header_t image_header = { 0 };
360 image_header.w = width;
361 image_header.h = height;
362 image_header.stride = row_stride;
363
364 lv_color_format_t fm = LV_COLOR_FORMAT_RGB888;
365 if(cinfo.out_color_space == JCS_CMYK) {
366 fm = LV_COLOR_FORMAT_XRGB8888;
367 }
368
369 /* Allocate the decoded draw buffer */
370 decoded = lv_draw_buf_create_ex(image_cache_draw_buf_handlers, width, height, fm, LV_STRIDE_AUTO);
371 if(decoded != NULL) {
372 uint32_t line_index = 0;
373 /* while (scan lines remain to be read) */
374 /* jpeg_read_scanlines(...); */
375
376 /* Here we use the library's state variable cinfo.output_scanline as the
377 * loop counter, so that we don't have to keep track ourselves.
378 */
379 while(cinfo.output_scanline < cinfo.output_height) {
380 /* jpeg_read_scanlines expects an array of pointers to scanlines.
381 * Here the array is only one element long, but you could ask for
382 * more than one scanline at a time if that's more convenient.
383 */
384 jpeg_read_scanlines(&cinfo, buffer, 1);
385
386 if(cinfo.out_color_space == JCS_CMYK) {
387 jpeg_cmyk_to_bgrx(buffer[0], decoded->header.w);
388 }
389
390 /* Assume put_scanline_someplace wants a pointer and sample count. */
391 process_buffer_orientation(image_orientation, decoded->data, &image_header, line_index, buffer[0],
392 decoded->header.stride);
393 line_index++;
394 }
395 }
396
397 /* Finish decompression */
398
399 jpeg_finish_decompress(&cinfo);
400
401 /* We can ignore the return value since suspension is not possible
402 * with the stdio data source.
403 */
404
405 /* Release JPEG decompression object */
406
407 /* This is an important step since it will release a good deal of memory. */
408 jpeg_destroy_decompress(&cinfo);
409
410 /* After finish_decompress, we can close the input file.
411 * Here we postpone it until after no more JPEG errors are possible,
412 * so as to simplify the setjmp error logic above. (Actually, I don't
413 * think that jpeg_destroy can do an error exit, but why assume anything...)
414 */
415 lv_free(data);
416
417 return decoded;
418}

445static bool get_jpeg_size(uint8_t * data, uint32_t data_size, uint32_t * width, uint32_t * height)
446{
447 struct jpeg_decompress_struct cinfo;
448 error_mgr_t jerr;
449
450 cinfo.err = jpeg_std_error(&jerr.pub);
451 jerr.pub.error_exit = error_exit;
452
453 if(setjmp(jerr.jb)) {
454 LV_LOG_WARN("read jpeg head failed");
455 jpeg_destroy_decompress(&cinfo);
456 return false;
457 }
458
459 /* jpeg_create_decompress */
460 jpeg_decompress_prepare(&cinfo, data, data_size);
461
462 int ret = jpeg_read_header(&cinfo, TRUE);
463 if(ret != JPEG_HEADER_OK) {
464 LV_LOG_WARN("read jpeg header failed: %d", ret);
465 jpeg_destroy_decompress(&cinfo);
466 return false;
467 }
468
469 /* read file exif orientation */
470 image_orientation_t op = jpeg_markers_reader(&cinfo);
471
472 *width = cinfo.image_width;
473 *height = cinfo.image_height;
474 convert_size_with_orientation(op, width, height);
475
476 jpeg_destroy_decompress(&cinfo);
477
478 return true;
479}

481static image_orientation_t get_jpeg_direction(uint8_t * data, uint32_t data_size)
482{
483 struct jpeg_decompress_struct cinfo;
484 error_mgr_t jerr;
485
486 cinfo.err = jpeg_std_error(&jerr.pub);
487 jerr.pub.error_exit = error_exit;
488
489 if(setjmp(jerr.jb)) {
490 LV_LOG_WARN("read jpeg orientation failed");
491 jpeg_destroy_decompress(&cinfo);
492 return IMAGE_CLOCKWISE_NONE;
493 }
494 /* jpeg_create_decompress */
495 jpeg_decompress_prepare(&cinfo, data, data_size);
496
497 int ret = jpeg_read_header(&cinfo, TRUE);
498 if(ret != JPEG_HEADER_OK) {
499 LV_LOG_WARN("read jpeg header failed: %d", ret);
500 jpeg_destroy_decompress(&cinfo);
501 return IMAGE_CLOCKWISE_NONE;
502 }
503
504 /* read file exif orientation */
505 image_orientation_t res = jpeg_markers_reader(&cinfo);
506
507 jpeg_destroy_decompress(&cinfo);
508
509 LV_LOG_INFO("read jpeg orientation data : %d", res);
510
511 return res;
512}

514static inline void process_buffer_orientation(image_orientation_t op, uint8_t * cur_pos, lv_image_header_t * header,
515 uint32_t line_index, uint8_t * buffer, uint32_t buf_stride)
516{
517 uint32_t dst_index = 0;
518 switch(op) {
519 case IMAGE_CLOCKWISE_NONE:
520 case IMAGE_CLOCKWISE_0:
521 lv_memcpy(cur_pos + line_index * buf_stride, buffer, header->stride);
522 break;
523 case IMAGE_CLOCKWISE_90:
524 for(uint32_t x = 0; x < header->h; x++) {
525 dst_index = x * buf_stride + (header->w - line_index - 1) * JPEG_PIXEL_SIZE;
526 lv_memcpy(cur_pos + dst_index, buffer + x * JPEG_PIXEL_SIZE, JPEG_PIXEL_SIZE);
527 }
528 break;
529 case IMAGE_CLOCKWISE_180:
530 for(uint32_t x = 0; x < header->w; x++) {
531 dst_index = (header->h - line_index - 1) * buf_stride + x * JPEG_PIXEL_SIZE;
532 lv_memcpy(cur_pos + dst_index, buffer + (header->w - x - 1) * JPEG_PIXEL_SIZE, JPEG_PIXEL_SIZE);
533 }
534 break;
535 case IMAGE_CLOCKWISE_270:
536 for(uint32_t x = 0; x < header->h; x++) {
537 dst_index = (header->h - x - 1) * buf_stride + line_index * JPEG_PIXEL_SIZE;
538 lv_memcpy(cur_pos + dst_index, buffer + x * JPEG_PIXEL_SIZE, JPEG_PIXEL_SIZE);
539 }
540 break;
541 case IMAGE_FLIP_VER:
542 dst_index = (header->h - line_index - 1) * buf_stride;
543 lv_memcpy(cur_pos + dst_index, buffer, header->stride);
544 break;
545 case IMAGE_FLIP_HOR:
546 for(uint32_t x = 0; x < header->w; x++) {
547 dst_index = line_index * buf_stride + (header->w - x - 1) * JPEG_PIXEL_SIZE;
548 lv_memcpy(cur_pos + dst_index, buffer + x * JPEG_PIXEL_SIZE, JPEG_PIXEL_SIZE);
549 }
550 break;
551 case IMAGE_TRANSPOSE:
552 for(uint32_t x = 0; x < header->h; x++) {
553 dst_index = x * buf_stride + line_index * JPEG_PIXEL_SIZE;
554 lv_memcpy(cur_pos + dst_index, buffer + x * JPEG_PIXEL_SIZE, JPEG_PIXEL_SIZE);
555 }
556 break;
557 case IMAGE_TRANSVERSE:
558 for(uint32_t x = 0; x < header->h; x++) {
559 dst_index = (header->h - x - 1) * buf_stride + (header->w - line_index - 1) * JPEG_PIXEL_SIZE;
560 lv_memcpy(cur_pos + dst_index, buffer + x * JPEG_PIXEL_SIZE, JPEG_PIXEL_SIZE);
561 }
562 break;
563 default:
564 lv_memcpy(cur_pos + line_index * buf_stride, buffer, header->stride);
565 break;
566 }
567}

569static image_orientation_t jpeg_markers_reader(struct jpeg_decompress_struct * cinfo)
570{
571 image_orientation_t res = IMAGE_CLOCKWISE_NONE;
572
573 if(cinfo == NULL || cinfo->marker_list == NULL) {
574 return res;
575 }
576
577 jpeg_saved_marker_ptr marker = cinfo->marker_list;
578 while(marker != NULL) {
579 if(marker->marker == APP1_MARKER) {
580 JOCTET FAR * app1_data = marker->data;
581 if(TRANS_32_VALUE(true, app1_data) == JPEG_EXIF) {
582 uint16_t endian_tag = TRANS_16_VALUE(true, app1_data + 4 + 2);
583 if(!(endian_tag == JPEG_LITTLE_ENDIAN_TAG || endian_tag == JPEG_BIG_ENDIAN_TAG)) {
584 return res;
585 }
586 bool is_big_endian = endian_tag == JPEG_BIG_ENDIAN_TAG;
587 /* first ifd offset addr : 4bytes(Exif) + 2bytes(0x00) + 2bytes(align) + 2bytes(tag mark) */
588 unsigned int offset = TRANS_32_VALUE(is_big_endian, app1_data + 8 + 2);
589 /* ifd base : 4bytes(Exif) + 2bytes(0x00) */
590 unsigned char * ifd = 0;
591 do {
592 /* ifd start: 4bytes(Exif) + 2bytes(0x00) + offset value(2bytes(align) + 2bytes(tag mark) + 4bytes(offset size)) */
593 unsigned int entry_offset = 4 + 2 + offset + 2;
594 if(entry_offset >= marker->data_length) {
595 return res;
596 }
597 ifd = app1_data + entry_offset;
598 unsigned short num_entries = TRANS_16_VALUE(is_big_endian, ifd - 2);
599 if(entry_offset + num_entries * 12 >= marker->data_length) {
600 return res;
601 }
602 for(int i = 0; i < num_entries; i++) {
603 unsigned short tag = TRANS_16_VALUE(is_big_endian, ifd);
604 if(tag == ORIENTATION_TAG) {
605 /* ifd entry: 12bytes = 2bytes(tag number) + 2bytes(kind of data) + 4bytes(number of components) + 4bytes(data)
606 * orientation kind(0x03) of data is unsigned short */
607 uint32_t dirc = TRANS_16_VALUE(is_big_endian, ifd + 2 + 2 + 4);
608 res = (dirc >= IMAGE_CLOCKWISE_0 && dirc <= IMAGE_CLOCKWISE_270) ? (image_orientation_t)dirc : IMAGE_CLOCKWISE_NONE;
609 }
610 ifd += 12;
611 }
612 offset = TRANS_32_VALUE(is_big_endian, ifd);
613 } while(offset != 0);
614 }
615 break;
616 }
617 marker = marker->next;
618 }
619
620 return res;
621}

623static void jpeg_cmyk_to_bgrx(uint8_t * cmyk_data, uint32_t pixel_count)
624{
625 uint8_t * ptr = cmyk_data;
626 for(uint32_t i = 0; i < pixel_count; i++) {
