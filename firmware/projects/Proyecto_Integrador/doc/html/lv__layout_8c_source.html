59uint32_t lv_layout_create(lv_layout_callbacks_t callbacks, void * user_data)
60{
61
62 layout_list_def = lv_realloc(layout_list_def, (layout_cnt + 1) * sizeof(lv_layout_dsc_t));
63 LV_ASSERT_MALLOC(layout_list_def);
64
65 layout_list_def[layout_cnt].callbacks = callbacks;

79bool lv_layout_get_min_size(lv_obj_t * obj, int32_t * size, bool width)
80{
81 LV_ASSERT(obj != NULL);
82 LV_ASSERT(size != NULL);
83 lv_layout_t layout_id = lv_obj_get_style_layout_internal(obj, LV_PART_MAIN);
84 if(layout_id > 0 && layout_id < layout_cnt) {
85 void * user_data = layout_list_def[layout_id].user_data;
86 if(layout_list_def[layout_id].callbacks.get_min_size_cb) {
87 return layout_list_def[layout_id].callbacks.get_min_size_cb(obj, size, width, user_data);
88 }
89 }
90 return false;
