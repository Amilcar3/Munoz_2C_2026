81static lv_result_t decoder_info(lv_image_decoder_t * decoder, lv_image_decoder_dsc_t * dsc, lv_image_header_t * header)
82{
83 LV_UNUSED(decoder);
84
85 const void * src = dsc->src;
86 lv_image_src_t src_type = dsc->src_type;
87
88 if(src_type == LV_IMAGE_SRC_VARIABLE) {
89 const lv_image_dsc_t * img_dsc = src;
90 uint8_t * raw_data = (uint8_t *)img_dsc->data;
91 const uint32_t raw_data_size = img_dsc->data_size;
92
93 if(is_jpg(raw_data, raw_data_size) == true) {
94#if LV_USE_FS_MEMFS
95 header->cf = LV_COLOR_FORMAT_RAW;
96 header->w = img_dsc->header.w;
97 header->h = img_dsc->header.h;
98 header->stride = img_dsc->header.w * 3;
99 return LV_RESULT_OK;
100#else
101 LV_LOG_WARN("LV_USE_FS_MEMFS needs to enabled to decode from data");
102 return LV_RESULT_INVALID;
103#endif
104 }
105 }
106 else if(src_type == LV_IMAGE_SRC_FILE) {
107 const char * fn = src;
108 const char * ext = lv_fs_get_ext(fn);
109 if((lv_strcmp(ext, "jpg") == 0) || (lv_strcmp(ext, "jpeg") == 0)) {
110 uint8_t * workb = lv_malloc(TJPGD_WORKBUFF_SIZE);
111 if(!workb) {
112 LV_LOG_WARN("failed to allocate memory for buffer");
113 return LV_RESULT_INVALID;
114 }
115 JDEC jd;
116 JRESULT rc = jd_prepare(&jd, input_func, workb, TJPGD_WORKBUFF_SIZE, &dsc->file);
117 lv_free(workb);
118 if(rc) {
119 LV_LOG_WARN("jd_prepare error: %d", rc);
120 return LV_RESULT_INVALID;
121 }
122 header->cf = LV_COLOR_FORMAT_RAW;
123 header->w = jd.width;
124 header->h = jd.height;
125 header->stride = jd.width * 3;
126
127 return LV_RESULT_OK;
128 }
129 }
130 return LV_RESULT_INVALID;
131}

133static size_t input_func(JDEC * jd, uint8_t * buff, size_t ndata)
134{
135 lv_fs_file_t * f = jd->device;
136 if(!f) return 0;
137
138 if(buff) {
139 uint32_t rn = 0;
140 lv_fs_read(f, buff, (uint32_t)ndata, &rn);
141 return rn;
142 }
143 else {
144 uint32_t pos;
145 lv_fs_tell(f, &pos);
146 lv_fs_seek(f, (uint32_t)(ndata + pos), LV_FS_SEEK_SET);
147 return ndata;
148 }
149 return 0;
150}

158static lv_result_t decoder_open(lv_image_decoder_t * decoder, lv_image_decoder_dsc_t * dsc)
159{
160 LV_UNUSED(decoder);
161 lv_fs_file_t * f = NULL;
162 if(dsc->src_type == LV_IMAGE_SRC_VARIABLE) {
163#if LV_USE_FS_MEMFS
164 const lv_image_dsc_t * img_dsc = dsc->src;
165 if(is_jpg(img_dsc->data, img_dsc->data_size) == true) {
166 f = lv_malloc(sizeof(lv_fs_file_t));
167 if(f == NULL) return LV_RESULT_INVALID;
168 lv_fs_path_ex_t path;
169 lv_fs_make_path_from_buffer(&path, LV_FS_MEMFS_LETTER, img_dsc->data, img_dsc->data_size, "bin");
170 lv_fs_res_t res;
171 res = lv_fs_open(f, (const char *)&path, LV_FS_MODE_RD);
172 if(res != LV_FS_RES_OK) {
173 lv_free(f);
174 return LV_RESULT_INVALID;
175 }
176 }
177#else
178 LV_LOG_WARN("LV_USE_FS_MEMFS needs to enabled to decode from data");
179#endif
180 }
181 else if(dsc->src_type == LV_IMAGE_SRC_FILE) {
182 const char * fn = dsc->src;
183 if((lv_strcmp(lv_fs_get_ext(fn), "jpg") == 0) || (lv_strcmp(lv_fs_get_ext(fn), "jpeg") == 0)) {
184 f = lv_malloc(sizeof(lv_fs_file_t));
185 if(f == NULL) return LV_RESULT_INVALID;
186 lv_fs_res_t res;
187 res = lv_fs_open(f, fn, LV_FS_MODE_RD);
188 if(res != LV_FS_RES_OK) {
189 lv_free(f);
190 return LV_RESULT_INVALID;
191 }
192 }
193 }
194 if(f == NULL) return LV_RESULT_INVALID;
195
196 uint8_t * workb_temp = lv_malloc(TJPGD_WORKBUFF_SIZE);
197 JDEC * jd = lv_malloc(sizeof(JDEC));
198 JRESULT rc = JDR_MEM1;
199
200 if(workb_temp != NULL && jd != NULL)
201 rc = jd_prepare(jd, input_func, workb_temp, (size_t)TJPGD_WORKBUFF_SIZE, f);
202
203 if(rc != JDR_OK) {
204 lv_fs_close(f);
205 lv_free(f);
206 lv_free(workb_temp);
207 lv_free(jd);
208 LV_LOG_WARN("jd_prepare error: %d", rc);
209 return LV_RESULT_INVALID;
210 }
211
212 dsc->user_data = jd;
213 dsc->header.cf = LV_COLOR_FORMAT_RGB888;
214 dsc->header.w = jd->width;
215 dsc->header.h = jd->height;
216 dsc->header.stride = jd->width * 3;
217
218 return LV_RESULT_OK;
219}

221static lv_result_t decoder_get_area(lv_image_decoder_t * decoder, lv_image_decoder_dsc_t * dsc,
222 const lv_area_t * full_area, lv_area_t * decoded_area)
223{
224 LV_UNUSED(decoder);
225 LV_UNUSED(full_area);
226
227 JDEC * jd = dsc->user_data;
228 lv_draw_buf_t * decoded = (void *)dsc->decoded;
229
230 uint32_t mx, my;
231 mx = jd->msx * 8;
232 my = jd->msy * 8; /* Size of the MCU (pixel) */
233 if(decoded_area->y1 == LV_COORD_MIN) {
234 decoded_area->y1 = 0;
235 decoded_area->y2 = my - 1;
236 decoded_area->x1 = -((int32_t)mx);
237 decoded_area->x2 = -1;
238 jd->scale = 0;
239 jd->dcv[2] = jd->dcv[1] = jd->dcv[0] = 0; /* Initialize DC values */
240 jd->rst = 0;
241 jd->rsc = 0;
242 if(decoded == NULL) {
243 decoded = lv_malloc_zeroed(sizeof(lv_draw_buf_t));
244 dsc->decoded = decoded;
245 }
246 else {
247 lv_fs_seek(jd->device, 0, LV_FS_SEEK_SET);
248 JRESULT rc = jd_prepare(jd, input_func, jd->pool_original, (size_t)TJPGD_WORKBUFF_SIZE, jd->device);
249 if(rc) {
250 LV_LOG_WARN("jd_prepare error: %d", rc);
251 return LV_RESULT_INVALID;
252 }
253 }
254 decoded->data = jd->workbuf;
255 decoded->header = dsc->header;
256 }
257
258 decoded_area->x1 += mx;
259 decoded_area->x2 += mx;
260
261 if(decoded_area->x1 >= jd->width) {
262 decoded_area->x1 = 0;
263 decoded_area->x2 = mx - 1;
264 decoded_area->y1 += my;
265 decoded_area->y2 += my;
266 }
267
268 if(decoded_area->x2 >= jd->width) decoded_area->x2 = jd->width - 1;
269 if(decoded_area->y2 >= jd->height) decoded_area->y2 = jd->height - 1;
270
271 decoded->header.w = lv_area_get_width(decoded_area);
272 decoded->header.h = lv_area_get_height(decoded_area);
273 decoded->header.stride = decoded->header.w * 3;
274 decoded->data_size = decoded->header.stride * decoded->header.h;
275
276 /* Process restart interval if enabled */
277 JRESULT rc;
278 if(jd->nrst && jd->rst++ == jd->nrst) {
279 rc = jd_restart(jd, jd->rsc++);
280 if(rc != JDR_OK) {
281 LV_LOG_WARN("jd_restart error: %d", rc);
282 return LV_RESULT_INVALID;
283 }
284 jd->rst = 1;
285 }
286
287 /* Load an MCU (decompress huffman coded stream, dequantize and apply IDCT) */
288 rc = jd_mcu_load(jd);
289 if(rc != JDR_OK) {
290 LV_LOG_WARN("jd_mcu_load error: %d", rc);
291 return LV_RESULT_INVALID;
292 }
293
294 /* Output the MCU (YCbCr to RGB, scaling and output) */
295 rc = jd_mcu_output(jd, NULL, decoded_area->x1, decoded_area->y1);
296 if(rc != JDR_OK) {
297 LV_LOG_WARN("jd_mcu_output error: %d", rc);
298 return LV_RESULT_INVALID;
299 }
300
301 return LV_RESULT_OK;
302}

320static int is_jpg(const uint8_t * raw_data, size_t len)
321{
322 const uint8_t jpg_signature[] = {0xFF, 0xD8, 0xFF};
323 if(len < sizeof(jpg_signature)) return false;
324 return memcmp(jpg_signature, raw_data, sizeof(jpg_signature)) == 0;
325}
