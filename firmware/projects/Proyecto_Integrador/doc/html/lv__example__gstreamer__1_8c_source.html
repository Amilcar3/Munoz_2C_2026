38void lv_example_gstreamer_1(void)
39{
40 static event_data_t event_data;
41
42 event_data.streamer = lv_gstreamer_create(lv_screen_active());
43
44 /* the gstreamer widget inherits the `lv_image` widget,
45 * meaning you can also provide it lv_image functions, like
46 lv_image_set_scale(event_data.streamer, 100);
47 lv_image_set_rotation(event_data.streamer, 100);
48 */
49
50 /* Set the current src of the streamer.
51 * Using the `URI` "factory", we can
52 * specify various URI schemes as media sources including local files (file://),
53 * web streams (http://, https://), RTSP streams (rtsp://), UDP streams (udp://),
54 * and many others. GStreamer's uridecodebin automatically selects the appropriate
55 * source element and decoder based on the URI scheme and media format.
56 * WebRTC streams, however, are provided via a dedicated WebRTC source (webrtcsrc)
57 * configured with a signalling server URI (e.g. ws://[ipsignallerserver]:[port])
58 * through its signaller-related properties, rather than via LV_GSTREAMER_FACTORY_URI_DECODE.
59 */
60 lv_gstreamer_set_src(event_data.streamer, LV_GSTREAMER_FACTORY_URI_DECODE, LV_GSTREAMER_PROPERTY_URI_DECODE,
61 "https://gstreamer.freedesktop.org/data/media/sintel_trailer-480p.webm");
62
63 lv_obj_center(event_data.streamer);
64
65 /* Create a slider to modify the stream volume and a label to visualize the current value */
66 volume_setter_create(&event_data);
67
68 /* Create a slider to see the position in the stream with 2 text label on each side
69 * One for the current position in the stream and the other for the total duration of the stream
70 * Also add a pause/play button*/
71 control_bar_create(&event_data);
72
73 /* The LV_EVENT_STATE_CHANGED will fire when the stream is ready at that point we can query the stream
74 * information like its resolution and duration. See `streamer_ready` */
75 lv_obj_add_event_cb(event_data.streamer, stream_state_changed, LV_EVENT_STATE_CHANGED, &event_data);
76
77 /* Play the stream immediately */
78 lv_gstreamer_play(event_data.streamer);
79
80 /* Create a timer that will update the slider position based on the stream position
81 * Make it 3 times faster than the refresh rate for a smoother effect */
82 lv_timer_create(update_position_slider, LV_DEF_REFR_PERIOD, &event_data);
83}

105 lv_obj_set_width(volume_label, 50);

110 lv_subject_set_int(volume_subject, 50);
111 lv_subject_add_observer_obj(volume_subject, volume_observer_cb, event_data->streamer, NULL);
112 lv_slider_bind_value(volume_slider, volume_subject);
113 lv_label_bind_text(volume_label, volume_subject, LV_SYMBOL_VOLUME_MID "\n%3" LV_PRId32 "%%");

177{

188static void play_pause_pressed(lv_event_t * e)
189{
190 event_data_t * event_data = (event_data_t *)lv_event_get_user_data(e);
191
192 if(lv_streq(lv_label_get_text(event_data->button_label), LV_SYMBOL_REFRESH)) {
193 lv_gstreamer_set_position(event_data->streamer, 0);
194 lv_gstreamer_play(event_data->streamer);
195 }
196 else if(lv_streq(lv_label_get_text(event_data->button_label), LV_SYMBOL_PLAY)) {
197 lv_gstreamer_play(event_data->streamer);
198 }
199 else {
200 lv_gstreamer_pause(event_data->streamer);
201 }

203static void stream_state_changed(lv_event_t * e)
204{
205 lv_event_code_t code = lv_event_get_code(e);
206 event_data_t * event_data = (event_data_t *)lv_event_get_user_data(e);
207
208 lv_obj_t * streamer = event_data->streamer;
209
210 if(code != LV_EVENT_STATE_CHANGED) {
211 return;
212 }
213
214 lv_gstreamer_stream_state_t stream_state = lv_gstreamer_get_stream_state(e);
215 switch(stream_state) {
216 case LV_GSTREAMER_STREAM_STATE_START: {
217 uint32_t duration = lv_gstreamer_get_duration(streamer);
218 LV_LOG_USER("Stream is starting");
219 LV_LOG_USER("\tStream resolution %" LV_PRId32 "x%" LV_PRId32, lv_image_get_src_width(streamer),
220 lv_image_get_src_height(streamer));
221 LV_LOG_USER("\tStream duration %" LV_PRIu32, duration);
222 update_duration_label(event_data->duration_label, duration);
223 break;
224 }
225 case LV_GSTREAMER_STREAM_STATE_END:
226 LV_LOG_USER("Stream is over");
227 lv_label_set_text_static(event_data->button_label, LV_SYMBOL_REFRESH);
228 break;
229 case LV_GSTREAMER_STREAM_STATE_PLAY:
230 lv_label_set_text_static(event_data->button_label, LV_SYMBOL_PAUSE);
231 break;
232 case LV_GSTREAMER_STREAM_STATE_PAUSE:
233 case LV_GSTREAMER_STREAM_STATE_STOP:
234 lv_label_set_text_static(event_data->button_label, LV_SYMBOL_PLAY);
235 break;
236 case LV_GSTREAMER_STREAM_STATE_INVALID:
237 break;
238 }
239}
