138lv_cache_entry_t * lv_cache_entry_alloc(uint32_t node_size, const lv_cache_t * cache)
139{
140 /* Align node_size so that accessing the entry data afterwards is done on an aligned byte address */
141 void * res = lv_malloc_zeroed(lv_cache_entry_get_size(node_size));
142 LV_ASSERT_MALLOC(res)
143 if(!res) {
144 LV_LOG_ERROR("malloc failed");
145 return NULL;
146 }
147 lv_cache_entry_t * entry = (lv_cache_entry_t *)((uint8_t *)res + LV_CACHE_ENTRY_ALIGN_NODE_SIZE(node_size));
148 lv_cache_entry_init(entry, cache, node_size);
149 return entry;
150}
