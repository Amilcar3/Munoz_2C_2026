65void lv_draw_nanovg_label_init(lv_draw_nanovg_unit_t * u)
66{
67 LV_ASSERT_NULL(u);
68 LV_ASSERT(u->letter_cache == NULL);
69 LV_ASSERT(u->letter_pending == NULL);
70
71 const lv_cache_ops_t ops = {
72 .compare_cb = (lv_cache_compare_cb_t)letter_compare_cb,
73 .create_cb = (lv_cache_create_cb_t)letter_create_cb,
74 .free_cb = (lv_cache_free_cb_t)letter_free_cb,
75 };
76
77 u->letter_cache = lv_cache_create(&lv_cache_class_lru_rb_count, sizeof(letter_item_t), LV_NANOVG_LETTER_CACHE_CNT, ops);

79 u->letter_pending = lv_pending_create(sizeof(lv_cache_entry_t *), 4);
80 lv_pending_set_free_cb(u->letter_pending, letter_cache_release_cb, u->letter_cache);

139static bool draw_letter_clip_areas(lv_draw_task_t * t, const lv_draw_glyph_dsc_t * dsc, lv_area_t * letter_area,
140 lv_area_t * clipped_area)
141{

143

147 .y = dsc->g->box_h + dsc->g->ofs_y

153 lv_area_get_height(dsc->letter_coords),
154 dsc->rotation,
155 LV_SCALE_NONE,

157 &pivot);

168static void draw_letter_bitmap(lv_draw_task_t * t, const lv_draw_glyph_dsc_t * dsc, int image_handle)
169{
170 LV_PROFILER_DRAW_BEGIN;
171
172 lv_area_t image_area;
173 lv_area_t clip_area;
174 if(!draw_letter_clip_areas(t, dsc, &image_area, &clip_area)) {
175 LV_PROFILER_DRAW_END;
176 return;
177 }
178
179 lv_draw_nanovg_unit_t * u = (lv_draw_nanovg_unit_t *)t->draw_unit;
180
181 if(!dsc->rotation) {
182 float x = dsc->letter_coords->x1;
183 float y = dsc->letter_coords->y1;
184 float w = lv_area_get_width(dsc->letter_coords);
185 float h = lv_area_get_height(dsc->letter_coords);
186
187 NVGpaint paint = nvgImagePattern(u->vg, x, y, w, h, 0, image_handle, 1.0f);
188 paint.innerColor = paint.outerColor = nvgRGBA(dsc->color.red, dsc->color.green, dsc->color.blue, dsc->opa);
189
190 nvgBeginPath(u->vg);
191 nvgRect(u->vg, x, y, w, h);
192 nvgFillPaint(u->vg, paint);
193 nvgFill(u->vg);
194 }
195 else {
196 /* TODO: draw rotated bitmap */
197 }
198
199 LV_PROFILER_DRAW_END;
200}

202static inline int letter_get_image_handle(lv_draw_nanovg_unit_t * u, lv_font_glyph_dsc_t * g_dsc)
203{
204 LV_PROFILER_DRAW_BEGIN;
205
206 letter_item_t search_key = { 0 };
207 search_key.u = u;
208 search_key.g_dsc = *g_dsc;
209
210 lv_cache_entry_t * cache_node_entry = lv_cache_acquire(u->letter_cache, &search_key, NULL);
211 if(cache_node_entry == NULL) {
212 /* check if the cache is full */
213 size_t free_size = lv_cache_get_free_size(u->letter_cache, NULL);
214 if(free_size == 0) {
215 LV_LOG_INFO("letter cache is full, release all pending cache entries");
216 lv_nanovg_end_frame(u);
217 }
218
219 cache_node_entry = lv_cache_acquire_or_create(u->letter_cache, &search_key, NULL);
220 if(cache_node_entry == NULL) {
221 LV_LOG_ERROR("letter cache creating failed");
222 LV_PROFILER_DRAW_END;
223 return -1;
224 }
225 }
226
227 /* Add the new entry to the pending list */
228 lv_pending_add(u->letter_pending, &cache_node_entry);
229
230 letter_item_t * letter_item = lv_cache_entry_get_data(cache_node_entry);
231
232 LV_PROFILER_DRAW_END;
233 return letter_item->image_handle;
234}

243static bool letter_create_cb(letter_item_t * item, void * user_data)
244{
245 LV_PROFILER_DRAW_BEGIN;
246 LV_UNUSED(user_data);
247 lv_font_glyph_dsc_t * g_dsc = &item->g_dsc;
248
249 const uint32_t w = g_dsc->box_w;
250 const uint32_t h = g_dsc->box_h;
251
252 lv_draw_buf_t * image_buf = lv_nanovg_reshape_global_image(item->u, LV_COLOR_FORMAT_A8, w, h);
253 if(!image_buf) {
254 LV_PROFILER_DRAW_END;
255 return false;
256 }
257
258 const lv_draw_buf_t * bitmap_draw_buf = (const lv_draw_buf_t *)lv_font_get_glyph_bitmap_internal(g_dsc, image_buf);
259 if(!bitmap_draw_buf) {
260 LV_PROFILER_DRAW_END;
261 return false;
262 }
263
264 LV_PROFILER_DRAW_BEGIN_TAG("nvgCreateImage");
265 item->image_handle = nvgCreateImage(item->u->vg, w, h, 0, NVG_TEXTURE_ALPHA, lv_draw_buf_goto_xy(bitmap_draw_buf, 0,
266 0));
267 LV_PROFILER_DRAW_END_TAG("nvgCreateImage");
268
269 lv_font_glyph_release_draw_data_internal(g_dsc);
270
271 LV_LOG_TRACE("image_handle: %d", item->image_handle);
272 LV_PROFILER_DRAW_END;
273 return true;
274}

292static void draw_letter_cb(lv_draw_task_t * t, lv_draw_glyph_dsc_t * glyph_draw_dsc,
293 lv_draw_fill_dsc_t * fill_draw_dsc, const lv_area_t * fill_area)
294{
295 lv_draw_nanovg_unit_t * u = (lv_draw_nanovg_unit_t *)t->draw_unit;
296
297 if(glyph_draw_dsc) {
298 switch(glyph_draw_dsc->format) {
299 case LV_FONT_GLYPH_FORMAT_A1:
300 case LV_FONT_GLYPH_FORMAT_A2:
301 case LV_FONT_GLYPH_FORMAT_A3:
302 case LV_FONT_GLYPH_FORMAT_A4:
303 case LV_FONT_GLYPH_FORMAT_A8: {
304 int image_handle = letter_get_image_handle(u, glyph_draw_dsc->g);
305 if(image_handle < 0) {
306 return;
307 }
308
309 draw_letter_bitmap(t, glyph_draw_dsc, image_handle);
310 }
311 break;
312
313#if LV_USE_FREETYPE
314 case LV_FONT_GLYPH_FORMAT_VECTOR: {
315 if(lv_freetype_is_outline_font(glyph_draw_dsc->g->resolved_font)) {
316 if(!glyph_draw_dsc->glyph_data) {
317 return;
318 }
319
320 /* TODO: draw_letter_outline(t, glyph_draw_dsc); */
321 }
322 }
323 break;
324#endif /* LV_USE_FREETYPE */
325
326 case LV_FONT_GLYPH_FORMAT_IMAGE: {
327 glyph_draw_dsc->glyph_data = lv_font_get_glyph_bitmap_internal(glyph_draw_dsc->g, glyph_draw_dsc->_draw_buf);
328 if(!glyph_draw_dsc->glyph_data) {
329 return;
330 }
331
332 lv_draw_image_dsc_t image_dsc;
333 lv_draw_image_dsc_init(&image_dsc);
334 image_dsc.opa = glyph_draw_dsc->opa;
335 image_dsc.src = glyph_draw_dsc->glyph_data;
336 image_dsc.rotation = glyph_draw_dsc->rotation;
337 lv_draw_nanovg_image(t, &image_dsc, glyph_draw_dsc->letter_coords, -1);
338 }
339 break;
340
341#if LV_USE_FONT_PLACEHOLDER
342 case LV_FONT_GLYPH_FORMAT_NONE: {
343 if(glyph_draw_dsc->bg_coords == NULL) break;
344 /* Draw a placeholder rectangle*/
345 lv_draw_border_dsc_t border_draw_dsc;
346 lv_draw_border_dsc_init(&border_draw_dsc);
347 border_draw_dsc.opa = glyph_draw_dsc->opa;
348 border_draw_dsc.color = glyph_draw_dsc->color;
349 border_draw_dsc.width = 1;
350 lv_draw_nanovg_border(t, &border_draw_dsc, glyph_draw_dsc->bg_coords);
351 }
352 break;
353#endif /* LV_USE_FONT_PLACEHOLDER */
354
355 default:
356 break;
357 }
358 }
359
360 if(fill_draw_dsc && fill_area) {
361 lv_draw_nanovg_fill(t, fill_draw_dsc, fill_area);
362 }
363}
