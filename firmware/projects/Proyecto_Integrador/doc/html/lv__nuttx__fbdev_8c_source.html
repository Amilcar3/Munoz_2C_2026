83lv_display_t * lv_nuttx_fbdev_create(void)
84{
85 lv_nuttx_fb_t * dsc = lv_malloc_zeroed(sizeof(lv_nuttx_fb_t));
86 LV_ASSERT_MALLOC(dsc);
87 if(dsc == NULL) return NULL;
88
89 lv_display_t * disp = lv_display_create(800, 480);
90 if(disp == NULL) {
91 lv_free(dsc);
92 return NULL;
93 }
94 dsc->fd = -1;
95 lv_display_set_driver_data(disp, dsc);
96 lv_display_add_event_cb(disp, display_release_cb, LV_EVENT_DELETE, disp);
97 lv_display_set_flush_cb(disp, flush_cb);
98 return disp;
99}

101int lv_nuttx_fbdev_set_file(lv_display_t * disp, const char * file)
102{
103 LV_CHECK_ARG(disp != NULL, return -EINVAL);
104 LV_CHECK_ARG(file != NULL, return -EINVAL);
105 int ret;
106 LV_ASSERT(disp && file);
107 lv_nuttx_fb_t * dsc = lv_display_get_driver_data(disp);
108
109 if(dsc->fd >= 0) close(dsc->fd);
110
111 /* Open the file for reading and writing*/
112
113 dsc->fd = open(file, O_RDWR);
114 if(dsc->fd < 0) {
115 LV_LOG_ERROR("Error: cannot open framebuffer device");
116 return -errno;
117 }
118 LV_LOG_USER("The framebuffer device was opened successfully");
119
120 if(ioctl(dsc->fd, FBIOGET_VIDEOINFO, (unsigned long)((uintptr_t)&dsc->vinfo)) < 0) {
121 LV_LOG_ERROR("ioctl(FBIOGET_VIDEOINFO) failed: %d", errno);
122 ret = -errno;
123 goto errout;
124 }
125
126 LV_LOG_USER("VideoInfo:");
127 LV_LOG_USER(" fmt: %u", dsc->vinfo.fmt);
128 LV_LOG_USER(" xres: %u", dsc->vinfo.xres);
129 LV_LOG_USER(" yres: %u", dsc->vinfo.yres);
130 LV_LOG_USER(" nplanes: %u", dsc->vinfo.nplanes);
131
132 if((ret = fbdev_get_pinfo(dsc->fd, &dsc->pinfo)) < 0) {
133 goto errout;
134 }
135
136 lv_color_format_t color_format = fb_fmt_to_color_format(dsc->vinfo.fmt);
137 if(color_format == LV_COLOR_FORMAT_UNKNOWN) {
138 goto errout;
139 }
140
141 dsc->mem = mmap(NULL, dsc->pinfo.fblen, PROT_READ | PROT_WRITE,
142 MAP_SHARED | MAP_FILE, dsc->fd, 0);
143 if(dsc->mem == MAP_FAILED) {
144 LV_LOG_ERROR("ioctl(FBIOGET_PLANEINFO) failed: %d", errno);
145 ret = -errno;
146 goto errout;
147 }
148
149 uint32_t w = dsc->vinfo.xres;
150 uint32_t h = dsc->vinfo.yres;
151 uint32_t stride = dsc->pinfo.stride;
152 uint32_t data_size = h * stride;
153 lv_draw_buf_init(&dsc->buf1, w, h, color_format, stride, dsc->mem, data_size);
154
155 /* Check buffer mode */
156 bool double_buffer = dsc->pinfo.yres_virtual >= (dsc->vinfo.yres * 2);
157 if(double_buffer) {
158 if((ret = fbdev_init_mem2(dsc)) < 0) {
159 goto errout;
160 }
161
162 lv_draw_buf_init(&dsc->buf2, w, h, color_format, stride, dsc->mem2, data_size);
163 lv_display_set_draw_buffers(disp, &dsc->buf1, &dsc->buf2);
164
165 bool triple_buffer = dsc->pinfo.yres_virtual >= (dsc->vinfo.yres * 3);
166 if(triple_buffer) {
167 if((ret = fbdev_init_mem3(dsc)) < 0) {
168 goto errout;
169 }
170
171 lv_draw_buf_init(&dsc->buf3, w, h, color_format, stride, dsc->mem3, data_size);
172 lv_display_set_3rd_draw_buffer(disp, &dsc->buf3);
173 }
174 }
175 else {
176 dsc->mem_off_screen = malloc(data_size);
177 LV_ASSERT_MALLOC(dsc->mem_off_screen);
178 if(!dsc->mem_off_screen) {
179 ret = -ENOMEM;
180 LV_LOG_ERROR("Failed to allocate memory for off-screen buffer");
181 goto errout;
182 }
183
184 LV_LOG_USER("Use off-screen mode, memory: %p, size: %" LV_PRIu32, dsc->mem_off_screen, data_size);
185 lv_draw_buf_init(&dsc->buf2, w, h, color_format, stride, dsc->mem_off_screen, data_size);
186 lv_display_set_draw_buffers(disp, &dsc->buf2, NULL);
187 }
188
189 lv_display_set_color_format(disp, color_format);
190 lv_display_set_render_mode(disp, LV_DISPLAY_RENDER_MODE_DIRECT);
191 lv_display_set_resolution(disp, dsc->vinfo.xres, dsc->vinfo.yres);
192 lv_timer_set_cb(disp->refr_timer, display_refr_timer_cb);
193
194#if LV_DRAW_TRANSFORM_USE_MATRIX
195 lv_display_set_matrix_rotation(disp, true);
196#endif
197
198 LV_LOG_USER("Resolution is set to %dx%d at %" LV_PRId32 "dpi",
199 dsc->vinfo.xres, dsc->vinfo.yres, lv_display_get_dpi(disp));
200 return 0;
201
202errout:
203 close(dsc->fd);
204 dsc->fd = -1;
205 return ret;
206}

213static void fbdev_join_inv_areas(lv_display_t * disp, lv_area_t * final_inv_area)
214{
215 uint16_t inv_index;
216
217 bool area_joined = false;
218
219 for(inv_index = 0; inv_index < disp->inv_p; inv_index++) {
220 if(disp->inv_area_joined[inv_index] == 0) {
221 const lv_area_t * area_p = &disp->inv_areas[inv_index];
222
223 /* Join to final_area */
224
225 if(!area_joined) {
226 /* copy first area */
227 *final_inv_area = *area_p;
228 area_joined = true;
229 }
230 else {
231 lv_area_join(final_inv_area,
232 final_inv_area,
233 area_p);
234 }
235 }
236 }
237}

240static void display_refr_timer_cb(lv_timer_t * tmr)
241{
242 lv_display_t * disp = lv_timer_get_user_data(tmr);
243 lv_nuttx_fb_t * dsc = lv_display_get_driver_data(disp);
244 struct pollfd pfds[1];
245
246 lv_memzero(pfds, sizeof(pfds));
247 pfds[0].fd = dsc->fd;
248 pfds[0].events = POLLOUT;
249
250 /* Query free fb to draw */
251
252 if(poll(pfds, 1, 0) < 0) {
253 return;
254 }
255
256 if(pfds[0].revents & POLLOUT) {
257 lv_display_refr_timer(tmr);
258 }
259}

261static void flush_cb(lv_display_t * disp, const lv_area_t * area, uint8_t * color_p)
262{
263 LV_UNUSED(color_p);
264 lv_nuttx_fb_t * dsc = lv_display_get_driver_data(disp);
265
266 if(dsc->mem_off_screen) {
267 /* When rendering in off-screen mode, copy the drawing buffer to fb */
268 /* buf2(off-screen buffer) -> buf1(fbmem)*/
269 lv_draw_buf_copy(&dsc->buf1, area, &dsc->buf2, area);
270 }
271
272 /* Skip the non-last flush */
273
274 if(!lv_display_flush_is_last(disp)) {
275 lv_display_flush_ready(disp);
276 return;
277 }
278
279#if defined(CONFIG_FB_UPDATE)
280 /*May be some direct update command is required*/
281 int yoffset = 0;
282 if(disp->buf_act == disp->buf_2) {
283 yoffset = dsc->mem2_yoffset;
284 }
285 else if(disp->buf_act == disp->buf_3) {
286 yoffset = dsc->mem3_yoffset;
287 }
288 /* Join the areas to update */
289 lv_area_t final_inv_area;
290 lv_memzero(&final_inv_area, sizeof(final_inv_area));
291 fbdev_join_inv_areas(disp, &final_inv_area);
292
293 struct fb_area_s fb_area;
294 fb_area.x = final_inv_area.x1;
295 fb_area.y = final_inv_area.y1 + yoffset;
296 fb_area.w = lv_area_get_width(&final_inv_area);
297 fb_area.h = lv_area_get_height(&final_inv_area);
298 if(ioctl(dsc->fd, FBIO_UPDATE, (unsigned long)((uintptr_t)&fb_area)) < 0) {
299 LV_LOG_ERROR("ioctl(FBIO_UPDATE) failed: %d", errno);
300 }
301#endif
302
303 /* double framebuffer */
304
305 if(dsc->mem2 != NULL) {
306 if(disp->buf_act == disp->buf_1) {
307 dsc->pinfo.yoffset = 0;
308 }
309 else if(disp->buf_act == disp->buf_2) {
310 dsc->pinfo.yoffset = dsc->mem2_yoffset;
311 }
312 else if(disp->buf_act == disp->buf_3) {
313 dsc->pinfo.yoffset = dsc->mem3_yoffset;
314 }
315
316 if(ioctl(dsc->fd, FBIOPAN_DISPLAY, (unsigned long)((uintptr_t) & (dsc->pinfo))) < 0) {
317 LV_LOG_ERROR("ioctl(FBIOPAN_DISPLAY) failed: %d", errno);
318 }
319 }
320 lv_display_flush_ready(disp);
321}

323static lv_color_format_t fb_fmt_to_color_format(int fmt)
324{
325 switch(fmt) {
326 case FB_FMT_RGB16_565:
327 return LV_COLOR_FORMAT_RGB565;
328 case FB_FMT_RGB24:
329 return LV_COLOR_FORMAT_RGB888;
330 case FB_FMT_RGB32:
331 return LV_COLOR_FORMAT_XRGB8888;
332 case FB_FMT_RGBA32:
333 return LV_COLOR_FORMAT_ARGB8888;
334 default:
335 break;
336 }
337
338 LV_LOG_ERROR("Unsupported color format: %d", fmt);
339
340 return LV_COLOR_FORMAT_UNKNOWN;
341}

360static int fbdev_init_mem2(lv_nuttx_fb_t * dsc)
361{
362 struct fb_planeinfo_s pinfo;
363 void * phy_mem1;
364 void * phy_mem2;
365
366 int ret;
367
368 /* Get display[0] planeinfo */
369 lv_memzero(&pinfo, sizeof(pinfo));
370 pinfo.display = dsc->pinfo.display;
371 if((ret = fbdev_get_pinfo(dsc->fd, &pinfo)) < 0) return ret;
372 phy_mem1 = pinfo.fbmem;
373
374 /* Get display[1] planeinfo */
375 lv_memzero(&pinfo, sizeof(pinfo));
376 pinfo.display = dsc->pinfo.display + 1;
377 if((ret = fbdev_get_pinfo(dsc->fd, &pinfo)) < 0) return ret;
378 phy_mem2 = pinfo.fbmem;
379
380 lv_uintptr_t offset = (lv_uintptr_t)phy_mem2 - (lv_uintptr_t)phy_mem1;
381 bool is_consecutive = offset == 0;
382
383 /* Check bpp */
384
385 if(pinfo.bpp != dsc->pinfo.bpp) {
386 LV_LOG_WARN("mem2 is incorrect");
387 return -EINVAL;
388 }
389
390 /* Check the buffer address offset,
391 * It needs to be divisible by pinfo.stride
392 */
393
394 if((offset % dsc->pinfo.stride) != 0) {
395 LV_LOG_WARN("It is detected that buf_offset(%" PRIuPTR ") "
396 "and stride(%d) are not divisible, please ensure "
397 "that the driver handles the address offset by itself.",
398 offset, dsc->pinfo.stride);
399 }
400
401 /* Calculate the address and yoffset of mem2 */
402
403 if(is_consecutive) {
404 dsc->mem2_yoffset = dsc->vinfo.yres;
405 offset = dsc->vinfo.yres * pinfo.stride;
406 }
407 else {
408 dsc->mem2_yoffset = offset / dsc->pinfo.stride;
409 }
410
411 uint32_t fblen = is_consecutive ? pinfo.fblen / 2 : pinfo.fblen;
412 void * mem2 = mmap(NULL, fblen, PROT_READ | PROT_WRITE, MAP_SHARED | MAP_FILE, dsc->fd, offset);
413 if(mem2 == MAP_FAILED) {
414 LV_LOG_ERROR("ERROR: mmap failed: %d, offset: %" PRIuPTR, errno, offset);
415 return -errno;
416 }
417 dsc->mem2 = mem2;
418
419 LV_LOG_USER("Use of %sconsecutive mem2 = %p, yoffset = %" LV_PRIu32, is_consecutive ? "" : "non-", dsc->mem2,
420 dsc->mem2_yoffset);
421
422 return 0;
423}

425static int fbdev_init_mem3(lv_nuttx_fb_t * dsc)
426{
427 uintptr_t buf_offset;
428 struct fb_planeinfo_s pinfo;
429 int ret;
430
431 lv_memzero(&pinfo, sizeof(pinfo));
432
433 /* Get display[2] planeinfo */
434
435 pinfo.display = dsc->pinfo.display + 2;
436
437 if((ret = fbdev_get_pinfo(dsc->fd, &pinfo)) < 0) {
438 return ret;
439 }
440
441 /* Check bpp */
442
443 if(pinfo.bpp != dsc->pinfo.bpp) {
444 LV_LOG_WARN("mem3 is incorrect");
445 return -EINVAL;
446 }
447
448 /* Check the buffer address offset,
449 * It needs to be divisible by pinfo.stride
450 */
451
452 buf_offset = (uintptr_t)pinfo.fbmem - (uintptr_t)dsc->mem;
453
454 if((buf_offset % dsc->pinfo.stride) != 0) {
455 LV_LOG_WARN("It is detected that buf_offset(%" PRIuPTR ") "
456 "and stride(%d) are not divisible, please ensure "
457 "that the driver handles the address offset by itself.",
458 buf_offset, dsc->pinfo.stride);
459 }
460
461 /* Calculate the address and yoffset of mem3 */
462
463 if(buf_offset == 0) {
464 dsc->mem3_yoffset = dsc->vinfo.yres * 2;
465 dsc->mem3 = (uint8_t *)pinfo.fbmem + dsc->mem3_yoffset * pinfo.stride;
466 LV_LOG_USER("Use consecutive mem3 = %p, yoffset = %" LV_PRIu32,
467 dsc->mem3, dsc->mem3_yoffset);
468 }
469 else {
470 dsc->mem3_yoffset = buf_offset / dsc->pinfo.stride;
471 dsc->mem3 = pinfo.fbmem;
472 LV_LOG_USER("Use non-consecutive mem3 = %p, yoffset = %" LV_PRIu32,
473 dsc->mem3, dsc->mem3_yoffset);
474 }
475
476 return 0;
477}

480{

494 free(dsc->mem_off_screen);

497

501}
