89const void * lv_font_get_bitmap_fmt_txt(lv_font_glyph_dsc_t * g_dsc, lv_draw_buf_t * draw_buf)
90{
91 /* TODO: We can't add an arg check here because this is used in a hot path for all fonts stored as a C array
92 * changing this will mean also changing the font converter so for now these assertions are an exception */
93 LV_ASSERT(g_dsc != NULL);
94
95 const lv_font_t * font = g_dsc->resolved_font;
96 LV_ASSERT(font != NULL);
97 lv_font_fmt_txt_dsc_t * fdsc = (lv_font_fmt_txt_dsc_t *)font->dsc;
98 LV_ASSERT(fdsc != NULL);
99
100 uint32_t gid = g_dsc->gid.index;
101 if(!gid) return NULL;
102
103 const lv_font_fmt_txt_glyph_dsc_t * gdsc = &fdsc->glyph_dsc[gid];
104
105 if(g_dsc->req_raw_bitmap) {
106 /*With a loader the bitmap is read into a scratch buffer which is overwritten by the next
107 *glyph, so the returned pointer is only valid until then. Such fonts have
108 *`static_bitmap == 0` which tells the draw units not to hold on to the pointer.*/
109 const uint8_t * raw = glyph_bitmap_acquire(fdsc, gdsc, gid);
110 if(raw == NULL) return NULL;
111
112 glyph_bitmap_release(fdsc);
113 return raw;
114 }
115
116 LV_ASSERT(draw_buf != NULL);
117 uint8_t * bitmap_out = draw_buf->data;
118 int32_t gsize = (int32_t) gdsc->box_w * gdsc->box_h;
119 if(gsize == 0) return NULL;
120
121 uint32_t stride_in = g_dsc->stride;
122
123
124 if(fdsc->bitmap_format == LV_FONT_FMT_TXT_PLAIN) {
125 LV_ASSERT_FORMAT_MSG(fdsc->bpp == 1 || fdsc->bpp == 2 || fdsc->bpp == 4 || fdsc->bpp == 8, "bpp was %d", fdsc->bpp);
126
127 const uint8_t * bitmap_in = glyph_bitmap_acquire(fdsc, gdsc, gid);
128 if(bitmap_in == NULL) return NULL;
129
130 uint8_t * bitmap_out_tmp = bitmap_out;
131 int32_t x, y;
132 uint32_t stride_out = lv_draw_buf_width_to_stride(gdsc->box_w, LV_COLOR_FORMAT_A8);
133 /*Whole source bytes are read at a time, so a pixel costs a shift and a table lookup
134 *instead of walking a compare chain. `shift` carries the bit position between rows,
135 *because without a stride the rows are packed continuously.*/
136 int32_t shift = 8;
137 uint8_t cur = 0;
138
139 for(y = 0; y < gdsc->box_h; y++) {
140 uint32_t used = 0;
141 if(stride_in) shift = 8; /*with a stride every row starts on a byte boundary*/
142
143 if(fdsc->bpp == 4) {
144 x = 0;
145 if(shift == 4) { /*the row starts mid byte, finish the low nibble*/
146 bitmap_out_tmp[x++] = opa4_table[cur & 0xF];
147 shift = 8;
148 }
149 for(; x + 1 < gdsc->box_w; x += 2) {
150 cur = *bitmap_in++;
151 used++;
152 bitmap_out_tmp[x] = opa4_table[cur >> 4];
153 bitmap_out_tmp[x + 1] = opa4_table[cur & 0xF];
154 }
155 if(x < gdsc->box_w) { /*odd width, the low nibble belongs to the next row*/
156 cur = *bitmap_in++;
157 used++;
158 bitmap_out_tmp[x] = opa4_table[cur >> 4];
159 shift = 4;
160 }
161 }
162 else if(fdsc->bpp == 8) {
163 for(x = 0; x < gdsc->box_w; x++) bitmap_out_tmp[x] = *bitmap_in++;
164 used = gdsc->box_w;
165 }
166 else if(fdsc->bpp == 1 || fdsc->bpp == 2) {
167 int32_t per = 8 / fdsc->bpp;
168 uint32_t mask = (1u << fdsc->bpp) - 1u;
169 x = 0;
170 while(x < gdsc->box_w) {
171 if(shift >= 8) {
172 cur = *bitmap_in++;
173 used++;
174 shift = 0;
175 }
176 int32_t n = per - shift / fdsc->bpp;
177 if(n > gdsc->box_w - x) n = gdsc->box_w - x;
178 while(n--) {
179 uint32_t v = (cur >> (8 - fdsc->bpp - shift)) & mask;
180 bitmap_out_tmp[x++] = fdsc->bpp == 1 ? (v ? 0xff : 0x00) : opa2_table[v];
181 shift += fdsc->bpp;
182 }
183 }
184 if(shift >= 8) shift = 8;
185 }
186
187 if(stride_in) {
188 bitmap_in += stride_in - used;
189 shift = 8;
190 }
191 bitmap_out_tmp += stride_out;
192 }
193
194 glyph_bitmap_release(fdsc);
195
196 lv_draw_buf_flush_cache(draw_buf, NULL);
197 return draw_buf;
198 }
199 /*Handle compressed bitmap*/
200 else {
201#if LV_USE_FONT_COMPRESSED
202 bool prefilter = fdsc->bitmap_format == LV_FONT_FMT_TXT_COMPRESSED;
203 const uint8_t * bitmap_in = glyph_bitmap_acquire(fdsc, gdsc, gid);
204 if(bitmap_in == NULL) return NULL;
205
206 decompress(bitmap_in, bitmap_out, gdsc->box_w, gdsc->box_h,
207 (uint8_t)fdsc->bpp, prefilter);
208 glyph_bitmap_release(fdsc);
209
210 lv_draw_buf_flush_cache(draw_buf, NULL);
211 return draw_buf;
212#else
213 LV_LOG_WARN("Compressed fonts is used but LV_USE_FONT_COMPRESSED is not enabled in lv_conf.h");
214 return NULL;
215#endif
216 }
217
218 /*If not returned earlier then the letter is not found in this font*/
219 return NULL;
220}

222bool lv_font_get_glyph_dsc_fmt_txt(const lv_font_t * font, lv_font_glyph_dsc_t * dsc_out, uint32_t unicode_letter,
223 uint32_t unicode_letter_next)
224{
225 /* TODO: We can't add an arg check here because this is used in a hot path for all fonts stored as a C array
226 * changing this will mean also changing the font converter so for now these assertions are an exception */
227 LV_ASSERT(font != NULL);
228 LV_ASSERT(dsc_out != NULL);
229 /*It fixes a strange compiler optimization issue: https://github.com/lvgl/lvgl/issues/4370*/
230 bool is_tab = unicode_letter == '\t';
231 if(is_tab) {
232 unicode_letter = ' ';
233 }
234 lv_font_fmt_txt_dsc_t * fdsc = (lv_font_fmt_txt_dsc_t *)font->dsc;
235 LV_ASSERT(fdsc != NULL);
236 uint32_t gid = get_glyph_dsc_id(font, unicode_letter);
237 if(!gid) return false;
238
239 int8_t kvalue = 0;
240 if(fdsc->kern_dsc) {
241 uint32_t gid_next = get_glyph_dsc_id(font, unicode_letter_next);
242 if(gid_next) {
243 kvalue = get_kern_value(font, gid, gid_next);
244 }
245 }
246
247 /*Put together a glyph dsc*/
248 const lv_font_fmt_txt_glyph_dsc_t * gdsc = &fdsc->glyph_dsc[gid];
249
250 int32_t kv = ((int32_t)((int32_t)kvalue * fdsc->kern_scale) >> 4);
251
252 uint32_t adv_w = gdsc->adv_w;
253 if(is_tab) adv_w *= 2;
254
255 adv_w = (uint32_t)LV_CLAMP(0, (int32_t)adv_w + kv, INT32_MAX);
256 adv_w = (adv_w + (1 << 3)) >> 4;
257
258 dsc_out->adv_w = adv_w;
259 dsc_out->box_h = gdsc->box_h;
260 dsc_out->box_w = gdsc->box_w;
261 dsc_out->ofs_x = gdsc->ofs_x;
262 dsc_out->ofs_y = gdsc->ofs_y;
263
264 if(fdsc->stride == 0) dsc_out->stride = 0;
265 else {
266 /*E.g. w = 5, bpp = 2, means 2 bytes/line*/
267 uint32_t bit_count = dsc_out->box_w * fdsc->bpp;
268 uint32_t width_in_bytes = (bit_count + 7) >> 3; /*No division round up*/
269
270 /*E.g. font_dsc stride == 4 means align to 4 byte boundary.
271 *In glyph_dsc store the actual line length in bytes*/
272 dsc_out->stride = LV_ROUND_UP(width_in_bytes, fdsc->stride);
273 }
274
275 dsc_out->format = (uint8_t)fdsc->bpp;
276 dsc_out->is_placeholder = false;
277 dsc_out->gid.index = gid;
278
279 if(is_tab) dsc_out->box_w = dsc_out->box_w * 2;
280
281 return true;
282}

297static const uint8_t * glyph_bitmap_acquire(lv_font_fmt_txt_dsc_t * fdsc,
298 const lv_font_fmt_txt_glyph_dsc_t * gdsc, uint32_t gid)
299{
300 if(!fdsc->are_glyphs_dynamic_loaded) {
301 const uint8_t * bitmap = fdsc->glyph_bitmap;
302 return &bitmap[gdsc->bitmap_index];
303 }
304
305 lv_font_fmt_txt_glyph_loader_t * loader = (lv_font_fmt_txt_glyph_loader_t *) fdsc->glyph_bitmap;
306 lv_mutex_lock(&loader->lock);
307
308 const uint8_t * bitmap = loader->get_glyph_bitmap_cb(loader, gid);
309 if(bitmap == NULL) {
310 LV_LOG_WARN("Couldn't load the bitmap of glyph %" LV_PRIu32, gid);
311 lv_mutex_unlock(&loader->lock);
312 }
313
314 return bitmap;
315}

331static uint32_t get_glyph_dsc_id(const lv_font_t * font, uint32_t letter)
332{
333 LV_ASSERT(font != NULL);
334 if(letter == '\0') return 0;
335
336 lv_font_fmt_txt_dsc_t * fdsc = (lv_font_fmt_txt_dsc_t *)font->dsc;
337
338 uint16_t i;
339 for(i = 0; i < fdsc->cmap_num; i++) {
340
341 /*Relative code point*/
342 uint32_t rcp = letter - fdsc->cmaps[i].range_start;
343 if(rcp >= fdsc->cmaps[i].range_length) continue;
344 uint32_t glyph_id = 0;
345 if(fdsc->cmaps[i].type == LV_FONT_FMT_TXT_CMAP_FORMAT0_TINY) {
346 glyph_id = fdsc->cmaps[i].glyph_id_start + rcp;
347 }
348 else if(fdsc->cmaps[i].type == LV_FONT_FMT_TXT_CMAP_FORMAT0_FULL) {
349 const uint8_t * gid_ofs_8 = fdsc->cmaps[i].glyph_id_ofs_list;
350 /* The first character is always valid and should have offset = 0
351 * However if a character is missing it also has offset=0.
352 * So if there is a 0 not on the first position then it's a missing character */
353 if(gid_ofs_8[rcp] == 0 && letter != fdsc->cmaps[i].range_start) continue;
354 glyph_id = fdsc->cmaps[i].glyph_id_start + gid_ofs_8[rcp];
355 }
356 else if(fdsc->cmaps[i].type == LV_FONT_FMT_TXT_CMAP_SPARSE_TINY) {
357 uint16_t key = rcp;
358 uint16_t * p = lv_utils_bsearch(&key, fdsc->cmaps[i].unicode_list, fdsc->cmaps[i].list_length,
359 sizeof(fdsc->cmaps[i].unicode_list[0]), unicode_list_compare);
360
361 if(p) {
362 lv_uintptr_t ofs = p - fdsc->cmaps[i].unicode_list;
363 glyph_id = fdsc->cmaps[i].glyph_id_start + (uint32_t) ofs;
364 }
365 }
366 else if(fdsc->cmaps[i].type == LV_FONT_FMT_TXT_CMAP_SPARSE_FULL) {
367 uint16_t key = rcp;
368 uint16_t * p = lv_utils_bsearch(&key, fdsc->cmaps[i].unicode_list, fdsc->cmaps[i].list_length,
369 sizeof(fdsc->cmaps[i].unicode_list[0]), unicode_list_compare);
370
371 if(p) {
372 lv_uintptr_t ofs = p - fdsc->cmaps[i].unicode_list;
373 const uint16_t * gid_ofs_16 = fdsc->cmaps[i].glyph_id_ofs_list;
374 glyph_id = fdsc->cmaps[i].glyph_id_start + gid_ofs_16[ofs];
375 }
376 }
377
378 return glyph_id;
379 }
380
381 return 0;
382
383}

385static int8_t get_kern_value(const lv_font_t * font, uint32_t gid_left, uint32_t gid_right)
386{
387 LV_ASSERT(font != NULL);
388 lv_font_fmt_txt_dsc_t * fdsc = (lv_font_fmt_txt_dsc_t *)font->dsc;
389 LV_ASSERT(fdsc != NULL);
390
391 int8_t value = 0;
392
393 if(fdsc->kern_classes == 0) {
394 /*Kern pairs*/
395 const lv_font_fmt_txt_kern_pair_t * kdsc = fdsc->kern_dsc;
396 if(kdsc->glyph_ids_size == 0) {
397 /*Use binary search to find the kern value.
398 *The pairs are ordered left_id first, then right_id secondly.*/
399 const uint16_t * g_ids = kdsc->glyph_ids;
400 kern_pair_ref_t g_id_both = {gid_left, gid_right};
401 uint16_t * kid_p = lv_utils_bsearch(&g_id_both, g_ids, kdsc->pair_cnt, 2, kern_pair_8_compare);
402
403 /*If the `g_id_both` were found get its index from the pointer*/
404 if(kid_p) {
405 lv_uintptr_t ofs = kid_p - g_ids;
406 value = kdsc->values[ofs];
407 }
408 }
409 else if(kdsc->glyph_ids_size == 1) {
410 /*Use binary search to find the kern value.
411 *The pairs are ordered left_id first, then right_id secondly.*/
412 const uint32_t * g_ids = kdsc->glyph_ids;
413 kern_pair_ref_t g_id_both = {gid_left, gid_right};
414 uint32_t * kid_p = lv_utils_bsearch(&g_id_both, g_ids, kdsc->pair_cnt, 4, kern_pair_16_compare);
415
416 /*If the `g_id_both` were found get its index from the pointer*/
417 if(kid_p) {
418 lv_uintptr_t ofs = kid_p - g_ids;
419 value = kdsc->values[ofs];
420 }
421
422 }
423 else {
424 /*Invalid value*/
425 }
426 }
427 else {
428 /*Kern classes*/
429 const lv_font_fmt_txt_kern_classes_t * kdsc = fdsc->kern_dsc;
430 uint8_t left_class = kdsc->left_class_mapping[gid_left];
431 uint8_t right_class = kdsc->right_class_mapping[gid_right];
432
433 /*If class = 0, kerning not exist for that glyph
434 *else got the value form `class_pair_values` 2D array*/
435 if(left_class > 0 && right_class > 0) {
436 value = kdsc->class_pair_values[(left_class - 1) * kdsc->right_class_cnt + (right_class - 1)];
437 }
438
439 }
440 return value;
441}

443static int kern_pair_8_compare(const void * ref, const void * element)
444{
445 LV_ASSERT(ref != NULL);
446 LV_ASSERT(element != NULL);
447 const kern_pair_ref_t * ref8_p = ref;
448 const uint8_t * element8_p = element;
449
450 /*If the MSB is different it will matter. If not return the diff. of the LSB*/
451 if(ref8_p->gid_left != element8_p[0]) return ref8_p->gid_left - element8_p[0];
452 else return ref8_p->gid_right - element8_p[1];
453}

478static void decompress(const uint8_t * in, uint8_t * out, int32_t w, int32_t h, uint8_t bpp, bool prefilter)
479{
480 LV_ASSERT(in != NULL);
481 LV_ASSERT(out != NULL);
482 const lv_opa_t * opa_table;
483 switch(bpp) {
484 case 2:
485 opa_table = opa2_table;
486 break;
487 case 3:
488 opa_table = opa3_table;
489 break;
490 case 4:
491 opa_table = opa4_table;
492 break;
493 default:
494 LV_LOG_WARN("%d bpp is not handled", bpp);
495 return;
496 }
497
498 rle_init(in, bpp);
499
500 uint8_t * line_buf1 = lv_malloc(w);
501
502 uint8_t * line_buf2 = NULL;
503
504 if(prefilter) {
505 line_buf2 = lv_malloc(w);
506 }
507
508 decompress_line(line_buf1, w);
509
510 int32_t y;
511 int32_t x;
512 uint32_t stride = lv_draw_buf_width_to_stride(w, LV_COLOR_FORMAT_A8);
513
514 for(x = 0; x < w; x++) {
515 out[x] = opa_table[line_buf1[x]];
516 }
517 out += stride;
518
519 for(y = 1; y < h; y++) {
520 if(prefilter) {
521 decompress_line(line_buf2, w);
522
523 for(x = 0; x < w; x++) {
524 line_buf1[x] = line_buf2[x] ^ line_buf1[x];
525 out[x] = opa_table[line_buf1[x]];
526 }
527 }
528 else {
529 decompress_line(line_buf1, w);
530
531 for(x = 0; x < w; x++) {
532 out[x] = opa_table[line_buf1[x]];
533 }
534 }
535 out += stride;
536 }
537
538 lv_free(line_buf1);
539 lv_free(line_buf2);
540}

563static inline uint8_t get_bits(const uint8_t * in, uint32_t bit_pos, uint8_t len)
564{
565 LV_ASSERT(in != NULL);
566 uint8_t bit_mask;
567 switch(len) {
568 case 1:
569 bit_mask = 0x1;
570 break;
571 case 2:
572 bit_mask = 0x3;
573 break;
574 case 3:
575 bit_mask = 0x7;
576 break;
577 case 4:
578 bit_mask = 0xF;
579 break;
580 case 8:
581 bit_mask = 0xFF;
582 break;
583 default:
584 bit_mask = (uint16_t)((uint16_t) 1 << len) - 1;
585 }
586
587 uint32_t byte_pos = bit_pos >> 3;
588 bit_pos = bit_pos & 0x7;
589
590 if(bit_pos + len >= 8) {
591 uint16_t in16 = (in[byte_pos] << 8) + in[byte_pos + 1];
592 return (in16 >> (16 - bit_pos - len)) & bit_mask;
593 }
594 else {
595 return (in[byte_pos] >> (8 - bit_pos - len)) & bit_mask;
596 }
597}

608 rle->count = 0;

611static inline uint8_t rle_next(void)
612{
613 uint8_t v = 0;
614 uint8_t ret = 0;
615 lv_font_fmt_rle_t * rle = &font_rle;
616
617 if(rle->state == RLE_STATE_SINGLE) {
618 ret = get_bits(rle->in, rle->rdp, rle->bpp);
619 if(rle->rdp != 0 && rle->prev_v == ret) {
620 rle->count = 0;
621 rle->state = RLE_STATE_REPEATED;
622 }
623
624 rle->prev_v = ret;
625 rle->rdp += rle->bpp;
626 }
627 else if(rle->state == RLE_STATE_REPEATED) {
628 v = get_bits(rle->in, rle->rdp, 1);
629 rle->count++;
630 rle->rdp += 1;
631 if(v == 1) {
632 ret = rle->prev_v;
