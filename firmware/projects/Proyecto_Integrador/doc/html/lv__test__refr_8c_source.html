74uint32_t refr_px_read(const uint8_t * buf, uint32_t stride, lv_color_format_t cf, int32_t x, int32_t y)
75{
76 uint32_t px_size = lv_color_format_get_size(cf);
77 const uint8_t * p = buf + (uint32_t)y * stride + (uint32_t)x * px_size;
78 switch(cf) {
79 case LV_COLOR_FORMAT_XRGB8888:
80 return 0xFF000000 | ((uint32_t)p[2] << 16) | ((uint32_t)p[1] << 8) | p[0];
81 case LV_COLOR_FORMAT_ARGB8888:
82 return ((uint32_t)p[3] << 24) | ((uint32_t)p[2] << 16) | ((uint32_t)p[1] << 8) | p[0];
83 case LV_COLOR_FORMAT_RGB888:
84 return 0xFF000000 | ((uint32_t)p[2] << 16) | ((uint32_t)p[1] << 8) | p[0];
85 case LV_COLOR_FORMAT_RGB565: {
86 uint16_t c = (uint16_t)(p[0] | ((uint32_t)p[1] << 8));
87 uint32_t r = ((c >> 11) & 0x1F) << 3;
88 uint32_t gg = ((c >> 5) & 0x3F) << 2;
89 uint32_t b = (c & 0x1F) << 3;
90 return 0xFF000000 | (r << 16) | (gg << 8) | b;
91 }
92 default:
93 return 0;
94 }
95}

197lv_display_t * refr_disp_create(int32_t w, int32_t h, lv_color_format_t cf,
198 lv_display_render_mode_t mode, uint32_t buf_cnt, int32_t buf_h)
199{
200 uint32_t i;
201
202 TEST_ASSERT_TRUE(buf_cnt >= 1 && buf_cnt <= 3);
203
204 refr_ctx_reset();
205 refr_ctx.cf = cf;
206 refr_ctx.hor_res = w;
207 refr_ctx.ver_res = h;
208 refr_ctx.buf_cnt = buf_cnt;
209
210 lv_display_t * disp = lv_display_create(w, h);
211 TEST_ASSERT_NOT_NULL(disp);
212 lv_display_set_color_format(disp, cf);
213
214 for(i = 0; i < buf_cnt; i++) {
215 refr_ctx.buf[i] = lv_draw_buf_create(w, buf_h, cf, LV_STRIDE_AUTO);
216 TEST_ASSERT_NOT_NULL(refr_ctx.buf[i]);
217 }
218 lv_display_set_draw_buffers(disp, refr_ctx.buf[0], buf_cnt > 1 ? refr_ctx.buf[1] : NULL);
219 if(buf_cnt > 2) lv_display_set_3rd_draw_buffer(disp, refr_ctx.buf[2]);
220 lv_display_set_render_mode(disp, mode);
221 lv_display_set_flush_cb(disp, refr_flush_cb);
222
223 refr_hide_sysmon(disp);
224
225 if(cf_is_readable(cf)) {
226 refr_ctx.full_frame_buffer_px_size = lv_color_format_get_size(cf);
227 refr_ctx.full_frame_buffer_stride = (uint32_t)w * refr_ctx.full_frame_buffer_px_size;
228 refr_ctx.full_frame_buffer = lv_malloc_zeroed(refr_ctx.full_frame_buffer_stride * (uint32_t)h);
229 TEST_ASSERT_NOT_NULL(refr_ctx.full_frame_buffer);
230 }
231
232 refr_ctx.disp = disp;
233 return disp;
234}

258uint32_t refr_max_row_of(int32_t area_w, int32_t area_h)
259{
260 uint32_t stride = lv_draw_buf_width_to_stride((uint32_t)area_w, refr_ctx.cf);

262 uint32_t max_row = (refr_ctx.disp->buf_act->data_size - overhead) / stride;
263 if(max_row > (uint32_t)area_h) max_row = (uint32_t)area_h;
264 return max_row;

267void refr_check_flushed_chunks(const lv_area_t * area, uint32_t max_row)
268{
269 uint32_t expected_cnt = ((uint32_t)lv_area_get_height(area) + max_row - 1) / max_row;
270 TEST_ASSERT_EQUAL_UINT32(expected_cnt, refr_ctx.flush_cnt);
271 /*Above `REFR_LOG_MAX` the flushed areas are not recorded any more*/
272 TEST_ASSERT_TRUE(refr_ctx.flush_cnt <= REFR_LOG_MAX);
273
274 uint32_t i;
275 for(i = 0; i < refr_ctx.flush_cnt; i++) {
276 lv_area_t expected;
277 expected.x1 = area->x1;
278 expected.x2 = area->x2;
279 expected.y1 = area->y1 + (int32_t)(i * max_row);
280 expected.y2 = expected.y1 + (int32_t)max_row - 1;
281 if(expected.y2 > area->y2) expected.y2 = area->y2;
282 TEST_ASSERT_TRUE(lv_area_is_equal(&expected, &refr_ctx.flush_area[i]));
283 /*Only the last part of the last area is marked*/
284 TEST_ASSERT_EQUAL(i == refr_ctx.flush_cnt - 1, refr_ctx.flush_is_last[i] != 0);
285 }

346static void full_frame_buffer_copy(lv_display_t * disp, const lv_area_t * area, const uint8_t * px_map)
347{
348 if(refr_ctx.full_frame_buffer == NULL) return;
349
350 /*The flushed area carries the display offset, the full frame buffer does not*/
351 lv_area_t a = *area;
352 lv_area_move(&a, -disp->offset_x, -disp->offset_y);
353 if(a.x1 < 0 || a.y1 < 0 || a.x2 >= refr_ctx.hor_res || a.y2 >= refr_ctx.ver_res) return;
354
355 uint32_t px_size = refr_ctx.full_frame_buffer_px_size;
356 uint32_t src_stride = disp->buf_act->header.stride;
357 bool partial = lv_display_get_render_mode(disp) == LV_DISPLAY_RENDER_MODE_PARTIAL;
358 int32_t w = lv_area_get_width(&a);
359 int32_t h = lv_area_get_height(&a);
360 int32_t y;
361
362 for(y = 0; y < h; y++) {
363 const uint8_t * src;
364 if(partial) src = px_map + (uint32_t)y * src_stride;
365 else src = px_map + (uint32_t)(a.y1 + y) * src_stride + (uint32_t)a.x1 * px_size;
366
367 lv_memcpy(refr_ctx.full_frame_buffer + (uint32_t)(a.y1 + y) * refr_ctx.full_frame_buffer_stride +
368 (uint32_t)a.x1 * px_size,
369 src, (uint32_t)w * px_size);
370 }
371}
