45lv_obj_t * lv_obj_class_create_obj(const lv_obj_class_t * class_p, lv_obj_t * parent)
46{
47 LV_CHECK_ARG(class_p != NULL, return NULL);
48
49 LV_TRACE_OBJ_CREATE("Creating object with %p class on %p parent", (void *)class_p, (void *)parent);
50 uint32_t s = get_instance_size(class_p);
51 lv_obj_t * obj = lv_malloc_zeroed(s);
52 LV_ASSERT_MALLOC(obj);
53 if(obj == NULL) return NULL;
54 obj->class_p = class_p;
55 obj->parent = parent;
56
57 /*Create a screen*/
58 if(parent == NULL) {
59 LV_TRACE_OBJ_CREATE("creating a screen");
60 lv_display_t * disp = lv_display_get_default();
61 if(!disp) {
62 LV_LOG_WARN("No display created yet. No place to assign the new screen");
63 lv_free(obj);
64 return NULL;
65 }
66
67 if(disp->screens == NULL) {
68 disp->screen_cnt = 0;
69 }
70
71 lv_obj_t ** screens = lv_realloc(disp->screens, sizeof(lv_obj_t *) * (disp->screen_cnt + 1));
72 LV_ASSERT_MALLOC(screens);
73 if(screens == NULL) {
74 LV_LOG_WARN("Failed to expand memory for screen array");
75 lv_free(obj);
76 return NULL;
77 }
78
79 disp->screen_cnt++;
80 disp->screens = screens;
81 disp->screens[disp->screen_cnt - 1] = obj;
82
83 /*Set coordinates to full screen size*/
84 obj->coords.x1 = 0;
85 obj->coords.y1 = 0;
86 obj->coords.x2 = lv_display_get_horizontal_resolution(disp) - 1;
87 obj->coords.y2 = lv_display_get_vertical_resolution(disp) - 1;
88 }
89 /*Create a normal object*/
90 else {
91 LV_TRACE_OBJ_CREATE("creating normal object");
92 LV_CHECK_OBJ(parent, MY_CLASS, return NULL);
93
94 if(!lv_obj_allocate_spec_attr(parent)) {
95 lv_free(obj);
96 return NULL;
97 }
98 if(lv_obj_add_child(parent, obj) != LV_RESULT_OK) {
99 lv_free(obj);
100 return NULL;
101 }
102 }
103
104 return obj;
105}

107void lv_obj_class_init_obj(lv_obj_t * obj)
108{
109 LV_CHECK_ARG(obj != NULL, return);
110
111 lv_obj_mark_layout_as_dirty(obj);
112 lv_obj_enable_style_refresh(false);
113
114 lv_theme_apply(obj);
115 lv_obj_construct(obj->class_p, obj);
116
117 lv_obj_enable_style_refresh(true);
118 lv_obj_refresh_style(obj, LV_PART_ANY, LV_STYLE_PROP_ANY);
119
120 lv_obj_refresh_self_size(obj);
121
122 lv_group_t * def_group = lv_group_get_default();
123 if(def_group && lv_obj_is_group_def(obj)) {
124 lv_group_add_obj(def_group, obj);
125 }
126
127 lv_obj_t * parent = lv_obj_get_parent(obj);
128 if(parent) {
129 /*Call the ancestor's event handler to the parent to notify it about the new child.
130 *Also triggers layout update*/
131 lv_obj_send_event(parent, LV_EVENT_CHILD_CHANGED, obj);
132 lv_obj_send_event(parent, LV_EVENT_CHILD_CREATED, obj);
133
134 /*Invalidate the area if not screen created*/
135 lv_obj_invalidate(obj);
136 }
137}

139void lv_obj_destruct(lv_obj_t * obj)
140{
141 LV_CHECK_ARG(obj != NULL, return);
142 LV_CHECK_ARG(obj->class_p != NULL, return);
143
144#if LV_USE_EXT_DATA
145 if(obj->ext_data.free_cb) {
146 obj->ext_data.free_cb(obj->ext_data.data);
147 obj->ext_data.data = NULL;
148 }
149#endif
150
151 if(obj->class_p->destructor_cb) obj->class_p->destructor_cb(obj->class_p, obj);
152
153 if(obj->class_p->base_class) {
154 /*Don't let the descendant methods run during destructing the ancestor type*/
155 obj->class_p = obj->class_p->base_class;
156
157 /*Call the base class's destructor too*/
158 lv_obj_destruct(obj);
159 }

162bool lv_obj_is_editable(const lv_obj_t * obj)
163{
164 LV_CHECK_ARG(obj != NULL, return false);
165
166 const lv_obj_class_t * class_p = obj->class_p;
167
168 /*Find a base in which editable is set*/
169 while(class_p && class_p->editable == LV_OBJ_CLASS_EDITABLE_INHERIT) class_p = class_p->base_class;
170
171 if(class_p == NULL) return false;
172
173 return class_p->editable == LV_OBJ_CLASS_EDITABLE_TRUE;
174}

204static void lv_obj_construct(const lv_obj_class_t * class_p, lv_obj_t * obj)
205{
206 LV_ASSERT_NULL(class_p);
207 LV_ASSERT_NULL(obj);
208 LV_ASSERT_NULL(obj->class_p);
209
210#if LV_USE_OBJ_NAME
211 LV_ASSERT_NULL(class_p->name);
212#endif
213
214#if LV_USE_EXT_DATA
215 obj->ext_data.free_cb = NULL;
216 obj->ext_data.data = NULL;
217#endif
218
219 if(obj->class_p->base_class) {
220 const lv_obj_class_t * original_class_p = obj->class_p;
221
222 /*Don't let the descendant methods run during constructing the ancestor type*/
223 obj->class_p = obj->class_p->base_class;
224
225 /*Construct the base first*/
226 lv_obj_construct(class_p, obj);
227
228 /*Restore the original class*/
229 obj->class_p = original_class_p;
230 }
231
232 if(obj->class_p->constructor_cb) obj->class_p->constructor_cb(class_p, obj);
233}

235static uint32_t get_instance_size(const lv_obj_class_t * class_p)
236{
237 /*Find a base in which instance size is set*/
238 while(class_p && class_p->instance_size == 0) class_p = class_p->base_class;
239
240 LV_ASSERT(class_p != NULL); /*Never happens: set at least in `lv_obj` class*/
241
242 return class_p->instance_size;
