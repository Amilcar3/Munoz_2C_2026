91 /*Animate out the previous content*/

102 uint32_t child_cnt_prev = lv_obj_get_child_count(cont);
103 for(i = 0; i < child_cnt_prev; i++) {

105 lv_anim_set_var(&a, child);

119 if(cur_v == 0) {

121 lv_obj_t * slider = lv_slider_create(cont);
122 lv_slider_bind_value(slider, slider_subject[i]);
123 lv_obj_align(slider, LV_ALIGN_TOP_MID, 0, 10 + i * 30);
124 }

136 lv_roller_bind_value(roller, roller_subject[i]);
137 lv_obj_align(roller, LV_ALIGN_CENTER, - 80 + i * 160, 0);
138 }
139 }

142 lv_anim_set_completed_cb(&a, NULL);
143 for(i = child_cnt_prev; i < lv_obj_get_child_count(cont); i++) {
144 lv_obj_t * child = lv_obj_get_child(cont, i);
145 lv_anim_set_var(&a, child);
