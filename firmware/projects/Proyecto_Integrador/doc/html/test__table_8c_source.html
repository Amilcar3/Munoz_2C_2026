137static void draw_part_event_cb(lv_event_t * e)
138{
139 lv_draw_task_t * draw_task = lv_event_get_draw_task(e);
140 lv_draw_dsc_base_t * base_dsc = draw_task->draw_dsc;
141 /*If the cells are drawn...*/
142 if(base_dsc->part == LV_PART_ITEMS) {
143 uint32_t row = base_dsc->id1;
144 uint32_t col = base_dsc->id2;
145
146 /*Make the texts in the first cell center aligned*/
147 if(row == 0) {
148 lv_draw_label_dsc_t * label_draw_dsc = lv_draw_task_get_label_dsc(draw_task);
149 if(label_draw_dsc) {
150 label_draw_dsc->align = LV_TEXT_ALIGN_CENTER;
151 }
152 lv_draw_fill_dsc_t * fill_draw_dsc = lv_draw_task_get_fill_dsc(draw_task);
153 if(fill_draw_dsc) {
154 fill_draw_dsc->color = lv_color_mix(lv_palette_main(LV_PALETTE_BLUE), fill_draw_dsc->color, LV_OPA_20);
155 fill_draw_dsc->opa = LV_OPA_COVER;
156 }
157 }
158 /*In the first column align the texts to the right*/
159 else if(col == 0) {
160 lv_draw_label_dsc_t * label_draw_dsc = lv_draw_task_get_label_dsc(draw_task);
161 if(label_draw_dsc) {
162 label_draw_dsc->align = LV_TEXT_ALIGN_RIGHT;
163 }
164 }
165
166 /*Make every 2nd row grayish*/
167 if((row != 0 && row % 2) == 0) {
168 lv_draw_fill_dsc_t * fill_draw_dsc = lv_draw_task_get_fill_dsc(draw_task);
169 if(fill_draw_dsc) {
170 fill_draw_dsc->color = lv_color_mix(lv_palette_main(LV_PALETTE_GREY), fill_draw_dsc->color, LV_OPA_10);
171 fill_draw_dsc->opa = LV_OPA_COVER;
172 }
173 }
174 }
