1"""Phase 2: assemble whole header files from the typed model.
2
3The :class:`Emitter` walks the Kconfig tree, tracking the top-level "group"
4menus (for banners) and the stack of open ``#if`` conditions (for dependency
5nesting). The actual ``#define`` text for each node comes from its typed
6:class:`~config_headers.config_entry.ConfigEntry` (``emit_template`` /
7``emit_internal``); this module only handles *structure*: banners, ``#if``
8blocks, the constant "Config options" block, and the static preamble/footer.

38class Emitter:
39 """Renders one target (``"template"``, ``"internal"``) of the tree."""

41 def __init__(self, kconf: Kconfig, target: str, entries):
42 assert target in ("internal", "template")
43 self.kconf = kconf
44 self.target = target
45 self.enum_choices = enum_backed_choices(kconf)
46 self.out: list[str] = []
47 self.cond_stack: list[str] = []
48 self.emitted: set[str] = set()
49 self.deferred: list[DerivedFlag] = [] # internal: emitted after the body
50 # internal: emitted after the compatibility block, once their selector
51 # macro is defined
52 self.derived_consts: list[DerivedConstToken] = []
53 # member symbol -> (macro, token): rewrites a `#if <member>` guard (only
54 # valid on the Kconfig path) into `<macro> == <token>` (valid on both).
55 self.guard: dict[str, tuple[str, str]] = enum_guard_map(entries)

59 def _render_cond(self, term) -> str:
60 if isinstance(term, tuple) and term[0] == NOT and isinstance(term[1], Symbol):
61 sym = term[1]
62 if sym.name in self.guard:
63 macro, token = self.guard[sym.name]
64 return f"{macro} != {token}"
65 if sym.choice is not None and sym.choice.name:
66 return f"{sym.choice.name} != {sym.name}"
67 return f"!{sym.name}"
68 if isinstance(term, Symbol):
69 if term.name in self.guard:
70 macro, token = self.guard[term.name]
71 return f"{macro} == {token}"
72 if term.choice is not None and term.choice.name:
73 return f"{term.choice.name} == {term.name}"
74 return term.name
75 return term_key(term)
76
77 def _sync_conditions(self, wanted: list, base_keys: set[str]):
78 def is_visible(sym):
79 return any(node.prompt for node in sym.nodes)
80
81 # Drop terms that shouldn't appear as conditions in the generated guard:
82 # - Choice objects (a member's implicit dep renders as `<choice>`, not valid C)
83 # - derived-combo symbols, which would gate their own selector members circularly
84 # - symbols with no prompt on any of their nodes
85 deduped, seen = [], set(base_keys)
86 for t in wanted:
87 if isinstance(t, Choice):
88 continue
89 if isinstance(t, Symbol) and not is_visible(t):
90 continue
91 k = term_key(t)
92 if k not in seen:
93 seen.add(k)
94 deduped.append(t)
95 wanted_conds = [self._render_cond(t) for t in deduped]
96
97 popped = False
98 while self.cond_stack and self.cond_stack[-1] not in wanted_conds:
99 self.out.append(f"#endif /*{self.cond_stack.pop()}*/")
100 popped = True
101 if popped:
102 self.out.append("")
103 for cond in wanted_conds:
104 if cond not in self.cond_stack:
105 self.out.append(f"#if {cond}")
106 self.cond_stack.append(cond)
107
108 def _close_all_conditions(self):
109 while self.cond_stack:
110 self.out.append(f"#endif /*{self.cond_stack.pop()}*/")
111 self.out.append("")

115 def _emit_entry(self, node, entry, base_keys: set[str]):
116 # Derived flags reference operands defined later in the file, so in the
117 # internal header they are deferred to the footer, not emitted inline.
118 if isinstance(entry, DerivedFlag):
119 if self.target == "internal":
120 self.deferred.append(entry)
121 return
122 if isinstance(entry, DerivedConstToken):
123 if self.target == "internal":
124 self.derived_consts.append(entry)
125 return
126 if self.target == "template" and entry.name in KCONFIG_ONLY_CONFIGS:
127 return
128 lines = (
129 entry.emit_template()
130 if self.target == "template"
131 else entry.emit_internal()
132 )
133 if not lines:
134 return
135 if self.target != "internal":
136 self._sync_conditions(dep_terms(node.dep), base_keys)
137 self.out += lines
138 self.out.append("")
139
140 def walk(self, node, base_keys: set[str]):
141 while node:
142 item = node.item
143 if item is MENU or item is COMMENT:
144 if node.list:
145 self.walk(node.list, base_keys)
146 else:
147 entry = classify(node, self.enum_choices)
148 if entry is not None and entry.name not in self.emitted:
149 self.emitted.add(entry.name)
150 self._emit_entry(node, entry, base_keys)
151 if node.list:
152 self.walk(node.list, base_keys)
153 node = node.next
154
155 def emit_group(self, menu_node):
156 title = menu_node.prompt[0] if menu_node.prompt else ""
157 base_keys = {term_key(t) for t in dep_terms(menu_node.dep)}
158 banner = "/*" + "=" * 76 + "\n" f" * {title.upper()}\n" " *" + "=" * 76 + "*/"
159 self.out.append(banner)
160 self.out.append("")
161 if menu_node.list:
162 self.walk(menu_node.list, base_keys)
163 self._close_all_conditions()
164 self.out.append("")

167def collect_groups(kconf: Kconfig) -> list:
168 """Top-level menu nodes under the root, in Kconfig order."""
169 root = kconf.top_node.list
170 if root and root.item is MENU and root.next is None:
171 root = root.list # descend into a single outer "LVGL" menu
172 groups, node = [], root
173 while node:
174 if node.item is MENU and node.prompt:
175 groups.append(node)
176 node = node.next
177 return groups

186def enum_guard_map(entries) -> dict:
187 """Map each enum choice *member* symbol to ``(macro, token)`` so a bare
188 ``#if <member>`` can be rewritten into ``<macro> == <token>`` - valid on
189 both the Kconfig and lv_conf.h paths."""
190 guard: dict[str, tuple[str, str]] = {}
191 for e in entries:
192 if isinstance(e, EnumChoice):
193 for m in e.members:
194 if m.member_name:
195 guard[m.member_name] = (e.name, m.token)
196 return guard

199def _refs_pointer_token(refs, guard) -> bool:
200 """True if any referenced symbol is an enum member whose token is a *pointer*
201 (the ``LV_FONT_DEFAULT_*`` font selectors): ``<macro> == <token>`` is then a
202 pointer comparison the C preprocessor can't evaluate, so the check that needs
203 it can't be expressed and must be skipped. Integer-valued members (the
204 ``LV_OS_*`` family etc.) are fine."""
205 for r in refs:
206 gm = guard.get(r.name)
207 if gm and gm[1] in templates.BUILTIN_FONTS:
208 return True
209 return False

212def constraint_checks(entries) -> list[ConstraintCheck]:
213 """Build the ``#error`` guards that replay Kconfig ``select`` / ``depends on``
214 on the hand-written ``lv_conf.h`` path. Only :class:`BoolConfig` options are
215 guarded (the constraint is "this on/off option needs ...").
216
217 * **select**: something ``select``s the option, so enabling the selector
218 requires the option on -> error when ``(selectors) && !option``.
219 * **depends on**: the option needs its dependency met -> error when
220 ``option && !(deps)``. Restricted to options that *default off*: a
221 default-on sub-option (e.g. ``LV_LOG_TRACE_*``) emits ``1`` even when its
222 parent feature is off - because ``bool_default`` strips ``depends on`` -
223 which would make the default/``LV_CONF_SKIP`` build error spuriously. A
224 default-off option emits ``0``, so its check can only fire when the user
225 explicitly turns it on - exactly the case worth catching.
226 """
227 guard = enum_guard_map(entries)
228 checks: list[ConstraintCheck] = []
229 for e in entries:
230 if not isinstance(e, BoolConfig) or e.node is None:
231 continue
232 sym = e.node.item
233 kn = sym.kconfig.n
234
235 # select: enabling a selector requires this option to be on
236 if sym.rev_dep is not kn:
237 refs: set = set()
238 collect_sym_refs(sym.rev_dep, refs)
239 sel = rev_dep_c_expr(sym.rev_dep, guard)
240 if sel is not None and not _refs_pointer_token(refs, guard):
241 checks.append(
242 ConstraintCheck(
243 sym.name,
244 f"({sel}) && !{sym.name}",
245 f"{sym.name} must be enabled: Kconfig selects it from "
246 f"{sel}",
247 node=e.node,
248 )
249 )
250
251 # depends on: the option may only be enabled if its dependency is met
252 dd = sym.direct_dep
253 if dd is not sym.kconfig.y and dd is not kn and bool_default(sym) == "0":
254 refs = set()
255 collect_sym_refs(dd, refs)
256 dep = rev_dep_c_expr(dd, guard)
257 if dep is not None and not _refs_pointer_token(refs, guard):
258 checks.append(
259 ConstraintCheck(
260 sym.name,
261 f"{sym.name} && !({dep})",
262 f"{sym.name} requires {dep} (Kconfig depends on)",
263 node=e.node,
264 )
265 )
266 return checks

274def render_config_options(entries) -> str:
275 """The constant ``#define`` block at the top of lv_conf_internal.h: built-in
276 font pointers, the named integer constants (:class:`ConstToken`, e.g.
277 ``LV_STDLIB_BUILTIN 0``), and the enum tokens whose values live inline in a
278 ``MEMBER_IS_TOKEN`` choice (``LV_OS_*``, ``LV_NEMA_*``, ...).
279
280 Each token is produced by exactly one entry - a ``ConstToken`` is defined by
281 its own config, and the remaining enum tokens belong to a single choice - so
282 there is nothing to de-duplicate.
283 """
284 blocks: list[str] = []
285
286 # The font pointer table (the only non-Kconfig data) comes first.
287 fonts = templates.BUILTIN_FONTS
288 fwidth = max(len(t) for t in fonts)
289 blocks.append(
290 "\n".join(
291 ["/* Built-in font selectors for LV_FONT_DEFAULT */"]
292 + [f"#define {t.ljust(fwidth)} {v}" for t, v in fonts.items()]
293 )
294 )
295
296 # Named integer constants (LV_STDLIB_BUILTIN -> 0, LV_LOG_LEVEL_NUM -> 5).
297 consts = [e for e in entries if isinstance(e, ConstToken)]
298 if consts:
299 cwidth = max(len(e.name) for e in consts)
300 blocks.append(
301 "\n".join(
302 ["/* Named constants */"]
303 + [f"#define {e.name.ljust(cwidth)} {e.value}" for e in consts]
304 )
305 )
306
307 # Enum tokens whose values live inline in the choice (LV_OS_*, LV_NEMA_*).
308 for e in entries:
309 if isinstance(e, EnumChoice):
310 lines = e.emit_internal_options()
311 if lines:
312 blocks.append("\n".join(lines))
313 return "\n\n".join(blocks)

321def _body(kconf: Kconfig, target: str, entries) -> Emitter:
322 em = Emitter(kconf, target, entries)
323 groups = collect_groups(kconf)
324 if groups:
325 for group in groups:
326 em.emit_group(group)
327 else:
328 # No top-level menus (e.g. a flat test fixture): walk the root directly.
329 em.walk(kconf.top_node.list, set())
330 em._close_all_conditions()
331 return em

371def custom_includes(entries) -> list[tuple[str, str]]:
372 """``(gate, path)`` pairs following the custom-include convention: a string
373 option ``LV_<X>_CUSTOM_INCLUDE`` with a companion bool ``LV_<X>_USE_CUSTOM_INCLUDE``.
374
375 These are optional user headers that override config macros (the sysmon /
376 font / nema pattern). The internal header ``#include``s ``path`` when ``gate``
377 is set, so individual source files don't each have to. An option that lacks
378 the companion bool (e.g. the enum-gated ``LV_OS_CUSTOM_INCLUDE``) is left to
379 its own subsystem and not auto-included here."""
380 names = {e.name for e in entries}
381 out: list[tuple[str, str]] = []
382 suffix = "_CUSTOM_INCLUDE"
383 for e in entries:
384 n = e.name
385 if not (n.endswith(suffix) and not n.endswith("_USE" + suffix)):
386 continue
387 gate = n[: -len(suffix)] + "_USE" + suffix
388 if gate not in names:
389 raise SystemExit(
390 f"{n}: missing companion gate {gate}. Add a bool "
391 f"`config {gate}`, or list {n} in CUSTOM_INCLUDE_NO_GATE if it "
392 f"is gated some other way."
393 )
394 if n in INCLUDED_BY_SUBSYSTEM:
395 continue # has a gate but its subsystem does the include (see below)
396 out.append((gate, n))
397 return out
398
399
400# Custom-include configs that DO follow the gate convention but are *not*
401# auto-included here because their subsystem includes them at a specific point.
402# LV_GLOBAL_CUSTOM_INCLUDE declares lv_global_default() (needs lv_global_t) and
403# defaults to "lv_global.h", so lv_global.h includes it after the type exists;
404# pulling it into lv_conf_internal.h (before any LVGL type) would not compile.
405INCLUDED_BY_SUBSYSTEM: set[str] = {"LV_GLOBAL_CUSTOM_INCLUDE"}

408def _order_deferred(flags):
409 """Order derived flags so one referencing another is emitted after it.
410
411 Most derived flags reference only ordinary options (defined earlier in the
412 body), but some aggregate other derived flags - e.g. LV_USE_EGL is the OR of
413 the per-driver LV_*_USE_EGL flags. Since a ``#if`` needs every macro already
414 defined, emit referenced flags first (stable topological order)."""
415 names = {f.name for f in flags}
416
417 def refs(f):
418 toks = set(re.findall(r"[A-Za-z_]\w*", f.selectors_expr or ""))
419 return {t for t in toks if t in names and t != f.name}
420
421 ordered, done, remaining = [], set(), list(flags)
422 while remaining:
423 ready = [f for f in remaining if refs(f) <= done]
424 assert ready
425 for f in ready:
426 ordered.append(f)
427 done.add(f.name)
428 remaining.remove(f)
429 return ordered

432def generate_internal(kconf: Kconfig, entries) -> str:
433 em = _body(kconf, "internal", entries)
434 options = render_config_options(entries)
435 preamble = templates.INTERNAL_PREAMBLE.replace("__CONFIG_OPTIONS__", options)
436
437 # Values derived from a selected option, emitted straight after the
438 # compatibility block: the selector macro is defined by the body above, and
439 # the derived capability flags below may read the result.
440 derived_consts: list[str] = []
441 if em.derived_consts:
442 derived_consts += [
443 "",
444 "/* Values fixed by another option's selected token (see LV_CONF_PASTE). */",
445 "",
446 ]
447 for const in em.derived_consts:
448 derived_consts += const.emit_internal()
449 derived_consts += ["", ""]
450
451 deferred: list[str] = []
452 if em.deferred:
453 for flag in em.deferred:
454 if flag.node is not None:
455 sym = flag.node.item
456 sel = rev_dep_c_expr(sym.rev_dep, em.guard)
457 dep = rev_dep_c_expr(sym.direct_dep, em.guard)
458 if sel is not None and dep is not None and dep != "1":
459 sel = f"({sel}) && ({dep})"
460 flag.selectors_expr = sel
461 deferred.append("/* Derived capability flags (set via Kconfig `select`). */")
462 for flag in _order_deferred(em.deferred):
463 deferred += flag.emit_internal()
464 deferred.append("")
465
466 guards: list[str] = []
467 checks = constraint_checks(entries)
468 if checks:
469 guards.append("")
470 guards.append(
471 "/* Kconfig enforces `depends on` / `select`; these checks catch a"
472 )
473 guards.append(" * hand-written lv_conf.h that violates them. */")
474 for c in checks:
475 guards += c.emit_internal()
476 guards.append("")
477
478 # Optional user headers that override config macros: include each one (once
479 # both its gate and path are defined) so source files don't have to.
480 custom_inc: list[str] = []
481 ci = custom_includes(entries)
482 if ci:
483 custom_inc.append("")
484 custom_inc.append(
485 "/* Optional user headers (LV_*_USE_CUSTOM_INCLUDE) overriding config macros. */"
486 )
487 for gate, path in ci:
488 custom_inc.append(f"#if {gate}")
489 custom_inc.append(f" #include {path}")
490 custom_inc.append("#endif")
491 custom_inc.append("")
492
493 return (
494 preamble
495 + "\n"
496 + "\n".join(em.out)
497 + "\n"
498 + templates.INTERNAL_COMPATIBILITY_BLOCK
499 + "\n"
500 + "\n".join(derived_consts)
501 + "\n".join(deferred)
502 + "\n".join(custom_inc)
503 + templates.INTERNAL_FOOTER
504 + "\n".join(guards)
505 + templates.INTERNAL_CLOSE
