47void lv_draw_vg_lite_img(lv_draw_task_t * t, const lv_draw_image_dsc_t * dsc,
48 const lv_area_t * coords, bool no_cache)
49{
50 lv_draw_vg_lite_unit_t * u = (lv_draw_vg_lite_unit_t *)t->draw_unit;
51
52 lv_area_t clip_area;
53 if(!lv_area_intersect(&clip_area, &t->_real_area, &t->clip_area)) {
54 /*Fully clipped, nothing to do*/
55 return;
56 }
57
58 LV_PROFILER_DRAW_BEGIN;
59
60 vg_lite_buffer_t src_buf;
61 lv_image_decoder_dsc_t decoder_dsc;
62
63 /* if not support blend normal, premultiply alpha */
64 bool premultiply = !lv_vg_lite_support_blend_normal();
65 if(!lv_vg_lite_buffer_open_image(&src_buf, &decoder_dsc, dsc->src, no_cache, premultiply)) {
66 LV_PROFILER_DRAW_END;
67 return;
68 }
69
70 vg_lite_color_t color = lv_vg_lite_image_recolor(&src_buf, dsc);
71
72 /* convert the blend mode to vg-lite blend mode, considering the premultiplied alpha */
73 bool has_pre_mul = lv_draw_buf_has_flag(decoder_dsc.decoded, LV_IMAGE_FLAGS_PREMULTIPLIED)
74 || (decoder_dsc.decoded->header.cf == LV_COLOR_FORMAT_ARGB8888_PREMULTIPLIED);
75 vg_lite_blend_t blend = lv_vg_lite_blend_mode(dsc->blend_mode, has_pre_mul);
76
77 /* original image matrix */
78 vg_lite_matrix_t image_matrix;
79 vg_lite_identity(&image_matrix);
80 lv_vg_lite_image_matrix(&image_matrix, coords->x1, coords->y1, dsc);
81
82 /* image drawing matrix */
83 vg_lite_matrix_t matrix = u->global_matrix;
84 lv_vg_lite_matrix_multiply(&matrix, &image_matrix);
85
86 const bool has_transform = matrix_has_transform(&matrix);
87 const bool has_perspective = lv_vg_lite_matrix_has_perspective(&matrix);
88 const vg_lite_filter_t filter = has_transform ? VG_LITE_FILTER_BI_LINEAR : VG_LITE_FILTER_POINT;
89
90 /* Use coords as the fallback image width and height */
91 const uint32_t img_w = dsc->header.w ? dsc->header.w : lv_area_get_width(coords);
92 const uint32_t img_h = dsc->header.h ? dsc->header.h : lv_area_get_height(coords);
93
94 const bool need_clip = !lv_area_is_in(&t->_real_area, &t->clip_area, false);
95
96 if(dsc->colorkey) {
97 lv_vg_lite_set_color_key(dsc->colorkey);
98 }
99
107 if(has_perspective && !dsc->tile) {
108 /* The scissor works in the coordinate system of the frame buffer */
109 lv_area_t phy_clip_area;
110 bool phy_clip_area_ok = lv_vg_lite_matrix_transform_area(&phy_clip_area, &u->global_matrix, &clip_area);
111 const lv_area_t scissor_area_ori = u->current_scissor_area;
112 bool scissor_changed = false;
113
114 if(phy_clip_area_ok && vg_lite_query_feature(gcFEATURE_BIT_VG_SCISSOR)) {
115 lv_area_t scissor_area;
116 if(!lv_area_intersect(&scissor_area, &phy_clip_area, &scissor_area_ori)) {
117 /*Fully clipped, nothing to do*/
118 if(dsc->colorkey) {
119 lv_vg_lite_set_color_key(NULL);
120 }
121 lv_vg_lite_pending_add(u->image_dsc_pending, &decoder_dsc);
122 LV_PROFILER_DRAW_END;
123 return;
124 }
125
126 phy_clip_area = scissor_area;
127
128 if(need_clip) {
129 lv_vg_lite_set_scissor_area(u, &phy_clip_area);
130 scissor_changed = true;
131 }
132 }
133
134 if(dsc->clip_radius > 0) {
140 lv_vg_lite_path_t * radius_path = lv_vg_lite_path_get(u, VG_LITE_FP32);
141 lv_vg_lite_path_set_transform(radius_path, &matrix);
142 lv_vg_lite_path_append_rect(
143 radius_path,
144 dsc->image_area.x1 - coords->x1, dsc->image_area.y1 - coords->y1,
145 lv_area_get_width(&dsc->image_area), lv_area_get_height(&dsc->image_area),
146 dsc->clip_radius);
147
148 if(phy_clip_area_ok) {
149 lv_vg_lite_path_set_bounding_box_area(radius_path, &phy_clip_area);
150 }
151 lv_vg_lite_path_end(radius_path);
152
153 vg_lite_matrix_t identity_matrix;
154 vg_lite_identity(&identity_matrix);
155
156 lv_vg_lite_draw_pattern(
157 &u->target_buffer,
158 lv_vg_lite_path_get_path(radius_path),
159 VG_LITE_FILL_EVEN_ODD,
160 &identity_matrix,
161 &src_buf,
162 &matrix,
163 blend,
164 VG_LITE_PATTERN_COLOR,
165 0,
166 color,
167 filter);
168
169 lv_vg_lite_path_drop(u, radius_path);
170 }
171 else {
172 lv_vg_lite_blit(
173 &u->target_buffer,
174 &src_buf,
175 &matrix,
176 blend,
177 color,
178 filter);
179 }
180
181 if(scissor_changed) {
182 /* Restore the scissor area of the layer */
183 lv_vg_lite_set_scissor_area(u, &scissor_area_ori);
184 }
185
186 if(dsc->colorkey) {
187 lv_vg_lite_set_color_key(NULL);
188 }
189
190 lv_vg_lite_pending_add(u->image_dsc_pending, &decoder_dsc);
191 LV_PROFILER_DRAW_END;
192 return;
193 }
194
195 /* If clipping is not required, blit directly */
196 if(!need_clip && dsc->clip_radius <= 0 && !dsc->tile) {
197 /* rect is used to crop the pixel-aligned padding area */
198 vg_lite_rectangle_t rect = {
199 .x = 0,
200 .y = 0,
201 .width = img_w,
202 .height = img_h,
203 };
204
205 lv_vg_lite_blit_rect(
206 &u->target_buffer,
207 &src_buf,
208 &rect,
209 &matrix,
210 blend,
211 color,
212 filter);
213
214 if(dsc->colorkey) {
215 lv_vg_lite_set_color_key(NULL);
216 }
217
218 lv_vg_lite_pending_add(u->image_dsc_pending, &decoder_dsc);
219 LV_PROFILER_DRAW_END;
220 return;
221 }
222
223 lv_vg_lite_path_t * path = lv_vg_lite_path_get(u, VG_LITE_FP32);
224
225 if(dsc->tile) {
226 /* When the image is tiled, use coords as the tile area and create a path around it */
227 lv_vg_lite_path_append_rect(
228 path,
229 coords->x1, coords->y1,
230 lv_area_get_width(coords), lv_area_get_height(coords),
231 dsc->clip_radius);
232 }
233 else if(has_transform || dsc->clip_radius) {
238 lv_vg_lite_path_set_transform(path, &image_matrix);
239
240 /* Each point will be transformed accordingly. */
241 lv_vg_lite_path_append_rect(
242 path,
243 dsc->image_area.x1 - coords->x1, dsc->image_area.y1 - coords->y1,
244 lv_area_get_width(&dsc->image_area), lv_area_get_height(&dsc->image_area),
245 dsc->clip_radius);
246 }
247 else {
248 /* append normal rect to the path */
249 lv_vg_lite_path_append_rect(
250 path,
251 clip_area.x1, clip_area.y1,
252 lv_area_get_width(&clip_area), lv_area_get_height(&clip_area),
253 0);
254 }
255
256 lv_vg_lite_path_set_bounding_box_area(path, &clip_area);
257 lv_vg_lite_path_end(path);
258
259 vg_lite_matrix_t path_matrix = u->global_matrix;
260
261 if(dsc->tile) {
262 lv_area_t tile_area;
263 if(lv_area_get_width(&dsc->image_area) >= 0) {
264 tile_area = dsc->image_area;
265 }
266 else {
267 tile_area = *coords;
268 }
269 lv_area_set_width(&tile_area, img_w);
270 lv_area_set_height(&tile_area, img_h);
271
276#if LV_USE_VG_LITE_THORVG
277 const int32_t tile_x_start = tile_area.x1;
278 while(tile_area.y1 <= coords->y2) {
279 while(tile_area.x1 <= coords->x2) {
280#endif
281 vg_lite_matrix_t tile_matrix;
282 vg_lite_identity(&tile_matrix);
283 vg_lite_translate(tile_area.x1 - coords->x1, tile_area.y1 - coords->y1, &tile_matrix);
284
285 vg_lite_matrix_t pattern_matrix = matrix;
286 lv_vg_lite_matrix_multiply(&pattern_matrix, &tile_matrix);
287
288 lv_area_t clipped_img_area;
289 if(!LV_USE_VG_LITE_THORVG || lv_area_intersect(&clipped_img_area, &tile_area, coords)) {
290 lv_vg_lite_draw_pattern(
291 &u->target_buffer,
292 lv_vg_lite_path_get_path(path),
293 VG_LITE_FILL_EVEN_ODD,
294 &path_matrix,
295 &src_buf,
296 &pattern_matrix,
297 blend,
298 VG_LITE_PATTERN_REPEAT,
299 0,
300 color,
301 filter);
302 }
303
304#if LV_USE_VG_LITE_THORVG
305 tile_area.x1 += img_w;
306 tile_area.x2 += img_w;
307 }
308
309 tile_area.y1 += img_h;
310 tile_area.y2 += img_h;
311 tile_area.x1 = tile_x_start;
312 tile_area.x2 = tile_x_start + img_w - 1;
313 }
314#endif
315 }
316 else {
317 lv_vg_lite_draw_pattern(
318 &u->target_buffer,
319 lv_vg_lite_path_get_path(path),
320 VG_LITE_FILL_EVEN_ODD,
321 &path_matrix,
322 &src_buf,
323 &matrix,
324 blend,
325 VG_LITE_PATTERN_COLOR,
326 0,
327 color,
328 filter);
329 }
330
331 if(dsc->colorkey) {
332 lv_vg_lite_set_color_key(NULL);
333 }
334
335 lv_vg_lite_path_drop(u, path);
336
337 lv_vg_lite_pending_add(u->image_dsc_pending, &decoder_dsc);
338 LV_PROFILER_DRAW_END;
339}

345static inline bool matrix_has_transform(const vg_lite_matrix_t * matrix)
346{
351 return !((matrix->m[0][0] == 1.0f || matrix->m[0][0] == -1.0f) &&
352 matrix->m[0][1] == 0.0f &&
353 matrix->m[1][0] == 0.0f &&
354 (matrix->m[1][1] == 1.0f || matrix->m[1][1] == -1.0f) &&
355 matrix->m[2][0] == 0.0f &&
356 matrix->m[2][1] == 0.0f &&
357 matrix->m[2][2] == 1.0f);
358}
