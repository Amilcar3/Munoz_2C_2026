1#!/usr/bin/env python3
2"""Wrap XML files as Markdown so they can be consumed by systems that accept
3.md but not .xml. Each `foo.xml` becomes `foo.md` with its contents wrapped in
4a fenced code block titled with the original filename, optionally annotated with
5a `source` URL pointing back at the original .xml:
6
7 ```xml title="foo.xml" source="https://.../foo.xml"
8 <the original xml>
9 ```
10
11The `source` URL is built from a base you pass on the CLI plus the file's path
12within the source tree. For example, with
13
14 --source-base https://github.com/lvgl/lvgl/blob/master/examples
15
16a file at SRC_DIR/scroll/scroll_scrollable/lv_example_scroll_scrollable.xml gets
17
18 source="https://github.com/lvgl/lvgl/blob/master/examples/scroll/scroll_scrollable/lv_example_scroll_scrollable.xml"
19
20Two ways to call it:
21
22 Single file (in-place rename, original .xml is removed):
23 python3 xml_to_md.py path/to/file.xml [--source-base URL]
24
25 Folder (recursive, mirrored to an output dir, source left untouched):
26 python3 xml_to_md.py SRC_DIR OUT_DIR [--source-base URL]
27
28The folder mode walks SRC_DIR recursively. For every .xml it finds it recreates
29the matching subfolder under OUT_DIR and writes the .md there. Subfolders that
30contain no XML are never created. Designed to run unattended in CI.
31
32Standalone: depends only on the Python standard library (Python 3.8+).
33"""
34

42def to_markdown(xml: str, title: str, source: str | None) -> str:
43 # Ensure exactly one trailing newline before the closing fence.
44 body = re.sub(r"\n+$", "", xml)
45 attrs = f'title="{title}"'
46 if source is not None:
47 attrs += f' source="{source}"'
48 return f"```xml {attrs}\n{body}\n```\n"

51def build_source(source_base: str | None, rel: str) -> str | None:
52 """Join the CLI-provided base URL with the file's relative path (always
53 using forward slashes, since this is a URL). Returns None if no base set."""
54 if source_base is None:
55 return None
56 return f"{source_base.rstrip('/')}/{rel.replace(os.sep, '/')}"

59def convert_file(src: str, out_path: str, source: str | None) -> None:
60 """Read XML at `src`, write the wrapped Markdown to `out_path`.
61
62 The fence title is always the original .xml filename and is never
63 overridden. Parent folders of `out_path` are created as needed.
64 """
65 with open(src, "r", encoding="utf-8") as f:
66 xml = f.read()
67
68 parent = os.path.dirname(out_path)
69 if parent:
70 os.makedirs(parent, exist_ok=True)
71
72 with open(out_path, "w", encoding="utf-8") as f:
73 f.write(to_markdown(xml, src, source))

76def convert_tree(src_dir: str, out_dir: str, source_base: str | None) -> int:
77 """Recursively convert every .xml under `src_dir` into `out_dir`, mirroring
78 the folder structure. Only folders that contain XML are created. Returns the
79 number of files converted. The source tree is left untouched."""
80 converted = 0
81 for root, _, files in os.walk(src_dir):
82 for name in files:
83 if not name.lower().endswith(".xml"):
84 continue
85 src = os.path.join(root, name)
86 rel = os.path.relpath(src, src_dir)
87 out_path = os.path.join(
88 out_dir, re.sub(r"\.xml$", ".md", rel, flags=re.IGNORECASE)
89 )
90 convert_file(src, out_path, build_source(source_base, rel))
91 print(f"{src} -> {out_path}")
92 converted += 1
93 return converted

96def parse_args(argv: list[str]):
97 """Pull out --source-base / -s and return (positionals, source_base)."""
98 positionals: list[str] = []
99 source_base: str | None = None
100 i = 0
101 while i < len(argv):
102 arg = argv[i]
103 if arg in ("--source-base", "-s"):
104 i += 1
105 if i >= len(argv):
106 print(f"{arg} needs a value", file=sys.stderr)
107 sys.exit(1)
108 source_base = argv[i]
109 elif arg.startswith("--source-base="):
110 source_base = arg.split("=", 1)[1]
111 elif arg.startswith("-"):
112 print(f"Unknown flag: {arg}", file=sys.stderr)
113 sys.exit(1)
114 else:
115 positionals.append(arg)
116 i += 1
117 return positionals, source_base

130def main() -> None:
131 args, source_base = parse_args(sys.argv[1:])
132
133 if not args or len(args) > 2:
134 usage()
135
136 src = args[0]
137
138 if os.path.isdir(src):
139 if len(args) != 2:
140 usage()
141 converted = convert_tree(src, args[1], source_base)
142 elif os.path.isfile(src):
143 if not src.lower().endswith(".xml"):
144 print(f"Not an .xml file: {src}", file=sys.stderr)
145 sys.exit(1)
146 source = build_source(source_base, src)
147 if len(args) == 2:
148 # Explicit destination given: keep the source, write .md there.
149 out_path = args[1]
150 if os.path.isdir(out_path) or out_path.endswith(os.sep):
151 out_path = os.path.join(
152 out_path,
153 re.sub(
154 r"\.xml$", ".md", os.path.basename(src), flags=re.IGNORECASE
155 ),
156 )
157 convert_file(src, out_path, source)
158 else:
159 # No destination: in-place rename, removing the original .xml.
160 out_path = re.sub(r"\.xml$", ".md", src, flags=re.IGNORECASE)
161 convert_file(src, out_path, source)
162 if out_path != src:
163 os.remove(src)
164 print(f"{src} -> {out_path}")
165 converted = 1
166 else:
167 print(f"Not a file or directory: {src}", file=sys.stderr)
168 sys.exit(1)
169
170 print(f"\nDone. Converted {converted} file{'' if converted == 1 else 's'}.")
