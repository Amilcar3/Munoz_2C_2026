115{
116 LV_CHECK_ARG(path != NULL, return NULL);
117
118 const lv_binfont_dsc_t dsc = {
119 .path = path,
120 };
121
122 return lv_binfont_create_ex(&dsc);

125lv_font_t * lv_binfont_create_ex(const lv_binfont_dsc_t * dsc)
126{
127 LV_CHECK_ARG(dsc != NULL, return NULL);
128#if LV_USE_FS_MEMFS
129 LV_CHECK_ARG(dsc->path != NULL || dsc->buffer != NULL, return NULL);
130#else
131 LV_CHECK_ARG(dsc->path != NULL, return NULL);
132#endif
133
134 const char * path = dsc->path;
135 lv_fs_path_ex_t mempath;
136 if(LV_USE_FS_MEMFS && path == NULL) {
137 lv_fs_make_path_from_buffer(&mempath, LV_FS_MEMFS_LETTER, (void *)dsc->buffer, dsc->buffer_size, "bin");
138 path = (const char *)&mempath;
139 }
140
141 lv_fs_file_t file;
142 lv_fs_res_t fs_res = lv_fs_open(&file, path, LV_FS_MODE_RD);
143 if(fs_res != LV_FS_RES_OK) return NULL;
144
145 lv_font_t * font = lv_zalloc(sizeof(*font));
146 LV_ASSERT_MALLOC(font);
147 if(!font) {
148 LV_LOG_ERROR("Failed to allocate memory for font");
149 lv_fs_close(&file);
150 return NULL;
151 }
152
153
154 bool loaded = load_font(&file, font, dsc->dynamic_glyph_load);
155
156 const lv_font_fmt_txt_dsc_t * font_dsc = font->dsc;
157 bool file_owned_by_loader = font_dsc != NULL && font_dsc->are_glyphs_dynamic_loaded;
158
159 if(!loaded) {
160 LV_LOG_WARN("Error loading font file: %s", path);
161 lv_binfont_destroy(font);
162 if(!file_owned_by_loader) lv_fs_close(&file);
163 return NULL;
164 }
165
166 if(!file_owned_by_loader) lv_fs_close(&file);
167
168 return font;
169}

185void lv_binfont_destroy(lv_font_t * font)
186{
187 if(font == NULL) return;
188
189 const lv_font_fmt_txt_dsc_t * dsc = font->dsc;
190 if(dsc == NULL) return;
191
192 if(dsc->kern_classes == 0) {
193 const lv_font_fmt_txt_kern_pair_t * kern_dsc = dsc->kern_dsc;
194 if(NULL != kern_dsc) {
195 lv_free((void *)kern_dsc->glyph_ids);
196 lv_free((void *)kern_dsc->values);
197 lv_free((void *)kern_dsc);
198 }
199 }
200 else {
201 const lv_font_fmt_txt_kern_classes_t * kern_dsc = dsc->kern_dsc;
202 if(NULL != kern_dsc) {
203 lv_free((void *)kern_dsc->class_pair_values);
204 lv_free((void *)kern_dsc->left_class_mapping);
205 lv_free((void *)kern_dsc->right_class_mapping);
206 lv_free((void *)kern_dsc);
207 }
208 }
209
210 const lv_font_fmt_txt_cmap_t * cmaps = dsc->cmaps;
211 if(NULL != cmaps) {
212 for(int i = 0; i < dsc->cmap_num; ++i) {
213 lv_free((void *)cmaps[i].glyph_id_ofs_list);
214 lv_free((void *)cmaps[i].unicode_list);
215 }
216 lv_free((void *)cmaps);
217 }
218
219 if(dsc->are_glyphs_dynamic_loaded) {
220 glyph_loader_delete((lv_binfont_glyph_loader_t *)dsc->glyph_bitmap);
221 }
222 else {
223 lv_free((void *)dsc->glyph_bitmap);
224 }
225
226 lv_free((void *)dsc->glyph_dsc);
227 lv_free((void *)dsc);
228 lv_free(font);
229}

235static bit_iterator_t init_bit_iterator(lv_fs_file_t * fp)
236{
237 LV_ASSERT(fp != NULL);
238 bit_iterator_t it;
239 it.fp = fp;
240 it.bit_pos = -1;
241 it.byte_value = 0;
242 return it;
243}

245static unsigned int read_bits(bit_iterator_t * it, int n_bits, lv_fs_res_t * res)
246{
247 LV_ASSERT(it != NULL);
248 LV_ASSERT(res != NULL);
249 unsigned int value = 0;
250 while(n_bits--) {
251 it->byte_value = it->byte_value << 1;
252 it->bit_pos--;
253
254 if(it->bit_pos < 0) {
255 it->bit_pos = 7;
256 *res = lv_fs_read(it->fp, &(it->byte_value), 1, NULL);
257 if(*res != LV_FS_RES_OK) {
258 return 0;
259 }
260 }
261 int8_t bit = (it->byte_value & 0x80) ? 1 : 0;
262
263 value |= (bit << n_bits);
264 }
265 *res = LV_FS_RES_OK;
266 return value;
267}

269static int read_bits_signed(bit_iterator_t * it, int n_bits, lv_fs_res_t * res)
270{
271 LV_ASSERT(it != NULL);
272 LV_ASSERT(res != NULL);
273 unsigned int value = read_bits(it, n_bits, res);
274 if(value & (1 << (n_bits - 1))) {
275 value |= ~0u << n_bits;
276 }
277 return value;
278}

280static int read_label(lv_fs_file_t * fp, int start, const char * label)
281{
282 LV_ASSERT(fp != NULL);
283 LV_ASSERT(label != NULL);
284 lv_fs_seek(fp, start, LV_FS_SEEK_SET);
285
286 uint32_t length;
287 char buf[4];
288
289 if(lv_fs_read(fp, &length, 4, NULL) != LV_FS_RES_OK
290 || lv_fs_read(fp, buf, 4, NULL) != LV_FS_RES_OK
291 || lv_memcmp(label, buf, 4) != 0) {
292 LV_LOG_WARN("Error reading '%s' label.", label);
293 return -1;
294 }
295
296 return length;
297}

299static bool load_cmaps_tables(lv_fs_file_t * fp, lv_font_fmt_txt_dsc_t * font_dsc,
300 uint32_t cmaps_start, cmap_table_bin_t * cmap_table)
301{
302 LV_ASSERT(fp != NULL);
303 LV_ASSERT(font_dsc != NULL);
304 LV_ASSERT(cmap_table != NULL);
305 if(lv_fs_read(fp, cmap_table, font_dsc->cmap_num * sizeof(cmap_table_bin_t), NULL) != LV_FS_RES_OK) {
306 return false;
307 }
308
309 for(unsigned int i = 0; i < font_dsc->cmap_num; ++i) {
310 lv_fs_res_t res = lv_fs_seek(fp, cmaps_start + cmap_table[i].data_offset, LV_FS_SEEK_SET);
311 if(res != LV_FS_RES_OK) {
312 return false;
313 }
314
315 lv_font_fmt_txt_cmap_t * cmap = (lv_font_fmt_txt_cmap_t *) & (font_dsc->cmaps[i]);
316
317 cmap->range_start = cmap_table[i].range_start;
318 cmap->range_length = cmap_table[i].range_length;
319 cmap->glyph_id_start = cmap_table[i].glyph_id_start;
320 cmap->type = cmap_table[i].format_type;
321
322 switch(cmap_table[i].format_type) {
323 case LV_FONT_FMT_TXT_CMAP_FORMAT0_FULL: {
324 uint32_t ids_size = (uint32_t)(sizeof(uint8_t) * cmap_table[i].data_entries_count);
325 uint8_t * glyph_id_ofs_list = lv_malloc(ids_size);
326 LV_ASSERT_MALLOC(glyph_id_ofs_list);
327 if(!glyph_id_ofs_list) {
328 return false;
329 }
330
331 cmap->glyph_id_ofs_list = glyph_id_ofs_list;
332
333 if(lv_fs_read(fp, glyph_id_ofs_list, ids_size, NULL) != LV_FS_RES_OK) {
334 return false;
335 }
336
337 cmap->list_length = cmap->range_length;
338 break;
339 }
340 case LV_FONT_FMT_TXT_CMAP_FORMAT0_TINY:
341 break;
342 case LV_FONT_FMT_TXT_CMAP_SPARSE_FULL:
343 case LV_FONT_FMT_TXT_CMAP_SPARSE_TINY: {
344 uint32_t list_size = sizeof(uint16_t) * cmap_table[i].data_entries_count;
345 uint16_t * unicode_list = (uint16_t *)lv_malloc(list_size);
346 LV_ASSERT_MALLOC(unicode_list);
347 if(!unicode_list) {
348 return false;
349 }
350
351 cmap->unicode_list = unicode_list;
352 cmap->list_length = cmap_table[i].data_entries_count;
353
354 if(lv_fs_read(fp, unicode_list, list_size, NULL) != LV_FS_RES_OK) {
355 return false;
356 }
357
358 if(cmap_table[i].format_type == LV_FONT_FMT_TXT_CMAP_SPARSE_FULL) {
359 uint16_t * buf = lv_malloc(sizeof(*buf) * cmap->list_length);
360 LV_ASSERT_MALLOC(buf);
361 if(!buf) {
362 return false;
363 }
364
365 cmap->glyph_id_ofs_list = buf;
366
367 if(lv_fs_read(fp, buf, sizeof(uint16_t) * cmap->list_length, NULL) != LV_FS_RES_OK) {
368 return false;
369 }
370 }
371 break;
372 }
373 default:
374 LV_LOG_WARN("Unknown cmaps format type %d.", cmap_table[i].format_type);
375 return false;
376 }
377 }
378 return true;
379}

381static int32_t load_cmaps(lv_fs_file_t * fp, lv_font_fmt_txt_dsc_t * font_dsc, uint32_t cmaps_start)
382{
383 LV_ASSERT(fp != NULL);
384 LV_ASSERT(font_dsc != NULL);
385 int32_t cmaps_length = read_label(fp, cmaps_start, "cmap");
386 if(cmaps_length < 0) {
387 return -1;
388 }
389
390
391 uint32_t cmaps_subtables_count;
392 if(lv_fs_read(fp, &cmaps_subtables_count, sizeof(uint32_t), NULL) != LV_FS_RES_OK) {
393 return -1;
394 }
395
396 lv_font_fmt_txt_cmap_t * cmaps = lv_zalloc(cmaps_subtables_count * sizeof(*cmaps));
397 LV_ASSERT_MALLOC(cmaps);
398 if(!cmaps) {
399 return -1;
400 }
401
402 cmap_table_bin_t * cmaps_tables = lv_malloc(cmaps_subtables_count * sizeof(*cmaps_tables));
403 LV_ASSERT_MALLOC(cmaps_tables);
404 if(!cmaps_tables) {
405 lv_free(cmaps);
406 return -1;
407 }
408
409 font_dsc->cmaps = cmaps;
410 font_dsc->cmap_num = cmaps_subtables_count;
411 bool success = load_cmaps_tables(fp, font_dsc, cmaps_start, cmaps_tables);
412
413 lv_free(cmaps_tables);
414
415 return success ? cmaps_length : -1;
416}

418static bool read_glyph_bitmap(lv_fs_file_t * fp, uint32_t pos, uint32_t nbits, uint8_t * buf, uint32_t buf_size)
419{
420 LV_ASSERT(fp != NULL);
421 LV_ASSERT(buf != NULL);
422 LV_ASSERT(buf_size > 0);
423
424
425 lv_fs_res_t res = lv_fs_seek(fp, pos, LV_FS_SEEK_SET);
426 if(res != LV_FS_RES_OK) return false;
427
428 bit_iterator_t bit_it = init_bit_iterator(fp);
429
430 /*Skip the glyph header*/
431 read_bits(&bit_it, nbits, &res);
432 if(res != LV_FS_RES_OK) return false;
433
434 if(nbits % 8 == 0) { /*Fast path: the bitmap is byte aligned in the file*/
435 return lv_fs_read(fp, buf, buf_size, NULL) == LV_FS_RES_OK;
436 }
437
438 for(uint32_t k = 0; k < buf_size - 1; ++k) {
439 buf[k] = read_bits(&bit_it, 8, &res);
440 if(res != LV_FS_RES_OK) return false;
441 }
442
443 buf[buf_size - 1] = read_bits(&bit_it, 8 - nbits % 8, &res);
444 if(res != LV_FS_RES_OK) return false;
445
446 /*The last fragment should be on the MSB but read_bits() will place it to the LSB*/
447 buf[buf_size - 1] = buf[buf_size - 1] << (nbits % 8);
448
449 return true;
450}

452static const uint8_t * load_glyph_bitmap_cb(lv_font_fmt_txt_glyph_loader_t * base, uint32_t gid)
453{
454 LV_ASSERT(base != NULL);
455 lv_binfont_glyph_loader_t * loader = (lv_binfont_glyph_loader_t *)base;
456
457 if(gid >= loader->loca_count) {
458 LV_LOG_WARN("Invalid glyph index: %" LV_PRIu32 " >= %" LV_PRIu32, gid, loader->loca_count);
459 return NULL;
460 }
461
462 uint32_t offset = loader->glyph_offset_table[gid];
463 uint32_t next_offset = gid < loader->loca_count - 1 ? loader->glyph_offset_table[gid + 1] : loader->glyph_length;
464 uint32_t header_size = loader->glyph_nbits / 8;
465
466 if(next_offset < offset + header_size) {
467 LV_LOG_WARN("Invalid glyph offsets for glyph %" LV_PRIu32, gid);
468 return NULL;
469 }
470
471 uint32_t bmp_size = next_offset - offset - header_size;
472 if(bmp_size == 0 || bmp_size > loader->bitmap_buf_size) {
473 LV_LOG_WARN("Invalid bitmap size for glyph %" LV_PRIu32, gid);
474 return NULL;
475 }
476
477 if(!read_glyph_bitmap(&loader->file, loader->glyph_start + offset, loader->glyph_nbits,
478 loader->bitmap_buf, bmp_size)) {
479 return NULL;
480 }
481
482 return loader->bitmap_buf;
483}

498static lv_binfont_glyph_loader_t * glyph_loader_create(lv_fs_file_t * fp,
499 uint32_t * glyph_offset_table,
500 uint32_t loca_count, uint32_t glyph_start, uint32_t glyph_length,
501 uint32_t glyph_nbits, uint32_t max_bmp_size)
502{
503 LV_ASSERT(fp != NULL);
504 LV_ASSERT(glyph_offset_table != NULL);
505
506 lv_binfont_glyph_loader_t * loader = lv_malloc_zeroed(sizeof(lv_binfont_glyph_loader_t));
507 LV_ASSERT_MALLOC(loader);
508 if(loader == NULL) {
509 LV_LOG_ERROR("Failed to allocate memory for dynamic glyph loader");
510 return NULL;
511 }
512
513 loader->bitmap_buf = lv_malloc(LV_MAX(max_bmp_size, 1));
514 LV_ASSERT_MALLOC(loader->bitmap_buf);
515 if(loader->bitmap_buf == NULL) {
516 LV_LOG_ERROR("Failed to allocate memory for scratch glyph buffer");
517 lv_free(loader);
518 return NULL;
519 }
520
521 if(lv_mutex_init(&loader->base.lock) != LV_RESULT_OK) {
522 LV_LOG_WARN("Couldn't create the mutex of the glyph loader");
523 lv_free(loader->bitmap_buf);
524 lv_free(loader);
525 return NULL;
526 }
527
528 loader->base.get_glyph_bitmap_cb = load_glyph_bitmap_cb;
529 loader->file = *fp;
530 loader->glyph_offset_table = glyph_offset_table;
531 loader->loca_count = loca_count;
532 loader->glyph_start = glyph_start;
533 loader->glyph_length = glyph_length;
534 loader->glyph_nbits = glyph_nbits;
535 loader->bitmap_buf_size = max_bmp_size;
536 return loader;
537}

546static int32_t load_glyph(lv_fs_file_t * fp, lv_font_fmt_txt_dsc_t * font_dsc,
547 uint32_t start, uint32_t * glyph_offset_table, uint32_t loca_count, font_header_bin_t * header,
548 bool dynamic_glyph_load, uint32_t * max_bmp_size)
549{
550 LV_ASSERT(fp != NULL);
551 LV_ASSERT(font_dsc != NULL);
552 LV_ASSERT(glyph_offset_table != NULL);
553 LV_ASSERT(header != NULL);
554 LV_ASSERT(max_bmp_size != NULL);
555
556 int32_t glyph_length = read_label(fp, start, "glyf");
557 if(glyph_length < 0) {
558 return -1;
559 }
560
561 lv_font_fmt_txt_glyph_dsc_t * glyph_dsc = lv_zalloc(loca_count * sizeof(*glyph_dsc));
562 LV_ASSERT_MALLOC(glyph_dsc);
563 if(!glyph_dsc) {
564 return -1;
565 }
566
567 font_dsc->glyph_dsc = glyph_dsc;
568
569 int32_t nbits = header->advance_width_bits + 2 * header->xy_bits + 2 * header->wh_bits;
570 int32_t cur_bmp_size = 0;
571 *max_bmp_size = 0;
572
573 for(unsigned int i = 0; i < loca_count; ++i) {
574 lv_font_fmt_txt_glyph_dsc_t * gdsc = &glyph_dsc[i];
575
576 lv_fs_res_t res = lv_fs_seek(fp, start + glyph_offset_table[i], LV_FS_SEEK_SET);
577 if(res != LV_FS_RES_OK) {
578 return -1;
579 }
580
581 bit_iterator_t bit_it = init_bit_iterator(fp);
582
583 if(header->advance_width_bits == 0) {
584 gdsc->adv_w = header->default_advance_width;
585 }
586 else {
587 gdsc->adv_w = read_bits(&bit_it, header->advance_width_bits, &res);
588 if(res != LV_FS_RES_OK) {
589 return -1;
590 }
591 }
592
593 if(header->advance_width_format == 0) {
594 gdsc->adv_w *= 16;
595 }
596
597 gdsc->ofs_x = read_bits_signed(&bit_it, header->xy_bits, &res);
598 if(res != LV_FS_RES_OK) {
599 return -1;
600 }
601
602 gdsc->ofs_y = read_bits_signed(&bit_it, header->xy_bits, &res);
603 if(res != LV_FS_RES_OK) {
604 return -1;
605 }
606
607 gdsc->box_w = read_bits(&bit_it, header->wh_bits, &res);
608 if(res != LV_FS_RES_OK) {
609 return -1;
610 }
611
612 gdsc->box_h = read_bits(&bit_it, header->wh_bits, &res);
613 if(res != LV_FS_RES_OK) {
614 return -1;
615 }
616
617 int next_offset = (i < loca_count - 1) ? glyph_offset_table[i + 1] : (uint32_t)glyph_length;
618 int bmp_size = next_offset - glyph_offset_table[i] - nbits / 8;
619
620 if(i == 0) {
621 gdsc->adv_w = 0;
622 gdsc->box_w = 0;
623 gdsc->box_h = 0;
624 gdsc->ofs_x = 0;
625 gdsc->ofs_y = 0;
626 }
627
628 gdsc->bitmap_index = cur_bmp_size;
629
630 if(gdsc->box_w * gdsc->box_h == 0) continue;
631
632 if(bmp_size <= 0) {
633 LV_LOG_WARN("Invalid bitmap size of glyph %u", i);
634 return -1;
635 }
636
637 if(dynamic_glyph_load) {
638 /* store the biggest bmp size so that we can allocate a buffer for it*/
639 *max_bmp_size = LV_MAX(*max_bmp_size, (uint32_t)bmp_size);
640 }
641 else {
642 cur_bmp_size += bmp_size;
643 }
644 }
645
646 if(dynamic_glyph_load) {
647 return glyph_length;
648 }
649
650 uint8_t * glyph_bmp = (uint8_t *)lv_malloc(LV_MAX(cur_bmp_size, 1));
651 LV_ASSERT_MALLOC(glyph_bmp);
652 if(!glyph_bmp) {
653 return -1;
654 }
655
656 font_dsc->glyph_bitmap = glyph_bmp;
657 font_dsc->are_glyphs_dynamic_loaded = false;
658
659 for(unsigned int i = 1; i < loca_count; ++i) {
660 if(glyph_dsc[i].box_w * glyph_dsc[i].box_h == 0) {
661 continue;
662 }
663
664 int next_offset = (i < loca_count - 1) ? glyph_offset_table[i + 1] : (uint32_t)glyph_length;
665 int bmp_size = next_offset - glyph_offset_table[i] - nbits / 8;
666
667 if(!read_glyph_bitmap(fp, start + glyph_offset_table[i], nbits,
668 &glyph_bmp[glyph_dsc[i].bitmap_index], bmp_size)) {
669 return -1;
670 }
671 }
672 return glyph_length;
673}

682/*
683 * Loads a `lv_font_t` from a binary file, given a `lv_fs_file_t`.
684 *
685 * Memory allocations on `lvgl_load_font` should be immediately zeroed and
686 * the pointer should be set on the `lv_font_t` data before any possible return.
687 *
688 * When something fails, it returns `false` and the memory on the `lv_font_t`
689 * still needs to be freed using `lv_binfont_destroy`.
690 *
691 * `lv_binfont_destroy` will assume that all non-null pointers are allocated and
692 * should be freed.
693 */
694static bool load_font(lv_fs_file_t * fp, lv_font_t * font, bool dynamic_glyph_load)
695{
696 LV_ASSERT(fp != NULL);
697 LV_ASSERT(font != NULL);
698 lv_font_fmt_txt_dsc_t * font_dsc = lv_zalloc(sizeof(*font_dsc));
699 LV_ASSERT_MALLOC(font_dsc);
700 if(!font_dsc) {
701 return false;
702 }
703
704 font->dsc = font_dsc;
705
706 /*header*/
707 int32_t header_length = read_label(fp, 0, "head");
708 if(header_length < 0) {
709 return false;
710 }
711
712 font_header_bin_t font_header;
713 if(lv_fs_read(fp, &font_header, sizeof(font_header_bin_t), NULL) != LV_FS_RES_OK) {
714 return false;
715 }
716
717 font->base_line = -font_header.descent;
718 font->line_height = font_header.ascent - font_header.descent;
719 font->get_glyph_dsc = lv_font_get_glyph_dsc_fmt_txt;
720 font->get_glyph_bitmap = lv_font_get_bitmap_fmt_txt;
721 font->release_glyph = release_glyph_cb;
722 font->subpx = font_header.subpixels_mode;
723 font->underline_position = (int8_t) font_header.underline_position;
724 font->underline_thickness = (int8_t) font_header.underline_thickness;
725
726 font_dsc->bpp = font_header.bits_per_pixel;
727 font_dsc->kern_scale = font_header.kerning_scale;
728 font_dsc->bitmap_format = font_header.compression_id;
729
730 /*cmaps*/
731 uint32_t cmaps_start = header_length;
732 int32_t cmaps_length = load_cmaps(fp, font_dsc, cmaps_start);
733 if(cmaps_length < 0) {
734 return false;
735 }
736
737 /*loca*/
738 uint32_t loca_start = cmaps_start + cmaps_length;
739 int32_t loca_length = read_label(fp, loca_start, "loca");
740 if(loca_length < 0) {
741 return false;
742 }
743
744 uint32_t loca_count;
745 if(lv_fs_read(fp, &loca_count, sizeof(uint32_t), NULL) != LV_FS_RES_OK) {
746 return false;
747 }
748
749 uint32_t * glyph_offset_table = lv_malloc(sizeof(uint32_t) * loca_count);
750 LV_ASSERT_MALLOC(glyph_offset_table);
751 if(!glyph_offset_table) {
752 LV_LOG_ERROR("Failed to load allocate glyph offset table");
753 return false;
754 }
755
756 switch(font_header.index_to_loc_format) {
757 case 0:
758 for(unsigned int i = 0; i < loca_count; ++i) {
759 uint16_t offset;
760 if(lv_fs_read(fp, &offset, sizeof(uint16_t), NULL) != LV_FS_RES_OK) {
761 lv_free(glyph_offset_table);
762 return false;
763 }
764 glyph_offset_table[i] = offset;
765 }
766 break;
767 case 1:
768 if(lv_fs_read(fp, glyph_offset_table, loca_count * sizeof(uint32_t), NULL) != LV_FS_RES_OK) {
769 lv_free(glyph_offset_table);
770 return false;
771 }
772 break;
773 default:
774 LV_LOG_WARN("Unknown index_to_loc_format: %d.", font_header.index_to_loc_format);
775 lv_free(glyph_offset_table);
776 return false;
777 }
778
779 /*glyph*/
780 uint32_t glyph_start = loca_start + loca_length;
781 uint32_t max_bmp_size = 0;
782 int32_t glyph_length = load_glyph(fp, font_dsc, glyph_start, glyph_offset_table, loca_count, &font_header,
783 dynamic_glyph_load, &max_bmp_size);
784
785 if(glyph_length < 0) {
786 lv_free(glyph_offset_table);
787 return false;
788 }
789
790 if(dynamic_glyph_load) {
791 uint32_t glyph_nbits = font_header.advance_width_bits + 2 * font_header.xy_bits + 2 * font_header.wh_bits;
792
793 lv_binfont_glyph_loader_t * loader = glyph_loader_create(fp,
794 glyph_offset_table,
795 loca_count,
796 glyph_start,
797 (uint32_t)glyph_length,
798 glyph_nbits,
799 max_bmp_size);
800
801 /*On success the loader takes over `fp` and `glyph_offset_table`*/
802 if(!loader) {
803 lv_free(glyph_offset_table);
804 return false;
805 }
806 font_dsc->glyph_bitmap = (const uint8_t *)loader;
807 font_dsc->are_glyphs_dynamic_loaded = true;
808 }
809 else {
810 lv_free(glyph_offset_table);
811 }
812
813 /*kerning*/
814 if(font_header.tables_count < 4) {
815 font_dsc->kern_dsc = NULL;
816 font_dsc->kern_classes = 0;
817 font_dsc->kern_scale = 0;
818 return true;
819 }
820
821 uint32_t kern_start = glyph_start + glyph_length;
822 int32_t kern_length = load_kern(fp, font_dsc, font_header.glyph_id_format, kern_start);
823
824 return kern_length >= 0;
825}

827int32_t load_kern(lv_fs_file_t * fp, lv_font_fmt_txt_dsc_t * font_dsc, uint8_t format, uint32_t start)
828{
829 LV_ASSERT(fp != NULL);
830 LV_ASSERT(font_dsc != NULL);
831 int32_t kern_length = read_label(fp, start, "kern");
832 if(kern_length < 0) {
833 return -1;
834 }
835
836 uint8_t kern_format_type;
837 int32_t padding;
838 if(lv_fs_read(fp, &kern_format_type, sizeof(uint8_t), NULL) != LV_FS_RES_OK ||
839 lv_fs_read(fp, &padding, 3 * sizeof(uint8_t), NULL) != LV_FS_RES_OK) {
840 return -1;
841 }
842
843 if(0 == kern_format_type) { /*sorted pairs*/
844 lv_font_fmt_txt_kern_pair_t * kern_pair = lv_zalloc(sizeof(*kern_pair));
845 LV_ASSERT_MALLOC(kern_pair);
846 if(!kern_pair) {
847 return -1;
848 }
849
850 font_dsc->kern_dsc = kern_pair;
851 font_dsc->kern_classes = 0;
852
853 uint32_t glyph_entries;
854 if(lv_fs_read(fp, &glyph_entries, sizeof(uint32_t), NULL) != LV_FS_RES_OK) {
855 return -1;
856 }
857
858 int ids_size;
859 if(format == 0) {
860 ids_size = sizeof(int8_t) * 2 * glyph_entries;
861 }
862 else {
863 ids_size = sizeof(int16_t) * 2 * glyph_entries;
864 }
865
866 uint8_t * glyph_ids = lv_malloc(ids_size);
867 LV_ASSERT_MALLOC(glyph_ids);
868 int8_t * values = lv_malloc(glyph_entries);
869 LV_ASSERT_MALLOC(values);
870 if(!glyph_ids || !values) {
871 lv_free(values);
872 lv_free(glyph_ids);
873 return -1;
874 }
875
876 kern_pair->glyph_ids_size = format;
877 kern_pair->pair_cnt = glyph_entries;
878 kern_pair->glyph_ids = glyph_ids;
879 kern_pair->values = values;
880
881 if(lv_fs_read(fp, glyph_ids, ids_size, NULL) != LV_FS_RES_OK) {
882 return -1;
883 }
884
885 if(lv_fs_read(fp, values, glyph_entries, NULL) != LV_FS_RES_OK) {
886 return -1;
887 }
888 }
889 else if(3 == kern_format_type) { /*array M*N of classes*/
890
891 lv_font_fmt_txt_kern_classes_t * kern_classes = lv_zalloc(sizeof(*kern_classes));
892 LV_ASSERT_MALLOC(kern_classes);
893 if(!kern_classes) {
894 return -1;
895 }
896
897 font_dsc->kern_dsc = kern_classes;
898 font_dsc->kern_classes = 1;
899
900 uint16_t kern_class_mapping_length;
901 uint8_t kern_table_rows;
902 uint8_t kern_table_cols;
903
904 if(lv_fs_read(fp, &kern_class_mapping_length, sizeof(uint16_t), NULL) != LV_FS_RES_OK ||
905 lv_fs_read(fp, &kern_table_rows, sizeof(uint8_t), NULL) != LV_FS_RES_OK ||
906 lv_fs_read(fp, &kern_table_cols, sizeof(uint8_t), NULL) != LV_FS_RES_OK) {
907 return -1;
908 }
909
910 int kern_values_length = sizeof(int8_t) * kern_table_rows * kern_table_cols;
911
912 uint8_t * kern_left = lv_malloc(kern_class_mapping_length);
913 LV_ASSERT_MALLOC(kern_left);
914 uint8_t * kern_right = lv_malloc(kern_class_mapping_length);
915 LV_ASSERT_MALLOC(kern_right);
916 int8_t * kern_values = lv_malloc(kern_values_length);
917 LV_ASSERT_MALLOC(kern_values);
918
919 if(!kern_left || !kern_right || !kern_values) {
920 lv_free(kern_left);
921 lv_free(kern_right);
922 lv_free(kern_values);
923 return -1;
924 }
925
926 kern_classes->left_class_mapping = kern_left;
927 kern_classes->right_class_mapping = kern_right;
928 kern_classes->left_class_cnt = kern_table_rows;
929 kern_classes->right_class_cnt = kern_table_cols;
930 kern_classes->class_pair_values = kern_values;
931
932 if(lv_fs_read(fp, kern_left, kern_class_mapping_length, NULL) != LV_FS_RES_OK ||
933 lv_fs_read(fp, kern_right, kern_class_mapping_length, NULL) != LV_FS_RES_OK ||
934 lv_fs_read(fp, kern_values, kern_values_length, NULL) != LV_FS_RES_OK) {
935 return -1;
936 }
937 }
938 else {
939 LV_LOG_WARN("Unknown kern_format_type: %d", kern_format_type);
940 return -1;
941 }
942
943 return kern_length;
944}
