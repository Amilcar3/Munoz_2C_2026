65static void epic_img_draw_core(lv_draw_task_t * task, const lv_draw_image_dsc_t * draw_dsc,
66 const lv_image_decoder_dsc_t * decoder_dsc, lv_draw_image_sup_t * sup,
67 const lv_area_t * img_coords, const lv_area_t * clipped_img_area)
68{
69 const lv_draw_buf_t * decoded = decoder_dsc->decoded;
70 const uint8_t * src_buf;
71 lv_color_format_t cf;
72 uint32_t img_total_width;
73 const bool is_ezip = (decoded != NULL) && ((decoded->header.flags & LV_IMAGE_FLAGS_EZIP) != 0U);
74#if defined(EPIC_INPUT_JPEG)
75 const bool is_jpeg = (decoded != NULL) && ((decoded->header.flags & LV_IMAGE_FLAGS_JPEG) != 0U);
76#else
77 const bool is_jpeg = false;
78#endif
79
80 if(decoded == NULL || decoded->data == NULL) {
81 return;
82 }
83
84 if(draw_dsc->clip_radius != 0 || draw_dsc->blend_mode != LV_BLEND_MODE_NORMAL) {
85 return;
86 }
87
88 lv_draw_buf_flush_cache(decoded, NULL);
89
90 src_buf = decoded->data;
91 cf = decoded->header.cf;
92
93 if(is_ezip || is_jpeg) {
94 img_total_width = decoded->header.w;
95 }
96 else {
97 img_total_width = lv_epic_stride_to_width(decoded->header.stride, cf);
98 }
99
100 EPIC_LayerConfigTypeDef input_layers[4];
101 EPIC_LayerConfigTypeDef output_layer;
102 uint8_t input_layer_cnt = 2;
103
104 if(lv_epic_setup_layers(&input_layers[0], &output_layer, task, clipped_img_area)) {
105 return;
106 }
107
108 HAL_EPIC_LayerConfigInit(&input_layers[1]);
109 input_layers[1].transform_cfg.angle = (draw_dsc->rotation + 3600) % 3600;
110 input_layers[1].transform_cfg.pivot_x = draw_dsc->pivot.x;
111 input_layers[1].transform_cfg.pivot_y = draw_dsc->pivot.y;
112 input_layers[1].transform_cfg.scale_x = LV_SCALE_NONE * EPIC_INPUT_SCALE_NONE / (uint32_t)draw_dsc->scale_x;
113 input_layers[1].transform_cfg.scale_y = LV_SCALE_NONE * EPIC_INPUT_SCALE_NONE / (uint32_t)draw_dsc->scale_y;
114 input_layers[1].alpha = draw_dsc->opa;
115 input_layers[1].x_offset = img_coords->x1;
116 input_layers[1].y_offset = img_coords->y1;
117
118 if(is_ezip) {
119 input_layers[1].color_mode = EPIC_INPUT_EZIP;
120 }
121#if defined(EPIC_INPUT_JPEG)
122 else if(is_jpeg) {
123 input_layers[1].color_mode = EPIC_INPUT_JPEG;
124 }
125#endif
126 else {
127 input_layers[1].color_mode = lv_img_cf_to_epic_cf(cf);
128 }
129
130 input_layers[1].data = (uint8_t *)src_buf;
131 input_layers[1].data_size = decoded->data_size;
132 input_layers[1].width = lv_area_get_width(img_coords);
133 input_layers[1].height = lv_area_get_height(img_coords);
134 input_layers[1].total_width = img_total_width;
135
136#ifdef EPIC_SUPPORT_A8
137 if(input_layers[1].color_mode == EPIC_INPUT_A8) {
138 input_layers[1].color_en = true;
139 input_layers[1].color_r = sup->alpha_color.red;
140 input_layers[1].color_g = sup->alpha_color.green;
141 input_layers[1].color_b = sup->alpha_color.blue;
142 input_layers[1].ax_mode = ALPHA_BLEND_RGBCOLOR;
143 }
144 else
145#endif
146#ifdef EPIC_SUPPORT_L8
147 if(input_layers[1].color_mode == EPIC_INPUT_L8) {
148 input_layers[1].lookup_table = (uint8_t *)decoder_dsc->palette;
149 input_layers[1].color_en = false;
150 }
151 else
152#endif
153 {
154 input_layers[1].color_en = false;
155 }
156
157#ifdef EPIC_SUPPORT_MASK
158 if(cf == LV_COLOR_FORMAT_RGB565A8) {
159 HAL_EPIC_LayerConfigInit(&input_layers[input_layer_cnt]);
160 input_layers[input_layer_cnt].alpha = draw_dsc->opa;
161 input_layers[input_layer_cnt].x_offset = img_coords->x1;
162 input_layers[input_layer_cnt].y_offset = img_coords->y1;
163 input_layers[input_layer_cnt].color_mode = EPIC_INPUT_A8;
164 input_layers[input_layer_cnt].data = ((uint8_t *)src_buf) + decoded->header.h * decoded->header.stride;
165 input_layers[input_layer_cnt].ax_mode = ALPHA_BLEND_OVERWRITE;
166 input_layers[input_layer_cnt].width = lv_area_get_width(img_coords);
167 input_layers[input_layer_cnt].height = lv_area_get_height(img_coords);
168 input_layers[input_layer_cnt].total_width = decoded->header.w;
169 input_layer_cnt++;
170 }
171#endif
172
173 lv_image_decoder_dsc_t mask_decoder_dsc;
174 bool mask_decoder_opened = false;
175
176 if(draw_dsc->bitmap_mask_src != NULL) {
177 lv_area_t mask_area;
178 const lv_image_dsc_t * mask_dsc = draw_dsc->bitmap_mask_src;
179 const lv_area_t * image_area = lv_area_get_width(&draw_dsc->image_area) < 0 ? img_coords : &draw_dsc->image_area;
180
181 if(lv_image_src_get_type(mask_dsc) == LV_IMAGE_SRC_VARIABLE &&
182 (mask_dsc->header.cf == LV_COLOR_FORMAT_A8 || mask_dsc->header.cf == LV_COLOR_FORMAT_L8)) {
183 const lv_image_header_t * header = &mask_dsc->header;
184
185 lv_area_set(&mask_area, 0, 0, header->w - 1, header->h - 1);
186 lv_area_align(image_area, &mask_area, LV_ALIGN_CENTER, 0, 0);
187
188 HAL_EPIC_LayerConfigInit(&input_layers[input_layer_cnt]);
189 input_layers[input_layer_cnt].alpha = LV_OPA_COVER;
190 input_layers[input_layer_cnt].x_offset = mask_area.x1;
191 input_layers[input_layer_cnt].y_offset = mask_area.y1;
192 input_layers[input_layer_cnt].color_mode = EPIC_INPUT_A8;
193 input_layers[input_layer_cnt].data = (uint8_t *)mask_dsc->data;
194 input_layers[input_layer_cnt].ax_mode = ALPHA_BLEND_OVERWRITE;
195 input_layers[input_layer_cnt].width = lv_area_get_width(&mask_area);
196 input_layers[input_layer_cnt].height = lv_area_get_height(&mask_area);
197 input_layers[input_layer_cnt].total_width = header->stride;
198 input_layer_cnt++;
199 }
200 else if(lv_image_decoder_open(&mask_decoder_dsc, draw_dsc->bitmap_mask_src, NULL) == LV_RESULT_OK &&
201 mask_decoder_dsc.decoded != NULL) {
202 const lv_draw_buf_t * mask_img = mask_decoder_dsc.decoded;
203 mask_decoder_opened = true;
204
205 if(mask_img->header.cf == LV_COLOR_FORMAT_A8 || mask_img->header.cf == LV_COLOR_FORMAT_L8) {
206 lv_area_set(&mask_area, 0, 0, mask_img->header.w - 1, mask_img->header.h - 1);
207 lv_area_align(image_area, &mask_area, LV_ALIGN_CENTER, 0, 0);
208
209 HAL_EPIC_LayerConfigInit(&input_layers[input_layer_cnt]);
210 input_layers[input_layer_cnt].alpha = LV_OPA_COVER;
211 input_layers[input_layer_cnt].x_offset = mask_area.x1;
212 input_layers[input_layer_cnt].y_offset = mask_area.y1;
213 input_layers[input_layer_cnt].color_mode = EPIC_INPUT_A8;
214 input_layers[input_layer_cnt].data = (uint8_t *)mask_img->data;
215 input_layers[input_layer_cnt].ax_mode = ALPHA_BLEND_OVERWRITE;
216 input_layers[input_layer_cnt].width = lv_area_get_width(&mask_area);
217 input_layers[input_layer_cnt].height = lv_area_get_height(&mask_area);
218 input_layers[input_layer_cnt].total_width = mask_img->header.stride;
219 input_layer_cnt++;
220 }
221 else {
222 }
223 }
224 else {
225 }
226 }
227
228 (void)lv_epic_blend(input_layers, input_layer_cnt, &output_layer);
229
230 if(mask_decoder_opened) {
231 lv_image_decoder_close(&mask_decoder_dsc);
232 }
233}
