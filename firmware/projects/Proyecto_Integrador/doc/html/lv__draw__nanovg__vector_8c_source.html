84static void draw_fill(lv_draw_nanovg_unit_t * u, const lv_vector_fill_dsc_t * fill_dsc, const lv_fpoint_t * offset,
85 enum NVGcompositeOperation comp_op)
86{
87 LV_PROFILER_DRAW_BEGIN;
88
89 const enum NVGwinding winding = lv_fill_to_nvg(fill_dsc->fill_rule);
90
91 lv_nanovg_transform(u->vg, &fill_dsc->matrix);
92
93 switch(fill_dsc->style) {
94 case LV_VECTOR_DRAW_STYLE_SOLID: {
95 lv_nanovg_fill(u->vg, winding, comp_op, lv_color32_to_nvg(fill_dsc->color, fill_dsc->opa));
96 }
97 break;
98 case LV_VECTOR_DRAW_STYLE_PATTERN: {
99 const lv_draw_image_dsc_t * img_dsc = &fill_dsc->img_dsc;
100 lv_image_header_t header;
101 int image_handle = lv_nanovg_image_cache_get_handle(u, img_dsc->src, 0, &header);
102 if(image_handle < 0) {
103 LV_PROFILER_DRAW_END;
104 return;
105 }
106
107 float offset_x = 0;
108 float offset_y = 0;
109
110 if(fill_dsc->fill_units == LV_VECTOR_FILL_UNITS_OBJECT_BOUNDING_BOX) {
111 offset_x = offset->x;
112 offset_y = offset->y;
113 }
114
115 NVGpaint paint = nvgImagePattern(u->vg, offset_x, offset_y, header.w, header.h, 0, image_handle,
116 img_dsc->opa / (float)LV_OPA_COVER);
117
118 nvgFillPaint(u->vg, paint);
119 nvgFill(u->vg);
120 }
121 break;
122 case LV_VECTOR_DRAW_STYLE_GRADIENT: {
123 lv_nanovg_draw_grad(u->vg, &fill_dsc->gradient, winding, comp_op);
124 }
125 break;
126 default:
127 LV_LOG_WARN("unsupported style: %d", fill_dsc->style);
128 break;
129 }
130
131 LV_PROFILER_DRAW_END;
132}

167static void task_draw_cb(void * ctx, const lv_vector_path_t * path, const lv_vector_path_ctx_t * dsc)
168{
169 LV_PROFILER_DRAW_BEGIN;
170 lv_draw_nanovg_unit_t * u = ctx;
171
172 /* clear area */
173 if(!path) {
174 NVGcolor c = lv_color32_to_nvg(dsc->fill_dsc.color, dsc->fill_dsc.opa);
175 nvgBeginPath(u->vg);
176 lv_nanovg_path_append_area(u->vg, &dsc->scissor_area);
177 lv_nanovg_fill(u->vg, NVG_CCW, NVG_COPY, c);
178 LV_PROFILER_DRAW_END;
179 return;
180 }
181
182 if(dsc->fill_dsc.opa == LV_OPA_TRANSP && dsc->stroke_dsc.opa == LV_OPA_TRANSP) {
183 LV_LOG_TRACE("Full transparent, no need to draw");
184 LV_PROFILER_DRAW_END;
185 return;
186 }
187
188 nvgSave(u->vg);
189 lv_nanovg_transform(u->vg, &dsc->matrix);
190
191 lv_fpoint_t offset = {0, 0};
192 lv_path_to_nvg(u->vg, path, &offset);
193
194 lv_nanovg_set_clip_area(u->vg, &dsc->scissor_area);
195
196 const enum NVGcompositeOperation comp_op = lv_blend_to_nvg(dsc->blend_mode);
197 nvgGlobalCompositeOperation(u->vg, comp_op);
198
199 if(dsc->fill_dsc.opa) {
200 draw_fill(u, &dsc->fill_dsc, &offset, comp_op);
201 }
202
203 if(dsc->stroke_dsc.opa) {
204 draw_stroke(u, &dsc->stroke_dsc);
205 }
206
207 nvgRestore(u->vg);
208
209 LV_PROFILER_DRAW_END;
210}

212static void lv_path_to_nvg(NVGcontext * ctx, const lv_vector_path_t * src, lv_fpoint_t * offset)
213{
214 LV_PROFILER_DRAW_BEGIN;
215
216 float min_x = FLT_MAX;
217 float min_y = FLT_MAX;
218 float max_x = -FLT_MAX;
219 float max_y = -FLT_MAX;
220
221#define CMP_BOUNDS(point) \
222 do { \
223 if((point)->x < min_x) min_x = (point)->x; \
224 if((point)->y < min_y) min_y = (point)->y; \
225 if((point)->x > max_x) max_x = (point)->x; \
226 if((point)->y > max_y) max_y = (point)->y; \
227 } while(0)
228
229 const lv_vector_path_op_t * ops = lv_array_front(&src->ops);
230 const lv_fpoint_t * point = lv_array_front(&src->points);
231 const uint32_t op_size = lv_array_size(&src->ops);
232
233 nvgBeginPath(ctx);
234
235 for(uint32_t i = 0; i < op_size; i++) {
236 switch(ops[i]) {
237 case LV_VECTOR_PATH_OP_MOVE_TO: {
238 nvgMoveTo(ctx, point->x, point->y);
239 CMP_BOUNDS(point);
240 point++;
241 }
242 break;
243 case LV_VECTOR_PATH_OP_LINE_TO: {
244 nvgLineTo(ctx, point->x, point->y);
245 CMP_BOUNDS(point);
246 point++;
247 }
248 break;
249 case LV_VECTOR_PATH_OP_QUAD_TO: {
250 nvgQuadTo(ctx, point[0].x, point[0].y, point[1].x, point[1].y);
251 CMP_BOUNDS(&point[0]);
252 CMP_BOUNDS(&point[1]);
253 point += 2;
254 }
255 break;
256 case LV_VECTOR_PATH_OP_CUBIC_TO: {
257 nvgBezierTo(ctx, point[0].x, point[0].y, point[1].x, point[1].y, point[2].x, point[2].y);
258 CMP_BOUNDS(&point[0]);
259 CMP_BOUNDS(&point[1]);
260 CMP_BOUNDS(&point[2]);
261 point += 3;
262 }
263 break;
264 case LV_VECTOR_PATH_OP_CLOSE: {
265 nvgClosePath(ctx);
266 }
267 break;
268 default:
269 LV_LOG_WARN("unknown op: %d", ops[i]);
270 break;
271 }
272 }
273
274 offset->x = lroundf(min_x);
275 offset->y = lroundf(min_y);
276 LV_PROFILER_DRAW_END;
277}

280{
281 switch(blend) {
