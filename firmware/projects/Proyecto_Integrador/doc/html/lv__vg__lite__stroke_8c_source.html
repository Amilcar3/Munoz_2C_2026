134lv_cache_entry_t * lv_vg_lite_stroke_get(struct _lv_draw_vg_lite_unit_t * unit,
135 struct _lv_vg_lite_path_t * path,
136 const lv_vector_stroke_dsc_t * dsc)
137{
138 LV_ASSERT_NULL(unit);
139 LV_ASSERT_NULL(path);
140 LV_ASSERT_NULL(dsc);
141
142 vg_lite_path_t * vg_path = lv_vg_lite_path_get_path(path);
143
144 if(vg_path->format != VG_LITE_FP32) {
145 LV_LOG_ERROR("only support VG_LITE_FP32 format");
146 return NULL;
147 }
148
149 /* prepare search key */
150 stroke_item_t search_key;
151 lv_memzero(&search_key, sizeof(search_key));
152 search_key.lv.cap = dsc->cap;
153 search_key.lv.join = dsc->join;
154 search_key.lv.width = dsc->width;
155 search_key.lv.miter_limit = dsc->miter_limit;
156
157 /* A one-time read-only array that only copies the pointer but not the content */
158 search_key.lv.dash_pattern = dsc->dash_pattern;
159 search_key.lv.path = path;
160
161 lv_cache_entry_t * cache_node_entry = lv_cache_acquire(unit->stroke_cache, &search_key, NULL);
162 if(cache_node_entry) {
163 return cache_node_entry;
164 }
165
166 cache_node_entry = lv_cache_acquire_or_create(unit->stroke_cache, &search_key, NULL);
167 if(cache_node_entry == NULL) {
168 LV_LOG_ERROR("stroke cache creating failed");
169 return NULL;
170 }
171
172 return cache_node_entry;
173}

202static bool stroke_create_cb(stroke_item_t * item, void * user_data)
203{
204 LV_UNUSED(user_data);
205 LV_ASSERT_NULL(item);
206
207 /* Check if stroke width is valid */
208 if(item->lv.width <= 0) {
209 LV_LOG_WARN("stroke width error: %f", item->lv.width);
210 return false;
211 }
212
213 /* Reset the dash pattern type */
214 item->dash_pattern_type = DASH_PATTERN_TYPE_LV;
215
216 /* dup the path */
217 item->vg.path = lv_vg_lite_path_create(VG_LITE_FP32);
218 lv_vg_lite_path_append_path(item->vg.path, item->lv.path);
219
220 /* dup the dash pattern */
221 vg_lite_float_t * vg_dash_pattern = NULL;
222 const uint32_t size = lv_array_size(&item->lv.dash_pattern);
223 if(size) {
224 /* Only support float dash pattern */
225 LV_ASSERT(item->lv.dash_pattern.element_size == sizeof(float));
226 lv_array_init(&item->vg.dash_pattern, size, sizeof(float));
227 lv_array_copy(&item->vg.dash_pattern, &item->lv.dash_pattern);
228
229 /* mark dash pattern has been duped */
230 item->dash_pattern_type = DASH_PATTERN_TYPE_VG;
231 vg_dash_pattern = lv_array_front(&item->vg.dash_pattern);
232 LV_ASSERT_NULL(vg_dash_pattern);
233 }
234
235 /* update parameters */
236 vg_lite_path_t * vg_path = lv_vg_lite_path_get_path(item->vg.path);
237 LV_VG_LITE_CHECK_ERROR(vg_lite_set_path_type(vg_path, VG_LITE_DRAW_STROKE_PATH), {});
238
239 vg_lite_error_t err = VG_LITE_SUCCESS;
240 LV_VG_LITE_CHECK_ERROR(
241 err = vg_lite_set_stroke(
242 vg_path,
243 lv_stroke_cap_to_vg(item->lv.cap),
244 lv_stroke_join_to_vg(item->lv.join),
245 item->lv.width,
246 item->lv.miter_limit,
247 vg_dash_pattern,
248 size,
249 item->lv.width / 2,
250 0),
251 /* Dump parameters */
252 {
253 lv_vg_lite_path_dump_info(vg_path);
254 LV_LOG_USER("Cap: 0x%X", (int)lv_stroke_cap_to_vg(item->lv.cap));
255 LV_LOG_USER("Join: 0x%X", (int)lv_stroke_join_to_vg(item->lv.join));
256 LV_LOG_USER("Width: %f", item->lv.width);
257 LV_LOG_USER("Miter limit: %d", item->lv.miter_limit);
258 if(vg_dash_pattern)
259 {
260 for(uint32_t i = 0; i < size; i++) {
261 LV_LOG_USER("Dash pattern[%" LV_PRIu32 "]: %f", i, vg_dash_pattern[i]);
262 }
263 }
264 });
265
266 if(err != VG_LITE_SUCCESS) {
267 stroke_free_cb(item, NULL);
268 return false;
269 }
270
271 const vg_lite_pointer * ori_path = vg_path->path;
272 const vg_lite_uint32_t ori_path_length = vg_path->path_length;
273
274 LV_PROFILER_DRAW_BEGIN_TAG("vg_lite_update_stroke");
275 LV_VG_LITE_CHECK_ERROR(err = vg_lite_update_stroke(vg_path), {
276 lv_vg_lite_path_dump_info(vg_path);
277 });
278 LV_PROFILER_DRAW_END_TAG("vg_lite_update_stroke");
279
280 /* check if path is changed */
281 LV_ASSERT_MSG(vg_path->path_length == ori_path_length, "vg_path->path_length should not change");
282 LV_ASSERT_MSG(vg_path->path == ori_path, "vg_path->path should not change");
283
284 if(err != VG_LITE_SUCCESS) {
285 stroke_free_cb(item, NULL);
286 return false;
287 }
288
289 return true;
290}

308static lv_cache_compare_res_t dash_pattern_compare(const stroke_item_t * lhs, const stroke_item_t * rhs)
309{
310 /* Select the dash pattern to compare */
311 const lv_array_t * lhs_dash_pattern = lhs->dash_pattern_type == DASH_PATTERN_TYPE_LV ?
312 &lhs->lv.dash_pattern :
313 &lhs->vg.dash_pattern;
314 const lv_array_t * rhs_dash_pattern = rhs->dash_pattern_type == DASH_PATTERN_TYPE_LV ?
315 &rhs->lv.dash_pattern :
316 &rhs->vg.dash_pattern;
317
318 const uint32_t lhs_dash_pattern_size = lv_array_size(lhs_dash_pattern);
319 const uint32_t rhs_dash_pattern_size = lv_array_size(rhs_dash_pattern);
320
321 if(lhs_dash_pattern_size != rhs_dash_pattern_size) {
322 return lhs_dash_pattern_size > rhs_dash_pattern_size ? 1 : -1;
323 }
324
325 if(lhs_dash_pattern_size == 0 && rhs_dash_pattern_size == 0) {
326 return 0;
327 }
328
329 /* Both dash pattern has the same size, compare them */
330 LV_ASSERT(lhs_dash_pattern->element_size == sizeof(float));
331 LV_ASSERT(rhs_dash_pattern->element_size == sizeof(float));
332
333 /* compare dash pattern data */
334 int cmp_res = lv_memcmp(
335 lv_array_front(lhs_dash_pattern),
336 lv_array_front(rhs_dash_pattern),
337 lhs_dash_pattern_size * sizeof(float));
338
339 if(cmp_res != 0) {
340 return cmp_res > 0 ? 1 : -1;
341 }
342
343 return 0;
344}

346static lv_cache_compare_res_t path_compare(const stroke_item_t * lhs, const stroke_item_t * rhs)
347{
348 /* Give priority to using dup vg.path */
349 const vg_lite_path_t * lhs_path = lhs->vg.path ?
350 lv_vg_lite_path_get_path(lhs->vg.path) :
351 lv_vg_lite_path_get_path(lhs->lv.path);
352 const vg_lite_path_t * rhs_path = rhs->vg.path ?
353 lv_vg_lite_path_get_path(rhs->vg.path) :
354 lv_vg_lite_path_get_path(rhs->lv.path);
355
356 LV_ASSERT(lhs_path->format == VG_LITE_FP32);
357 LV_ASSERT(rhs_path->format == VG_LITE_FP32);
358
359 if(lhs_path->path_length != rhs_path->path_length) {
360 return lhs_path->path_length > rhs_path->path_length ? 1 : -1;
361 }
362
363 int cmp_res = lv_memcmp(lhs_path->path, rhs_path->path, lhs_path->path_length);
364 if(cmp_res != 0) {
365 return cmp_res > 0 ? 1 : -1;
366 }
367
368 return 0;
369}

371static lv_cache_compare_res_t stroke_compare_cb(const stroke_item_t * lhs, const stroke_item_t * rhs)
372{
373 if(lhs->lv.width != rhs->lv.width) {
374 return lhs->lv.width > rhs->lv.width ? 1 : -1;
375 }
376
377 if(lhs->lv.cap != rhs->lv.cap) {
378 return lhs->lv.cap > rhs->lv.cap ? 1 : -1;
379 }
380
381 if(lhs->lv.join != rhs->lv.join) {
382 return lhs->lv.join > rhs->lv.join ? 1 : -1;
383 }
384
385 if(lhs->lv.miter_limit != rhs->lv.miter_limit) {
386 return lhs->lv.miter_limit > rhs->lv.miter_limit ? 1 : -1;
387 }
388
389 lv_cache_compare_res_t dash_pattern_res = dash_pattern_compare(lhs, rhs);
390 if(dash_pattern_res != 0) {
391 return dash_pattern_res;
392 }
393
394 return path_compare(lhs, rhs);
395}
