1#!/usr/bin/env python3
2"""
3Reject C++ style (`//`) comments in C source and headers.
4
5LVGL is a C99 library; its `.c` and `.h` files must use block comments
6(`/* ... */`) only. C++ files (`.cpp`/`.hpp`) are allowed to use `//`.
7
8The scan uses a small character-level state machine so that `//` sequences
9inside string literals (`"http://..."`) and inside block comments
10(`/* see https://... */`) are NOT reported -- only genuine `//` comments are.
11
12Usage:
13 check_comments.py [path ...]
14
15 With no arguments the default first-party roots are scanned
16 (src, demos, examples, tests). Any path argument (file or directory)
17 restricts the scan to those paths, e.g.:
18
19 check_comments.py src/core
20 check_comments.py src/misc/lv_math.c
21
22Exit codes:
23 0 - no `//` comments found
24 1 - one or more `//` comments found
25"""
26

36# Only C sources/headers are checked. C++ translation units may use `//`.
37CHECKED_EXTENSIONS = {".c", ".h"}
38
39# Default roots scanned when no path argument is given (mirrors code-format.py).
40DEFAULT_ROOTS = ("src", "demos", "examples", "tests")

42# Third-party and generated files legitimately contain `//` comments and are
43# excluded from formatting in scripts/code-format.cfg. Keep this in sync.
44# A path is excluded if any of these fragments appears in its POSIX path.
45EXCLUDED_FRAGMENTS = (
46 # generated
47 "src/lv_conf_internal.h",
48 "src/core/lv_obj_style_gen.c",

69 posix = path.as_posix()

77def find_line_comments(text: str) -> list[int]:
78 """
79 Return the 1-based line numbers that start a `//` comment.
80
81 A character-level state machine ignores `//` that appears inside string
82 literals, character literals, and block comments.
83 """
84 CODE, STRING, CHAR, BLOCK, LINE = range(5)
85 state = CODE
86 line = 1
87 i = 0
88 n = len(text)
89 hits: list[int] = []
90
91 while i < n:
92 c = text[i]
93 nxt = text[i + 1] if i + 1 < n else ""
94
95 if c == "\n":
96 line += 1
97 # A `//` comment runs to end of line.
98 if state == LINE:
99 state = CODE
100 i += 1
101 continue
102
103 if state == CODE:
104 if c == "/" and nxt == "/":
105 hits.append(line)
106 state = LINE
107 i += 2
108 continue
109 if c == "/" and nxt == "*":
110 state = BLOCK
111 i += 2
112 continue
113 if c == '"':
114 state = STRING
115 i += 1
116 continue
117 if c == "'":
118 state = CHAR
119 i += 1
120 continue
121 i += 1
122 continue
123
124 if state == STRING:
125 if c == "\\":
126 i += 2 # skip escaped char (e.g. \" )
127 continue
128 if c == '"':
129 state = CODE
130 i += 1
131 continue
132
133 if state == CHAR:
134 if c == "\\":
135 i += 2 # skip escaped char (e.g. \' )
136 continue
137 if c == "'":
138 state = CODE
139 i += 1
140 continue
141
142 if state == BLOCK:
143 if c == "*" and nxt == "/":
144 state = CODE
145 i += 2
146 continue
147 i += 1
148 continue
149
150 # state == LINE: consume until newline (handled above)
151 i += 1
152
153 return hits

160def collect_files(roots: list[Path]) -> list[Path]:
161 files: list[Path] = []
162 for root in roots:
163 if root.is_file():
164 if root.suffix in CHECKED_EXTENSIONS and not is_excluded(root):
165 files.append(root)
166 continue
167 for ext in CHECKED_EXTENSIONS:
168 for path in root.rglob(f"*{ext}"):
169 if not is_excluded(path):
170 files.append(path)
171 return sorted(set(files))

217 if violations:
218 total = sum(len(lines) for _, lines in violations)
219 print(
220 f"Found {total} C++ style (//) comment(s) in "
221 f"{len(violations)} file(s). Use block comments (/* ... */):\n"
222 )
223 for path, lines in violations:
224 try:
225 shown = path.relative_to(repo_root)
226 except ValueError:
227 shown = path
228 for ln in lines:
229 print(f" {shown}:{ln}")
230 sys.exit(1)

232 print(f"OK: no // comments in {len(files)} C source/header file(s)")
