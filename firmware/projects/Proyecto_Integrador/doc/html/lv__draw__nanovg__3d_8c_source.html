44 LV_PROFILER_DRAW_BEGIN;

46 lv_draw_nanovg_unit_t * u = (lv_draw_nanovg_unit_t *)t->draw_unit;
47
48 /* End NanoVG frame temporarily to allow direct OpenGL rendering */
49 lv_nanovg_end_frame(u);
50
51 lv_layer_t * layer = t->target_layer;
52
53 /* Get target layer info */
54 int32_t layer_w = lv_area_get_width(&layer->buf_area);
55 int32_t layer_h = lv_area_get_height(&layer->buf_area);
56
57 /* Calculate destination area relative to layer */
58 lv_area_t dest_area = *coords;
59 lv_area_move(&dest_area, -layer->buf_area.x1, -layer->buf_area.y1);
60
61 /* Calculate clip area relative to layer */
62 lv_area_t clip_area = t->clip_area;
63 lv_area_move(&clip_area, -layer->buf_area.x1, -layer->buf_area.y1);
64
65 /* Reinitialize OpenGL ES driver state after NanoVG modified it */
66 lv_opengles_reinit_state();
67
68 /* Use LVGL's OpenGL ES rendering infrastructure */
69 lv_opengles_viewport(0, 0, layer_w, layer_h);
70
71 lv_opengles_render_params_t params;
72 lv_opengles_render_params_init(&params);
73 params.texture = dsc->tex_id;
74 params.texture_area = &dest_area;
75 params.opa = dsc->opa;
76 params.disp_w = layer_w;
77 params.disp_h = layer_h;
78 params.texture_clip_area = &clip_area;
79 params.h_flip = dsc->h_flip;
80 params.v_flip = dsc->v_flip;
81 params.rb_swap = true;
82#if LV_DRAW_TRANSFORM_USE_MATRIX

84#endif
