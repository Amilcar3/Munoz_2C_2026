46lv_display_t * lv_wayland_window_create(uint32_t hor_res, uint32_t ver_res, char * title,
47 lv_wayland_display_close_cb_t close_cb)
48{
49 LV_CHECK_ARG(title != NULL, return NULL);
50
51 lv_result_t res = lv_wayland_init();
52 if(res != LV_RESULT_OK) {
53 return NULL;
54 }
55
56 if(close_cb) {
57 LV_LOG_DEPRECATED("'lv_wayland_display_close_cb_t' is deprecated and will be removed in the next release. "
58 "Bind an LV_EVENT_DELETE to the display returned by `lv_wayland_window_create` instead.");
59 }
60
61 lv_wl_window_t * window = lv_ll_ins_tail(&lv_wl_ctx.window_ll);
62 LV_ASSERT_MALLOC(window);
63 if(!window) {
64 LV_LOG_ERROR("Failed to allocate memory for window");
65 goto alloc_window_err;
66 }
67
68 lv_memset(window, 0, sizeof(*window));
69
70 window->close_cb = close_cb;
71
72 window->lv_disp = lv_display_create(hor_res, ver_res);
73 if(!window->lv_disp) {
74 LV_LOG_ERROR("failed to create lvgl display");
75 goto create_display_error;
76 }
77
78 window->body = wl_compositor_create_surface(lv_wl_ctx.wl_compositor);
79 if(!window->body) {
80 LV_LOG_ERROR("Failed to create window body");
81 goto create_surface_err;
82 }
83
84 if(lv_wl_xdg_create_window(lv_wl_ctx.xdg_wm, window, title) != LV_RESULT_OK) {
85 LV_LOG_ERROR("Failed to create window");
86 goto create_window_err;
87 }
88
89 lv_display_set_driver_data(window->lv_disp, window);
90
91 /* Initialize display driver */
92 res = lv_wayland_backend_init_display(&window->backend_ddata,
93 window->lv_disp, hor_res,
94 ver_res);
95 if(res != LV_RESULT_OK) {
96 LV_LOG_ERROR("Failed to create display");
97 goto init_display_err;
98 }
99
100 /*Assert here so that we can freely use these operations afterwards*/
101 LV_ASSERT(window->backend_ddata.ops != NULL);
102
103 lv_wayland_xdg_configure_surface(window);
104
105 lv_display_add_event_cb(window->lv_disp, res_changed_event, LV_EVENT_COLOR_FORMAT_CHANGED, NULL);
106 lv_display_add_event_cb(window->lv_disp, res_changed_event, LV_EVENT_RESOLUTION_CHANGED, NULL);
107 lv_display_add_event_cb(window->lv_disp, refr_start_event, LV_EVENT_REFR_START, NULL);
108 lv_display_add_event_cb(window->lv_disp, refr_end_event, LV_EVENT_REFR_READY, NULL);
109 lv_display_add_event_cb(window->lv_disp, delete_event, LV_EVENT_DELETE, NULL);
110
111 /* Register input */
112 window->lv_indev_pointer = lv_wayland_pointer_create();
113 lv_indev_set_display(window->lv_indev_pointer, window->lv_disp);
114
115 if(!window->lv_indev_pointer) {
116 LV_LOG_ERROR("failed to register pointer indev");
117 }
118
119 window->lv_indev_pointeraxis = lv_wayland_pointer_axis_create();
120 lv_indev_set_display(window->lv_indev_pointeraxis, window->lv_disp);
121
122 if(!window->lv_indev_pointeraxis) {
123 LV_LOG_ERROR("failed to register pointeraxis indev");
124 }
125
126 window->lv_indev_touch = lv_wayland_touch_create();
127 lv_indev_set_display(window->lv_indev_touch, window->lv_disp);
128
129 if(!window->lv_indev_touch) {
130 LV_LOG_ERROR("failed to register touch indev");
131 }
132
133 window->lv_indev_keyboard = lv_wayland_keyboard_create();
134 lv_indev_set_display(window->lv_indev_keyboard, window->lv_disp);
135
136 if(!window->lv_indev_keyboard) {
137 LV_LOG_ERROR("failed to register keyboard indev");
138 }
139 return window->lv_disp;
140init_display_err:
141 lv_wayland_xdg_delete_window(&window->xdg);
142create_window_err:
143 wl_surface_destroy(window->body);
144create_surface_err:
145 lv_display_delete(window->lv_disp);
146create_display_error:
147 lv_ll_remove(&lv_wl_ctx.window_ll, window);
148 lv_free(window);
149alloc_window_err:
150 return NULL;
151}

222void lv_wayland_window_set_physical_display(lv_display_t * display, uint8_t phys_display)
223{
224 LV_CHECK_ARG(display != NULL, return);
225 lv_wl_window_t * window = lv_display_get_driver_data(display);
226 LV_CHECK_ARG_MSG(window != NULL, return, "Invalid display");
227 LV_CHECK_ARG_FORMAT_MSG(phys_display == LV_WAYLAND_PHYSICAL_DISPLAY_ANY ||
228 phys_display < lv_wl_ctx.wl_output_count, return,
229 "Invalid display number '%d'. Expected '0'..'%d' or LV_WAYLAND_PHYSICAL_DISPLAY_ANY",
230 phys_display, lv_wl_ctx.wl_output_count - 1);
231
232 if(phys_display == LV_WAYLAND_PHYSICAL_DISPLAY_ANY) {
233 window->physical_output = NULL;
234 }
235 else {
236 window->physical_output = lv_wl_ctx.physical_outputs[phys_display]->wl_output;
237 }
238}

296static void delete_event(lv_event_t * e)
297{
298 LV_ASSERT(e != NULL);
299 lv_display_t * display = lv_event_get_target(e);
300 LV_ASSERT(display != NULL);
301 lv_wl_window_t * window = lv_display_get_driver_data(display);
302
303 if(window == NULL) {
304 return;
305 }
306
307 if(window->close_cb) {
308 window->close_cb(window->lv_disp);
309 }
310 lv_wayland_xdg_delete_window(&window->xdg);
311
312 /* Commit a NULL buffer to the body surface so that we release buffers*/
313 wl_surface_attach(window->body, NULL, 0, 0);
314 wl_surface_commit(window->body);
315 wl_display_roundtrip(lv_wl_ctx.wl_display);
316
317 wl_surface_destroy(window->body);
318 window->body = NULL;
319
320 /* Make sure buffer is correctly released*/
321 wl_display_roundtrip(lv_wl_ctx.wl_display);
322 lv_wayland_backend_deinit_display(&window->backend_ddata, window->lv_disp);
323
324 lv_indev_t * indevs[] = { window->lv_indev_keyboard,
325 window->lv_indev_pointer,
326 window->lv_indev_pointeraxis,
327 window->lv_indev_touch
328 };
329
330 for(size_t i = 0; i < sizeof(indevs) / sizeof(indevs[0]); ++i) {
331 lv_indev_delete(indevs[i]);
332 }
333
334 lv_display_set_driver_data(window->lv_disp, NULL);
335 lv_ll_remove(&lv_wl_ctx.window_ll, window);
336 lv_free(window);
337
338 if(LV_WAYLAND_DIRECT_EXIT && lv_ll_is_empty(&lv_wl_ctx.window_ll)) {
339 /* lv_deinit will deinit the wayland driver*/
340 lv_deinit();
341 exit(0);
342 }
343}

365static void res_changed_event(lv_event_t * e)
366{
367 LV_ASSERT(e != NULL);
368 lv_display_t * display = (lv_display_t *) lv_event_get_target(e);
369 LV_ASSERT(display != NULL);
370 lv_wl_window_t * window = lv_display_get_driver_data(display);
371 LV_ASSERT(window != NULL);
372
373 void * display_data = window->backend_ddata.ops->resize_display(window->backend_ddata.backend_data, display);
374 if(!display_data) {
375 /* The backend kept the display data of the previous resolution alive */
376 LV_LOG_ERROR("Failed to resize the display, keeping the previous configuration");
377 return;
378 }
379
380 window->backend_ddata.display_data = display_data;
381}
