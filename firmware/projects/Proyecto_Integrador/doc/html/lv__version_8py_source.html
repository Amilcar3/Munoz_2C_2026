20def version_from_sources():
21 """The version read from the lv_version.h the target was built against.
22
23 Neither way of asking the binary is dependable. lv_version_major() is
24 `static inline`, so it is dropped unless something calls it, and the macros
25 need -g3 *and* a stop inside a translation unit that included lv_version.h -
26 lv_timer.c, where a breakpoint on lv_timer_handler lands, does not.
27
28 DWARF does record the path of every source file, so the LVGL tree can be
29 found through any of its symbols and the version header read from disk.
30 Returns (major, minor, patch, info) or None.
31 """
32 for symbol in ("lv_obj_create", "lv_timer_handler", "lv_init"):
33 try:
34 found = gdb.lookup_global_symbol(symbol)
35 if found is None or found.symtab is None:
36 continue
37 # DWARF records whatever separator the compiler used, which is not
38 # always the host's: a MinGW build writes forward slashes.
39 path = found.symtab.fullname().replace("\\", "/")
40 marker = "/src/"
41 if marker not in path:
42 continue
43 root = path[:path.rindex(marker)]
44 # Since 9.6 the public headers live in include/lvgl/ and the
45 # top-level lv_version.h is a deprecated wrapper that only includes
46 # the real one, so the defines are not in it. Read that one first
47 # and keep the top-level file for the releases that still have it.
48 for header in (os.path.join(root, "include", "lvgl", "lv_version.h"),
49 os.path.join(root, "lv_version.h")):
50 if not os.path.exists(header):
51 continue
52 with open(header) as f:
53 macros = dict(_VERSION_MACRO.findall(f.read()))
54 if not {"MAJOR", "MINOR", "PATCH"} <= set(macros):
55 continue
56 return (int(macros["MAJOR"]), int(macros["MINOR"]),
57 int(macros["PATCH"]), macros.get("INFO", '""').strip('"'))
58 except (gdb.error, OSError, ValueError):
59 continue
60 return None

63def lvgl_version():
64 """The LVGL version of the target, as (major, minor, patch, info).
65
66 lv_version_major() and friends in lvgl.h are the intended way to ask, with
67 the version macros as a second try. Both depend on how the target was
68 built, so lv_version.h itself is the third and most dependable one.
69 Fields that cannot be read come back as None.
70 """
71 parts = []
72 for name in ("major", "minor", "patch"):
73 value = _call(f"lv_version_{name}()")
74 if value is None:
75 value = _call(f"LVGL_VERSION_{name.upper()}")
76 parts.append(int(value) if value is not None else None)
77
78 info = _call("lv_version_info()") or _call("LVGL_VERSION_INFO")
79 try:
80 info = info.string() if info is not None else None
81 except gdb.error:
82 info = None
83
84 # `info` counts as missing too: a build that hides it loses the "-dev" that
85 # tells a development version from a release.
86 if any(p is None for p in parts) or info is None:
87 from_sources = version_from_sources()
88 if from_sources is not None:
89 return from_sources
90
91 return (*parts, info)
