54void lv_draw_arc(lv_layer_t * layer, const lv_draw_arc_dsc_t * dsc)
55{
56 LV_CHECK_ARG(layer != NULL, return);
57 LV_CHECK_ARG(dsc != NULL, return);
58
59 if(dsc->opa <= LV_OPA_MIN) return;
60 if(dsc->width == 0) return;
61 if(dsc->start_angle == dsc->end_angle) return;
62
63 LV_PROFILER_DRAW_BEGIN;
64 lv_area_t a;
65 a.x1 = dsc->center.x - dsc->radius;
66 a.y1 = dsc->center.y - dsc->radius;
67 a.x2 = dsc->center.x + dsc->radius - 1;
68 a.y2 = dsc->center.y + dsc->radius - 1;
69
70 if(dsc->base.drop_shadow_opa) {
71 lv_layer_t * ds_layer = lv_draw_layer_create_drop_shadow(layer, &dsc->base, &a);
72 LV_ASSERT_NULL(ds_layer);
73 lv_draw_arc_dsc_t ds_dsc = *dsc;
74 ds_dsc.base.drop_shadow_opa = 0; /*Disable drop shadow so rendering below will render plain arc*/
75 lv_draw_arc(ds_layer, &ds_dsc);
76 lv_draw_layer_finish_drop_shadow(ds_layer, &dsc->base);
77 }
78
79 lv_draw_task_t * t = lv_draw_add_task(layer, &a, LV_DRAW_TASK_TYPE_ARC);
80
81 lv_memcpy(t->draw_dsc, dsc, sizeof(*dsc));
82
83 lv_draw_finalize_task_creation(layer, t);
84
85 LV_PROFILER_DRAW_END;
86}

88void lv_draw_arc_get_area(int32_t x, int32_t y, uint16_t radius, lv_value_precise_t start_angle,
89 lv_value_precise_t end_angle,
90 int32_t w, bool rounded, lv_area_t * area)
91{
92 LV_CHECK_ARG(area != NULL, return);
93
94 int32_t rout = radius;
95 int32_t start_angle_int = (int32_t) start_angle;
96 int32_t end_angle_int = (int32_t) end_angle;
97
98 /*Guard: radius too small for stroke width — return outer-radius bounding box
99 *to avoid negative rin causing inverted trig calculations.
100 *The arc never draws outside its outer radius regardless of stroke width. */
101 if(radius <= w) {
102 area->x1 = x - rout;
103 area->y1 = y - rout;
104 area->x2 = x + rout;
105 area->y2 = y + rout;
106 return;
107 }
108
109 /*Special case: full arc invalidation */
110 if(end_angle_int == start_angle_int + 360) {
111 area->x1 = x - rout;
112 area->y1 = y - rout;
113 area->x2 = x + rout;
114 area->y2 = y + rout;
115 return;
116 }
117
118 if(start_angle_int > 360) start_angle_int -= 360;
119 if(end_angle_int > 360) end_angle_int -= 360;
120
121 int32_t rin = radius - w;
122 int32_t extra_area = rounded ? w / 2 + 1 : 0;
123 uint8_t start_quarter = start_angle_int / 90;
124 uint8_t end_quarter = end_angle_int / 90;
125
126 /*360 deg still counts as quarter 3 (360 / 90 would be 4)*/
127 if(start_quarter == 4) start_quarter = 3;
128 if(end_quarter == 4) end_quarter = 3;
129
130 if(start_quarter == end_quarter && start_angle_int <= end_angle_int) {
131 if(start_quarter == 0) {
132 area->y1 = y + ((lv_trigo_sin(start_angle_int) * rin) >> LV_TRIGO_SHIFT) - extra_area;
133 area->x2 = x + ((lv_trigo_sin(start_angle_int + 90) * rout) >> LV_TRIGO_SHIFT) + extra_area;
134
135 area->y2 = y + ((lv_trigo_sin(end_angle_int) * rout) >> LV_TRIGO_SHIFT) + extra_area;
136 area->x1 = x + ((lv_trigo_sin(end_angle_int + 90) * rin) >> LV_TRIGO_SHIFT) - extra_area;
137 }
138 else if(start_quarter == 1) {
139 area->y2 = y + ((lv_trigo_sin(start_angle_int) * rout) >> LV_TRIGO_SHIFT) + extra_area;
140 area->x2 = x + ((lv_trigo_sin(start_angle_int + 90) * rin) >> LV_TRIGO_SHIFT) + extra_area;
141
142 area->y1 = y + ((lv_trigo_sin(end_angle_int) * rin) >> LV_TRIGO_SHIFT) - extra_area;
143 area->x1 = x + ((lv_trigo_sin(end_angle_int + 90) * rout) >> LV_TRIGO_SHIFT) - extra_area;
144 }
145 else if(start_quarter == 2) {
146 area->x1 = x + ((lv_trigo_sin(start_angle_int + 90) * rout) >> LV_TRIGO_SHIFT) - extra_area;
147 area->y2 = y + ((lv_trigo_sin(start_angle_int) * rin) >> LV_TRIGO_SHIFT) + extra_area;
148
149 area->y1 = y + ((lv_trigo_sin(end_angle_int) * rout) >> LV_TRIGO_SHIFT) - extra_area;
150 area->x2 = x + ((lv_trigo_sin(end_angle_int + 90) * rin) >> LV_TRIGO_SHIFT) + extra_area;
151 }
152 else if(start_quarter == 3) {
153 area->x1 = x + ((lv_trigo_sin(start_angle_int + 90) * rin) >> LV_TRIGO_SHIFT) - extra_area;
154 area->y1 = y + ((lv_trigo_sin(start_angle_int) * rout) >> LV_TRIGO_SHIFT) - extra_area;
155
156 area->x2 = x + ((lv_trigo_sin(end_angle_int + 90) * rout) >> LV_TRIGO_SHIFT) + extra_area;
157 area->y2 = y + ((lv_trigo_sin(end_angle_int) * rin) >> LV_TRIGO_SHIFT) + extra_area;
158 }
159 }
160 else if(start_quarter == 0 && end_quarter == 1) {
161 area->x1 = x + ((lv_trigo_sin(end_angle_int + 90) * rout) >> LV_TRIGO_SHIFT) - extra_area;
162 area->y1 = y + ((LV_MIN(lv_trigo_sin(end_angle_int),
163 lv_trigo_sin(start_angle_int)) * rin) >> LV_TRIGO_SHIFT) - extra_area;
164 area->x2 = x + ((lv_trigo_sin(start_angle_int + 90) * rout) >> LV_TRIGO_SHIFT) + extra_area;
165 area->y2 = y + rout + extra_area;
166 }
167 else if(start_quarter == 1 && end_quarter == 2) {
168 area->x1 = x - rout - extra_area;
169 area->y1 = y + ((lv_trigo_sin(end_angle_int) * rout) >> LV_TRIGO_SHIFT) - extra_area;
170 area->x2 = x + ((LV_MAX(lv_trigo_sin(start_angle_int + 90),
171 lv_trigo_sin(end_angle_int + 90)) * rin) >> LV_TRIGO_SHIFT) + extra_area;
172 area->y2 = y + ((lv_trigo_sin(start_angle_int) * rout) >> LV_TRIGO_SHIFT) + extra_area;
173 }
174 else if(start_quarter == 2 && end_quarter == 3) {
175 area->x1 = x + ((lv_trigo_sin(start_angle_int + 90) * rout) >> LV_TRIGO_SHIFT) - extra_area;
176 area->y1 = y - rout - extra_area;
177 area->x2 = x + ((lv_trigo_sin(end_angle_int + 90) * rout) >> LV_TRIGO_SHIFT) + extra_area;
178 area->y2 = y + (LV_MAX(lv_trigo_sin(end_angle_int) * rin,
179 lv_trigo_sin(start_angle_int) * rin) >> LV_TRIGO_SHIFT) + extra_area;
180 }
181 else if(start_quarter == 3 && end_quarter == 0) {
182 area->x1 = x + ((LV_MIN(lv_trigo_sin(end_angle_int + 90),
183 lv_trigo_sin(start_angle_int + 90)) * rin) >> LV_TRIGO_SHIFT) - extra_area;
184 area->y1 = y + ((lv_trigo_sin(start_angle_int) * rout) >> LV_TRIGO_SHIFT) - extra_area;
185 area->x2 = x + rout + extra_area;
186 area->y2 = y + ((lv_trigo_sin(end_angle_int) * rout) >> LV_TRIGO_SHIFT) + extra_area;
187
188 }
189 else {
190 area->x1 = x - rout;
191 area->y1 = y - rout;
192 area->x2 = x + rout;
193 area->y2 = y + rout;
194 }
195}
