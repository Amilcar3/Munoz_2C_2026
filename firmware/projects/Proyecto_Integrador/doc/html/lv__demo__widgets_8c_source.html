134static void color_changer_create(lv_obj_t * parent)
135{
136 static lv_palette_t palette[] = {
137 LV_PALETTE_BLUE, LV_PALETTE_GREEN, LV_PALETTE_BLUE_GREY, LV_PALETTE_ORANGE,
138 LV_PALETTE_RED, LV_PALETTE_PURPLE, LV_PALETTE_TEAL, LV_PALETTE_LAST
139 };
140
141 lv_obj_t * color_cont = lv_obj_create(parent);
142 lv_obj_remove_style_all(color_cont);
143 lv_obj_set_flex_flow(color_cont, LV_FLEX_FLOW_ROW);
144 lv_obj_set_flex_align(color_cont, LV_FLEX_ALIGN_SPACE_EVENLY, LV_FLEX_ALIGN_CENTER, LV_FLEX_ALIGN_CENTER);
145 lv_obj_set_floating(color_cont, true);
146
147 lv_obj_set_style_bg_color(color_cont, lv_color_white(), 0);
148 lv_obj_set_style_pad_right(color_cont, disp_size == DISP_SMALL ? LV_DPX(47) : LV_DPX(55), 0);
149 lv_obj_set_style_bg_opa(color_cont, LV_OPA_COVER, 0);
150 lv_obj_set_style_radius(color_cont, LV_RADIUS_CIRCLE, 0);
151
152 if(disp_size == DISP_SMALL) lv_obj_set_size(color_cont, LV_DPX(52), LV_DPX(52));
153 else lv_obj_set_size(color_cont, LV_DPX(60), LV_DPX(60));
154
155 lv_obj_align(color_cont, LV_ALIGN_BOTTOM_RIGHT, - LV_DPX(10), - LV_DPX(10));
156
157 uint32_t i;
158 for(i = 0; palette[i] != LV_PALETTE_LAST; i++) {
159 lv_obj_t * c = lv_button_create(color_cont);
160 lv_obj_set_style_bg_color(c, lv_palette_main(palette[i]), 0);
161 lv_obj_set_style_radius(c, LV_RADIUS_CIRCLE, 0);
162 lv_obj_set_style_opa(c, LV_OPA_TRANSP, 0);
163 lv_obj_set_size(c, 20, 20);
164 lv_obj_add_event_cb(c, color_event_cb, LV_EVENT_ALL, &palette[i]);
165 lv_obj_set_scroll_on_focus(c, false);
166 }
167
168 lv_obj_t * btn = lv_button_create(parent);
169 lv_obj_set_floating(btn, true);
170 lv_obj_set_clickable(btn, true);
171 lv_obj_set_style_bg_color(btn, lv_color_white(), LV_STATE_CHECKED);
172 lv_obj_set_style_pad_all(btn, 10, 0);
173 lv_obj_set_style_radius(btn, LV_RADIUS_CIRCLE, 0);
174 lv_obj_add_event_cb(btn, color_changer_event_cb, LV_EVENT_ALL, color_cont);
175 lv_obj_set_style_shadow_width(btn, 0, 0);
176 lv_obj_set_style_bg_image_src(btn, LV_SYMBOL_TINT, 0);
177
178 if(disp_size == DISP_SMALL) {
179 lv_obj_set_size(btn, LV_DPX(42), LV_DPX(42));
180 lv_obj_align(btn, LV_ALIGN_BOTTOM_RIGHT, -LV_DPX(15), -LV_DPX(15));
181 }
182 else {
183 lv_obj_set_size(btn, LV_DPX(50), LV_DPX(50));
184 lv_obj_align(btn, LV_ALIGN_BOTTOM_RIGHT, -LV_DPX(15), -LV_DPX(15));
185 }
186}

188static void color_changer_anim_cb(void * var, int32_t v)
189{
190 lv_obj_t * obj = var;
191 int32_t max_w = lv_obj_get_width(lv_obj_get_parent(obj)) - LV_DPX(20);
192 int32_t w;
193
194 if(disp_size == DISP_SMALL) {
195 w = lv_map(v, 0, 256, LV_DPX(52), max_w);
196 lv_obj_set_width(obj, w);
197 lv_obj_align(obj, LV_ALIGN_BOTTOM_RIGHT, - LV_DPX(10), - LV_DPX(10));
198 }
199 else {
200 w = lv_map(v, 0, 256, LV_DPX(60), max_w);
201 lv_obj_set_width(obj, w);
202 lv_obj_align(obj, LV_ALIGN_BOTTOM_RIGHT, - LV_DPX(10), - LV_DPX(10));
203 }
204
205 if(v > LV_OPA_COVER) v = LV_OPA_COVER;
206
207 uint32_t i;
208 for(i = 0; i < lv_obj_get_child_count(obj); i++) {
209 lv_obj_set_style_opa(lv_obj_get_child(obj, i), v, 0);
210 }
211

238static void color_event_cb(lv_event_t * e)
239{
240 lv_event_code_t code = lv_event_get_code(e);
241 lv_obj_t * obj = lv_event_get_target(e);
242
243 if(code == LV_EVENT_FOCUSED) {
244 lv_obj_t * color_cont = lv_obj_get_parent(obj);
245 if(lv_obj_get_width(color_cont) < LV_HOR_RES / 2) {
246 lv_anim_t a;
247 lv_anim_init(&a);
248 lv_anim_set_var(&a, color_cont);
249 lv_anim_set_exec_cb(&a, color_changer_anim_cb);
250 lv_anim_set_values(&a, 0, 256);
251 lv_anim_set_duration(&a, 200);
252 lv_anim_start(&a);
253 }
254 }
255 else if(code == LV_EVENT_CLICKED) {
256 extern lv_obj_t * chart1;
257 extern lv_obj_t * chart2;
258
259 extern lv_chart_series_t * ser1;
260 extern lv_chart_series_t * ser3;
261
262
263 lv_palette_t * palette_primary = lv_event_get_user_data(e);
264 lv_palette_t palette_secondary = (*palette_primary) + 3; /*Use another palette as secondary*/
265 if(palette_secondary >= LV_PALETTE_LAST) palette_secondary = 0;
266#if LV_USE_THEME_DEFAULT
267 lv_theme_default_init(lv_obj_get_display(obj), lv_palette_main(*palette_primary), lv_palette_main(palette_secondary),
268 LV_THEME_DEFAULT_DARK, font_normal);
269#endif
270 lv_color_t color = lv_palette_main(*palette_primary);
271 lv_style_set_text_color(&style_icon, color);
272 lv_chart_set_series_color(chart1, ser1, color);
273 lv_chart_set_series_color(chart2, ser3, color);
274 }
275}

282static void slideshow_anim_completed_cb(lv_anim_t * a_old)
283{
284 LV_UNUSED(a_old);
285
286 lv_obj_t * cont = lv_tabview_get_content(tv);
287 uint32_t tab_id = lv_tabview_get_tab_active(tv);
288 tab_id += 1;
289 if(tab_id > 2) tab_id = 0;
290 lv_tabview_set_active(tv, tab_id, LV_ANIM_ON);
291
292 lv_obj_t * tab = lv_obj_get_child(cont, tab_id);
293 lv_obj_scroll_to_y(tab, 0, LV_ANIM_OFF);
294 lv_obj_update_layout(tv);
295
296 int32_t v = lv_obj_get_scroll_bottom(tab);
297 int32_t dpi = lv_display_get_dpi(lv_display_get_default());
298 uint32_t t = lv_anim_speed(dpi);
299
300 lv_anim_t a;
301 lv_anim_init(&a);
302 lv_anim_set_exec_cb(&a, scroll_anim_y_cb);
303 lv_anim_set_duration(&a, t);
304 lv_anim_set_reverse_duration(&a, t);
305 lv_anim_set_values(&a, 0, v);
306 lv_anim_set_var(&a, tab);
307 lv_anim_set_completed_cb(&a, slideshow_anim_completed_cb);
308 lv_anim_start(&a);
309}
