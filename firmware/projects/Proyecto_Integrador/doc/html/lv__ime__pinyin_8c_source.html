427void lv_ime_pinyin_set_keyboard(lv_obj_t * obj, lv_obj_t * kb)
428{
429 if(kb) {
430 LV_CHECK_OBJ(kb, &lv_keyboard_class, return);
431 }
432
433 LV_CHECK_OBJ(obj, MY_CLASS, return);
434 lv_ime_pinyin_t * pinyin_ime = (lv_ime_pinyin_t *)obj;
435
436 pinyin_ime->kb = kb;
437 lv_obj_null_on_delete(&pinyin_ime->kb);
438 lv_obj_set_parent(obj, lv_obj_get_parent(kb));
439 lv_obj_set_parent(pinyin_ime->cand_panel, lv_obj_get_parent(kb));
440 lv_obj_add_event_cb(pinyin_ime->kb, lv_ime_pinyin_kb_event, LV_EVENT_VALUE_CHANGED, obj);
441 lv_obj_align_to(pinyin_ime->cand_panel, pinyin_ime->kb, LV_ALIGN_OUT_TOP_MID, 0, 0);
442}

817static void pinyin_page_proc(lv_obj_t * obj, uint16_t dir)
818{
819 LV_ASSERT(obj != NULL);
820 lv_ime_pinyin_t * pinyin_ime = (lv_ime_pinyin_t *)obj;
821 uint16_t page_num = pinyin_ime->cand_num / LV_IME_PINYIN_CAND_TEXT_NUM;
822 uint16_t remainder = pinyin_ime->cand_num % LV_IME_PINYIN_CAND_TEXT_NUM;
823
824 if(!pinyin_ime->cand_str) return;
825
826 if(dir == 0) {
827 if(pinyin_ime->py_page) {
828 pinyin_ime->py_page--;
829 }
830 }
831 else {
832 if(remainder == 0) {
833 page_num -= 1;
834 }
835 if(pinyin_ime->py_page < page_num) {
836 pinyin_ime->py_page++;
837 }
838 else return;
839 }
840
841 for(uint8_t i = 0; i < LV_IME_PINYIN_CAND_TEXT_NUM; i++) {
842 lv_memset(lv_pinyin_cand_str[i], 0x00, sizeof(lv_pinyin_cand_str[i]));
843 lv_pinyin_cand_str[i][0] = ' ';
844 }
845
846 // fill buf
847 uint16_t offset = pinyin_ime->py_page * (3 * LV_IME_PINYIN_CAND_TEXT_NUM);
848 for(uint8_t i = 0; (i < pinyin_ime->cand_num && i < LV_IME_PINYIN_CAND_TEXT_NUM); i++) {
849 if((remainder > 0) && (pinyin_ime->py_page == page_num)) {
850 if(i >= remainder)
851 break;
852 }
853 for(uint8_t j = 0; j < 3; j++) {
854 lv_pinyin_cand_str[i][j] = pinyin_ime->cand_str[offset + (i * 3) + j];
855 }
856 }

873static void init_pinyin_dict(lv_obj_t * obj, const lv_pinyin_dict_t * dict)
874{
875 LV_ASSERT(obj != NULL);
876 LV_ASSERT(dict != NULL);
877 lv_ime_pinyin_t * pinyin_ime = (lv_ime_pinyin_t *)obj;
878
879 char headletter = 'a';
880 uint16_t offset_sum = 0;
881 uint16_t offset_count = 0;
882 uint16_t letter_calc = 0;
883
884 pinyin_ime->dict = dict;
885
886 for(uint16_t i = 0; ; i++) {
887 if((NULL == (dict[i].py)) || (NULL == (dict[i].py_mb))) {
888 headletter = dict[i - 1].py[0];
889 letter_calc = headletter - 'a';
890 pinyin_ime->py_num[letter_calc] = offset_count;
891 break;
892 }
893
894 if(headletter == (dict[i].py[0])) {
895 offset_count++;
896 }
897 else {
898 headletter = dict[i].py[0];
899 pinyin_ime->py_num[letter_calc] = offset_count;
900 letter_calc = headletter - 'a';
901 offset_sum += offset_count;
902 pinyin_ime->py_pos[letter_calc] = offset_sum;
903
904 offset_count = 1;
905 }
906 }

909static char * pinyin_search_matching(lv_obj_t * obj, char * py_str, uint16_t * cand_num)
910{
911 LV_ASSERT(obj != NULL);
912 LV_ASSERT(py_str != NULL);
913 LV_ASSERT(cand_num != NULL);
914 lv_ime_pinyin_t * pinyin_ime = (lv_ime_pinyin_t *)obj;
915
916 const lv_pinyin_dict_t * cpHZ;
917 uint8_t index, len = 0, offset;
918 volatile uint8_t count = 0;
919
920 if(*py_str == '\0') return NULL;
921 if(*py_str == 'i') return NULL;
922 if(*py_str == 'u') return NULL;
923 if(*py_str == 'v') return NULL;
924 if(*py_str == ' ') return NULL;
925
926 offset = py_str[0] - 'a';
927 len = lv_strlen(py_str);
928
929 cpHZ = &pinyin_ime->dict[pinyin_ime->py_pos[offset]];
930 count = pinyin_ime->py_num[offset];
931
932 while(count--) {
933 for(index = 0; index < len; index++) {
934 if(*(py_str + index) != *((cpHZ->py) + index)) {
935 break;
936 }
937 }
938
939 // perfect match
940 if(len == 1 || index == len) {
941 // The Chinese character in UTF-8 encoding format is 3 bytes
942 * cand_num = lv_strlen((const char *)(cpHZ->py_mb)) / 3;
943 return (char *)(cpHZ->py_mb);
944 }
945 cpHZ++;
946 }
947 return NULL;
948}
