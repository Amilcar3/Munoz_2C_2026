130void test_refr_clip_corner_splits_the_children(void)
131{
132 refr_disp_create(100, 100, LV_COLOR_FORMAT_XRGB8888, LV_DISPLAY_RENDER_MODE_DIRECT, 1, 100);
133 refr_screen_set_color(REFR_COLOR_BLACK);
134 lv_obj_t * parent = refr_rect_create(refr_screen(), 10, 10, 60, 60, REFR_COLOR_BLUE);
135 lv_obj_set_style_radius(parent, 20, 0);
136 lv_obj_set_style_clip_corner(parent, true, 0);
137
138 lv_obj_t * child = refr_rect_create(parent, 0, 0, 60, 60, REFR_COLOR_RED);
139 uint32_t child_draw_cnt = 0;
140 lv_obj_add_event_cb(child, refr_draw_main_counter_cb, LV_EVENT_DRAW_MAIN, &child_draw_cnt);
141
142 lv_obj_invalidate(refr_screen());
143 refr_frame();
144
145 /*The children are rendered three times: the top, the middle and the bottom part*/
146 TEST_ASSERT_EQUAL_UINT32(3, child_draw_cnt);
147 /*The corner of the child is cut off, the middle is not*/
148 ASSERT_PX_EQ(REFR_COLOR_BLACK, refr_screen_px(11, 11));
149 ASSERT_PX_EQ(REFR_COLOR_RED, refr_screen_px(40, 40));
150 ASSERT_PX_EQ(REFR_COLOR_RED, refr_screen_px(40, 12));
151
152 /*Without a radius there is nothing to clip*/
153 child_draw_cnt = 0;
154 lv_obj_set_style_radius(parent, 0, 0);
155 lv_obj_invalidate(refr_screen());
156 refr_frame();
157 TEST_ASSERT_EQUAL_UINT32(1, child_draw_cnt);
158 ASSERT_PX_EQ(REFR_COLOR_RED, refr_screen_px(11, 11));
159}
