105static inline void lv_global_init(lv_global_t * global)
106{
107 LV_ASSERT_NULL(global);
108
109 if(global == NULL) {
110 LV_LOG_ERROR("lv_global cannot be null");
111 return;
112 }
113
114 lv_memzero(global, sizeof(lv_global_t));
115
116 lv_ll_init(&(global->disp_ll), sizeof(lv_display_t));
117 lv_ll_init(&(global->indev_ll), sizeof(lv_indev_t));
118
119 global->memory_zero = ZERO_MEM_SENTINEL;
120 global->style_refresh = true;
121 global->layout_count = LV_LAYOUT_LAST;
122 global->style_last_custom_prop_id = (uint32_t)LV_STYLE_LAST_BUILT_IN_PROP;
123 global->event_last_register_id = LV_EVENT_LAST;
124 lv_rand_set_seed(0x1234ABCD);
125
126#ifdef LV_LOG_PRINT_CB
127 void LV_LOG_PRINT_CB(lv_log_level_t, const char * txt);
128 global->custom_log_print_cb = LV_LOG_PRINT_CB;
129#endif
130
131#if defined(LV_DRAW_SW_SHADOW_CACHE_SIZE) && LV_DRAW_SW_SHADOW_CACHE_SIZE > 0
132 global->sw_shadow_cache.cache_size = -1;
133 global->sw_shadow_cache.cache_r = -1;
134#endif
135}
