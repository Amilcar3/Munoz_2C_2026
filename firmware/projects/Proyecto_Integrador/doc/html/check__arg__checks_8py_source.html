48SKIPPED_HEADER_DIRS = (

76def _validation_patterns(param: str) -> list[re.Pattern]:
77 p = re.escape(param)
78 macros = "|".join(CHECK_MACROS)
79 cond_macros = "|".join(CONDITION_CHECK_MACROS)
80 pats = [
81 # LV_CHECK_ARG(p != NULL, ...) / LV_CHECK_OBJ(p, cls, ...)
82 re.compile(r"\b(?:" + macros + r")\s*\‍(\s*&?\s*" + p + r"\b"),
83 # a compound check that mentions the parameter, e.g.
84 # LV_CHECK_ARG(len == 0 || dst != NULL, ...)
85 re.compile(
86 r"\b(?:" + cond_macros + r")\s*\‍([^;]*?\b" + p + r"\s*(?:==|!=)\s*NULL"
87 ),
88 re.compile(
89 r"\b(?:" + cond_macros + r")\s*\‍([^;]*?NULL\s*(?:==|!=)\s*" + p + r"\b"
90 ),
91 ]
92 return pats

95def is_validated(body: str, param: str) -> bool:
96 """Whether `param` is checked in `body`. Pass `body` through `strip_noncode`."""
97 patterns = _validation_patterns(param)
98 return any(pat.search(body) for pat in patterns)

101def _checkable_params(func: Function) -> list:
102 """Pointer parameters of `func` that are required to be validated."""
103 out = []
104 for _, param in enumerate(func.params):
105 if not param.is_pointer_like or not param.name:
106 continue
107 if NULLABLE_MARKER in (param.doc or ""):
108 continue
109 out.append(param)
110 return out

118def collect(repo_root: Path, xml_dir: Path | None = None) -> dict:
119 api = PublicApi.parse(repo_root, xml_dir=xml_dir)
120
121 missing = []
122 no_impl = 0
123 checked = 0
124 considered = 0
125
126 for func in sorted(api.public_functions(), key=lambda f: f.name):
127 if func.header and any(func.header.startswith(d) for d in SKIPPED_HEADER_DIRS):
128 continue
129 if any(func.name == x for x in SKIPPED_FUNCTIONS):
130 continue
131 if not func.has_body:
132 if any(p.is_pointer_like for p in func.params):
133 no_impl += 1
134 continue
135
136 body = func.body(repo_root)
137 for param in _checkable_params(func):
138 considered += 1
139
140 if is_validated(body, param.name):
141 checked += 1
142 else:
143 missing.append(
144 {
145 "function": func.name,
146 "param": param.name,
147 "type": param.type,
148 "header": func.header,
149 "header_line": func.header_line,
150 "impl": func.body_file,
151 "impl_line": func.body_start,
152 "null_mentioned_in_docs": "NULL" in param.doc,
153 }
154 )
155
156 return {
157 "total_public_functions": len(api.public_functions()),
158 "missing_but_documented_nullable_in_prose": sum(
159 1 for m in missing if m["null_mentioned_in_docs"]
160 ),
161 "pointer_params_considered": considered,
162 "pointer_params_validated": checked,
163 "pointer_params_missing": len(missing),
164 "functions_without_implementation": no_impl,
165 "missing": missing,
166 }

169def check_audit(repo_root: Path, xml_dir: Path | None = None) -> bool:
170 report = collect(repo_root, xml_dir)
171 missing = report["missing"]
172
173 by_file: dict[str, list[dict]] = {}
174 for item in missing:
175 by_file.setdefault(item["impl"], []).append(item)
176
177 for path in sorted(by_file):
178 print(f"\n{C.BOLD}{path}{C.RESET}")
179 for item in sorted(by_file[path], key=lambda i: (i["function"], i["param"])):
180 print(
181 f" {C.YELLOW}{item['function']}{C.RESET}"
182 f"({C.CYAN}{item['param']}{C.RESET})\n"
183 f" {C.DIM}{item['header']}:{item['header_line']}{C.RESET}\n"
184 f" {C.DIM}{item['impl']}:{item['impl_line']}{C.RESET}"
185 )
186
187 print()
188 print(f"{C.BOLD}Public API argument checks{C.RESET}")
189 print(f" public functions : {report['total_public_functions']}")
190 print(f" pointer parameters : {report['pointer_params_considered']}")
191 print(
192 f" validated : {C.GREEN}{report['pointer_params_validated']}{C.RESET}"
193 )
194 print(
195 f" {C.BOLD}unvalidated{C.RESET} : "
196 f"{C.RED if missing else C.GREEN}{report['pointer_params_missing']}{C.RESET}"
197 )
198 if report["missing_but_documented_nullable_in_prose"]:
199 print(
200 f" {C.DIM}of those, whose docs already mention NULL: "
201 f"{report['missing_but_documented_nullable_in_prose']}"
202 f" -- mark these `{NULLABLE_MARKER}`{C.RESET}"
203 )
204 if report["functions_without_implementation"]:
205 print(
206 f" {C.DIM}no implementation in tree : "
207 f"{report['functions_without_implementation']} (not counted){C.RESET}"
208 )
209 return not missing
