36lv_color_t LV_ATTRIBUTE_FAST_MEM lv_color_mix(lv_color_t c1, lv_color_t c2, uint8_t mix)
37{
38 lv_color_t ret;
39
40 ret.red = LV_UDIV255((uint16_t)c1.red * mix + c2.red * (255 - mix) + LV_COLOR_MIX_ROUND_OFS);
41 ret.green = LV_UDIV255((uint16_t)c1.green * mix + c2.green * (255 - mix) + LV_COLOR_MIX_ROUND_OFS);
42 ret.blue = LV_UDIV255((uint16_t)c1.blue * mix + c2.blue * (255 - mix) + LV_COLOR_MIX_ROUND_OFS);
43 return ret;
44}

46lv_color32_t lv_color_mix32(lv_color32_t fg, lv_color32_t bg)
47{
48 if(fg.alpha >= LV_OPA_MAX) {
49 fg.alpha = bg.alpha;
50 return fg;
51 }
52 if(fg.alpha <= LV_OPA_MIN) {
53 return bg;
54 }
55 bg.red = LV_UDIV255((uint32_t)((uint32_t)fg.red * fg.alpha + (uint32_t)bg.red * (255 - fg.alpha)));
56 bg.green = LV_UDIV255((uint32_t)((uint32_t)fg.green * fg.alpha + (uint32_t)bg.green * (255 - fg.alpha)));
57 bg.blue = LV_UDIV255((uint32_t)((uint32_t)fg.blue * fg.alpha + (uint32_t)bg.blue * (255 - fg.alpha)));
58 return bg;
59}

61lv_color32_t lv_color_mix32_premultiplied(lv_color32_t fg, lv_color32_t bg)
62{
63 if(fg.alpha >= LV_OPA_MAX) {
64 return fg; /* Fully opaque foreground replaces background */
65 }
66 if(fg.alpha <= LV_OPA_MIN) {
67 return bg; /* Fully transparent foreground, return background */
68 }
69
70 uint32_t inv_fg_alpha = LV_OPA_MAX - fg.alpha;
71
72 /* Premultiplied blending */
73 bg.red = fg.red + ((bg.red * inv_fg_alpha) >> 8);
74 bg.green = fg.green + ((bg.green * inv_fg_alpha) >> 8);
75 bg.blue = fg.blue + ((bg.blue * inv_fg_alpha) >> 8);
76
77 return bg;
78}

93lv_color32_t lv_color_over32(lv_color32_t fg, lv_color32_t bg)
94{
95 if(fg.alpha >= LV_OPA_MAX || bg.alpha <= LV_OPA_MIN) {
96 return fg;
97 }
98 else if(fg.alpha <= LV_OPA_MIN) {
99 return bg;
100 }
101 else if(bg.alpha == 255) {
102 return lv_color_mix32(fg, bg);
103 }
104
105 lv_opa_t res_alpha = 255 - LV_OPA_MIX2(255 - fg.alpha, 255 - bg.alpha);
106 lv_opa_t ratio = (uint32_t)((uint32_t)fg.alpha * 255) / res_alpha;
107 fg.alpha = ratio;
108 lv_color32_t res = lv_color_mix32(fg, bg);
109 res.alpha = res_alpha;
110
111 return res;
112}
