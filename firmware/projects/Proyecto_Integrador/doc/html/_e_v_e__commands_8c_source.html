331static void CoprocessorFaultRecover(void)
332{
333#if EVE_GEN > 2
334 uint16_t copro_patch_pointer;
335 copro_patch_pointer = EVE_memRead16(REG_COPRO_PATCH_PTR);
336#endif
337
338 EVE_memWrite8(REG_CPURESET, 1U); /* hold coprocessor engine in the reset condition */
339 EVE_memWrite16(REG_CMD_READ, 0U); /* set REG_CMD_READ to 0 */
340 EVE_memWrite16(REG_CMD_WRITE, 0U); /* set REG_CMD_WRITE to 0 */
341 EVE_memWrite16(REG_CMD_DL, 0U); /* reset REG_CMD_DL to 0 as required by the BT81x programming guide, should not hurt FT8xx */
342
343#if EVE_GEN > 2
344 EVE_memWrite16(REG_COPRO_PATCH_PTR, copro_patch_pointer);
345
346 /* restore REG_PCLK in case it was set to zero by an error */
347#if (EVE_GEN > 3) && (defined EVE_PCLK_FREQ)
348 EVE_memWrite16(REG_PCLK_FREQ, (uint16_t) EVE_PCLK_FREQ);
349 EVE_memWrite8(REG_PCLK, 1U); /* enable extsync mode */
350#else
351 EVE_memWrite8(REG_PCLK, EVE_PCLK);
352#endif
353
354#endif
355 EVE_memWrite8(REG_CPURESET, 0U); /* set REG_CPURESET to 0 to restart the coprocessor engine*/
356 DELAY_MS(10U); /* just to be safe */
357}

367uint8_t EVE_busy(void)
368{
369 uint16_t space;
370 uint8_t ret = EVE_IS_BUSY;
371
372#if defined (EVE_DMA)
373 if (0 == EVE_dma_busy)
374 {
375#endif
376
377 space = EVE_memRead16(REG_CMDB_SPACE);
378
379 /* (REG_CMDB_SPACE & 0x03) != 0 -> we have a coprocessor fault */
380 if ((space & 3U) != 0U) /* we have a coprocessor fault, make EVE play with us again */
381 {
382 ret = EVE_FAULT_RECOVERED;
383 fault_recovered = EVE_FAULT_RECOVERED; /* save fault recovery state */
384 CoprocessorFaultRecover();
385 }
386 else
387 {
388 if (0xffcU == space)
389 {
390 ret = E_OK;
391 }
392 else if (space > 0x800U)
393 {
394 ret = EVE_FIFO_HALF_EMPTY;
395 }
396 else
397 {
398 ret = EVE_IS_BUSY;
399 }
400 }
401
402#if defined (EVE_DMA)
403 }
404#endif
405
406 return (ret);
407}

449static void private_block_write(const uint8_t *p_data, uint16_t len)
450{
451 uint8_t padding;
452
453 padding = (uint8_t) (len & 3U); /* 0, 1, 2 or 3 */
454 padding = 4U - padding; /* 4, 3, 2 or 1 */
455 padding &= 3U; /* 3, 2 or 1 */
456
457 for (uint16_t count = 0U; count < len; count++)
458 {
459 spi_transmit(fetch_flash_byte(&p_data[count]));
460 }
461
462 while (padding > 0U)
463 {
464 spi_transmit(0U);
465 padding--;
466 }
467}

469static void block_transfer(const uint8_t *p_data, uint32_t len); /* prototype to comply with MISRA */
470
471static void block_transfer(const uint8_t *p_data, uint32_t len)
472{
473 uint32_t bytes_left;
474 uint32_t offset = 0U;
475
476 bytes_left = len;
477 while (bytes_left > 0U)
478 {
479 uint32_t block_len;
480
481 block_len = (bytes_left > 3840UL) ? 3840UL : bytes_left;
482
483 EVE_cs_set();
484 spi_transmit((uint8_t) 0xB0U); /* high-byte of REG_CMDB_WRITE + MEM_WRITE */
485 spi_transmit((uint8_t) 0x25U); /* middle-byte of REG_CMDB_WRITE */
486 spi_transmit((uint8_t) 0x78U); /* low-byte of REG_CMDB_WRITE */
487 private_block_write(&p_data[offset], (uint16_t) block_len);
488 EVE_cs_clear();
489 offset += block_len;
490 bytes_left -= block_len;
491 EVE_execute_cmd();
492 }
493}

1589{

1591
1592 EVE_pdn_set();
1593 DELAY_MS(6U); /* minimum time for power-down is 5ms */
1594 EVE_pdn_clear();
1595 DELAY_MS(21U); /* minimum time to allow from rising PD_N to first access is 20ms */
1596
1597#if defined (EVE_GD3X)
1598 EVE_cmdWrite(EVE_RST_PULSE,0U); /* reset, only required for warm-start if PowerDown line is not used */
1599#endif

1603 }
1604 else {

1606 }

1608#if EVE_GEN > 2

1610#endif
1611
1612 EVE_cmdWrite(EVE_ACTIVE, 0U); /* start EVE */
1613 DELAY_MS(40U); /* give EVE a moment of silence to power up */

1616 if (E_OK == ret)
1617 {

1622#if EVE_GEN > 2
1623 EVE_memWrite32(REG_FREQUENCY, 72000000UL);
1624#endif

1626/* we have a display with a Goodix GT911 / GT9271 touch-controller on it,
1627 so we patch our FT811 or FT813 according to AN_336 or setup a BT815 / BT817 accordingly */
1628 if(EVE_HAS_GT911) {
1629 use_gt911();
1630 }
1631
1632#if defined (EVE_ADAM101)
1633 EVE_memWrite8(REG_PWM_DUTY, 0x80U); /* turn off backlight for Glyn ADAM101 module, it uses inverted values */
1634#else
1635 EVE_memWrite8(REG_PWM_DUTY, 0U); /* turn off backlight for any other module */
1636#endif
1637 EVE_write_display_parameters();
1638
1639 /* disable Audio for now */
1640 EVE_memWrite8(REG_VOL_PB, 0U); /* turn recorded audio volume down, reset-default is 0xff */
1641 EVE_memWrite8(REG_VOL_SOUND, 0U); /* turn synthesizer volume down, reset-default is 0xff */
1642 EVE_memWrite16(REG_SOUND, EVE_MUTE); /* set synthesizer to mute */

1646 EVE_memWrite32(EVE_RAM_DL + 4U, (DL_CLEAR | CLR_COL | CLR_STN | CLR_TAG));

1648 EVE_memWrite32(REG_DLSWAP, EVE_DLSWAP_FRAME);

1650
1651#if defined (EVE_GD3X)

1660 DELAY_MS(1U);
1661 EVE_execute_cmd(); /* just to be safe, wait for EVE to not be busy */
1662
1663#if defined (EVE_DMA)
1664 EVE_init_dma(); /* prepare DMA */
1665#endif
1666 }
1667 }
1668
1669 return (ret);

1721/* write a string to coprocessor memory in context of a command: */
1722/* no chip-select, just plain SPI-transfers */
1723static void private_string_write(const char *p_text)
1724{
1725 /* treat the array as bunch of bytes */
1726 const uint8_t *const p_bytes = (const uint8_t *)p_text;
1727
1728 if (0U == cmd_burst)
1729 {
1730 uint8_t textindex = 0U;
1731 uint8_t padding;
1732
1733 /* either leave on Zero or when the string is too long */
1734 while ((textindex < 249U) && (p_bytes[textindex] != 0U))
1735 {
1736 spi_transmit(p_bytes[textindex]);
1737 textindex++;
1738 }
1739
1740 /* transmit at least one 0x00 byte */
1741 /* and up to four if the string happens to be 4-byte aligned already */
1742 padding = textindex & 3U; /* 0, 1, 2 or 3 */
1743 padding = 4U - padding; /* 4, 3, 2 or 1 */
1744
1745 while (padding > 0U)
1746 {
1747 spi_transmit(0U);
1748 padding--;
1749 }
1750 }
1751 else /* we are in burst mode, so every transfer is 32 bits */
1752 {
1753 for (uint8_t textindex = 0U; textindex < 249U; textindex += 4U)
1754 {
1755 uint32_t calc = 0U;
1756
1757 for (uint8_t index = 0U; index < 4U; index++)
1758 {
1759 uint8_t data;
1760
1761 data = p_bytes[textindex + index];
1762
1763 if (0U == data)
1764 {
1765 spi_transmit_burst(calc);
1766 return; /* MISRA 2012 rule 15.5 (advisory) violation */
1767 }
1768
1769 calc += ((uint32_t)data) << (index * 8U);
1770 }
1771
1772 spi_transmit_burst(calc);
1773 }
1774
1775 spi_transmit_burst(0U); /* executed when the line is too long */
1776 }
1777}

3796/* This is meant to be called outside display-list building. */
3797/* This function displays an interactive calibration screen, calculates the calibration values */
3798/* and writes the new values to the touch matrix registers of EVE.*/
3799/* Unlike the built-in cmd_calibrate() of EVE this also works with displays that are cut down from larger ones like
3800 * EVE2-38A / EVE2-38G. */
3801/* The dimensions are needed as parameter as EVE_VSIZE for the EVE2-38 is 272 but the visible size is only 116. */
3802/* So the call would be EVE_calibrate_manual(EVE_HSIZE, 116); for the EVE2-38A and EVE2-38G while for most other
3803 * displays */
3804/* using EVE_calibrate_manual(EVE_VSIZE, EVE_VSIZE) would work - but for normal displays the built-in cmd_calibrate
3805 * would work as expected anyways */
3806/* This code was taken from the MatrixOrbital EVE2-Library on Github, adapted and modified */
3807void EVE_calibrate_manual(uint16_t width, uint16_t height)
3808{
3809 int32_t display_x[3U];
3810 int32_t display_y[3U];
3811 int32_t touch_x[3U];
3812 int32_t touch_y[3U];
3813 uint32_t touch_value;
3814 int32_t tmp;
3815 int32_t divi;
3816 int32_t trans_matrix[6U];
3817 uint8_t count = 0U;
3818 uint8_t calc = 0U;
3819 uint32_t calc32 = 0U;
3820 char num[4U];
3821 uint8_t touch_lock = 1U;
3822
3823 /* these values determine where your calibration points will be drawn on your display */
3824 display_x[0U] = (int32_t) width / 6;
3825 display_y[0U] = (int32_t) height / 6;
3826
3827 display_x[1U] = (int32_t) width - ((int32_t) width / 8);
3828 display_y[1U] = (int32_t) height / 2;
3829
3830 display_x[2U] = (int32_t) width / 2;
3831 display_y[2U] = (int32_t) height - ((int32_t) height / 8);
3832
3833 while (count < 3U)
3834 {
3835 EVE_cmd_dl(CMD_DLSTART);
3836 EVE_cmd_dl(DL_CLEAR_COLOR_RGB);
3837 EVE_cmd_dl(DL_CLEAR | CLR_COL | CLR_STN | CLR_TAG);
3838 EVE_cmd_dl(DL_VERTEX_FORMAT); /* set to 0 - reduce precision for VERTEX2F to 1 pixel instead of 1/16 pixel default */
3839
3840 /* draw Calibration Point on screen */
3841 EVE_cmd_dl(DL_COLOR_RGB | 0x0000ffUL);
3842 EVE_cmd_dl(POINT_SIZE(15U * 16U));
3843 EVE_cmd_dl((DL_BEGIN | EVE_POINTS));
3844
3845 int16_t xc0;
3846 int16_t yc0;
3847
3848 xc0 = (int16_t) display_x[count];
3849 yc0 = (int16_t) display_y[count];
3850 EVE_cmd_dl(VERTEX2F(xc0, yc0));
3851 EVE_cmd_dl(DL_END);
3852 EVE_cmd_dl(DL_COLOR_RGB | 0xffffffUL);
3853 EVE_cmd_text((int16_t) width / 2, 20, 26U, EVE_OPT_CENTER, "tap on the dot");
3854 calc = count + 0x31U;
3855 num[0U] = (char) calc;
3856 num[1U] = (char) 0U; /* null terminated string of one character */
3857 EVE_cmd_text((int16_t) display_x[count], (int16_t) display_y[count], 27U, EVE_OPT_CENTER, num);
3858
3859 EVE_cmd_dl(DL_DISPLAY);
3860 EVE_cmd_dl(CMD_SWAP);
3861 EVE_execute_cmd();
3862
3863 for (;;)
3864 {
3865 touch_value = EVE_memRead32(REG_TOUCH_DIRECT_XY); /* read for any new touch tag inputs */
3866
3867 if (touch_lock != 0U)
3868 {
3869 if ((touch_value & 0x80000000UL) != 0UL) /* check if we have no touch */
3870 {
3871 touch_lock = 0U;
3872 }
3873 }
3874 else
3875 {
3876 if (0UL == (touch_value & 0x80000000UL)) /* check if a touch is detected */
3877 {
3878 calc32 = ((touch_value >> 16U) & 0x03FFUL);
3879 touch_x[count] = (int32_t) calc32; /* raw Touchscreen X coordinate */
3880 calc32 = touch_value & 0x03FFUL;
3881 touch_y[count] = (int32_t) calc32; /* raw Touchscreen Y coordinate */
3882 touch_lock = 1U;
3883 count++;
3884 break; /* leave for (;;) */
3885 }
3886 }
3887 }
3888 }
3889
3890 divi = ((touch_x[0U] - touch_x[2U]) * (touch_y[1U] - touch_y[2U])) - ((touch_x[1U] - touch_x[2U]) * (touch_y[0U] - touch_y[2U]));
3891
3892 tmp = (((display_x[0U] - display_x[2U]) * (touch_y[1U] - touch_y[2U])) -
3893 ((display_x[1U] - display_x[2U]) * (touch_y[0U] - touch_y[2U])));
3894 trans_matrix[0U] = (int32_t) (((int64_t) tmp * 65536) / divi);
