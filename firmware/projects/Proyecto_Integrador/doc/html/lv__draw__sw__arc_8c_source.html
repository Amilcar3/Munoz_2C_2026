52void lv_draw_sw_arc(lv_draw_task_t * t, const lv_draw_arc_dsc_t * dsc, const lv_area_t * coords)
53{
54#if LV_DRAW_SW_COMPLEX
55 if(dsc->opa <= LV_OPA_MIN) return;
56 if(dsc->width == 0) return;
57 if(dsc->start_angle == dsc->end_angle) return;
58
59 int32_t width = dsc->width;
60 if(width > dsc->radius) width = dsc->radius;
61
62 lv_area_t area_out = *coords;
63 lv_area_t clipped_area;
64 if(!lv_area_intersect(&clipped_area, &area_out, &t->clip_area)) return;
65
66 /*Draw a full ring*/
67 if(dsc->img_src == NULL &&
68 (dsc->start_angle + 360 == dsc->end_angle || dsc->start_angle == dsc->end_angle + 360)) {
69 lv_draw_border_dsc_t cir_dsc;
70 lv_draw_border_dsc_init(&cir_dsc);
71 cir_dsc.opa = dsc->opa;
72 cir_dsc.color = dsc->color;
73 cir_dsc.width = width;
74 cir_dsc.radius = LV_RADIUS_CIRCLE;
75 cir_dsc.side = LV_BORDER_SIDE_FULL;
76 lv_draw_sw_border(t, &cir_dsc, &area_out);
77 return;
78 }
79
80 lv_area_t area_in;
81 area_in = area_out;
82 area_in.x1 += dsc->width;
83 area_in.y1 += dsc->width;
84 area_in.x2 -= dsc->width;
85 area_in.y2 -= dsc->width;
86
87 /*Optimization: Minimize clipped area*/
88 lv_area_t arc_area;
89 lv_draw_arc_get_area(dsc->center.x, dsc->center.y, dsc->radius, dsc->start_angle, dsc->end_angle,
90 width, dsc->rounded, &arc_area);
91 if(!lv_area_intersect(&clipped_area, &clipped_area, &arc_area)) return;
92
93 int32_t start_angle = (int32_t)dsc->start_angle;
94 int32_t end_angle = (int32_t)dsc->end_angle;
95 while(start_angle >= 360) start_angle -= 360;
96 while(end_angle >= 360) end_angle -= 360;
97
98 void * mask_list[4] = {0};
99 /*Create an angle mask*/
100 lv_draw_sw_mask_angle_param_t mask_angle_param;
101 lv_draw_sw_mask_angle_init(&mask_angle_param, dsc->center.x, dsc->center.y, start_angle, end_angle);
102 mask_list[0] = &mask_angle_param;
103
104 /*Create an outer mask*/
105 lv_draw_sw_mask_radius_param_t mask_out_param;
106 lv_result_t res = lv_draw_sw_mask_radius_init(&mask_out_param, &area_out, LV_RADIUS_CIRCLE, false);
107 mask_list[1] = &mask_out_param;
108 if(res != LV_RESULT_OK) {
109 LV_LOG_WARN("Couldn't create outer arc mask");
110 goto outer_arc_mask_fail;
111 }
112
113 /*Create inner the mask*/
114 lv_draw_sw_mask_radius_param_t mask_in_param;
115 bool mask_in_param_valid = false;
116 if(lv_area_get_width(&area_in) > 0 && lv_area_get_height(&area_in) > 0) {
117 res = lv_draw_sw_mask_radius_init(&mask_in_param, &area_in, LV_RADIUS_CIRCLE, true);
118 if(res != LV_RESULT_OK) {
119 LV_LOG_WARN("Couldn't create inner arc mask");
120 goto inner_arc_mask_fail;
121 }
122 mask_list[2] = &mask_in_param;
123 mask_in_param_valid = true;
124 }
125
126 int32_t blend_h = lv_area_get_height(&clipped_area);
127 int32_t blend_w = lv_area_get_width(&clipped_area);
128 int32_t h;
129 lv_opa_t * mask_buf = lv_malloc(blend_w);
130 if(!mask_buf) {
131 LV_LOG_WARN("Couldn't allocate arc mask buffer");
132 goto mask_buf_alloc_fail;
133 }
134
135 lv_area_t blend_area = clipped_area;
136 lv_area_t img_area;
137 lv_draw_sw_blend_dsc_t blend_dsc = {0};
138 blend_dsc.mask_buf = mask_buf;
139 blend_dsc.opa = dsc->opa;
140 blend_dsc.blend_area = &blend_area;
141 blend_dsc.mask_area = &blend_area;
142
143 const uint8_t * img_mask = NULL;
144 lv_image_decoder_dsc_t decoder_dsc;
145 if(dsc->img_src == NULL) {
146 blend_dsc.color = dsc->color;
147 }
148 else {
149 res = lv_image_decoder_open(&decoder_dsc, dsc->img_src, NULL);
150 if(res == LV_RESULT_INVALID || decoder_dsc.decoded == NULL) {
151 LV_LOG_WARN("Can't decode the background image");
152 blend_dsc.color = dsc->color;
153 }
154 else {
155 img_area.x1 = 0;
156 img_area.y1 = 0;
157 img_area.x2 = decoder_dsc.decoded->header.w - 1;
158 img_area.y2 = decoder_dsc.decoded->header.h - 1;
159 int32_t ofs = decoder_dsc.decoded->header.w / 2;
160 lv_area_move(&img_area, dsc->center.x - ofs, dsc->center.y - ofs);
161 blend_dsc.src_area = &img_area;
162 blend_dsc.src_buf = decoder_dsc.decoded->data;
163 blend_dsc.src_stride = decoder_dsc.decoded->header.stride;
164 blend_dsc.src_color_format = decoder_dsc.decoded->header.cf;
165 if(blend_dsc.src_color_format == LV_COLOR_FORMAT_RGB565A8) {
166 blend_dsc.src_color_format = LV_COLOR_FORMAT_RGB565;
167 img_mask = (uint8_t *)blend_dsc.src_buf + blend_dsc.src_stride * lv_area_get_height(blend_dsc.src_area);
168 }
169 }
170 }
171
172 lv_area_t round_area_1;
173 lv_area_t round_area_2;
174 lv_opa_t * circle_mask = NULL;
175 if(dsc->rounded) {
176 circle_mask = allocate_circle_mask(dsc, start_angle, end_angle, width, &round_area_1, &round_area_2);
177 if(!circle_mask) {
178 LV_LOG_WARN("Failed to allocate circle mask. Arc will look degraded");
179 }
180 }
181
182 blend_area.y2 = blend_area.y1;
183 for(h = 0; h < blend_h; h++) {
184 lv_memset(mask_buf, 0xff, blend_w);
185 blend_dsc.mask_res = lv_draw_sw_mask_apply(mask_list, mask_buf, blend_area.x1, blend_area.y1, blend_w);
186
187 if(circle_mask) {
188 if(blend_area.y1 >= round_area_1.y1 && blend_area.y1 <= round_area_1.y2) {
189 if(blend_dsc.mask_res == LV_DRAW_SW_MASK_RES_TRANSP) {
190 lv_memzero(mask_buf, blend_w);
191 blend_dsc.mask_res = LV_DRAW_SW_MASK_RES_CHANGED;
192 }
193 add_circle(circle_mask, &blend_area, &round_area_1, mask_buf, width);
194 }
195 if(blend_area.y1 >= round_area_2.y1 && blend_area.y1 <= round_area_2.y2) {
196 if(blend_dsc.mask_res == LV_DRAW_SW_MASK_RES_TRANSP) {
197 lv_memzero(mask_buf, blend_w);
198 blend_dsc.mask_res = LV_DRAW_SW_MASK_RES_CHANGED;
199 }
200 add_circle(circle_mask, &blend_area, &round_area_2, mask_buf, width);
201 }
202 }
203
204 /*If it was an RGB565A8 image use consider its A8 part on the mask*/
205 if(img_mask && blend_dsc.mask_res != LV_DRAW_SW_MASK_RES_TRANSP) {
206 const uint8_t * img_mask_tmp = img_mask;
207 img_mask_tmp += blend_dsc.src_stride / 2 * (blend_area.y1 - blend_dsc.src_area->y1);
208 img_mask_tmp += blend_area.x1 - blend_dsc.src_area->x1;
209
210 int32_t i;
211 for(i = 0; i < blend_w; i++) {
212 mask_buf[i] = LV_OPA_MIX2(mask_buf[i], img_mask_tmp[i]);
213 }
214 if(blend_dsc.mask_res == LV_DRAW_SW_MASK_RES_FULL_COVER) {
215 blend_dsc.mask_res = LV_DRAW_SW_MASK_RES_CHANGED;
216 }
217 }
218
219 lv_draw_sw_blend(t, &blend_dsc);
220
221 blend_area.y1 ++;
222 blend_area.y2 ++;
223 }
224
225 lv_draw_sw_mask_free_param(&mask_angle_param);
226 lv_draw_sw_mask_free_param(&mask_out_param);
227 if(mask_in_param_valid) {
228 lv_draw_sw_mask_free_param(&mask_in_param);
229 }
230
231 lv_free(mask_buf);
232 if(dsc->img_src) lv_image_decoder_close(&decoder_dsc);
233 if(circle_mask) lv_free(circle_mask);
234 return;
235
236mask_buf_alloc_fail:
237 if(mask_in_param_valid) {
238 lv_draw_sw_mask_free_param(&mask_in_param);
239 }
240inner_arc_mask_fail:
241 lv_draw_sw_mask_free_param(&mask_out_param);
242outer_arc_mask_fail:
243 lv_draw_sw_mask_free_param(&mask_angle_param);
244
245#else
246 LV_LOG_WARN("Can't draw arc with LV_DRAW_SW_COMPLEX == 0");
247 LV_UNUSED(center);
248 LV_UNUSED(radius);
249 LV_UNUSED(start_angle);
250 LV_UNUSED(end_angle);
251 LV_UNUSED(layer);
252 LV_UNUSED(dsc);
253#endif /*LV_DRAW_SW_COMPLEX*/
254}

260static void add_circle(const lv_opa_t * circle_mask, const lv_area_t * blend_area, const lv_area_t * circle_area,
261 lv_opa_t * mask_buf, int32_t width)
262{
263 lv_area_t circle_common_area;
264 if(lv_area_intersect(&circle_common_area, circle_area, blend_area)) {
265 const lv_opa_t * circle_mask_tmp = circle_mask + width * (circle_common_area.y1 - circle_area->y1);
266 circle_mask_tmp += circle_common_area.x1 - circle_area->x1;
267
268 lv_opa_t * mask_buf_tmp = mask_buf + circle_common_area.x1 - blend_area->x1;
269
270 uint32_t x;
271 uint32_t w = lv_area_get_width(&circle_common_area);
272 for(x = 0; x < w; x++) {
273 uint32_t res = mask_buf_tmp[x] + circle_mask_tmp[x];
274 mask_buf_tmp[x] = res > 255 ? 255 : res;
275 }
276 }

280static void get_rounded_area(int16_t angle, int32_t radius, uint8_t thickness, lv_area_t * res_area)
281{
282 int32_t thick_half = thickness / 2;
283 uint8_t thick_corr = (thickness & 0x01) ? 0 : 1;
284
285 int32_t cir_x;
286 int32_t cir_y;
287
288 cir_x = ((radius - thick_half) * lv_trigo_cos(angle)) >> (LV_TRIGO_SHIFT - 8);
289 cir_y = ((radius - thick_half) * lv_trigo_sin(angle)) >> (LV_TRIGO_SHIFT - 8);
290
291 /*The center of the pixel need to be calculated so apply 1/2 px offset*/
292 if(cir_x > 0) {
293 cir_x = (cir_x - 128) >> 8;
294 res_area->x1 = cir_x - thick_half + thick_corr;
295 res_area->x2 = cir_x + thick_half;
296 }
297 else {
298 cir_x = (cir_x + 128) >> 8;
299 res_area->x1 = cir_x - thick_half;
300 res_area->x2 = cir_x + thick_half - thick_corr;
301 }
302
303 if(cir_y > 0) {
304 cir_y = (cir_y - 128) >> 8;
305 res_area->y1 = cir_y - thick_half + thick_corr;
306 res_area->y2 = cir_y + thick_half;
307 }
308 else {
309 cir_y = (cir_y + 128) >> 8;
310 res_area->y1 = cir_y - thick_half;
311 res_area->y2 = cir_y + thick_half - thick_corr;
312 }
313}

315static inline lv_opa_t * allocate_circle_mask(const lv_draw_arc_dsc_t * dsc, int32_t start_angle, int32_t end_angle,
316 uint32_t width, lv_area_t * out_round_area_1, lv_area_t * out_round_area_2)
317{
318
319 lv_opa_t * circle_mask = lv_malloc(width * width);
320 if(!circle_mask) {
321 return NULL;
322 }
323 lv_memset(circle_mask, 0xff, width * width);
324 lv_area_t circle_area = {0, 0, width - 1, width - 1};
325 lv_draw_sw_mask_radius_param_t circle_mask_param;
326 lv_result_t result = lv_draw_sw_mask_radius_init(&circle_mask_param, &circle_area, width / 2, false);
327 if(result != LV_RESULT_OK) {
328 lv_free(circle_mask);
329 return NULL;
330 }
331 void * circle_mask_list[2] = {&circle_mask_param, NULL};
332
333 lv_opa_t * circle_mask_tmp = circle_mask;
334 for(uint32_t h = 0; h < width; h++) {
335 lv_draw_sw_mask_res_t res = lv_draw_sw_mask_apply(circle_mask_list, circle_mask_tmp, 0, h, width);
336 if(res == LV_DRAW_SW_MASK_RES_TRANSP) {
337 lv_memzero(circle_mask_tmp, width);
338 }
339
340 circle_mask_tmp += width;
341 }
342 lv_draw_sw_mask_free_param(&circle_mask_param);
343
344 get_rounded_area(start_angle, dsc->radius, width, out_round_area_1);
345 lv_area_move(out_round_area_1, dsc->center.x, dsc->center.y);
346 get_rounded_area(end_angle, dsc->radius, width, out_round_area_2);
347 lv_area_move(out_round_area_2, dsc->center.x, dsc->center.y);
348 return circle_mask;
349}
