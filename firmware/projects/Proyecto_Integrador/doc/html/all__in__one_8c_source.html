232static void draw_hollow_hand(lv_layer_t *layer, int32_t cx, int32_t cy, float deg,
233 int len, int tail, float hw, lv_color_t col)
234{
235 float rad = deg * (float)M_PI / 180.0f;
236 float dx = sinf(rad);
237 float dy = -cosf(rad);
238 float px = -dy; /* perpendicular */
239 float py = dx;
240
241 float ax = cx - dx * tail - px * hw, ay = cy - dy * tail - py * hw;
242 float bx = cx - dx * tail + px * hw, by = cy - dy * tail + py * hw;
243 float qx = cx + dx * len + px * hw, qy = cy + dy * len + py * hw;
244 float rx = cx + dx * len - px * hw, ry = cy + dy * len - py * hw;
245
246 lv_draw_line_dsc_t d;
247 lv_draw_line_dsc_init(&d);
248 d.color = col;
249 d.width = 2;
250 d.opa = LV_OPA_COVER;
251 d.round_start = 1;
252 d.round_end = 1;
253
254 draw_line(layer, &d, ax, ay, bx, by);
255 draw_line(layer, &d, bx, by, qx, qy);
256 draw_line(layer, &d, qx, qy, rx, ry);
257 draw_line(layer, &d, rx, ry, ax, ay);
258}

265static void dial_draw_cb(lv_event_t *e)
266{
267 const dial_cfg_t *cfg = (const dial_cfg_t *)lv_event_get_user_data(e);
268 lv_layer_t *layer = lv_event_get_layer(e);
269 lv_obj_t *obj = lv_event_get_target_obj(e);
270
271 lv_area_t coords;
272 lv_obj_get_coords(obj, &coords);
273 const int32_t cx = coords.x1 + WF_CX;
274 const int32_t cy = coords.y1 + WF_CY;
275
276 /* Círculo interior */
277 if(cfg->inner_circle) {
278 draw_circle(layer, cx - R_INNER, cy - R_INNER, cx + R_INNER - 1, cy + R_INNER - 1,
279 lv_color_hex(C_INNER), lv_color_hex(C_INNER_BORDER), 1);
280 }
281
282 /* Ticks */
283 lv_draw_line_dsc_t ld;
284 lv_draw_line_dsc_init(&ld);
285 ld.round_start = 1;
286 ld.round_end = 1;
287 ld.opa = LV_OPA_COVER;
288 for(int i = 0; i < 60; i++) {
289 bool major = (i % 5) == 0;
290 int r_in = major ? R_TICK_MAJOR_IN : R_TICK_MINOR_IN;
291 ld.color = lv_color_hex(major ? C_TICK_MAJOR : C_TICK_MINOR);
292 ld.width = major ? 3 : 1;
293 draw_line(layer, &ld,
294 cx + r_in * s_sin[i], cy - r_in * s_cos[i],
295 cx + R_TICK_OUT * s_sin[i], cy - R_TICK_OUT * s_cos[i]);
296 }
297
298 /* Números 1..12 */
299 lv_draw_label_dsc_t td;
300 lv_draw_label_dsc_init(&td);
301 td.font = cfg->num_font;
302 td.color = lv_color_hex(C_NUMBER);
303 td.opa = LV_OPA_COVER;
304 td.align = LV_TEXT_ALIGN_CENTER;
305 const int32_t fh = lv_font_get_line_height(td.font);
306 for(int h = 1; h <= 12; h++) {
307 int idx = (h % 12) * 5;
308 int32_t px = cx + (int32_t)lroundf(R_NUMBERS * s_sin[idx]);
309 int32_t py = cy - (int32_t)lroundf(R_NUMBERS * s_cos[idx]);
310 td.text = s_hour_txt[h - 1];
311 lv_area_t ta = { px - 14, py - fh / 2, px + 13, py - fh / 2 + fh - 1 };
312 lv_draw_label(layer, &td, &ta);
313 }
314
315 /* Íconos: corazón (HR) y burbuja (SpO2) de la vista ALL */
316 if(cfg->icons) {
317 draw_heart(layer, cx - 50, cy - 14, lv_color_hex(C_HEART));
318 draw_bubble(layer, cx + 30, cy - 38, lv_color_hex(C_BUBBLE));
319 }
320}

338static void hands_draw_cb(lv_event_t *e)
339{
340 lv_layer_t *layer = lv_event_get_layer(e);
341 lv_obj_t *obj = lv_event_get_target_obj(e);
342
343 lv_area_t coords;
344 lv_obj_get_coords(obj, &coords);
345 const int32_t cx = coords.x1 + WF_CX;
346 const int32_t cy = coords.y1 + WF_CY;
347
348 float hour_deg = (float)(s_h % 12) * 30.0f + (float)s_m * 0.5f;
349 float min_deg = (float)s_m * 6.0f + (float)s_s * 0.1f;
350 float sec_deg = (float)s_s * 6.0f;
351
352 lv_color_t hand = lv_color_hex(C_HAND);
353 draw_hollow_hand(layer, cx, cy, hour_deg, HOUR_LEN, HOUR_TAIL, HOUR_HALF_W, hand);
354 draw_hollow_hand(layer, cx, cy, min_deg, MIN_LEN, MIN_TAIL, MIN_HALF_W, hand);
355
356 /* Segundero: frente azul, cola roja */
357 float rad = sec_deg * (float)M_PI / 180.0f;
358 float dx = sinf(rad), dy = -cosf(rad);
359 lv_draw_line_dsc_t sd;
360 lv_draw_line_dsc_init(&sd);
361 sd.width = 1;
362 sd.opa = LV_OPA_COVER;
363 sd.round_start = 1;
364 sd.round_end = 1;
365
366 sd.color = lv_color_hex(C_SEC_FRONT);
367 draw_line(layer, &sd, cx, cy, cx + dx * SEC_LEN, cy + dy * SEC_LEN);
368 sd.color = lv_color_hex(C_SEC_TAIL);
369 draw_line(layer, &sd, cx, cy, cx - dx * SEC_TAIL, cy - dy * SEC_TAIL);
370
371 /* Eje central */
372 draw_circle(layer, cx - 6, cy - 6, cx + 5, cy + 5,

572lv_obj_t *all_in_one_create(lv_obj_t *parent)
573{
574 if(parent == NULL) parent = lv_screen_active();
575

683 lv_anim_set_values(&a, 0, 100);
684 lv_anim_set_duration(&a, dur);

699static void demo_cb(lv_timer_t *t)
700{
701 (void)t;
702 static float phase = 0.0f;
703 float prev = phase;
704
705 phase += 0.02f / 0.83f; /* 50 Hz, latido de ~0.83 s (72 BPM) */
706 if(phase >= 1.0f) phase -= 1.0f;
707
708 /* El latido se dispara justo en el pico de la onda */
709 if(prev < 0.20f && phase >= 0.20f) all_in_one_beat();
710
711 float a = (phase - 0.20f) / 0.06f;
712 float b = (phase - 0.45f) / 0.08f;
713 float v = 0.25f + 0.65f * expf(-a * a) + 0.18f * expf(-b * b);
714 all_in_one_ppg_push((int)(v * 1000.0f));
715}
