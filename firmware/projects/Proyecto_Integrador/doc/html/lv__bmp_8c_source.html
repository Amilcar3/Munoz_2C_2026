91static lv_result_t decoder_info(lv_image_decoder_t * decoder, lv_image_decoder_dsc_t * dsc, lv_image_header_t * header)
92{
93 LV_UNUSED(decoder);
94
95 const void * src = dsc->src;
96 lv_image_src_t src_type = dsc->src_type; /*Get the source type*/
97
98 /*If it's a BMP file...*/
99 if(src_type == LV_IMAGE_SRC_FILE) {
100 const char * fn = src;
101 if(lv_strcmp(lv_fs_get_ext(fn), "bmp") == 0) { /*Check the extension*/
102 /*Save the data in the header*/
103 uint8_t headers[54];
104
105 uint32_t rn = 0;
106 lv_fs_res_t res = lv_fs_read(&dsc->file, headers, 54, &rn);
107 if(res != LV_FS_RES_OK || rn < 54) {
108 return LV_RESULT_INVALID;
109 }
110
111 uint32_t w;
112 uint32_t h;
113 lv_memcpy(&w, headers + 18, 4);
114 lv_memcpy(&h, headers + 22, 4);
115 header->w = w;
116 header->h = h;
117
118 uint16_t bpp;
119 lv_memcpy(&bpp, headers + 28, 2);
120 switch(bpp) {
121 case 16:
122 header->cf = LV_COLOR_FORMAT_RGB565;
123 break;
124 case 24:
125 header->cf = LV_COLOR_FORMAT_RGB888;
126 break;
127 case 32:
128 header->cf = LV_COLOR_FORMAT_ARGB8888;
129 break;
130 default:
131 LV_LOG_WARN("Not supported bpp: %d", bpp);
132 return LV_RESULT_OK;
133 }
134 return LV_RESULT_OK;
135 }
136 }
137 /* BMP file as data not supported for simplicity.
138 * Convert them to LVGL compatible C arrays directly. */
139 else if(src_type == LV_IMAGE_SRC_VARIABLE) {
140 return LV_RESULT_INVALID;
141 }
142
143 return LV_RESULT_INVALID; /*If didn't succeeded earlier then it's an error*/
144}

152static lv_result_t decoder_open(lv_image_decoder_t * decoder, lv_image_decoder_dsc_t * dsc)
153{
154 LV_UNUSED(decoder);
155
156 /*If it's a BMP file...*/
157 if(dsc->src_type == LV_IMAGE_SRC_FILE) {
158 const char * fn = dsc->src;
159
160 if(lv_strcmp(lv_fs_get_ext(fn), "bmp") != 0) {
161 return LV_RESULT_INVALID; /*Check the extension*/
162 }
163
164 bmp_dsc_t b;
165 lv_memset(&b, 0x00, sizeof(b));
166
167 lv_fs_res_t res = lv_fs_open(&b.f, dsc->src, LV_FS_MODE_RD);
168 if(res != LV_FS_RES_OK) return LV_RESULT_INVALID;
169
170 uint8_t header[54];
171 lv_fs_read(&b.f, header, 54, NULL);
172
173 if(0x42 != header[0] || 0x4d != header[1]) {
174 lv_fs_close(&b.f);
175 return LV_RESULT_INVALID;
176 }
177
178 lv_memcpy(&b.px_offset, header + 10, 4);
179 lv_memcpy(&b.px_width, header + 18, 4);
180 lv_memcpy(&b.px_height, header + 22, 4);
181 lv_memcpy(&b.bpp, header + 28, 2);
182 b.row_size_bytes = ((b.bpp * b.px_width + 31) / 32) * 4;
183
184 dsc->user_data = lv_malloc(sizeof(bmp_dsc_t));
185 LV_ASSERT_MALLOC(dsc->user_data);
186 if(dsc->user_data == NULL) return LV_RESULT_INVALID;
187 lv_memcpy(dsc->user_data, &b, sizeof(b));
188 return LV_RESULT_OK;
189 }
190 /* BMP file as data not supported for simplicity.
191 * Convert them to LVGL compatible C arrays directly. */
192 else if(dsc->src_type == LV_IMAGE_SRC_VARIABLE) {
193 return LV_RESULT_INVALID;
194 }
195
196 return LV_RESULT_INVALID; /*If not returned earlier then it failed*/
197}

199static lv_result_t decoder_get_area(lv_image_decoder_t * decoder, lv_image_decoder_dsc_t * dsc,
200 const lv_area_t * full_area, lv_area_t * decoded_area)
201{
202 LV_UNUSED(decoder);
203 bmp_dsc_t * b = dsc->user_data;
204 lv_draw_buf_t * decoded = (void *)dsc->decoded;
205
206 if(decoded_area->y1 == LV_COORD_MIN) {
207 *decoded_area = *full_area;
208 decoded_area->y2 = decoded_area->y1;
209 int32_t w_px = lv_area_get_width(full_area);
210 lv_draw_buf_t * reshaped = lv_draw_buf_reshape(decoded, dsc->header.cf, w_px, 1, LV_STRIDE_AUTO);
211 if(reshaped == NULL) {
212 if(decoded != NULL) {
213 lv_draw_buf_destroy(decoded);
214 decoded = NULL;
215 dsc->decoded = NULL;
216 }
217 decoded = lv_draw_buf_create_ex(image_cache_draw_buf_handlers, w_px, 1, dsc->header.cf, LV_STRIDE_AUTO);
218 if(decoded == NULL) return LV_RESULT_INVALID;
219 }
220 else {
221 decoded = reshaped;
222 }
223 dsc->decoded = decoded;
224 }
225 else {
226 decoded_area->y1++;
227 decoded_area->y2++;
228 }
229
230 if(decoded_area->y1 > full_area->y2) {
231 return LV_RESULT_INVALID;
232 }
233 else {
234 int32_t y = (b->px_height - 1) - (decoded_area->y1); /*BMP images are stored upside down*/
235 uint32_t p = b->px_offset + b->row_size_bytes * y;
236 p += (decoded_area->x1) * (b->bpp / 8);
237 lv_fs_seek(&b->f, p, LV_FS_SEEK_SET);
238 uint32_t line_width_byte = lv_area_get_width(full_area) * (b->bpp / 8);
239 lv_fs_read(&b->f, decoded->data, line_width_byte, NULL);
240
241 return LV_RESULT_OK;
242 }
243}
