13# Type aliases for field specs
14# ---------------------------------------------------------------------------
15
16# A field spec is a tuple describing one line of print_info output:
17# (label, key_or_fmt)
18#
19# - label: the left-hand label string (e.g. "var")
20# - key_or_fmt: either a dict key string, or a callable(d) -> str
21#
22# Special forms:
23# ("_title", callable(d) -> str) — title line (no indent)
24# ("_children", key, child_fields) — nested list of dicts
25# ("_skip_if", key, sentinel, (label, key_or_fmt))
26# — only print if d[key] != sentinel
27
28FieldSpec = Union[
29 Tuple[str, Union[str, Callable]],
30 Tuple[str, str, Callable],
31 Tuple[str, str, str, Tuple],
32 Tuple[str, str, list],
33]

36def print_info(d, fields: Sequence = None, indent: int = 0) -> None:
37 """Print a snapshot dict using a declarative field spec list.
38
39 Fields resolution order:
40 1. Explicit *fields* parameter (highest priority)
41 2. d._display_spec["info"] (Snapshot self-describing)
42 3. None → default rendering (iterate dict keys as "key = value")
43 """
44 if fields is None:
45 spec = getattr(d, "_display_spec", None)
46 if spec:
47 fields = spec["info"]
48
49 # Default rendering: iterate dict keys
50 if fields is None:
51 prefix = " " * indent
52 items = d.items() if hasattr(d, "items") else vars(d).items()
53 for key, value in items:
54 print(f"{prefix} {key} = {value}")
55 return
56
57 prefix = " " * indent
58 for spec in fields:
59 # Bare string shorthand: "key" → ("key", "key")
60 if isinstance(spec, str):
61 spec = (spec, spec)
62
63 tag = spec[0]
64
65 if tag == "_title":
66 fmt_fn = spec[1]
67 print(f"{prefix}{fmt_fn(d)}")
68
69 elif tag == "_children":
70 key, child_fields = spec[1], spec[2]
71 children = d.get(key, [])
72 for child in children:
73 print_info(child, fields=child_fields, indent=indent + 1)
74
75 elif tag == "_skip_if":
76 key, sentinel, inner = spec[1], spec[2], spec[3]
77 # Bare string shorthand for inner: "key" → ("key", "key")
78 if isinstance(inner, str):
79 inner = (inner, inner)
80 if d.get(key) != sentinel:
81 _print_field(prefix, inner, d)
82
83 else:
84 _print_field(prefix, spec, d)

87def _print_field(prefix: str, spec: tuple, d) -> None:
88 """Print a single (label, key_or_fmt) field."""
89 label, key_or_fmt = spec[0], spec[1]
90 if callable(key_or_fmt):
91 value = key_or_fmt(d)
92 else:
93 value = d.get(key_or_fmt, "")
94 print(f"{prefix} {label:14s} = {value}")

102def print_table(
103 entries: List,
104 columns: List[str],
105 row_fn: Callable,
106 empty_msg: str,
107 align: str = "l",
108 numbered: bool = True,
109 col_align: dict = None,
110) -> None:
111 """Generic helper for printing a PrettyTable from snapshot data.
112
113 Args:
114 entries: list of dict-like objects (Snapshot or plain dict).
115 columns: column header names (excluding '#' if numbered=True).
116 row_fn: callable(index, entry) -> list of cell values.
117 empty_msg: message to print when entries is empty.
118 align: default column alignment.
119 numbered: if True, prepend a '#' column with row index.
120 col_align: optional per-column alignment overrides, e.g. {"src": "l"}.
121 """
122 table = PrettyTable()
123 table.field_names = (["#"] + columns) if numbered else columns
124 table.align = align
125 if col_align:
126 for col, a in col_align.items():
127 if col in table.field_names:
128 table.align[col] = a
129
130 for i, d in enumerate(entries):
131 row = row_fn(i, d)
132 if numbered:
133 row = [i] + row
134 table.add_row(row)
135
136 if not table.rows:
137 print(empty_msg)
138 else:
139 print(table)

141def resolve_table_columns(spec: dict) -> tuple:
142 """Resolve Display_Spec into (column_headers, auto_row_fn).
143
144 Merges the ``info`` field list with ``table`` overrides to produce a flat
145 list of column headers and a row-extraction function with ``(i, d)``
146 signature suitable for :func:`print_table`.
147
148 Algorithm:
149 1. Walk ``spec["info"]``, skip ``_title`` / ``_children``, normalise
150 bare strings, and collect ``(label, key_or_fmt, skip_info)`` triples.
151 2. Walk ``spec["table"]``, build an override map (keyed by label) and
152 an append list for labels not present in the base columns.
153 3. Apply overrides in-place, then append new columns.
154 4. Return ``(headers, auto_row_fn)``.
155 """
156 # -- 1. base columns from info ----------------------------------------
157 base_columns = []
158 base_labels = set()
159 for entry in spec["info"]:
160 if isinstance(entry, str):
161 entry = (entry, entry)
162
163 tag = entry[0]
164 if tag in ("_title", "_children"):
165 continue
166
167 if tag == "_skip_if":
168 skip_key, sentinel, inner = entry[1], entry[2], entry[3]
169 if isinstance(inner, str):
170 inner = (inner, inner)
171 label, key_or_fmt = inner[0], inner[1]
172 base_columns.append((label, key_or_fmt, (skip_key, sentinel)))
173 else:
174 label, key_or_fmt = entry[0], entry[1]
175 base_columns.append((label, key_or_fmt, None))
176
177 base_labels.add(base_columns[-1][0])
178
179 # -- 2. table overrides and appends -----------------------------------
180 override_map = {}
181 append_list = []
182 for entry in spec.get("table", []):
183 if isinstance(entry, str):
184 entry = (entry, entry)
185
186 tag = entry[0]
187 if tag == "_skip_if":
188 skip_key, sentinel, inner = entry[1], entry[2], entry[3]
189 if isinstance(inner, str):
190 inner = (inner, inner)
191 label, key_or_fmt = inner[0], inner[1]
192 item = (label, key_or_fmt, (skip_key, sentinel))
193 else:
194 label, key_or_fmt = entry[0], entry[1]
195 item = (label, key_or_fmt, None)
196
197 if label in base_labels:
198 override_map[label] = item
199 else:
200 append_list.append(item)
201
202 # -- 3. merge ---------------------------------------------------------
203 resolved = []
204 for label, key_or_fmt, skip in base_columns:
205 if label in override_map:
206 resolved.append(override_map[label])
207 else:
208 resolved.append((label, key_or_fmt, skip))
209 resolved.extend(append_list)
210
211 # -- 4. build outputs -------------------------------------------------
212 headers = [label for label, _, _ in resolved]
213
214 def auto_row_fn(i, d):
215 row = []
216 for _label, key_or_fmt, skip in resolved:
217 if skip is not None:
218 sk, sentinel = skip
219 if d.get(sk) == sentinel:
220 row.append("")
221 continue
222 if callable(key_or_fmt):
223 row.append(key_or_fmt(d))
224 else:
225 row.append(d.get(key_or_fmt, ""))
226 return row
227
228 return (headers, auto_row_fn)

230def print_spec_table(
231 entries: list,
232 spec: dict = None,
233 align: str = "l",
234 numbered: bool = True,
235 col_align: dict = None,
236 extra_columns: list = None,
237 extra_row_fn: Callable = None,
238) -> None:
239 """Render a PrettyTable from a Display_Spec.
240
241 Resolves columns from *spec* via :func:`resolve_table_columns`, optionally
242 prepends *extra_columns* (used by DumpCache for dynamic fields), then
243 delegates to :func:`print_table`.
244
245 If *spec* is not provided, it is read from the first entry's
246 ``_display_spec`` attribute.
247 """
248 if spec is None and entries:
249 spec = getattr(entries[0], "_display_spec", None)
250 if spec is None:
251 print("")
252 return
253 columns, auto_row_fn = resolve_table_columns(spec)
254
255 if extra_columns and extra_row_fn:
256 full_columns = extra_columns + columns
257
258 def combined_row_fn(i, d):
259 return extra_row_fn(d) + auto_row_fn(i, d)
260
261 row_fn = combined_row_fn
262 else:
263 full_columns = columns
264 row_fn = auto_row_fn
265
266 print_table(
267 entries,
268 full_columns,
269 row_fn,
270 spec["empty_msg"],
271 align=align,
272 numbered=numbered,
273 col_align=col_align,
274 )
