7static int32_t content_size(lv_obj_t * cont, bool horizontal)
8{
9 int32_t total = (int32_t)lv_obj_get_child_count(cont);
10 int32_t gap = horizontal ? lv_obj_get_style_pad_column(cont, LV_PART_MAIN)
11 : lv_obj_get_style_pad_row(cont, LV_PART_MAIN);
12 int32_t s = 0;
13 for(int32_t i = 0; i < total; i++) {
14 lv_obj_t * child = lv_obj_get_child(cont, i);
15 s += horizontal ? lv_obj_get_width(child) : lv_obj_get_height(child);
16 if(i < total - 1) s += gap;
17 }
18 if(horizontal) {
19 return s + lv_obj_get_style_pad_left(cont, LV_PART_MAIN) + lv_obj_get_style_pad_right(cont, LV_PART_MAIN);
20 }
21 return s + lv_obj_get_style_pad_top(cont, LV_PART_MAIN) + lv_obj_get_style_pad_bottom(cont, LV_PART_MAIN);
22}

24static void wrap_scroll_cb(lv_event_t * e)
25{
26 static bool adjusting = false;
27 if(adjusting) return;
28
29 lv_obj_t * cont = lv_event_get_current_target_obj(e);
30 bool horizontal = lv_obj_get_style_flex_flow(cont, LV_PART_MAIN) == LV_FLEX_FLOW_ROW;
31
32 int32_t scroll = horizontal ? lv_obj_get_scroll_x(cont) : lv_obj_get_scroll_y(cont);
33 int32_t view = horizontal ? lv_obj_get_width(cont) : lv_obj_get_height(cont);
34 int32_t content = content_size(cont, horizontal);
35 int32_t last = (int32_t)lv_obj_get_child_count(cont) - 1;
36
37 adjusting = true;
38 if(scroll <= 0) {
39 /* Reached the start: move the last child to the front. */
40 lv_obj_move_to_index(lv_obj_get_child(cont, last), 0);
41 if(horizontal) lv_obj_scroll_to_x(cont, scroll + ITEM_SIZE, LV_ANIM_OFF);
42 else lv_obj_scroll_to_y(cont, scroll + ITEM_SIZE, LV_ANIM_OFF);
43 }
44 else if(scroll > content - view) {
45 /* Reached the end: move the first child to the back. */
46 lv_obj_move_to_index(lv_obj_get_child(cont, 0), last);
47 if(horizontal) lv_obj_scroll_to_x(cont, scroll - ITEM_SIZE, LV_ANIM_OFF);
48 else lv_obj_scroll_to_y(cont, scroll - ITEM_SIZE, LV_ANIM_OFF);
49 }
50 adjusting = false;
51}
