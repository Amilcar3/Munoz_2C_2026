84static lv_cache_compare_res_t sdl_texture_cache_compare_cb(const cache_data_t * lhs, const cache_data_t * rhs)
85{
86 if(lhs == rhs) return 0;
87
88 if(lhs->w != rhs->w) {
89 return lhs->w > rhs->w ? 1 : -1;
90 }
91 if(lhs->h != rhs->h) {
92 return lhs->h > rhs->h ? 1 : -1;
93 }
94
95 if(lhs->draw_dsc == NULL || rhs->draw_dsc == NULL) {
96 if(lhs->draw_dsc == rhs->draw_dsc) return 0;
97 if(lhs->draw_dsc == NULL) return -1;
98 return 1;
99 }
100
101 uint32_t lhs_dsc_size = lhs->draw_dsc->dsc_size;
102 uint32_t rhs_dsc_size = rhs->draw_dsc->dsc_size;
103
104 if(lhs_dsc_size != rhs_dsc_size) {
105 return lhs_dsc_size > rhs_dsc_size ? 1 : -1;
106 }
107
108 const uint8_t * left_draw_dsc = (const uint8_t *)lhs->draw_dsc;
109 const uint8_t * right_draw_dsc = (const uint8_t *)rhs->draw_dsc;
110 left_draw_dsc += sizeof(lv_draw_dsc_base_t);
111 right_draw_dsc += sizeof(lv_draw_dsc_base_t);
112
113 int cmp_res = lv_memcmp(left_draw_dsc, right_draw_dsc, lhs->draw_dsc->dsc_size - sizeof(lv_draw_dsc_base_t));
114
115 if(cmp_res != 0) {
116 return cmp_res > 0 ? 1 : -1;
117 }
118
119 return 0;
120}

123{

143static int32_t dispatch(lv_draw_unit_t * draw_unit, lv_layer_t * layer)
144{
145 lv_draw_sdl_unit_t * draw_sdl_unit = (lv_draw_sdl_unit_t *) draw_unit;
146
147 /*Return immediately if it's busy with a draw task*/
148 if(draw_sdl_unit->task_act) return 0;
149
150 lv_draw_task_t * t = NULL;
151 t = lv_draw_get_available_task(layer, NULL, DRAW_UNIT_ID_SDL);
152 if(t == NULL) return -1;
153
154 lv_display_t * disp = lv_refr_get_disp_refreshing();
155 SDL_Texture * texture = layer_get_texture(layer);
156 if(layer != disp->layer_head && texture == NULL) {
157 void * buf = lv_draw_layer_alloc_buf(layer);
158 if(buf == NULL) {
159 t->state = LV_DRAW_TASK_STATE_FAILED;
160 return -1;
161 }
162
163 SDL_Renderer * renderer = lv_sdl_window_get_renderer(disp);
164 int32_t w = lv_area_get_width(&layer->buf_area);
165 int32_t h = lv_area_get_height(&layer->buf_area);
166 layer->user_data = SDL_CreateTexture(renderer, SDL_PIXELFORMAT_ARGB8888,
167 SDL_TEXTUREACCESS_TARGET, w, h);
168 }
169
170 t->state = LV_DRAW_TASK_STATE_IN_PROGRESS;
171 draw_sdl_unit->task_act = t;
172
173 execute_drawing(draw_sdl_unit);
174
175 draw_sdl_unit->task_act->state = LV_DRAW_TASK_STATE_FINISHED;
176 draw_sdl_unit->task_act = NULL;
177
178 /*The draw unit is free now. Request a new dispatching as it can get a new task*/
179 lv_draw_dispatch_request();
180 return 1;
181}

183static int32_t evaluate(lv_draw_unit_t * draw_unit, lv_draw_task_t * task)
184{
185 LV_UNUSED(draw_unit);
186
187 switch(task->type) {
188 case LV_DRAW_TASK_TYPE_FILL:
189 case LV_DRAW_TASK_TYPE_BORDER:
190 case LV_DRAW_TASK_TYPE_BOX_SHADOW:
191 case LV_DRAW_TASK_TYPE_LABEL:
192 case LV_DRAW_TASK_TYPE_IMAGE:
193 case LV_DRAW_TASK_TYPE_LAYER:
194 case LV_DRAW_TASK_TYPE_LINE:
195 case LV_DRAW_TASK_TYPE_ARC:
196 case LV_DRAW_TASK_TYPE_TRIANGLE:
197 break;
198 default:
199 return 0;
200 }
201
202 if(task->type == LV_DRAW_TASK_TYPE_IMAGE &&
203 ((lv_draw_image_dsc_t *)task->draw_dsc)->header.cf >= LV_COLOR_FORMAT_PROPRIETARY_START) {
204 return 0;
205 }
206
207 /*If not refreshing the display probably it's a canvas rendering
208 *which his not support in SDL as it's not a texture.*/
209 if(lv_refr_get_disp_refreshing() == NULL) return 0;
210
211 if(((lv_draw_dsc_base_t *)task->draw_dsc)->user_data == NULL) {
212 task->preference_score = 0;
213 task->preferred_draw_unit_id = DRAW_UNIT_ID_SDL;
214 }
215 return 0;
216}

218static bool draw_to_texture(lv_draw_sdl_unit_t * u, cache_data_t * cache_data)
219{
220 lv_draw_task_t * task = u->task_act;
221
222 lv_layer_t dest_layer;
223 lv_layer_init(&dest_layer);
224
225 int32_t texture_w = lv_area_get_width(&task->_real_area);
226 int32_t texture_h = lv_area_get_height(&task->_real_area);
227
228 if(!lv_draw_buf_reshape(&u->render_draw_buf, LV_COLOR_FORMAT_ARGB8888, texture_w, texture_h, LV_STRIDE_AUTO)) {
229 uint8_t * data = u->render_draw_buf.unaligned_data;
230 uint32_t data_size = LV_DRAW_BUF_SIZE(texture_w, texture_h, LV_COLOR_FORMAT_ARGB8888);
231 if(data == NULL) data = malloc(data_size);
232 else data = realloc(data, data_size);
233
234 LV_ASSERT_MALLOC(data);
235 lv_result_t init_result = lv_draw_buf_init(&u->render_draw_buf, texture_w, texture_h, LV_COLOR_FORMAT_ARGB8888,
236 LV_STRIDE_AUTO, data, data_size);
237 LV_ASSERT(init_result == LV_RESULT_OK);
238 }
239
240
241 dest_layer.draw_buf = &u->render_draw_buf;
242 dest_layer.color_format = LV_COLOR_FORMAT_ARGB8888;
243 dest_layer.buf_area = task->_real_area;
244 dest_layer._clip_area = task->_real_area;
245 dest_layer.phy_clip_area = task->_real_area;
246 lv_memzero(u->render_draw_buf.data, lv_area_get_size(&task->_real_area) * 4);
247
248 lv_display_t * disp = lv_refr_get_disp_refreshing();
249
250 lv_obj_t * obj = ((lv_draw_dsc_base_t *)task->draw_dsc)->obj;
251 bool original_send_draw_task_event = false;
252 if(obj) {
253 original_send_draw_task_event = lv_obj_is_send_draw_task_events(obj);
254 lv_obj_set_send_draw_task_events(obj, false);
255 }
256
257 switch(task->type) {
258 case LV_DRAW_TASK_TYPE_FILL: {
259 lv_draw_fill_dsc_t * fill_dsc = task->draw_dsc;
260 lv_draw_rect_dsc_t rect_dsc;
261 lv_draw_rect_dsc_init(&rect_dsc);
262 rect_dsc.base.user_data = lv_sdl_window_get_renderer(disp);
263 rect_dsc.bg_color = fill_dsc->color;
264 rect_dsc.bg_grad = fill_dsc->grad;
265 rect_dsc.radius = fill_dsc->radius;
266 rect_dsc.bg_opa = fill_dsc->opa;
267
268 lv_draw_rect(&dest_layer, &rect_dsc, &task->area);
269 }
270 break;
271 case LV_DRAW_TASK_TYPE_BORDER: {
272 lv_draw_border_dsc_t * border_dsc = task->draw_dsc;
273 lv_draw_rect_dsc_t rect_dsc;
274 lv_draw_rect_dsc_init(&rect_dsc);
275 rect_dsc.base.user_data = lv_sdl_window_get_renderer(disp);
276 rect_dsc.bg_opa = LV_OPA_TRANSP;
277 rect_dsc.radius = border_dsc->radius;
278 rect_dsc.border_color = border_dsc->color;
279 rect_dsc.border_opa = border_dsc->opa;
280 rect_dsc.border_side = border_dsc->side;
281 rect_dsc.border_width = border_dsc->width;
282 lv_draw_rect(&dest_layer, &rect_dsc, &task->area);
283 break;
284 }
285 case LV_DRAW_TASK_TYPE_BOX_SHADOW: {
286 lv_draw_box_shadow_dsc_t * box_shadow_dsc = task->draw_dsc;
287 lv_draw_rect_dsc_t rect_dsc;
288 lv_draw_rect_dsc_init(&rect_dsc);
289 rect_dsc.base.user_data = lv_sdl_window_get_renderer(disp);
290 rect_dsc.bg_opa = LV_OPA_0;
291 rect_dsc.radius = box_shadow_dsc->radius;
292 rect_dsc.bg_color = box_shadow_dsc->color;
293 rect_dsc.shadow_opa = box_shadow_dsc->opa;
294 rect_dsc.shadow_width = box_shadow_dsc->width;
295 rect_dsc.shadow_spread = box_shadow_dsc->spread;
296 rect_dsc.shadow_offset_x = box_shadow_dsc->ofs_x;
297 rect_dsc.shadow_offset_y = box_shadow_dsc->ofs_y;
298 lv_draw_rect(&dest_layer, &rect_dsc, &task->area);
299 break;
300 }
301 case LV_DRAW_TASK_TYPE_LABEL: {
302 lv_draw_label_dsc_t label_dsc;
303 lv_memcpy(&label_dsc, task->draw_dsc, sizeof(label_dsc));
304 label_dsc.base.user_data = lv_sdl_window_get_renderer(disp);
305 lv_draw_label(&dest_layer, &label_dsc, &task->area);
306 }
307 break;
308 case LV_DRAW_TASK_TYPE_ARC: {
309 lv_draw_arc_dsc_t arc_dsc;
310 lv_memcpy(&arc_dsc, task->draw_dsc, sizeof(arc_dsc));
311 arc_dsc.base.user_data = lv_sdl_window_get_renderer(disp);
312 lv_draw_arc(&dest_layer, &arc_dsc);
313 }
314 break;
315 case LV_DRAW_TASK_TYPE_LINE: {
316 lv_draw_line_dsc_t line_dsc;
317 lv_memcpy(&line_dsc, task->draw_dsc, sizeof(line_dsc));
318 line_dsc.base.user_data = lv_sdl_window_get_renderer(disp);
319 lv_draw_line(&dest_layer, &line_dsc);
320 }
321 break;
322 case LV_DRAW_TASK_TYPE_TRIANGLE: {
323 lv_draw_triangle_dsc_t triangle_dsc;
324 lv_memcpy(&triangle_dsc, task->draw_dsc, sizeof(triangle_dsc));
325 triangle_dsc.base.user_data = lv_sdl_window_get_renderer(disp);
326 lv_draw_triangle(&dest_layer, &triangle_dsc);
327 }
328 break;
329 case LV_DRAW_TASK_TYPE_IMAGE: {
330 lv_draw_image_dsc_t image_dsc;
331 lv_memcpy(&image_dsc, task->draw_dsc, sizeof(image_dsc));
332 image_dsc.base.user_data = lv_sdl_window_get_renderer(disp);
333 lv_draw_image(&dest_layer, &image_dsc, &task->area);
334 break;
335 }
336 default:
337 /*Keep in sync with evaluate(): every type it claims must have a case above*/
338 LV_UNREACHABLE();
339 }
340
341 while(dest_layer.draw_task_head) {
342 lv_draw_dispatch_layer(disp, &dest_layer);
343 if(dest_layer.draw_task_head) {
344 lv_draw_dispatch_wait_for_request();
345 }
346 }
347
348 SDL_Texture * texture = SDL_CreateTexture(lv_sdl_window_get_renderer(disp), SDL_PIXELFORMAT_ARGB8888,
349 SDL_TEXTUREACCESS_STATIC, texture_w, texture_h);
350 SDL_SetTextureBlendMode(texture, SDL_BLENDMODE_BLEND);
351 SDL_UpdateTexture(texture, NULL, u->render_draw_buf.data, texture_w * 4);
352
353 lv_draw_dsc_base_t * base_dsc = task->draw_dsc;
354
355 cache_data->draw_dsc = lv_malloc(base_dsc->dsc_size);
356 lv_memcpy((void *)cache_data->draw_dsc, base_dsc, base_dsc->dsc_size);
357 cache_data->dsc_owned = true;
358 cache_data->w = texture_w;
359 cache_data->h = texture_h;
360 cache_data->texture = texture;
361
362 if(obj) {
363 lv_obj_set_send_draw_task_events(obj, original_send_draw_task_event);
364 }
365
366 return true;
367}

369static void blend_texture_layer(lv_draw_task_t * t)
370{
371 lv_display_t * disp = lv_refr_get_disp_refreshing();
372 SDL_Renderer * renderer = lv_sdl_window_get_renderer(disp);
373
374 SDL_Rect clip_rect;
375 clip_rect.x = t->clip_area.x1;
376 clip_rect.y = t->clip_area.y1;
377 clip_rect.w = lv_area_get_width(&t->clip_area);
378 clip_rect.h = lv_area_get_height(&t->clip_area);
379
380 lv_draw_image_dsc_t * draw_dsc = t->draw_dsc;
381 SDL_Rect rect;
382 rect.w = (lv_area_get_width(&t->area) * draw_dsc->scale_x) / 256;
383 rect.h = (lv_area_get_height(&t->area) * draw_dsc->scale_y) / 256;
384
385 rect.x = -draw_dsc->pivot.x;
386 rect.y = -draw_dsc->pivot.y;
387 rect.x = (rect.x * draw_dsc->scale_x) / 256;
388 rect.y = (rect.y * draw_dsc->scale_y) / 256;
389 rect.x += t->area.x1 + draw_dsc->pivot.x;
390 rect.y += t->area.y1 + draw_dsc->pivot.y;
391
392 lv_layer_t * src_layer = (lv_layer_t *)draw_dsc->src;
393 SDL_Texture * src_texture = layer_get_texture(src_layer);
394
395 SDL_SetTextureAlphaMod(src_texture, draw_dsc->opa);
396 SDL_SetTextureBlendMode(src_texture, SDL_BLENDMODE_BLEND);
397 SDL_SetRenderTarget(renderer, layer_get_texture(t->target_layer));
398 SDL_RenderSetClipRect(renderer, &clip_rect);
399
400 SDL_Point center = {draw_dsc->pivot.x, draw_dsc->pivot.y};
401 SDL_RenderCopyEx(renderer, src_texture, NULL, &rect, draw_dsc->rotation / 10, &center, SDL_FLIP_NONE);
402
403 SDL_DestroyTexture(src_texture);
404 SDL_RenderSetClipRect(renderer, NULL);
405}

407static void draw_from_cached_texture(lv_draw_sdl_unit_t * u)
408{
409 lv_draw_task_t * t = u->task_act;
410
411 cache_data_t data_to_find;
412 data_to_find.draw_dsc = (lv_draw_dsc_base_t *)t->draw_dsc;
413
414 data_to_find.w = lv_area_get_width(&t->_real_area);
415 data_to_find.h = lv_area_get_height(&t->_real_area);
416 data_to_find.texture = NULL;
417 data_to_find.dsc_owned = false;
418
419 /*user_data stores the renderer to differentiate it from SW rendered tasks.
420 *However the cached texture is independent from the renderer so use NULL user_data*/
421 void * user_data_saved = data_to_find.draw_dsc->user_data;
422 data_to_find.draw_dsc->user_data = NULL;
423
424 /*Absolute coordinates are different for the same draw_dsc on a different position.
425 *So make everything relative to 0;0 before caching*/
426 lv_area_t a = t->area;
427 if(t->type == LV_DRAW_TASK_TYPE_IMAGE) {
428 lv_draw_image_dsc_t * img_dsc = (lv_draw_image_dsc_t *)data_to_find.draw_dsc;
429 lv_area_move(&img_dsc->image_area, -t->area.x1, -t->area.y1);
430 }
431 else if(t->type == LV_DRAW_TASK_TYPE_TRIANGLE) {
432 lv_draw_triangle_dsc_t * tri_dsc = (lv_draw_triangle_dsc_t *)data_to_find.draw_dsc;
433 tri_dsc->p[0].x -= t->area.x1;
434 tri_dsc->p[0].y -= t->area.y1;
435 tri_dsc->p[1].x -= t->area.x1;
436 tri_dsc->p[1].y -= t->area.y1;
437 tri_dsc->p[2].x -= t->area.x1;
438 tri_dsc->p[2].y -= t->area.y1;
439 }
440 else if(t->type == LV_DRAW_TASK_TYPE_LINE) {
441 lv_draw_line_dsc_t * line_dsc = (lv_draw_line_dsc_t *)data_to_find.draw_dsc;
442 line_dsc->p1.x -= t->area.x1;
443 line_dsc->p1.y -= t->area.y1;
444 line_dsc->p2.x -= t->area.x1;
445 line_dsc->p2.y -= t->area.y1;
446 }
447 else if(t->type == LV_DRAW_TASK_TYPE_ARC) {
448 lv_draw_arc_dsc_t * arc_dsc = (lv_draw_arc_dsc_t *)data_to_find.draw_dsc;
449 arc_dsc->center.x -= t->area.x1;
450 arc_dsc->center.y -= t->area.y1;
451 }
452
453 lv_area_move(&t->area, -a.x1, -a.y1);
454 lv_area_move(&t->_real_area, -a.x1, -a.y1);
455
456 lv_cache_entry_t * entry_cached = lv_cache_acquire_or_create(u->texture_cache, &data_to_find, u);
457
458 lv_area_move(&t->area, a.x1, a.y1);
459 lv_area_move(&t->_real_area, a.x1, a.y1);
460
461 if(!entry_cached) {
462 return;
463 }
464
465 data_to_find.draw_dsc->user_data = user_data_saved;
466
467 cache_data_t * data_cached = lv_cache_entry_get_data(entry_cached);
468 SDL_Texture * texture = data_cached->texture;
469 lv_display_t * disp = lv_refr_get_disp_refreshing();
470 SDL_Renderer * renderer = lv_sdl_window_get_renderer(disp);
471
472 lv_layer_t * dest_layer = t->target_layer;
473 SDL_Rect clip_rect;
474 clip_rect.x = t->clip_area.x1 - dest_layer->buf_area.x1;
475 clip_rect.y = t->clip_area.y1 - dest_layer->buf_area.y1;
476 clip_rect.w = lv_area_get_width(&t->clip_area);
477 clip_rect.h = lv_area_get_height(&t->clip_area);
478
479 SDL_Rect rect;
480
481 SDL_SetRenderTarget(renderer, layer_get_texture(dest_layer));
482
483 rect.x = t->_real_area.x1 - dest_layer->buf_area.x1;
484 rect.y = t->_real_area.y1 - dest_layer->buf_area.y1;
485 rect.w = data_cached->w;
486 rect.h = data_cached->h;
487
488 SDL_RenderSetClipRect(renderer, &clip_rect);
489 SDL_RenderCopy(renderer, texture, NULL, &rect);
490
491 SDL_RenderSetClipRect(renderer, NULL);
492
493 lv_cache_release(u->texture_cache, entry_cached, u);
494
495 /*Do not cache non static (const) texts as the text's pointer can be freed/reallocated
496 *at any time resulting in a wild pointer in the cached draw dsc. */
497 if(t->type == LV_DRAW_TASK_TYPE_LABEL) {
498 lv_draw_label_dsc_t * label_dsc = t->draw_dsc;
499 if(!label_dsc->text_static) {
500 lv_cache_drop(u->texture_cache, &data_to_find, NULL);
501 }
502 }
503 /*Do not cache lines rendered from points at dsc->points will be freed*/
504 else if(t->type == LV_DRAW_TASK_TYPE_LINE) {
505 lv_draw_line_dsc_t * line_dsc = t->draw_dsc;
506 if(line_dsc->points) {
507 lv_cache_drop(u->texture_cache, &data_to_find, u);
508 }
509 }
510}

512static void execute_drawing(lv_draw_sdl_unit_t * u)
513{
514 lv_draw_task_t * t = u->task_act;
515
516 if(t->type == LV_DRAW_TASK_TYPE_FILL) {
517 lv_draw_fill_dsc_t * fill_dsc = t->draw_dsc;
518 if(fill_dsc->radius == 0 && fill_dsc->grad.dir == LV_GRAD_DIR_NONE) {
519 SDL_Rect rect;
520 lv_layer_t * layer = t->target_layer;
521 lv_area_t fill_area;
522 lv_area_intersect(&fill_area, &t->area, &t->clip_area);
523
524 rect.x = fill_area.x1 - layer->buf_area.x1;
525 rect.y = fill_area.y1 - layer->buf_area.y1;
526 rect.w = lv_area_get_width(&fill_area);
