47void lv_draw_sw_line(lv_draw_task_t * t, const lv_draw_line_dsc_t * dsc)
48{
49 if(dsc->width == 0) return;
50 if(dsc->opa <= LV_OPA_MIN) return;
51
52 if(dsc->p1.x == dsc->p2.x && dsc->p1.y == dsc->p2.y) return;
53
54 lv_area_t clip_line;
55 clip_line.x1 = (int32_t)LV_MIN(dsc->p1.x, dsc->p2.x) - dsc->width / 2;
56 clip_line.x2 = (int32_t)LV_MAX(dsc->p1.x, dsc->p2.x) + dsc->width / 2;
57 clip_line.y1 = (int32_t)LV_MIN(dsc->p1.y, dsc->p2.y) - dsc->width / 2;
58 clip_line.y2 = (int32_t)LV_MAX(dsc->p1.y, dsc->p2.y) + dsc->width / 2;
59
60 bool is_common;
61 is_common = lv_area_intersect(&clip_line, &clip_line, &t->clip_area);
62 if(!is_common) return;
63
64 LV_PROFILER_DRAW_BEGIN;
65 if((int32_t)dsc->p1.y == (int32_t)dsc->p2.y) draw_line_hor(t, dsc);
66 else if((int32_t)dsc->p1.x == (int32_t)dsc->p2.x) draw_line_ver(t, dsc);
67 else draw_line_skew(t, dsc);
68
69 if(dsc->round_end || dsc->round_start) {
70 lv_draw_fill_dsc_t cir_dsc;
71 lv_draw_fill_dsc_init(&cir_dsc);
72 cir_dsc.color = dsc->color;
73 cir_dsc.radius = LV_RADIUS_CIRCLE;
74 cir_dsc.opa = dsc->opa;
75
76 int32_t r = (dsc->width >> 1);
77 int32_t r_corr = (dsc->width & 1) ? 0 : 1;
78 lv_area_t cir_area;
79
80 if(dsc->round_start) {
81 cir_area.x1 = (int32_t)dsc->p1.x - r;
82 cir_area.y1 = (int32_t)dsc->p1.y - r;
83 cir_area.x2 = (int32_t)dsc->p1.x + r - r_corr;
84 cir_area.y2 = (int32_t)dsc->p1.y + r - r_corr ;
85 lv_draw_sw_fill(t, &cir_dsc, &cir_area);
86 }
87
88 if(dsc->round_end) {
89 cir_area.x1 = (int32_t)dsc->p2.x - r;
90 cir_area.y1 = (int32_t)dsc->p2.y - r;
91 cir_area.x2 = (int32_t)dsc->p2.x + r - r_corr;
92 cir_area.y2 = (int32_t)dsc->p2.y + r - r_corr ;
93 lv_draw_sw_fill(t, &cir_dsc, &cir_area);
94 }
95 }
96 LV_PROFILER_DRAW_END;
97}

102static void LV_ATTRIBUTE_FAST_MEM draw_line_hor(lv_draw_task_t * t, const lv_draw_line_dsc_t * dsc)
103{
104 int32_t w = dsc->width - 1;
105 int32_t w_half0 = w >> 1;
106 int32_t w_half1 = w_half0 + (w & 0x1); /*Compensate rounding error*/
107
108 lv_area_t blend_area;
109 blend_area.x1 = (int32_t)LV_MIN(dsc->p1.x, dsc->p2.x);
110 blend_area.x2 = (int32_t)LV_MAX(dsc->p1.x, dsc->p2.x) - 1;
111 blend_area.y1 = (int32_t)dsc->p1.y - w_half1;
112 blend_area.y2 = (int32_t)dsc->p1.y + w_half0;
113
114 bool is_common;
115 is_common = lv_area_intersect(&blend_area, &blend_area, &t->clip_area);
116 if(!is_common) return;
117
118 bool dashed = dsc->dash_gap && dsc->dash_width;
119
120 lv_draw_sw_blend_dsc_t blend_dsc;
121 lv_memzero(&blend_dsc, sizeof(blend_dsc));
122 blend_dsc.blend_area = &blend_area;
123 blend_dsc.color = dsc->color;
124 blend_dsc.opa = dsc->opa;
125
126 /*If there is no mask then simply draw a rectangle*/
127 if(!dashed) {
128 lv_draw_sw_blend(t, &blend_dsc);
129 }
130#if LV_DRAW_SW_COMPLEX
131 /*If there other mask apply it*/
132 else {
133
134 int32_t blend_area_w = lv_area_get_width(&blend_area);
135
136 int32_t y2 = blend_area.y2;
137 blend_area.y2 = blend_area.y1;
138
139 int32_t dash_start = blend_area.x1 % (dsc->dash_gap + dsc->dash_width);
140
141 lv_opa_t * mask_buf = lv_malloc(blend_area_w);
142 blend_dsc.mask_buf = mask_buf;
143 blend_dsc.mask_area = &blend_area;
144 blend_dsc.mask_res = LV_DRAW_SW_MASK_RES_CHANGED;
145 int32_t h;
146 for(h = blend_area.y1; h <= y2; h++) {
147 lv_memset(mask_buf, 0xff, blend_area_w);
148
149 int32_t dash_cnt = dash_start;
150 int32_t i;
151 for(i = 0; i < blend_area_w;) {
152 if(dash_cnt < dsc->dash_width) {
153 int16_t diff = dsc->dash_width - dash_cnt;
154 i += diff;
155 dash_cnt += diff;
156 }
157 else {
158 mask_buf[i] = 0x00;
159 i++;
160 dash_cnt++;
161 }
162 if(dash_cnt >= dsc->dash_gap + dsc->dash_width) {
163 dash_cnt = 0;
164 }
165
166 blend_dsc.mask_res = LV_DRAW_SW_MASK_RES_CHANGED;
167 }
168
169 lv_draw_sw_blend(t, &blend_dsc);
170
171 blend_area.y1++;
172 blend_area.y2++;
173 }
174 lv_free(mask_buf);
175 }
176#endif /*LV_DRAW_SW_COMPLEX*/
177}

201 blend_dsc.opa = dsc->opa;

236 dash_cnt = 0;

240 lv_draw_sw_blend(t, &blend_dsc);
241
242 blend_area.y1++;
243 blend_area.y2++;
244 }
245 lv_free(mask_buf);

250static void LV_ATTRIBUTE_FAST_MEM draw_line_skew(lv_draw_task_t * t, const lv_draw_line_dsc_t * dsc)
251{
252#if LV_DRAW_SW_COMPLEX
253 /*Keep the great y in p1*/
254 lv_point_t p1;
255 lv_point_t p2;
256 if(dsc->p1.y < dsc->p2.y) {
257 p1 = lv_point_from_precise(&dsc->p1);
258 p2 = lv_point_from_precise(&dsc->p2);
259 }
260 else {
261 p1 = lv_point_from_precise(&dsc->p2);
262 p2 = lv_point_from_precise(&dsc->p1);
263 }
264
265 int32_t xdiff = p2.x - p1.x;
266 int32_t ydiff = p2.y - p1.y;
267 bool flat = LV_ABS(xdiff) > LV_ABS(ydiff);
268
269 static const uint8_t wcorr[] = {
270 128, 128, 128, 129, 129, 130, 130, 131,
271 132, 133, 134, 135, 137, 138, 140, 141,
272 143, 145, 147, 149, 151, 153, 155, 158,
273 160, 162, 165, 167, 170, 173, 175, 178,
274 181,
275 };
276
277 int32_t w = dsc->width;
278 int32_t wcorr_i = 0;
279 if(flat) wcorr_i = (LV_ABS(ydiff) << 5) / LV_ABS(xdiff);
280 else wcorr_i = (LV_ABS(xdiff) << 5) / LV_ABS(ydiff);
281
282 w = (w * wcorr[wcorr_i] + 63) >> 7; /*+ 63 for rounding*/
283 int32_t w_half0 = w >> 1;
284 int32_t w_half1 = w_half0 + (w & 0x1); /*Compensate rounding error*/
285
286 lv_area_t blend_area;
287 blend_area.x1 = LV_MIN(p1.x, p2.x) - w;
288 blend_area.x2 = LV_MAX(p1.x, p2.x) + w;
289 blend_area.y1 = LV_MIN(p1.y, p2.y) - w;
290 blend_area.y2 = LV_MAX(p1.y, p2.y) + w;
291
292 /*Get the union of `coords` and `clip`*/
293 /*`clip` is already truncated to the `draw_buf` size
294 *in 'lv_refr_area' function*/
295 bool is_common = lv_area_intersect(&blend_area, &blend_area, &t->clip_area);
296 if(is_common == false) return;
297
298 lv_draw_sw_mask_line_param_t mask_left_param;
299 lv_draw_sw_mask_line_param_t mask_right_param;
300 lv_draw_sw_mask_line_param_t mask_top_param;
301 lv_draw_sw_mask_line_param_t mask_bottom_param;
302
303 void * masks[5] = {&mask_left_param, & mask_right_param, NULL, NULL, NULL};
304
305 if(flat) {
306 if(xdiff > 0) {
307 lv_draw_sw_mask_line_points_init(&mask_left_param, p1.x, p1.y - w_half0, p2.x, p2.y - w_half0,
308 LV_DRAW_SW_MASK_LINE_SIDE_LEFT);
309 lv_draw_sw_mask_line_points_init(&mask_right_param, p1.x, p1.y + w_half1, p2.x, p2.y + w_half1,
310 LV_DRAW_SW_MASK_LINE_SIDE_RIGHT);
311 }
312 else {
313 lv_draw_sw_mask_line_points_init(&mask_left_param, p1.x, p1.y + w_half1, p2.x, p2.y + w_half1,
314 LV_DRAW_SW_MASK_LINE_SIDE_LEFT);
315 lv_draw_sw_mask_line_points_init(&mask_right_param, p1.x, p1.y - w_half0, p2.x, p2.y - w_half0,
316 LV_DRAW_SW_MASK_LINE_SIDE_RIGHT);
317 }
318 }
319 else {
320 lv_draw_sw_mask_line_points_init(&mask_left_param, p1.x + w_half1, p1.y, p2.x + w_half1, p2.y,
321 LV_DRAW_SW_MASK_LINE_SIDE_LEFT);
322 lv_draw_sw_mask_line_points_init(&mask_right_param, p1.x - w_half0, p1.y, p2.x - w_half0, p2.y,
323 LV_DRAW_SW_MASK_LINE_SIDE_RIGHT);
324
325 }
326
327 /*Use the normal vector for the endings*/
328
329 if(!dsc->raw_end) {
330 lv_draw_sw_mask_line_points_init(&mask_top_param, p1.x, p1.y, p1.x - ydiff, p1.y + xdiff,
331 LV_DRAW_SW_MASK_LINE_SIDE_BOTTOM);
332 lv_draw_sw_mask_line_points_init(&mask_bottom_param, p2.x, p2.y, p2.x - ydiff, p2.y + xdiff,
333 LV_DRAW_SW_MASK_LINE_SIDE_TOP);
334 masks[2] = &mask_top_param;
335 masks[3] = &mask_bottom_param;
336 }
337
338
339 /*Draw the background line by line*/
340 int32_t h;
341 size_t mask_buf_size = LV_MIN((int32_t)lv_area_get_size(&blend_area), lv_area_get_width(&blend_area));
342 lv_opa_t * mask_buf = lv_malloc(mask_buf_size);
343
344 /*The real draw area is around the line.
345 *It's easy to calculate with steep lines, but the area can be very wide with very flat lines.
346 *So deal with it only with steep lines.*/
347 size_t draw_area_w = LV_MIN((size_t)lv_area_get_width(&blend_area), mask_buf_size);
348
349 int32_t y2 = blend_area.y2;
350 blend_area.y2 = blend_area.y1;
351
352 uint32_t mask_p = 0;
353 lv_memset(mask_buf, 0xff, mask_buf_size);
354
355 lv_draw_sw_blend_dsc_t blend_dsc;
356 lv_memzero(&blend_dsc, sizeof(blend_dsc));
357 blend_dsc.blend_area = &blend_area;
358 blend_dsc.color = dsc->color;
359 blend_dsc.opa = dsc->opa;
360 blend_dsc.mask_buf = mask_buf;
361 blend_dsc.mask_area = &blend_area;
362
363 /*Fill the first row with 'color'*/
364 for(h = blend_area.y1; h <= y2; h++) {
365 blend_dsc.mask_res = lv_draw_sw_mask_apply(masks, &mask_buf[mask_p], blend_area.x1, h, draw_area_w);
366 if(blend_dsc.mask_res == LV_DRAW_SW_MASK_RES_TRANSP) {
367 lv_memzero(&mask_buf[mask_p], draw_area_w);
368 }
369
370 mask_p += draw_area_w;
371 if((uint32_t) mask_p + draw_area_w < mask_buf_size) {
372 blend_area.y2 ++;
373 }
374 else {
375 blend_dsc.mask_res = LV_DRAW_SW_MASK_RES_CHANGED;
376 lv_draw_sw_blend(t, &blend_dsc);
377
378 blend_area.y1 = blend_area.y2 + 1;
379 blend_area.y2 = blend_area.y1;
380 mask_p = 0;
381 lv_memset(mask_buf, 0xff, mask_buf_size);
382 }
383 }
384
385 /*Flush the last part*/
386 if(blend_area.y1 != blend_area.y2) {
387 blend_area.y2--;
388 blend_dsc.mask_res = LV_DRAW_SW_MASK_RES_CHANGED;
389 lv_draw_sw_blend(t, &blend_dsc);
390 }
391
392 lv_free(mask_buf);
393
394 lv_draw_sw_mask_free_param(&mask_left_param);
395 lv_draw_sw_mask_free_param(&mask_right_param);
396 if(!dsc->raw_end) {
397 lv_draw_sw_mask_free_param(&mask_top_param);
398 lv_draw_sw_mask_free_param(&mask_bottom_param);
399 }
400#else
401 LV_UNUSED(t);
402 LV_UNUSED(dsc);
403 LV_LOG_WARN("Can't draw skewed line with LV_DRAW_SW_COMPLEX == 0");
404#endif /*LV_DRAW_SW_COMPLEX*/
405}

407#endif /*LV_USE_DRAW_SW*/
