311void lv_label_get_letter_pos(const lv_obj_t * obj, uint32_t char_id, lv_point_t * pos)
312{
313 LV_CHECK_OBJ(obj, MY_CLASS, return);
314 LV_CHECK_ARG(pos != NULL, return);
315
316 lv_label_t * label = (lv_label_t *)obj;
317 const char * txt = lv_label_get_text(obj);
318 const lv_text_align_t align = lv_obj_calculate_style_text_align_internal(obj, LV_PART_MAIN, txt);
319
320 if(txt[0] == '\0') {
321 pos->y = 0;
322 switch(align) {
323 case LV_TEXT_ALIGN_LEFT:
324 pos->x = 0;
325 break;
326 case LV_TEXT_ALIGN_RIGHT:
327 pos->x = lv_obj_get_content_width(obj);
328 break;
329 case LV_TEXT_ALIGN_CENTER:
330 pos->x = lv_obj_get_content_width(obj) / 2;
331 break;
332 default:
333 pos->x = 0;
334 break;
335 }
336 return;
337 }
338
339
340 const uint32_t byte_id = lv_text_encoded_get_byte_id(txt, char_id);
341 /*Search the line of the index letter*/
342 lv_text_attributes_t attributes = {0};
343 attributes.text_flags = get_label_flags(label);
344 attributes.line_space = lv_obj_get_style_text_line_space_internal(obj, LV_PART_MAIN);
345 attributes.letter_space = lv_obj_get_style_text_letter_space_internal(obj, LV_PART_MAIN);
346
347 const lv_font_t * font = lv_obj_get_style_text_font_internal(obj, LV_PART_MAIN);
348 const int32_t letter_height = lv_font_get_line_height_internal(font);
349
350 lv_area_t txt_coords;
351 lv_obj_get_content_coords(obj, &txt_coords);
352 const int32_t max_h = lv_area_get_height(&txt_coords);
353
354 attributes.max_width = lv_area_get_width(&txt_coords);
355
356 int32_t y = 0;
357 uint32_t line_start = 0;
358 uint32_t new_line_start = 0;
359 while(txt[new_line_start] != '\0') {
360 bool last_line = y + letter_height + attributes.line_space + letter_height > max_h;
361 if(last_line && label->long_mode == LV_LABEL_LONG_MODE_DOTS) attributes.text_flags |= LV_TEXT_FLAG_BREAK_ALL;
362
363 new_line_start += lv_text_get_next_line(&txt[line_start], LV_TEXT_LEN_MAX, font, NULL, &attributes);
364
365 if(byte_id < new_line_start || txt[new_line_start] == '\0')
366 break; /*The line of 'index' letter begins at 'line_start'*/
367
368 y += letter_height + attributes.line_space;
369 line_start = new_line_start;
370 }
371
372 /*If the last character is line break then go to the next line*/
373 if(byte_id > 0) {
374 if((txt[byte_id - 1] == '\n' || txt[byte_id - 1] == '\r') && txt[byte_id] == '\0') {
375 y += letter_height + attributes.line_space;
376 line_start = byte_id;
377 }
378 }
379
380 const char * bidi_txt;
381 uint32_t visual_byte_pos;
382#if LV_USE_BIDI
383 lv_base_dir_t base_dir = lv_obj_get_style_base_dir_internal(obj, LV_PART_MAIN);
384 if(base_dir == LV_BASE_DIR_AUTO) base_dir = lv_bidi_detect_base_dir(txt);
385
386 char * mutable_bidi_txt = NULL;
387 /*Handle Bidi*/
388 if(new_line_start == byte_id) {
389 visual_byte_pos = base_dir == LV_BASE_DIR_RTL ? 0 : byte_id - line_start;
390 bidi_txt = &txt[line_start];
391 }
392 else {
393 uint32_t line_char_id = lv_text_encoded_get_char_id(&txt[line_start], byte_id - line_start);
394
395 bool is_rtl;
396 uint32_t visual_char_pos = lv_bidi_get_visual_pos(&txt[line_start], &mutable_bidi_txt, new_line_start - line_start,
397 base_dir, line_char_id, &is_rtl);
398 bidi_txt = mutable_bidi_txt;
399 if(is_rtl) visual_char_pos++;
400
401 visual_byte_pos = lv_text_encoded_get_byte_id(bidi_txt, visual_char_pos);
402 }
403#else
404 bidi_txt = &txt[line_start];
405 visual_byte_pos = byte_id - line_start;
406#endif
407
408 /*Calculate the x coordinate*/
409 int32_t x = lv_text_get_width(bidi_txt, visual_byte_pos, font, &attributes);
410 if(char_id != line_start) x += attributes.letter_space;
411
412 uint32_t length = new_line_start - line_start;
413 calculate_x_coordinate(&x, align, bidi_txt, length, font, &txt_coords, &attributes);
414 pos->x = x;
415 pos->y = y;
416
417 lv_text_leading_trim_t leading_trim_mode =
418 lv_obj_get_style_text_leading_trim_internal(obj, LV_PART_MAIN);
419 if(leading_trim_mode != LV_TEXT_LEADING_TRIM_NONE) {
420 pos->y -= lv_font_get_top_trim_internal(font, leading_trim_mode);
421 }
422
423#if LV_USE_BIDI
424 if(mutable_bidi_txt) lv_free(mutable_bidi_txt);
425#endif
426}

428uint32_t lv_label_get_letter_on(const lv_obj_t * obj, lv_point_t * pos_in, bool bidi)
429{
430 LV_UNUSED(bidi);
431 LV_CHECK_OBJ(obj, MY_CLASS, return 0);
432 LV_CHECK_ARG(pos_in != NULL, return 0);
433 lv_label_t * label = (lv_label_t *)obj;
434
435 lv_point_t pos;
436 pos.x = pos_in->x - lv_obj_get_style_pad_left_internal(obj, LV_PART_MAIN);
437 pos.y = pos_in->y - lv_obj_get_style_pad_top_internal(obj, LV_PART_MAIN);
438
439 lv_area_t txt_coords;
440 lv_obj_get_content_coords(obj, &txt_coords);
441 const char * txt = lv_label_get_text(obj);
442 uint32_t line_start = 0;
443 uint32_t new_line_start = 0;
444 int32_t max_h = lv_area_get_height(&txt_coords);
445 const lv_font_t * font = lv_obj_get_style_text_font_internal(obj, LV_PART_MAIN);
446
447 lv_text_leading_trim_t leading_trim_mode =
448 lv_obj_get_style_text_leading_trim_internal(obj, LV_PART_MAIN);
449 if(leading_trim_mode != LV_TEXT_LEADING_TRIM_NONE) {
450 pos.y += lv_font_get_top_trim_internal(font, leading_trim_mode);
451 }
452
453 const int32_t letter_height = lv_font_get_line_height_internal(font);
454 int32_t y = 0;
455
456 lv_text_attributes_t attributes = {0};
457 attributes.letter_space = lv_obj_get_style_text_letter_space_internal(obj, LV_PART_MAIN);
458 attributes.line_space = lv_obj_get_style_text_line_space_internal(obj, LV_PART_MAIN);
459 attributes.text_flags = get_label_flags(label);
460 attributes.max_width = lv_area_get_width(&txt_coords);
461
462 /*Search the line of the index letter*/;
463 while(txt[line_start] != '\0') {
464 /*If dots will be shown, break the last visible line anywhere,
465 *not only at word boundaries.*/
466 bool last_line = y + letter_height + attributes.line_space + letter_height > max_h;
467 if(last_line && label->long_mode == LV_LABEL_LONG_MODE_DOTS) attributes.text_flags |= LV_TEXT_FLAG_BREAK_ALL;
468
469 new_line_start += lv_text_get_next_line(&txt[line_start], LV_TEXT_LEN_MAX, font, NULL, &attributes);
470
471 if(pos.y <= y + letter_height) {
472 /*The line is found (stored in 'line_start')*/
473 /*Include the NULL terminator in the last line*/
474 uint32_t tmp = new_line_start;
475 uint32_t letter;
476 letter = lv_text_encoded_prev(txt, &tmp);
477 if(letter != '\n' && txt[new_line_start] == '\0') new_line_start++;
478 break;
479 }
480 y += letter_height + attributes.line_space;
481
482 line_start = new_line_start;
483 }
484
485 char * bidi_txt;
486
487#if LV_USE_BIDI
488 uint32_t txt_len = 0;
489 if(bidi) {
490 bidi_txt = lv_malloc(new_line_start - line_start + 1);
491 txt_len = new_line_start - line_start;
492 if(new_line_start > 0 && txt[new_line_start - 1] == '\0' && txt_len > 0) txt_len--;
493 lv_bidi_process_paragraph(txt + line_start, bidi_txt, txt_len, lv_obj_get_style_base_dir_internal(obj, LV_PART_MAIN),
494 NULL, 0);
495 }
496 else
497#endif
498 {
499 bidi_txt = (char *)txt + line_start;
500 }
501
502 /*Calculate the x coordinate*/
503 int32_t x = 0;
504 const lv_text_align_t align = lv_obj_calculate_style_text_align_internal(obj, LV_PART_MAIN, label->text);
505 uint32_t length = new_line_start - line_start;
506 calculate_x_coordinate(&x, align, bidi_txt, length, font, &txt_coords, &attributes);
507
508 lv_text_cmd_state_t cmd_state = LV_TEXT_CMD_STATE_WAIT;
509
510 uint32_t i = 0;
511 uint32_t i_act = i;
512
513 if(new_line_start > 0) {
514 while(i + line_start < new_line_start) {
515 uint32_t letter;
516 uint32_t letter_next;
517 /*Get the current letter and the next letter for kerning*/
518 /*Be careful 'i' already points to the next character*/
519 lv_text_encoded_letter_next_2(bidi_txt, &letter, &letter_next, &i);
520
521 if((attributes.text_flags & LV_TEXT_FLAG_RECOLOR) != 0) {
522 if(lv_text_is_cmd(&cmd_state, bidi_txt[i]) != false) {
523 continue; /*Skip the letter if it is part of a command*/
524 }
525 }
526
527 int32_t gw = lv_font_get_glyph_width(font, letter, letter_next);
528
529 /*Finish if the x position or the last char of the next line is reached*/
530 if(pos.x < x + gw || i + line_start == new_line_start || txt[i_act + line_start] == '\0') {
531 i = i_act;
532 break;
533 }
534 x += gw;
535 x += attributes.letter_space;
536 i_act = i;
537 }
538 }
539
540 uint32_t logical_pos;
541#if LV_USE_BIDI
542 if(bidi) {
543 /*Handle Bidi*/
544 uint32_t cid = lv_text_encoded_get_char_id(bidi_txt, i);
545 if(txt[line_start + i] == '\0') {
546 logical_pos = i;
547 }
548 else {
549 bool is_rtl;
550 logical_pos = lv_bidi_get_logical_pos(&txt[line_start], NULL,
551 txt_len, lv_obj_get_style_base_dir_internal(obj, LV_PART_MAIN), cid, &is_rtl);
552 if(is_rtl) logical_pos++;
553 }
554 lv_free(bidi_txt);
555 }
556 else
557#endif
558 {
559 logical_pos = lv_text_encoded_get_char_id(bidi_txt, i);
560 }
561
562 return logical_pos + lv_text_encoded_get_char_id(txt, line_start);
563}

565bool lv_label_is_char_under_pos(const lv_obj_t * obj, lv_point_t * pos)
566{
567 LV_CHECK_OBJ(obj, MY_CLASS, return false);
568 LV_CHECK_ARG(pos != NULL, return false);
569
570 lv_area_t txt_coords;
571 lv_obj_get_content_coords(obj, &txt_coords);
572 lv_text_attributes_t attributes = {0};
573 const char * txt = lv_label_get_text(obj);
574 lv_label_t * label = (lv_label_t *)obj;
575 uint32_t line_start = 0;
576 uint32_t new_line_start = 0;
577 const lv_font_t * font = lv_obj_get_style_text_font_internal(obj, LV_PART_MAIN);
578 const int32_t letter_height = lv_font_get_line_height_internal(font);
579 const int32_t max_h = lv_area_get_height(&txt_coords);
580
581 attributes.max_width = lv_area_get_width(&txt_coords);
582 attributes.line_space = lv_obj_get_style_text_line_space_internal(obj, LV_PART_MAIN);
583 attributes.letter_space = lv_obj_get_style_text_letter_space_internal(obj, LV_PART_MAIN);
584 attributes.text_flags = get_label_flags(label);
585
586 /*Search the line of the index letter*/
587 int32_t y = 0;
588 while(txt[line_start] != '\0') {
589 bool last_line = y + letter_height + attributes.line_space + letter_height > max_h;
590 if(last_line && label->long_mode == LV_LABEL_LONG_MODE_DOTS) attributes.text_flags |= LV_TEXT_FLAG_BREAK_ALL;
591
592 new_line_start += lv_text_get_next_line(&txt[line_start], LV_TEXT_LEN_MAX, font, NULL, &attributes);
593
594 if(pos->y <= y + letter_height) break; /*The line is found (stored in 'line_start')*/
595 y += letter_height + attributes.line_space;
596
597 line_start = new_line_start;
598 }
599
600 /*Calculate the x coordinate*/
601 const lv_text_align_t align = lv_obj_calculate_style_text_align_internal(obj, LV_PART_MAIN, label->text);
602
603 int32_t x = 0;
604 if(align == LV_TEXT_ALIGN_CENTER) {
605 const int32_t line_w = lv_text_get_line_width(&txt[line_start], new_line_start - line_start, font, &attributes);
606 x += lv_area_get_width(&txt_coords) / 2 - line_w / 2;
607 }
608 else if(align == LV_TEXT_ALIGN_RIGHT) {
609 const int32_t line_w = lv_text_get_line_width(&txt[line_start], new_line_start - line_start, font, &attributes);
610 x += lv_area_get_width(&txt_coords) - line_w;
611 }
612
613 lv_text_cmd_state_t cmd_state = LV_TEXT_CMD_STATE_WAIT;
614
615 int32_t last_x = 0;
616 uint32_t i = line_start;
617 uint32_t i_current = i;
618 uint32_t letter = '\0';
619 uint32_t letter_next = '\0';
620
621 if(new_line_start > 0) {
622 while(i <= new_line_start - 1) {
623 /*Get the current letter and the next letter for kerning*/
624 /*Be careful 'i' already points to the next character*/
625 lv_text_encoded_letter_next_2(txt, &letter, &letter_next, &i);
626
627 if((attributes.text_flags & LV_TEXT_FLAG_RECOLOR) != 0) {
628 if(lv_text_is_cmd(&cmd_state, txt[i]) != false) {
629 continue; /*Skip the letter if it is part of a command*/
630 }
631 }
632
633 last_x = x;
634 x += lv_font_get_glyph_width(font, letter, letter_next);
635 if(pos->x < x) {
636 i = i_current;
637 break;
638 }
639 x += attributes.letter_space;
640 i_current = i;
641 }
642 }
643
644 const int32_t max_diff = lv_font_get_glyph_width(font, letter, letter_next) + attributes.letter_space + 1;
645 return (pos->x >= (last_x - attributes.letter_space) && pos->x <= (last_x + max_diff));
646}

818static void lv_label_event(const lv_obj_class_t * class_p, lv_event_t * e)
819{
820 LV_ASSERT(e != NULL);
821 LV_UNUSED(class_p);
822
823 /*Call the ancestor's event handler*/
824 const lv_result_t res = lv_obj_event_base(MY_CLASS, e);
825 if(res != LV_RESULT_OK) return;
826
827 const lv_event_code_t code = lv_event_get_code(e);
828 lv_obj_t * obj = lv_event_get_current_target(e);
829
830 if((code == LV_EVENT_STYLE_CHANGED) || (code == LV_EVENT_SIZE_CHANGED)) {
831 lv_label_mark_need_refr_text(obj);
832 }
833 else if(code == LV_EVENT_REFR_EXT_DRAW_SIZE) {
834 /* Italic or other non-typical letters can be drawn of out of the object.
835 * It happens if box_w + ofs_x > adw_w in the glyph.
836 * To avoid this add some extra draw area.
837 * font_h / 4 is an empirical value. */
838 const lv_font_t * font = lv_obj_get_style_text_font_internal(obj, LV_PART_MAIN);
839 const int32_t font_h = lv_font_get_line_height_internal(font);
840 lv_event_set_ext_draw_size(e, font_h / 4);
841 }
842 else if(code == LV_EVENT_GET_SELF_SIZE) {
843 lv_label_t * label = (lv_label_t *)obj;
844 if(label->invalid_size_cache) {
845 const lv_font_t * font = lv_obj_get_style_text_font_internal(obj, LV_PART_MAIN);
846 int32_t letter_space = lv_obj_get_style_text_letter_space_internal(obj, LV_PART_MAIN);
847
848 int32_t line_space = lv_obj_get_style_text_line_space_internal(obj, LV_PART_MAIN);
849 lv_text_flag_t flag = LV_TEXT_FLAG_NONE;
850 if(label->recolor != 0) flag |= LV_TEXT_FLAG_RECOLOR;
851 if(label->expand != 0) flag |= LV_TEXT_FLAG_EXPAND;
852
853 int32_t w;
854 if(lv_obj_get_style_width_internal(obj, LV_PART_MAIN) == LV_SIZE_CONTENT && !obj->w_layout) w = LV_COORD_MAX;
855 else w = lv_obj_get_content_width(obj);
856 w = LV_MIN(w, lv_obj_get_style_max_width_internal(obj, LV_PART_MAIN));
857
858 uint32_t dot_begin = label->dot_begin;
859 lv_label_revert_dots(obj);
860
861 lv_text_attributes_t attributes = {0};
862
863 attributes.letter_space = letter_space;
864 attributes.line_space = line_space;
865 attributes.text_flags = flag;
866 attributes.max_width = w;
867
868 lv_text_get_size_attributes(&label->size_cache, label->text, font, &attributes);
869 lv_label_set_dots(obj, dot_begin);
870
871 if(label->max_lines > 0) {
872 label->size_cache.y = LV_MIN(label->size_cache.y,
873 lv_font_get_line_height_internal(font) * label->max_lines + line_space * (label->max_lines - 1));
874 }
875
876 lv_text_leading_trim_t leading_trim =
877 lv_obj_get_style_text_leading_trim_internal(obj, LV_PART_MAIN);
878 if(leading_trim != LV_TEXT_LEADING_TRIM_NONE) {
879 int32_t top_trim = lv_font_get_top_trim_internal(font, leading_trim);
880 int32_t bottom_trim = lv_font_get_bottom_trim_internal(font, leading_trim);
881 label->size_cache.y -= (top_trim + bottom_trim);
882 }
883
884 label->size_cache.y = LV_MIN(label->size_cache.y, lv_obj_get_style_max_height_internal(obj, LV_PART_MAIN));
885
886 label->invalid_size_cache = false;
887 }
888
889 lv_point_t * self_size = lv_event_get_param(e);
890 self_size->x = LV_MAX(self_size->x, label->size_cache.x);
891 self_size->y = LV_MAX(self_size->y, label->size_cache.y);
892 }
893 else if(code == LV_EVENT_DRAW_MAIN) {
894 draw_main(e);
895
896 }
897#if LV_USE_TRANSLATION
898 else if(code == LV_EVENT_TRANSLATION_LANGUAGE_CHANGED) {
899 lv_label_t * label = (lv_label_t *)obj;
900 if(label->translation_tag) {
901 const char * new_text = lv_tr(label->translation_tag);
902 set_text_internal(obj, new_text);
903 }
904 }
905#endif
906}

908static void draw_main(lv_event_t * e)
909{
910 LV_ASSERT(e != NULL);
911 lv_obj_t * obj = lv_event_get_current_target(e);
912 lv_label_t * label = (lv_label_t *)obj;
913 lv_layer_t * layer = lv_event_get_layer(e);
914
915 lv_area_t txt_coords;
916 lv_obj_get_content_coords(obj, &txt_coords);
917
918 /*Shift the text drawing origin up by the top trim amount so the
919 *cap-height or x-height aligns with the top of the trimmed boundary.
920 *Extend the bottom by the bottom trim so descenders are not clipped.*/
921 lv_text_leading_trim_t leading_trim =
922 lv_obj_get_style_text_leading_trim_internal(obj, LV_PART_MAIN);
923 if(leading_trim != LV_TEXT_LEADING_TRIM_NONE) {
924 const lv_font_t * trim_font =
925 lv_obj_get_style_text_font_internal(obj, LV_PART_MAIN);
926 int32_t top_trim = lv_font_get_top_trim_internal(trim_font, leading_trim);
927 int32_t bottom_trim = lv_font_get_bottom_trim_internal(trim_font, leading_trim);
928 txt_coords.y1 -= top_trim;
929 txt_coords.y2 += bottom_trim;
930 }
931
932 lv_text_flag_t flag = get_label_flags(label);
933
934 lv_draw_label_dsc_t label_draw_dsc;
935 lv_draw_label_dsc_init(&label_draw_dsc);
936 label_draw_dsc.text = label->text;
937 label_draw_dsc.text_static = label->static_txt;
938 label_draw_dsc.ofs_x = label->offset.x;
939 label_draw_dsc.ofs_y = label->offset.y;
940 label_draw_dsc.text_size = label->text_size;
941#if LV_LABEL_LONG_TXT_HINT
942 if(label->long_mode != LV_LABEL_LONG_MODE_SCROLL_CIRCULAR &&
943 lv_area_get_height(&txt_coords) >= LV_LABEL_HINT_HEIGHT_LIMIT) {
944 label_draw_dsc.hint = &label->hint;
945 }
946#endif
947
948 label_draw_dsc.flag = flag;
949 label_draw_dsc.base.layer = layer;
950 lv_obj_init_draw_label_dsc(obj, LV_PART_MAIN, &label_draw_dsc);
951 lv_bidi_calculate_align_internal(&label_draw_dsc.align, &label_draw_dsc.bidi_dir, label->text);
952
953 label_draw_dsc.sel_start = lv_label_get_text_selection_start(obj);
954 label_draw_dsc.sel_end = lv_label_get_text_selection_end(obj);
955 if(label_draw_dsc.sel_start != LV_DRAW_LABEL_NO_TXT_SEL && label_draw_dsc.sel_end != LV_DRAW_LABEL_NO_TXT_SEL) {
956 label_draw_dsc.sel_color = lv_obj_get_style_text_color_filtered_internal(obj, LV_PART_SELECTED);
957 label_draw_dsc.sel_bg_color = lv_obj_get_style_bg_color_internal(obj, LV_PART_SELECTED);
958 }
959
960 /* get the style attributes of a letter outline */
961 label_draw_dsc.outline_stroke_color = lv_obj_get_style_text_outline_stroke_color_internal(obj, LV_PART_MAIN);
962 label_draw_dsc.outline_stroke_opa = lv_obj_get_style_text_outline_stroke_opa_internal(obj, LV_PART_MAIN);
963 label_draw_dsc.outline_stroke_width = lv_obj_get_style_text_outline_stroke_width_internal(obj, LV_PART_MAIN);
964
965
966 /* In SCROLL and SCROLL_CIRCULAR mode the CENTER and RIGHT are pointless, so remove them.
967 * (In addition, they will create misalignment in this situation)*/
968 if((label->long_mode == LV_LABEL_LONG_MODE_SCROLL || label->long_mode == LV_LABEL_LONG_MODE_SCROLL_CIRCULAR) &&
969 (label_draw_dsc.align == LV_TEXT_ALIGN_CENTER || label_draw_dsc.align == LV_TEXT_ALIGN_RIGHT)) {
970 lv_point_t size = label->text_size;
971 if(size.x > lv_area_get_width(&txt_coords)) {
972#if LV_USE_BIDI
973 const lv_base_dir_t base_dir = lv_obj_get_style_base_dir_internal(obj, LV_PART_MAIN);
974 label_draw_dsc.align = base_dir == LV_BASE_DIR_RTL ? LV_TEXT_ALIGN_RIGHT : LV_TEXT_ALIGN_LEFT;
975#else
976 label_draw_dsc.align = LV_TEXT_ALIGN_LEFT;
977#endif
978 }
979 }
980
981 lv_area_t txt_clip;
982 bool is_common = lv_area_intersect(&txt_clip, &txt_coords, &layer->_clip_area);
983 if(!is_common) {
984 return;
985 }
986
987 if(label->long_mode == LV_LABEL_LONG_MODE_WRAP) {
988 int32_t s = lv_obj_get_scroll_top(obj);
989 lv_area_move(&txt_coords, 0, -s);
990 txt_coords.y2 = obj->coords.y2;
991 }
992
993 /*Clip to the text in some cases to avoid ugly overflows*/
994 if(label->long_mode == LV_LABEL_LONG_MODE_SCROLL ||
995 label->long_mode == LV_LABEL_LONG_MODE_SCROLL_CIRCULAR ||
996 label->long_mode == LV_LABEL_LONG_MODE_CLIP) {
997 const lv_area_t clip_area_ori = layer->_clip_area;
998 layer->_clip_area = txt_clip;
999 lv_draw_label(layer, &label_draw_dsc, &txt_coords);
1000 layer->_clip_area = clip_area_ori;
1001 }
1002 /*Do not clip to make the drop shadow visible*/
1003 else if(label_draw_dsc.base.drop_shadow_opa > 0) {
1004 lv_draw_label(layer, &label_draw_dsc, &txt_coords);
1005 }
1006 /*Labels have some extra draw area by default to not clip characters with
1007 *italic, handwritten and other less standard fonts.
1008 *However, with most of the fonts typically it's safe to clip at least to bottom side*/
1009 else {
1010 const lv_area_t clip_area_ori = layer->_clip_area;
1011 layer->_clip_area.y2 = txt_clip.y2;
1012 lv_draw_label(layer, &label_draw_dsc, &txt_coords);
1013 layer->_clip_area = clip_area_ori;
1014 }
1015
1016 lv_area_t clip_area_ori = layer->_clip_area;
1017 layer->_clip_area = txt_clip;
1018
1019 if(label->long_mode == LV_LABEL_LONG_MODE_SCROLL_CIRCULAR) {
1020 lv_point_t size = label->text_size;
1021
1022 /*Draw the text again on label to the original to make a circular effect */
1023 if(size.x > lv_area_get_width(&txt_coords)) {
1024 label_draw_dsc.ofs_x = label->offset.x + size.x +
1025 lv_font_get_glyph_width(label_draw_dsc.font, ' ', ' ') * LV_LABEL_WAIT_CHAR_COUNT;
1026 label_draw_dsc.ofs_y = label->offset.y;
1027
1028 lv_draw_label(layer, &label_draw_dsc, &txt_coords);
1029 }
1030
1031 /*Draw the text again below the original to make a circular effect */
1032 if(size.y > lv_area_get_height(&txt_coords)) {
1033 label_draw_dsc.ofs_x = label->offset.x;
1034 label_draw_dsc.ofs_y = label->offset.y + size.y + lv_font_get_line_height_internal(label_draw_dsc.font);
1035
1036 lv_draw_label(layer, &label_draw_dsc, &txt_coords);
1037 }
1038 }
1039
1040 layer->_clip_area = clip_area_ori;
1041}

1043static void set_text_internal(lv_obj_t * obj, const char * text)
1044{
1045 LV_ASSERT(obj != NULL);
1046 lv_label_t * label = (lv_label_t *)obj;
1047
1048 /*If text is NULL then just refresh with the current text*/
1049 if(text == NULL) text = label->text;
1050 if(text == NULL) {
1051 lv_label_mark_need_refr_text(obj);
1052 return;
1053 }
1054
1055 lv_label_revert_dots(obj); /*In case text == label->text*/
1056 const size_t text_len = get_text_length(text);
1057
1058 /*If set its own text then reallocate it (maybe its size changed)*/
1059 if(label->text == text && label->static_txt == 0) {
1060 label->text = lv_realloc(label->text, text_len);
1061 LV_ASSERT_MALLOC(label->text);
1062 if(label->text == NULL) return;
1063
1064#if LV_USE_ARABIC_PERSIAN_CHARS
1065 lv_text_ap_proc(label->text, label->text);
1066#endif
1067
1068 }
1069 else {
1070 /*Free the old text*/
1071 if(label->text != NULL && label->static_txt == 0) {
1072 lv_free(label->text);
1073 label->text = NULL;
1074 }
1075
1076 label->text = lv_malloc(text_len);
1077 LV_ASSERT_MALLOC(label->text);
1078 if(label->text == NULL) return;
1079
1080 copy_text_to_label(label, text);
1081
1082 /*Now the text is dynamically allocated*/
1083 label->static_txt = 0;
1084 }
1085
1086 lv_label_mark_need_refr_text(obj);
1087}
