125static lv_obj_t * card_create(void)
126{
127 lv_obj_t * card = lv_obj_create(lv_screen_active());
128 lv_obj_add_style(card, &style_card, 0);
129 lv_obj_set_clickable(card, false);
130 lv_obj_set_scrollable(card, false);
131 lv_obj_set_scroll_chain_hor(card, false);
132
133 lv_obj_t * avatar = lv_image_create(card);
134 lv_image_set_src(avatar, &img_transform_avatar_15);
135 lv_obj_set_grid_cell(avatar, LV_GRID_ALIGN_CENTER, 0, 1, LV_GRID_ALIGN_CENTER, 0, 4);
136 lv_obj_add_style(avatar, &style_avatar, 0);
137
138 lv_obj_t * name = lv_label_create(card);
139 lv_label_set_text(name, "Pavel Svoboda");
140 lv_obj_set_grid_cell(name, LV_GRID_ALIGN_START, 2, 1, LV_GRID_ALIGN_CENTER, 0, 1);
141 lv_obj_set_style_text_font(name, LV_FONT_DEFAULT, 0);
142
143 lv_obj_t * btn = lv_button_create(card);
144 lv_obj_set_scroll_chain_hor(card, false);
145 lv_obj_set_grid_cell(btn, LV_GRID_ALIGN_START, 2, 1, LV_GRID_ALIGN_CENTER, 3, 1);
146 lv_obj_add_style(btn, &style_btn, 0);
147
148 LV_IMAGE_DECLARE(img_multilang_like);
149 lv_obj_t * btn_img = lv_image_create(btn);
150 lv_image_set_src(btn_img, &img_multilang_like);
151 lv_obj_align(btn_img, LV_ALIGN_LEFT_MID, 30, 0);
152
153 lv_obj_t * btn_label = lv_label_create(btn);
154 lv_label_set_text(btn_label, "Like");
155 lv_obj_align(btn_label, LV_ALIGN_LEFT_MID, 60, 1);
156
157 lv_obj_scroll_to_view(card, LV_ANIM_OFF);
158
159 return card;
160}
