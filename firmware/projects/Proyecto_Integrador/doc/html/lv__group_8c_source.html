77void lv_group_delete(lv_group_t * group)
78{
79 if(group == NULL) return;
80
81 /*Defocus the currently focused object*/
82 if(group->obj_focus != NULL) {
83 lv_obj_send_event(*group->obj_focus, LV_EVENT_DEFOCUSED, get_indev(group));
84 lv_obj_invalidate(*group->obj_focus);
85 }
86
87 /*Remove the objects from the group*/
88 lv_obj_t ** obj;
89 LV_LL_READ(&group->obj_ll, obj) {
90 if((*obj)->spec_attr)(*obj)->spec_attr->group_p = NULL;
91 }
92
93 /*Remove the group from any indev devices */
94 lv_indev_t * indev = lv_indev_get_next(NULL);
95 while(indev) {
96 if(lv_indev_get_group(indev) == group) {
97 lv_indev_set_group(indev, NULL);
98 }
99 indev = lv_indev_get_next(indev);
100 }
101
102 /*If the group is the default group, set the default group as NULL*/
103 if(group == lv_group_get_default()) lv_group_set_default(NULL);
104
105 lv_ll_clear(&(group->obj_ll));
106 lv_ll_remove(group_ll_p, group);
107#if LV_USE_EXT_DATA
108 if(group->ext_data.free_cb) {
109 group->ext_data.free_cb(group->ext_data.data);
110 group->ext_data.data = NULL;
111 }
112#endif
113 lv_free(group);
114}

126void lv_group_add_obj(lv_group_t * group, lv_obj_t * obj)
127{
128 LV_CHECK_ARG(group != NULL, return);
129 LV_CHECK_ARG(obj != NULL, return);
130
131 LV_LOG_TRACE("begin");
132
133 if(!lv_obj_allocate_spec_attr(obj)) {
134 return;
135 }
136
137 LV_ASSERT(obj->spec_attr != NULL);
138
139 /*Be sure the object is removed from its current group*/
140 lv_group_remove_obj(obj);
141 obj->spec_attr->group_p = group;
142
143 lv_obj_t ** next = lv_ll_ins_tail(&group->obj_ll);
144 LV_ASSERT_MALLOC(next);
145 if(next == NULL) return;
146 *next = obj;
147
148 /*If the head and the tail is equal then there is only one object in the linked list.
149 *In this case automatically activate it*/
150 if(lv_ll_get_head(&group->obj_ll) == next) {
151 lv_group_refocus(group);
152 }
153
154 LV_LOG_TRACE("finished");
155}

157void lv_group_swap_obj(lv_obj_t * obj1, lv_obj_t * obj2)
158{
159 LV_CHECK_ARG(obj1 != NULL, return);
160 LV_CHECK_ARG(obj2 != NULL, return);
161
162 lv_group_t * g1 = lv_obj_get_group(obj1);
163 lv_group_t * g2 = lv_obj_get_group(obj2);
164
165 if(g1 != g2) return;
166 if(g1 == NULL) return;
167
168 /*Do not add the object twice*/
169 lv_obj_t ** obj_i;
170 LV_LL_READ(&g1->obj_ll, obj_i) {
171 if((*obj_i) == obj1)(*obj_i) = obj2;
172 else if((*obj_i) == obj2)(*obj_i) = obj1;
173 }
174
175 /* Swap the focus as well. */
176 lv_obj_t * focused = lv_group_get_focused(g1);
177 if(focused == obj1) lv_group_focus_obj(obj2);
178 else if(focused == obj2) lv_group_focus_obj(obj1);
179}

181void lv_group_remove_obj(lv_obj_t * obj)
182{
183 LV_CHECK_ARG(obj != NULL, return);
184
185 lv_group_t * g = lv_obj_get_group(obj);
186 if(g == NULL) return;
187
188 LV_LOG_TRACE("begin");
189
190 /* If we are removing the focused object */
191 if(g->obj_focus && *g->obj_focus == obj) {
192 if(g->frozen) g->frozen = 0;
193
194 /*If there are more objects in the group, try to move focus to the next/prev object*/
195 if(lv_ll_get_head(&g->obj_ll) != g->obj_focus || lv_ll_get_tail(&g->obj_ll) != g->obj_focus) {
196 lv_group_refocus(g);
197 }
198 }
199
200 /*If the focused object is still the one being removed (either it was the only object,
201 *or refocus couldn't find a focusable candidate), defocus it and clear the focus.*/
202 if(g->obj_focus && *g->obj_focus == obj) {
203 lv_obj_send_event(*g->obj_focus, LV_EVENT_DEFOCUSED, get_indev(g));
204 g->obj_focus = NULL;
205 }
206
207 /*Search the object and remove it from its group*/
208 lv_obj_t ** i;
209 LV_LL_READ(&g->obj_ll, i) {
210 if(*i == obj) {
211 lv_ll_remove(&g->obj_ll, i);
212 lv_free(i);
213 if(obj->spec_attr) obj->spec_attr->group_p = NULL;
214 break;
215 }
216 }
217 LV_LOG_TRACE("finished");
218}

240void lv_group_focus_obj(lv_obj_t * obj)
241{
242 LV_CHECK_ARG(obj != NULL, return);
243
244 lv_group_t * g = lv_obj_get_group(obj);
245
246 if(g == NULL) return;
247 if(g->frozen) return;
248
249 /*On defocus edit mode must be leaved*/
250 lv_group_set_editing(g, false);
251
252 lv_obj_t ** i;
253 LV_LL_READ(&g->obj_ll, i) {
254 if(*i == obj) {
255 if(g->obj_focus != NULL) {
256 if(*g->obj_focus == obj) break; /*Refocusing same object*/
257 lv_result_t res = lv_obj_send_event(*g->obj_focus, LV_EVENT_DEFOCUSED, get_indev(g));
258 if(res != LV_RESULT_OK) return;
259 lv_obj_invalidate(*g->obj_focus);
260 }
261
262 g->obj_focus = i;
263
264 if(g->focus_cb) g->focus_cb(g);
265 lv_result_t res = lv_obj_send_event(*g->obj_focus, LV_EVENT_FOCUSED, get_indev(g));
266 if(res != LV_RESULT_OK) return;
267 lv_obj_invalidate(*g->obj_focus);
268 break;
269 }
270 }

458static void lv_group_refocus(lv_group_t * g)
459{
460 LV_ASSERT(g != NULL);
461 /*Refocus must temporarily allow wrapping to work correctly*/
462 uint8_t temp_wrap = g->wrap;
463 g->wrap = 1;
464
465 if(g->refocus_policy == LV_GROUP_REFOCUS_POLICY_NEXT)
466 lv_group_focus_next(g);
467 else if(g->refocus_policy == LV_GROUP_REFOCUS_POLICY_PREV)
468 lv_group_focus_prev(g);
469 /*Restore wrap property*/
470 g->wrap = temp_wrap;
471}

473static bool focus_next_core(lv_group_t * group, void * (*begin)(const lv_ll_t *),
474 void * (*move)(const lv_ll_t *, const void *))
475{
476 LV_ASSERT(group != NULL);
477 LV_ASSERT(begin != NULL);
478 LV_ASSERT(move != NULL);
479 if(group->frozen) return false;
480
481 bool focus_changed = false;
482
483 lv_obj_t ** obj_next = group->obj_focus;
484 lv_obj_t ** obj_sentinel = NULL;
485 bool can_move = true;
486 bool can_begin = true;
487
488 for(;;) {
489 if(obj_next == NULL) {
490 if(group->wrap || obj_sentinel == NULL) {
491 if(!can_begin) return focus_changed;
492 obj_next = begin(&group->obj_ll);
493 can_move = false;
494 can_begin = false;
495 }
496 else {
497 /*Currently focused object is the last/first in the group, keep it that way*/
498 return focus_changed;
499 }
500 }
501
502 if(obj_sentinel == NULL) {
503 obj_sentinel = obj_next;
504 if(obj_sentinel == NULL) return focus_changed; /*Group is empty*/
505 }
506
507 if(can_move) {
508 obj_next = move(&group->obj_ll, obj_next);
509
510 /*Give up if we walked the entire list and haven't found another visible object*/
511 if(obj_next == obj_sentinel) return focus_changed;
512 }
513
514 can_move = true;
515
516 if(obj_next == NULL) continue;
517 if(lv_obj_get_state(*obj_next) & LV_STATE_DISABLED) continue;
518
519 /*Hidden objects don't receive focus.
520 *If any parent is hidden, the object is also hidden)*/
521 lv_obj_t * parent = *obj_next;
522 while(parent) {
523 if(lv_obj_is_hidden(parent)) break;
524 parent = lv_obj_get_parent(parent);
525 }
526
527 if(parent && lv_obj_is_hidden(parent)) continue;
528
529 /*If we got her a good candidate is found*/
530 break;
531 }
532
533 if(obj_next == group->obj_focus) return focus_changed; /*There's only one visible object and it's already focused*/
534
535 if(group->obj_focus) {
536 lv_result_t res = lv_obj_send_event(*group->obj_focus, LV_EVENT_DEFOCUSED, get_indev(group));
537 if(res != LV_RESULT_OK) return focus_changed;
538 lv_obj_invalidate(*group->obj_focus);
539 }
540
541 group->obj_focus = obj_next;
542
543 lv_result_t res = lv_obj_send_event(*group->obj_focus, LV_EVENT_FOCUSED, get_indev(group));
544 if(res != LV_RESULT_OK) return focus_changed;
545
546 lv_obj_invalidate(*group->obj_focus);
547
548 if(group->focus_cb) group->focus_cb(group);
549 focus_changed = true;
550 return focus_changed;
551}

560static lv_indev_t * get_indev(const lv_group_t * g)
561{
562 LV_ASSERT(g != NULL);
563 lv_indev_t * indev_guess = NULL;
564 lv_indev_t * indev = lv_indev_get_next(NULL);
565
566 while(indev) {
567 lv_indev_type_t indev_type = lv_indev_get_type(indev);
568 /*Prefer POINTER*/
569 if(indev_type == LV_INDEV_TYPE_POINTER) return indev;
570 if(lv_indev_get_group(indev) == g) {
571 indev_guess = indev;
572 }
573 indev = lv_indev_get_next(indev);
574 }
575
576 return indev_guess;
577}
