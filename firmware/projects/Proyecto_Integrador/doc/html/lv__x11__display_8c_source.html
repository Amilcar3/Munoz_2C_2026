112static void x11_flush_cb(lv_display_t * disp, const lv_area_t * area, uint8_t * px_map)
113{
114 x11_disp_data_t * xd = lv_display_get_driver_data(disp);
115 LV_ASSERT(xd != NULL);
116
117 /* build display update area until lv_display_flush_is_last */
118 xd->flush_area.x1 = MIN(xd->flush_area.x1, area->x1);
119 xd->flush_area.x2 = MAX(xd->flush_area.x2, area->x2);
120 xd->flush_area.y1 = MIN(xd->flush_area.y1, area->y1);
121 xd->flush_area.y2 = MAX(xd->flush_area.y2, area->y2);
122
123 int32_t hor_res = lv_display_get_horizontal_resolution(disp);
124
125 uint32_t dst_offs;
126 lv_color32_t * dst_data;
127 color_t * src_data = (color_t *)px_map + (LV_X11_RENDER_MODE == LV_DISPLAY_RENDER_MODE_PARTIAL ? 0 : hor_res *
128 area->y1 + area->x1);
129 for(int16_t y = area->y1; y <= area->y2; y++) {
130 dst_offs = area->x1 + y * hor_res;
131 dst_data = &((lv_color32_t *)(xd->xdata))[dst_offs];
132 for(int16_t x = area->x1; x <= area->x2; x++, src_data++, dst_data++) {
133 *dst_data = get_px(*src_data);
134 }
135 src_data += (LV_X11_RENDER_MODE == LV_DISPLAY_RENDER_MODE_PARTIAL ? 0 : hor_res - (area->x2 - area->x1 + 1));
136 }
137
138 if(lv_display_flush_is_last(disp)) {
139 LV_LOG_TRACE("(%d/%d), %dx%d)", xd->flush_area.x1, xd->flush_area.y1, xd->flush_area.x2 + 1 - xd->flush_area.x1,
140 xd->flush_area.y2 + 1 - xd->flush_area.y1);
141
142 /* refresh collected display update area only */
143 int16_t upd_w = xd->flush_area.x2 - xd->flush_area.x1 + 1;
144 int16_t upd_h = xd->flush_area.y2 - xd->flush_area.y1 + 1;
145 XPutImage(xd->hdr.display, xd->window, xd->gc, xd->ximage, xd->flush_area.x1, xd->flush_area.y1, xd->flush_area.x1,
146 xd->flush_area.y1, upd_w, upd_h);
147
148 /* invalidate collected area */
149 x11_reset_flush_area(xd);
150 }
151 /* Inform the graphics library that you are ready with the flushing */
152 lv_display_flush_ready(disp);
153}

159static void x11_resolution_evt_cb(lv_event_t * e)
160{
161 lv_display_t * disp = lv_event_get_user_data(e);
162 x11_disp_data_t * xd = lv_display_get_driver_data(disp);
163 LV_ASSERT(xd != NULL);
164
165 int32_t hor_res = lv_display_get_horizontal_resolution(disp);
166 int32_t ver_res = lv_display_get_vertical_resolution(disp);
167
168 if(LV_X11_RENDER_MODE != LV_DISPLAY_RENDER_MODE_PARTIAL) {
169 /* update lvgl full-screen display draw buffers for new display size */
170 int sz_buffers = (hor_res * ver_res * (LV_COLOR_DEPTH + 7) / 8);
171 xd->buffer[0] = realloc(xd->buffer[0], sz_buffers);
172 xd->buffer[1] = (LV_X11_DOUBLE_BUFFER ? realloc(xd->buffer[1], sz_buffers) : NULL);
173 lv_display_set_buffers(disp, xd->buffer[0], xd->buffer[1], sz_buffers, LV_X11_RENDER_MODE);
174 }
175
176 /* re-create cache image with new size */
177 XDestroyImage(xd->ximage);
178 size_t sz_buffers = hor_res * ver_res * sizeof(lv_color32_t);
179 xd->xdata = calloc(1, sz_buffers); /* use clib method here, x11 memory not part of device footprint */
180 xd->ximage = XCreateImage(xd->hdr.display, xd->visual, xd->dplanes, ZPixmap, 0, xd->xdata,
181 hor_res, ver_res, lv_color_format_get_bpp(LV_COLOR_FORMAT_ARGB8888), 0);
182 x11_reset_flush_area(xd);
183}

235static int is_disp_event(Display * disp, XEvent * event, XPointer arg)
236{
237 LV_UNUSED(disp);
238 LV_UNUSED(arg);
239 return (event->type == Expose
240 || (event->type >= DestroyNotify && event->type <= CirculateNotify) /* events from StructureNotifyMask */
241 || event->type == ClientMessage);
242}
243static void x11_event_handler(lv_timer_t * t)
244{
245 lv_display_t * disp = lv_timer_get_user_data(t);
246 x11_disp_data_t * xd = lv_display_get_driver_data(disp);
247 LV_ASSERT(xd != NULL);
248
249 /* handle all outstanding X events */
250 XEvent event;
251 while(XCheckIfEvent(xd->hdr.display, &event, is_disp_event, NULL)) {
252 LV_LOG_TRACE("Display Event %d", event.type);
253 switch(event.type) {
254 case Expose:
255 if(event.xexpose.count == 0) {
256 XPutImage(xd->hdr.display, xd->window, xd->gc, xd->ximage, 0, 0, 0, 0, event.xexpose.width, event.xexpose.height);
257 }
258 break;
259 case ConfigureNotify:
260 if(event.xconfigure.width != lv_display_get_horizontal_resolution(disp)
261 || event.xconfigure.height != lv_display_get_vertical_resolution(disp)) {
262 lv_display_set_resolution(disp, event.xconfigure.width, event.xconfigure.height);
263 lv_refr_now(disp);
264 }
265 break;
266 case ClientMessage:
267 if(event.xclient.data.l[0] == (long)xd->wmDeleteMessage) {
268 xd->terminated = true;
269 void * ret = NULL;
270 pthread_join(xd->thr_tick, &ret);
271 lv_display_delete(disp);
272 return;
273 }
274 break;
275 case MapNotify:
276 case ReparentNotify:
277 /*suppress unhandled warning*/
278 break;
279 default:
280 LV_LOG_WARN("unhandled x11 event: %d", event.type);
281 }
282 }
283}

297static void x11_window_create(lv_display_t * disp, char const * title)
298{
299 x11_disp_data_t * xd = lv_display_get_driver_data(disp);
300 LV_ASSERT(xd != NULL);
301
302 /* setup display/screen */
303 xd->hdr.display = XOpenDisplay(NULL);
304 int screen = XDefaultScreen(xd->hdr.display);
305 xd->visual = XDefaultVisual(xd->hdr.display, screen);
306
307 /* create window */
308 int32_t hor_res = lv_display_get_horizontal_resolution(disp);
309 int32_t ver_res = lv_display_get_vertical_resolution(disp);
310#if 0
311 /* drawing contexts for an window */
312 unsigned long col_fg = BlackPixel(xd->hdr.display, screen);
313 unsigned long col_bg = WhitePixel(xd->hdr.display, screen);
314
315 xd->window = XCreateSimpleWindow(xd->hdr.display, DefaultRootWindow(xd->hdr.display),
316 0, 0, hor_res, ver_res, 0, col_fg, col_bg);
317#else
318 xd->window = XCreateWindow(xd->hdr.display, XDefaultRootWindow(xd->hdr.display),
319 0, 0, hor_res, ver_res, 0,
320 XDefaultDepth(xd->hdr.display, screen), InputOutput,
321 xd->visual, 0, NULL);
322#endif
323 /* window manager properties (yes, use of StdProp is obsolete) */
324 XSetStandardProperties(xd->hdr.display, xd->window, title, NULL, None, NULL, 0, NULL);
325 xd->gc = XCreateGC(xd->hdr.display, xd->window, 0, 0);
326
327 /* allow receiving mouse, keyboard and window change/close events */
328 XSelectInput(xd->hdr.display, xd->window,
329 PointerMotionMask | ButtonPressMask | ButtonReleaseMask | KeyPressMask | KeyReleaseMask | ExposureMask |
330 StructureNotifyMask);
331 xd->wmDeleteMessage = XInternAtom(xd->hdr.display, "WM_DELETE_WINDOW", False);
332 XSetWMProtocols(xd->hdr.display, xd->window, &xd->wmDeleteMessage, 1);
333
334 x11_hide_cursor(disp);
335
336 /* create cache XImage */
337 size_t sz_buffers = hor_res * ver_res * sizeof(lv_color32_t);
338 xd->dplanes = XDisplayPlanes(xd->hdr.display, screen);
339 xd->xdata = calloc(1, sz_buffers); /* use clib method here, x11 memory not part of device footprint */
340 xd->ximage = XCreateImage(xd->hdr.display, xd->visual, xd->dplanes, ZPixmap, 0, xd->xdata,
341 hor_res, ver_res, lv_color_format_get_bpp(LV_COLOR_FORMAT_ARGB8888), 0);
342 x11_reset_flush_area(xd);
343
344 /* finally bring window on top of the other windows */
345 XMapRaised(xd->hdr.display, xd->window);
346
347#if LV_X11_DIRECT_EXIT
348 count_windows++;
349#endif
350}

356lv_display_t * lv_x11_window_create(char const * title, int32_t hor_res, int32_t ver_res)
357{
358 LV_CHECK_ARG(title != NULL, return NULL);
359
360 x11_disp_data_t * xd = lv_malloc_zeroed(sizeof(x11_disp_data_t));
361 LV_ASSERT_MALLOC(xd);
362 if(NULL == xd) return NULL;
363
364 lv_display_t * disp = lv_display_create(hor_res, ver_res);
365 if(NULL == disp) {
366 lv_free(xd);
367 return NULL;
368 }
369 lv_display_set_driver_data(disp, xd);
370 lv_display_set_flush_cb(disp, x11_flush_cb);
371 lv_display_add_event_cb(disp, x11_resolution_evt_cb, LV_EVENT_RESOLUTION_CHANGED, disp);
372 lv_display_add_event_cb(disp, x11_disp_delete_evt_cb, LV_EVENT_DELETE, disp);
373
374 x11_window_create(disp, title);
375
376 int sz_buffers = (hor_res * ver_res * (LV_COLOR_DEPTH + 7) / 8);
377 if(LV_X11_RENDER_MODE == LV_DISPLAY_RENDER_MODE_PARTIAL) {
378 sz_buffers /= 10;
379 }
380 xd->buffer[0] = malloc(sz_buffers);
381 xd->buffer[1] = (LV_X11_DOUBLE_BUFFER ? malloc(sz_buffers) : NULL);
382 lv_display_set_buffers(disp, xd->buffer[0], xd->buffer[1], sz_buffers, LV_X11_RENDER_MODE);
383
384 xd->timer = lv_timer_create(x11_event_handler, 5, disp);
385
386 /* initialize Tick simulation */
387 xd->terminated = false;
388 pthread_create(&xd->thr_tick, NULL, x11_tick_thread, xd);
389
390 return disp;
391}
