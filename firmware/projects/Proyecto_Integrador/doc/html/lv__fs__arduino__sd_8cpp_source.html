60static void * fs_open(lv_fs_drv_t * drv, const char * path, lv_fs_mode_t mode)
61{
62 LV_ASSERT(path != NULL);
63 LV_UNUSED(drv);
64
65 const char * flags;
66 if(mode == LV_FS_MODE_WR)
67 flags = FILE_WRITE;
68 else if(mode == LV_FS_MODE_RD)
69 flags = FILE_READ;
70 else if(mode == (LV_FS_MODE_WR | LV_FS_MODE_RD))
71 flags = FILE_WRITE;
72 else {
73 LV_LOG_WARN("Unsupported mode: 0x%x", (unsigned)mode);
74 return NULL;
75 }
76
77 char buf[LV_FS_MAX_PATH_LEN];
78 lv_snprintf(buf, sizeof(buf), LV_FS_ARDUINO_SD_PATH "%s", path);
79
80 File file = SD.open(buf, flags);
81 if(!file) {
82 return NULL;
83 }
84
85 SdFile * lf = new SdFile{file};
86
87 return (void *)lf;
88}

158{
159 LV_ASSERT(file_p != NULL);
160 LV_UNUSED(drv);
161 SeekMode mode;

163 mode = SeekSet;
164 else if(whence == LV_FS_SEEK_CUR)
165 mode = SeekCur;

167 mode = SeekEnd;
168
