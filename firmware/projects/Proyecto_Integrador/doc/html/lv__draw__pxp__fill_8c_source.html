49void lv_draw_pxp_fill(lv_draw_task_t * t)
50{
51 const lv_draw_fill_dsc_t * dsc = t->draw_dsc;
52 const lv_area_t * coords = &t->area;
53
54 if(dsc->opa <= (lv_opa_t)LV_OPA_MIN)
55 return;
56
57 lv_layer_t * layer = t->target_layer;
58 lv_draw_buf_t * draw_buf = layer->draw_buf;
59
60 lv_area_t rel_coords;
61 rel_coords = *coords;
62 lv_area_move(&rel_coords, -layer->buf_area.x1, -layer->buf_area.y1);
63
64 lv_area_t rel_clip_area;
65 rel_clip_area = t->clip_area;
66 lv_area_move(&rel_clip_area, -layer->buf_area.x1, -layer->buf_area.y1);
67
68 lv_area_t blend_area;
69 if(!lv_area_intersect(&blend_area, &rel_coords, &rel_clip_area))
70 return; /*Fully clipped, nothing to do*/
71
72 _pxp_fill(draw_buf->data, &blend_area, draw_buf->header.stride, draw_buf->header.cf, dsc);
73}

79static void _pxp_fill(uint8_t * dest_buf, const lv_area_t * dest_area, int32_t dest_stride,
80 lv_color_format_t dest_cf, const lv_draw_fill_dsc_t * dsc)
81{
82 int32_t dest_w = lv_area_get_width(dest_area);
83 int32_t dest_h = lv_area_get_height(dest_area);
84
85 lv_pxp_reset();
86
87 uint8_t px_size = lv_color_format_get_size(dest_cf);
88
89 /*OUT buffer configure*/
90 pxp_output_buffer_config_t outputConfig = {
91 .pixelFormat = pxp_get_out_px_format(dest_cf),
92 .interlacedMode = kPXP_OutputProgressive,
93 .buffer0Addr = (uint32_t)(dest_buf + dest_stride * dest_area->y1 + px_size * dest_area->x1),
94 .buffer1Addr = (uint32_t)NULL,
95 .pitchBytes = dest_stride,
96 .width = dest_w,
97 .height = dest_h
98 };
99
100 PXP_SetOutputBufferConfig(PXP_ID, &outputConfig);
101
102 if(dsc->opa >= (lv_opa_t)LV_OPA_MAX) {
103 /*Simple color fill without opacity - AS disabled*/
104 PXP_SetAlphaSurfacePosition(PXP_ID, 0xFFFFU, 0xFFFFU, 0U, 0U);
105
106 }
107 else {
108 /*Fill with opacity - AS used as source (same as OUT)*/
109 pxp_as_buffer_config_t asBufferConfig = {
110 .pixelFormat = pxp_get_as_px_format(dest_cf),
111 .bufferAddr = outputConfig.buffer0Addr,
112 .pitchBytes = outputConfig.pitchBytes
113 };
114
115 PXP_SetAlphaSurfaceBufferConfig(PXP_ID, &asBufferConfig);
116 PXP_SetAlphaSurfacePosition(PXP_ID, 0U, 0U, dest_w - 1U, dest_h - 1U);
117 }
118
119 /*Disable PS, use as color generator*/
120 PXP_SetProcessSurfacePosition(PXP_ID, 0xFFFFU, 0xFFFFU, 0U, 0U);
121 PXP_SetProcessSurfaceBackGroundColor(PXP_ID, lv_color_to_u32(dsc->color));
122
130 pxp_porter_duff_config_t pdConfig = {
131 .enable = 1,
132 .dstColorMode = kPXP_PorterDuffColorNoAlpha,
133 .srcColorMode = kPXP_PorterDuffColorNoAlpha,
134 .dstGlobalAlphaMode = kPXP_PorterDuffGlobalAlpha,
135 .srcGlobalAlphaMode = kPXP_PorterDuffGlobalAlpha,
136 .dstFactorMode = kPXP_PorterDuffFactorStraight,
137 .srcFactorMode = (dsc->opa >= (lv_opa_t)LV_OPA_MAX) ? kPXP_PorterDuffFactorStraight : kPXP_PorterDuffFactorInversed,
138 .dstGlobalAlpha = dsc->opa,
139 .srcGlobalAlpha = dsc->opa,
140 .dstAlphaMode = kPXP_PorterDuffAlphaStraight, /*don't care*/
141 .srcAlphaMode = kPXP_PorterDuffAlphaStraight /*don't care*/
142 };
143
144 PXP_SetPorterDuffConfig(PXP_ID, &pdConfig);

147}
