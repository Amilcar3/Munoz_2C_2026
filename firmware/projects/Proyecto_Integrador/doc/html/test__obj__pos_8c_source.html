263 lv_obj_set_style_bg_color(rect_bottom_left, lv_palette_main(LV_PALETTE_GREEN), LV_PART_MAIN);
264 lv_obj_set_style_bg_opa(rect_bottom_left, LV_OPA_COVER, LV_PART_MAIN);
265
266 lv_obj_t * rect_bottom_right = lv_obj_create(cont);
267 lv_obj_remove_style(rect_bottom_right, NULL, LV_PART_MAIN);
268 lv_obj_set_size(rect_bottom_right, 10, 10);
269 lv_obj_set_align(rect_bottom_right, LV_ALIGN_BOTTOM_RIGHT);
270 lv_obj_set_style_bg_color(rect_bottom_right, lv_palette_main(LV_PALETTE_RED), LV_PART_MAIN);
271 lv_obj_set_style_bg_opa(rect_bottom_right, LV_OPA_COVER, LV_PART_MAIN);
272
273 return cont;
274}
