51uint32_t lv_gltf_data_get_animation_total_time(lv_gltf_model_t * data, uint32_t index)
52{
53 LV_ASSERT(data != NULL);
54 LV_ASSERT(data->asset.animations.size() > index);
55 auto & animation = data->asset.animations[index];
56 float max_time = -1.0f;
57 for(uint64_t i = 0; i < animation.channels.size(); i++) {
58 auto & accessor = data->asset.accessors[animation.samplers[i].inputAccessor];
59 max_time = std::max(max_time, fastgltf::getAccessorElement<float>(data->asset, accessor, accessor.count - 1));
60 }
61 return (uint32_t)(max_time * 1000);
62}

64std::vector<uint32_t> * lv_gltf_data_animation_get_channel_set(std::size_t anim_num, lv_gltf_model_t * data,
65 fastgltf::Node * node)
66{
67 LV_ASSERT(data != NULL);
68 LV_ASSERT(node != NULL);
69 const auto & asset = lv_gltf_data_get_asset(data);
70 size_t animation_count = lv_gltf_model_get_animation_count(data);
71 if(data->channel_set_cache.find(node) == data->channel_set_cache.end()) {
72 std::vector<uint32_t> new_cache = std::vector<uint32_t>();
73 if(animation_count > anim_num) {
74 auto & anim = asset->animations[anim_num];
75
76 for(uint64_t c = 0; c < anim.channels.size(); c++) {
77 auto & channel = anim.channels[c];
78 if(&(asset->nodes[channel.nodeIndex.value()]) == node) {
79 new_cache.push_back(c);
80 }
81 }
82 }
83 data->channel_set_cache[node] = new_cache;
84 }
85 return &data->channel_set_cache[node];
86}

89void lv_gltf_data_animation_matrix_apply(float timestamp, std::size_t anim_num, lv_gltf_model_t * gltf_data,
90 fastgltf::Node * node,
91 fastgltf::math::fmat4x4 & matrix)
92{
93 LV_ASSERT(gltf_data != NULL);
94 LV_ASSERT(node != NULL);
95 const auto & asset = lv_gltf_data_get_asset(gltf_data);
96
97 size_t animation_count = lv_gltf_model_get_animation_count(gltf_data);
98 auto _channel_set = lv_gltf_data_animation_get_channel_set(anim_num, gltf_data, node);
99 if(_channel_set->size() == 0) {
100 return;
101 }
102 if(animation_count > anim_num) {
103 auto & anim = asset->animations[anim_num];
104 bool need_rot_recalc = false;
105 int32_t translation_comp_index = -1;
106 int32_t rotation_comp_index = -1;
107 int32_t scale_comp_index = -1;
108
109 for(const auto & c : (*_channel_set)) {
110 switch(anim.channels[c].path) {
111 case fastgltf::AnimationPath::Translation:
112 translation_comp_index = c;
113 break;
114 case fastgltf::AnimationPath::Rotation:
115 rotation_comp_index = c;
116 need_rot_recalc = true;
117 break;
118 case fastgltf::AnimationPath::Scale:
119 scale_comp_index = c;
120 need_rot_recalc = true;
121 break;
122 case fastgltf::AnimationPath::Weights:
123 LV_LOG_WARN("Unhandled weights animation");
124 break;
125 }
126 }
127
128 if(need_rot_recalc) {
129 fastgltf::math::fvec3 new_scale;
130 fastgltf::math::fquat new_quat;
131 if(!((scale_comp_index > -1) && (rotation_comp_index > -1))) {
132 fastgltf::math::fvec3 unused;
133 fastgltf::math::decomposeTransformMatrix(matrix, new_scale, new_quat, unused);
134 }
135 if(scale_comp_index > -1) new_scale = animation_get_vec3_at_timestamp(gltf_data, &anim.samplers[scale_comp_index],
136 timestamp);
137 if(rotation_comp_index > -1) new_quat = animation_get_quat_at_timestamp(gltf_data, &anim.samplers[rotation_comp_index],
138 timestamp);
139
140 float sx = new_scale[0], sy = new_scale[1], sz = new_scale[2];
141 float qx = new_quat[0], qy = new_quat[1], qz = new_quat[2], qw = new_quat[3];
142
143 float x2 = qx + qx, y2 = qy + qy, z2 = qz + qz;
144 float xx = qx * x2, xy = qx * y2, xz = qx * z2;
145 float yy = qy * y2, yz = qy * z2, zz = qz * z2;
146 float wx = qw * x2, wy = qw * y2, wz = qw * z2;
147
148 matrix[0][0] = (1 - (yy + zz)) * sx;
149 matrix[0][1] = (xy + wz) * sx;
150 matrix[0][2] = (xz - wy) * sx;
151
152 matrix[1][0] = (xy - wz) * sy;
153 matrix[1][1] = (1 - (xx + zz)) * sy;
154 matrix[1][2] = (yz + wx) * sy;
155
156 matrix[2][0] = (xz + wy) * sz;
157 matrix[2][1] = (yz - wx) * sz;
158 matrix[2][2] = (1 - (xx + yy)) * sz;
159
160 /* These entries should not be necessary */
161 //matrix[0][3] = 0.f;
162 //matrix[1][3] = 0.f;
163 //matrix[2][3] = 0.f;
164 }
165
166 if(translation_comp_index > -1) {
167 fastgltf::math::fvec3 new_translation = animation_get_vec3_at_timestamp(gltf_data,
168 &anim.samplers[translation_comp_index], timestamp);
169 matrix[3][0] = new_translation[0];
170 matrix[3][1] = new_translation[1];
171 matrix[3][2] = new_translation[2];
172 }
173 matrix[3][3] = 1.f;
174 }
175}

181static fastgltf::math::fquat animation_get_quat_at_timestamp(lv_gltf_model_t * data,
182 fastgltf::AnimationSampler * sampler,
183 float _seconds)
184{
185 LV_ASSERT(data != NULL);
186 LV_ASSERT(sampler != NULL);
187 const auto & asset = lv_gltf_data_get_asset(data);
188 auto & _inAcc = asset->accessors[sampler->inputAccessor];
189 auto & _outAcc = asset->accessors[sampler->outputAccessor];
190 std::size_t _inAccCount = _inAcc.count;
191 float _maxTime = fastgltf::getAccessorElement<float>(*asset, _inAcc, _inAccCount - 1);
192 std::size_t _lowerIndex = 0;
193 float _lowerTimestamp = 0.0f;
194
195 if(_seconds < 0.001f) {
196 _lowerIndex = 0;
197 }
198 else {
199 std::size_t _firstCheckOffset = 0;
200 std::size_t _lastCheckOffset = _inAccCount;
201 std::size_t _prepassLeft = TIME_LOC_PREPASS_COUNT;
202 while(_prepassLeft > 0) {
203 _prepassLeft -= 1;
204 if(_seconds >= fastgltf::getAccessorElement<float>(*asset, _inAcc, (_firstCheckOffset + _lastCheckOffset) >> 1)) {
205 _firstCheckOffset = (_firstCheckOffset + _lastCheckOffset) >> 1;
206 }
207 else {
208 _lastCheckOffset = (_firstCheckOffset + _lastCheckOffset) >> 1;
209 if(_lastCheckOffset <= _firstCheckOffset + 1) {
210 _prepassLeft = 0;
211 }
212 }
213 }
214 for(uint64_t ii = _firstCheckOffset; ii < _inAccCount; ii++) {
215 float _stampTime = fastgltf::getAccessorElement<float>(*asset, _inAcc, ii);
216 if(_stampTime > _seconds) {
217 _lowerIndex = ii - 1;
218 break;
219 }
220 _lowerTimestamp = _stampTime;
221 }
222 }
223
224 fastgltf::math::fquat _lowerValue = fastgltf::getAccessorElement<fastgltf::math::fquat>(*asset, _outAcc, _lowerIndex);
225 if(_seconds >= _maxTime || _seconds <= 0.0f) {
226 return _lowerValue;
227 }
228 std::size_t _upperIndex = _lowerIndex + 1;
229 float _linDist = fastgltf::getAccessorElement<float>(*asset, _inAcc, _upperIndex) - _lowerTimestamp;
230 return fastgltf::math::slerp(_lowerValue, fastgltf::getAccessorElement<fastgltf::math::fquat>(*asset, _outAcc,
231 _upperIndex), (_seconds - _lowerTimestamp) / _linDist);
232}
