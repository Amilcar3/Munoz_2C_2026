58void lv_gltf_model_delete(lv_gltf_model_t * model)
59{
60 if(!model) {
61 return;
62 }
63
64 /* Each call drops one entry, and with it one viewer from the array being walked */
65 while(!lv_array_is_empty(&model->viewers)) {
66 lv_gltf_detach_model(*(lv_obj_t **)lv_array_at(&model->viewers, 0), model);
67 }
68
69 lv_timer_delete(model->animation_update_timer);
70 model->animation_update_timer = NULL;
71
72 lv_gltf_data_delete_textures(model);
73
74 /* The geometry buffers belong to the model. The material textures do not, they are
75 * owned and reused by the loader cache. */
76 for(auto & mesh : model->meshes) {
77 for(auto & primitive : mesh.primitives) {
78 GL_CALL(glDeleteVertexArrays(1, &primitive.vertexArray));
79 const GLuint buffers[2] = { primitive.vertexBuffer, primitive.indexBuffer };
80 GL_CALL(glDeleteBuffers(2, buffers));
81 }
82 }
83
84 uint32_t node_count = lv_array_size(&model->nodes);
85 for(uint32_t i = 0; i < node_count; ++i) {
86 lv_gltf_model_node_t * node = (lv_gltf_model_node_t *) lv_array_at(&model->nodes, i);
87 lv_gltf_model_node_deinit(node);
88 }
89 lv_array_deinit(&model->viewers);
90 lv_array_deinit(&model->nodes);
91
92 /* Explicitly call destructors for C++ objects initialized with placement new */
93 model->ibm_by_skin_then_node.~IbmBySkinThenNodeMap();
94 model->textures.~vector();
95 model->meshes.~vector();
96 model->node_by_light_index.~vector();
97 model->local_mesh_to_center_points_by_primitive.~map();
98 model->skin_tex.~vector();
99 model->validated_skins.~vector();
100 model->blended_nodes_by_material_index.~map();
101 model->opaque_nodes_by_material_index.~map();
102 model->transforms.~map();
103 model->channel_set_cache.~map();
104 model->asset.~Asset();
105
106 lv_free(model);
107}

157lv_result_t lv_gltf_model_play_animation(lv_gltf_model_t * model, size_t index)
158{
159 LV_CHECK_ARG(model != NULL, return LV_RESULT_INVALID);
160 LV_CHECK_ARG(index < model->asset.animations.size(), return LV_RESULT_INVALID);
161
162 if(lv_timer_get_paused(model->animation_update_timer)) {
163 model->last_tick = lv_tick_get();
164 lv_timer_resume(model->animation_update_timer);
165 }
166
167 model->current_animation_max_time = lv_gltf_data_get_animation_total_time(model, index);
168 model->current_animation = index;
169 model->is_animation_enabled = true;
170 return LV_RESULT_OK;
171}

211lv_gltf_model_t * lv_gltf_data_create_internal(const char * gltf_path,
212 fastgltf::Asset && asset)
213{
214 LV_ASSERT(gltf_path != NULL);
215 lv_gltf_model_t * data = (lv_gltf_model_t *)lv_zalloc(sizeof(*data));
216 LV_ASSERT_MALLOC(data);
217 new(data) lv_gltf_model_t;
218 new(&data->asset) fastgltf::Asset(std::move(asset));
219 data->filename = gltf_path;
220 data->last_anim_num = -5;
221 data->current_animation_max_time = 0;
222 data->local_timestamp = 0.0f;
223
224 data->animation_speed_ratio = LV_GLTF_ANIM_SPEED_NORMAL;
225 data->animation_update_timer = lv_timer_create(update_animation_cb, LV_DEF_REFR_PERIOD, data);
226 lv_timer_pause(data->animation_update_timer);
227 LV_ASSERT_NULL(data->animation_update_timer);
228
229 new(&data->transforms) NodeTransformMap();
230 new(&data->opaque_nodes_by_material_index) MaterialIndexMap();
231 new(&data->blended_nodes_by_material_index) MaterialIndexMap();
232 new(&data->validated_skins) LongVector();
233 new(&data->skin_tex) IntVector();
234 new(&data->local_mesh_to_center_points_by_primitive)
235 NodePrimCenterMap();
236 new(&data->node_by_light_index) NodeVector();
237 new(&data->meshes) std::vector<lv_gltf_mesh_data_t>();
238 new(&data->textures) std::vector<GLuint>();
239 new(&data->ibm_by_skin_then_node) std::map<int32_t, std::map<fastgltf::Node *, fastgltf::math::fmat4x4>>;
240
241 lv_array_init(&data->viewers, 1, sizeof(lv_gltf_t *));
242 return data;
243}

344static void update_animation_cb(lv_timer_t * timer)
345{
346 LV_ASSERT(timer != NULL);
347 lv_gltf_model_t * model = (lv_gltf_model_t *)lv_timer_get_user_data(timer);
348 LV_ASSERT(model != NULL);
349
350 const uint32_t current_tick = lv_tick_get();
351 const uint32_t delta = lv_tick_diff(current_tick, model->last_tick);
352
353 model->last_tick = current_tick;
354
355 uint64_t next_timestamp = model->local_timestamp + ((delta * model->animation_speed_ratio) / 1000);
356 if(next_timestamp >= model->current_animation_max_time) {
357 next_timestamp = LV_GLTF_ANIMATION_RESET_TIME;
358 }
359 lv_gltf_model_set_animation_time(model, (uint32_t)next_timestamp);
360}
