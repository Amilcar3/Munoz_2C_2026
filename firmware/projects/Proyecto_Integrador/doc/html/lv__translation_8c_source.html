49void lv_translation_deinit(void)
50{
51 lv_translation_pack_t * pack;
52 LV_LL_READ(&packs_ll, pack) {
53 if(pack->is_static == false) {
54 size_t i;
55 size_t trans_cnt = lv_array_size(&pack->translation_array);
56 for(i = 0; i < trans_cnt; i++) {
57 lv_translation_tag_dsc_t * tag = lv_array_at(&pack->translation_array, i);
58 lv_free((void *)tag->tag);
59
60 size_t j;
61 for(j = 0; j < pack->language_cnt; j++) {
62 lv_free((void *)tag->translations[j]); /*Free each translation of the tag*/
63 }
64 lv_free(tag->translations);
65 }
66
67 lv_array_deinit(&pack->translation_array);
68
69 for(i = 0; i < pack->language_cnt; i++) {
70 lv_free((void *)pack->languages[i]);
71 }
72 lv_free(pack->languages);
73 }
74 }
75
76 lv_ll_clear(&packs_ll);
77
78 lv_free((void *)selected_lang);
79 selected_lang = NULL;
80}

82lv_translation_pack_t * lv_translation_add_static(const char * const languages[], const char * const tags[],
83 const char * const translations[])
84{
85 LV_CHECK_ARG(languages != NULL, return NULL);
86 LV_CHECK_ARG(tags != NULL, return NULL);
87 LV_CHECK_ARG(translations != NULL, return NULL);
88
89 lv_translation_pack_t * pack = lv_ll_ins_head(&packs_ll);
90 LV_ASSERT_MALLOC(pack);
91 if(pack == NULL) return NULL;
92 lv_memzero(pack, sizeof(lv_translation_pack_t));
93 pack->is_static = 1;
94
95 /*Count the languages*/
96 while(languages[pack->language_cnt]) {
97 pack->language_cnt++;
98 }
99
100 pack->languages = (const char **)languages;
101 pack->tag_p = (const char **)tags;
102 pack->translation_p = (const char **)translations;
103 return pack;
104}

153lv_result_t lv_translation_add_language(lv_translation_pack_t * pack, const char * lang)
154{
155 LV_CHECK_ARG(pack != NULL, return LV_RESULT_INVALID);
156 LV_CHECK_ARG(lang != NULL, return LV_RESULT_INVALID);
157
158 if(pack->is_static) {
159 LV_LOG_WARN("Can't add language `%s` to static translation pack `%p`", lang, (void *)pack);
160 return LV_RESULT_INVALID;
161 }
162
163 pack->language_cnt++;
164 pack->languages = lv_realloc(pack->languages, sizeof(const char *) * pack->language_cnt);
165 LV_ASSERT_MALLOC(pack->languages);
166 if(pack->languages == NULL) {
167 LV_LOG_WARN("Couldn't allocate languages in `%p`", (void *)pack);
168 return LV_RESULT_INVALID;
169 }
170
171 pack->languages[pack->language_cnt - 1] = lv_strdup(lang);
172 LV_ASSERT_MALLOC(pack->languages[pack->language_cnt - 1]);
173 if(pack->languages[pack->language_cnt - 1] == NULL) {
174 LV_LOG_WARN("Couldn't allocate the new language in `%p`", (void *)pack);
175 return LV_RESULT_INVALID;
176 }
177
178 return LV_RESULT_OK;
179}

195lv_translation_tag_dsc_t * lv_translation_add_tag(lv_translation_pack_t * pack, const char * tag_name)
196{
197 LV_CHECK_ARG(pack != NULL, return NULL);
198 LV_CHECK_ARG(tag_name != NULL, return NULL);
199 LV_CHECK_ARG_FORMAT_MSG(!pack->is_static, return NULL, "Can't add tag `%s` to static translation pack", tag_name);
200
201 lv_translation_tag_dsc_t tag;
202 tag.tag = lv_strdup(tag_name);
203 LV_ASSERT_MALLOC(tag.tag);
204 tag.translations = lv_zalloc(pack->language_cnt * sizeof(const char *));
205 LV_ASSERT_MALLOC(tag.translations);
206
207 if(tag.tag == NULL || tag.translations == NULL) {
208 LV_LOG_WARN("Couldn't allocate memory for the tag's data in `%p`", (void *)pack);
209 lv_free((void *)tag.tag);
210 lv_free((void *)tag.translations);
211 return NULL;
212 }
213
214 lv_result_t res = lv_array_push_back(&pack->translation_array, &tag);
215
216 if(res != LV_RESULT_OK) {
217 LV_LOG_WARN("Couldn't add the tag in `%p`", (void *)pack);
218 lv_free((void *)tag.tag);
219 lv_free((void *)tag.translations);
220 return NULL;
221 }
222
223 return lv_array_back(&pack->translation_array);
224}

226lv_result_t lv_translation_set_tag_translation(lv_translation_pack_t * pack, lv_translation_tag_dsc_t * tag,
227 uint32_t lang_idx, const char * trans)
228{
229 LV_CHECK_ARG(pack != NULL, return LV_RESULT_INVALID);
230 LV_CHECK_ARG(tag != NULL, return LV_RESULT_INVALID);
231 LV_CHECK_ARG(trans != NULL, return LV_RESULT_INVALID);
232 LV_CHECK_ARG_FORMAT_MSG(!pack->is_static, return LV_RESULT_INVALID,
233 "Can't set tag translation `%s` in static translation pack",
234 trans);
235 LV_CHECK_ARG_FORMAT_MSG(lang_idx < pack->language_cnt, return LV_RESULT_INVALID,
236 "Can't set the translation for language %" LV_PRIu32 " as there are only %" LV_PRIu32 " languages defined",
237 lang_idx, pack->language_cnt);
238 LV_UNUSED(pack);
239
240 size_t new_len = lv_strlen(trans) + 1;
241 char * new_trans = lv_realloc((void *)tag->translations[lang_idx], new_len);
242 if(!new_trans) {
243 LV_LOG_WARN("Couldn't allocate the new translation in tag `%p` in pack `%p`", (void *)tag, (void *) pack);
244 return LV_RESULT_INVALID;
245 }
246 lv_strcpy(new_trans, trans);
247 tag->translations[lang_idx] = new_trans;
248 return LV_RESULT_OK;
249}

261static const char * get_translation(const char * tag)
262{
263 LV_ASSERT(tag != NULL);
264
265 if(selected_lang == NULL) {
266 LV_LOG_WARN("No language is selected to get the translation of `%s`", tag);
267 return tag;
268 }
269
270 lv_translation_pack_t * pack;
271 bool lang_found = false;
272 LV_LL_READ(&packs_ll, pack) {
273 uint32_t lang;
274 for(lang = 0; lang < pack->language_cnt; lang++) {
275 /*Does this pack contains the language?*/
276 if(lv_streq(pack->languages[lang], selected_lang)) {
277 lang_found = true;
278 /*Find the tag*/
279 if(pack->is_static) {
280 uint32_t t;
281 for(t = 0; pack->tag_p[t]; t++) {
282 if(lv_streq(pack->tag_p[t], tag)) {
283 /*Find the "row" of the tag */
284 const char ** tr_row = pack->translation_p + pack->language_cnt * t;
285 const char * tr = tr_row[lang];
286 if(tr) return tr; /*Found directly*/
287
288 LV_LOG_WARN("`%s` tag is not found. Using the tag as translation.", tag);
289 return tag; /*Return the tag as a fall back*/
290 }
291 }
292 }
293 else {
294 size_t trans_cnt = lv_array_size(&pack->translation_array);
295 size_t i;
296 for(i = 0; i < trans_cnt; i++) {
297 lv_translation_tag_dsc_t * tag_dsc = lv_array_at(&pack->translation_array, i);
298 if(lv_streq(tag_dsc->tag, tag)) {
299 const char * tr = tag_dsc->translations[lang];
300 if(tr) return tr; /*Found directly*/
301
302 LV_LOG_WARN("`%s` tag is not found. Using the tag as translation.", tag);
303 return tag; /*Return the tag as a worst case option*/
304 }
305 }
306 }
307 }
308 }
309 }
310
311 if(lang_found) {
312 LV_LOG_WARN("`%s` tag is not found, using the tag as translation.", tag);
313 }
314 else {
315 LV_LOG_WARN("`%s` language is not found, using the `%s` as translation.", selected_lang, tag);
316 }
317
318 return tag;
319}
