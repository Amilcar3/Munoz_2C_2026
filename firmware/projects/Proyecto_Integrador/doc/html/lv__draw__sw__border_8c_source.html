49void lv_draw_sw_border(lv_draw_task_t * t, const lv_draw_border_dsc_t * dsc, const lv_area_t * coords)
50{
51 if(dsc->opa <= LV_OPA_MIN) return;
52 if(dsc->width == 0) return;
53 if(dsc->side == LV_BORDER_SIDE_NONE) return;
54
55 int32_t coords_w = lv_area_get_width(coords);
56 int32_t coords_h = lv_area_get_height(coords);
57 int32_t rout = dsc->radius;
58 int32_t short_side = LV_MIN(coords_w, coords_h);
59 if(rout > short_side >> 1) rout = short_side >> 1;
60
61 /*Get the inner area*/
62 lv_area_t area_inner;
63 area_inner = *coords;
64 area_inner.x1 += ((dsc->side & LV_BORDER_SIDE_LEFT) ? dsc->width : - (dsc->width + rout));
65 area_inner.x2 -= ((dsc->side & LV_BORDER_SIDE_RIGHT) ? dsc->width : - (dsc->width + rout));
66 area_inner.y1 += ((dsc->side & LV_BORDER_SIDE_TOP) ? dsc->width : - (dsc->width + rout));
67 area_inner.y2 -= ((dsc->side & LV_BORDER_SIDE_BOTTOM) ? dsc->width : - (dsc->width + rout));
68
69 int32_t rin = rout - dsc->width;
70 if(rin < 0) rin = 0;
71
72 if(rout == 0 && rin == 0) {
73 draw_border_simple(t, coords, &area_inner, dsc->color, dsc->opa);
74 }
75 else {
76 draw_border_complex(t, coords, &area_inner, rout, rin, dsc->color, dsc->opa);
77 }
78
79}

85void draw_border_complex(lv_draw_task_t * t, const lv_area_t * outer_area, const lv_area_t * inner_area,
86 int32_t rout, int32_t rin, lv_color_t color, lv_opa_t opa)
87{
88#if LV_DRAW_SW_COMPLEX
89 /*Get clipped draw area which is the real draw area.
90 *It is always the same or inside `coords`*/
91 lv_area_t draw_area;
92 if(!lv_area_intersect(&draw_area, outer_area, &t->clip_area)) return;
93 int32_t draw_area_w = lv_area_get_width(&draw_area);
94
95 lv_draw_sw_blend_dsc_t blend_dsc;
96 lv_memzero(&blend_dsc, sizeof(blend_dsc));
97 lv_opa_t * mask_buf = lv_malloc(draw_area_w);
98 blend_dsc.mask_buf = mask_buf;
99
100 void * mask_list[3] = {0};
101
102 /*Create mask for the inner mask*/
103 lv_draw_sw_mask_radius_param_t mask_rin_param;
104 lv_draw_sw_mask_radius_init(&mask_rin_param, inner_area, rin, true);
105 mask_list[0] = &mask_rin_param;
106
107 /*Create mask for the outer area*/
108 lv_draw_sw_mask_radius_param_t mask_rout_param;
109 if(rout > 0) {
110 lv_draw_sw_mask_radius_init(&mask_rout_param, outer_area, rout, false);
111 mask_list[1] = &mask_rout_param;
112 }
113
114 int32_t h;
115 lv_area_t blend_area;
116 blend_dsc.blend_area = &blend_area;
117 blend_dsc.mask_area = &blend_area;
118 blend_dsc.color = color;
119 blend_dsc.opa = opa;
120
121 /*Calculate the x and y coordinates where the straight parts area is*/
122 lv_area_t core_area;
123 core_area.x1 = LV_MAX(outer_area->x1 + rout, inner_area->x1);
124 core_area.x2 = LV_MIN(outer_area->x2 - rout, inner_area->x2);
125 core_area.y1 = LV_MAX(outer_area->y1 + rout, inner_area->y1);
126 core_area.y2 = LV_MIN(outer_area->y2 - rout, inner_area->y2);
127 int32_t core_w = lv_area_get_width(&core_area);
128
129 bool top_side = outer_area->y1 <= inner_area->y1;
130 bool bottom_side = outer_area->y2 >= inner_area->y2;
131
132 /*No masks*/
133 bool left_side = outer_area->x1 <= inner_area->x1;
134 bool right_side = outer_area->x2 >= inner_area->x2;
135
136 bool split_hor = true;
137 if(left_side && right_side && top_side && bottom_side &&
138 core_w < SPLIT_LIMIT) {
139 split_hor = false;
140 }
141
142 blend_dsc.mask_res = LV_DRAW_SW_MASK_RES_FULL_COVER;
143 /*Draw the straight lines first if they are long enough*/
144 if(top_side && split_hor) {
145 blend_area.x1 = core_area.x1;
146 blend_area.x2 = core_area.x2;
147 blend_area.y1 = outer_area->y1;
148 blend_area.y2 = inner_area->y1 - 1;
149 lv_draw_sw_blend(t, &blend_dsc);
150 }
151
152 if(bottom_side && split_hor) {
153 blend_area.x1 = core_area.x1;
154 blend_area.x2 = core_area.x2;
155 blend_area.y1 = inner_area->y2 + 1;
156 blend_area.y2 = outer_area->y2;
157 lv_draw_sw_blend(t, &blend_dsc);
158 }
159
160 /*If the border is very thick and the vertical sides overlap horizontally draw a single rectangle*/
161 if(inner_area->x1 >= inner_area->x2 && left_side && right_side) {
162 blend_area.x1 = outer_area->x1;
163 blend_area.x2 = outer_area->x2;
164 blend_area.y1 = core_area.y1;
165 blend_area.y2 = core_area.y2;
166 lv_draw_sw_blend(t, &blend_dsc);
167 }
168 else {
169 if(left_side) {
170 blend_area.x1 = outer_area->x1;
171 blend_area.x2 = inner_area->x1 - 1;
172 blend_area.y1 = core_area.y1;
173 blend_area.y2 = core_area.y2;
174 lv_draw_sw_blend(t, &blend_dsc);
175 }
176
177 if(right_side) {
178 blend_area.x1 = inner_area->x2 + 1;
179 blend_area.x2 = outer_area->x2;
180 blend_area.y1 = core_area.y1;
181 blend_area.y2 = core_area.y2;
182 lv_draw_sw_blend(t, &blend_dsc);
183 }
184 }
185
186 /*Draw the corners*/
187 int32_t blend_w;
188
189 /*Left and right corner together if they are close to each other*/
190 if(!split_hor) {
191 /*Calculate the top corner and mirror it to the bottom*/
192 blend_area.x1 = draw_area.x1;
193 blend_area.x2 = draw_area.x2;
194 int32_t max_h = LV_MAX(rout, inner_area->y1 - outer_area->y1);
195 for(h = 0; h < max_h; h++) {
196 int32_t top_y = outer_area->y1 + h;
197 int32_t bottom_y = outer_area->y2 - h;
198 if(top_y < draw_area.y1 && bottom_y > draw_area.y2) continue; /*This line is clipped now*/
199
200 lv_memset(mask_buf, 0xff, draw_area_w);
201 blend_dsc.mask_res = lv_draw_sw_mask_apply(mask_list, mask_buf, blend_area.x1, top_y, draw_area_w);
202
203 if(top_y >= draw_area.y1) {
204 blend_area.y1 = top_y;
205 blend_area.y2 = top_y;
206 lv_draw_sw_blend(t, &blend_dsc);
207 }
208
209 if(bottom_y <= draw_area.y2) {
210 blend_area.y1 = bottom_y;
211 blend_area.y2 = bottom_y;
212 lv_draw_sw_blend(t, &blend_dsc);
213 }
214 }
215 }
216 else {
217 /*Left corners*/
218 blend_area.x1 = draw_area.x1;
219 blend_area.x2 = LV_MIN(draw_area.x2, core_area.x1 - 1);
220 blend_w = lv_area_get_width(&blend_area);
221 if(blend_w > 0) {
222 if(left_side || top_side) {
223 for(h = draw_area.y1; h < core_area.y1; h++) {
224 blend_area.y1 = h;
225 blend_area.y2 = h;
226
227 lv_memset(mask_buf, 0xff, blend_w);
228 blend_dsc.mask_res = lv_draw_sw_mask_apply(mask_list, mask_buf, blend_area.x1, h, blend_w);
229 lv_draw_sw_blend(t, &blend_dsc);
230 }
231 }
232
233 if(left_side || bottom_side) {
234 for(h = core_area.y2 + 1; h <= draw_area.y2; h++) {
235 blend_area.y1 = h;
236 blend_area.y2 = h;
237
238 lv_memset(mask_buf, 0xff, blend_w);
239 blend_dsc.mask_res = lv_draw_sw_mask_apply(mask_list, mask_buf, blend_area.x1, h, blend_w);
240 lv_draw_sw_blend(t, &blend_dsc);
241 }
242 }
243 }
244
245 /*Right corners*/
246 blend_area.x1 = LV_MAX(draw_area.x1, blend_area.x2 + 1); /*To not overlap with the left side*/
247 blend_area.x1 = LV_MAX(draw_area.x1, core_area.x2 + 1);
248
249 blend_area.x2 = draw_area.x2;
250 blend_w = lv_area_get_width(&blend_area);
251
252 if(blend_w > 0) {
253 if(right_side || top_side) {
254 for(h = draw_area.y1; h < core_area.y1; h++) {
255 blend_area.y1 = h;
256 blend_area.y2 = h;
257
258 lv_memset(mask_buf, 0xff, blend_w);
259 blend_dsc.mask_res = lv_draw_sw_mask_apply(mask_list, mask_buf, blend_area.x1, h, blend_w);
260 lv_draw_sw_blend(t, &blend_dsc);
261 }
262 }
263
264 if(right_side || bottom_side) {
265 for(h = core_area.y2 + 1; h <= draw_area.y2; h++) {
266 blend_area.y1 = h;
267 blend_area.y2 = h;
268
269 lv_memset(mask_buf, 0xff, blend_w);
270 blend_dsc.mask_res = lv_draw_sw_mask_apply(mask_list, mask_buf, blend_area.x1, h, blend_w);
271 lv_draw_sw_blend(t, &blend_dsc);
272 }
273 }
274 }
275 }
276
277 lv_draw_sw_mask_free_param(&mask_rin_param);
278 if(rout > 0) lv_draw_sw_mask_free_param(&mask_rout_param);
279 lv_free(mask_buf);
280
281#else
282 LV_UNUSED(t);
283 LV_UNUSED(outer_area);
284 LV_UNUSED(inner_area);
285 LV_UNUSED(rout);
286 LV_UNUSED(rin);
287 LV_UNUSED(color);
288 LV_UNUSED(opa);
289#endif /*LV_DRAW_SW_COMPLEX*/
290}
291static void draw_border_simple(lv_draw_task_t * t, const lv_area_t * outer_area, const lv_area_t * inner_area,
292 lv_color_t color, lv_opa_t opa)
293{
294 lv_area_t a;
295 lv_draw_sw_blend_dsc_t blend_dsc;
296 lv_memzero(&blend_dsc, sizeof(lv_draw_sw_blend_dsc_t));
297 blend_dsc.blend_area = &a;
298 blend_dsc.color = color;
299 blend_dsc.opa = opa;
300
301 bool top_side = outer_area->y1 <= inner_area->y1;
302 bool bottom_side = outer_area->y2 >= inner_area->y2;
303 bool left_side = outer_area->x1 <= inner_area->x1;
304 bool right_side = outer_area->x2 >= inner_area->x2;
305
306 /*Top*/
307 a.x1 = outer_area->x1;
308 a.x2 = outer_area->x2;
309 a.y1 = outer_area->y1;
310 a.y2 = inner_area->y1 - 1;
311 if(top_side) {
312 lv_draw_sw_blend(t, &blend_dsc);
313 }
314
315 /*Bottom*/
316 a.y1 = inner_area->y2 + 1;
317 a.y2 = outer_area->y2;
318 if(bottom_side) {
319 lv_draw_sw_blend(t, &blend_dsc);
320 }
321
322 /*Left*/
323 a.x1 = outer_area->x1;
324 a.x2 = inner_area->x1 - 1;
325 a.y1 = (top_side) ? inner_area->y1 : outer_area->y1;
326 a.y2 = (bottom_side) ? inner_area->y2 : outer_area->y2;
327 if(left_side) {
328 lv_draw_sw_blend(t, &blend_dsc);
329 }
330
331 /*Right*/
332 a.x1 = inner_area->x2 + 1;
333 a.x2 = outer_area->x2;
334 if(right_side) {
335 lv_draw_sw_blend(t, &blend_dsc);
336 }
337}
