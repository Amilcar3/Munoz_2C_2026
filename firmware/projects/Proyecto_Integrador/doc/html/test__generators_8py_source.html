22def write_enum(directory: Path, body: str) -> Path:
23 """A header holding one typedef enum, as parse_enum expects to find it."""
24 path = directory / "lv_test.h"
25 path.write_text("typedef enum {\n" + body + "\n} lv_test_t;\n")
26 return path

29class TestParseEnum(unittest.TestCase):
30 def test_a_trailing_comment_is_not_part_of_the_value(self):
31 with tempfile.TemporaryDirectory() as tmp:
32 path = write_enum(Path(tmp), """
33 LV_TEST_A = 0x10, /** the first one, which runs
34 * on to the next line */
35 LV_TEST_B = 0x20,
36 """)
37 self.assertEqual(
38 parse_enum(path, "lv_test_t", "LV_TEST_"),
39 {0x10: "A", 0x20: "B"},
40 )
41
42 def test_a_comment_between_operands_does_not_truncate_the_value(self):
43 with tempfile.TemporaryDirectory() as tmp:
44 path = write_enum(Path(tmp), """
45 LV_TEST_A = 0x10,
46 LV_TEST_B = LV_TEST_A /* keep going */ | 0x01,
47 """)
48 self.assertEqual(
49 parse_enum(path, "lv_test_t", "LV_TEST_"),
50 {0x10: "A", 0x11: "B"},

53 def test_a_skipped_member_can_still_be_referenced(self):
54 with tempfile.TemporaryDirectory() as tmp:
55 path = write_enum(Path(tmp), """
56 LV_TEST_A = 0x10,
57 LV_TEST_HIDDEN = 0x20,
58 LV_TEST_B = LV_TEST_HIDDEN | 0x01,
59 """)
60 entries = parse_enum(path, "lv_test_t", "LV_TEST_",
61 skip={"LV_TEST_HIDDEN"})
62 self.assertEqual(entries, {0x10: "A", 0x21: "B"})

73class TestLvglPaths(unittest.TestCase):
74 def make_tree(self, tmp: str, *files: str) -> Path:
75 root = Path(tmp)
76 for name in files:
77 path = root / name
78 path.parent.mkdir(parents=True, exist_ok=True)
79 path.write_text("")
80 return root
81
82 def test_a_checkout_is_found(self):
83 with tempfile.TemporaryDirectory() as tmp:
84 root = self.make_tree(tmp, "lv_version.h", "src/lv_init.c")
85 self.assertTrue(_is_lvgl(root))
