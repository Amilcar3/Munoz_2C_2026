71void lv_nanovg_image_cache_init(struct _lv_draw_nanovg_unit_t * u)
72{
73 LV_ASSERT_NULL(u);
74 LV_ASSERT(u->image_cache == NULL);
75 LV_ASSERT(u->image_pending == NULL);
76
77 const lv_cache_ops_t ops = {
78 .compare_cb = (lv_cache_compare_cb_t)image_compare_cb,
79 .create_cb = (lv_cache_create_cb_t)image_create_cb,
80 .free_cb = (lv_cache_free_cb_t)image_free_cb,
81 };
82
83 u->image_cache = lv_cache_create(&lv_cache_class_lru_rb_count, sizeof(image_item_t), LV_NANOVG_IMAGE_CACHE_CNT, ops);
84 lv_cache_set_name(u->image_cache, "NVG_IMAGE");
85 u->image_pending = lv_pending_create(sizeof(lv_cache_entry_t *), 4);
86 lv_pending_set_free_cb(u->image_pending, image_cache_release_cb, u->image_cache);
87
88 /* Non-cached (modifiable) images upload a fresh texture each frame; track their
89 * handles here so they're deleted at end of frame (same lifecycle as image_pending). */
90 u->image_uncached_pending = lv_pending_create(sizeof(int), 4);
91 lv_pending_set_free_cb(u->image_uncached_pending, image_uncached_free_cb, u);
92
93 lv_ll_init(&u->image_drop_ll, sizeof(image_item_t));
94}

112int lv_nanovg_image_cache_get_handle(struct _lv_draw_nanovg_unit_t * u,
113 const void * src,
114 int image_flags,
115 lv_image_header_t * header)
116{
117 LV_PROFILER_DRAW_BEGIN;
118
119 LV_ASSERT_NULL(u);
120 LV_ASSERT_NULL(src);
121
122 lv_image_decoder_args_t args;
123 lv_memzero(&args, sizeof(lv_image_decoder_args_t));
124
125 lv_image_decoder_dsc_t decoder_dsc;
126 lv_result_t res = lv_image_decoder_open(&decoder_dsc, src, &args);
127 if(res != LV_RESULT_OK) {
128 lv_image_src_t type = lv_image_src_get_type(src);
129 LV_UNUSED(type);
130 LV_LOG_WARN("Failed to open image: type: %d, src: %p (%s)", type, src,
131 type == LV_IMAGE_SRC_FILE ? (const char *)src : "var");
132 LV_PROFILER_DRAW_END;
133 return -1;
134 }
135
136 const lv_draw_buf_t * decoded = decoder_dsc.decoded;
137 if(decoded == NULL || decoded->data == NULL) {
138 lv_image_decoder_close(&decoder_dsc);
139 LV_LOG_ERROR("image data is NULL");
140 LV_PROFILER_DRAW_END;
141 return -1;
142 }
143
144 if(header) {
145 *header = decoder_dsc.header;
146 }
147
148 image_item_t item = { 0 };
149 item.u = u;
150 item.src_buf = *decoded;
151 item.image_flags = image_flags;
152 item.src = src;
153 item.src_type = lv_image_src_get_type(src);
154 item.image_handle = -1; /* Mark as not yet created */
155
156 /* Cache the GPU texture only for sources the image decoder cached (stable identity).
157 * "Use directly" sources (canvas/snapshot/lottie draw buffers) have no cache_entry:
158 * their pixels can change in place at the same address, so caching would go stale. */
159 if(!decoder_dsc.cache_entry) {
160 int image_handle = image_get_handle_uncached(u, &item, src);
161 lv_image_decoder_close(&decoder_dsc);
162 LV_PROFILER_DRAW_END;
163 return image_handle;
164 }
165
166 lv_cache_entry_t * cache_node_entry = lv_cache_acquire(u->image_cache, &item, NULL);
167 if(cache_node_entry == NULL) {
168 /* check if the cache is full */
169 size_t free_size = lv_cache_get_free_size(u->image_cache, NULL);
170 if(free_size == 0) {
171 LV_LOG_INFO("image cache is full, release all pending cache entries");
172 lv_nanovg_end_frame(u);
173 }
174
175 cache_node_entry = lv_cache_acquire_or_create(u->image_cache, &item, NULL);
176 if(cache_node_entry == NULL) {
177 LV_LOG_ERROR("image cache creating failed");
178 lv_image_decoder_close(&decoder_dsc);
179 LV_PROFILER_DRAW_END;
180 return -1;
181 }
182 }
183
184 lv_image_decoder_close(&decoder_dsc);
185
186 /* Add the new entry to the pending list */
187 lv_pending_add(u->image_pending, &cache_node_entry);
188
189 image_item_t * image_item = lv_cache_entry_get_data(cache_node_entry);
190
191 LV_PROFILER_DRAW_END;
192 return image_item->image_handle;
193}

195void lv_nanovg_image_cache_drop(struct _lv_draw_nanovg_unit_t * u, const void * src)
196{
197 LV_ASSERT_NULL(u);
198 LV_UNUSED(src);
199 if(src == NULL) {
200 lv_cache_drop_all(u->image_cache, NULL);
201 return;
202 }
203
204 u->image_drop_src = src;
205
206 lv_iter_t * iter = lv_cache_iter_create(u->image_cache);
207 LV_ASSERT_NULL(iter);
208
209 /* Collect all cache entries that match the drop source */
210 lv_iter_inspect(iter, image_cache_drop_collect_cb);
211
212 image_item_t * drop_item;
213 LV_LL_READ(&u->image_drop_ll, drop_item) {
214 lv_cache_drop(u->image_cache, drop_item, NULL);
215 }
216
217 lv_ll_clear(&u->image_drop_ll);
218 lv_iter_destroy(iter);
219 u->image_drop_src = NULL;
220}

248static int image_get_handle_uncached(struct _lv_draw_nanovg_unit_t * u, image_item_t * item, const void * src)
249{
250 if(!create_image_item(item)) {
251 return -1;
252 }
253
254 /* Queue the fresh texture for deletion at end of frame. */
255 lv_pending_add(u->image_uncached_pending, &item->image_handle);
256
257 /* create_image_item() strdup's the path for FILE sources so a cached entry can own
258 * it. This item is not cached, so free that copy to avoid leaking it. */
259 if(item->src_type == LV_IMAGE_SRC_FILE && item->src != src) {
260 lv_free((void *)item->src);
261 item->src = NULL;
262 }
263
264 return item->image_handle;
265}

273static bool create_image_item(image_item_t * item)
274{
275 const uint32_t w = item->src_buf.header.w;
276 const uint32_t h = item->src_buf.header.h;
277 const lv_color_format_t cf = item->src_buf.header.cf;
278 const uint32_t stride = item->src_buf.header.stride;
279 enum NVGtexture nvg_tex_type = NVG_TEXTURE_BGRA;
280 item->image_handle = -1;
281
282 /* Determine texture type and pixel size based on color format */
283 switch(cf) {
284 case LV_COLOR_FORMAT_A8:
285 nvg_tex_type = NVG_TEXTURE_ALPHA;
286 break;
287
288 case LV_COLOR_FORMAT_ARGB8888:
289 case LV_COLOR_FORMAT_ARGB8888_PREMULTIPLIED:
290 nvg_tex_type = NVG_TEXTURE_BGRA;
291 break;
292
293 case LV_COLOR_FORMAT_XRGB8888:
294 nvg_tex_type = NVG_TEXTURE_BGRX;
295 break;
296
297 case LV_COLOR_FORMAT_RGB888:
298 nvg_tex_type = NVG_TEXTURE_BGR;
299 break;
300
301 case LV_COLOR_FORMAT_RGB565:
302 nvg_tex_type = NVG_TEXTURE_RGB565;
303 break;
304
305 default:
306 LV_LOG_ERROR("Unsupported image format: %d", cf);
307 return false;
308 }
309
310 void * data = NULL;
311
312 /* Check if stride is tightly packed */
313 uint32_t tight_stride = (w * lv_color_format_get_bpp(cf) + 7) >> 3;
314 if(stride == tight_stride) {
315 /* Stride matches, use source buffer directly (zero-copy) */
316 data = lv_draw_buf_goto_xy(&item->src_buf, 0, 0);
317 LV_LOG_TRACE("Image stride matches: %" LV_PRIu32, stride);
318 }
319 else {
320 /* Stride doesn't match, need to copy with tight alignment */
321 lv_draw_buf_t * tmp_buf = lv_nanovg_reshape_global_image(item->u, cf, w, h);
322 if(!tmp_buf) {
323 LV_LOG_ERROR("Failed to allocate temp buffer for stride conversion");
324 return false;
325 }
326
327 lv_draw_buf_copy(tmp_buf, NULL, &item->src_buf, NULL);
328 data = lv_draw_buf_goto_xy(tmp_buf, 0, 0);
329 LV_LOG_TRACE("Image stride converted: %" LV_PRIu32 " -> %" LV_PRIu32, stride, tight_stride);
330 }
331
332 int flags = item->image_flags;
333 if(cf == LV_COLOR_FORMAT_ARGB8888_PREMULTIPLIED
334 || lv_draw_buf_has_flag(&item->src_buf, LV_IMAGE_FLAGS_PREMULTIPLIED)) {
335 flags |= NVG_IMAGE_PREMULTIPLIED;
336 }
337
338 LV_PROFILER_DRAW_BEGIN_TAG("nvgCreateImage");
339 int image_handle = nvgCreateImage(item->u->vg, w, h, flags, nvg_tex_type, data);
340 LV_PROFILER_DRAW_END_TAG("nvgCreateImage");
341
342 /* nvgCreateImage() returns 0 on failure in the GL backends; valid handles are > 0. */
343 if(image_handle <= 0) {
344 return false;
345 }
346
347 if(item->src_type == LV_IMAGE_SRC_FILE) {
348 item->src = lv_strdup(item->src);
349 LV_ASSERT_MALLOC(item->src);
350 }
351
352 item->image_handle = image_handle;
353 return true;
354}

356static void image_free_cb(image_item_t * item, void * user_data)
357{
358 LV_UNUSED(user_data);
359 LV_PROFILER_DRAW_BEGIN;
360 LV_LOG_TRACE("image_handle: %d", item->image_handle);
361
362 /* free_cb also runs for entries whose create_cb failed (e.g. unsupported format):
363 * those keep image_handle == -1 and a borrowed src pointer that must not be freed.
364 * nvgCreateImage() returns 0 on failure, so a created image always has handle > 0. */
365 bool created = (item->image_handle > 0);
366
367 nvgDeleteImage(item->u->vg, item->image_handle); /* safe no-op for invalid handles */
368 item->image_handle = -1;
369
370 /* For FILE entries src is strdup'd in create_cb, so only free it when create succeeded. */
371 if(item->src_type == LV_IMAGE_SRC_FILE && created) {
372 lv_free((void *)item->src);
373 }
374 item->src = NULL;
375 item->src_type = LV_IMAGE_SRC_UNKNOWN;
376
377 LV_PROFILER_DRAW_END;
378}

380static lv_cache_compare_res_t image_compare_cb(const image_item_t * lhs, const image_item_t * rhs)
381{
382 if(lhs->image_flags != rhs->image_flags) {
383 return lhs->image_flags > rhs->image_flags ? 1 : -1;
384 }
385
386 int cmp_res = lv_memcmp(&lhs->src_buf, &rhs->src_buf, sizeof(lv_draw_buf_t));
387 if(cmp_res != 0) {
388 return cmp_res > 0 ? 1 : -1;
389 }
390
391 return 0;
392}

394static void image_cache_drop_collect_cb(void * elem)
395{
401 LV_ASSERT_NULL(elem);
402 image_item_t * item = elem;
403 const void * src = item->u->image_drop_src;
404 LV_ASSERT_NULL(src);
405
406 /* A live cache entry always has a valid src (set in get_handle/create_cb). */
407 LV_ASSERT_NULL(item->src);
408
409 /* Use the entry's own src_type to decide comparison method.
410 * Using the incoming src's type would incorrectly strcmp on VARIABLE entries
411 * whose src pointer may have been freed by lv_draw_buf_destroy(). */
412 if((item->src_type == LV_IMAGE_SRC_FILE && lv_image_src_get_type(src) == LV_IMAGE_SRC_FILE
413 && lv_strcmp(item->src, src) == 0)
414 || (item->src_type == LV_IMAGE_SRC_VARIABLE && item->src == src)) {
415 image_item_t * drop_item = lv_ll_ins_tail(&item->u->image_drop_ll);
416 LV_ASSERT_MALLOC(drop_item);
417 *drop_item = *item;
418 }
419}
