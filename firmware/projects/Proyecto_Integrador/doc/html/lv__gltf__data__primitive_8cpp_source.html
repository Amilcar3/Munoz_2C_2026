69fastgltf::math::fvec4 lv_gltf_get_primitive_centerpoint(lv_gltf_model_t * data,
70 fastgltf::Mesh & mesh,
71 uint32_t prim_num)
72{
73 LV_ASSERT(data != NULL);
74 fastgltf::math::fvec4 result{ 0.f };
75 fastgltf::math::fvec3 v_min{ 999999999.f };
76 fastgltf::math::fvec3 v_max{ -999999999.f };
77 fastgltf::math::fvec3 v_cen{ 0.f };
78 float radius = 0.f;
79
80 if(mesh.primitives.size() <= prim_num) {
81 return result;
82 }
83 const auto & it = mesh.primitives[prim_num];
84 const auto & asset = data->asset;
85
86 const auto * positionIt = it.findAttribute("POSITION");
87 const auto & positionAccessor =
88 asset.accessors[positionIt->accessorIndex];
89 if(!positionAccessor.bufferViewIndex.has_value()) {
90 return result;
91 }
92
93 if(!(positionAccessor.min.has_value() &&
94 positionAccessor.max.has_value())) {
95 LV_LOG_ERROR(
96 "Could not get primitive center point. Missing min/max values");
97 return result;
98 }
99
100 fastgltf::math::fvec4 t_min{
101 (float)(positionAccessor.min.value().get<double>((size_t)0)),
102 (float)(positionAccessor.min.value().get<double>((size_t)1)),
103 (float)(positionAccessor.min.value().get<double>((size_t)2)),
104 0.f
105 };
106 fastgltf::math::fvec4 t_max{
107 (float)(positionAccessor.max.value().get<double>((size_t)0)),
108 (float)(positionAccessor.max.value().get<double>((size_t)1)),
109 (float)(positionAccessor.max.value().get<double>((size_t)2)),
110 0.f
111 };
112
113 v_max[0] = LV_MAX(t_min.x(), t_max.x());
114 v_max[1] = LV_MAX(t_min.y(), t_max.y());
115 v_max[2] = LV_MAX(t_min.z(), t_max.z());
116 v_min[0] = LV_MIN(t_min.x(), t_max.x());
117 v_min[1] = LV_MIN(t_min.y(), t_max.y());
118 v_min[2] = LV_MIN(t_min.z(), t_max.z());
119 v_cen[0] = (v_max[0] + v_min[0]) / 2.0f;
120 v_cen[1] = (v_max[1] + v_min[1]) / 2.0f;
121 v_cen[2] = (v_max[2] + v_min[2]) / 2.0f;
122 float size_x = v_max[0] - v_min[0];
123 float size_y = v_max[1] - v_min[1];
124 float size_z = v_max[2] - v_min[2];
125 radius = std::sqrt((size_x * size_x) + (size_y * size_y) +
126 (size_z * size_z)) /
127 2.0f;
128 result[0] = v_cen[0];
129 result[1] = v_cen[1];
130 result[2] = v_cen[2];
131 result[3] = radius;
132 return result;
133}
