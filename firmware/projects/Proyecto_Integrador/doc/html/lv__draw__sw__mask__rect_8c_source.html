41void lv_draw_sw_mask_rect(lv_draw_task_t * t, const lv_draw_mask_rect_dsc_t * dsc)
42{
43 lv_area_t draw_area;
44 if(!lv_area_intersect(&draw_area, &dsc->area, &t->clip_area)) {
45 return;
46 }
47
48 lv_layer_t * target_layer = t->target_layer;
49 lv_area_t * buf_area = &target_layer->buf_area;
50 lv_area_t clear_area;
51
52 void * draw_buf = target_layer->draw_buf;
53
54 if(dsc->keep_outside == 0) {
55 /*Clear the top part*/
56 lv_area_set(&clear_area, t->clip_area.x1, t->clip_area.y1, t->clip_area.x2,
57 dsc->area.y1 - 1);
58 lv_area_move(&clear_area, -buf_area->x1, -buf_area->y1);
59 lv_draw_buf_clear(draw_buf, &clear_area);
60
61 /*Clear the bottom part*/
62 lv_area_set(&clear_area, t->clip_area.x1, dsc->area.y2 + 1, t->clip_area.x2,
63 t->clip_area.y2);
64 lv_area_move(&clear_area, -buf_area->x1, -buf_area->y1);
65 lv_draw_buf_clear(draw_buf, &clear_area);
66
67 /*Clear the left part*/
68 lv_area_set(&clear_area, t->clip_area.x1, dsc->area.y1, dsc->area.x1 - 1, dsc->area.y2);
69 lv_area_move(&clear_area, -buf_area->x1, -buf_area->y1);
70 lv_draw_buf_clear(draw_buf, &clear_area);
71
72 /*Clear the right part*/
73 lv_area_set(&clear_area, dsc->area.x2 + 1, dsc->area.y1, t->clip_area.x2, dsc->area.y2);
74 lv_area_move(&clear_area, -buf_area->x1, -buf_area->y1);
75 lv_draw_buf_clear(draw_buf, &clear_area);
76 }
77
78 /*The rows below are addressed relative to the buffer, so clamp the draw area to it*/
79 if(!lv_area_intersect(&draw_area, &draw_area, buf_area)) {
80 return;
81 }
82
83 lv_draw_sw_mask_radius_param_t param;
84 lv_draw_sw_mask_radius_init(&param, &dsc->area, dsc->radius, false);
85
86 void * masks[2] = {0};
87 masks[0] = &param;
88
89 uint32_t area_w = lv_area_get_width(&draw_area);
90 lv_opa_t * mask_buf = lv_malloc(area_w);
91
92 int32_t y;
93 for(y = draw_area.y1; y <= draw_area.y2; y++) {
94 lv_memset(mask_buf, 0xff, area_w);
95 lv_draw_sw_mask_res_t res = lv_draw_sw_mask_apply(masks, mask_buf, draw_area.x1, y, area_w);
96 if(res == LV_DRAW_SW_MASK_RES_FULL_COVER) continue;
97
98 lv_color32_t * c32_buf = lv_draw_layer_go_to_xy(target_layer, draw_area.x1 - buf_area->x1,
99 y - buf_area->y1);
100
101 if(res == LV_DRAW_SW_MASK_RES_TRANSP) {
102 lv_memzero(c32_buf, area_w * sizeof(lv_color32_t));
103 }
104 else {
105 uint32_t i;
106 for(i = 0; i < area_w; i++) {
107 if(mask_buf[i] != LV_OPA_COVER) {
108 c32_buf[i].alpha = LV_OPA_MIX2(c32_buf[i].alpha, mask_buf[i]);
109 }
110 }
111 }
112 }
113
114 lv_free(mask_buf);
115 lv_draw_sw_mask_free_param(&param);
116}
