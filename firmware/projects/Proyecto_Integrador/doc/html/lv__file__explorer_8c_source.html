95#if LV_FILE_EXPLORER_QUICK_ACCESS
96void lv_file_explorer_set_quick_access_path(lv_obj_t * obj, lv_file_explorer_dir_t dir, const char * path)
97{
98 LV_LOG_DEPRECATED(LV_FILE_EXPLORER_DEPRECATED_MSG);
99 LV_CHECK_OBJ(obj, MY_CLASS, return);
100 LV_CHECK_ARG(path != NULL, return);
101
102 lv_file_explorer_t * explorer = (lv_file_explorer_t *)obj;
103
104 /*If path is unavailable */
105 if(lv_strlen(path) <= 0) return;
106
107 char ** dir_str = NULL;
108 switch(dir) {
109 case LV_EXPLORER_HOME_DIR:
110 dir_str = &(explorer->home_dir);
111 break;
112 case LV_EXPLORER_MUSIC_DIR:
113 dir_str = &(explorer->music_dir);
114 break;
115 case LV_EXPLORER_PICTURES_DIR:
116 dir_str = &(explorer->pictures_dir);
117 break;
118 case LV_EXPLORER_VIDEO_DIR:
119 dir_str = &(explorer->video_dir);
120 break;
121 case LV_EXPLORER_DOCS_DIR:
122 dir_str = &(explorer->docs_dir);
123 break;
124 case LV_EXPLORER_FS_DIR:
125 dir_str = &(explorer->fs_dir);
126 break;
127
128 default:
129 return;
130 break;
131 }
132
133 /*Free the old text*/
134 if(*dir_str != NULL) {
135 lv_free(*dir_str);
136 *dir_str = NULL;
137 }
138
139 /*Allocate space for the new text*/
140 *dir_str = lv_strdup(path);
141}

478#if LV_FILE_EXPLORER_QUICK_ACCESS
479static void quick_access_event_handler(lv_event_t * e)
480{
481 lv_event_code_t code = lv_event_get_code(e);
482 lv_obj_t * btn = lv_event_get_current_target(e);
483 lv_obj_t * obj = lv_event_get_user_data(e);
484
485 lv_file_explorer_t * explorer = (lv_file_explorer_t *)obj;
486
487 if(code == LV_EVENT_CLICKED) {
488 char ** path = NULL;
489 lv_obj_t * label = lv_obj_get_child(btn, -1);
490 char * label_text = lv_label_get_text(label);
491
492 if((lv_strcmp(label_text, LV_SYMBOL_HOME " HOME") == 0)) {
493 path = &(explorer->home_dir);
494 }
495 else if((lv_strcmp(label_text, LV_SYMBOL_VIDEO " Video") == 0)) {
496 path = &(explorer->video_dir);
497 }
498 else if((lv_strcmp(label_text, LV_SYMBOL_IMAGE " Pictures") == 0)) {
499 path = &(explorer->pictures_dir);
500 }
501 else if((lv_strcmp(label_text, LV_SYMBOL_AUDIO " Music") == 0)) {
502 path = &(explorer->music_dir);
503 }
504 else if((lv_strcmp(label_text, LV_SYMBOL_FILE " Documents") == 0)) {
505 path = &(explorer->docs_dir);
506 }
507 else if((lv_strcmp(label_text, LV_SYMBOL_DRIVE " File System") == 0)) {
508 path = &(explorer->fs_dir);
509 }
510
511 if(path != NULL)
512 show_dir(obj, *path);
513 }
514}

533static void browser_file_event_handler(lv_event_t * e)
534{
535 lv_event_code_t code = lv_event_get_code(e);
536 lv_obj_t * obj = lv_event_get_user_data(e);
537
538 lv_file_explorer_t * explorer = (lv_file_explorer_t *)obj;
539
540 lv_indev_t * indev = lv_indev_active();
541 lv_indev_type_t type = indev != NULL ? lv_indev_get_type(indev) : LV_INDEV_TYPE_NONE;
542 lv_event_code_t active_code = type == LV_INDEV_TYPE_POINTER ||
543 type == LV_INDEV_TYPE_BUTTON ? LV_EVENT_VALUE_CHANGED : LV_EVENT_CLICKED;
544
545 if(code == active_code) {
546 char file_name[LV_FILE_EXPLORER_PATH_MAX_LEN];
547 const char * selected_text = NULL;
548 const lv_file_explorer_file_table_entry_data_t * file_entry_user_data = NULL;
549 uint32_t row;
550 uint32_t col;
551 uint8_t navigate_to_parent_dir = 0;
552 uint8_t navigate_to_child = 0;
553
554 lv_memzero(file_name, sizeof(file_name));
555 lv_table_get_selected_cell(explorer->file_table, &row, &col);
556 selected_text = lv_table_get_cell_value(explorer->file_table, row, col);
557 file_entry_user_data = lv_table_get_cell_user_data(explorer->file_table, row, col);
558
559 selected_text = selected_text + 5; /* skip table cell format */
560
561 /* Three navigation modes are supported:
562 * - Navigate to current directory
563 * - Navigate to parent directory
564 * - Navigate to (current directory) child */
565 navigate_to_parent_dir = (lv_strcmp(selected_text, LV_FILE_NAVIGATION_PARENT_DIR) == 0);
566 navigate_to_child = !navigate_to_parent_dir;

577 *(file_name + stripped_file_name_length) = '/';

579 *(file_name + stripped_file_name_length + 1) = '\0';
580 }
581 }
582 else {
583 if(navigate_to_child) {
584 lv_snprintf((char *)file_name, sizeof(file_name), "%s%s", explorer->current_path, selected_text);
585 }
586 else if(navigate_to_parent_dir) { /* We are most likely in the drive letter directory, doesn't have parent directory */
587 return;
588 }
589 else { /* Nothing to do*/ }
590 }
591
592 /* Navigate to the file path if this is a directory */
593 if(file_entry_user_data->file_kind == LV_FILE_EXPLORER_FILE_KIND_DIR) {
594 lv_fs_dir_t dir;
595 lv_fs_res_t res = lv_fs_dir_open(&dir, file_name);
596 if(res == LV_FS_RES_OK) {
597 lv_fs_dir_close(&dir);
598 show_dir(obj, (char *)file_name);

600 else {
601 LV_LOG_USER("Could not navigate to %s (error: %d)", file_name, res);
602 }
603 }
604 else {
605 if(navigate_to_child) {
606 explorer->sel_fn = selected_text;
607 lv_obj_send_event(obj, LV_EVENT_VALUE_CHANGED, NULL);
608 }
609 }
610 }
611 else if(code == LV_EVENT_SIZE_CHANGED) {
612 lv_table_set_column_width(explorer->file_table, 0, lv_obj_get_width(explorer->file_table));
613 }
614 else if(code == LV_EVENT_DELETE) {
615 clear_table_cells_user_data(explorer);
616 }

632static void show_dir(lv_obj_t * obj, const char * path)
633{
634 lv_file_explorer_t * explorer = (lv_file_explorer_t *)obj;
635
636 char fn[LV_FILE_EXPLORER_PATH_MAX_LEN];
637 uint16_t index = 0;
638 lv_fs_dir_t dir;
639 lv_fs_res_t res;
640 lv_file_explorer_file_table_entry_data_t * file_entry_user_data;
641
642 res = lv_fs_dir_open(&dir, path);
643 if(res != LV_FS_RES_OK) {
644 LV_LOG_USER("Open dir error %d!", res);
645 return;
646 }
647
648 clear_table_cells_user_data(explorer);
649 if(explorer->show_back_button) {
650 lv_table_set_cell_value(explorer->file_table, index++, 0, LV_SYMBOL_LEFT " " LV_FILE_NAVIGATION_PARENT_DIR);
651 file_entry_user_data = lv_malloc(sizeof(lv_file_explorer_file_table_entry_data_t));
652 LV_ASSERT_MALLOC(file_entry_user_data);
653 file_entry_user_data->file_kind = LV_FILE_EXPLORER_FILE_KIND_DIR;
654 lv_table_set_cell_user_data(explorer->file_table, 0, 0, file_entry_user_data);
655 }
656
657 while(1) {
658 res = lv_fs_dir_read(&dir, fn, sizeof(fn));
659 if(res != LV_FS_RES_OK) {
660 LV_LOG_USER("Driver, file or directory does not exist %d!", res);
661 break;
662 }
663
664 /*fn is empty, if not more files to read*/
665 if(lv_strlen(fn) == 0) {
666 LV_LOG_USER("No more files to read!");
667 break;
668 }
669
670 file_entry_user_data = lv_malloc(sizeof(lv_file_explorer_file_table_entry_data_t));
671 LV_ASSERT_MALLOC(file_entry_user_data);
672
673 if((is_end_with(fn, ".png") == true) || (is_end_with(fn, ".PNG") == true) || \
674 (is_end_with(fn, ".jpg") == true) || (is_end_with(fn, ".JPG") == true) || \
675 (is_end_with(fn, ".bmp") == true) || (is_end_with(fn, ".BMP") == true) || \
676 (is_end_with(fn, ".gif") == true) || (is_end_with(fn, ".GIF") == true)) {
677 lv_table_set_cell_value_fmt(explorer->file_table, index, 0, LV_SYMBOL_IMAGE " %s", fn);
678 file_entry_user_data->file_kind = LV_FILE_EXPLORER_FILE_KIND_IMAGE;
679 }
680 else if((is_end_with(fn, ".mp3") == true) || (is_end_with(fn, ".MP3") == true) || \
681 (is_end_with(fn, ".wav") == true) || (is_end_with(fn, ".WAV") == true)) {
682 lv_table_set_cell_value_fmt(explorer->file_table, index, 0, LV_SYMBOL_AUDIO " %s", fn);
683 file_entry_user_data->file_kind = LV_FILE_EXPLORER_FILE_KIND_AUDIO;
684 }
685 else if((is_end_with(fn, ".mp4") == true) || (is_end_with(fn, ".MP4") == true)) {
686 lv_table_set_cell_value_fmt(explorer->file_table, index, 0, LV_SYMBOL_VIDEO " %s", fn);
687 file_entry_user_data->file_kind = LV_FILE_EXPLORER_FILE_KIND_VIDEO;
688 }
689 else if((is_end_with(fn, ".") == true) || (is_end_with(fn, "..") == true)) {
690 /*is dir*/
691 continue;
692 }
693 else if(fn[0] == '/') {/*is dir*/
694 lv_table_set_cell_value_fmt(explorer->file_table, index, 0, LV_SYMBOL_DIRECTORY " %s", fn + 1);
695 file_entry_user_data->file_kind = LV_FILE_EXPLORER_FILE_KIND_DIR;
696 }
697 else {
698 lv_table_set_cell_value_fmt(explorer->file_table, index, 0, LV_SYMBOL_FILE " %s", fn);
699 file_entry_user_data->file_kind = LV_FILE_EXPLORER_FILE_KIND_FILE;
700 }
701
702 lv_table_set_cell_user_data(explorer->file_table, index, 0, file_entry_user_data);
703
704 index++;
705 }
706
707 lv_fs_dir_close(&dir);
708
709 lv_table_set_row_count(explorer->file_table, index);
710 file_explorer_sort(obj);
711 lv_obj_send_event(obj, LV_EVENT_READY, NULL);
712
713 /*Move the table to the top*/
714 lv_obj_scroll_to_y(explorer->file_table, 0, LV_ANIM_OFF);
715
716 lv_strncpy(explorer->current_path, path, sizeof(explorer->current_path));
717 lv_label_set_text_fmt(explorer->path_label, LV_SYMBOL_EYE_OPEN" %s", path);
718
719 size_t current_path_len = lv_strlen(explorer->current_path);
720 if(current_path_len > 0 &&
721 explorer->current_path[current_path_len - 1] != '/' &&
722 current_path_len + 1 < sizeof(explorer->current_path)) {
723 explorer->current_path[current_path_len] = '/';
724 explorer->current_path[current_path_len + 1] = '\0';
725 }
726}

745static void exch_table_item(lv_obj_t * tb, int16_t i, int16_t j)
746{
747 if(i == j) return;

763static void file_explorer_sort(lv_obj_t * obj)
764{
765 LV_CHECK_OBJ(obj, MY_CLASS, return);
766
767 lv_file_explorer_t * explorer = (lv_file_explorer_t *)obj;
768
769 uint16_t sum = lv_table_get_row_count(explorer->file_table);
770
771 if(sum > 1) {
772 switch(explorer->sort) {
773 case LV_EXPLORER_SORT_NONE:
774 break;
775 case LV_EXPLORER_SORT_KIND:
776 sort_by_file_kind(explorer->file_table, 0, (sum - 1));
777 break;
778 default:
779 break;
780 }
781 }
782}

784/*Quick sort 3 way*/
785static void sort_by_file_kind(lv_obj_t * tb, int16_t lo, int16_t hi)
786{
787 if(lo >= hi) return;
788
789 int16_t lt = lo;
790 int16_t i = lo + 1;
791 int16_t gt = hi;
792 lv_file_explorer_file_table_entry_data_t * v = lv_table_get_cell_user_data(tb, lo, 0);
793 while(i <= gt) {
794 lv_file_explorer_file_table_entry_data_t * x = lv_table_get_cell_user_data(tb, i, 0);
795 if(x->file_kind < v->file_kind)
796 exch_table_item(tb, lt++, i++);
797 else if(x->file_kind > v->file_kind)
798 exch_table_item(tb, i, gt--);
799 else
800 i++;
801 }
802
803 sort_by_file_kind(tb, lo, lt - 1);
804 sort_by_file_kind(tb, gt + 1, hi);
805}

807static bool is_end_with(const char * str1, const char * str2)
808{
809 if(str1 == NULL || str2 == NULL)
810 return false;
811
812 size_t len1 = lv_strlen(str1);
813 size_t len2 = lv_strlen(str2);
814 if((len1 < len2) || (len1 == 0 || len2 == 0))
815 return false;
816
817 while(len2 >= 1) {
818 if(str2[len2 - 1] != str1[len1 - 1])
819 return false;
820
821 len2--;
822 len1--;
823 }
824
825 return true;
826}
