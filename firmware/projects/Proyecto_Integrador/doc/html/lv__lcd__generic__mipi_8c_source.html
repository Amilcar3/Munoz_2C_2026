49lv_display_t * lv_lcd_generic_mipi_create(uint32_t hor_res, uint32_t ver_res, lv_lcd_flag_t flags,
50 lv_lcd_send_cmd_cb_t send_cmd_cb, lv_lcd_send_color_cb_t send_color_cb)
51{
52 lv_display_t * disp = lv_display_create(hor_res, ver_res);
53 if(disp == NULL) {
54 return NULL;
55 }
56
57 lv_lcd_generic_mipi_driver_t * drv = (lv_lcd_generic_mipi_driver_t *)lv_malloc(sizeof(lv_lcd_generic_mipi_driver_t));
58 if(drv == NULL) {
59 lv_display_delete(disp);
60 return NULL;
61 }
62
63 /* init driver struct */
64 drv->disp = disp;
65 drv->send_cmd = send_cmd_cb;
66 drv->send_color = send_color_cb;
67 lv_display_set_driver_data(disp, (void *)drv);
68
69 /* init controller */
70 init(drv, flags);
71
72 /* register resolution change callback (NOTE: this handles screen rotation as well) */
73 lv_display_add_event_cb(disp, res_chg_event_cb, LV_EVENT_RESOLUTION_CHANGED, NULL);
74
75 /* register object deletion callback for freeing driver struct */
76 lv_display_add_event_cb(disp, delete_cb, LV_EVENT_DELETE, NULL);
77
78 /* register flush callback */
79 lv_display_set_flush_cb(disp, flush_cb);
80
81 return disp;
82}

123void lv_lcd_generic_mipi_send_cmd_list(lv_display_t * disp, const uint8_t * cmd_list)
124{
125 LV_CHECK_ARG(disp != NULL, return);
126 LV_CHECK_ARG(cmd_list != NULL, return);
127 lv_lcd_generic_mipi_driver_t * drv = get_driver(disp);
128 while(1) {
129 uint8_t cmd = *cmd_list++;
130 uint8_t num = *cmd_list++;
131 if(cmd == LV_LCD_CMD_DELAY_MS) {
132 if(num == LV_LCD_CMD_EOF) /* end of list */
133 break;
134 else { /* delay in 10 ms units*/
135 lv_delay_ms((uint32_t)(num) * 10);
136 }
137 }
138 else {
139 drv->send_cmd(drv->disp, &cmd, 1, cmd_list, num);
140 cmd_list += num;
141 }
142 }
143}

180static void init(lv_lcd_generic_mipi_driver_t * drv, lv_lcd_flag_t flags)
181{
182 drv->x_gap = 0;
183 drv->y_gap = 0;
184
185 /* init color mode and RGB order */
186 drv->madctl_reg = flags & LV_LCD_FLAG_BGR ? LV_LCD_BIT_RGB_ORDER__BGR : LV_LCD_BIT_RGB_ORDER__RGB;
187 drv->colmod_reg = flags & LV_LCD_FLAG_RGB666 ? LV_LCD_PIXEL_FORMAT_RGB666 : LV_LCD_PIXEL_FORMAT_RGB565;
188
189 /* init orientation */
190 drv->mirror_x = flags & LV_LCD_FLAG_MIRROR_X;
191 drv->mirror_y = flags & LV_LCD_FLAG_MIRROR_Y;
192 drv->swap_xy = false;
193 /* update madctl_reg */
194 set_swap_xy(drv, drv->swap_xy);
195 set_mirror(drv, drv->mirror_x, drv->mirror_y);
196
197 /* enter sleep mode first */
198 send_cmd(drv, LV_LCD_CMD_ENTER_SLEEP_MODE, NULL, 0);
199 lv_delay_ms(10);
200
201 /* perform software reset */
202 send_cmd(drv, LV_LCD_CMD_SOFT_RESET, NULL, 0);
203 lv_delay_ms(200);
204
205 /* LCD goes into sleep mode and display will be turned off after power on reset, exit sleep mode first */
206 send_cmd(drv, LV_LCD_CMD_EXIT_SLEEP_MODE, NULL, 0);
207 lv_delay_ms(300);
208
209 send_cmd(drv, LV_LCD_CMD_ENTER_NORMAL_MODE, NULL, 0);
210
211 send_cmd(drv, LV_LCD_CMD_SET_ADDRESS_MODE, (uint8_t[]) {
212 drv->madctl_reg,
213 }, 1);
214 send_cmd(drv, LV_LCD_CMD_SET_PIXEL_FORMAT, (uint8_t[]) {
215 drv->colmod_reg,
216 }, 1);
217 send_cmd(drv, LV_LCD_CMD_SET_DISPLAY_ON, NULL, 0);
218}

226static void set_mirror(lv_lcd_generic_mipi_driver_t * drv, bool mirror_x, bool mirror_y)
227{
228 uint8_t mad = drv->madctl_reg & ~(LV_LCD_MASK_COLUMN_ADDRESS_ORDER | LV_LCD_MASK_PAGE_ADDRESS_ORDER);

260static void flush_cb(lv_display_t * disp, const lv_area_t * area, uint8_t * px_map)
261{
262 lv_lcd_generic_mipi_driver_t * drv = get_driver(disp);
263
264 int32_t x_start = area->x1;
265 int32_t x_end = area->x2 + 1;
266 int32_t y_start = area->y1;
267 int32_t y_end = area->y2 + 1;
268
269 LV_ASSERT((x_start < x_end) && (y_start < y_end) && "start position must be smaller than end position");
270
271 x_start += drv->x_gap;
272 x_end += drv->x_gap;
273 y_start += drv->y_gap;
274 y_end += drv->y_gap;
275
276 /* define an area of frame memory where MCU can access */
277 send_cmd(drv, LV_LCD_CMD_SET_COLUMN_ADDRESS, (uint8_t[]) {
278 (x_start >> 8) & 0xFF,
279 x_start & 0xFF,
280 ((x_end - 1) >> 8) & 0xFF,
281 (x_end - 1) & 0xFF,
282 }, 4);
283 send_cmd(drv, LV_LCD_CMD_SET_PAGE_ADDRESS, (uint8_t[]) {
284 (y_start >> 8) & 0xFF,
285 y_start & 0xFF,
286 ((y_end - 1) >> 8) & 0xFF,
287 (y_end - 1) & 0xFF,
288 }, 4);
289 /* transfer frame buffer */
290 size_t len = (x_end - x_start) * (y_end - y_start) * lv_color_format_get_size(lv_display_get_color_format(disp));
291 send_color(drv, LV_LCD_CMD_WRITE_MEMORY_START, px_map, len);
292}

299static void set_rotation(lv_lcd_generic_mipi_driver_t * drv, lv_display_rotation_t rot)
300{
301 switch(rot) {
302 case LV_DISPLAY_ROTATION_0:
303 set_swap_xy(drv, drv->swap_xy);
304 set_mirror(drv, drv->mirror_x, drv->mirror_y);
305 break;
306 case LV_DISPLAY_ROTATION_90:
307 set_swap_xy(drv, !drv->swap_xy);
308 set_mirror(drv, drv->mirror_x, !drv->mirror_y);
309 break;
310 case LV_DISPLAY_ROTATION_180:
311 set_swap_xy(drv, drv->swap_xy);
312 set_mirror(drv, !drv->mirror_x, !drv->mirror_y);
313 break;
314 case LV_DISPLAY_ROTATION_270:
315 set_swap_xy(drv, !drv->swap_xy);
316 set_mirror(drv, !drv->mirror_x, drv->mirror_y);
317 break;
318 }
319 send_cmd(drv, LV_LCD_CMD_SET_ADDRESS_MODE, (uint8_t[]) {
320 drv->madctl_reg
321 }, 1);
322}
