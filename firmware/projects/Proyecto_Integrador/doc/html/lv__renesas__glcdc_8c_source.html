144static lv_display_t * glcdc_create(void * buf1, void * buf2, uint32_t buf_size, lv_display_render_mode_t render_mode)
145{
146#ifdef _RENESAS_RA_
147 glcdc_init();
148#else
149 g_display0_cfg.input->format = LCD_CH0_IN_GR2_FORMAT;
150 _SemaphoreVsync = xSemaphoreCreateBinary();
151
152 glcdc_init();
153 enable_dave2d_drw_interrupt();
154#endif /*_RENESAS_RA_*/
155
156 lv_display_t * display = lv_display_create(DISPLAY_HSIZE_INPUT0, DISPLAY_VSIZE_INPUT0);
157
158 if(render_mode == LV_DISPLAY_RENDER_MODE_DIRECT) {
159 lv_display_set_flush_cb(display, flush_direct);
160 lv_display_set_flush_wait_cb(display, flush_wait_direct);
161 }
162 else if(render_mode == LV_DISPLAY_RENDER_MODE_PARTIAL) {
163 lv_display_set_flush_cb(display, flush_partial);
164 lv_display_set_flush_wait_cb(display, flush_wait_partial);
165 }
166 else {
167 LV_ASSERT(0);
168 }
169
170 lv_display_set_buffers_with_stride(display, buf1, buf2, buf_size, DISPLAY_BUFFER_STRIDE_BYTES_INPUT0, render_mode);
171
172 return display;
173}

189static void glcdc_init(void)
190{
191 /* Fill the Frame buffer with black colour (0x0000 in RGB565), for a clean start after previous runs */
192 lv_memzero(fb_background, sizeof(fb_background));
193
194#ifdef _RENESAS_RA_
195 /* Initialize GLCDC driver */
196 uint8_t * p_fb = &fb_background[1][0];
197 fsp_err_t err;
198
199 err = R_GLCDC_Open(&g_display0_ctrl, &g_display0_cfg);
200 if(FSP_SUCCESS != err) {
201 __BKPT(0);
202 }
203
204 err = R_GLCDC_Start(&g_display0_ctrl);
205 if(FSP_SUCCESS != err) {
206 __BKPT(0);
207 }
208
209 do {
210 err =
211 R_GLCDC_BufferChange(&g_display0_ctrl,
212 (uint8_t *) p_fb,
213 (display_frame_layer_t) 0);
214 } while(FSP_ERR_INVALID_UPDATE_TIMING == err);
215#else /* RX */
216 glcdc_err_t err;
217 glcdc_runtime_cfg_t layer_change;
218
219 R_GLCDC_PinSet();
220
221 err = R_GLCDC_Open(&g_config);
222 if(GLCDC_SUCCESS != err) {
223 while(1);
224 }
225
226 err = R_GLCDC_Control(GLCDC_CMD_START_DISPLAY, &g_config);
227 if(GLCDC_SUCCESS != err) {
228 while(1);
229 }
230
231 g_layer_change.input = g_config.input[GLCDC_FRAME_LAYER_2];
232 g_layer_change.chromakey = g_config.chromakey[GLCDC_FRAME_LAYER_2];
233 g_layer_change.blend = g_config.blend[GLCDC_FRAME_LAYER_2];
234
235 layer_change.input.p_base = (uint32_t *)&fb_background[1][0];
236
237 do {
238 err = R_GLCDC_LayerChange(GLCDC_FRAME_LAYER_2, &g_layer_change);
239 } while(GLCDC_ERR_INVALID_UPDATE_TIMING == err);
240#endif /*_RENESAS_RA_*/
241}

243static void flush_direct(lv_display_t * display, const lv_area_t * area, uint8_t * px_map)
244{
245 FSP_PARAMETER_NOT_USED(area);
246 /*Display the frame buffer pointed by px_map*/
247
248 if(!lv_display_flush_is_last(display)) return;
249
250#if defined(RENESAS_CORTEX_M85) && (BSP_CFG_DCACHE_ENABLED)
251 /* Invalidate cache - so the HW can access any data written by the CPU */
252 SCB_CleanInvalidateDCache_by_Addr(px_map, sizeof(fb_background[0]));
253#endif
254
255#ifdef _RENESAS_RA_
256 R_GLCDC_BufferChange(&g_display0_ctrl,
257 (uint8_t *) px_map,
258 (display_frame_layer_t) 0);
259#else /* RX */
260 glcdc_err_t err;
261
262 g_layer_change.input.p_base = (uint32_t *)px_map;
263
264 do {
265 err = R_GLCDC_LayerChange(GLCDC_FRAME_LAYER_2, &g_layer_change);
266 } while(GLCDC_ERR_INVALID_UPDATE_TIMING == err);
267#endif /*_RENESAS_RA_*/
268}

270static void flush_wait_direct(lv_display_t * display)
271{
272 if(!lv_display_flush_is_last(display)) return;
273
274#if USE_FREE_RTOS
275 /*If Vsync semaphore has already been set, clear it then wait to avoid tearing*/
276 if(uxSemaphoreGetCount(_SemaphoreVsync)) {
277 xSemaphoreTake(_SemaphoreVsync, 10);
278 }
279
280 xSemaphoreTake(_SemaphoreVsync, portMAX_DELAY);
281#endif /*USE_FREE_RTOS*/
282
283}

285static void flush_partial(lv_display_t * display, const lv_area_t * area, uint8_t * px_map)
286{
287 uint16_t * img = (uint16_t *)px_map;
288
289 lv_area_t rotated_area;
290 lv_color_format_t cf = lv_display_get_color_format(display);
291 lv_display_rotation_t rotation = lv_display_get_rotation(display);
292
293 if(rotation != LV_DISPLAY_ROTATION_0) {
294 int32_t w = lv_area_get_width(area);
295 int32_t h = lv_area_get_height(area);
296 uint32_t w_stride = lv_draw_buf_width_to_stride(w, cf);
297 uint32_t h_stride = lv_draw_buf_width_to_stride(h, cf);
298
299 // only allocate if rotation is actually being used
300 if(!rotation_buffer) {
301 rotation_buffer = lv_malloc(partial_buffer_size);
302 LV_ASSERT_MALLOC(rotation_buffer);
303 }
304
305 if(rotation == LV_DISPLAY_ROTATION_180)
306 lv_draw_rotate(img, rotation_buffer, w, h, w_stride, w_stride, rotation, cf);
307 else /* 90 or 270 */
308 lv_draw_rotate(img, rotation_buffer, w, h, w_stride, h_stride, rotation, cf);
309
310 img = rotation_buffer;
311
312 rotated_area = *area;
313 lv_display_rotate_area(display, &rotated_area);
314 area = &rotated_area;
315 }
316
317 int32_t w = lv_area_get_width(area);
318 int32_t h = lv_area_get_height(area);
319
320 uint16_t * fb = (uint16_t *)fb_background[1];
321
322 fb = fb + area->y1 * DISPLAY_HSIZE_INPUT0;
323 fb = fb + area->x1;
324
325 int32_t i;
326 for(i = 0; i < h; i++) {
327 lv_memcpy(fb, img, w * BYTES_PER_PIXEL);
328
329#if defined(RENESAS_CORTEX_M85) && (BSP_CFG_DCACHE_ENABLED)
330 SCB_CleanInvalidateDCache_by_Addr(fb, w * BYTES_PER_PIXEL);
331#endif
332 fb += DISPLAY_HSIZE_INPUT0;
333 img += w;
334 }
335}
