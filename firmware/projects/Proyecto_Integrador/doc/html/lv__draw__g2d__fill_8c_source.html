59void lv_draw_g2d_fill(lv_draw_task_t * t)
60{
61 lv_draw_fill_dsc_t * dsc = t->draw_dsc;
62
63 if(dsc->opa <= (lv_opa_t)LV_OPA_MIN)
64 return;
65
66 lv_layer_t * layer = t->target_layer;
67 lv_draw_buf_t * draw_buf = layer->draw_buf;
68 lv_area_t * coords = &t->area;
69
70 lv_area_t rel_coords;
71 rel_coords = *coords;
72 lv_area_move(&rel_coords, -layer->buf_area.x1, -layer->buf_area.y1);
73
74 lv_area_t rel_clip_area;
75 rel_clip_area = t->clip_area;
76 lv_area_move(&rel_clip_area, -layer->buf_area.x1, -layer->buf_area.y1);
77
78 lv_area_t blend_area;
79 if(!lv_area_intersect(&blend_area, &rel_coords, &rel_clip_area))
80 return; /*Fully clipped, nothing to do*/
81
82 /* G2D takes stride in pixels. */
83 int32_t stride = draw_buf->header.stride / (lv_color_format_get_bpp(draw_buf->header.cf) / 8);
84
85 /* Destination buffer */
86 struct g2d_buf * dst_buf = g2d_search_buf_map(draw_buf->data);
87
88 bool has_opa = (dsc->opa < (lv_opa_t)LV_OPA_MAX);
89 struct g2d_surface dst_surf;
90 _g2d_set_dst_surf(&dst_surf, dst_buf, &blend_area, stride, dsc->color, draw_buf->header.cf);
91
92 void * handle = g2d_get_handle();
93
94 if(has_opa) {
95 struct g2d_buf * tmp_buf = g2d_alloc(lv_area_get_width(&blend_area) * lv_area_get_height(
96 &blend_area) * lv_color_format_get_size(draw_buf->header.cf), 1);
97 G2D_ASSERT_MSG(tmp_buf, "Failed to alloc temporary buffer.");
98 struct g2d_surface src_surf;
99 _g2d_set_src_surf(&src_surf, tmp_buf, &blend_area, dsc->color, dsc->opa, draw_buf->header.cf);
100 _g2d_fill_with_opa(handle, &dst_surf, &src_surf);
101 g2d_free(tmp_buf);
102 }
103 else {
104 _g2d_fill(handle, &dst_surf);
105 }
106}

112static void _g2d_set_src_surf(struct g2d_surface * src_surf, struct g2d_buf * buf, const lv_area_t * area,
113 lv_color_t color, lv_opa_t opa, lv_color_format_t cf)
114{
115 int32_t width = lv_area_get_width(area);
116 int32_t height = lv_area_get_height(area);
117
118 src_surf->format = g2d_get_buf_format(cf);
119
120 src_surf->left = 0;
121 src_surf->top = 0;
122 src_surf->right = width;
123 src_surf->bottom = height;
124 src_surf->stride = width;
125 src_surf->width = width;
126 src_surf->height = height;
127
128 src_surf->planes[0] = buf->buf_paddr;
129 src_surf->rot = G2D_ROTATION_0;
130
131 src_surf->clrcolor = g2d_rgba_to_u32(color);
132 src_surf->global_alpha = opa;
133 src_surf->blendfunc = G2D_ONE | G2D_PRE_MULTIPLIED_ALPHA;
134}

160static void _g2d_fill_with_opa(void * handle, struct g2d_surface * dst_surf, struct g2d_surface * src_surf)
161{
162 g2d_clear(handle, src_surf);
163
164 g2d_enable(handle, G2D_BLEND);
165 g2d_enable(handle, G2D_GLOBAL_ALPHA);
166 g2d_blit(handle, src_surf, dst_surf);
167 g2d_finish(handle);
168 g2d_disable(handle, G2D_GLOBAL_ALPHA);
169 g2d_disable(handle, G2D_BLEND);
170}
