92 @property
93 def flags_raw(self) -> int:
94 """Return raw flags bitmask, 0 if corrupted.
95
96 Up to LVGL 9.5 this was a single `flags` bitmask. Since 9.6 each flag
97 is its own bitfield named after it, so the mask is rebuilt from them.
98 Without this every object reports no flags at all, because safe_field()
99 cannot tell a missing field from a clear one.
100 """
101 from .lv_obj_flag_consts import OBJ_FLAG_NAMES
102
103 raw = self.safe_field("flags")
104 if raw is not None:
105 return int(raw)
106
107 value = 0
108 for bit, name in OBJ_FLAG_NAMES.items():
109 # Composite names (SCROLL_CHAIN) have no field of their own.
110 field = self.safe_field(name.lower())
111 if field is not None and int(field):
112 value |= bit
113 return value | self._user_flags_mask()
114
115 def _user_flags_mask(self) -> int:
116 """The eight flags 9.6 keeps in spec_attr->user_flags, as a flag mask.
117
118 These have no bitfield of their own, so the rebuild above misses all of
119 them - and LAYOUT_1 is LV_OBJ_FLAG_FLEX_IN_NEW_TRACK, so losing it costs
120 a flex layout its wrapping.
121 """
122 from .lv_obj_flag_consts import OBJ_FLAG_NAMES
123
124 raw = self._get_spec_int("user_flags")
125 if not raw:
126 return 0
127
128 # The order lv_obj_add_flag() writes them in, from bit 0 up.
129 names = ("LAYOUT_1", "LAYOUT_2", "WIDGET_1", "WIDGET_2",
130 "USER_1", "USER_2", "USER_3", "USER_4")
131 bit_of = {name: bit for bit, name in OBJ_FLAG_NAMES.items()}
132 value = 0
133 for index, name in enumerate(names):
134 if raw & (1 << index) and name in bit_of:
135 value |= bit_of[name]
136 return value
