54lv_result_t lv_image_header_cache_init(uint32_t count)

119inline static lv_cache_compare_res_t image_cache_common_compare(const void * lhs_src, lv_image_src_t lhs_src_type,
120 const void * rhs_src, lv_image_src_t rhs_src_type)
121{
122 if(lhs_src_type == rhs_src_type) {
123 if(lhs_src_type == LV_IMAGE_SRC_FILE) {
124 int32_t cmp_res = lv_strcmp(lhs_src, rhs_src);
125 if(cmp_res != 0) {
126 return cmp_res > 0 ? 1 : -1;
127 }
128 }
129 else if(lhs_src_type == LV_IMAGE_SRC_VARIABLE) {
130 if(lhs_src != rhs_src) {
131 return lhs_src > rhs_src ? 1 : -1;
132 }
133 }
134 return 0;
135 }
136 return lhs_src_type > rhs_src_type ? 1 : -1;
137}

153static void iter_inspect_cb(void * elem)
154{
155 lv_image_cache_data_t * data = (lv_image_cache_data_t *)elem;
156 lv_draw_buf_t * decoded = (lv_draw_buf_t *)data->decoded;
157 lv_image_header_t * header = &decoded->header;
158 lv_cache_entry_t * entry = lv_cache_entry_get_entry(data, img_header_cache_p->node_size);
159
160 LV_UNUSED(decoded);
161 LV_UNUSED(header);
162 LV_UNUSED(entry);
163
164 /* size data_size cf rc type decoded src*/
165#define IMAGE_CACHE_DUMP_FORMAT " %4dx%-4d %9"LV_PRIu32" %d %" LV_PRId32 " "
166 switch(data->src_type) {
167 case LV_IMAGE_SRC_FILE:
168 LV_LOG_USER(IMAGE_CACHE_DUMP_FORMAT "file\t%-12p\t%s", header->w, header->h, decoded->data_size, header->cf,
169 lv_cache_entry_get_ref(entry), (void *)data->decoded, (char *)data->src);
170 break;
171 case LV_IMAGE_SRC_VARIABLE:
172 LV_LOG_USER(IMAGE_CACHE_DUMP_FORMAT "var \t%-12p\t%p", header->w, header->h, decoded->data_size, header->cf,
173 lv_cache_entry_get_ref(entry), (void *)data->decoded, data->src);
174 break;
175 default:
176 LV_LOG_USER(IMAGE_CACHE_DUMP_FORMAT "unkn\t%-12p\t%p", header->w, header->h, decoded->data_size, header->cf,
177 lv_cache_entry_get_ref(entry), (void *)data->decoded, data->src);
178 break;
179 }
