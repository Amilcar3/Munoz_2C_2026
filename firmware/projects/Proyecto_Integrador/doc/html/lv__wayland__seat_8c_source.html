62void lv_wayland_seat_init(lv_wl_seat_t * seat, struct wl_registry * registry, uint32_t name, uint32_t version)
63{
64 LV_CHECK_ARG(seat != NULL, return);
65 LV_UNUSED(version);
66 seat->wl_seat = wl_registry_bind(registry, name, &wl_seat_interface, (version < SEAT_VERSION) ? version : SEAT_VERSION);
67 wl_seat_add_listener(seat->wl_seat, &seat_listener, seat);

114static lv_wl_seat_pointer_t * create_pointer(struct wl_seat * wl_seat)
115{
116 struct wl_surface * surface = wl_compositor_create_surface(lv_wl_ctx.wl_compositor);
117 if(!surface) {
118 LV_LOG_WARN("Failed to get surface for pointer");
119 return NULL;
120 }
121 lv_wl_seat_pointer_t * seat_pointer = lv_wayland_seat_pointer_create(wl_seat, surface);
122 if(!seat_pointer) {
123 LV_LOG_WARN("Failed to create seat pointer");
124 wl_surface_destroy(surface);
125 return NULL;
126 }
127 return seat_pointer;
128}

136static void seat_handle_capabilities(void * data, struct wl_seat * wl_seat, enum wl_seat_capability caps)
137{
138 lv_wl_seat_t * seat = (lv_wl_seat_t *) data;
139
140 if((caps & WL_SEAT_CAPABILITY_POINTER) && !seat->pointer) {
141 seat->pointer = create_pointer(wl_seat);
142 }
143 else if(!(caps & WL_SEAT_CAPABILITY_POINTER) && seat->pointer) {
144 delete_pointer(seat->pointer);
145 seat->pointer = NULL;
146 }
147
148 if((caps & WL_SEAT_CAPABILITY_KEYBOARD) && !seat->keyboard) {
149 seat->keyboard = lv_wayland_seat_keyboard_create(wl_seat);
150 }
151 else if(!(caps & WL_SEAT_CAPABILITY_KEYBOARD) && seat->keyboard) {
152 lv_wayland_seat_keyboard_delete(seat->keyboard);
153 seat->keyboard = NULL;
154 }
155
156 if((caps & WL_SEAT_CAPABILITY_TOUCH) && !seat->touch) {
157 seat->touch = lv_wayland_seat_touch_create(wl_seat);
158 }
159 else if(!(caps & WL_SEAT_CAPABILITY_TOUCH) && seat->touch) {
160 lv_wayland_seat_touch_delete(seat->touch);
161 seat->touch = NULL;
162 }
163}
