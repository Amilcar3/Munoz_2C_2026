77uint32_t lv_xkb_process_key(lv_xkb_t * dsc, uint32_t scancode, bool down)
78{
79 LV_CHECK_ARG(dsc != NULL, return 0);
80
81 /* Offset the evdev scancode by 8, see https://xkbcommon.org/doc/current/xkbcommon_8h.html#ac29aee92124c08d1953910ab28ee1997 */
82 xkb_keycode_t keycode = scancode + 8;
83
84 uint32_t result = 0;
85
86 switch(xkb_state_key_get_one_sym(dsc->state, keycode)) {
87 case XKB_KEY_BackSpace:
88 result = LV_KEY_BACKSPACE;
89 break;
90 case XKB_KEY_Return:
91 case XKB_KEY_KP_Enter:
92 result = LV_KEY_ENTER;
93 break;
94 case XKB_KEY_Prior:
95 case XKB_KEY_KP_Prior:
96 result = LV_KEY_PREV;
97 break;
98 case XKB_KEY_Next:
99 case XKB_KEY_KP_Next:
100 result = LV_KEY_NEXT;
101 break;
102 case XKB_KEY_Up:
103 case XKB_KEY_KP_Up:
104 result = LV_KEY_UP;
105 break;
106 case XKB_KEY_Left:
107 case XKB_KEY_KP_Left:
108 result = LV_KEY_LEFT;
109 break;
110 case XKB_KEY_Right:
111 case XKB_KEY_KP_Right:
112 result = LV_KEY_RIGHT;
113 break;
114 case XKB_KEY_Down:
115 case XKB_KEY_KP_Down:
116 result = LV_KEY_DOWN;
117 break;
118 case XKB_KEY_Tab:
119 case XKB_KEY_KP_Tab:
120 result = LV_KEY_NEXT;
121 break;
122 case XKB_KEY_ISO_Left_Tab: /* Sent on SHIFT + TAB */
123 result = LV_KEY_PREV;
124 break;
125 case XKB_KEY_Home:
126 case XKB_KEY_KP_Home:
127 result = LV_KEY_HOME;
128 break;
129 case XKB_KEY_End:
130 case XKB_KEY_KP_End:
131 result = LV_KEY_END;
132 break;
133 default:
134 break;
135 }
136
137 if(result == 0) {
138 char buffer[4] = { 0, 0, 0, 0 };
139 int size = xkb_state_key_get_utf8(dsc->state, keycode, NULL, 0) + 1;
140 if(size > 1) {
141 xkb_state_key_get_utf8(dsc->state, keycode, buffer, size);
142 memcpy(&result, buffer, 4);
143 }
144 }
145
146 xkb_state_update_key(dsc->state, keycode, down ? XKB_KEY_DOWN : XKB_KEY_UP);
147
148 return result;
149}

155static bool _set_keymap(lv_xkb_t * dsc, struct xkb_rule_names names)
156{
157 LV_ASSERT(dsc != NULL);
158 if(dsc->keymap) {
159 xkb_keymap_unref(dsc->keymap);
160 dsc->keymap = NULL;
161 }
162
163 dsc->keymap = xkb_keymap_new_from_names(context, &names, XKB_KEYMAP_COMPILE_NO_FLAGS);
164 if(!dsc->keymap) {
165 LV_LOG_ERROR("xkb_keymap_new_from_names failed: %s", strerror(errno));
166 return false;
167 }
168
169 if(dsc->state) {
170 xkb_state_unref(dsc->state);
171 dsc->state = NULL;
172 }
173
174 dsc->state = xkb_state_new(dsc->keymap);
175 if(!dsc->state) {
176 LV_LOG_ERROR("xkb_state_new failed: %s", strerror(errno));
177 return false;
178 }
179
180 return true;
181}
