433static int glad_gl_has_extension(const char *exts, char **exts_i, const char *ext) {
434 if(exts_i) {
435 unsigned int index;
436 for(index = 0; exts_i[index]; index++) {
437 const char *e = exts_i[index];
438 if(strcmp(e, ext) == 0) {
439 return 1;
440 }
441 }

443 const char *extensions;

445 const char *terminator;
446 extensions = exts;

448 return 0;
449 }
450 while(1) {
