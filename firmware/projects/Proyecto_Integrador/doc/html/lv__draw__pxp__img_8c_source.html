68void lv_draw_pxp_img(lv_draw_task_t * t)
69{
70 const lv_draw_image_dsc_t * dsc = t->draw_dsc;
71 const lv_area_t * coords = &t->area;
72
73 if(dsc->opa <= (lv_opa_t)LV_OPA_MIN)
74 return;
75
76 bool is_tiled = (dsc->tile != 0);
77 if(is_tiled)
78 lv_draw_image_tiled_helper(t, dsc, coords, _pxp_draw_core_cb, NULL);
79 else
80 lv_draw_image_normal_helper(t, dsc, coords, _pxp_draw_core_cb, NULL);
81}

83/**********************

86static void _pxp_draw_core_cb(lv_draw_task_t * t, const lv_draw_image_dsc_t * draw_dsc,
87 const lv_image_decoder_dsc_t * decoder_dsc, lv_draw_image_sup_t * sup,
88 const lv_area_t * img_coords, const lv_area_t * clipped_img_area)
89{
90 lv_layer_t * layer = t->target_layer;
91 lv_draw_buf_t * draw_buf = layer->draw_buf;
92 const lv_draw_buf_t * decoded = decoder_dsc->decoded;
93
94 lv_area_t rel_clip_area;
95 rel_clip_area = *clipped_img_area;
96 lv_area_move(&rel_clip_area, -img_coords->x1, -img_coords->y1);
97
98 lv_area_t rel_img_coords;
99 rel_img_coords = *img_coords;
100 lv_area_move(&rel_img_coords, -img_coords->x1, -img_coords->y1);
101
102 const uint8_t * src_buf = decoded->data;
103
104 lv_area_t src_area;
105 if(!lv_area_intersect(&src_area, &rel_clip_area, &rel_img_coords))
106 return;
107
108 int32_t src_stride = decoded->header.stride;
109 lv_color_format_t src_cf = decoded->header.cf;
110
111 lv_area_t rel_dest_area = *clipped_img_area;
112 lv_area_move(&rel_dest_area, -layer->buf_area.x1, -layer->buf_area.y1);
113
114 uint8_t * dest_buf = draw_buf->data;
115 int32_t dest_stride = draw_buf->header.stride;
116 lv_color_format_t dest_cf = draw_buf->header.cf;
117 bool has_recolor = (draw_dsc->recolor_opa > LV_OPA_MIN);
118 bool has_transform = (draw_dsc->rotation != 0 || draw_dsc->scale_x != LV_SCALE_NONE ||
119 draw_dsc->scale_y != LV_SCALE_NONE);
120
121 if(has_recolor && !has_transform)
122 _pxp_blit_recolor(dest_buf, &rel_dest_area, dest_stride, dest_cf,
123 src_buf, &src_area, src_stride, src_cf, draw_dsc);
124 else if(has_transform)
125 _pxp_blit_transform(dest_buf, &rel_dest_area, dest_stride, dest_cf,
126 src_buf, &src_area, src_stride, src_cf, draw_dsc);
127 else
128 _pxp_blit(dest_buf, &rel_dest_area, dest_stride, dest_cf,
129 src_buf, &src_area, src_stride, src_cf, draw_dsc->opa);
130}

132static void _pxp_blit_recolor(uint8_t * dest_buf, const lv_area_t * dest_area, int32_t dest_stride,
133 lv_color_format_t dest_cf, const uint8_t * src_buf, const lv_area_t * src_area,
134 int32_t src_stride, lv_color_format_t src_cf, const lv_draw_image_dsc_t * dsc)
135{
136
137 int32_t dest_w = lv_area_get_width(dest_area);
138 int32_t dest_h = lv_area_get_height(dest_area);
139 int32_t src_w = lv_area_get_width(src_area);
140 int32_t src_h = lv_area_get_height(src_area);
141
142 bool src_has_alpha = (src_cf == LV_COLOR_FORMAT_ARGB8888);
143 uint8_t src_px_size = lv_color_format_get_size(src_cf);
144 uint8_t dest_px_size = lv_color_format_get_size(dest_cf);
145
146 lv_pxp_reset();
147
148 /*AS buffer - source image*/
149 pxp_as_buffer_config_t asBufferConfig = {
150 .pixelFormat = pxp_get_as_px_format(src_cf),
151 .bufferAddr = (uint32_t)(src_buf + src_stride * src_area->y1 + src_px_size * src_area->x1),
152 .pitchBytes = src_stride
153 };
154 PXP_SetAlphaSurfaceBufferConfig(PXP_ID, &asBufferConfig);
155 PXP_SetAlphaSurfacePosition(PXP_ID, 0U, 0U, src_w - 1U, src_h - 1U);
156
157 /*Disable PS, use as color generator*/
158 PXP_SetProcessSurfacePosition(PXP_ID, 0xFFFFU, 0xFFFFU, 0U, 0U);
159 PXP_SetProcessSurfaceBackGroundColor(PXP_ID, lv_color_to_u32(dsc->recolor));
160
161 /*Output buffer*/
162 pxp_output_buffer_config_t outputBufferConfig = {
163 .pixelFormat = pxp_get_out_px_format(dest_cf),
164 .interlacedMode = kPXP_OutputProgressive,
165 .buffer0Addr = (uint32_t)(dest_buf + dest_stride * dest_area->y1 + dest_px_size * dest_area->x1),
166 .buffer1Addr = (uint32_t)0U,
167 .pitchBytes = dest_stride,
168 .width = dest_w,
169 .height = dest_h
170 };
171 PXP_SetOutputBufferConfig(PXP_ID, &outputBufferConfig);
172
180 pxp_porter_duff_config_t pdConfig = {
181 .enable = 1,
182 .dstColorMode = kPXP_PorterDuffColorWithAlpha,
183 .srcColorMode = kPXP_PorterDuffColorWithAlpha,
184 .dstGlobalAlphaMode = kPXP_PorterDuffGlobalAlpha,
185 .srcGlobalAlphaMode = src_has_alpha ? kPXP_PorterDuffLocalAlpha : kPXP_PorterDuffGlobalAlpha,
186 .dstFactorMode = kPXP_PorterDuffFactorStraight,
187 .srcFactorMode = kPXP_PorterDuffFactorInversed,
188 .dstGlobalAlpha = dsc->recolor_opa,
189 .srcGlobalAlpha = 0xff,
190 .dstAlphaMode = kPXP_PorterDuffAlphaStraight, /*don't care*/
191 .srcAlphaMode = kPXP_PorterDuffAlphaStraight
192 };
193 PXP_SetPorterDuffConfig(PXP_ID, &pdConfig);
194
195 lv_pxp_run();
196}

198static void _pxp_blit_transform(uint8_t * dest_buf, const lv_area_t * dest_area, int32_t dest_stride,
199 lv_color_format_t dest_cf, const uint8_t * src_buf, const lv_area_t * src_area,
200 int32_t src_stride, lv_color_format_t src_cf, const lv_draw_image_dsc_t * dsc)
201{
202 int32_t src_w = lv_area_get_width(src_area);
203 int32_t src_h = lv_area_get_height(src_area);
204 int32_t dest_w = lv_area_get_width(dest_area);
205 int32_t dest_h = lv_area_get_height(dest_area);
206
207 bool has_rotation = (dsc->rotation != 0);
208 bool has_scale = (dsc->scale_x != LV_SCALE_NONE || dsc->scale_y != LV_SCALE_NONE);
209 uint8_t src_px_size = lv_color_format_get_size(src_cf);
210 uint8_t dest_px_size = lv_color_format_get_size(dest_cf);
211
212 lv_pxp_reset();
213
214 if(has_rotation) {
215 /*Convert rotation angle and calculate offsets caused by pivot*/
216 pxp_rotate_degree_t pxp_angle;
217 switch(dsc->rotation) {
218 case 0:
219 pxp_angle = kPXP_Rotate0;
220 break;
221 case 900:
222 pxp_angle = kPXP_Rotate90;
223 break;
224 case 1800:
225 pxp_angle = kPXP_Rotate180;
226 break;
227 case 2700:
228 pxp_angle = kPXP_Rotate270;
229 break;
230 default:
231 pxp_angle = kPXP_Rotate0;
232 }
233 /*PS buffer rotation and decimation does not function at the same time*/
234 PXP_SetRotateConfig(PXP_ID, kPXP_RotateOutputBuffer, pxp_angle, kPXP_FlipDisable);
235 }
236
237 /*PS buffer - source image*/
238 pxp_ps_buffer_config_t psBufferConfig = {
239 .pixelFormat = pxp_get_ps_px_format(src_cf),
240 .swapByte = false,
241 .bufferAddr = (uint32_t)(src_buf + src_stride * src_area->y1 + src_px_size * src_area->x1),
242 .bufferAddrU = 0,
243 .bufferAddrV = 0,
244 .pitchBytes = src_stride
245 };
246 PXP_SetProcessSurfaceBufferConfig(PXP_ID, &psBufferConfig);
247 PXP_SetProcessSurfacePosition(PXP_ID, 0U, 0U, dest_w - 1U, dest_h - 1U);
248
249 if(has_scale)
250 PXP_SetProcessSurfaceScaler(PXP_ID, src_w, src_h, dest_w, dest_h);
251
252 /*AS disabled */
253 PXP_SetAlphaSurfacePosition(PXP_ID, 0xFFFFU, 0xFFFFU, 0U, 0U);
254
255 /*Output buffer*/
256 pxp_output_buffer_config_t outputBufferConfig = {
257 .pixelFormat = pxp_get_out_px_format(dest_cf),
258 .interlacedMode = kPXP_OutputProgressive,
259 .buffer0Addr = (uint32_t)(dest_buf + dest_stride * (dest_area->y1) + dest_px_size * (dest_area->x1)),
260 .buffer1Addr = (uint32_t)0U,
261 .pitchBytes = dest_stride,
262 .width = dest_w,
263 .height = dest_h
264 };
265 PXP_SetOutputBufferConfig(PXP_ID, &outputBufferConfig);
266
267 lv_pxp_run();
268}

270static void _pxp_blit(uint8_t * dest_buf, const lv_area_t * dest_area, int32_t dest_stride,
271 lv_color_format_t dest_cf, const uint8_t * src_buf, const lv_area_t * src_area,
272 int32_t src_stride, lv_color_format_t src_cf, lv_opa_t opa)
273{
274 int32_t dest_w = lv_area_get_width(dest_area);
275 int32_t dest_h = lv_area_get_height(dest_area);
276 int32_t src_w = lv_area_get_width(src_area);
277 int32_t src_h = lv_area_get_height(src_area);
278
279 bool src_has_alpha = (src_cf == LV_COLOR_FORMAT_ARGB8888);
280 uint8_t src_px_size = lv_color_format_get_size(src_cf);
281 uint8_t dest_px_size = lv_color_format_get_size(dest_cf);
282
283 lv_pxp_reset();
284
285 pxp_as_blend_config_t asBlendConfig = {
286 .alpha = opa,
287 .invertAlpha = false,
288 .alphaMode = kPXP_AlphaRop,
289 .ropMode = kPXP_RopMergeAs
290 };
291
292 if(opa >= (lv_opa_t)LV_OPA_MAX && !src_has_alpha) {
293 /*Simple blit, no effect - Disable PS buffer*/
294 PXP_SetProcessSurfacePosition(PXP_ID, 0xFFFFU, 0xFFFFU, 0U, 0U);
295 }
296 else {
297 /*PS must be enabled to fetch background pixels.
298 PS and OUT buffers are the same, blend will be done in-place*/
299 pxp_ps_buffer_config_t psBufferConfig = {
300 .pixelFormat = pxp_get_ps_px_format(dest_cf),
301 .swapByte = false,
302 .bufferAddr = (uint32_t)(dest_buf + dest_stride * dest_area->y1 + dest_px_size * dest_area->x1),
303 .bufferAddrU = 0U,
304 .bufferAddrV = 0U,
305 .pitchBytes = dest_stride
306 };
307
308 if(opa >= (lv_opa_t)LV_OPA_MAX)
309 asBlendConfig.alphaMode = src_has_alpha ? kPXP_AlphaEmbedded : kPXP_AlphaOverride;
310 else
311 asBlendConfig.alphaMode = src_has_alpha ? kPXP_AlphaMultiply : kPXP_AlphaOverride;
312
313 PXP_SetProcessSurfaceBufferConfig(PXP_ID, &psBufferConfig);
314 PXP_SetProcessSurfacePosition(PXP_ID, 0U, 0U, dest_w - 1U, dest_h - 1U);
315 }
316
317 /*AS buffer - source image*/
318 pxp_as_buffer_config_t asBufferConfig = {
319 .pixelFormat = pxp_get_as_px_format(src_cf),
320 .bufferAddr = (uint32_t)(src_buf + src_stride * src_area->y1 + src_px_size * src_area->x1),
321 .pitchBytes = src_stride
322 };
323 PXP_SetAlphaSurfaceBufferConfig(PXP_ID, &asBufferConfig);
324 PXP_SetAlphaSurfacePosition(PXP_ID, 0U, 0U, src_w - 1U, src_h - 1U);
325 PXP_SetAlphaSurfaceBlendConfig(PXP_ID, &asBlendConfig);
326 PXP_EnableAlphaSurfaceOverlayColorKey(PXP_ID, false);
327
328 /*Output buffer.*/
329 pxp_output_buffer_config_t outputBufferConfig = {
330 .pixelFormat = pxp_get_out_px_format(dest_cf),
331 .interlacedMode = kPXP_OutputProgressive,
332 .buffer0Addr = (uint32_t)(dest_buf + dest_stride * dest_area->y1 + dest_px_size * dest_area->x1),
333 .buffer1Addr = (uint32_t)0U,
334 .pitchBytes = dest_stride,
335 .width = dest_w,
336 .height = dest_h
337 };
338 PXP_SetOutputBufferConfig(PXP_ID, &outputBufferConfig);
339
340 lv_pxp_run();
341}
