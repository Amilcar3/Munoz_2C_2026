81static lv_tree_node_t * _lv_tree_class_create_node(const lv_tree_class_t * class_p, lv_tree_node_t * parent)
82{
83 uint32_t s = get_instance_size(class_p);
84 lv_tree_node_t * node = lv_malloc(s);
85 if(node == NULL) return NULL;
86 lv_memzero(node, s);
87 node->class_p = class_p;
88 node->parent = parent;
89 node->child_cap = INIT_CHILDREN_CAP;
90 node->children = lv_malloc(sizeof(lv_tree_node_t *) * node->child_cap);
91 if(parent != NULL) {
92 parent->child_cnt++;
93 if(parent->child_cnt == parent->child_cap) {
94 parent->child_cap <<= 1;
95 parent->children = lv_realloc(parent->children, sizeof(lv_tree_node_t *) * parent->child_cap);
96 }
97 parent->children[parent->child_cnt - 1] = node;
98 }
99 return node;
100}

139bool lv_tree_walk(const lv_tree_node_t * node,
140 lv_tree_walk_mode_t mode,
141 lv_tree_traverse_cb_t cb,
142 lv_tree_before_cb_t bcb,
143 lv_tree_after_cb_t acb,
144 void * user_data)
145{
146 if(node && cb) {
147 if(mode == LV_TREE_WALK_PRE_ORDER) {
148 if(bcb && !bcb(node, user_data)) {
149 return false;
150 }
151 if(!cb(node, user_data)) {
152 return false;
153 }
154 }
155 for(uint32_t i = 0; i < node->child_cnt; i++) {
156 if(!lv_tree_walk(node->children[i], mode, cb, bcb, acb, user_data)) {
157 return false;
158 }
159 }
160 if(mode == LV_TREE_WALK_PRE_ORDER) {
161 if(acb) {
162 acb(node, user_data);
163 }
164 }
165
166 if(mode == LV_TREE_WALK_POST_ORDER) {
167 if(bcb && !bcb(node, user_data)) {
168 return false;
169 }
170 if(!cb(node, user_data)) {
171 return false;
172 }
173 if(acb) {
174 acb(node, user_data);
175 }
176 return true;
177 }
178 else {
179 return true;
180 }
181 }
182 else {
183 return true;
184 }
