18void test_refr_sync_areas_of_double_buffered_direct_mode(void)
19{
20 refr_disp_create(64, 64, LV_COLOR_FORMAT_XRGB8888, LV_DISPLAY_RENDER_MODE_DIRECT, 2, 64);
21 refr_screen_set_color(REFR_COLOR_BLACK);
22 lv_obj_invalidate(refr_screen());
23 refr_frame();
24
25 /*Render a red rectangle into one of the buffers only*/
26 const lv_draw_buf_t * rendered_buf = refr_ctx.disp->buf_act;
27 refr_rect_create(refr_screen(), 0, 0, 20, 20, REFR_COLOR_RED);
28 lv_obj_update_layout(refr_screen());
29 lv_area_t obj_area = {0, 0, 19, 19};
30 refr_ctx.disp->inv_p = 0;
31 lv_inv_area(refr_ctx.disp, &obj_area);
32 refr_frame();
33 ASSERT_PX_EQ(REFR_COLOR_RED, refr_buf_px(rendered_buf, 10, 10));
34
35 const lv_draw_buf_t * next_buf = refr_ctx.disp->buf_act;
36 TEST_ASSERT_NOT_EQUAL(rendered_buf, next_buf);
37 ASSERT_PX_EQ(REFR_COLOR_BLACK, refr_buf_px(next_buf, 10, 10));
38
39 /*Refresh a different area. Before rendering, the area updated in the previous
40 *frame is copied to the buffer that is now being rendered.*/
41 lv_area_t other = {40, 40, 59, 59};
42 refr_ctx.disp->inv_p = 0;
43 lv_inv_area(refr_ctx.disp, &other);
44 refr_frame();
45
46 ASSERT_PX_EQ(REFR_COLOR_RED, refr_buf_px(next_buf, 10, 10));
47}

49void test_refr_sync_area_is_dropped_when_redrawn(void)
50{
51 refr_disp_create(64, 64, LV_COLOR_FORMAT_XRGB8888, LV_DISPLAY_RENDER_MODE_DIRECT, 2, 64);
52 lv_display_set_sync_cb(refr_ctx.disp, refr_sync_cb);
53 refr_screen_set_color(REFR_COLOR_BLACK);
54 lv_obj_invalidate(refr_screen());
55 refr_frame();
56 refr_log_reset();
57
58 /*The previous frame covered the whole screen. Everything around the newly
59 *rendered area still has to be synchronized.*/
60 lv_area_t area = {10, 10, 29, 29};
61 refr_ctx.disp->inv_p = 0;
62 lv_inv_area(refr_ctx.disp, &area);
63 refr_frame();
64 TEST_ASSERT_EQUAL_UINT32(4, refr_ctx.sync_cnt);
65 uint32_t i;
66 for(i = 0; i < refr_ctx.sync_cnt; i++) {
67 lv_area_t common;
68 TEST_ASSERT_FALSE(lv_area_intersect(&common, &refr_ctx.sync_area[i], &area));
69 }
70
71 refr_log_reset();
72 /*Redraw exactly the same area: nothing is left to synchronize*/
73 refr_ctx.disp->inv_p = 0;
74 lv_inv_area(refr_ctx.disp, &area);
75 refr_frame();
76 TEST_ASSERT_EQUAL_UINT32(0, refr_ctx.sync_cnt);
77}
