73void lv_calendar_header_dropdown_set_year_list(lv_obj_t * parent, const char * years_list)
74{
75 LV_CHECK_OBJ(parent, &lv_calendar_class, return);
76 LV_CHECK_ARG(years_list != NULL, return);
77
78 /* Search for the header dropdown */
79 lv_obj_t * header = lv_obj_get_child_by_type(parent, 0, &lv_calendar_header_dropdown_class);
80 if(NULL == header) {
81 /* Header not found */
82 return;
83 }
84
85 /* Search for the year dropdown
86 * Index is 0 because in the header dropdown constructor the year dropdown (year_dd)
87 * is the first created child of the header */
88 const int32_t year_dropdown_index = 0;
89 lv_obj_t * year_dropdown = lv_obj_get_child_by_type(header, year_dropdown_index, &lv_dropdown_class);
90 if(NULL == year_dropdown) {
91 /* year dropdown not found */
92 return;
93 }
94
95 lv_dropdown_clear_options(year_dropdown);
96 lv_dropdown_set_options(year_dropdown, years_list);
97
98 lv_obj_invalidate(parent);
99}

105static void my_constructor(const lv_obj_class_t * class_p, lv_obj_t * obj)
106{
107 LV_TRACE_OBJ_CREATE("begin");
108
109 LV_UNUSED(class_p);
110 LV_ASSERT(obj != NULL);
111
112 lv_obj_t * calendar = lv_obj_get_parent(obj);
113 LV_ASSERT(calendar != NULL);
114 lv_obj_move_to_index(obj, 0);
115 lv_obj_set_flex_flow(obj, LV_FLEX_FLOW_ROW);
116
117 lv_obj_t * year_dd = lv_dropdown_create(obj);
118 lv_dropdown_set_options(year_dd, year_list);
119 lv_obj_add_event_cb(year_dd, year_event_cb, LV_EVENT_VALUE_CHANGED, calendar);
120 lv_obj_set_flex_grow(year_dd, 1);
121
122 lv_obj_t * month_dd = lv_dropdown_create(obj);
123 lv_dropdown_set_options(month_dd, month_list);
124 lv_obj_add_event_cb(month_dd, month_event_cb, LV_EVENT_VALUE_CHANGED, calendar);
125 lv_obj_set_flex_grow(month_dd, 1);
126
127 lv_obj_add_event_cb(obj, value_changed_event_cb, LV_EVENT_VALUE_CHANGED, NULL);
128
129 /*Refresh the drop down*/
130 lv_obj_send_event(obj, LV_EVENT_VALUE_CHANGED, NULL);
131}

133static void month_event_cb(lv_event_t * e)
134{
135 LV_ASSERT(e != NULL);
136 lv_obj_t * dropdown = lv_event_get_current_target(e);
137 lv_obj_t * calendar = lv_event_get_user_data(e);
138 LV_ASSERT(dropdown != NULL);
139 LV_ASSERT(calendar != NULL);
140
141 uint32_t sel = lv_dropdown_get_selected(dropdown);
142
143 const lv_calendar_date_t * d;
144 d = lv_calendar_get_showed_date(calendar);
145 lv_calendar_date_t newd = *d;
146 newd.month = sel + 1;
147
148 lv_calendar_set_month_shown(calendar, newd.year, newd.month);
149}

151static void year_event_cb(lv_event_t * e)
152{
153 LV_ASSERT(e != NULL);
154 lv_obj_t * dropdown = lv_event_get_current_target(e);
155 lv_obj_t * calendar = lv_event_get_user_data(e);
156 LV_ASSERT(dropdown != NULL);
157 LV_ASSERT(calendar != NULL);
158
159 uint32_t sel = lv_dropdown_get_selected(dropdown);
160
161 const char * year_p = lv_dropdown_get_options(dropdown);
162
163 LV_ASSERT(3 + sel * 5 < lv_strlen(year_p));
164
165 /* NOTE: Assumes YYYY format */
166 const uint32_t year = (year_p[0 + sel * 5] - '0') * 1000 +
167 (year_p[1 + sel * 5] - '0') * 100 +
168 (year_p[2 + sel * 5] - '0') * 10 +
169 (year_p[3 + sel * 5] - '0');
170
171 const lv_calendar_date_t * d;
172 d = lv_calendar_get_showed_date(calendar);
173 lv_calendar_date_t newd = *d;
174 newd.year = year;
175
176 lv_calendar_set_month_shown(calendar, newd.year, newd.month);
177}

179static void value_changed_event_cb(lv_event_t * e)
180{
181 LV_ASSERT(e != NULL);
182 lv_obj_t * header = lv_event_get_current_target(e);
183 LV_ASSERT(header != NULL);
184 lv_obj_t * calendar = lv_obj_get_parent(header);
185 LV_ASSERT(calendar != NULL);
186 const lv_calendar_date_t * cur_date = lv_calendar_get_showed_date(calendar);
187 lv_obj_t * year_dd = lv_obj_get_child(header, 0);
188 lv_obj_t * month_dd = lv_obj_get_child(header, 1);
189 const char * year_p = lv_dropdown_get_options(year_dd);
190 /* NOTE: Assumes YYYY format */
191 const int32_t first_year = (year_p[0] - '0') * 1000 +
192 (year_p[1] - '0') * 100 +
193 (year_p[2] - '0') * 10 + (year_p[3] - '0');
194 lv_dropdown_set_selected(year_dd, LV_ABS(first_year - cur_date->year));
195 lv_dropdown_set_selected(month_dd, cur_date->month - 1);
196}
