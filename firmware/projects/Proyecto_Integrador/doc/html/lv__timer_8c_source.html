58LV_ATTRIBUTE_TIMER_HANDLER uint32_t lv_timer_handler(void)
59{
60 LV_TRACE_TIMER("begin");
61
62 lv_timer_state_t * state_p = &state;
63 /*Avoid concurrent running of the timer handler*/
64 if(state_p->already_running) {
65 LV_TRACE_TIMER("already running, concurrent calls are not allow, returning");
66 return 1;
67 }
68 state_p->already_running = true;
69
70 if(state_p->lv_timer_run == false) {
71 state_p->already_running = false; /*Release mutex*/
72 return 1;
73 }
74
75 LV_PROFILER_TIMER_BEGIN;
76 lv_lock();
77
78 uint32_t handler_start = lv_tick_get();
79
80 if(handler_start == 0) {
81 state.run_cnt++;
82 if(state.run_cnt > 100) {
83 state.run_cnt = 0;
84 LV_LOG_WARN("It seems lv_tick_inc() is not called.");
85 }
86 }
87
88 /*Run all timer from the list*/
89 lv_timer_t * next;
90 lv_timer_t * timer_active;
91 lv_ll_t * timer_head = timer_ll_p;
92 do {
93 state_p->timer_deleted = false;
94 state_p->timer_created = false;
95
96 timer_active = lv_ll_get_head(timer_head);
97 while(timer_active) {
98 /*The timer might be deleted if it runs only once ('repeat_count = 1')
99 *So get next element until the current is surely valid*/
100 next = lv_ll_get_next(timer_head, timer_active);
101
102 if(lv_timer_exec(timer_active)) {
103 /*If a timer was created or deleted then this or the next item might be corrupted*/
104 if(state_p->timer_created || state_p->timer_deleted) {
105 LV_TRACE_TIMER("Start from the first timer again because a timer was created or deleted");
106 break;
107 }
108 }
109
110 timer_active = next; /*Load the next timer*/
111 }
112 } while(timer_active);
113
114 uint32_t time_until_next = lv_timer_get_time_to_next();
115
116 state_p->busy_time += lv_tick_elaps(handler_start);
117 uint32_t idle_period_time = lv_tick_elaps(state_p->idle_period_start);
118 if(idle_period_time >= IDLE_MEAS_PERIOD) {
119 state_p->idle_last = (state_p->busy_time * 100) / idle_period_time; /*Calculate the busy percentage*/
120 state_p->idle_last = state_p->idle_last > 100 ? 0 : 100 - state_p->idle_last; /*But we need idle time*/
121 state_p->busy_time = 0;
122 state_p->idle_period_start = lv_tick_get();
123 }
124
125 state_p->timer_time_until_next = time_until_next;
126 state_p->already_running = false; /*Release the mutex*/
127
128 LV_TRACE_TIMER("finished (%" LV_PRIu32 " ms until the next timer call)", time_until_next);
129 lv_unlock();
130
131 LV_PROFILER_TIMER_END;
132 return time_until_next;
133}

135LV_ATTRIBUTE_TIMER_HANDLER uint32_t lv_timer_get_time_to_next(void)
136{
137 uint32_t time_until_next = LV_NO_TIMER_READY;
138 lv_ll_t * timer_head = timer_ll_p;
139 lv_timer_t * next = lv_ll_get_head(timer_head);
140
141 while(next && time_until_next) {
142 if(!next->paused) {
143 uint32_t timer_remaining = lv_timer_time_remaining(next);
144 time_until_next = LV_MIN(time_until_next, timer_remaining);
145 }
146
147 next = lv_ll_get_next(timer_head, next); /*Find the next timer*/
148 }
149
150 return time_until_next;
151}

168lv_timer_t * lv_timer_create(lv_timer_cb_t timer_xcb, uint32_t period, void * user_data)
169{
170 lv_timer_t * new_timer = NULL;
171
172 new_timer = lv_ll_ins_head(timer_ll_p);
173 LV_ASSERT_MALLOC(new_timer);
174 if(new_timer == NULL) return NULL;
175
176 new_timer->period = period;
177 new_timer->timer_cb = timer_xcb;
178 new_timer->repeat_count = -1;
179 new_timer->paused = 0;
180 new_timer->last_run = lv_tick_get();
181 new_timer->user_data = user_data;
182 new_timer->auto_delete = true;
183#if LV_USE_EXT_DATA
184 new_timer->ext_data.free_cb = NULL;
185 new_timer->ext_data.data = NULL;
186#endif
187
188 state.timer_created = true;
189
190 lv_timer_handler_resume();
191
192 return new_timer;
193}

339static bool lv_timer_exec(lv_timer_t * timer)
340{
341 if(timer->paused) return false;
342
343 bool exec = false;
344 if(lv_timer_time_remaining(timer) == 0) {
345 /* Decrement the repeat count before executing the timer_cb.
346 * If any timer is deleted `if(timer->repeat_count == 0)` is not executed below
347 * but at least the repeat count is zero and the timer can be deleted in the next round*/
348 int32_t original_repeat_count = timer->repeat_count;
349 if(timer->repeat_count > 0) timer->repeat_count--;
350 timer->last_run = lv_tick_get();
351 LV_TRACE_TIMER("calling timer callback: %p", *((void **)&timer->timer_cb));
352
353 if(timer->timer_cb && original_repeat_count != 0) {
354 LV_PROFILER_TIMER_BEGIN_TAG("timer_cb");
355 timer->timer_cb(timer);
356 LV_PROFILER_TIMER_END_TAG("timer_cb");
357 }
358
359 if(!state.timer_deleted) {
360 LV_TRACE_TIMER("timer callback %p finished", *((void **)&timer->timer_cb));
361 }
362 else {
363 LV_TRACE_TIMER("timer callback finished");
364 }
365
366 LV_ASSERT_MEM_INTEGRITY();
367 exec = true;
368 }
369
370 if(state.timer_deleted == false) { /*The timer might be deleted by itself as well*/
371 if(timer->repeat_count == 0) { /*The repeat count is over, delete the timer*/
372 if(timer->auto_delete) {
373 LV_TRACE_TIMER("deleting timer with %p callback because the repeat count is over", *((void **)&timer->timer_cb));
374 lv_timer_delete(timer);
375 }
376 else {
377 LV_TRACE_TIMER("pausing timer with %p callback because the repeat count is over", *((void **)&timer->timer_cb));
378 lv_timer_pause(timer);
379 }
380 }
381 }
382
383 return exec;
384}

391static uint32_t lv_timer_time_remaining(lv_timer_t * timer)
392{
393 /*Check if at least 'period' time elapsed*/
394 uint32_t elp = lv_tick_elaps(timer->last_run);
395 if(elp >= timer->period)
396 return 0;
397 return timer->period - elp;
