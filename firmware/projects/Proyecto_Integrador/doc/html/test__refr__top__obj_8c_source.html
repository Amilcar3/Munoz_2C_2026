18void test_refr_top_obj_finds_the_covering_object(void)
19{
20 refr_disp_create(64, 64, LV_COLOR_FORMAT_XRGB8888, LV_DISPLAY_RENDER_MODE_DIRECT, 1, 64);
21 refr_screen_set_color(REFR_COLOR_BLACK);
22 lv_obj_t * obj = refr_rect_create(refr_screen(), 10, 10, 40, 40, REFR_COLOR_RED);
23 lv_obj_update_layout(refr_screen());
24
25 /*Fully inside the object*/
26 lv_area_t inside = {20, 20, 29, 29};
27 TEST_ASSERT_EQUAL_PTR(obj, lv_refr_get_top_obj(&inside, refr_screen()));
28
29 /*Only partially on the object, so the screen covers it*/
30 lv_area_t crossing = {5, 5, 20, 20};
31 TEST_ASSERT_EQUAL_PTR(refr_screen(), lv_refr_get_top_obj(&crossing, refr_screen()));
32
33 /*Searching from the object itself finds the object*/
34 TEST_ASSERT_EQUAL_PTR(obj, lv_refr_get_top_obj(&inside, obj));
35
36 /*Outside of the given object*/
37 lv_area_t outside = {55, 55, 60, 60};
38 TEST_ASSERT_NULL(lv_refr_get_top_obj(&outside, obj));
39}
