1"""Runtime access to the field metadata behind the generated wrappers.
2
3The wrappers carry the values; what the values *mean* - which fields matter
4first, what an enum's numbers are called, how a widget summarises itself - lives
5in the generator's widget_specs.json. The dashboard already ships that file to
6the browser, so reading the same file here keeps the terminal output and the
7HTML inspector telling one story instead of two that drift apart.
8
9Everything degrades to the raw value when the file is missing, which is the case
10when the plugin is vendored without its scripts/ directory.
11"""

44def order_fields(class_name: str, keys) -> list:
45 """Order widget_data keys: the spec's primary fields first, rest after."""
46 primary = widget_spec(class_name).get("primary", [])
47 keys = list(keys)

52def format_value(class_name: str, field: str, value) -> str:
53 """Render one widget_data value: enums by name, points and areas inline."""
54 if value is None:
55 return "-"
56
57 names = widget_spec(class_name).get("enums", {}).get(field)
58 if names is not None and isinstance(value, int) and not isinstance(value, bool):
59 if 0 <= value < len(names):
60 return f"{names[value]} ({value})"
61 return str(value)
62
63 if isinstance(value, dict):
64 if {"x1", "y1", "x2", "y2"} <= value.keys():
65 return (
66 f"({value['x1']},{value['y1']})-({value['x2']},{value['y2']})"
67 )
68 if {"x", "y"} <= value.keys():
69 return f"({value['x']},{value['y']})"
70
71 if field_types(class_name).get(field) == "bool":
72 return str(bool(value))
73
74 if isinstance(value, str):
75 return f'"{escape(value)}"'
76
77 return str(value)

81 """Keep a value on one line: a label's text may well be several."""

85def summary(class_name: str, widget_data: dict, limit: int = 32) -> str:
