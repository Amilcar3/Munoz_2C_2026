37void * lv_utils_bsearch(const void * key, const void * base, size_t n, size_t size,
38 int (*cmp)(const void * pRef, const void * pElement))
39{
40 const char * middle;
41 int32_t c;
42
43 for(middle = base; n != 0;) {
44 middle += (n / 2) * size;
45 if((c = (*cmp)(key, middle)) > 0) {
46 n = (n / 2) - ((n & 1) == 0);
47 base = (middle += size);
48 }
49 else if(c < 0) {
50 n /= 2;
51 middle = base;
52 }
53 else {
54 return (char *)middle;
55 }
56 }
57 return NULL;
58}

60lv_result_t lv_draw_buf_save_to_file(const lv_draw_buf_t * draw_buf, const char * path)
61{
62 lv_fs_file_t file;
63 lv_fs_res_t res = lv_fs_open(&file, path, LV_FS_MODE_WR);
64 if(res != LV_FS_RES_OK) {
65 LV_LOG_ERROR("create file %s failed", path);
66 return LV_RESULT_INVALID;
67 }
68
69 /*Image content modified, invalidate image cache.*/
70 lv_image_cache_drop(path);
71
72 uint32_t bw;
73 res = lv_fs_write(&file, &draw_buf->header, sizeof(draw_buf->header), &bw);
74 if(res != LV_FS_RES_OK || bw != sizeof(draw_buf->header)) {
75 LV_LOG_ERROR("write draw_buf->header failed");
76 lv_fs_close(&file);
77 return LV_RESULT_INVALID;
78 }
79
80 res = lv_fs_write(&file, draw_buf->data, draw_buf->data_size, &bw);
81 if(res != LV_FS_RES_OK || bw != draw_buf->data_size) {
82 LV_LOG_ERROR("write draw_buf->data failed");
83 lv_fs_close(&file);
84 return LV_RESULT_INVALID;
85 }
86
87 lv_fs_close(&file);
88 LV_LOG_TRACE("saved draw_buf to %s", path);
89 return LV_RESULT_OK;
90}
