50void lv_demo_keypad_encoder(void)
51{
52 g = lv_group_create();
53 lv_group_set_default(g);
54
55 lv_indev_t * indev = NULL;
56 for(;;) {
57 indev = lv_indev_get_next(indev);
58 if(!indev) {
59 break;
60 }
61
62 lv_indev_type_t indev_type = lv_indev_get_type(indev);
63 if(indev_type == LV_INDEV_TYPE_KEYPAD) {
64 lv_indev_set_group(indev, g);
65 }
66
67 if(indev_type == LV_INDEV_TYPE_ENCODER) {
68 lv_indev_set_group(indev, g);
69 }
70 }
71
72 tv = lv_tabview_create(lv_screen_active());
73
74 t1 = lv_tabview_add_tab(tv, "Selectors");
75 t2 = lv_tabview_add_tab(tv, "Text input");
76
77 selectors_create(t1);
78 text_input_create(t2);
79
80 msgbox_create();
81}

195static void ta_event_cb(lv_event_t * e)
196{
197 lv_indev_t * indev = lv_indev_active();
198 if(indev == NULL) return;
199 lv_indev_type_t indev_type = lv_indev_get_type(indev);
200
201 lv_event_code_t code = lv_event_get_code(e);
202 lv_obj_t * ta = lv_event_get_target(e);
203 lv_obj_t * kb = lv_event_get_user_data(e);
204
205 if(code == LV_EVENT_CLICKED && indev_type == LV_INDEV_TYPE_ENCODER) {
206 lv_keyboard_set_textarea(kb, ta);
207 lv_obj_set_hidden(kb, false);
208 lv_group_focus_obj(kb);
209 lv_group_set_editing(lv_obj_get_group(kb), kb != NULL);
210 lv_obj_set_height(tv, LV_VER_RES / 2);
211 lv_obj_align(kb, LV_ALIGN_BOTTOM_MID, 0, 0);
212 }
213
214 if(code == LV_EVENT_READY || code == LV_EVENT_CANCEL) {
215 lv_obj_set_hidden(kb, true);
216 lv_obj_set_height(tv, LV_VER_RES);
217 }
218}
