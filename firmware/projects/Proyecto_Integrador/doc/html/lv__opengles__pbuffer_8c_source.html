63lv_display_t * lv_opengles_pbuffer_create(int32_t hor_res, int32_t ver_res)
64{
65 lv_pbuffer_ctx_t * ctx = lv_zalloc(sizeof(*ctx));
66 LV_ASSERT_MALLOC(ctx);
67 if(!ctx) {
68 LV_LOG_ERROR("Failed to allocate the pbuffer context");
69 return NULL;
70 }
71
72 ctx->width = hor_res;
73 ctx->height = ver_res;
74
75 lv_display_t * disp = lv_display_create(hor_res, ver_res);
76 if(!disp) {
77 LV_LOG_ERROR("Failed to create the pbuffer display");
78 lv_free(ctx);
79 return NULL;
80 }
81
82 ctx->egl_interface = (lv_egl_interface_t) {
83 .driver_data = ctx,
84 .native_display = EGL_DEFAULT_DISPLAY,
85 .egl_platform = EGL_PLATFORM_SURFACELESS_MESA,
86 .select_config = select_config_cb,
87 .flip_cb = flip_cb,
88 .create_surface_cb = create_surface_cb,
89 .destroy_surface_cb = NULL, /* default eglDestroySurface */
90 .create_window_cb = NULL,
91 .destroy_window_cb = NULL,
92 };
93
94 ctx->egl_ctx = lv_opengles_egl_context_create(&ctx->egl_interface);
95 if(!ctx->egl_ctx) {
96 LV_LOG_ERROR("Failed to create the EGL context for the pbuffer display");
97 lv_display_delete(disp);
98 lv_free(ctx);
99 return NULL;
100 }
101
102 if(init_render_target(ctx) != LV_RESULT_OK) {
103 lv_display_delete(disp);
104 lv_opengles_pbuffer_destroy(ctx);
105 return NULL;
106 }
107
108 ctx->draw_buf = lv_draw_buf_create(hor_res, ver_res, LV_COLOR_FORMAT_XRGB8888, LV_STRIDE_AUTO);
109 if(!ctx->draw_buf) {
110 LV_LOG_ERROR("Failed to create the pbuffer draw buffer");
111 lv_display_delete(disp);
112 lv_opengles_pbuffer_destroy(ctx);
113 return NULL;
114 }
115
116 lv_display_set_driver_data(disp, ctx);
117 lv_display_set_color_format(disp, LV_COLOR_FORMAT_XRGB8888);
118 lv_display_set_draw_buffers(disp, ctx->draw_buf, NULL);
119 lv_display_set_render_mode(disp, LV_DISPLAY_RENDER_MODE_FULL);
120 lv_display_set_flush_cb(disp, flush_cb);
121
122 LV_LOG_INFO("EGL pbuffer display created (%" LV_PRId32 "x%" LV_PRId32 ", OpenGL ES %d)",
123 hor_res, ver_res, (int)lv_opengles_egl_get_gles_version(ctx->egl_ctx));
124
125 return disp;
126}

128lv_result_t lv_opengles_pbuffer_read_to_draw_buf(lv_display_t * display, lv_draw_buf_t * draw_buf)
129{
130 lv_pbuffer_ctx_t * ctx = lv_display_get_driver_data(display);
131 if(!ctx || !draw_buf) {
132 return LV_RESULT_INVALID;
133 }
134
135 const int32_t w = draw_buf->header.w;
136 const int32_t h = draw_buf->header.h;
137
138 if(w != ctx->width || h != ctx->height) {
139 LV_LOG_ERROR("Draw buffer size mismatch: expected %" LV_PRId32 "x%" LV_PRId32
140 ", got %" LV_PRId32 "x%" LV_PRId32,
141 ctx->width, ctx->height, w, h);
142 return LV_RESULT_INVALID;
143 }
144
145 const lv_color_format_t cf = draw_buf->header.cf;
146 if(cf != LV_COLOR_FORMAT_XRGB8888 && cf != LV_COLOR_FORMAT_ARGB8888) {
147 LV_LOG_ERROR("Unsupported color format for readback: %d", (int)cf);
148 return LV_RESULT_INVALID;
149 }
150
151 GL_CALL(glBindFramebuffer(GL_FRAMEBUFFER, ctx->fbo));
152
153 /* OpenGL's origin is bottom-left and LVGL's is top-left,
154 * so the rows are read back in reverse.
155 */
156 for(int32_t y = 0; y < h; y++) {
157 uint8_t * row = lv_draw_buf_goto_xy(draw_buf, 0, h - 1 - y);
158 GL_CALL(glReadPixels(0, y, w, 1, GL_RGBA, GL_UNSIGNED_BYTE, row));
159
160 /* GL returns RGBA byte order
161 * LVGL XRGB8888/ARGB8888 is BGRA
162 */
163 uint8_t * px = row;
164 for(int32_t x = 0; x < w; x++) {
165 const uint8_t r = px[0];
166 px[0] = px[2];
167 px[2] = r;
168 px += 4;
169 }
170 }
171
172 return LV_RESULT_OK;
173}

210static lv_result_t init_render_target(lv_pbuffer_ctx_t * ctx)
211{
212 /* Render into an FBO rather than straight into the pbuffer. Mesa refuses
213 * glReadPixels on a multisampled pbuffer, and the FBO carries its own stencil
214 * buffer for NanoVG's clipping regardless of the config.
215 */
216 GL_CALL(glGenFramebuffers(1, &ctx->fbo));
217 GL_CALL(glBindFramebuffer(GL_FRAMEBUFFER, ctx->fbo));
218
219 GL_CALL(glGenTextures(1, &ctx->color_tex));
220 GL_CALL(glBindTexture(GL_TEXTURE_2D, ctx->color_tex));
221 GL_CALL(glTexImage2D(GL_TEXTURE_2D, 0, GL_RGBA, ctx->width, ctx->height, 0,
222 GL_RGBA, GL_UNSIGNED_BYTE, NULL));
223 GL_CALL(glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_NEAREST));
224 GL_CALL(glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_NEAREST));
225 GL_CALL(glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_S, GL_CLAMP_TO_EDGE));
226 GL_CALL(glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_T, GL_CLAMP_TO_EDGE));
227 GL_CALL(glFramebufferTexture2D(GL_FRAMEBUFFER, GL_COLOR_ATTACHMENT0, GL_TEXTURE_2D, ctx->color_tex, 0));
228
229 GL_CALL(glGenRenderbuffers(1, &ctx->stencil_rbo));
230 GL_CALL(glBindRenderbuffer(GL_RENDERBUFFER, ctx->stencil_rbo));
231 GL_CALL(glRenderbufferStorage(GL_RENDERBUFFER, GL_STENCIL_INDEX8, ctx->width, ctx->height));
232 GL_CALL(glFramebufferRenderbuffer(GL_FRAMEBUFFER, GL_STENCIL_ATTACHMENT, GL_RENDERBUFFER, ctx->stencil_rbo));
233
234 const GLenum status = glCheckFramebufferStatus(GL_FRAMEBUFFER);
235 if(status != GL_FRAMEBUFFER_COMPLETE) {
236 LV_LOG_ERROR("Pbuffer framebuffer is incomplete: %#x", status);
237 deinit_render_target(ctx);
238 return LV_RESULT_INVALID;
239 }
240
241 GL_CALL(glViewport(0, 0, ctx->width, ctx->height));
242
243 return LV_RESULT_OK;
244}

247{

255 }

258 ctx->fbo = 0;
259 }

292static size_t select_config_cb(void * driver_data, const lv_egl_config_t * configs, size_t config_count)
293{
294 lv_pbuffer_ctx_t * ctx = (lv_pbuffer_ctx_t *)driver_data;
295
296 for(size_t i = 0; i < config_count; ++i) {
297 LV_LOG_TRACE("Got config %zu %#x %dx%d RGBA(%d,%d,%d,%d) buffer size %d depth %d samples %d stencil %d "
298 "surface type %#x renderable type %#x",
299 i, configs[i].id, configs[i].max_width, configs[i].max_height,
300 configs[i].r_bits, configs[i].g_bits, configs[i].b_bits, configs[i].a_bits,
301 configs[i].buffer_size, configs[i].depth, configs[i].samples, configs[i].stencil,
302 configs[i].surface_type, configs[i].renderable_type);
303 }
304
305 for(size_t i = 0; i < config_count; ++i) {
306 const bool is_pbuffer = (configs[i].surface_type & EGL_PBUFFER_BIT) != 0;
307 const bool resolution_fits = configs[i].max_width >= ctx->width &&
308 configs[i].max_height >= ctx->height;
309 const bool is_rgb8 = configs[i].r_bits == 8 && configs[i].g_bits == 8 && configs[i].b_bits == 8;
310 const bool is_es2_capable = (configs[i].renderable_type & EGL_OPENGL_ES2_BIT) != 0;
311
312 if(is_pbuffer && resolution_fits && is_rgb8 && is_es2_capable && configs[i].samples == 0) {
313 LV_LOG_INFO("Choosing pbuffer config %zu (%#x)", i, configs[i].id);
314 return i;
315 }
316 }
317
318 LV_LOG_ERROR("Failed to find a suitable pbuffer EGL config");
319 return config_count;
320}
