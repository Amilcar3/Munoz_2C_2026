60void lv_gridnav_add(lv_obj_t * obj, lv_gridnav_ctrl_t ctrl)
61{
62 LV_CHECK_ARG(obj != NULL, return);
63
64 lv_gridnav_remove(obj); /*Be sure to not add gridnav twice*/
65
66 lv_gridnav_dsc_t * dsc = lv_malloc(sizeof(lv_gridnav_dsc_t));
67 LV_ASSERT_MALLOC(dsc);
68 dsc->ctrl = ctrl;
69 dsc->focused_obj = NULL;
70 lv_obj_add_event_cb(obj, gridnav_event_cb, LV_EVENT_ALL, dsc);
71
72 lv_obj_set_scroll_with_arrow(obj, false);
73}

75void lv_gridnav_remove(lv_obj_t * obj)
76{
77 LV_CHECK_ARG(obj != NULL, return);
78
79 lv_event_dsc_t * event_dsc = NULL;
80 uint32_t event_cnt = lv_obj_get_event_count(obj);
81 uint32_t i;
82 for(i = 0; i < event_cnt; i++) {
83 event_dsc = lv_obj_get_event_dsc(obj, i);
84 if(lv_event_dsc_get_cb(event_dsc) == gridnav_event_cb) {
85 lv_free(lv_event_dsc_get_user_data(event_dsc));
86 lv_obj_remove_event(obj, i);
87 break;
88 }
89 }

93void lv_gridnav_set_focused(lv_obj_t * cont, lv_obj_t * to_focus, lv_anim_enable_t anim_en)
94{
95 LV_CHECK_ARG(cont != NULL, return);
96 LV_CHECK_ARG(to_focus != NULL, return);
97
98 uint32_t i;
99 uint32_t event_cnt = lv_obj_get_event_count(cont);
100 lv_gridnav_dsc_t * dsc = NULL;
101 for(i = 0; i < event_cnt; i++) {
102 lv_event_dsc_t * event_dsc = lv_obj_get_event_dsc(cont, i);
103 if(lv_event_dsc_get_cb(event_dsc) == gridnav_event_cb) {
104 dsc = lv_event_dsc_get_user_data(event_dsc);
105 break;
106 }
107 }
108
109 if(dsc == NULL) {
110 LV_LOG_WARN("`cont` is not a gridnav container");
111 return;
112 }
113
114 if(obj_is_focusable(to_focus) == false) {
115 LV_LOG_WARN("The object to focus is not focusable");
116 return;
117 }
118
119 if(dsc->focused_obj) {
120 lv_obj_remove_state(dsc->focused_obj, LV_STATE_FOCUSED | LV_STATE_FOCUS_KEY);
121 }
122
123 lv_obj_add_state(to_focus, (lv_state_t)(LV_STATE_FOCUSED | LV_STATE_FOCUS_KEY));
124 lv_obj_scroll_to_view(to_focus, anim_en);
125 dsc->focused_obj = to_focus;
126

133static void gridnav_event_cb(lv_event_t * e)
134{
135 LV_ASSERT(e != NULL);
136 lv_obj_t * obj = lv_event_get_current_target(e);
137 lv_gridnav_dsc_t * dsc = lv_event_get_user_data(e);
138 LV_ASSERT(obj != NULL);
139 LV_ASSERT(dsc != NULL);
140 lv_event_code_t code = lv_event_get_code(e);
141
142 if(code == LV_EVENT_KEY) {
143 uint32_t child_cnt = lv_obj_get_child_count(obj);
144 if(child_cnt == 0) return;
145
146 if(dsc->focused_obj == NULL) dsc->focused_obj = find_first_focusable(obj);
147 if(dsc->focused_obj == NULL) return;
148
149 uint32_t key = lv_event_get_key(e);
150 lv_obj_t * guess = NULL;
151
152 if(key == LV_KEY_RIGHT && !(dsc->ctrl & LV_GRIDNAV_CTRL_VERTICAL_MOVE_ONLY)) {
153 if((dsc->ctrl & LV_GRIDNAV_CTRL_SCROLL_FIRST) && lv_obj_is_scrollable(dsc->focused_obj) &&
154 lv_obj_get_scroll_right(dsc->focused_obj) > 0) {
155 int32_t d = lv_obj_get_width(dsc->focused_obj) / 4;
156 if(d <= 0) d = 1;
157 lv_obj_scroll_by_bounded(dsc->focused_obj, -d, 0, LV_ANIM_ON);
158 }
159 else {
160 guess = find_chid(obj, dsc->focused_obj, FIND_RIGHT);
161 if(guess == NULL) {
162 if(dsc->ctrl & LV_GRIDNAV_CTRL_ROLLOVER) {
163 guess = find_chid(obj, dsc->focused_obj, FIND_NEXT_ROW_FIRST_ITEM);
164 if(guess == NULL) guess = find_first_focusable(obj);
165 }
166 else {
167 lv_group_focus_next(lv_obj_get_group(obj));
168 }
169 }
170 }
171 }
172 else if(key == LV_KEY_LEFT && !(dsc->ctrl & LV_GRIDNAV_CTRL_VERTICAL_MOVE_ONLY)) {
173 if((dsc->ctrl & LV_GRIDNAV_CTRL_SCROLL_FIRST) && lv_obj_is_scrollable(dsc->focused_obj) &&
174 lv_obj_get_scroll_left(dsc->focused_obj) > 0) {
175 int32_t d = lv_obj_get_width(dsc->focused_obj) / 4;
176 if(d <= 0) d = 1;
177 lv_obj_scroll_by_bounded(dsc->focused_obj, d, 0, LV_ANIM_ON);
178 }
179 else {
180 guess = find_chid(obj, dsc->focused_obj, FIND_LEFT);
181 if(guess == NULL) {
182 if(dsc->ctrl & LV_GRIDNAV_CTRL_ROLLOVER) {
183 guess = find_chid(obj, dsc->focused_obj, FIND_PREV_ROW_LAST_ITEM);
184 if(guess == NULL) guess = find_last_focusable(obj);
185 }
186 else {
187 lv_group_focus_prev(lv_obj_get_group(obj));
188 }
189 }
190 }
191 }
192 else if(key == LV_KEY_DOWN && !(dsc->ctrl & LV_GRIDNAV_CTRL_HORIZONTAL_MOVE_ONLY)) {
193 if((dsc->ctrl & LV_GRIDNAV_CTRL_SCROLL_FIRST) && lv_obj_is_scrollable(dsc->focused_obj) &&
194 lv_obj_get_scroll_bottom(dsc->focused_obj) > 0) {
195 int32_t d = lv_obj_get_height(dsc->focused_obj) / 4;
196 if(d <= 0) d = 1;
197 lv_obj_scroll_by_bounded(dsc->focused_obj, 0, -d, LV_ANIM_ON);
198 }
199 else {
200 guess = find_chid(obj, dsc->focused_obj, FIND_BOTTOM);
201 if(guess == NULL) {
202 if(dsc->ctrl & LV_GRIDNAV_CTRL_ROLLOVER) {
203 guess = find_chid(obj, dsc->focused_obj, FIND_FIRST_ROW);
204 }
205 else {
206 lv_group_focus_next(lv_obj_get_group(obj));
207 }
208 }
209 }
210 }
211 else if(key == LV_KEY_UP && !(dsc->ctrl & LV_GRIDNAV_CTRL_HORIZONTAL_MOVE_ONLY)) {
212 if((dsc->ctrl & LV_GRIDNAV_CTRL_SCROLL_FIRST) && lv_obj_is_scrollable(dsc->focused_obj) &&
213 lv_obj_get_scroll_top(dsc->focused_obj) > 0) {
214 int32_t d = lv_obj_get_height(dsc->focused_obj) / 4;
215 if(d <= 0) d = 1;
216 lv_obj_scroll_by_bounded(dsc->focused_obj, 0, d, LV_ANIM_ON);
217 }
218 else {
219 guess = find_chid(obj, dsc->focused_obj, FIND_TOP);
220 if(guess == NULL) {
221 if(dsc->ctrl & LV_GRIDNAV_CTRL_ROLLOVER) {
222 guess = find_chid(obj, dsc->focused_obj, FIND_LAST_ROW);
223 }
224 else {
225 lv_group_focus_prev(lv_obj_get_group(obj));
226 }
227 }
228 }
229 }
230 else {
231 lv_group_t * group = lv_obj_get_group(obj);
232 if(group && lv_group_get_focused(group) == obj) {
233 lv_obj_send_event(dsc->focused_obj, LV_EVENT_KEY, &key);
234 }
235 }
236
237 if(guess && guess != dsc->focused_obj) {
238 lv_obj_remove_state(dsc->focused_obj, LV_STATE_FOCUSED | LV_STATE_FOCUS_KEY);
239 lv_obj_send_event(dsc->focused_obj, LV_EVENT_DEFOCUSED, lv_indev_active());
240 lv_obj_add_state(guess, (lv_state_t)(LV_STATE_FOCUSED | LV_STATE_FOCUS_KEY));
241 lv_obj_send_event(guess, LV_EVENT_FOCUSED, lv_indev_active());
242 lv_obj_scroll_to_view(guess, LV_ANIM_ON);
243 dsc->focused_obj = guess;
244 }
245 }
246 else if(code == LV_EVENT_FOCUSED) {
247 if(dsc->focused_obj == NULL) dsc->focused_obj = find_first_focusable(obj);
248 if(dsc->focused_obj) {
249 lv_obj_add_state(dsc->focused_obj, (lv_state_t)(LV_STATE_FOCUSED | LV_STATE_FOCUS_KEY));
250 lv_obj_remove_state(dsc->focused_obj, LV_STATE_PRESSED); /*Be sure the focuses obj is not stuck in pressed state*/
251 lv_obj_scroll_to_view(dsc->focused_obj, LV_ANIM_OFF);
252 }
253 }
254 else if(code == LV_EVENT_DEFOCUSED) {
255 if(dsc->focused_obj) {
256 lv_obj_remove_state(dsc->focused_obj, LV_STATE_FOCUSED | LV_STATE_FOCUS_KEY);
257 }
258 }
259 else if(code == LV_EVENT_CHILD_CREATED) {
260 lv_obj_t * child = lv_event_get_target(e);
261 if(lv_obj_get_parent(child) == obj) {
262 if(dsc->focused_obj == NULL) {
263 dsc->focused_obj = child;
264 if(lv_obj_has_state(obj, LV_STATE_FOCUSED)) {
265 lv_obj_add_state(child, (lv_state_t)(LV_STATE_FOCUSED | LV_STATE_FOCUS_KEY));
266 lv_obj_scroll_to_view(child, LV_ANIM_OFF);
267 }
268 }
269 }
270 }
271 else if(code == LV_EVENT_CHILD_DELETED) {
272 /*This event bubble, so be sure this object's child was deleted.
273 *As we don't know which object was deleted we can't make the next focused.
274 *So make the first object focused*/
275 lv_obj_t * target = lv_event_get_target(e);
276 if(target == obj) {
277 dsc->focused_obj = find_first_focusable(obj);
278 }
279 }
280 else if(code == LV_EVENT_DELETE) {
281 lv_gridnav_remove(obj);
282 }
283 else if(code == LV_EVENT_PRESSED || code == LV_EVENT_PRESSING || code == LV_EVENT_PRESS_LOST ||
284 code == LV_EVENT_SHORT_CLICKED || code == LV_EVENT_LONG_PRESSED || code == LV_EVENT_LONG_PRESSED_REPEAT ||
285 code == LV_EVENT_CLICKED || code == LV_EVENT_RELEASED) {
286 if(lv_group_get_focused(lv_obj_get_group(obj)) == obj) {
287 /*Forward press/release related event too*/
288 lv_indev_t * indev = lv_event_get_indev(e);
289 lv_indev_type_t t = indev != NULL ? lv_indev_get_type(indev) : LV_INDEV_TYPE_NONE;
290 if(t == LV_INDEV_TYPE_ENCODER || t == LV_INDEV_TYPE_KEYPAD) {
291 lv_obj_send_event(dsc->focused_obj, code, indev);
292 }
293 }
294 }
295}

297static lv_obj_t * find_chid(lv_obj_t * obj, lv_obj_t * start_child, find_mode_t mode)
298{
299 LV_ASSERT(obj != NULL);
300 LV_ASSERT(start_child != NULL);
301 int32_t x_start = get_x_center(start_child);
302 int32_t y_start = get_y_center(start_child);
303 uint32_t child_cnt = lv_obj_get_child_count(obj);
304 lv_obj_t * guess = NULL;
305 int32_t x_err_guess = LV_COORD_MAX;
306 int32_t y_err_guess = LV_COORD_MAX;
307 int32_t h_half = lv_obj_get_height(start_child) / 2;
308 int32_t h_max = lv_obj_get_height(obj) + lv_obj_get_scroll_top(obj) + lv_obj_get_scroll_bottom(obj);
309 uint32_t i;
310 for(i = 0; i < child_cnt; i++) {
311 lv_obj_t * child = lv_obj_get_child(obj, i);
312 if(child == start_child) continue;
313 if(obj_is_focusable(child) == false) continue;
314
315 int32_t x_err = 0;
316 int32_t y_err = 0;
317 switch(mode) {
318 case FIND_LEFT:
319 x_err = get_x_center(child) - x_start;
320 y_err = get_y_center(child) - y_start;
321 if(x_err >= 0) continue; /*It's on the right*/
322 if(LV_ABS(y_err) > h_half) continue; /*Too far*/
323 break;
324 case FIND_RIGHT:
325 x_err = get_x_center(child) - x_start;
326 y_err = get_y_center(child) - y_start;
327 if(x_err <= 0) continue; /*It's on the left*/
328 if(LV_ABS(y_err) > h_half) continue; /*Too far*/
329 break;
330 case FIND_TOP:
331 x_err = get_x_center(child) - x_start;
332 y_err = get_y_center(child) - y_start;
333 if(y_err >= 0) continue; /*It's on the bottom*/
334 break;
335 case FIND_BOTTOM:
336 x_err = get_x_center(child) - x_start;
337 y_err = get_y_center(child) - y_start;
338 if(y_err <= 0) continue; /*It's on the top*/
339 break;
340 case FIND_NEXT_ROW_FIRST_ITEM:
341 y_err = get_y_center(child) - y_start;
342 if(y_err <= 0) continue; /*It's on the top*/
343 x_err = lv_obj_get_x(child);
344 break;
345 case FIND_PREV_ROW_LAST_ITEM:
346 y_err = get_y_center(child) - y_start;
347 if(y_err >= 0) continue; /*It's on the bottom*/
348 x_err = obj->coords.x2 - child->coords.x2;
349 break;
350 case FIND_FIRST_ROW:
351 x_err = get_x_center(child) - x_start;
352 y_err = lv_obj_get_y(child);
353 break;
354 case FIND_LAST_ROW:
355 x_err = get_x_center(child) - x_start;
356 y_err = h_max - lv_obj_get_y(child);
357 }
358
359 if(guess == NULL ||
360 (y_err * y_err + x_err * x_err < y_err_guess * y_err_guess + x_err_guess * x_err_guess)) {
361 guess = child;
362 x_err_guess = x_err;
363 y_err_guess = y_err;
364 }
365 }
366 return guess;
367}

369static lv_obj_t * find_first_focusable(lv_obj_t * obj)
370{
371 LV_ASSERT(obj != NULL);
372 uint32_t child_cnt = lv_obj_get_child_count(obj);
373 uint32_t i;
374 for(i = 0; i < child_cnt; i++) {
375 lv_obj_t * child = lv_obj_get_child(obj, i);
376 if(obj_is_focusable(child)) return child;
377
378 }
379 return NULL;

383{
384 LV_ASSERT(obj != NULL);
385 uint32_t child_cnt = lv_obj_get_child_count(obj);
386 int32_t i;
387 for(i = child_cnt - 1; i >= 0; i--) {
388 lv_obj_t * child = lv_obj_get_child(obj, i);
389 if(obj_is_focusable(child)) return child;
390 }
391 return NULL;
