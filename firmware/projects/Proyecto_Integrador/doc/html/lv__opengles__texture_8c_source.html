56lv_display_t * lv_opengles_texture_create(int32_t w, int32_t h)
57{
58 lv_display_t * display = lv_opengles_texture_create_common(w, h);
59 if(!display) {
60 LV_LOG_ERROR("Failed to create display");
61 return NULL;
62 }
63 lv_opengles_texture_t * texture = lv_display_get_driver_data(display);
64 unsigned int texture_id = create_texture(w, h, lv_display_get_color_format(display));
65 if(texture_id == GL_NONE) {
66 LV_LOG_ERROR("Failed to create texture");
67 lv_display_delete(display);
68 return NULL;
69 }
70 texture->texture_id = texture_id;
71 texture->is_texture_owner = true;
72 /* Attach the texture to the display after the texture id has been set*/
73 lv_opengles_texture_attach_to_display(texture, display);
74 return display;
75}

92lv_result_t lv_opengles_texture_reshape(lv_opengles_texture_t * texture, lv_display_t * display, int32_t width,
93 int32_t height)
94{
95 LV_ASSERT(display != NULL);
96 LV_ASSERT(texture != NULL);
97 unsigned int new_texture = create_texture(width, height, lv_display_get_color_format(display));
98 if(new_texture == GL_NONE) {
99 LV_LOG_ERROR("Failed to reshape texture. Couldn't acquire new texture from GPU");
100 return LV_RESULT_INVALID;
101 }
102
103#if LV_USE_DRAW_NANOVG
104 static size_t LV_ATTRIBUTE_MEM_ALIGN dummy_buf;
105 lv_display_set_buffers(display, &dummy_buf, NULL,
106 lv_draw_buf_width_to_stride(width, lv_display_get_color_format(display)) * height,
107 lv_display_get_render_mode(display));
108 lv_display_set_draw_buf_handlers(display, lv_draw_nanovg_get_draw_buf_handlers());
109#elif LV_USE_DRAW_OPENGLES
110 static size_t LV_ATTRIBUTE_MEM_ALIGN dummy_buf;
111 lv_display_set_buffers(display, &dummy_buf, NULL, width * height * 4, LV_DISPLAY_RENDER_MODE_DIRECT);
112 lv_display_set_draw_buf_handlers(display, lv_draw_opengles_get_draw_buf_handlers());
113#else
114 uint32_t stride = lv_draw_buf_width_to_stride(width, lv_display_get_color_format(display));
115 uint32_t buf_size = stride * height;
116 uint8_t * buffer = lv_realloc(texture->fb1, buf_size);
117 LV_ASSERT_MALLOC(buffer);
118 if(!buffer) {
119 GL_CALL(glDeleteTextures(1, &new_texture));
120 LV_LOG_ERROR("Failed to reshape texture. Couldn't resize buffer");
121 return LV_RESULT_INVALID;
122 }
123 texture->fb1 = buffer;
124
125 lv_display_set_buffers(display, texture->fb1, NULL, buf_size, lv_display_get_render_mode(display));
126#endif /*LV_USE_DRAW_NANOVG*/
127
128 if(texture->is_texture_owner && texture->texture_id != 0) {
129 GL_CALL(glDeleteTextures(1, &texture->texture_id));
130 }
131 texture->texture_id = new_texture;
132 lv_opengles_texture_attach_to_display(texture, display);
133 return LV_RESULT_OK;
134}

145static lv_result_t lv_opengles_texture_create_draw_buffers(lv_opengles_texture_t * texture, lv_display_t * display)
146{
147 int32_t w = lv_display_get_horizontal_resolution(display);
148 int32_t h = lv_display_get_vertical_resolution(display);
149
150#if LV_USE_DRAW_NANOVG
151 LV_UNUSED(texture);
152 static size_t LV_ATTRIBUTE_MEM_ALIGN dummy_buf;
153 lv_display_set_buffers(display, &dummy_buf, NULL,
154 lv_draw_buf_width_to_stride(w, lv_display_get_color_format(display)) * h,
155 LV_DISPLAY_RENDER_MODE_DIRECT);
156 lv_display_set_draw_buf_handlers(display, lv_draw_nanovg_get_draw_buf_handlers());
157#elif LV_USE_DRAW_OPENGLES
158 LV_UNUSED(texture);
159 static size_t LV_ATTRIBUTE_MEM_ALIGN dummy_buf;
160 lv_display_set_buffers(display, &dummy_buf, NULL, w * h * 4, LV_DISPLAY_RENDER_MODE_DIRECT);
161 lv_display_set_draw_buf_handlers(display, lv_draw_opengles_get_draw_buf_handlers());
162#else
163 uint32_t stride = lv_draw_buf_width_to_stride(w, lv_display_get_color_format(display));
164 uint32_t buf_size = stride * h;
165 texture->fb1 = lv_malloc(buf_size);
166 LV_ASSERT_MALLOC(texture->fb1);
167 if(!texture->fb1) {
168 return LV_RESULT_INVALID;
169 }
170 lv_display_set_buffers(display, texture->fb1, NULL, buf_size, LV_DISPLAY_RENDER_MODE_DIRECT);
171#endif /*LV_USE_DRAW_NANOVG*/
172 return LV_RESULT_OK;
173}

186void lv_opengles_texture_render_display(lv_opengles_texture_t * texture, lv_display_t * display)
187{
188 LV_ASSERT(texture != NULL);
189 LV_ASSERT(display != NULL);
190
191 /*The rotation is applied while rendering, so the viewport is never rotated*/
192 lv_opengles_viewport(0, 0, lv_display_get_original_horizontal_resolution(display),
193 lv_display_get_original_vertical_resolution(display));
194
195#if LV_USE_DRAW_OPENGLES
196 /*The draw unit has rendered into the texture of the display already*/
197 LV_UNUSED(texture);
198 lv_opengles_render_display_texture_internal(display, false, true);
199#elif LV_USE_DRAW_NANOVG
200 /*NanoVG has rendered into the framebuffer directly*/
201 LV_UNUSED(texture);
202#else
203 /*Upload the software rendered buffer of the display, then render it*/
204 if(lv_opengles_texture_upload_display_buf(texture->texture_id, display, texture->fb1) != LV_RESULT_OK) {
205 return;
206 }
207
208 const lv_color_format_t cf = lv_display_get_color_format(display);
209 lv_opengles_render_params_t params;
210 lv_opengles_render_params_init(&params);
211 params.h_flip = false;
212 params.v_flip = false;
213 params.rb_swap = lv_opengles_color_format_is_rb_swap(cf);
214 params.cf = cf;
215 lv_opengles_render_display(display, &params);
216#endif /*LV_USE_DRAW_OPENGLES*/
217}

241static lv_display_t * lv_opengles_texture_create_common(int32_t w, int32_t h)
242{
243 lv_display_t * disp = lv_display_create(w, h);
244 if(!disp) {
245 LV_LOG_ERROR("Failed to create display");
246 return NULL;
247 }
248 lv_opengles_texture_t * texture = lv_malloc_zeroed(sizeof(lv_opengles_texture_t));
249 LV_ASSERT_MALLOC(texture);
250 if(!texture) {
251 LV_LOG_ERROR("Failed to create texture");
252 lv_display_delete(disp);
253 return NULL;
254 }
255 lv_result_t res = lv_opengles_texture_create_draw_buffers(texture, disp);
256 if(res != LV_RESULT_OK) {
257 LV_LOG_ERROR("Failed to create draw buffers");
258 lv_free(texture);
259 lv_display_delete(disp);
260 return NULL;
261 }
262 lv_display_set_resolution(disp, w, h);
263 lv_display_set_flush_cb(disp, flush_cb);
264 lv_display_set_driver_data(disp, texture);
265 lv_display_add_event_cb(disp, release_disp_cb, LV_EVENT_DELETE, disp);
266
267 lv_opengles_init();
268 return disp;
269}

271static unsigned int create_texture(int32_t w, int32_t h, lv_color_format_t cf)
272{
273 unsigned int texture;
274 GL_CALL(glGenTextures(1, &texture));
275 GL_CALL(glBindTexture(GL_TEXTURE_2D, texture));
276 GL_CALL(glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_NEAREST));
277 GL_CALL(glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_NEAREST));
278 GL_CALL(glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_S, GL_CLAMP_TO_EDGE));
279 GL_CALL(glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_T, GL_CLAMP_TO_EDGE));
280
281 /* set the dimensions and format to complete the texture */
282 if(lv_opengles_texture_upload_buf(texture, NULL, w, h, 0, cf) != LV_RESULT_OK) {
283 GL_CALL(glDeleteTextures(1, &texture));
284 return GL_NONE;
285 }
286
287#if 0
288 GL_CALL(glGenerateMipmap(GL_TEXTURE_2D));
289 GL_CALL(glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAX_LEVEL, 20));
290 GL_CALL(glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_NEAREST_MIPMAP_NEAREST));
291#else
292 GL_CALL(glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_NEAREST));
293#endif
294
295 GL_CALL(glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_NEAREST));
296 GL_CALL(glBindTexture(GL_TEXTURE_2D, GL_NONE));
297 return texture;
298}
