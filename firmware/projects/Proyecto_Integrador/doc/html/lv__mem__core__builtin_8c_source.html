68void lv_mem_init(void)
69{
70#if LV_USE_OS
71 lv_mutex_init(&state.mutex);
72#endif
73
74#if LV_MEM_ADR == 0
75#ifdef LV_MEM_POOL_ALLOC
76 state.tlsf = lv_tlsf_create_with_pool((void *)LV_MEM_POOL_ALLOC(LV_MEM_SIZE), LV_MEM_SIZE);
77#else
78 /*Allocate a large array to store the dynamically allocated data*/
79 static MEM_UNIT work_mem_int[LV_MEM_SIZE / sizeof(MEM_UNIT)] LV_ATTRIBUTE_LARGE_RAM_ARRAY;
80 state.tlsf = lv_tlsf_create_with_pool((void *)work_mem_int, LV_MEM_SIZE);
81#endif
82#else
83 state.tlsf = lv_tlsf_create_with_pool((void *)LV_MEM_ADR, LV_MEM_SIZE);
84#endif
85
86 lv_ll_init(&state.pool_ll, sizeof(lv_pool_t));
87
88 /*Record the first pool*/
89 lv_pool_t * pool_p = lv_ll_ins_tail(&state.pool_ll);
90 LV_ASSERT_MALLOC(pool_p);
91 *pool_p = lv_tlsf_get_pool(state.tlsf);
92
93#if LV_MEM_ADD_JUNK
94 LV_LOG_WARN("LV_MEM_ADD_JUNK is enabled which makes LVGL much slower");
95#endif
96}

107lv_mem_pool_t lv_mem_add_pool(void * mem, size_t bytes)
108{
109 LV_CHECK_ARG(mem != NULL, return NULL);
110 LV_CHECK_ARG(bytes > 0, return NULL);
111
112 lv_mem_pool_t new_pool = lv_tlsf_add_pool(state.tlsf, mem, bytes);
113 if(!new_pool) {
114 LV_LOG_WARN("failed to add memory pool, address: %p, size: %zu", mem, bytes);
115 return NULL;
116 }
117
118 lv_pool_t * pool_p = lv_ll_ins_tail(&state.pool_ll);
119 LV_ASSERT_MALLOC(pool_p);
120 *pool_p = new_pool;
121
122 return new_pool;
123}

125void lv_mem_remove_pool(lv_mem_pool_t pool)
126{
127 lv_pool_t * pool_p;
128 LV_LL_READ(&state.pool_ll, pool_p) {
129 if(*pool_p == pool) {
130 lv_ll_remove(&state.pool_ll, pool_p);
131 lv_free(pool_p);
132 lv_tlsf_remove_pool(state.tlsf, pool);
133 return;
134 }
135 }
136 LV_LOG_WARN("invalid pool: %p", pool);
137}

158{
159#if LV_USE_OS
160 lv_mutex_lock(&state.mutex);
161#endif
162
163 size_t old_size = lv_tlsf_block_size(p);
164 void * p_new = lv_tlsf_realloc(state.tlsf, p, new_size);
165
166 if(p_new) {
167 state.cur_used -= old_size;

175 return p_new;

178void lv_free_core(void * p)
179{
180 if(p == NULL) return;
181
182#if LV_USE_OS
183 lv_mutex_lock(&state.mutex);
184#endif
185
186#if LV_MEM_ADD_JUNK
187 lv_memset(p, 0xbb, lv_tlsf_block_size(p));
188#endif
189 size_t size = lv_tlsf_block_size(p);
190 lv_tlsf_free(state.tlsf, p);
191 if(state.cur_used > size) state.cur_used -= size;
192 else state.cur_used = 0;
193
194#if LV_USE_OS
195 lv_mutex_unlock(&state.mutex);
196#endif
197}

199void lv_mem_monitor_core(lv_mem_monitor_t * mon_p)
200{
201 LV_CHECK_ARG(mon_p != NULL, return);
202
203 /*Init the data*/
204 lv_memzero(mon_p, sizeof(lv_mem_monitor_t));
205 LV_TRACE_MEM("begin");
206
207 lv_pool_t * pool_p;
208 LV_LL_READ(&state.pool_ll, pool_p) {
209 lv_tlsf_walk_pool(*pool_p, lv_mem_walker, mon_p);
210 }
211
212 mon_p->used_pct = 100 - (uint64_t)100U * mon_p->free_size / mon_p->total_size;
213 if(mon_p->free_size > 0) {
214 mon_p->frag_pct = (uint64_t)mon_p->free_biggest_size * 100U / mon_p->free_size;
215 mon_p->frag_pct = 100 - mon_p->frag_pct;
216 }
217 else {
218 mon_p->frag_pct = 0; /*no fragmentation if all the RAM is used*/
219 }
220
221 mon_p->max_used = state.max_used;
222
223 LV_TRACE_MEM("finished");
224}

261static void lv_mem_walker(void * ptr, size_t size, int used, void * user)
262{
263 LV_UNUSED(ptr);
264
265 lv_mem_monitor_t * mon_p = user;
266 mon_p->total_size += size;
267 if(used) {
268 mon_p->used_cnt++;
269 }
270 else {
271 mon_p->free_cnt++;
272 mon_p->free_size += size;
273 if(size > mon_p->free_biggest_size)
274 mon_p->free_biggest_size = size;
275 }
276}
