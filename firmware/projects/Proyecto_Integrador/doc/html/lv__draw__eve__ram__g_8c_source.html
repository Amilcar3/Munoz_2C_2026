55bool lv_draw_eve_ramg_get_addr(uint32_t * addr_dst, uintptr_t key,
56 uint32_t addr_size, uint32_t addr_align)
57{
58 LV_ASSERT(key != 0);
59
60 lv_draw_eve_ramg_t * ramg = &lv_draw_eve_unit_g->ramg;
61
62 if(ramg->hash_table_cell_count == 0) {
63 ramg_init(ramg);
64 }
65
66 uint32_t key_hash = hash_key(key);
67 uint32_t table_index = key_hash % ramg->hash_table_cell_count;
68 lv_draw_eve_ramg_hash_table_cell_t * cell;
69
70 while(1) {
71 cell = &ramg->hash_table[table_index];
72
73 if(cell->key == key) {
74#if RAMG_DEBUG
75 ramg_debug(ramg, key_hash, table_index);
76#endif
77
78 *addr_dst = cell->addr;
79 return true;
80 }
81
82 if(cell->key == 0) {
83 break;
84 }
85
86 table_index++;
87 if(table_index >= ramg->hash_table_cell_count) table_index = 0;
88 }
89
90 uint32_t addr_ret = LV_ALIGN_UP(ramg->ramg_addr_end, addr_align);
91 uint32_t addr_new_end = addr_ret + addr_size;
92
93 if(addr_new_end > 1024 * 1024) {
94 LV_LOG_WARN("EVE on-chip 1 MB RAM_G for images and fonts has run out.");
95 *addr_dst = LV_DRAW_EVE_RAMG_OUT_OF_RAMG;
96 return false;
97 }
98
99 ramg->ramg_addr_end = addr_new_end;
100 ramg->hash_table_cells_occupied++;
101
102 cell->key = key;
103 cell->addr = addr_ret;
104 *addr_dst = addr_ret;
105
106#if RAMG_DEBUG
107 ramg_debug(ramg, key_hash, table_index);
108#endif
109
110 if(ramg->hash_table_cells_occupied > ramg->hash_table_cell_count / 4 * 3) {
111 grow_hash_table(ramg);
112 }
113
114 return false;
115}

128static uint32_t hash_key(uintptr_t key)
129{
130 return fnv_1a_hash(&key, sizeof(key));
131}

133static uint32_t fnv_1a_hash(const void * src, size_t len)
134{
135 const uint8_t * src_u8 = src;
136 uint32_t hash = 2166136261u;
137 for(size_t i = 0; i < len; i++) {
138 hash ^= src_u8[i];
139 hash *= 16777619u;
140 }
141 return hash;
142}

144static void grow_hash_table(lv_draw_eve_ramg_t * ramg)
145{
146 uint32_t old_cell_count = ramg->hash_table_cell_count;
147 lv_draw_eve_ramg_hash_table_cell_t * old_hash_table = ramg->hash_table;
148
149 ramg->hash_table_cell_count += ramg->hash_table_cell_count / 2;
150 ramg->hash_table = lv_calloc(ramg->hash_table_cell_count,
151 sizeof(lv_draw_eve_ramg_hash_table_cell_t));
152 LV_ASSERT_MALLOC(ramg->hash_table);
153
154 for(uint32_t i = 0; i < old_cell_count; i++) {
155 lv_draw_eve_ramg_hash_table_cell_t * old_cell = &old_hash_table[i];
156
157 if(old_cell->key == 0) continue;
158
159 uint32_t key_hash = hash_key(old_cell->key);
160 uint32_t new_table_index = key_hash % ramg->hash_table_cell_count;
161 lv_draw_eve_ramg_hash_table_cell_t * new_cell_dst = &ramg->hash_table[new_table_index];
162
163 while(new_cell_dst->key != 0) {
164 new_table_index++;
165 if(new_table_index >= ramg->hash_table_cell_count) new_table_index = 0;
166 new_cell_dst = &ramg->hash_table[new_table_index];
167 }
168 *new_cell_dst = *old_cell;
169 }
170
171 lv_free(old_hash_table);
172}

174#if RAMG_DEBUG
175/*
176Print tables like this:
177 113 kB of RAM_G used
178 ================================-==-=====---=---===---=====-=-=-=-==--=======----=-==-==---=--===--=-=-=========================
179 ========^$==========--=-==-=-=--=--=========---=----========-------===--=====----=======--=====--====--=====-=--=-=
180
181'-' unoccupied cells
182'=' occupied cells
183'^' where the hash pointed to in the table initially and linear probing started
184'$' where linear probing ended because a matching or unoccupied cell
185 was found. This symbol is not shown if the initial guess was correct.
186
187This example has 244 cells. Each cell uses 8 bytes of local RAM, so just under 2 kB.
188Each cell represents an allocation in EVE RAM_G. The RAM_G allocation sizes are not
189represented in this table, except for the overall "113 kB of RAM_G used" message.
190*/
191static void ramg_debug(lv_draw_eve_ramg_t * ramg, uint32_t key_hash, uint32_t table_index)
192{
193 uint32_t table_index_initial_guess = key_hash % ramg->hash_table_cell_count;
194
195 lv_log("%u kB of RAM_G used\n", (unsigned) ramg->ramg_addr_end / 1024);
196
197 for(uint32_t i = 0; i < ramg->hash_table_cell_count; i++) {
198 if(i != 0 && i % 128 == 0) {
199 lv_log("\n");
200 }
201
202 if(i == table_index_initial_guess) {
203 lv_log("^");
204 }
205 else if(i == table_index) {
206 lv_log("$");
207 }
208 else if(ramg->hash_table[i].key) {
209 lv_log("=");
210 }
211 else {
212 lv_log("-");
213 }
214 }
215
216 lv_log("\n\n");
217}
