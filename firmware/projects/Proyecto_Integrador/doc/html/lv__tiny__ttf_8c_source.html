157void lv_tiny_ttf_set_size(lv_font_t * font, int32_t font_size)
158{
159 LV_CHECK_ARG(font != NULL, return);
160
161 if(font_size <= 0) {
162 LV_LOG_ERROR("invalid font size: %"LV_PRIx32, font_size);
163 return;
164 }
165 ttf_font_desc_t * dsc = (ttf_font_desc_t *)font->dsc;
166 dsc->scale = stbtt_ScaleForMappingEmToPixels(&dsc->info, font_size);
167 int line_gap = 0;
168 stbtt_GetFontVMetrics(&dsc->info, &dsc->ascent, &dsc->descent, &line_gap);
169 font->line_height = (int32_t)(dsc->scale * (dsc->ascent - dsc->descent + line_gap));
170 font->base_line = (int32_t)(dsc->scale * (line_gap - dsc->descent));
171
172 /* size change means cache needs to be invalidated. */
173
174 if(dsc->glyph_cache) {
175 lv_cache_destroy(dsc->glyph_cache, NULL);
176 dsc->glyph_cache = NULL;
177 }
178
179 if(dsc->draw_data_cache) {
180 lv_cache_destroy(dsc->draw_data_cache, NULL);
181 dsc->draw_data_cache = NULL;
182 }
183 if(dsc->kerning_cache) {
184 lv_cache_destroy(dsc->kerning_cache, NULL);
185 dsc->kerning_cache = NULL;
186 }
187
188 lv_tiny_ttf_cache_create(dsc);
189}
190
191void lv_tiny_ttf_destroy(lv_font_t * font)
192{
193 if(font == NULL) return;
194
195 if(font->dsc != NULL) {
196 ttf_font_desc_t * ttf = (ttf_font_desc_t *)font->dsc;
197#if LV_TINY_TTF_FILE_SUPPORT != 0
198 lv_mutex_lock(&ttf->stream.mutex);
199 if(ttf->stream.file != NULL) {
200 lv_fs_close(&ttf->file);
201 }
202 lv_mutex_unlock(&ttf->stream.mutex);
203 lv_mutex_delete(&ttf->stream.mutex);
204#endif
205 lv_cache_destroy(ttf->glyph_cache, NULL);
206 lv_cache_destroy(ttf->draw_data_cache, NULL);
207 lv_cache_destroy(ttf->kerning_cache, NULL);
208 lv_free(ttf);
209 font->dsc = NULL;
210 }
211
212 lv_free(font);
213}

220static void ttf_cb_stream_read(ttf_cb_stream_t * stream, void * data, size_t to_read)
221{
222 if(stream->file != NULL) {
223 uint32_t br;
224 lv_fs_read(stream->file, data, to_read, &br);
225 }
226 else {
227 if(to_read + stream->position >= stream->size) {
228 to_read = stream->size - stream->position;
229 }
230 memcpy(data, ((const unsigned char *)stream->data + stream->position), to_read);
231 stream->position += to_read;
232 }
233}
234static void ttf_cb_stream_seek(ttf_cb_stream_t * stream, size_t position)
235{
236 if(stream->file != NULL) {
237 lv_fs_seek(stream->file, position, LV_FS_SEEK_SET);
238 }
239 else {
240 if(position > stream->size) {
241 stream->position = stream->size;
242 }
243 else {
244 stream->position = position;
245 }
246 }
247}

253 ttf_cb_stream_seek(stream, position);

260static inline uint16_t ttf_calculate_kerning_width(float scale, uint16_t adv_w, int k)
261{
262
263 /*Horizontal space required by the glyph in [px]*/;
264 return (uint16_t)(scale * (adv_w + k) + 0.5f);

267static uint16_t ttf_get_glyph_pair_kerning_width(const ttf_font_desc_t * dsc, uint32_t g1, uint32_t g2, int adv_w)
268{
269 tiny_ttf_kerning_cache_data_t kerning_cache_search_key = {
270 .glyph1_idx = g1,
271 .glyph2_idx = g2,
272 };
273
274 tiny_ttf_kerning_cache_create_data_t kerning_cache_create_data = {
275 .adv_w = adv_w,
276 .dsc = dsc,
277 };
278
279 if(dsc->kerning_cache->max_size == 0) {
280 /* No cache, call the create function directly */
281 bool ret = tiny_ttf_kerning_cache_create_cb(&kerning_cache_search_key, (void *)&kerning_cache_create_data);
282 LV_ASSERT(ret);
283 LV_UNUSED(ret);
284 return kerning_cache_search_key.adv_w16;
285 }
286
287 LV_ASSERT_NULL(dsc->kerning_cache);
288 lv_cache_entry_t * kerning_entry = lv_cache_acquire_or_create(dsc->kerning_cache, &kerning_cache_search_key,
289 (void *)&kerning_cache_create_data);
290 LV_ASSERT_NULL(kerning_entry);
291 tiny_ttf_kerning_cache_data_t * data = lv_cache_entry_get_data(kerning_entry);
292 LV_ASSERT_NULL(data);
293
294 uint16_t adv_w16 = data->adv_w16;
295 lv_cache_release(dsc->kerning_cache, kerning_entry, NULL);
296 return adv_w16;
297}

299static bool ttf_get_glyph_dsc_cb(const lv_font_t * font, lv_font_glyph_dsc_t * dsc_out, uint32_t unicode_letter,
300 uint32_t unicode_letter_next)
301{
302 if(unicode_letter < 0x20 ||
303 unicode_letter == 0xf8ff || /*LV_SYMBOL_DUMMY*/
304 unicode_letter == 0x200c) { /*ZERO WIDTH NON-JOINER*/
305 dsc_out->box_w = 0;
306 dsc_out->adv_w = 0;
307 dsc_out->box_h = 0; /*height of the bitmap in [px]*/
308 dsc_out->ofs_x = 0; /*X offset of the bitmap in [pf]*/
309 dsc_out->ofs_y = 0; /*Y offset of the bitmap in [pf]*/
310 dsc_out->format = LV_FONT_GLYPH_FORMAT_NONE;
311 dsc_out->is_placeholder = false;
312 return true;
313 }
314
315 ttf_font_desc_t * dsc = (ttf_font_desc_t *)font->dsc;
316
317 tiny_ttf_glyph_cache_data_t search_key = {
318 .unicode = unicode_letter,
319 };
320
321
322 int adv_w;
323 lv_cache_entry_t * entry = lv_cache_acquire_or_create(dsc->glyph_cache, &search_key, (void *)dsc);
324
325 if(entry == NULL) {
326 if(!dsc->cache_size) { /* no cache, do everything directly */
327 int g1 = stbtt_FindGlyphIndex(&dsc->info, (int)unicode_letter);
328 if(g1 == 0) {
329 return false;
330 }
331 tiny_ttf_glyph_cache_create_cb(&search_key, dsc);
332 *dsc_out = search_key.glyph_dsc;
333 adv_w = search_key.adv_w;
334
335 /*Kerning correction*/
336 if(font->kerning == LV_FONT_KERNING_NORMAL &&
337 unicode_letter_next != 0) {
338 int g2 = stbtt_FindGlyphIndex(&dsc->info, (int)unicode_letter_next); /* not using cache, only do glyph id lookup */
339 if(g2) {
340 dsc_out->adv_w = ttf_get_glyph_pair_kerning_width(dsc, g1, g2, adv_w);
341 }
342 }
343
344 dsc_out->entry = NULL;
345 return true;
346 }
347 LV_LOG_ERROR("cache not allocated");
348 return false;
349 }
350
351 tiny_ttf_glyph_cache_data_t * data = lv_cache_entry_get_data(entry);
352 *dsc_out = data->glyph_dsc;
353 adv_w = data->adv_w;
354 lv_cache_release(dsc->glyph_cache, entry, NULL);
355
356 if(dsc_out->gid.index == 0) {
357 return false;
358 }
359
360 /*Kerning correction*/
361 if(font->kerning == LV_FONT_KERNING_NORMAL &&
362 unicode_letter_next != 0) { /* check if we need to do any kerning calculations */
363 uint32_t g1 = dsc_out->gid.index;
364
365 int g2 = 0;
366 search_key.unicode = unicode_letter_next; /* reuse search key */
367 lv_cache_entry_t * entry_next = lv_cache_acquire_or_create(dsc->glyph_cache, &search_key, (void *)dsc);
368
369 if(entry_next == NULL) {
370 g2 = stbtt_FindGlyphIndex(&dsc->info, (int)unicode_letter_next);
371 }
372 else {
373 tiny_ttf_glyph_cache_data_t * data_next = lv_cache_entry_get_data(entry_next);
374 g2 = data_next->glyph_dsc.gid.index;
375 lv_cache_release(dsc->glyph_cache, entry_next, NULL);
376 }
377 if(g2) {
378 dsc_out->adv_w = ttf_get_glyph_pair_kerning_width(dsc, g1, g2, adv_w);
379 }
380 }
381
382 dsc_out->entry = NULL;
383 return true; /*true: glyph found; false: glyph was not found*/
384}

386static const void * ttf_get_glyph_bitmap_cb(lv_font_glyph_dsc_t * g_dsc, lv_draw_buf_t * draw_buf)
387{
388 LV_UNUSED(draw_buf);
389 uint32_t glyph_index = g_dsc->gid.index;
390 const lv_font_t * font = g_dsc->resolved_font;
391 ttf_font_desc_t * dsc = (ttf_font_desc_t *)font->dsc;
392 tiny_ttf_cache_data_t search_key = {
393 .glyph_index = glyph_index,
394 .size = font->line_height,
395 };
396
397 lv_cache_entry_t * entry = lv_cache_acquire_or_create(dsc->draw_data_cache, &search_key, (void *)font->dsc);
398 if(entry == NULL) {
399 if(!dsc->cache_size) { /* no cache, do everything directly */
400 if(tiny_ttf_draw_data_cache_create_cb(&search_key, (void *)font->dsc)) {
401 /* use the cache entry to store the buffer if no cache specified */
402 g_dsc->entry = (lv_cache_entry_t *)search_key.draw_buf;
403 return g_dsc->entry;
404 }
405 else {
406 return NULL;
407 }
408 }
409 LV_LOG_ERROR("cache not allocated");
410 return NULL;
411 }
412
413 g_dsc->entry = entry;
414 tiny_ttf_cache_data_t * cached_data = lv_cache_entry_get_data(entry);
415 return cached_data->draw_buf;
416}

418static void ttf_release_glyph_cb(const lv_font_t * font, lv_font_glyph_dsc_t * g_dsc)
419{
420 LV_ASSERT_NULL(font);
421 if(!g_dsc) {
422 return;
423 }
424
425 ttf_font_desc_t * dsc = (ttf_font_desc_t *)font->dsc;
426 if(!dsc->cache_size) { /* no cache, do everything directly */
427 lv_draw_buf_destroy((lv_draw_buf_t *)g_dsc->entry);
428 }
429 else {
430 if(g_dsc->entry == NULL) {
431 return;
432 }
433 lv_cache_release(dsc->draw_data_cache, g_dsc->entry, NULL);
434 }
435 g_dsc->entry = NULL;
436}

467static lv_font_t * lv_tiny_ttf_create(const char * path, const void * data, size_t data_size, int32_t font_size,
468 lv_font_kerning_t kerning, size_t cache_size)
469{
470 LV_ASSERT(path != NULL || data != NULL);
471 LV_ASSERT(font_size > 0);
472 ttf_font_desc_t * dsc = lv_malloc_zeroed(sizeof(ttf_font_desc_t));
473 if(dsc == NULL) {
474 LV_LOG_ERROR("tiny_ttf: out of memory");
475 return NULL;
476 }
477#if LV_TINY_TTF_FILE_SUPPORT
478 lv_mutex_init(&dsc->stream.mutex);
479 if(path != NULL) {
480 if(LV_FS_RES_OK != lv_fs_open(&dsc->file, path, LV_FS_MODE_RD)) {
481 lv_free(dsc);
482 LV_LOG_ERROR("tiny_ttf: unable to open %s\n", path);
483 return NULL;
484 }
485 dsc->stream.file = &dsc->file;
486 }
487 else {
488 dsc->stream.data = (const uint8_t *)data;
489 dsc->stream.size = data_size;
490 }
491 if(0 == stbtt_InitFont(&dsc->info, &dsc->stream, stbtt_GetFontOffsetForIndex(&dsc->stream, 0))) {
492 lv_free(dsc);
493 LV_LOG_ERROR("tiny_ttf: init failed");
494 return NULL;
495 }
496
497#else
498 LV_UNUSED(data_size);
499 dsc->stream = (const uint8_t *)data;
500 if(0 == stbtt_InitFont(&dsc->info, dsc->stream, stbtt_GetFontOffsetForIndex(dsc->stream, 0))) {
501 lv_free(dsc);
502 LV_LOG_ERROR("tiny_ttf: init failed");
503 return NULL;
504 }
505#endif
506
507 dsc->cache_size = cache_size;
508
509 lv_font_t * out_font = lv_malloc_zeroed(sizeof(lv_font_t));
510 if(out_font == NULL) {
511 lv_free(dsc);
512 LV_LOG_ERROR("tiny_ttf: out of memory");
513 return NULL;
514 }
515
516 /* check if font has kerning tables to use, else disable kerning automatically. */
517 if(kerning != LV_FONT_KERNING_NONE && stbtt_KernTableCheck(&dsc->info) == 0) {
518 /* disable kerning if font has no tables. */
519 LV_LOG_INFO("Disabling kerning as font doesn't support it.");
520 kerning = LV_FONT_KERNING_NONE;
521 }
522
523 dsc->kerning = kerning;
524 out_font->kerning = kerning;
525
526 out_font->get_glyph_dsc = ttf_get_glyph_dsc_cb;
527 out_font->get_glyph_bitmap = ttf_get_glyph_bitmap_cb;
528 out_font->release_glyph = ttf_release_glyph_cb;
529 out_font->dsc = dsc;
530 lv_tiny_ttf_set_size(out_font, font_size);
531 return out_font;
532}

574static bool tiny_ttf_glyph_cache_create_cb(tiny_ttf_glyph_cache_data_t * node, void * user_data)
575{
576 LV_ASSERT(node != NULL);
577 LV_ASSERT(user_data != NULL);
578 ttf_font_desc_t * dsc = (ttf_font_desc_t *)user_data;
579 lv_font_glyph_dsc_t * dsc_out = &node->glyph_dsc;
580
581 uint32_t unicode_letter = node->unicode;
582
583 int g1 = stbtt_FindGlyphIndex(&dsc->info, (int)unicode_letter);
584 if(g1 == 0) {
585 /* Glyph not found */
586 lv_memzero(dsc_out, sizeof(*dsc_out));
587 node->adv_w = 0;
588 dsc_out->gid.index = 0;
589 return true;
590 }
591 int x1, y1, x2, y2;
592
593 stbtt_GetGlyphBitmapBox(&dsc->info, g1, dsc->scale, dsc->scale, &x1, &y1, &x2, &y2);
594
595 int advw;
596 int lsb;
597 stbtt_GetGlyphHMetrics(&dsc->info, g1, &advw, &lsb);
598 if(dsc->kerning != LV_FONT_KERNING_NORMAL) { /* calculate default advance */
599 dsc_out->adv_w = ttf_get_glyph_pair_kerning_width(dsc, g1, 0, advw);
600 }
601 else {
602 dsc_out->adv_w = ttf_calculate_kerning_width(dsc->scale, advw, 0);
603 }
604 /* precalculate no kerning value */
605 node->adv_w = advw;
606 dsc_out->box_w = (x2 - x1 + 1); /*width of the bitmap in [px]*/
607 dsc_out->box_h = (y2 - y1 + 1); /*height of the bitmap in [px]*/
608 dsc_out->ofs_x = x1; /*X offset of the bitmap in [pf]*/
609 dsc_out->ofs_y = -y2; /*Y offset of the bitmap measured from the as line*/
610 dsc_out->format = LV_FONT_GLYPH_FORMAT_A8;
611 dsc_out->is_placeholder = false;
612 dsc_out->gid.index = (uint32_t)g1;
613
614 return true;
615}
