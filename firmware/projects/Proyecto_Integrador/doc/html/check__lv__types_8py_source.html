1#!/usr/bin/env python3
2"""
3Check that every forward-declared struct/enum in include/lvgl/lv_types.h
4has a real definition (with a body) somewhere inside src/.
5
6Rules:
7 - Only `typedef struct _Tag` and `typedef enum _Tag` entries are checked.
8 - A definition is a line containing `struct _Tag {` or `enum _Tag {`.
9 - The `{` must be on the same line as the tag name.
10 - lv_types.h itself is never used as evidence of a definition.
11
12Exit codes:
13 0 – all forward declarations have a matching definition
14 1 – one or more forward declarations are dangling
15"""
16
17from __future__ import annotations
18
19import subprocess

29# Matches:
30# typedef struct _lv_obj_t lv_obj_t;
31# typedef enum _lv_align_t lv_align_t;
32FORWARD_DECL_RE = re.compile(
33 r"^\s*typedef\s+(struct|enum)\s+(\S+)\s+\S+\s*;",
34 re.MULTILINE,
35)

38def parse_forward_declarations(lv_types_h: Path) -> list[tuple[str, str]]:
39 """
40 Return [(kind, tag), ...] for every typedef struct/enum in lv_types.h.
41 kind is 'struct' or 'enum', tag is the backing name e.g. '_lv_obj_t'.
42 """
43 content = lv_types_h.read_text(encoding="utf-8", errors="ignore")
44 return [
45 (m.group(1), m.group(2))
46 for m in FORWARD_DECL_RE.finditer(content)
47 ]

50# ---------------------------------------------------------------------------
51# Searching
52# ---------------------------------------------------------------------------
53
54SOURCE_EXTENSIONS = {".c", ".cpp", ".h", ".hpp"}
55
56
57def find_definitions_slow(repo_root: Path, kind: str, tag: str) -> list[Path]:
58 """
59 Return all files under src_dir that contain a line like:
60 struct _lv_obj_t {
61 enum _lv_align_t {
62 (whitespace between tag and { is allowed, but both must be on one line)
63 """
64
65 src_dir = repo_root / "src"
66 include_dir = repo_root / "include" / "lvgl"
67 # e.g. struct\s+_lv_obj_t\s*\{
68 pattern = re.compile(
69 rf"\b{re.escape(kind)}\s+{re.escape(tag)}\s*\{{"
70 )
71 found = []
72
73 src_source_files = src_dir.rglob("*")
74 include_headers = include_dir.rglob("*")
75
76 for path in sorted(itertools.chain(src_source_files, include_headers)):
77 if path.suffix not in SOURCE_EXTENSIONS:
78 continue
79 try:
80 content = path.read_text(encoding="utf-8", errors="ignore")
81 except OSError:
82 continue
83 if pattern.search(content):
84 found.append(path)
85 return found

87def find_definitions(repo_root: Path, kind: str, tag: str) -> list[Path]:
88 # Construct the search pattern for grep
89 # \b ensures we match word boundaries
90 pattern = rf"\b{kind}\s+{tag}\s*\{{"
91
92 # Define the search paths
93 search_paths = [
94 str(repo_root / "src"),
95 str(repo_root / "include" / "lvgl")
96 ]
97
98 # Build the grep command:
99 # -r: recursive
100 # -l: only print names of files with matches
101 # -E: use extended regular expressions
102 # --include: filter for specific extensions
103 cmd = ["grep", "-rlE", pattern]
104 for ext in SOURCE_EXTENSIONS:
105 # Converts ".h" to "*.h" for grep
106 cmd.append(f"--include=*{ext}")
107
108 cmd.extend(search_paths)
109
110 try:
111 result = subprocess.run(
112 cmd,
113 capture_output=True,
114 text=True,
115 check=False
116 )
117
118 if result.stdout:
119 # Split lines and convert back to Path objects
120 return [Path(p) for p in result.stdout.splitlines()]
121 except FileNotFoundError:
122 # Fallback or error if grep isn't installed (e.g., on Windows)
123 print("Grep not found. Falling back to slow search.")
124 return find_definitions_slow(repo_root, kind, tag)
125
126 return []

133def check_forward_declarations(repo_root: Path) -> bool:
134 lv_types_h = repo_root / "include" / "lvgl" / "lv_types.h"
135
136 if not lv_types_h.exists():
137 print(f"ERROR: {lv_types_h} not found")
138 return False
139
140 declarations = parse_forward_declarations(lv_types_h)
141
142 if not declarations:
143 print("WARNING: no typedef struct/enum entries found in lv_types.h")
144 return True
145
146 print(f"Found {len(declarations)} forward declarations in lv_types.h")
147
148 dangling: list[tuple[str, str]] = []
149
150 for kind, tag in declarations:
151 print(f"{kind} {tag} ->", end= " ")
152 hits = find_definitions(repo_root, kind, tag)
153 if not hits:
154 print(f"KO")
155 dangling.append((kind, tag))
156 else:
157 for h in hits:
158 print(f"{h.relative_to(repo_root)} OK")
159
160 if dangling:
161 print(
162 f"DANGLING forward declarations in lv_types.h "
163 f"({len(dangling)} of {len(declarations)}):\n"
164 )
165 for kind, tag in dangling:
166 print(f" {kind} {tag}")
167 return False
168
169 print(
170 f"OK: all {len(declarations)} forward declarations in lv_types.h "
171 f"have a matching definition"
172 )
173 return True
