59void lv_draw_nanovg_blur(lv_draw_task_t * t, const lv_draw_blur_dsc_t * dsc, const lv_area_t * coords)
60{
61 if(dsc->blur_radius <= 0) return;
62
63 if(dsc->blur_radius > 256) {
64 LV_LOG_WARN("nanovg blur: radius %d exceeds backend limit (256), skipping",
65 (int)dsc->blur_radius);
66 return;
67 }
68
69 LV_PROFILER_DRAW_BEGIN;
70
71 lv_draw_nanovg_unit_t * u = (lv_draw_nanovg_unit_t *)t->draw_unit;
72 lv_layer_t * layer = t->target_layer;
73 NVGLUblurState * state = u->blur_state;
74
75 if(state == NULL) {
76 LV_LOG_WARN("nanovg blur: state not initialized (FBO not supported?), skipping");
77 LV_PROFILER_DRAW_END;
78 return;
79 }
80
81 /* Clip the blur area to clip_area and layer extent */
82 lv_area_t clip;
83 if(!lv_area_intersect(&clip, coords, &t->clip_area)) {
84 LV_PROFILER_DRAW_END;
85 return;
86 }
87 if(!lv_area_intersect(&clip, &clip, &layer->buf_area)) {
88 LV_PROFILER_DRAW_END;
89 return;
90 }
91
92 const int blur_w = lv_area_get_width(&clip);
93 const int blur_h = lv_area_get_height(&clip);
94 if(blur_w <= 0 || blur_h <= 0) {
95 LV_PROFILER_DRAW_END;
96 return;
97 }
98
99 /* Convert LVGL top-down coords to GL bottom-up coords */
100 const int layer_h = lv_area_get_height(&layer->buf_area);
101 const int rel_x = clip.x1 - layer->buf_area.x1;
102 const int rel_y_top = clip.y1 - layer->buf_area.y1;
103 const int gl_y = layer_h - rel_y_top - blur_h;
104
105 /* Determine source FBO */
106 const bool is_root = (layer->user_data == NULL);
107 NVGLUframebuffer * src_fb = NULL;
108 if(!is_root) {
109 src_fb = lv_nanovg_fbo_cache_entry_to_fb(layer->user_data);
110 if(src_fb == NULL) {
111 LV_PROFILER_DRAW_END;
112 return;
113 }
114 }
115
116 /* Flush pending NanoVG draws before raw GL operations */
117 lv_nanovg_end_frame(u);
118
119 /* Build blur params */
120 NVGLUblurParams params;
121 params.radius = dsc->blur_radius;
122 params.quality = (dsc->quality == LV_BLUR_QUALITY_SPEED)
123 ? NVGLU_BLUR_QUALITY_SPEED : NVGLU_BLUR_QUALITY_NORMAL;
124
125 /* Recolor for drop shadows */
126 if(dsc->base.drop_shadow_opa > 0) {
127 params.recolor = nvgRGBA(dsc->base.drop_shadow_color.red,
128 dsc->base.drop_shadow_color.green,
129 dsc->base.drop_shadow_color.blue,
130 255);
131 }
132 else {
133 params.recolor = nvgRGBA(0, 0, 0, 0);
134 }
135
136 /* Call the NanoVG blur utility */
137 int ret = nvgluBlurRegion(state, u->vg, src_fb, rel_x, gl_y, blur_w, blur_h, &params);
138 if(ret != 0) {
139 LV_LOG_WARN("nanovg blur: nvgluBlurRegion failed (ret=%d)", ret);
140 }
141
142 LV_PROFILER_DRAW_END;
143}
