111lv_result_t lv_gstreamer_set_src(lv_obj_t * obj, const char * factory_name, const char * property, const char * source)
112{
113 LV_CHECK_OBJ(obj, MY_CLASS, return LV_RESULT_INVALID);
114 LV_CHECK_ARG(factory_name != NULL, return LV_RESULT_INVALID);
115
116 lv_gstreamer_t * streamer = (lv_gstreamer_t *)obj;
117
118 if(streamer->pipeline) {
119 LV_LOG_WARN("LVGL doesn't allow modifying the GStreamer source. Create a new widget with a new src instead");
120 return LV_RESULT_INVALID;
121 }
122 GstElement * pipeline = gst_pipeline_new("lv_gstreamer_pipeline");
123 if(!pipeline) {
124 LV_LOG_ERROR("Failed to create gstreamer pipeline");
125 return LV_RESULT_INVALID;
126 }
127
128 GstElement * head = gst_element_factory_make(factory_name, "lv_gstreamer_source");
129 if(!head) {
130 gst_object_unref(pipeline);
131 LV_LOG_ERROR("Failed to create source from factory '%s'", factory_name);
132 return LV_RESULT_INVALID;
133 }
134
135 if(!gst_bin_add(GST_BIN(pipeline), head)) {
136 gst_object_unref(head);
137 gst_object_unref(pipeline);
138 LV_LOG_ERROR("Failed to add source element to pipeline");
139 return LV_RESULT_INVALID;
140 }
141
142 if(property != NULL && source != NULL) {
143 /* LV_GSTREAMER_PROPERTY_WEBRTCSRC is a child-proxy property path */
144 if(lv_streq(property, LV_GSTREAMER_PROPERTY_WEBRTCSRC)) {
145 if(!gstreamer_set_child_proxy_string(head, property, source)) {
146 gst_object_unref(pipeline);
147 LV_LOG_ERROR("Failed to set '%s' via child proxy", property);
148 return LV_RESULT_INVALID;
149 }
150 }
151 else {
152 g_object_set(G_OBJECT(head), property, source, NULL);
153 }
154 }
155
156 /* The uri decode source element will automatically handle parsing and decoding for us
157 * for other source types, we need to add a parser and a decoder ourselves element*/
158 if(!lv_streq(LV_GSTREAMER_FACTORY_URI_DECODE, factory_name)) {
159 /* webrtcsrc plugins require its own handling to be able to connect on the first available video stream */
160 if(lv_streq(LV_GSTREAMER_FACTORY_WEBRTCSRC, factory_name)) {
161 LV_LOG_INFO("Setting up webrtc pipeline");
162 if(gstreamer_element_has_property(head, "connect-to-first-producer")) {
163 g_object_set(G_OBJECT(head), "connect-to-first-producer", TRUE, NULL);
164 }
165 else {
166 LV_LOG_WARN("webrtcsrc property 'connect-to-first-producer' is not available in this plugin build");
167 }
168 }
169 else {
170 GstElement * decodebin = gst_element_factory_make("decodebin", "lv_gstreamer_decodebin");
171 if(!decodebin) {
172 gst_object_unref(pipeline);
173 LV_LOG_ERROR("Failed to create decodebin element");
174 return LV_RESULT_INVALID;
175 }
176 if(!gst_bin_add(GST_BIN(pipeline), decodebin)) {
177 gst_object_unref(decodebin);
178 gst_object_unref(pipeline);
179 LV_LOG_ERROR("Failed to add decodebin element to pipeline");
180 return LV_RESULT_INVALID;
181 }
182
183 if(!gst_element_link(head, decodebin)) {
184 gst_object_unref(pipeline);
185 LV_LOG_ERROR("Failed to link source with parsebin elements");
186 return LV_RESULT_INVALID;
187 }
188 head = decodebin;
189 }
190 }
191
192 /* At this point we don't yet know the input format
193 * Once the source starts receiving the data, it will create the necessary pads,
194 * i.e one pad for audio and one for video
195 * We add a callback so that we automatically connect to the data once it's figured out*/
196 g_signal_connect(head, "pad-added", G_CALLBACK(on_decode_pad_added), streamer);
197
198 streamer->color_format = gstreamer_resolve_color_format(obj);
199
200 streamer->pipeline = pipeline;
201 return LV_RESULT_OK;
202}

317 LV_CHECK_OBJ(obj, MY_CLASS, return LV_GSTREAMER_STATE_NULL);
318 lv_gstreamer_t * streamer = (lv_gstreamer_t *)obj;

337 return LV_GSTREAMER_STATE_PAUSED;
338 case GST_STATE_PLAYING:
339 return LV_GSTREAMER_STATE_PLAYING;

341 return LV_GSTREAMER_STATE_NULL;

432static lv_result_t gstreamer_poll_bus(lv_gstreamer_t * streamer)
433{
434 LV_ASSERT(streamer != NULL);
435 GstBus * bus = gst_element_get_bus(streamer->pipeline);
436 GstMessage * msg;
437
438 while((msg = gst_bus_pop(bus)) != NULL) {
439 const GstMessageType message_type = GST_MESSAGE_TYPE(msg);
440 switch(message_type) {
441 case GST_MESSAGE_ERROR: {
442 GError * err;
443 gchar * debug;
444 gst_message_parse_error(msg, &err, &debug);
445 GstObject * src = GST_MESSAGE_SRC(msg);
446 const gchar * name = NULL;
447 if(src != NULL) {
448 name = GST_OBJECT_NAME(src);
449 }
450 if(!name) {
451 name = "unknown";
452 }
453 LV_LOG_ERROR("GStreamer error from %s: %s", name, err->message);
454 if(debug && debug[0] != '\0') {
455 LV_LOG_ERROR("GStreamer error details: %s", debug);
456 }
457 g_error_free(err);
458 g_free(debug);
459 break;
460 }
461 case GST_MESSAGE_EOS:
462 if(gstreamer_send_state_changed(streamer, LV_GSTREAMER_STREAM_STATE_END) == LV_RESULT_INVALID) {
463 /* Object deleted inside event handler */
464 gst_object_unref(bus);
465 gst_message_unref(msg);
466 return LV_RESULT_INVALID;
467 }
468 break;
469 case GST_MESSAGE_STATE_CHANGED: {
470 GstState old_state, new_state;
471 gst_message_parse_state_changed(msg, &old_state, &new_state, NULL);
472 LV_LOG_INFO("State changed: %s -> %s", gst_element_state_get_name(old_state),
473 gst_element_state_get_name(new_state));
474 break;
475 }
476 default:
477 LV_LOG_TRACE("Received message %d", message_type);
478 break;
479 }
480 gst_message_unref(msg);
481 }
482 gst_object_unref(bus);
483 return LV_RESULT_OK;
484}

486static void gstreamer_update_frame(lv_gstreamer_t * streamer)
487{
488 LV_ASSERT(streamer != NULL);
489 GstSample * sample = g_async_queue_try_pop(streamer->frame_queue);
490
491 if(!sample) {
492 return;
493 }
494
495 const bool first_frame = !streamer->is_video_info_valid;
496 if(first_frame) {
497 GstCaps * caps = gst_sample_get_caps(sample);
498 if(!caps || !gst_video_info_from_caps(&streamer->video_info, caps)) {
499 LV_LOG_ERROR("Failed to get video info from caps");
500 gst_sample_unref(sample);
501 return;
502 }
503 streamer->is_video_info_valid = true;
504 }
505
506
507 GstBuffer * buffer = gst_sample_get_buffer(sample);
508 GstMapInfo map;
509 if(!buffer || !gst_buffer_map(buffer, &map, GST_MAP_READ)) {
510 gst_sample_unref(sample);
511 }
512 bool stored = gstreamer_store_frame(streamer, sample, buffer, &map);
513 if(first_frame && !stored) {
514 streamer->is_video_info_valid = false;
515 return;
516 }
517 lv_image_set_src((lv_obj_t *)streamer, &streamer->frame);
518
519 /* We send the event AFTER setting the image source so that users can query the
520 * resolution on this specific event callback */
521 if(first_frame) {
522 if(gstreamer_send_state_changed(streamer, LV_GSTREAMER_STREAM_STATE_START) == LV_RESULT_INVALID) {
523 /* Object deleted inside event handler */
524 return;
525 }
526 /*Send READY event for backwards compatibility with v9.4*/
527 lv_obj_send_event((lv_obj_t *)streamer, LV_EVENT_READY, streamer);
528 }

532static bool gstreamer_store_frame(lv_gstreamer_t * streamer, GstSample * sample, GstBuffer * buffer, GstMapInfo * map)
533{
534 const uint32_t w = GST_VIDEO_INFO_WIDTH(&streamer->video_info);
535 const uint32_t h = GST_VIDEO_INFO_HEIGHT(&streamer->video_info);
536 const uint32_t src_stride = GST_VIDEO_INFO_PLANE_STRIDE(&streamer->video_info, 0);
537
538 const bool aligned = (lv_uintptr_t)map->data % LV_DRAW_BUF_ALIGN == 0
539 && src_stride == lv_draw_buf_width_to_stride(w, streamer->color_format);
540
541 gstreamer_release_frame(streamer);
542
543 if(aligned) {
544 streamer->last_buffer = buffer;
545 streamer->last_map_info = *map;
546 streamer->last_sample = sample;
547 }
548 else {
549 const bool copied = gstreamer_copy_to_aligned_frame(streamer, map, w, h, src_stride);
550 gst_buffer_unmap(buffer, map);
551 gst_sample_unref(sample);
552 if(!copied) {
553 return false;
554 }
555 }
556
557 const lv_draw_buf_t * copy = streamer->aligned_frame;
558 streamer->frame = (lv_image_dsc_t) {
559 .data = aligned ? map->data : copy->data,
560 .data_size = aligned ? map->size : copy->data_size,
561 .header = {
562 .magic = LV_IMAGE_HEADER_MAGIC,
563 .cf = streamer->color_format,
564 .flags = LV_IMAGE_FLAGS_MODIFIABLE,
565 .h = h,
566 .w = w,
567 .stride = aligned ? src_stride : copy->header.stride,
568 }
569 };
570 return true;
571}

573static bool gstreamer_copy_to_aligned_frame(lv_gstreamer_t * streamer, const GstMapInfo * map, uint32_t w, uint32_t h,
574 uint32_t src_stride)
575{
576 lv_draw_buf_t * dest = streamer->aligned_frame;
577 if(dest && (dest->header.w != w || dest->header.h != h)) {
578 lv_draw_buf_destroy(dest);
579 dest = NULL;
580 streamer->aligned_frame = NULL;
581 }
582
583 if(!dest) {
584 dest = lv_draw_buf_create(w, h, streamer->color_format, LV_STRIDE_AUTO);
585 if(!dest) {
586 LV_LOG_ERROR("Failed to allocate an aligned %" LV_PRIu32 "x%" LV_PRIu32 " frame", w, h);
587 return false;
588 }
589 streamer->aligned_frame = dest;
590 }
591
592 const uint32_t dest_stride = dest->header.stride;
593 const uint32_t row_size = LV_MIN(src_stride, dest_stride);
594 for(uint32_t y = 0; y < h; y++) {
595 lv_memcpy(dest->data + y * dest_stride, map->data + y * src_stride, row_size);
596 }
597
598 lv_draw_buf_flush_cache(dest, NULL);
599 return true;
600}

629static void lv_gstreamer_destructor(const lv_obj_class_t * class_p, lv_obj_t * obj)
630{
631 LV_ASSERT(obj != NULL);
632 LV_UNUSED(class_p);
633 lv_gstreamer_t * streamer = (lv_gstreamer_t *)obj;
634
635 if(streamer->pipeline) {
636 gst_element_set_state(streamer->pipeline, GST_STATE_NULL);
637 gst_object_unref(streamer->pipeline);
638 }
639 gstreamer_release_frame(streamer);
640 if(streamer->aligned_frame) {
641 lv_draw_buf_destroy(streamer->aligned_frame);
642 }
643 if(streamer->frame_queue) {
644 GstSample * sample;
645 while((sample = g_async_queue_try_pop(streamer->frame_queue)) != NULL) {
646 gst_sample_unref(sample);
647 }
648 g_async_queue_unref(streamer->frame_queue);
649 streamer->frame_queue = NULL;
650 }
651 lv_timer_delete(streamer->gstreamer_timer);
652}

654static lv_result_t gstreamer_make_and_add_to_pipeline(lv_gstreamer_t * streamer,
655 const lv_gstreamer_pipeline_element_t * elements, size_t element_count)
656{
657 for(size_t i = 0; i < element_count; ++i) {
658 GstElement * el = gst_element_factory_make(elements[i].factory, elements[i].name);
659 if(!el) {
660 /* The previous elements were added to the pipeline so we don't need to unref them explicitly
661 * Unrefing the pipeline is enough and is done by caller*/
662 LV_LOG_ERROR("Failed to create %s element", elements[i].name);
663 return LV_RESULT_INVALID;
664 }
665 *(elements[i].store) = el;
666 if(!gst_bin_add(GST_BIN(streamer->pipeline), el)) {
667 gst_object_unref(el);
668 LV_LOG_ERROR("Failed to add %s element to pipeline", elements[i].name);
669 return LV_RESULT_INVALID;
670 }
671 }
672 return LV_RESULT_OK;
673}

742static void on_decode_pad_added(GstElement * element, GstPad * pad, gpointer user_data)
743{
744 LV_ASSERT(element != NULL);
745 LV_ASSERT(pad != NULL);
746 LV_UNUSED(element);
747 lv_gstreamer_t * streamer = (lv_gstreamer_t *)user_data;
748 GstCaps * caps = gst_pad_query_caps(pad, NULL);
749
750 GstStructure * structure = gst_caps_get_structure(caps, 0);
751 const gchar * name = gst_structure_get_name(structure);
752
753 LV_LOG_TRACE("Pad discovered %s", name);
754
755 if(g_str_has_prefix(name, "video/")) {
756 if(!streamer->video_convert) {
757 GstElement * video_app_sink;
758 GstElement * video_rate;
759 GstElement * video_queue;
760 const lv_gstreamer_pipeline_element_t elements[] = {
761 {"videoconvert", "lv_gstreamer_video_convert", &streamer->video_convert},
762 {"videorate", "lv_gstreamer_video_rate", &video_rate},
763 {"queue", "lv_gstreamer_video_queue", &video_queue},
764 {"appsink", "lv_gstreamer_video_sink", &video_app_sink},
765 };
766 const size_t element_count = sizeof(elements) / sizeof(elements[0]);
767 if(gstreamer_make_and_add_to_pipeline(streamer, elements, element_count) != LV_RESULT_OK) {
768 goto exit;
769 }
770
771 /* Here we set the fps we want the pipeline to produce and the color format
772 * This is achieved by the video_convert and video_rate elements that will automatically throttle and
773 * convert the image to the format we desire*/
774 uint32_t target_fps = 1000 / LV_DEF_REFR_PERIOD;
775 char caps_str[128];
776 lv_snprintf(caps_str, sizeof(caps_str), "video/x-raw,format=%s,framerate=%" LV_PRIu32 "/1",
777 gstreamer_get_gst_format(streamer->color_format), target_fps);
778
779 GstCaps * appsink_caps = gst_caps_from_string(caps_str);
780 g_object_set(G_OBJECT(video_app_sink), "emit-signals", TRUE, "sync", TRUE, "max-buffers", 1, "drop", TRUE, "caps",
781 appsink_caps, NULL);
782 gst_caps_unref(appsink_caps);
783
784 if(!gst_element_link_many(streamer->video_convert, video_rate, video_queue, video_app_sink, NULL)) {
785 LV_LOG_ERROR("Failed to link video convert to sink");
786 goto exit;
787 }
788 for(size_t i = 0; i < element_count; ++i) {
789 gst_element_sync_state_with_parent(*elements[i].store);
790 }
791 g_signal_connect(video_app_sink, "new-sample", G_CALLBACK(on_new_sample), streamer);
792 }
793
794 GstPad * video_convert_sink_pad = gst_element_get_static_pad(streamer->video_convert, "sink");
795 if(gst_pad_is_linked(video_convert_sink_pad)) {
796 LV_LOG_WARN("Received another video pad '%s' but our video pipeline is already linked - Ignoring", name);
797 }
798 else if(gst_pad_link(pad, video_convert_sink_pad) != GST_PAD_LINK_OK) {
799 LV_LOG_ERROR("Failed to link discovered pad '%s' to videoconvert", name);
800 }
801 gst_object_unref(video_convert_sink_pad);
802 }
803 else if(g_str_has_prefix(name, "audio/")) {
804 if(!streamer->audio_convert) {
805 GstElement * audio_resample;
806 GstElement * audio_sink;
807 const lv_gstreamer_pipeline_element_t elements[] = {
808 {"audioconvert", "lv_gstreamer_audio_convert", &streamer->audio_convert},
809 {"volume", "lv_gstreamer_audio_volume", &streamer->audio_volume},
810 {"audioresample", "lv_gstreamer_audio_resample", &audio_resample},
811 {"autoaudiosink", "lv_gstreamer_audio_sink", &audio_sink},
812 };
813 const size_t element_count = sizeof(elements) / sizeof(elements[0]);
814 if(gstreamer_make_and_add_to_pipeline(streamer, elements, element_count) != LV_RESULT_OK) {
815 goto exit;
816 }
817 if(!gst_element_link_many(streamer->audio_convert, audio_resample, streamer->audio_volume, audio_sink, NULL)) {
818 LV_LOG_ERROR("Failed to link audio convert to sink");
819 goto exit;
820 }
821 for(size_t i = 0; i < element_count; ++i) {
822 gst_element_sync_state_with_parent(*elements[i].store);
823 }
824 }
825
826 GstPad * audio_convert_sink_pad = gst_element_get_static_pad(streamer->audio_convert, "sink");
827 if(!gst_pad_is_linked(audio_convert_sink_pad)) {
828 if(gst_pad_link(pad, audio_convert_sink_pad) != GST_PAD_LINK_OK) {
829 LV_LOG_ERROR("Failed to link discovered pad '%s' to audioconvert", name);
830 }
831 }
832 else {
833 LV_LOG_WARN("Received another audio pad '%s' but our audio pipeline is already linked - Ignoring", name);
834 }
835 gst_object_unref(audio_convert_sink_pad);
836 }
837
838exit:
839 gst_caps_unref(caps);
840}

842static GstFlowReturn on_new_sample(GstElement * sink, gpointer user_data)
843{
844 LV_ASSERT(sink != NULL);
845 /* This function is called from a thread other than the main one so we can't call anything related to LVGL here
846 * Instead, we acquire the new sample (the new frame) and push it to the queue so that we can retrieve it from an LVGL timer
847 * Note that the pipeline spits out a new frame every LV_DEF_REFR_PERIOD as per the way it's set up so we shouldn't ever lose any
848 * frames with this method*/
849 lv_gstreamer_t * streamer = (lv_gstreamer_t *)user_data;
850 GstSample * sample;
851
852 g_signal_emit_by_name(sink, "pull-sample", &sample);
853 if(!sample) {
854 return GST_FLOW_OK;
855 }
856
857 g_async_queue_lock(streamer->frame_queue);
858 g_async_queue_push_unlocked(streamer->frame_queue, sample);
859 while(g_async_queue_length_unlocked(streamer->frame_queue) > LV_GSTREAMER_MAX_QUEUED_FRAMES) {
860 GstSample * dropped = g_async_queue_try_pop_unlocked(streamer->frame_queue);
861 if(!dropped) {
862 break;
863 }
864 gst_sample_unref(dropped);
865 }
866 g_async_queue_unlock(streamer->frame_queue);
867
868 return GST_FLOW_OK;
869}
