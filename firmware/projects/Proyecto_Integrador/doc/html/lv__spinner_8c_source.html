155static void arc_anim_angles(void * obj, int32_t v)
156{
157 lv_spinner_t * spinner = (lv_spinner_t *)obj;
158
159 /*Start angle (bezier)*/
160 int32_t step = lv_cubic_bezier(v, LV_BEZIER_VAL_FLOAT(0.42f), LV_BEZIER_VAL_FLOAT(0.58f),
161 LV_BEZIER_VAL_FLOAT(0.0f), LV_BEZIER_VAL_FLOAT(1.0f));
162 lv_value_precise_t start = (lv_value_precise_t)((step * 360) >> LV_BEZIER_VAL_SHIFT);
163
164 /*End angle (linear)*/
165 lv_value_precise_t end = spinner->angle + ((v * 360) >> LV_BEZIER_VAL_SHIFT);
166
167 lv_arc_set_angles(obj, start, end);
168}
