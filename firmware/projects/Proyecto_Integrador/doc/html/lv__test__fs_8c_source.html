102static void * fs_open(lv_fs_drv_t * drv, const char * path, lv_fs_mode_t mode)
103{
104 LV_ASSERT(path != NULL);
105 LV_UNUSED(drv);
106 lv_fs_file_t * file_p = lv_malloc(sizeof(lv_fs_file_t));
107 LV_ASSERT_MALLOC(file_p);
108 if(file_p == NULL) {
109 return NULL;
110 }
111
112 lv_fs_res_t res = lv_fs_open(file_p, path, mode);
113 if(res != LV_FS_RES_OK) {
114 lv_free(file_p);
115 return NULL;
116 }
117
118 return file_p;
119}
