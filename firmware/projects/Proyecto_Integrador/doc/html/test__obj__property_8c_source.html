31void test_obj_property_set_get_should_match(void)
32{
33#if LV_USE_OBJ_PROPERTY
34 lv_obj_t * obj = lv_obj_create(lv_screen_active());
35 lv_obj_t * root = lv_obj_create(lv_screen_active());
36 lv_property_t prop = { };
37 lv_color_t color = {.red = 0x11, .green = 0x22, .blue = 0x33};
38
39 /* Style property should work */
40 /* int type */
41 prop.id = LV_PROPERTY_STYLE_X;
42 prop.num = 0xaabb;
43 TEST_ASSERT_TRUE(lv_obj_set_property(obj, &prop) == LV_RESULT_OK);
44 TEST_ASSERT_EQUAL_UINT32(0xaabb, lv_obj_get_style_x(obj, LV_PART_MAIN));
45 TEST_ASSERT_EQUAL_UINT32(0xaabb, lv_obj_get_property(obj, LV_PROPERTY_STYLE_X).num);
46
47 /* color type */
48 prop.id = LV_PROPERTY_STYLE_BG_COLOR;
49 prop.color = color;
50 TEST_ASSERT_TRUE(lv_obj_set_property(obj, &prop) == LV_RESULT_OK);
51 TEST_ASSERT_EQUAL_COLOR(color, lv_obj_get_style_bg_color(obj, LV_PART_MAIN));
52 TEST_ASSERT_EQUAL_COLOR(color, lv_obj_get_property(obj, LV_PROPERTY_STYLE_BG_COLOR).color);
53
54 /* pointer type */
55 prop.id = LV_PROPERTY_STYLE_TEXT_FONT;
56 prop.ptr = &lv_font_montserrat_26;
57 TEST_ASSERT_TRUE(lv_obj_set_property(obj, &prop) == LV_RESULT_OK);
58 TEST_ASSERT_EQUAL_PTR(&lv_font_montserrat_26, lv_obj_get_style_text_font(obj, LV_PART_MAIN));
59 TEST_ASSERT_EQUAL_PTR(&lv_font_montserrat_26, lv_obj_get_property(obj, LV_PROPERTY_STYLE_TEXT_FONT).ptr);
60
61 /* Object flags (deprecated flag API kept for backward compatibility) */
62 LV_DEPRECATIONS_IGNORE_BEGIN
63 prop.id = LV_PROPERTY_OBJ_FLAG_HIDDEN ;
64 prop.num = 1;
65 TEST_ASSERT_TRUE(lv_obj_set_property(obj, &prop) == LV_RESULT_OK);
66 TEST_ASSERT_TRUE(lv_obj_has_flag(obj, LV_OBJ_FLAG_HIDDEN));
67 TEST_ASSERT_TRUE(lv_obj_get_property(obj, LV_PROPERTY_OBJ_FLAG_HIDDEN).num);
68
69 prop.id = LV_PROPERTY_OBJ_FLAG_CLICKABLE;
70 prop.num = 0;
71 TEST_ASSERT_TRUE(lv_obj_set_property(obj, &prop) == LV_RESULT_OK);
72 TEST_ASSERT_FALSE(lv_obj_has_flag(obj, LV_OBJ_FLAG_CLICKABLE));
73 TEST_ASSERT_FALSE(lv_obj_get_property(obj, LV_PROPERTY_OBJ_FLAG_CLICKABLE).num);
74 LV_DEPRECATIONS_IGNORE_END
75
76 /* Obj property */
77 prop.id = LV_PROPERTY_OBJ_PARENT;
78 prop.ptr = root;
79 TEST_ASSERT_TRUE(lv_obj_set_property(obj, &prop) == LV_RESULT_OK);
80 TEST_ASSERT_EQUAL_PTR(root, lv_obj_get_parent(obj));
81 TEST_ASSERT_EQUAL_PTR(root, lv_obj_get_property(obj, LV_PROPERTY_OBJ_PARENT).ptr);
82
83 /* Derived widget could use same property */
84 lv_obj_t * img = lv_image_create(obj);
85 prop.id = LV_PROPERTY_OBJ_PARENT;
86 prop.ptr = root;
87 TEST_ASSERT_TRUE(lv_obj_set_property(img, &prop) == LV_RESULT_OK);
88 TEST_ASSERT_EQUAL_PTR(root, lv_obj_get_parent(img));
89 TEST_ASSERT_EQUAL_PTR(root, lv_obj_get_property(img, LV_PROPERTY_OBJ_PARENT).ptr);

93 prop.num = 0x1234;
