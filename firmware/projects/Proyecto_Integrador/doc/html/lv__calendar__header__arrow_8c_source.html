108static void month_event_cb(lv_event_t * e)
109{
110 LV_ASSERT(e != NULL);
111 lv_obj_t * btn = lv_event_get_current_target(e);
112
113 lv_obj_t * header = lv_obj_get_parent(btn);
114 LV_ASSERT(header != NULL);
115 lv_obj_t * calendar = lv_obj_get_parent(header);
116 LV_ASSERT(calendar != NULL);
117
118 const lv_calendar_date_t * d;
119 d = lv_calendar_get_showed_date(calendar);
120 lv_calendar_date_t newd = *d;
121
122 LV_ASSERT_FORMAT_MSG(newd.month >= 1 && newd.month <= 12,
123 "Invalid month: %d (expected 1-12)", newd.month);
124
125 /*The last child is the right button*/
126 if(lv_obj_get_child(header, 0) == btn) {
127 if(newd.month == 1) {
128 newd.month = 12;
129 newd.year --;
130 }
131 else {
132 newd.month --;
133 }
134 }
135 else {
136 if(newd.month == 12) {
137 newd.month = 1;
138 newd.year ++;
139 }
140 else {
141 newd.month ++;
142 }
143 }
144
145 lv_calendar_set_month_shown(calendar, newd.year, newd.month);
146
147 lv_obj_t * label = lv_obj_get_child(header, 1);
148 lv_label_set_text_fmt(label, "%d %s", newd.year, month_names_def[newd.month - 1]);
149}

163 lv_obj_t * label = lv_obj_get_child(header, 1);
164 lv_label_set_text_fmt(label, "%d %s", date->year, month_names_def[date->month - 1]);
165}
