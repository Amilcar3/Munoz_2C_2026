66void lv_draw_layer(lv_layer_t * layer, const lv_draw_image_dsc_t * dsc, const lv_area_t * coords)
67{
68 LV_CHECK_ARG(layer != NULL, return);
69 LV_CHECK_ARG(dsc != NULL, return);
70 LV_CHECK_ARG(coords != NULL, return);
71
72 if(dsc->scale_x <= 0 || dsc->scale_y <= 0) {
73 /* NOT draw if scale is negative or zero */
74 return;
75 }
76
77 LV_PROFILER_DRAW_BEGIN;
78
79 lv_draw_task_t * t = lv_draw_add_task(layer, coords, LV_DRAW_TASK_TYPE_LAYER);
80 lv_draw_image_dsc_t * new_image_dsc = t->draw_dsc;
81 lv_memcpy(new_image_dsc, dsc, sizeof(*dsc));
82 t->state = LV_DRAW_TASK_STATE_BLOCKED;
83
84 lv_image_buf_get_transformed_area(&t->_real_area, lv_area_get_width(coords), lv_area_get_height(coords),
85 dsc->rotation, dsc->scale_x, dsc->scale_y, &dsc->pivot);
86 lv_area_move(&t->_real_area, coords->x1, coords->y1);
87
88 /*If the image_area is not set assume that it's the same as the rendering area */
89 if(new_image_dsc->image_area.x2 == LV_COORD_MIN) {
90 new_image_dsc->image_area = *coords;
91 }
92
93 lv_layer_t * layer_to_draw = (lv_layer_t *)dsc->src;
94 layer_to_draw->all_tasks_added = true;
95
96 lv_draw_finalize_task_creation(layer, t);
97
98 LV_PROFILER_DRAW_END;
99}

101void lv_draw_image(lv_layer_t * layer, const lv_draw_image_dsc_t * dsc, const lv_area_t * coords)
102{
103 LV_CHECK_ARG(layer != NULL, return);
104 LV_CHECK_ARG(dsc != NULL, return);
105 LV_CHECK_ARG(coords != NULL, return);
106
107 if(dsc->src == NULL) {
108 LV_LOG_WARN("Image draw: src is NULL");
109 return;
110 }
111 if(dsc->opa <= LV_OPA_MIN) return;
112
113 if(dsc->scale_x <= 0 || dsc->scale_y <= 0) {
114 /* NOT draw if scale is negative or zero */
115 return;
116 }
117
118 LV_PROFILER_DRAW_BEGIN;
119
120 if(dsc->base.drop_shadow_opa) {
121 lv_layer_t * ds_layer = lv_draw_layer_create_drop_shadow(layer, &dsc->base, coords);
122 LV_ASSERT_NULL(ds_layer);
123 lv_draw_image_dsc_t ds_dsc = *dsc;
124 ds_dsc.base.drop_shadow_opa = 0; /*Disable drop shadow so rendering below will render plain image*/
125 lv_draw_image(ds_layer, &ds_dsc, coords);
126 lv_draw_layer_finish_drop_shadow(ds_layer, &dsc->base);
127 }
128
129 lv_draw_image_dsc_t new_image_dsc;
130 lv_memcpy(&new_image_dsc, dsc, sizeof(*dsc));
131 lv_result_t res = lv_image_decoder_get_info(new_image_dsc.src, &new_image_dsc.header);
132 if(res != LV_RESULT_OK) {
133 LV_LOG_WARN("Couldn't get info about the image");
134 LV_PROFILER_DRAW_END;
135 return;
136 }
137
138 /*If the image_area is not set assume that it's the same as the rendering area */
139 if(new_image_dsc.image_area.x2 == LV_COORD_MIN) {
140 new_image_dsc.image_area = *coords;
141 }
142
143 /*Typical case, draw the image as bitmap*/
144 if(!(new_image_dsc.header.flags & LV_IMAGE_FLAGS_CUSTOM_DRAW)) {
145 lv_draw_task_t * t = lv_draw_add_task(layer, coords, LV_DRAW_TASK_TYPE_IMAGE);
146 lv_memcpy(t->draw_dsc, &new_image_dsc, sizeof(lv_draw_image_dsc_t));
147
148 lv_image_buf_get_transformed_area(&t->_real_area, lv_area_get_width(coords), lv_area_get_height(coords),
149 dsc->rotation, dsc->scale_x, dsc->scale_y, &dsc->pivot);
150 lv_area_move(&t->_real_area, coords->x1, coords->y1);
151
152 lv_draw_finalize_task_creation(layer, t);
153 }
154 /*Use a custom draw callback*/
155 else {
156
157 lv_image_decoder_dsc_t decoder_dsc;
158 res = lv_image_decoder_open(&decoder_dsc, new_image_dsc.src, NULL);
159 if(res != LV_RESULT_OK) {
160 LV_LOG_ERROR("Failed to open image");
161 LV_PROFILER_DRAW_END;
162 return;
163 }
164
165 if(decoder_dsc.decoder && decoder_dsc.decoder->custom_draw_cb) {
166 lv_area_t draw_area = layer->buf_area;
167 lv_area_t coords_area = *coords;
168
169 lv_area_t obj_area = dsc->base.obj->coords;
170 if(layer->parent) { /* child layer */
171 if(lv_area_intersect(&coords_area, &coords_area, &obj_area)) {
172 int32_t xpos = coords->x1 - draw_area.x1;
173 int32_t ypos = coords->y1 - draw_area.y1;
174
175 lv_area_move(&coords_area, -(coords->x1 - xpos), -(coords->y1 - ypos));
176 layer->_clip_area = coords_area;
177 decoder_dsc.decoder->custom_draw_cb(layer, &decoder_dsc, &coords_area, &new_image_dsc, &coords_area);
178 }
179 }
180 else {
181 lv_area_t clip_area = draw_area;
182 if(lv_area_intersect(&clip_area, &clip_area, &coords_area)) {
183
184 lv_image_buf_get_transformed_area(&coords_area, lv_area_get_width(coords), lv_area_get_height(coords),
185 dsc->rotation, dsc->scale_x, dsc->scale_y, &dsc->pivot);
186 lv_area_move(&coords_area, coords->x1, coords->y1);
187
188 lv_image_buf_get_transformed_area(&clip_area, lv_area_get_width(coords), lv_area_get_height(coords),
189 dsc->rotation, dsc->scale_x, dsc->scale_y, &dsc->pivot);
190 lv_area_move(&clip_area, coords->x1, coords->y1);
191
192 if(lv_area_intersect(&clip_area, &clip_area, &obj_area)) {
193 decoder_dsc.decoder->custom_draw_cb(layer, &decoder_dsc, &coords_area, &new_image_dsc, &clip_area);
194 }
195 }
196 }
197
198 }
199
200 lv_image_decoder_close(&decoder_dsc);
201 }
202
203 LV_PROFILER_DRAW_END;
204}

206lv_image_src_t lv_image_src_get_type(const void * src)
207{
208 if(src == NULL) {
209 return LV_IMAGE_SRC_UNKNOWN;
210 }
211
212 const uint8_t first_byte = *(const uint8_t *)src;
213
214 /*The first byte shows the type of the image source*/
215 if(first_byte >= 0x20 && first_byte <= 0x7F) {
216 return LV_IMAGE_SRC_FILE; /*If it's an ASCII character then it's file name*/
217 }
218
219 if(first_byte >= 0x80) {
220 return LV_IMAGE_SRC_SYMBOL; /*Symbols begins after 0x7F*/
221 }
222
227 if(first_byte == LV_IMAGE_HEADER_MAGIC || first_byte == LV_IMAGE_HEADER_LEGACY) {
228 return LV_IMAGE_SRC_VARIABLE;
229 }
230
231 LV_LOG_ERROR("image src %p invalid magic: 0x%02X", src, first_byte);
232 return LV_IMAGE_SRC_UNKNOWN;
233}

235void lv_draw_image_normal_helper(lv_draw_task_t * t, const lv_draw_image_dsc_t * draw_dsc,
236 const lv_area_t * coords, lv_draw_image_core_cb draw_core_cb,
237 const lv_image_decoder_args_t * decoder_args)
238{
239 if(draw_core_cb == NULL) {
240 LV_LOG_WARN("draw_core_cb is NULL");
241 return;
242 }
243
244 lv_area_t draw_area;
245 draw_area = *coords;
246 if(draw_dsc->rotation || draw_dsc->scale_x != LV_SCALE_NONE || draw_dsc->scale_y != LV_SCALE_NONE) {
247 int32_t w = lv_area_get_width(coords);
248 int32_t h = lv_area_get_height(coords);
249
250 lv_image_buf_get_transformed_area(&draw_area, w, h, draw_dsc->rotation, draw_dsc->scale_x, draw_dsc->scale_y,
251 &draw_dsc->pivot);
252
253 draw_area.x1 += coords->x1;
254 draw_area.y1 += coords->y1;
255 draw_area.x2 += coords->x1;
256 draw_area.y2 += coords->y1;
257 }
258
259 lv_area_t clipped_img_area;
260 if(!lv_area_intersect(&clipped_img_area, &draw_area, &t->clip_area)) {
261 return;
262 }
263
264 lv_image_decoder_dsc_t decoder_dsc;
265 lv_result_t res = lv_image_decoder_open(&decoder_dsc, draw_dsc->src, decoder_args);
266 if(res != LV_RESULT_OK) {
267 LV_LOG_ERROR("Failed to open image");
268 return;
269 }
270
271 img_decode_and_draw(t, draw_dsc, &decoder_dsc, NULL, coords, &clipped_img_area, draw_core_cb);
272
273 lv_image_decoder_close(&decoder_dsc);
274}

276void lv_draw_image_tiled_helper(lv_draw_task_t * t, const lv_draw_image_dsc_t * draw_dsc,
277 const lv_area_t * coords, lv_draw_image_core_cb draw_core_cb,
278 const lv_image_decoder_args_t * decoder_args)
279{
280 if(draw_core_cb == NULL) {
281 LV_LOG_WARN("draw_core_cb is NULL");
282 return;
283 }
284
285 lv_image_decoder_dsc_t decoder_dsc;
286 lv_result_t res = lv_image_decoder_open(&decoder_dsc, draw_dsc->src, decoder_args);
287 if(res != LV_RESULT_OK) {
288 LV_LOG_ERROR("Failed to open image");
289 return;
290 }
291
292 int32_t img_w = draw_dsc->header.w;
293 int32_t img_h = draw_dsc->header.h;
294
295 lv_area_t tile_area;
296 if(lv_area_get_width(&draw_dsc->image_area) >= 0) {
297 tile_area = draw_dsc->image_area;
298 }
299 else {
300 tile_area = *coords;
301 }
302 lv_area_set_width(&tile_area, img_w);
303 lv_area_set_height(&tile_area, img_h);
304
305 int32_t tile_x_start = tile_area.x1;
306
307 lv_area_t relative_decoded_area = {
308 .x1 = LV_COORD_MIN,
309 .y1 = LV_COORD_MIN,
310 .x2 = LV_COORD_MIN,
311 .y2 = LV_COORD_MIN,
312 };
313
314 while(tile_area.y1 <= coords->y2) {
315 while(tile_area.x1 <= coords->x2) {
316
317 lv_area_t clipped_img_area;
318 if(lv_area_intersect(&clipped_img_area, &tile_area, coords)) {
319 img_decode_and_draw(t, draw_dsc, &decoder_dsc, &relative_decoded_area, &tile_area, &clipped_img_area,
320 draw_core_cb);
321 }
322
323 tile_area.x1 += img_w;
324 tile_area.x2 += img_w;
325 }
326
327 tile_area.y1 += img_h;
328 tile_area.y2 += img_h;
329 tile_area.x1 = tile_x_start;
330 tile_area.x2 = tile_x_start + img_w - 1;
331 }
332
333 lv_image_decoder_close(&decoder_dsc);
334}

336void lv_image_buf_get_transformed_area(lv_area_t * res, int32_t w, int32_t h, int32_t angle,
337 uint16_t scale_x, uint16_t scale_y, const lv_point_t * pivot)
338{
339 if(angle == 0 && scale_x == LV_SCALE_NONE && scale_y == LV_SCALE_NONE) {
340 res->x1 = 0;
341 res->y1 = 0;
342 res->x2 = w - 1;
343 res->y2 = h - 1;
344 return;
345 }
346
347 lv_point_t p[4] = {
348 {0, 0},
349 {w - 1, 0},
350 {0, h - 1},
351 {w - 1, h - 1},
352 };
353 lv_point_transform(&p[0], angle, scale_x, scale_y, pivot, true);
354 lv_point_transform(&p[1], angle, scale_x, scale_y, pivot, true);
355 lv_point_transform(&p[2], angle, scale_x, scale_y, pivot, true);
356 lv_point_transform(&p[3], angle, scale_x, scale_y, pivot, true);
357 res->x1 = LV_MIN4(p[0].x, p[1].x, p[2].x, p[3].x);
358 res->x2 = LV_MAX4(p[0].x, p[1].x, p[2].x, p[3].x);
359 res->y1 = LV_MIN4(p[0].y, p[1].y, p[2].y, p[3].y);
360 res->y2 = LV_MAX4(p[0].y, p[1].y, p[2].y, p[3].y);
361}

367static void img_decode_and_draw(lv_draw_task_t * t, const lv_draw_image_dsc_t * draw_dsc,
368 lv_image_decoder_dsc_t * decoder_dsc, lv_area_t * relative_decoded_area,
369 const lv_area_t * img_area, const lv_area_t * clipped_img_area,
370 lv_draw_image_core_cb draw_core_cb)
371{
372 lv_draw_image_sup_t sup;
373 sup.alpha_color = draw_dsc->recolor;
374 sup.palette = decoder_dsc->palette;
375 sup.palette_size = decoder_dsc->palette_size;
376
377 /*The whole image is available, just draw it*/
378 if(decoder_dsc->decoded && (relative_decoded_area == NULL || relative_decoded_area->x1 == LV_COORD_MIN)) {
379 draw_core_cb(t, draw_dsc, decoder_dsc, &sup, img_area, clipped_img_area);
380 }
381 /*Draw in smaller pieces*/
382 else {
383 if(draw_dsc->rotation != 0 || draw_dsc->scale_x != LV_SCALE_NONE || draw_dsc->scale_y != LV_SCALE_NONE) {
384 LV_LOG_WARN("Partially decoded images cannot be transformed");
385 return;
386 }
387
388 lv_area_t relative_full_area_to_decode = *clipped_img_area;
389 lv_area_move(&relative_full_area_to_decode, -img_area->x1, -img_area->y1);
390 lv_area_t tmp;
391 if(relative_decoded_area == NULL) relative_decoded_area = &tmp;
392 relative_decoded_area->x1 = LV_COORD_MIN;
393 relative_decoded_area->y1 = LV_COORD_MIN;
394 relative_decoded_area->x2 = LV_COORD_MIN;
395 relative_decoded_area->y2 = LV_COORD_MIN;
396 lv_result_t res = LV_RESULT_OK;
397
398 while(res == LV_RESULT_OK) {
399 res = lv_image_decoder_get_area(decoder_dsc, &relative_full_area_to_decode, relative_decoded_area);
400
401 lv_area_t absolute_decoded_area = *relative_decoded_area;
402 lv_area_move(&absolute_decoded_area, img_area->x1, img_area->y1);
403 if(res == LV_RESULT_OK) {
404 /*Limit draw area to the current decoded area and draw the image*/
405 lv_area_t clipped_img_area_sub;
406 if(lv_area_intersect(&clipped_img_area_sub, clipped_img_area, &absolute_decoded_area)) {
407 draw_core_cb(t, draw_dsc, decoder_dsc, &sup,
408 &absolute_decoded_area, &clipped_img_area_sub);
409 }
410 }
411 }
412 }
