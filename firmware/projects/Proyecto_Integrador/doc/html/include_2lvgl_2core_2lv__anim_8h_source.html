29/*
30 * Macros used to set cubic-bezier anim parameter.
31 * Parameters come from https://easings.net/
32 *
33 * Usage:
34 *
35 * lv_anim_t a;
36 * lv_anim_init(&a);
37 * ...
38 * lv_anim_set_path_cb(&a, lv_anim_path_custom_bezier3);
39 * LV_ANIM_SET_EASE_IN_SINE(&a); //Set cubic-bezier anim parameter to easeInSine
40 * ...
41 * lv_anim_start(&a);
42 */
43
44#define _PARA(a, x1, y1, x2, y2) ((a)->parameter.bezier3 = \
45(lv_anim_bezier3_para_t) { \
46 LV_BEZIER_VAL_FLOAT(x1), LV_BEZIER_VAL_FLOAT(y1), \
47 LV_BEZIER_VAL_FLOAT(x2), LV_BEZIER_VAL_FLOAT(y2) } \
48 )
49
