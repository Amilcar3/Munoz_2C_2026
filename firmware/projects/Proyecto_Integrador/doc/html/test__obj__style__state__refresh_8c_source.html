26/* Build a 2-level subtree with a `LV_EVENT_STYLE_CHANGED` counter on every
27 * descendant (not on `parent`, which gets its own event on a state change).
28 * Render it once: state changes only refresh already-rendered objects, so
29 * without a render the toggles below would be skipped and the tests would pass
30 * for the wrong reason. */
31static lv_obj_t * build_rendered_subtree(void)
32{
33 lv_obj_t * parent = lv_obj_create(lv_screen_active());
34 lv_obj_set_size(parent, 200, 200);
35
36 for(uint32_t i = 0; i < 3; i++) {
37 lv_obj_t * child = lv_obj_create(parent);
38 lv_obj_add_event_cb(child, count_style_changed_cb, LV_EVENT_STYLE_CHANGED, NULL);
39 for(uint32_t j = 0; j < 2; j++) {
40 lv_obj_t * grandchild = lv_obj_create(child);
41 lv_obj_add_event_cb(grandchild, count_style_changed_cb, LV_EVENT_STYLE_CHANGED, NULL);
42 }
43 }
44
45 lv_refr_now(NULL);
46 return parent;
47}

49/* A redraw-only, scrollbar-only state change must not reach any descendant. */
50void test_scrollbar_redraw_state_does_not_cascade_to_children(void)
51{
52 lv_obj_t * parent = build_rendered_subtree();
53
54 lv_style_t scrollbar_scrolled;
55 lv_style_init(&scrollbar_scrolled);
56 lv_style_set_bg_opa(&scrollbar_scrolled, LV_OPA_COVER); /* redraw-only property */
57 lv_obj_add_style(parent, &scrollbar_scrolled, LV_PART_SCROLLBAR | LV_STATE_SCROLLED);
58
59 /* Reset after the subtree setup (which emits its own STYLE_CHANGED events)
60 * so the count reflects only the state change below. */
61 child_style_changed_count = 0;
62 lv_obj_add_state(parent, LV_STATE_SCROLLED);
63 lv_obj_remove_state(parent, LV_STATE_SCROLLED);
64
65 TEST_ASSERT_EQUAL(0, child_style_changed_count);
66
67 lv_obj_remove_style(parent, &scrollbar_scrolled, LV_PART_SCROLLBAR | LV_STATE_SCROLLED);
68 lv_style_reset(&scrollbar_scrolled);
69}
