47/* GStreamer decodes on the real clock in its own threads, so unlike
48 * lv_test_wait() these have to burn real time while LVGL's simulated tick and
49 * its timers keep running. */
50static void pump(uint32_t ms)
51{
52 while(ms--) {
53 usleep(1000);
54 lv_tick_inc(1);
55 lv_timer_handler();
56 }
57 lv_refr_now(NULL);
58}

60static bool pump_until(const uint32_t * counter, uint32_t target, uint32_t timeout_ms)
61{
62 for(uint32_t i = 0; i < timeout_ms; i++) {
63 if(*counter >= target) break;
64 usleep(1000);
65 lv_tick_inc(1);
66 lv_timer_handler();
67 }
68 lv_refr_now(NULL);
69 return *counter >= target;
70}

72static void assert_frame_is_video_color(const lv_image_dsc_t * frame)
73{
74 const uint8_t * px = frame->data + frame->header.stride * (TEST_VIDEO_SIZE / 2);
75
76 switch(frame->header.cf) {
77 case LV_COLOR_FORMAT_ARGB8888:
78 case LV_COLOR_FORMAT_XRGB8888:
79 px += (TEST_VIDEO_SIZE / 2) * 4;
80 TEST_ASSERT_INT_WITHIN(TEST_VIDEO_COLOR_TOLERANCE, TEST_VIDEO_B, px[0]);
81 TEST_ASSERT_INT_WITHIN(TEST_VIDEO_COLOR_TOLERANCE, TEST_VIDEO_G, px[1]);
82 TEST_ASSERT_INT_WITHIN(TEST_VIDEO_COLOR_TOLERANCE, TEST_VIDEO_R, px[2]);
83 break;
84 case LV_COLOR_FORMAT_RGB565: {
85 px += (TEST_VIDEO_SIZE / 2) * 2;
86 uint16_t color;
87 lv_memcpy(&color, px, sizeof(color));
88 TEST_ASSERT_INT_WITHIN(1, TEST_VIDEO_R >> 3, (color >> 11) & 0x1f);
89 TEST_ASSERT_INT_WITHIN(1, TEST_VIDEO_G >> 2, (color >> 5) & 0x3f);
90 TEST_ASSERT_INT_WITHIN(1, TEST_VIDEO_B >> 3, color & 0x1f);
91 break;
92 }
93 default:
94 TEST_FAIL_MESSAGE("frame decoded into an unexpected color format");
95 }
96}

208 lv_gstreamer_play(player);

236void test_gstreamer_queue_stays_bounded(void)
237{
238 lv_obj_t * player = create_player();
239 TEST_ASSERT_EQUAL(LV_RESULT_OK, lv_gstreamer_set_src(player, LV_GSTREAMER_FACTORY_TEST_VIDEO, NULL, NULL));
240
241 lv_gstreamer_play(player);
242 TEST_ASSERT_TRUE(pump_until(&event_cnt[LV_GSTREAMER_STREAM_STATE_START], 1, 3000));
243
244 /* Stall the UI thread. The pipeline keeps decoding on its own thread, and nothing
245 * takes the frames it produces out of the queue */
246 usleep(1000000);
247
248 const lv_gstreamer_t * streamer = (const lv_gstreamer_t *)player;
249 TEST_ASSERT_LESS_OR_EQUAL(LV_GSTREAMER_MAX_QUEUED_FRAMES, g_async_queue_length(streamer->frame_queue));
250
251 /* The widget still displays what the pipeline produced after the stall */
252 event_cnt[LV_GSTREAMER_STREAM_STATE_START] = 0;
253 pump(100);
254 TEST_ASSERT_EQUAL(LV_GSTREAMER_STATE_PLAYING, lv_gstreamer_get_state(player));
255 TEST_ASSERT_NOT_NULL(lv_image_get_src(player));
256}
