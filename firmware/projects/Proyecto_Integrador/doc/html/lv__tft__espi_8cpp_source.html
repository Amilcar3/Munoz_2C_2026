43lv_display_t * lv_tft_espi_create(uint32_t hor_res, uint32_t ver_res, void * buf, uint32_t buf_size_bytes)
44{
45 LV_CHECK_ARG(buf != NULL, return NULL);
46 lv_tft_espi_t * dsc = (lv_tft_espi_t *)lv_malloc_zeroed(sizeof(lv_tft_espi_t));
47 LV_ASSERT_MALLOC(dsc);
48 if(dsc == NULL) return NULL;
49
50 lv_display_t * disp = lv_display_create(hor_res, ver_res);
51 if(disp == NULL) {
52 lv_free(dsc);
53 return NULL;
54 }
55
56 dsc->tft = new TFT_eSPI(hor_res, ver_res);
57 dsc->tft->begin(); /* TFT init */
58 dsc->tft->setRotation(0);
59 lv_display_set_driver_data(disp, (void *)dsc);
60 lv_display_set_flush_cb(disp, flush_cb);
61 lv_display_add_event_cb(disp, resolution_changed_event_cb, LV_EVENT_RESOLUTION_CHANGED, NULL);
62 lv_display_set_buffers(disp, (void *)buf, NULL, buf_size_bytes, LV_DISPLAY_RENDER_MODE_PARTIAL);
63 return disp;
64}

70static void flush_cb(lv_display_t * disp, const lv_area_t * area, uint8_t * px_map)
71{
72 lv_tft_espi_t * dsc = (lv_tft_espi_t *)lv_display_get_driver_data(disp);
73
74 uint32_t w = (area->x2 - area->x1 + 1);
75 uint32_t h = (area->y2 - area->y1 + 1);
76
77 dsc->tft->startWrite();
78 dsc->tft->setAddrWindow(area->x1, area->y1, w, h);
79 dsc->tft->pushColors((uint16_t *)px_map, w * h, true);
80 dsc->tft->endWrite();
81
82 lv_display_flush_ready(disp);
83
84}
