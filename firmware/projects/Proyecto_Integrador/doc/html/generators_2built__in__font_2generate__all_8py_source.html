122def main():
123 for tool in ("lv_font_conv", "astyle"):
124 if shutil.which(tool) is None:
125 sys.exit(f"error: '{tool}' is required but was not found in PATH")
126
127 fonts = generate_all()
128
129 # The fonts end up in src/font/, so point lv_font_conv's include of the
130 # LVGL header at the repository root instead of an installed lvgl/ prefix
131 for font in fonts:
132 font.write_text(
133 font.read_text().replace('#include "lvgl/lvgl.h"', '#include "../../lvgl.h"')
134 )
135
136 print()
137 subprocess.run(
138 [
139 "astyle",
140 "--ignore-exclude-errors",
141 f"--options={CODE_FORMAT_CFG}",
142 *[str(font) for font in fonts],
143 ],
144 check=True,
145 )
146
147 for font in fonts:
148 shutil.move(font, OUTPUT_DIR / font.name)
