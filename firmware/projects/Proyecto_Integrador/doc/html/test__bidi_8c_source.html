19/*
20 * Helper: call lv_bidi_process_paragraph with a guarded output buffer.
21 * Allocates (len + 1 + 2*guard) bytes, fills guard zones with 0xAA,
22 * runs the function, then checks guard zones are intact.
23 * Returns true if no overflow detected.
24 */
25#define GUARD_SIZE 16
26#define GUARD_BYTE 0xAA
27
28static bool bidi_process_check_overflow(const char * str_in, uint32_t len, lv_base_dir_t base_dir)
29{
30 /* Allocate buffer with guard zones before and after */
31 uint32_t total = GUARD_SIZE + len + 1 + GUARD_SIZE;
32 uint8_t * buf = lv_malloc(total);
33 TEST_ASSERT_NOT_NULL(buf);
34
35 /* Fill entire buffer with guard byte */
36 memset(buf, GUARD_BYTE, total);
37
38 /* str_out points to the usable region */
39 char * str_out = (char *)(buf + GUARD_SIZE);
40
41 /* Clear the usable region */
42 memset(str_out, 0, len + 1);
43
44 lv_bidi_process_paragraph(str_in, str_out, len, base_dir, NULL, 0);
45
46 /* Check pre-guard */
47 bool ok = true;
48 for(uint32_t i = 0; i < GUARD_SIZE; i++) {
49 if(buf[i] != GUARD_BYTE) {
50 ok = false;
51 break;
52 }
53 }
54
55 /* Check post-guard */
56 for(uint32_t i = 0; i < GUARD_SIZE; i++) {
57 if(buf[GUARD_SIZE + len + 1 + i] != GUARD_BYTE) {
58 ok = false;
59 break;
60 }
61 }
62
63 lv_free(buf);
64 return ok;
65}
