158void test_spinbox_event_key(void)
159{
160 /* Spinbox should increment it's value by one after receiving the LV_KEY_UP event */
161 lv_spinbox_set_value(spinbox_events, 0);
162 uint32_t key = LV_KEY_UP;
163 lv_obj_send_event(spinbox_events, LV_EVENT_KEY, (void *) &key);
164
165 TEST_ASSERT_EQUAL(1, lv_spinbox_get_value(spinbox_events));
166
167 /* Spinbox should decrement it's value by one after receiving the LV_KEY_DOWN event */
168 key = LV_KEY_DOWN;
169 lv_obj_send_event(spinbox_events, LV_EVENT_KEY, (void *) &key);
170
171 TEST_ASSERT_EQUAL(0, lv_spinbox_get_value(spinbox_events));
172
173 /* Spinbox should multiply it's step vale by 10 after receiving the LV_KEY_LEFT event */
174 int32_t step = lv_spinbox_get_step(spinbox_events);
175 key = LV_KEY_LEFT;
176 lv_obj_send_event(spinbox_events, LV_EVENT_KEY, (void *) &key);
177
178 TEST_ASSERT_EQUAL(step * 10, lv_spinbox_get_step(spinbox_events));
179
180 /* Spinbox should divide it's step vale by 10 after receiving the LV_KEY_RIGHT event */
181 step = lv_spinbox_get_step(spinbox_events);
182 key = LV_KEY_RIGHT;
183 lv_obj_send_event(spinbox_events, LV_EVENT_KEY, (void *) &key);
184
185 TEST_ASSERT_EQUAL(step / 10, lv_spinbox_get_step(spinbox_events));
186}
