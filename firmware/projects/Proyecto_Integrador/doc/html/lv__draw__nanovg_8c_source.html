103void lv_draw_nanovg_init(void)
104{
105 static bool initialized = false;
106 if(initialized) return;
107 initialized = true;
108
109 lv_draw_nanovg_unit_t * unit = lv_draw_create_unit(sizeof(lv_draw_nanovg_unit_t));
110 unit->base_unit.dispatch_cb = draw_dispatch;
111 unit->base_unit.evaluate_cb = draw_evaluate;
112 unit->base_unit.delete_cb = draw_delete;
113 unit->base_unit.event_cb = draw_event_cb;
114 unit->base_unit.name = "NANOVG";
115
116 unit->vg = NVG_CTX_CREATE(0);
117 LV_ASSERT_MSG(unit->vg != NULL, "NanoVG init failed");
118
119 lv_nanovg_utils_init(unit);
120 lv_nanovg_image_cache_init(unit);
121 lv_nanovg_fbo_cache_init(unit);
122 lv_draw_nanovg_label_init(unit);
123 lv_draw_nanovg_blur_init(unit);
124
125 lv_draw_buf_init_with_default_handlers(&gpu_draw_buf_handlers);
126 gpu_draw_buf_handlers.buf_clear_cb = gpu_buf_clear_cb;
127
128 g_unit = unit;

146static void draw_execute(lv_draw_nanovg_unit_t * u, lv_draw_task_t * t)
147{
148 /* remember draw unit for access to unit's context */
149 t->draw_unit = (lv_draw_unit_t *)u;
150 lv_layer_t * layer = t->target_layer;
151
152 lv_matrix_t global_matrix;
153 lv_matrix_identity(&global_matrix);
154 if(layer->buf_area.x1 || layer->buf_area.y1) {
155 lv_matrix_translate(&global_matrix, -layer->buf_area.x1, -layer->buf_area.y1);
156 }
157
158#if LV_DRAW_TRANSFORM_USE_MATRIX
159 lv_matrix_t layer_matrix = t->matrix;
160 lv_matrix_multiply(&global_matrix, &layer_matrix);
161#endif
162
163 /* NanoVG will output premultiplied image, set the flag correspondingly. */
164 if(layer->draw_buf) {
165 lv_draw_buf_set_flag(layer->draw_buf, LV_IMAGE_FLAGS_PREMULTIPLIED);
166 }
167
168 nvgReset(u->vg);
169 lv_nanovg_transform(u->vg, &global_matrix);
170
171 lv_nanovg_set_clip_area(u->vg, &t->clip_area);
172
173 switch(t->type) {
174 case LV_DRAW_TASK_TYPE_FILL:
175 lv_draw_nanovg_fill(t, t->draw_dsc, &t->area);
176 break;
177
178 case LV_DRAW_TASK_TYPE_BORDER:
179 lv_draw_nanovg_border(t, t->draw_dsc, &t->area);
180 break;
181
182 case LV_DRAW_TASK_TYPE_BOX_SHADOW:
183 lv_draw_nanovg_box_shadow(t, t->draw_dsc, &t->area);
184 break;
185
186 case LV_DRAW_TASK_TYPE_LETTER:
187 lv_draw_nanovg_letter(t, t->draw_dsc, &t->area);
188 break;
189
190 case LV_DRAW_TASK_TYPE_LABEL:
191 lv_draw_nanovg_label(t, t->draw_dsc, &t->area);
192 break;
193
194 case LV_DRAW_TASK_TYPE_IMAGE:
195 lv_draw_nanovg_image(t, t->draw_dsc, &t->area, -1);
196 break;
197
198 case LV_DRAW_TASK_TYPE_LAYER:
199 lv_draw_nanovg_layer(t, t->draw_dsc, &t->area);
200 break;
201
202 case LV_DRAW_TASK_TYPE_LINE:
203 lv_draw_line_iterate(t, t->draw_dsc, lv_draw_nanovg_line);
204 break;
205
206 case LV_DRAW_TASK_TYPE_ARC:
207 lv_draw_nanovg_arc(t, t->draw_dsc, &t->area);
208 break;
209
210 case LV_DRAW_TASK_TYPE_TRIANGLE:
211 lv_draw_nanovg_triangle(t, t->draw_dsc);
212 break;
213
214 case LV_DRAW_TASK_TYPE_MASK_RECTANGLE:
215 lv_draw_nanovg_mask_rect(t, t->draw_dsc);
216 break;
217
218#if LV_USE_VECTOR_GRAPHIC
219 case LV_DRAW_TASK_TYPE_VECTOR:
220 lv_draw_nanovg_vector(t, t->draw_dsc);
221 break;
222#endif
223
224#if LV_USE_3DTEXTURE
225 case LV_DRAW_TASK_TYPE_3D:
226 lv_draw_nanovg_3d(t, t->draw_dsc, &t->area);
227 break;
228#endif
229
230 case LV_DRAW_TASK_TYPE_BLUR:
231 lv_draw_nanovg_blur(t, t->draw_dsc, &t->area);
232 break;
233
234 default:
235 LV_LOG_ERROR("unknown draw task type: %d", t->type);
236 break;
237 }

269static void gpu_buf_clear_cb(lv_draw_buf_t * draw_buf, const lv_area_t * a, lv_layer_t * layer)
270{
271 LV_UNUSED(draw_buf);
272
273 if(g_unit == NULL || layer == NULL) return;
274
275 LV_PROFILER_DRAW_BEGIN;
276
277 lv_nanovg_end_frame(g_unit);
278
279 bind_layer_framebuffer(layer);
280 g_unit->current_layer = NULL;
281
282 const int32_t buf_w = lv_area_get_width(&layer->buf_area);
283 const int32_t buf_h = lv_area_get_height(&layer->buf_area);
284
285 lv_area_t buf_bounds;
286 lv_area_set(&buf_bounds, 0, 0, buf_w - 1, buf_h - 1);
287
288 /* `a` is relative to the draw buffer, which matches the framebuffer origin. */
289 lv_area_t clear_area;
290 if(a == NULL) {
291 clear_area = buf_bounds;
292 }
293 else if(!lv_area_intersect(&clear_area, a, &buf_bounds)) {
294 LV_PROFILER_DRAW_END;
295 return;
296 }
297
298 const int32_t w = lv_area_get_width(&clear_area);
299 const int32_t h = lv_area_get_height(&clear_area);
300 if(w <= 0 || h <= 0) {
301 LV_PROFILER_DRAW_END;
302 return;
303 }
304
305 GLboolean prev_scissor_test = glIsEnabled(GL_SCISSOR_TEST);
306 GLint prev_scissor_box[4] = {0};
307 GLfloat prev_clear_color[4] = {0};
308 glGetIntegerv(GL_SCISSOR_BOX, prev_scissor_box);
309 glGetFloatv(GL_COLOR_CLEAR_VALUE, prev_clear_color);
310
311 LV_PROFILER_DRAW_BEGIN_TAG("glClear");
312 glEnable(GL_SCISSOR_TEST);
313 /* GL puts the origin at the bottom left */
314 glScissor(clear_area.x1, buf_h - clear_area.y1 - h, w, h);
315 glClearColor(0, 0, 0, 0);
316 glClear(GL_COLOR_BUFFER_BIT | GL_STENCIL_BUFFER_BIT);
317 LV_PROFILER_DRAW_END_TAG("glClear");
318
319 if(!prev_scissor_test) glDisable(GL_SCISSOR_TEST);
320 glScissor(prev_scissor_box[0], prev_scissor_box[1], prev_scissor_box[2], prev_scissor_box[3]);
321 glClearColor(prev_clear_color[0], prev_clear_color[1], prev_clear_color[2], prev_clear_color[3]);
322
323 LV_PROFILER_DRAW_END;
324}

326static void on_layer_readback(lv_draw_nanovg_unit_t * u, lv_layer_t * layer)
327{
328 LV_PROFILER_DRAW_BEGIN;
329 LV_UNUSED(u);
330 LV_ASSERT_NULL(layer);
331
332 lv_cache_entry_t * entry = layer->user_data;
333
334 if(!entry) {
335 LV_LOG_WARN("No entry available for layer: %p", (void *)layer);
336 LV_PROFILER_DRAW_END;
337 return;
338 }
339
340 if(!layer->draw_buf) {
341 LV_LOG_WARN("No draw buffer available for layer: %p", (void *)layer);
342 LV_PROFILER_DRAW_END;
343 return;
344 }
345
346 struct NVGLUframebuffer * fb = lv_nanovg_fbo_cache_entry_to_fb(entry);
347 if(!fb) {
348 LV_LOG_ERROR("No framebuffer available for layer: %p", (void *)layer);
349 LV_PROFILER_DRAW_END;
350 return;
351 }
352
353 /* Bind the FBO for reading */
354 nvgluBindFramebuffer(fb);
355
356 int32_t w = lv_area_get_width(&layer->buf_area);
357 int32_t h = lv_area_get_height(&layer->buf_area);
358 lv_draw_buf_t * draw_buf = layer->draw_buf;
359
360 /* Read pixels from FBO */
361 GLenum format;
362 GLenum type;
363
364 /* OpenGL reads bottom-to-top, but LVGL expects top-to-bottom */
365 switch(draw_buf->header.cf) {
366 case LV_COLOR_FORMAT_ARGB8888:
367 case LV_COLOR_FORMAT_XRGB8888:
368 case LV_COLOR_FORMAT_ARGB8888_PREMULTIPLIED:
369 format = GL_BGRA;
370 type = GL_UNSIGNED_BYTE;
371 break;
372
373 case LV_COLOR_FORMAT_RGB888:
374 format = GL_RGB;
375 type = GL_UNSIGNED_BYTE;
376 break;
377
378 case LV_COLOR_FORMAT_RGB565:
379 format = GL_RGB;
380 type = GL_UNSIGNED_SHORT_5_6_5;
381 break;
382
383 default:
384 LV_LOG_WARN("Unsupported color format: %d", draw_buf->header.cf);
385 LV_PROFILER_DRAW_END;
386 return;
387 }
388
389 for(int32_t y = 0; y < h; y++) {
390 /* Reverse Y coordinate */
391 void * row = lv_draw_buf_goto_xy(draw_buf, 0, h - 1 - y);
392 LV_PROFILER_DRAW_BEGIN_TAG("glReadPixels");
393 glReadPixels(0, y, w, 1, format, type, row);
394 LV_PROFILER_DRAW_END_TAG("glReadPixels");
395
396 if(draw_buf->header.cf == LV_COLOR_FORMAT_RGB888) {
397 /* Swizzle RGB -> BGR */
398 lv_color_t * px = row;
399 for(int32_t x = 0; x < w; x++) {
400 uint8_t r = px->blue;
401 px->blue = px->red;
402 px->red = r;
403 px++;
404 }
405 }
406 }
407
408 /* Bind back to default framebuffer */
409 nvgluBindFramebuffer(NULL);
410
411 /* Mark draw_buf as modified */
412 lv_draw_buf_flush_cache(draw_buf, NULL);
413
414 LV_PROFILER_DRAW_END;
415}

417static int32_t draw_dispatch(lv_draw_unit_t * draw_unit, lv_layer_t * layer)
418{
419 lv_draw_nanovg_unit_t * u = (lv_draw_nanovg_unit_t *)draw_unit;
420
421 lv_draw_task_t * t = lv_draw_get_available_task(layer, NULL, NANOVG_DRAW_UNIT_ID);
422 if(!t || t->preferred_draw_unit_id != NANOVG_DRAW_UNIT_ID) {
423 lv_nanovg_end_frame(u);
424 return LV_DRAW_UNIT_IDLE;
425 }
426
427 if(u->current_layer != layer) {
428 /* Flush any draws still queued for the previous layer before
429 * rebinding to the new layer's FBO. Otherwise those queued draws
430 * get rerouted to the new FBO when nvgEndFrame is eventually
431 * called (and the previous layer ends up missing them). */
432 lv_nanovg_end_frame(u);
433 on_layer_changed(layer);
434 u->current_layer = layer;
435 }
436
437 if(!u->is_started) {
438 const int32_t buf_w = lv_area_get_width(&layer->buf_area);
439 const int32_t buf_h = lv_area_get_height(&layer->buf_area);
440
441 glViewport(0, 0, buf_w, buf_h);
442 LV_PROFILER_DRAW_BEGIN_TAG("nvgBeginFrame");
443 nvgBeginFrame(u->vg, buf_w, buf_h, 1.0f);
444 LV_PROFILER_DRAW_END_TAG("nvgBeginFrame");
445 u->is_started = true;
446 }
447
448 t->state = LV_DRAW_TASK_STATE_IN_PROGRESS;
449
450 draw_execute(u, t);
451
452 t->state = LV_DRAW_TASK_STATE_FINISHED;
453
454 /*The draw unit is free now. Request a new dispatching as it can get a new task*/
455 lv_draw_dispatch_request();
456
457 return 1;
458}

460static int32_t draw_evaluate(lv_draw_unit_t * draw_unit, lv_draw_task_t * task)
461{
462 lv_draw_nanovg_unit_t * u = (lv_draw_nanovg_unit_t *)draw_unit;
463
464 switch(task->type) {
465 case LV_DRAW_TASK_TYPE_FILL:
466 case LV_DRAW_TASK_TYPE_BORDER:
467 case LV_DRAW_TASK_TYPE_BOX_SHADOW:
468 case LV_DRAW_TASK_TYPE_LETTER:
469 case LV_DRAW_TASK_TYPE_LABEL:
470 case LV_DRAW_TASK_TYPE_IMAGE:
471 case LV_DRAW_TASK_TYPE_LAYER:
472 case LV_DRAW_TASK_TYPE_LINE:
473 case LV_DRAW_TASK_TYPE_ARC:
474 case LV_DRAW_TASK_TYPE_TRIANGLE:
475 case LV_DRAW_TASK_TYPE_MASK_RECTANGLE:
476#if LV_USE_VECTOR_GRAPHIC
477 case LV_DRAW_TASK_TYPE_VECTOR:
478#endif
479#if LV_USE_3DTEXTURE
480 case LV_DRAW_TASK_TYPE_3D:
481#endif
482 break;
483
484 case LV_DRAW_TASK_TYPE_BLUR:
485 /* Only accept blur if FBO support is available (blur_state was created) */
486 if(u->blur_state == NULL) return 0;
487 break;
488
489 default:
490 /*The draw unit is not able to draw this task. */
491 return 0;
492 }
493
494 if(task->preference_score > 80) {
495 /* The draw unit is able to draw this task. */
496 task->preference_score = 80;
497 task->preferred_draw_unit_id = NANOVG_DRAW_UNIT_ID;
498 }
499
500 return 1;
501}

516static void draw_event_cb(lv_event_t * e)
517{
518 lv_draw_nanovg_unit_t * u = lv_event_get_current_target(e);
519 lv_layer_t * layer = lv_event_get_param(e);
520
521 switch(lv_event_get_code(e)) {
522 case LV_EVENT_CANCEL:
523 LV_PROFILER_DRAW_BEGIN_TAG("nvgCancelFrame");
524 nvgCancelFrame(u->vg);
525 LV_PROFILER_DRAW_END_TAG("nvgCancelFrame");
526 lv_nanovg_clean_up(u);
527 break;
528 case LV_EVENT_CHILD_CREATED: {
529 /* The internal rendering uses RGBA format, which is switched to LVGL BGRA format during readback. */
530 lv_cache_entry_t * entry = lv_nanovg_fbo_cache_get(u, lv_area_get_width(&layer->buf_area),
531 lv_area_get_height(&layer->buf_area), 0, NVG_TEXTURE_RGBA);
532 layer->user_data = entry;
533 }
534 break;
535 case LV_EVENT_CHILD_DELETED: {
536 lv_cache_entry_t * entry = layer->user_data;
537 if(entry) {
538 lv_nanovg_fbo_cache_release(u, entry);
539 layer->user_data = NULL;
540 }
541
546 if(u->current_layer == layer) {
547 u->current_layer = NULL;
548 }
549 }
550 break;
551 case LV_EVENT_SCREEN_LOAD_START:
552 on_layer_readback(u, layer);
553 break;
554 case LV_EVENT_INVALIDATE_AREA:
555 lv_nanovg_image_cache_drop(u, lv_event_get_param(e));
556 break;
557 default:
558 break;
559 }
560}
