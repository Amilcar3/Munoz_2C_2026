92Result Fill::colorStops(const ColorStop* colorStops, uint32_t cnt) noexcept
93{
94 if ((!colorStops && cnt > 0) || (colorStops && cnt == 0)) return Result::InvalidArguments;
95
96 if (cnt == 0) {
97 if (pImpl->colorStops) {
98 free(pImpl->colorStops);
99 pImpl->colorStops = nullptr;
100 pImpl->cnt = 0;
101 }
102 return Result::Success;
103 }
104
105 if (pImpl->cnt != cnt) {
106 pImpl->colorStops = static_cast<ColorStop*>(lv_realloc(pImpl->colorStops, cnt * sizeof(ColorStop)));
107 LV_ASSERT_MALLOC(pImpl->colorStops);
108 }
109
110 pImpl->cnt = cnt;
111 memcpy(pImpl->colorStops, colorStops, cnt * sizeof(ColorStop));
112
113 return Result::Success;
114}
