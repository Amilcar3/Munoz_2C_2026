3LVGL header consistency checker.
4
5Subcommands (each exits 0 on success, 1 on failure):
6 public-api Check every header in include/lvgl/ is included by include/lvgl/lvgl.h

11 src/ and include/lvgl/; fail if any entry is unused anywhere
12 deprecated Check no source file includes a deprecated header

336def _source_files(directory: Path):
337 """Yield all source/header files under *directory*."""
338 for path in sorted(directory.rglob("*")):
339 if path.suffix in SOURCE_EXTENSIONS:

343def _read_lines(path: Path) -> list[str]:
344 try:
345 return path.read_text(encoding="utf-8", errors="ignore").splitlines()
346 except OSError:
347 return []

350def _collect_includes(umbrella: Path) -> set[str]:
351 """
352 Return the set of header filenames directly included by *umbrella*.
353 Single-level only -- the umbrella is expected to include everything directly.
354 """
355 includes: set[str] = set()
356 for line in _read_lines(umbrella):
357 m = INCLUDE_QUOTED_RE.search(line) or INCLUDE_ANGLE_RE.search(line)
358 if m:
359 includes.add(Path(m.group(1)).name)
360 return includes

363def _grep(pattern: str, directory: Path) -> list[tuple[Path, int, str]]:
364 """
365 Fast search using the system grep binary with a pure-Python fallback.
366 Returns [(path, lineno, line), ...] for every matching line.
367 """
368 try:
369 result = subprocess.run(
370 [
371 "grep",
372 "-rn",
373 "--include=*.h",
374 "--include=*.c",
375 "--include=*.cpp",
376 "--include=*.hpp",
377 "-E",
378 pattern,
379 str(directory),
380 ],
381 capture_output=True,
382 text=True,
383 )
384 hits: list[tuple[Path, int, str]] = []
385 for raw in result.stdout.splitlines():
386 # grep output format: path:lineno:line
387 parts = raw.split(":", 2)
388 if len(parts) == 3:
389 try:
390 hits.append((Path(parts[0]), int(parts[1]), parts[2]))
391 except ValueError:
392 pass
393 return hits
394 except FileNotFoundError:
395 # grep not available -- fall back to pure Python
396 compiled = re.compile(pattern)
397 hits = []
398 for path in _source_files(directory):
399 for lineno, line in enumerate(_read_lines(path), start=1):
400 if compiled.search(line):
401 hits.append((path, lineno, line))
402 return hits

405def _find_deprecated_headers(folder: Path) -> set[Path]:
406 deprecated: set[Path] = set()
407 for path in folder.rglob("*.h"):
408 try:
409 content = path.read_text(encoding="utf-8", errors="ignore")
410 except OSError:
411 continue
412 if any(marker in content for marker in DEPRECATION_MARKERS):
413 # Store the path relative to the folder so it matches include directives
414 deprecated.add(path.relative_to(folder))
415 return deprecated

490def check_include_paths(repo_root: Path) -> bool:
491 """
492 For every source file under src/:
493 - Quoted includes must resolve to an existing file.
494 - Angle-bracket includes must be in ALLOWED_EXTERNAL_HEADERS and must
495 NOT be in FORBIDDEN_SYSTEM_HEADERS (use LV_*_INCLUDE macros instead).
496 - Macro-style includes (LV_*_INCLUDE) are always fine.
497 """
498 src_dir = repo_root / "src"
499 ok = True
500 disallowed_headers = set()
501 for path in _source_files(src_dir):
502 relative = path.relative_to(src_dir)
503 # Is this specific file in the whitelist ?
504 if any(relative == Path(entry) for entry in HEADER_CHECK_WHITELIST):
505 continue
506
507 # Is this file inside a whitelisted folder ?
508 if not path.name.startswith("lv") and any(
509 relative == Path(entry) or relative.is_relative_to(entry)
510 for entry in HEADER_CHECK_WHITELIST
511 ):
512 continue
513
514 for lineno, line in enumerate(_read_lines(path), start=1):
515 if INCLUDE_MACRO_RE.search(line):
516 continue
517
518 angle = INCLUDE_ANGLE_RE.search(line)
519 if angle:
520 header_name = angle.group(1)
521 loc = f"{path.relative_to(repo_root)}:{lineno}"
522 if header_name in FORBIDDEN_SYSTEM_HEADERS:
523 _error(
524 loc,
525 f"forbidden angle-bracket include <{angle.group(1)}>. "
526 "Use the LV_*_INCLUDE macro instead.",
527 )
528 ok = False
529 elif not is_allowed_header(header_name):
530 disallowed_headers.add(header_name)
531 _error(
532 loc,
533 f"unlisted angle-bracket include <{angle.group(1)}>. '{header_name}' "
534 "Add to ALLOWED_EXTERNAL_HEADERS if intentional.",
535 )
536 ok = False
537 continue
538
539 quoted = INCLUDE_QUOTED_RE.search(line)
540 if quoted:
541 include_path = quoted.group(1)
542 resolved = (path.parent / include_path).resolve()
543 if not resolved.exists():
544 _error(
545 f"{path.relative_to(repo_root)}:{lineno}",
546 f'missing quoted include "{include_path}" '
547 f"(resolved to {resolved})",
548 )
549 ok = False
550
551 disallowed_headers = list(disallowed_headers)
552 disallowed_headers.sort()
553 for header in disallowed_headers:
554 print(f"'{header}',")
555 if ok:
556 log_ok("all include paths in src/ are valid")
557 return ok

565def check_deprecated(repo_root: Path) -> bool:
566 """No source file anywhere in the repo may include a deprecated header."""
567 src_dir = repo_root / "src"
568 deprecated = _find_deprecated_headers(repo_root)
569
570 if not deprecated:
571 log_ok("no deprecated headers found in repo")
572 return True
573
574 include_re = re.compile(r'#\s*include\s*[<"]([^>"]+)[>"]')
575 offenders: dict[Path, list[tuple[int, str]]] = {}
576
577 for path in _source_files(repo_root):
578 hits = []
579 for lineno, line in enumerate(_read_lines(path), start=1):
580 m = include_re.search(line)
581 if m:
582 included_path = Path(m.group(1))
583 # Match against full relative paths, not just filenames
584 if included_path in deprecated:
585 hits.append((lineno, line.strip()))
586 if hits:
587 offenders[path] = hits
588
589 if not offenders:
590 log_ok("no files include deprecated headers")
591 return True
592
593 for path, hits in sorted(offenders.items()):
594 for lineno, line in hits:
595 _error(
596 f"{path.relative_to(repo_root)}:{lineno}", f"deprecated include: {line}"
597 )
598 return False

605
606def check_no_direct_public_include(repo_root: Path) -> bool:
607 """
608 No file under src/ may include a header from include/lvgl/ directly.
609 Internal code must go through lvgl_private.h which pulls in lvgl.h.
610 """
611 src_dir = repo_root / "src"
612 include_dir = (repo_root / "include" / "lvgl").resolve()
613 ok = True
614 deprecated = [p.name for p in _find_deprecated_headers(src_dir)]
615
616 for path in _source_files(src_dir):
617 for lineno, line in enumerate(_read_lines(path), start=1):
618 quoted = INCLUDE_QUOTED_RE.search(line)
619 if not quoted:
620 continue
621 resolved = (path.parent / quoted.group(1)).resolve()
622 if path.name == "lvgl_public.h":
623 continue
624 if resolved.is_relative_to(include_dir) and path.name not in deprecated:
625 _error(
626 f"{path.relative_to(repo_root)}:{lineno}",
627 f'direct public include "{quoted.group(1)}". '
628 "Internal code must include via lvgl_public.h, not include/lvgl/ directly.",
629 )
630 ok = False
631
632 if ok:
633 log_ok("no src/ file includes include/lvgl/ directly")
634 return ok

638# Check 6 -- allowed-list audit
639# ---------------------------------------------------------------------------

641
642def check_allowed_list_audit(repo_root: Path) -> bool:
643 """
644 For every header in ALLOWED_EXTERNAL_HEADERS, search for literal
645 #include <header> in both src/ and include/lvgl/.
646
647 Report:
648 green -- used, all usages are in src/
649 yellow -- used, but at least one usage is in include/lvgl/ (leaks to users)
650 red -- never used anywhere (should be removed from the allow-list)
651
652 Fails if any header is unused.
653 """
654 if not ALLOWED_EXTERNAL_HEADERS:
655 log_ok("ALLOWED_EXTERNAL_HEADERS is empty, nothing to audit")
656 return True
657
658 src_dir = repo_root / "src"
659 include_dir = repo_root / "include" / "lvgl"
660 ok = True
661
662 col = max(len(h) for h in ALLOWED_EXTERNAL_HEADERS) + 4 # padding for <header>
663
664 print(f"\n{C.BOLD}Allowed system header usage report:{C.RESET}\n")
665
666 for header in sorted(ALLOWED_EXTERNAL_HEADERS):
667 pattern = rf"#\s*include\s*<{re.escape(header)}>"
668 src_hits = _grep(pattern, src_dir)
669 inc_hits = _grep(pattern, include_dir)
670
671 padded = f"<{header}>".ljust(col)
672
673 if not src_hits and not inc_hits:
674 print(
675 f" {C.RED}{C.BOLD}{padded}{C.RESET} "
676 f"{C.RED}unused -- remove from ALLOWED_EXTERNAL_HEADERS{C.RESET}"
677 )
678 ok = False
679
680 elif inc_hits:
681 print(
682 f" {C.YELLOW}{C.BOLD}{padded}{C.RESET} "
683 f"{C.YELLOW}leaks into public headers "
684 f"({len(src_hits)} src, {len(inc_hits)} include){C.RESET}"
685 )
686 for path, lineno, line in sorted(inc_hits):
687 _warn(f" {path.relative_to(repo_root)}:{lineno}", line.strip())
688
689 else:
690 print(
691 f" {C.GREEN}{padded}{C.RESET} " f"used in {len(src_hits)} src file(s)"
692 )
693
694 print()
695 if ok:
696 log_ok("all allowed system headers are in use")
697 return ok
