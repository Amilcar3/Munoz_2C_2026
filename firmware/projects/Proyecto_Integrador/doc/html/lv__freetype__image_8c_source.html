93 LV_ASSERT(font != NULL);

95 LV_ASSERT_FREETYPE_FONT_DSC(dsc);
96
97 FT_UInt glyph_index = (FT_UInt)g_dsc->gid.index;
98
99 lv_cache_t * cache = dsc->cache_node->draw_data_cache;
100
101 lv_freetype_image_cache_data_t search_key = {
102 .glyph_index = glyph_index,
103 .size = dsc->size,
104 };
105
106 lv_cache_entry_t * entry = lv_cache_acquire_or_create(cache, &search_key, dsc);
107 if(entry == NULL) {
108 LV_LOG_ERROR("glyph bitmap lookup failed for glyph_index = 0x%" LV_PRIx32, (uint32_t)glyph_index);
109 LV_PROFILER_FONT_END;
110 return NULL;
111 }
112
113 g_dsc->entry = entry;
114 lv_freetype_image_cache_data_t * cache_node = lv_cache_entry_get_data(entry);
115
116 LV_PROFILER_FONT_END;
117 return cache_node->draw_buf;
118}

133static bool freetype_image_create_cb(lv_freetype_image_cache_data_t * data, void * user_data)
134{
135 LV_ASSERT(data != NULL);
136 LV_ASSERT(user_data != NULL);
137 LV_PROFILER_FONT_BEGIN;
138
139 lv_freetype_font_dsc_t * dsc = (lv_freetype_font_dsc_t *)user_data;
140
141 FT_Error error;
142
143 lv_mutex_lock(&dsc->cache_node->face_lock);
144
145 FT_Face face = dsc->cache_node->face;
146 if(FT_IS_SCALABLE(face)) {
147 error = FT_Set_Pixel_Sizes(face, 0, dsc->size);
148 }
149 else {
150 error = FT_Select_Size(face, 0);
151 }
152 if(error) {
153 FT_ERROR_MSG("FT_Set_Pixel_Sizes", error);
154 lv_mutex_unlock(&dsc->cache_node->face_lock);
155 return false;
156 }
157 error = FT_Load_Glyph(face, data->glyph_index,
158 FT_LOAD_COLOR | FT_LOAD_RENDER | FT_LOAD_TARGET_NORMAL | FT_LOAD_NO_AUTOHINT);
159 if(error) {
160 FT_ERROR_MSG("FT_Load_Glyph", error);
161 lv_mutex_unlock(&dsc->cache_node->face_lock);
162 LV_PROFILER_FONT_END;
163 return false;
164 }
165 error = FT_Render_Glyph(face->glyph, FT_RENDER_MODE_NORMAL);
166 if(error) {
167 FT_ERROR_MSG("FT_Render_Glyph", error);
168 lv_mutex_unlock(&dsc->cache_node->face_lock);
169 LV_PROFILER_FONT_END;
170 return false;
171 }
172
173 FT_Glyph glyph;
174 error = FT_Get_Glyph(face->glyph, &glyph);
175 if(error) {
176 FT_ERROR_MSG("FT_Get_Glyph", error);
177 lv_mutex_unlock(&dsc->cache_node->face_lock);
178 LV_PROFILER_FONT_END;
179 return false;
180 }
181
182 FT_BitmapGlyph glyph_bitmap = (FT_BitmapGlyph)glyph;
183
184 uint16_t box_h = glyph_bitmap->bitmap.rows; /*Height of the bitmap in [px]*/
185 uint16_t box_w = glyph_bitmap->bitmap.width; /*Width of the bitmap in [px]*/
186
187 lv_color_format_t col_format;
188 if(glyph_bitmap->bitmap.pixel_mode == FT_PIXEL_MODE_BGRA) {
189 col_format = LV_COLOR_FORMAT_ARGB8888;
190 }
191 else {
192 col_format = LV_COLOR_FORMAT_A8;
193 }
194 uint32_t pitch = glyph_bitmap->bitmap.pitch;
195 uint32_t stride = lv_draw_buf_width_to_stride(box_w, col_format);
196 data->draw_buf = lv_draw_buf_create_ex(font_draw_buf_handlers, box_w, box_h, col_format, stride);
197 if(!data->draw_buf) {
198 LV_LOG_WARN("Could not create draw buffer");
199 FT_Done_Glyph(glyph);
200 lv_mutex_unlock(&dsc->cache_node->face_lock);
201 LV_PROFILER_FONT_END;
202 return false;
203 }
204 lv_draw_buf_clear(data->draw_buf, NULL);
205
206 for(int y = 0; y < box_h; ++y) {
207 lv_memcpy((uint8_t *)(data->draw_buf->data) + y * stride, glyph_bitmap->bitmap.buffer + y * pitch,
208 pitch);
209 }
210
211 lv_draw_buf_flush_cache(data->draw_buf, NULL);
212 FT_Done_Glyph(glyph);
213 lv_mutex_unlock(&dsc->cache_node->face_lock);
214 LV_PROFILER_FONT_END;
215 return true;
216}

223static lv_cache_compare_res_t freetype_image_compare_cb(const lv_freetype_image_cache_data_t * lhs,
224 const lv_freetype_image_cache_data_t * rhs)
225{
226 LV_ASSERT(lhs != NULL);
227 LV_ASSERT(rhs != NULL);
228 if(lhs->glyph_index != rhs->glyph_index) {
229 return lhs->glyph_index > rhs->glyph_index ? 1 : -1;
230 }
231 if(lhs->size != rhs->size) {
232 return lhs->size > rhs->size ? 1 : -1;
233 }
234 return 0;
235}
