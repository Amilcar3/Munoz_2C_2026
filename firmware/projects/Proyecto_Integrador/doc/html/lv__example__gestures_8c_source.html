83void lv_example_gestures(void)
84{
85 lv_obj_t * root_view;
86
87 label_width = RECT_INIT_WIDTH;
88 label_height = RECT_INIT_HEIGHT;
89 label_x = LV_HOR_RES / 2 - (label_width / 2);
90 label_y = LV_VER_RES / 2 - (label_height / 2);
91
92 root_view = lv_screen_active();
93
94 lv_obj_set_style_bg_color(root_view, lv_color_hex(0xffffff), LV_PART_MAIN);
95 label = lv_label_create(root_view);
96 lv_obj_set_scrollable(root_view, false);
97
98 lv_label_set_text(label, "Zoom, rotate or move");
99 lv_obj_set_clickable(label, true);
100
101 lv_style_init(&label_style);
102 lv_style_set_bg_color(&label_style, lv_color_hex(RECT_COLOR));
103 lv_style_set_bg_opa(&label_style, LV_OPA_COVER);
104
105 lv_style_set_width(&label_style, (int)label_width);
106 lv_style_set_height(&label_style, (int)label_height);
107
108 lv_style_set_x(&label_style, (int)label_x);
109 lv_style_set_y(&label_style, (int)label_y);
110
111 lv_obj_add_style(label, &label_style, LV_STATE_DEFAULT);
112
113 lv_obj_add_event_cb(label, label_rotate, LV_EVENT_GESTURE, label);
114 lv_obj_add_event_cb(label, label_scale, LV_EVENT_GESTURE, label);

126static void label_swipe(lv_event_t * gesture_event)
127{
128
129 lv_dir_t dir;
130 lv_indev_gesture_state_t state;
131 const char * text;
132
133 if(lv_event_get_gesture_type(gesture_event) != LV_INDEV_GESTURE_TWO_FINGERS_SWIPE) {
134 return;
135 }
136
137 state = lv_event_get_gesture_state(gesture_event, LV_INDEV_GESTURE_TWO_FINGERS_SWIPE);
138 dir = lv_event_get_two_fingers_swipe_dir(gesture_event);
139
140 if(state == LV_INDEV_GESTURE_STATE_ENDED) {
141
142 text = "NONE";
143 lv_label_set_text(label, text);
144 }
145 else if(state == LV_INDEV_GESTURE_STATE_RECOGNIZED) {
146
147 switch(dir) {
148 case LV_DIR_LEFT:
149 text = "LEFT";
150 break;
151 case LV_DIR_RIGHT:
152 text = "RIGHT";
153 break;
154 case LV_DIR_TOP:
155 text = "TOP";
156 break;
157 case LV_DIR_BOTTOM:
158 text = "BOTTOM";
159 break;
160 default:
161 text = "???";
162 break;
163 }
164
165 lv_label_set_text_fmt(label, "%s - %f", text, lv_event_get_two_fingers_swipe_distance(gesture_event));
166 }

174static void label_scale(lv_event_t * gesture_event)
175{
176
177 static int initial_w = -1;
178 static int initial_h = -1;
179 lv_indev_gesture_state_t state;
180 static lv_point_t center_pnt;
181 static float base_scale = 1.0;
182 float scale;
183 float label_width_float;
184 float label_height_float;
185
186 /* Ensure the gesture is a pinch */
187 if(lv_event_get_gesture_type(gesture_event) != LV_INDEV_GESTURE_PINCH) {
188 return;
189 }
190
191 state = lv_event_get_gesture_state(gesture_event, LV_INDEV_GESTURE_PINCH);
192
193 scale = base_scale * lv_event_get_pinch_scale(gesture_event);
194
195 /* Reset state when the gesture ended */
196 if(state == LV_INDEV_GESTURE_STATE_ENDED) {
197
198 /* Pinch gesture has ended - reset the width/height for the next pinch gesture*/
199 initial_w = -1;
200 initial_h = -1;
201
202 base_scale = scale;
203
204 return;
205 }
206
207 /* The first time the gesture is recognized, save its center */
208 if(state == LV_INDEV_GESTURE_STATE_RECOGNIZED) {
209
210 if((initial_h == -1 || initial_w == -1)) {
211
212 /* Pinch gesture has been recognized - this is the first event in a series of recognized events */
213 /* The scaling is applied relative to the original width/height of the rectangle */
214 initial_w = label_width;
215 initial_h = label_height;
216 center_pnt.x = lv_obj_get_x(label) + label_width / 2;
217 center_pnt.y = lv_obj_get_y(label) + label_height / 2;
218 }
219
220 /* The gesture is recognized, we can now use the scale */
221
222 /* Avoids a situation where the rectangle becomes too small or too big,
223 * adding limits */
224 if(scale < 0.4f) {
225 scale = 0.4f;
226 }
227 else if(scale > 2.0f) {
228 scale = 2.0f;
229 }
230
231 label_x = center_pnt.x - label_width / 2;
232 label_y = center_pnt.y - label_height / 2;
233
234 label_width_float = (float)RECT_INIT_WIDTH * scale;
235 label_height_float = (float)RECT_INIT_HEIGHT * scale;
236
237 /* Update position and size */
238 lv_style_set_width(&label_style, (int)label_width_float);
239 lv_style_set_height(&label_style, (int)label_height_float);
240 lv_style_set_x(&label_style, (int)label_x);
241 lv_style_set_y(&label_style, (int)label_y);
242
243 lv_obj_add_style(label, &label_style, LV_STATE_DEFAULT);
244
245 label_width = (int)label_width_float;
246 label_height = (int)label_height_float;
247 }
248}

255static void label_rotate(lv_event_t * gesture_event)
256{
257
258 float angle_degrees = 0.f;
259 static float start_angle = 0.f;
260 lv_indev_gesture_state_t state;
261
262 if(lv_event_get_gesture_type(gesture_event) != LV_INDEV_GESTURE_ROTATE) {
263 return;
264 }
265
266 state = lv_event_get_gesture_state(gesture_event, LV_INDEV_GESTURE_ROTATE);
267
268 /* Calculate new angle. The x10 is due to lv_obj_set_style_transform_rotation using x10 angle in parameter */
269 angle_degrees = start_angle + 10.0f * (lv_event_get_rotation(gesture_event) * 180.0f / M_PI);
270
271 /* Once the gesture ends, save the current angle to the start_angle */
272 if(state == LV_INDEV_GESTURE_STATE_ENDED) {
273
274 start_angle = angle_degrees;
275 }
276
277 /* If the gesture is recognized, rotate the label */
278 if(state == LV_INDEV_GESTURE_STATE_RECOGNIZED) {
279
280 /* Need to set the pivot to the center of the widget to not rotate
281 around the top-left corner */
282 lv_obj_set_style_transform_pivot_x(label, lv_obj_get_width(label) / 2, 0);
283 lv_obj_set_style_transform_pivot_y(label, lv_obj_get_height(label) / 2, 0);
284
285 lv_obj_set_style_transform_rotation(label, (int) angle_degrees, 0);
286 }
287
288 return;
289}
