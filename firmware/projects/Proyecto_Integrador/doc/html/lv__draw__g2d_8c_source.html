189static int32_t _g2d_evaluate(lv_draw_unit_t * u, lv_draw_task_t * t)
190{
191 LV_UNUSED(u);
192 if(!init_successful) {
193 return 0;
194 }
195
196 const lv_draw_dsc_base_t * draw_dsc_base = (lv_draw_dsc_base_t *) t->draw_dsc;
197 lv_draw_buf_t * draw_buf = draw_dsc_base->layer->draw_buf;
198
199 if(!_g2d_dest_cf_supported(draw_dsc_base->layer->color_format))
200 return 0;
201
202 if(draw_buf && !g2d_search_buf_map(draw_buf->data))
203 return 0;
204
205 switch(t->type) {
206 case LV_DRAW_TASK_TYPE_FILL: {
207 const lv_draw_fill_dsc_t * draw_dsc = (lv_draw_fill_dsc_t *) t->draw_dsc;
208
209 /* Most simple case: just a plain rectangle (no radius, no gradient). */
210 if((draw_dsc->radius != 0) || (draw_dsc->grad.dir != (lv_grad_dir_t)LV_GRAD_DIR_NONE))
211 return 0;
212
213 if(t->preference_score > 70) {
214 t->preference_score = 70;
215 t->preferred_draw_unit_id = DRAW_UNIT_ID_G2D;
216 }
217 return 1;
218 }
219 case LV_DRAW_TASK_TYPE_IMAGE: {
220 /* TODO: Fix issue where images rendered to a different layer don't render in the final layer.*/
221 return 0;
222#if 0

225 if(!_g2d_src_cf_supported(draw_dsc->header.cf))

246static int32_t _g2d_dispatch(lv_draw_unit_t * draw_unit, lv_layer_t * layer)
247{
248 lv_draw_g2d_unit_t * draw_g2d_unit = (lv_draw_g2d_unit_t *) draw_unit;
249
250#if LV_USE_G2D_DRAW_THREAD
251 lv_draw_sw_thread_dsc_t * thread_dsc = &draw_g2d_unit->thread_dsc;
252
253 /* Return immediately if it's busy with draw task. */
254 if(thread_dsc->task_act)
255 return 0;
256#else
257 /* Return immediately if it's busy with draw task. */
258 if(draw_g2d_unit->task_act)
259 return 0;
260#endif
261
262 /* Try to get an ready to draw. */
263 lv_draw_task_t * t = lv_draw_get_available_task(layer, NULL, DRAW_UNIT_ID_G2D);
264
265 if(t == NULL || t->preferred_draw_unit_id != DRAW_UNIT_ID_G2D)
266 return LV_DRAW_UNIT_IDLE;
267
268 if(lv_draw_layer_alloc_buf(layer) == NULL) {
269 t->state = LV_DRAW_TASK_STATE_FAILED;
270 return LV_DRAW_UNIT_IDLE;
271 }
272
273 t->state = LV_DRAW_TASK_STATE_IN_PROGRESS;
274 t->draw_unit = draw_unit;
275
276#if LV_USE_G2D_DRAW_THREAD
277 thread_dsc->task_act = t;
278
279 /* Let the render thread work. */
280 lv_thread_sync_signal(&thread_dsc->new_task_sync);
281#else
282 draw_g2d_unit->task_act = t;
283
284 _g2d_execute_drawing(t);
285
286 draw_g2d_unit->task_act->state = LV_DRAW_TASK_STATE_FINISHED;
287 draw_g2d_unit->task_act = NULL;
288
289 /* The draw unit is free now. Request a new dispatching as it can get a new task. */
290 lv_draw_dispatch_request();
291#endif
292
293 return 1;
294}

339static void _g2d_render_thread_cb(void * ptr)
340{
341 lv_draw_sw_thread_dsc_t * thread_dsc = ptr;
342
343 while(1) {
344 /* Wait for sync if there is no task set. */
345 while(thread_dsc->task_act == NULL) {
346 if(thread_dsc->exit_status)
347 break;
348
349 lv_thread_sync_wait(&thread_dsc->new_task_sync);
350 }
351
352 if(thread_dsc->exit_status) {
353 LV_LOG_INFO("Ready to exit G2D draw thread.");
354 break;
355 }
356
357 _g2d_execute_drawing(thread_dsc->task_act);
358
359 /* Signal the ready state to dispatcher. */
360 thread_dsc->task_act->state = LV_DRAW_TASK_STATE_FINISHED;
361
362 /* Cleanup. */
363 thread_dsc->task_act = NULL;
364
365 lv_thread_sync_signal(&thread_dsc->task_done_sync);
366
367 /* The draw unit is free now. Request a new dispatching as it can get a new task. */
368 lv_draw_dispatch_request();
369 }
370
371 LV_LOG_INFO("Exit G2D draw thread.");
372}
