4static void scroll_event_cb(lv_event_t * e)
5{
6 lv_obj_t * cont = lv_event_get_target_obj(e);
7
8 lv_area_t cont_a;
9 lv_obj_get_coords(cont, &cont_a);
10 int32_t cont_y_center = cont_a.y1 + lv_area_get_height(&cont_a) / 2;
11
12 int32_t r = lv_obj_get_height(cont) * 7 / 10;
13 int32_t child_cnt = (int32_t)lv_obj_get_child_count(cont);
14 for(int32_t i = 0; i < child_cnt; i++) {
15 lv_obj_t * child = lv_obj_get_child(cont, i);
16 lv_area_t child_a;
17 lv_obj_get_coords(child, &child_a);
18
19 int32_t child_y_center = child_a.y1 + lv_area_get_height(&child_a) / 2;
20 int32_t diff_y = LV_ABS(child_y_center - cont_y_center);
21
22 /* Map the vertical distance from the centre onto a circle of radius r. */
23 int32_t x;
24 if(diff_y >= r) {
25 x = r;
26 }
27 else {
28 uint32_t x_sqr = r * r - diff_y * diff_y;
29 lv_sqrt_res_t res;
30 lv_sqrt(x_sqr, &res, 0x8000);
31 x = r - res.i;
32 }
33
34 lv_obj_set_style_translate_x(child, x, 0);
35
36 /* Fade items further from the centre. */
37 lv_opa_t opa = (lv_opa_t)lv_map(x, 0, r, LV_OPA_TRANSP, LV_OPA_COVER);
38 lv_obj_set_style_opa(child, LV_OPA_COVER - opa, 0);
39 }
