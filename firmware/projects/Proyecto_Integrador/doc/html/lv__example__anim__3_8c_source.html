68static int32_t anim_path_bezier3_cb(const lv_anim_t * a)
69{
70 int32_t t = lv_map(a->act_time, 0, a->duration, 0, 1024);
71 int32_t step = lv_bezier3(t, 0, ginfo.p1, ginfo.p2, 1024);
72 int32_t new_value;
73 new_value = step * (a->end_value - a->start_value);
74 new_value = new_value >> 10;
75 new_value += a->start_value;
76 return new_value;
77}

79static void refer_chart_cubic_bezier(void)
80{
81 for(uint16_t i = 0; i < CHART_POINTS_NUM; i ++) {
82 int32_t t = (int32_t)i * 1024 / (CHART_POINTS_NUM - 1);
83 int32_t step = lv_bezier3(t, 0, ginfo.p1, ginfo.p2, 1024);
84 lv_chart_set_series_value_by_id2(ginfo.chart, ginfo.ser1, i, t, step);
85 }
86 lv_chart_refresh(ginfo.chart);
87}
