81bool lv_demos_create(char * info[], int size)
82{
83 const int demos_count = LV_DEMOS_COUNT;
84
85 if(demos_count <= 0) {
86 LV_LOG_ERROR("Please enable some lv_demos firstly!");
87 return false;
88 }
89
90 const demo_entry_info_t * entry_info = NULL;
91 if(size <= 0) { /* default: first demo*/
92 entry_info = &demos_entry_info[0];
93 }
94 else if(entry_info == NULL && info) {
95 const char * name = info[0];
96 for(int i = 0; i < demos_count; i++) {
97 if(lv_strcmp(name, demos_entry_info[i].name) == 0) {
98 entry_info = &demos_entry_info[i];
99 }
100 }
101 }
102
103 if(entry_info == NULL) {
104 LV_LOG_ERROR("lv_demos create(%s) failure!", size > 0 ? info[0] : "");
105 return false;
106 }
107
108 if(entry_info->entry_cb) {
109 entry_info->entry_cb();
110 return true;
111 }
112
113 return false;
114}
