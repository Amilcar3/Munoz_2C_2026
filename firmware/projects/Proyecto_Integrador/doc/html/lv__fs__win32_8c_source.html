116{
117 lv_fs_res_t res;
118
119 switch(error) {

121 res = LV_FS_RES_OK;
122 break;
123 case ERROR_BAD_UNIT:
124 case ERROR_NOT_READY:
125 case ERROR_CRC:
126 case ERROR_SEEK:
127 case ERROR_NOT_DOS_DISK:
128 case ERROR_WRITE_FAULT:
129 case ERROR_READ_FAULT:
130 case ERROR_GEN_FAILURE:
131 case ERROR_WRONG_DISK:
132 res = LV_FS_RES_HW_ERR;
133 break;
134 case ERROR_INVALID_HANDLE:
135 case ERROR_INVALID_TARGET_HANDLE:
136 res = LV_FS_RES_FS_ERR;
137 break;
138 case ERROR_FILE_NOT_FOUND:
139 case ERROR_PATH_NOT_FOUND:
140 case ERROR_INVALID_DRIVE:
141 case ERROR_NO_MORE_FILES:
142 case ERROR_SECTOR_NOT_FOUND:
143 case ERROR_BAD_NETPATH:
144 case ERROR_BAD_NET_NAME:
145 case ERROR_BAD_PATHNAME:
146 case ERROR_FILENAME_EXCED_RANGE:
147 res = LV_FS_RES_NOT_EX;
148 break;
149 case ERROR_DISK_FULL:
150 res = LV_FS_RES_FULL;
151 break;
152 case ERROR_SHARING_VIOLATION:
153 case ERROR_LOCK_VIOLATION:
154 case ERROR_DRIVE_LOCKED:
155 res = LV_FS_RES_LOCKED;
156 break;
157 case ERROR_ACCESS_DENIED:
158 case ERROR_CURRENT_DIRECTORY:
159 case ERROR_WRITE_PROTECT:
160 case ERROR_NETWORK_ACCESS_DENIED:
161 case ERROR_CANNOT_MAKE:
162 case ERROR_FAIL_I24:

167 break;
168 case ERROR_BUSY:
169 res = LV_FS_RES_BUSY;

173 break;
174 case ERROR_NOT_SAME_DEVICE:
175 case ERROR_DIRECT_ACCESS_HANDLE:

177 break;
178 case ERROR_TOO_MANY_OPEN_FILES:
179 case ERROR_ARENA_TRASHED:
180 case ERROR_NOT_ENOUGH_MEMORY:

211static void * fs_open(lv_fs_drv_t * drv, const char * path, lv_fs_mode_t mode)
212{
213 LV_ASSERT(path != NULL);
214 LV_UNUSED(drv);
215
216 DWORD desired_access = 0;
217
218 if(mode & LV_FS_MODE_RD) {
219 desired_access |= GENERIC_READ;
220 }
221
222 if(mode & LV_FS_MODE_WR) {
223 desired_access |= GENERIC_WRITE;
224 }
225
226 /*Make the path relative to the current directory (the projects root folder)*/
227
228 char buf[MAX_PATH];
229 lv_snprintf(buf, sizeof(buf), LV_FS_WIN32_PATH "%s", path);
230
231 return (void *)CreateFileA(
232 buf,
233 desired_access,
234 FILE_SHARE_READ,
235 NULL,
236 OPEN_EXISTING,
237 FILE_ATTRIBUTE_NORMAL,
238 NULL);
239}

306static lv_fs_res_t fs_seek(lv_fs_drv_t * drv, void * file_p, uint32_t pos, lv_fs_whence_t whence)
307{
308 LV_ASSERT(file_p != NULL);
309 LV_UNUSED(drv);
310
311 DWORD move_method = (DWORD) -1;
312 if(whence == LV_FS_SEEK_SET) {
313 move_method = FILE_BEGIN;
314 }
315 else if(whence == LV_FS_SEEK_CUR) {
316 move_method = FILE_CURRENT;
317 }
318 else if(whence == LV_FS_SEEK_END) {
319 move_method = FILE_END;
320 }
321
322 LARGE_INTEGER distance_to_move;
323 distance_to_move.QuadPart = pos;
324 return SetFilePointerEx((HANDLE)file_p, distance_to_move, NULL, move_method)
325 ? LV_FS_RES_OK
326 : fs_error_from_win32(GetLastError());
327}

337static lv_fs_res_t fs_tell(lv_fs_drv_t * drv, void * file_p, uint32_t * pos_p)
338{
339 LV_ASSERT(file_p != NULL);
340 LV_ASSERT(pos_p != NULL);
341 LV_UNUSED(drv);
342
343 LARGE_INTEGER file_pointer;
344 file_pointer.QuadPart = 0;
345
346 LARGE_INTEGER distance_to_move;
347 distance_to_move.QuadPart = 0;
348 if(SetFilePointerEx(
349 (HANDLE)file_p,
350 distance_to_move,
351 &file_pointer,
352 FILE_CURRENT)) {
353 if(file_pointer.QuadPart > LONG_MAX) {
354 return LV_FS_RES_INV_PARAM;
355 }
356 else {
357 *pos_p = file_pointer.LowPart;
358 return LV_FS_RES_OK;
359 }
360 }
361 else {
362 return fs_error_from_win32(GetLastError());
363 }

372static void * fs_dir_open(lv_fs_drv_t * drv, const char * path)
373{
374 LV_ASSERT(path != NULL);
375 LV_UNUSED(drv);
376 dir_handle_t * handle = (dir_handle_t *)lv_malloc(sizeof(dir_handle_t));
377 handle->dir_p = INVALID_HANDLE_VALUE;
378 handle->next_error = LV_FS_RES_OK;
379 WIN32_FIND_DATAA fdata;
380
381 /*Make the path relative to the current directory (the projects root folder)*/
382 char buf[LV_FS_MAX_PATH_LEN];
383 lv_snprintf(buf, sizeof(buf), LV_FS_WIN32_PATH "%s\\*", path);
384
385 lv_strcpy(handle->next_fn, "");
386 handle->dir_p = FindFirstFileA(buf, &fdata);
387
388 if(handle->dir_p != INVALID_HANDLE_VALUE) {
389 do {
390 if(is_dots_name(fdata.cFileName)) {
391 continue;
392 }
393 else {
394 if(fdata.dwFileAttributes & FILE_ATTRIBUTE_DIRECTORY) {
395 lv_snprintf(handle->next_fn, sizeof(handle->next_fn), "/%s", fdata.cFileName);
396 }
397 else {
398 lv_snprintf(handle->next_fn, sizeof(handle->next_fn), "%s", fdata.cFileName);
399 }
400 break;
401 }
402 } while(FindNextFileA(handle->dir_p, &fdata));
403 }
404
405 if(handle->dir_p == INVALID_HANDLE_VALUE) {
406 lv_free(handle);
407 handle->next_error = fs_error_from_win32(GetLastError());
408 return INVALID_HANDLE_VALUE;
409 }
410 else {
411 handle->next_error = LV_FS_RES_OK;
412 return handle;
413 }
414}

425static lv_fs_res_t fs_dir_read(lv_fs_drv_t * drv, void * dir_p, char * fn, uint32_t fn_len)
426{
427 LV_ASSERT(dir_p != NULL);
428 LV_ASSERT(fn != NULL);
429 LV_UNUSED(drv);
430 LV_ASSERT(fn_len > 0);
431
432 dir_handle_t * handle = (dir_handle_t *)dir_p;
433 lv_strlcpy(fn, handle->next_fn, fn_len);
434 lv_fs_res_t current_error = handle->next_error;
435 lv_strcpy(handle->next_fn, "");
436
437 WIN32_FIND_DATAA fdata;
438
439 while(FindNextFileA(handle->dir_p, &fdata)) {
440 if(is_dots_name(fdata.cFileName)) {
441 continue;
442 }
443 else {
444 if(fdata.dwFileAttributes & FILE_ATTRIBUTE_DIRECTORY) {
445 lv_snprintf(handle->next_fn, sizeof(handle->next_fn), "/%s", fdata.cFileName);
446 }
447 else {
448 lv_snprintf(handle->next_fn, sizeof(handle->next_fn), "%s", fdata.cFileName);
449 }
450 break;
451 }
452 }
453
454 if(handle->next_fn[0] == '\0') {
455 handle->next_error = fs_error_from_win32(GetLastError());
456 }
457
458 return current_error;
459}
