62static lv_result_t init_display(lv_display_t * display)
63{
64 lv_sdl_texture_display_data_t * ddata = lv_malloc_zeroed(sizeof(*ddata));
65 if(!ddata) {
66 LV_LOG_WARN("No memory for display data");
67 return LV_RESULT_INVALID;
68 }
69 ddata->renderer = SDL_CreateRenderer(lv_sdl_window_get_window(display), -1,
70 LV_SDL_ACCELERATED ? SDL_RENDERER_ACCELERATED : SDL_RENDERER_SOFTWARE);
71 if(!ddata->renderer) {
72 LV_LOG_ERROR("Failed to create SDL renderer '%s'", SDL_GetError());
73 lv_free(ddata);
74 return LV_RESULT_INVALID;
75 }
76 lv_sdl_backend_set_display_data(display, ddata);
77
78 /*It will render directly to default Texture, so the buffer is not used, so just set something*/
79 static lv_draw_buf_t draw_buf;
80 static uint8_t dummy_buf; /*It won't be used as it will render to the SDL textures directly*/
81 lv_draw_buf_init(&draw_buf, 4096, 4096, LV_COLOR_FORMAT_ARGB8888, 4096 * 4, &dummy_buf, 4096 * 4096 * 4);
82
83 lv_display_set_draw_buffers(display, &draw_buf, NULL);
84 lv_display_set_render_mode(display, LV_DISPLAY_RENDER_MODE_FULL);
85 lv_display_set_flush_cb(display, flush_cb);
86 return LV_RESULT_OK;
