52lv_result_t lv_obj_send_event(lv_obj_t * obj, lv_event_code_t event_code, void * param)
53{
54 LV_CHECK_OBJ(obj, MY_CLASS, return 0);
55
56 lv_event_t e;
57 e.current_target = obj;
58 e.original_target = obj;
59 e.code = event_code;
60 e.user_data = NULL;
61 e.param = param;
62 e.deleted = 0;
63 e.stop_bubbling = 0;
64 e.stop_processing = 0;
65 e.stop_trickling = 0;
66
67 lv_event_push(&e);
68
69 /*Send the event*/
70 lv_result_t res = event_send_core(&e);
71
72 /*Remove this element from the list*/
73 lv_event_pop(&e);
74
75 return res;
76}

78lv_result_t lv_obj_event_base(const lv_obj_class_t * class_p, lv_event_t * e)
79{
80 LV_CHECK_ARG(e != NULL, return LV_RESULT_INVALID);
81 LV_CHECK_ARG(e->current_target != NULL, return LV_RESULT_INVALID);
82
83 const lv_obj_class_t * base;
84 if(class_p == NULL) base = ((lv_obj_t *)e->current_target)->class_p;
85 else base = class_p->base_class;
86
87 /*Find a base in which call the ancestor's event handler_cb if set*/
88 while(base && base->event_cb == NULL) base = base->base_class;
89
90 if(base == NULL) return LV_RESULT_OK;
91 if(base->event_cb == NULL) return LV_RESULT_OK;
92
93 /*Call the actual event callback*/
94 e->user_data = NULL;
95 LV_PROFILER_EVENT_BEGIN_TAG(lv_event_code_get_name(e->code));
96 base->event_cb(base, e);
97 LV_PROFILER_EVENT_END_TAG(lv_event_code_get_name(e->code));
98
99 lv_result_t res = LV_RESULT_OK;
100 /*Stop if the object is deleted*/
101 if(e->deleted) res = LV_RESULT_INVALID;
102
103 return res;
104}

148uint32_t lv_obj_remove_event_cb(lv_obj_t * obj, lv_event_cb_t event_cb)
149{
150 LV_CHECK_ARG(obj != NULL, return 0);
151
152 uint32_t event_cnt = lv_obj_get_event_count(obj);
153 uint32_t removed_count = 0;
154 int32_t i;
155
156 if(event_cnt == 0) return 0;
157
158 for(i = event_cnt - 1; i >= 0; i--) {
159 lv_event_dsc_t * dsc = lv_obj_get_event_dsc(obj, i);
160 if(dsc && (event_cb == NULL || dsc->cb == event_cb)) {
161 lv_obj_remove_event(obj, i);
162 removed_count++;
163 }
164 }
165
166 return removed_count;

350static lv_result_t event_send_core(lv_event_t * e)
351{
352 LV_TRACE_EVENT("Sending event %d to %p with %p param", e->code, (void *)e->original_target, e->param);
353
354 lv_indev_t * indev_act = lv_indev_active();
355 if(indev_act) {
356 if(e->stop_processing) return LV_RESULT_OK;
357 if(e->deleted) return LV_RESULT_INVALID;
358 }
359
360 lv_obj_t * target = e->current_target;
361 lv_result_t res = LV_RESULT_OK;
362 lv_event_list_t * list = target->spec_attr ? &target->spec_attr->event_list : NULL;
363
364 res = lv_event_send(list, e, true);
365 if(res != LV_RESULT_OK || e->stop_processing) return res;
366
367 res = lv_obj_event_base(NULL, e);
368 if(res != LV_RESULT_OK || e->stop_processing) return res;
369
370 res = lv_event_send(list, e, false);
371 if(res != LV_RESULT_OK || e->stop_processing) return res;
372
373 lv_obj_t * parent = lv_obj_get_parent(e->current_target);
374 if(parent && event_is_bubbled(e)) {
375 e->current_target = parent;
376 res = event_send_core(e);
377 }
378 if(res != LV_RESULT_OK) return res;
379
380 /*Trickle down to children if enabled*/
381 if(event_is_trickled(e)) {
382 uint32_t child_count = lv_obj_get_child_count(target);
383
384 /* we don't want the event to bubble up again when trickling down */
385 e->stop_bubbling = 1;
386
387 for(uint32_t i = 0; i < child_count && res == LV_RESULT_OK && !e->stop_processing; i++) {
388 lv_obj_t * child = lv_obj_get_child(target, i);
389 if(child) {
390 e->current_target = child;
391 res = event_send_core(e);
392 if(res != LV_RESULT_OK) {
393 LV_LOG_WARN("Trickle down event %d to child %p failed", e->code, (void *)child);
394 break;
395 }
396 }
397 }
398 }
399
400 return res;
401}

403static bool event_is_bubbled(lv_event_t * e)
404{
405 if(e->stop_bubbling) return false;
406
407 /*Event codes that always bubble*/
408 switch(e->code) {
409 case LV_EVENT_CHILD_CREATED:
410 case LV_EVENT_CHILD_DELETED:
411 return true;
412 default:
413 break;
414 }
415
416 /*Check other codes only if bubbling is enabled*/
417 if(lv_obj_is_event_bubble(e->current_target) == false) return false;
418
419 switch(e->code) {
420 case LV_EVENT_HIT_TEST:
421 case LV_EVENT_COVER_CHECK:
422 case LV_EVENT_REFR_EXT_DRAW_SIZE:
423 case LV_EVENT_DRAW_MAIN_BEGIN:
424 case LV_EVENT_DRAW_MAIN:
425 case LV_EVENT_DRAW_MAIN_END:
426 case LV_EVENT_DRAW_POST_BEGIN:
427 case LV_EVENT_DRAW_POST:
428 case LV_EVENT_DRAW_POST_END:
429 case LV_EVENT_DRAW_TASK_ADDED:
430 case LV_EVENT_REFRESH:
431 case LV_EVENT_DELETE:
432 case LV_EVENT_CHILD_CREATED:
433 case LV_EVENT_CHILD_DELETED:
434 case LV_EVENT_CHILD_CHANGED:
435 case LV_EVENT_SIZE_CHANGED:
436 case LV_EVENT_STYLE_CHANGED:
437 case LV_EVENT_GET_SELF_SIZE:
438 return false;
439 default:
440 return true;
441 }
442}
