44def _scan_enum_types() -> set[str]:
45 """Scan LVGL headers to find all typedef enum and int-like alias type names."""
46 result = set()
47 # In a checkout LVGL_INC *is* LVGL_SRC, so the two trees have to be merged
48 # rather than concatenated or every header is read twice.
49 headers = sorted({*LVGL_SRC.rglob("*.h"), *LVGL_INC.rglob("*.h")})
50 for h in headers:
51 text = h.read_text(errors="ignore")
52 # typedef enum { ... } lv_xxx_t;
53 for m in re.finditer(
54 r"typedef\s+enum\s*\{[^}]*\}\s*(lv_\w+_t)", text, re.DOTALL
55 ):
56 result.add(m.group(1))
57 # typedef <int-like> lv_xxx_t;
58 for m in re.finditer(
59 r"typedef\s+((?:unsigned\s+)?\w+)\s+(lv_\w+_t)\s*;", text
60 ):
61 base = m.group(1).strip()
62 if base in (
63 "unsigned int", "unsigned char", "unsigned short",
64 "unsigned long", "int", "char", "short", "long",
65 "uint8_t", "uint16_t", "uint32_t", "uint64_t",
66 "int8_t", "int16_t", "int32_t", "int64_t", "size_t",
67 ):
68 result.add(m.group(2))
69 return result

120def parse_struct_fields(body: str) -> list[StructField]:
121 fields = []
122 depth = 0
123 clean_lines = []
124 for line in body.splitlines():
125 stripped = line.strip()
126 if re.match(r"(struct|union)\s*\{", stripped):
127 depth += 1
128 continue
129 if stripped.startswith("}") and depth > 0:
130 depth -= 1
131 continue
132 if depth > 0:
133 continue
134 clean_lines.append(stripped)
135
136 for line in clean_lines:
137 line = line.rstrip(";").strip()
138 if not line or line.startswith("/*") or line.startswith("//") or line.startswith("*") or line.startswith("#"):
139 continue
140
141 comment = ""
142 cm = re.search(r"/\*\*?<?\s*(.*?)\s*\*/", line)
143 if cm:
144 comment = cm.group(1)
145 line = line[:cm.start()].strip().rstrip(";").strip()
146 if not line:
147 continue
148
149 if re.search(r"\‍[.*\‍]", line):
150 am = re.match(r"(?:const\s+)?(\w[\w\s\*]*?)\s+(\w+)\s*\‍[", line)
151 if am:
152 fields.append(StructField(
153 name=am.group(2), c_type=am.group(1).strip(),
154 is_array=True, comment=comment,
155 ))
156 continue
157
158 bm = re.match(r"(?:const\s+)?(\w[\w\s\*]*?)\s+(\w+)\s*:\s*(\d+)", line)
159 if bm:
160 fields.append(StructField(
161 name=bm.group(2), c_type=bm.group(1).strip(),
162 is_bitfield=True, bitfield_width=int(bm.group(3)),
163 comment=comment,
164 ))
165 continue
166
167 fm = re.match(r"((?:const\s+)?[\w][\w\s]*?(?:\s*\*\s*(?:const\s*)?)*\*?)\s+(\*?\w+)", line)
168 if fm:
169 c_type = fm.group(1).strip()
170 name = fm.group(2).strip().lstrip("*")
171 is_ptr = "*" in c_type or fm.group(2).startswith("*")
172 fields.append(StructField(
173 name=name, c_type=c_type,
174 is_pointer=is_ptr,
175 is_obj_pointer=is_ptr and "lv_obj_t" in c_type,
176 is_string=is_ptr and "char" in c_type and c_type.count("*") <= 1,
177 comment=comment,
178 ))
179
180 return fields

183def _find_structs(text: str):
184 """Find top-level struct _lv_*_t definitions with brace-balanced matching."""
185 for m in re.finditer(r"struct\s+_lv_(\w+)_t\s*\{", text):
186 name = m.group(1)
187 start = m.end()
188 depth = 1
189 i = start
190 while i < len(text) and depth > 0:
191 if text[i] == "{":
192 depth += 1
193 elif text[i] == "}":
194 depth -= 1
195 i += 1
196 if depth == 0:
197 yield name, text[start:i - 1]

200def parse_widgets() -> dict[str, WidgetDef]:
201 # Pass 1: collect all structs with parent type
202 candidates = []
203 for private_h in sorted(WIDGETS_DIR.glob("*/lv_*_private.h")):
204 text = private_h.read_text()
205 widget_dir = private_h.parent.name
206 for raw_name, body in _find_structs(text):
207 struct_name = f"lv_{raw_name}_t"
208 first_line = ""
209 for line in body.splitlines():
210 line = line.strip()
211 if line and not line.startswith(("/*", "//", "*", "#")):
212 first_line = line.rstrip(";").strip()
213 break
214 parent_match = re.match(r"(lv_\w+_t)\s+\w+", first_line)
215 if parent_match:
216 candidates.append((struct_name, parent_match.group(1), body, widget_dir))
217
218 # Pass 2: resolve inheritance from lv_obj_t (order-independent)
219 widget_types = {"lv_obj_t"}
220 changed = True
221 while changed:
222 changed = False
223 for name, parent, _, _ in candidates:
224 if name not in widget_types and parent in widget_types:
225 widget_types.add(name)
226 changed = True
227
228 # Pass 3: build WidgetDef for discovered widgets
229 widgets = {}
230 for struct_name, parent_type, body, widget_dir in candidates:
231 if struct_name in widget_types and struct_name != "lv_obj_t":
232 all_fields = parse_struct_fields(body)
233 widgets[struct_name] = WidgetDef(
234 struct_name=struct_name, c_type=struct_name,
235 parent_type=parent_type,
236 fields=all_fields[1:] if all_fields else [],
237 widget_dir=widget_dir,
238 )
239 return widgets

242def _field_expr(f: StructField, attr: str = "_wv") -> str | None:
243 """Return snapshot expression for a field, or None to skip.
244
245 `attr` is the class's own cast of the object. Each class needs its own,
246 because a widget struct embeds its ancestor as the first member: casting
247 an lv_slider_t to lv_bar_t is valid, but reading lv_bar_t's fields off an
248 lv_slider_t cast is not, and safe_field() would quietly return the default.
249 """
250 if f.is_array:
251 return None
252 if f.is_obj_pointer:
253 return f'ptr_or_none(self.{attr}.safe_field("{f.name}"))'
254 if f.is_string:
255 return f'safe_string(self.{attr}, "{f.name}")'
256 if f.is_pointer:
257 return f'ptr_or_none(self.{attr}.safe_field("{f.name}"))'
258 if f.c_type == "lv_color_t":
259 return f'safe_color(self.{attr}, "{f.name}")'
260 if f.c_type == "lv_area_t":
261 return f'safe_area(self.{attr}, "{f.name}")'
262 if f.c_type == "lv_point_t":
263 return f'safe_point(self.{attr}, "{f.name}")'
264 # Known wrapper types — use snapshot() for rich output
265 _WRAPPER_TYPES = {
266 "lv_draw_buf_t": ("lvglgdb.lvgl.draw.lv_draw_buf", "LVDrawBuf"),
267 "lv_ll_t": ("lvglgdb.lvgl.misc.lv_ll", "LVList"),
268 "lv_anim_t": ("lvglgdb.lvgl.misc.lv_anim", "LVAnim"),
269 "lv_array_t": ("lvglgdb.lvgl.misc.lv_array", "LVArray"),
270 }
271 if f.c_type in _WRAPPER_TYPES:
272 mod, cls = _WRAPPER_TYPES[f.c_type]
273 return f'safe_wrapper(self.{attr}, "{f.name}", "{mod}", "{cls}")'
274 if f.is_bitfield or f.c_type in SIMPLE_INT_TYPES or f.c_type.startswith(("uint", "int")):
275 return f'int(self.{attr}.safe_field("{f.name}", 0))'
276 # TODO: implement generic struct expansion (Value.to_dict) for
277 # non-enum lv_*_t types like lv_calendar_date_t.
278 if f.c_type in _INT_SAFE_TYPES:
279 return f'int(self.{attr}.safe_field("{f.name}", 0))'
280 return None

283def _field_type_name(f: StructField) -> str | None:
284 """Infer a spec type name for a field. Returns None to skip."""
285 if f.is_array:
286 return None
287 if f.is_obj_pointer:
288 return "pointer"
289 if f.is_string:
290 return "string"
291 if f.is_pointer:
292 return "pointer"
293 if f.c_type == "lv_color_t":
294 return "color"
295 if f.c_type == "lv_area_t":
296 return "area"
297 if f.c_type == "lv_point_t":
298 return "point"
299 if f.is_bitfield:
300 return "bool" if f.bitfield_width == 1 else f"enum:{f.bitfield_width}"
301 if f.c_type == "bool":
302 return "bool"
303 if f.c_type in SIMPLE_INT_TYPES or f.c_type.startswith(("uint", "int")):
304 return "int"
305 if f.c_type in _INT_SAFE_TYPES:
306 return "int"
307 return None

310def _topo_sort(widgets: dict[str, WidgetDef]) -> list[WidgetDef]:
311 result, visited = [], set()
312 def visit(w):
313 if w.c_type in visited:
314 return
315 visited.add(w.c_type)
316 if w.parent_type in widgets:
317 visit(widgets[w.parent_type])
318 result.append(w)
319 for w in widgets.values():
320 visit(w)
321 return result

384def safe_wrapper(obj, field_name, module_path, class_name):
385 """Read a struct field using its known Value wrapper, return snapshot dict.
386
387 Not every wrapper has a snapshot(); LVArray does not. Value forwards an
388 unknown attribute to gdb.Value.__getitem__, so asking for one raises
389 gdb.error rather than AttributeError, and the caller would otherwise lose
390 the whole widget - and its subtree - over a single field.
391 """
392 import gdb
393
394 val = obj.safe_field(field_name)
395 if val is None or not getattr(val, 'is_ok', True):
396 return None
397 import importlib
398 try:
399 mod = importlib.import_module(module_path)
400 cls = getattr(mod, class_name)
401 wrapper = cls(val)
402 if getattr(type(wrapper), "snapshot", None) is None:
403 # An embedded struct - lv_scale_t.needles - is not convertible to an
404 # int, so the address has to come from the field itself.
405 if val.type.strip_typedefs().code != gdb.TYPE_CODE_PTR:
406 val = val.address
407 return {"addr": hex(int(val)), "type": class_name}
408 return wrapper.snapshot().as_dict()
409 except (gdb.error, gdb.MemoryError, ImportError, AttributeError, TypeError):
410 return None
411'''

414def gen_widget_file(wdef: WidgetDef, widgets: dict[str, WidgetDef]) -> str:
415 """Generate a single widget module file."""
416 attr = f"_wv_{wdef.c_type}"
417 lines = [
418 '"""',
419 f"Auto-generated wrapper for {wdef.c_type}.",
420 "",
421 "Do not edit manually. Regenerate from the GDB script root with:",
422 " python3 scripts/generate_all.py",
423 '"""',
424 "",
425 ]
426
427 # Import parent
428 if wdef.parent_type == "lv_obj_t":
429 lines.append("from lvglgdb.lvgl.core.lv_obj import LVObject")
430 else:
431 parent_def = widgets[wdef.parent_type]
432 lines.append(f"from .{parent_def.module_name} import {parent_def.class_name}")
433
434 # Import helpers only if needed
435 needs = set()
436 for f in wdef.fields:
437 expr = _field_expr(f, attr)
438 if expr is None:
439 continue
440 if "ptr_or_none" in expr:
441 needs.add("ptr_or_none")
442 if "safe_string" in expr:
443 needs.add("safe_string")
444 if "safe_color" in expr:
445 needs.add("safe_color")
446 if "safe_area" in expr:
447 needs.add("safe_area")
448 if "safe_point" in expr:
449 needs.add("safe_point")
450 if "safe_wrapper" in expr:
451 needs.add("safe_wrapper")
452
453 if needs:
454 imports = ", ".join(sorted(needs))
455 lines.append(f"from ._helpers import {imports}")
456
457 lines.append("")
458 lines.append("")
459
460 parent_cls = wdef.parent_class_name
461 lines.append(f"class {wdef.class_name}({parent_cls}):")
462 lines.append(f' """LVGL {wdef.widget_dir} widget ({wdef.c_type})."""')
463 lines.append("")
464 lines.append(f" def __init__(self, obj):")
465 lines.append(f" super().__init__(obj)")
466 lines.append(f' self.{attr} = self.cast("{wdef.c_type}", ptr=True) or self')
467 lines.append(f" self._wv = self.{attr}")
468 lines.append("")
469
470 # Properties
471 snapshot_fields = []
472 for f in wdef.fields:
473 expr = _field_expr(f, attr)
474 if expr is None:
475 continue
476 lines.append(f" @property")
477 lines.append(f" def {f.name}(self):")
478 if f.comment:
479 lines.append(f' """{f.comment}"""')
480 lines.append(f" return {expr}")
481 lines.append("")
482 snapshot_fields.append(f.name)
483
484 # snapshot — override to include widget-specific fields in widget_data
485 lines.append(f" def snapshot(self, include_children=False, include_styles=False):")
486 lines.append(f' """Snapshot with widget-specific fields in widget_data."""')
487 lines.append(f" s = super().snapshot(include_children=include_children, include_styles=include_styles)")
488
489 if snapshot_fields:
490 lines.append(f" d = s.get('widget_data') or {{}}")
491 for fname in snapshot_fields:
492 lines.append(f' d["{fname}"] = self.{fname}')
493 lines.append(f" s['widget_data'] = d")
494
495 lines.append(f" return s")
496 lines.append("")
497
498 return "\n".join(lines)

567def update_specs_json(widgets: dict[str, WidgetDef]) -> dict:
568 """Merge auto-generated field info with hand-written specs.
569
570 - _auto section is always regenerated
571 - Hand-written keys (summary_tpl, primary, enums) are preserved
572 - New widgets get only _auto (hand-write summary_tpl/primary/enums to complete)
573 - Removed widgets get _removed: true
574 """
575 existing = {}
576 if SPECS_JSON.exists():
577 existing = json.loads(SPECS_JSON.read_text())
578
579 result = {}
580 # Preserve _comment
581 if "_comment" in existing:
582 result["_comment"] = existing["_comment"]
583 else:
584 result["_comment"] = "Auto-generated + hand-written widget specs. _auto is regenerated; other keys are preserved."
585
586 seen = set()
587 for wdef in sorted(widgets.values(), key=lambda w: w.c_class_name):
588 key = wdef.c_class_name # e.g. "lv_label"
589 seen.add(key)
590 old = existing.get(key, {})
591 entry = {"_auto": _build_auto_spec(wdef)}
592 # Preserve hand-written keys
593 for k in ("summary_tpl", "primary", "enums"):
594 if k in old:
595 entry[k] = old[k]
596 result[key] = entry
597
598 # Mark removed widgets
599 for key, val in existing.items():
600 if key.startswith("_") or key in seen:
601 continue
602 val["_removed"] = True
603 result[key] = val
604
605 SPECS_JSON.write_text(json.dumps(result, indent=2, ensure_ascii=False) + "\n")
606 return result
