108static void * fs_open(lv_fs_drv_t * drv, const char * path, lv_fs_mode_t mode)
109{
110 LV_ASSERT(path != NULL);
111 LV_UNUSED(drv);
112 uint8_t flags = 0;
113
114 if(mode == LV_FS_MODE_WR) flags = FA_WRITE | FA_OPEN_ALWAYS;
115 else if(mode == LV_FS_MODE_RD) flags = FA_READ;
116 else if(mode == (LV_FS_MODE_WR | LV_FS_MODE_RD)) flags = FA_READ | FA_WRITE | FA_OPEN_ALWAYS;
117
118 FIL * f = lv_malloc(sizeof(FIL));
119 if(f == NULL) return NULL;
120
121 char buf[LV_FS_MAX_PATH_LEN];
122 lv_snprintf(buf, sizeof(buf), LV_FS_FATFS_PATH "%s", path);
123
124 FRESULT res = f_open(f, buf, flags);
125 if(res == FR_OK) {
126 return f;
127 }
128 else {
129 lv_free(f);
130 return NULL;
131 }
132}

200static lv_fs_res_t fs_seek(lv_fs_drv_t * drv, void * file_p, uint32_t pos, lv_fs_whence_t whence)
201{
202 LV_ASSERT(file_p != NULL);
203 LV_UNUSED(drv);
204 switch(whence) {
205 case LV_FS_SEEK_SET:
206 f_lseek(file_p, pos);
207 break;
208 case LV_FS_SEEK_CUR:
209 f_lseek(file_p, f_tell((FIL *)file_p) + pos);
210 break;
211 case LV_FS_SEEK_END:
212 f_lseek(file_p, f_size((FIL *)file_p) + pos);
213 break;
214 default:
215 break;
216 }
217 return LV_FS_RES_OK;
218}

270static lv_fs_res_t fs_dir_read(lv_fs_drv_t * drv, void * dir_p, char * fn, uint32_t fn_len)
271{
272 LV_ASSERT(dir_p != NULL);
273 LV_ASSERT(fn != NULL);
274 LV_UNUSED(drv);
275 LV_ASSERT(fn_len > 0);
276
277 FRESULT res;
278 FILINFO fno;
279 fn[0] = '\0';
280
281 do {
282 res = f_readdir(dir_p, &fno);
283 if(res != FR_OK) return LV_FS_RES_UNKNOWN;
284
285 if(fno.fname[0] == 0) break; /* End of the directory */
286
287 if(fno.fattrib & AM_DIR) {
288 lv_snprintf(fn, fn_len, "/%s", fno.fname);
289 }
290 else lv_strlcpy(fn, fno.fname, fn_len);
291
292 } while(lv_strcmp(fn, "/.") == 0 || lv_strcmp(fn, "/..") == 0);
293
294 return LV_FS_RES_OK;
295}
