73lv_display_t * lv_linux_drm_create(void)
74{
75 lv_tick_set_cb(tick_cb);
76 lv_drm_ctx_t * ctx = lv_zalloc(sizeof(*ctx));
77 LV_ASSERT_MALLOC(ctx);
78 if(!ctx) {
79 LV_LOG_ERROR("Failed to create drm context");
80 return NULL;
81 }
82
83 ctx->display = lv_display_create(1, 1);
84
85 if(!ctx->display) {
86 LV_LOG_ERROR("Failed to create display");
87 lv_free(ctx);
88 return NULL;
89 }
90
91 lv_display_set_driver_data(ctx->display, ctx);
92 lv_display_add_event_cb(ctx->display, event_cb, LV_EVENT_DELETE, NULL);
93 return ctx->display;
94}

96lv_result_t lv_linux_drm_set_file(lv_display_t * disp, const char * file, int64_t connector_id)
97{
98 LV_CHECK_ARG(disp != NULL, return LV_RESULT_INVALID);
99 LV_CHECK_ARG(file != NULL, return LV_RESULT_INVALID);
100 LV_UNUSED(connector_id);
101 lv_drm_ctx_t * ctx = lv_display_get_driver_data(disp);
102
103 LV_CHECK_ARG_MSG(ctx != NULL, return LV_RESULT_INVALID, "Invalid display");
104
105 lv_result_t err = drm_device_init(ctx, file);
106 if(err != LV_RESULT_OK) {
107 LV_LOG_ERROR("Failed to initialize DRM device");
108 return LV_RESULT_INVALID;
109 }
110
111 lv_display_set_resolution(disp, ctx->drm_mode->hdisplay, ctx->drm_mode->vdisplay);
112
113 lv_opengles_egl_set_display_color_format(disp);
114
115 ctx->egl_interface = drm_get_egl_interface(ctx);
116 ctx->egl_ctx = lv_opengles_egl_context_create(&ctx->egl_interface);
117 if(!ctx->egl_ctx) {
118 LV_LOG_ERROR("Failed to create egl context");
119 return LV_RESULT_INVALID;
120 }
121
122 /* Let the opengles texture driver handle the texture lifetime */
123 ctx->texture.is_texture_owner = true;
124 /*Initialize the draw buffers and texture*/
125 lv_result_t res = lv_opengles_texture_reshape(&ctx->texture, disp, ctx->drm_mode->hdisplay, ctx->drm_mode->vdisplay);
126 if(res != LV_RESULT_OK) {
127 LV_LOG_ERROR("Failed to create draw buffers");
128 lv_opengles_egl_context_destroy(ctx->egl_ctx);
129 ctx->egl_ctx = NULL;
130 return LV_RESULT_INVALID;
131 }
132
133 lv_display_set_flush_cb(disp, flush_cb);
134 lv_display_set_render_mode(disp, LV_USE_DRAW_NANOVG ? LV_DISPLAY_RENDER_MODE_FULL : LV_DISPLAY_RENDER_MODE_DIRECT);
135
136 lv_display_add_event_cb(ctx->display, event_cb, LV_EVENT_RESOLUTION_CHANGED, NULL);
137 lv_display_add_event_cb(ctx->display, event_cb, LV_EVENT_DELETE, NULL);
138
139 return LV_RESULT_OK;
140}

154static void event_cb(lv_event_t * e)
155{
156 lv_event_code_t code = lv_event_get_code(e);
157 lv_display_t * display = (lv_display_t *) lv_event_get_target(e);
158 lv_drm_ctx_t * ctx = lv_display_get_driver_data(display);
159 switch(code) {
160 case LV_EVENT_DELETE:
161 if(ctx) {
162 lv_opengles_egl_context_destroy(ctx->egl_ctx);
163 ctx->egl_ctx = NULL;
164 lv_opengles_texture_deinit(&ctx->texture);
165 drm_device_deinit(ctx);
166 lv_display_set_driver_data(display, NULL);
167 }
168 break;
169 case LV_EVENT_RESOLUTION_CHANGED: {
170 lv_result_t res = lv_opengles_texture_reshape(&ctx->texture, display, lv_display_get_horizontal_resolution(display),
171 lv_display_get_vertical_resolution(display));
172
173 if(res != LV_RESULT_OK) {
174 LV_LOG_ERROR("Failed to resize display");
175 }
176 }
177 break;
178 default:
179 return;
180 }
181}

202void drm_device_deinit(lv_drm_ctx_t * ctx)
203{
204 if(ctx->drm_crtc) {
205 drmModeSetCrtc(ctx->fd,
206 ctx->drm_crtc->crtc_id,
207 ctx->drm_crtc->buffer_id,
208 ctx->drm_crtc->x,
209 ctx->drm_crtc->y,
210 &ctx->drm_connector->connector_id,
211 1,
212 &ctx->drm_crtc->mode);
213 drmModeFreeCrtc(ctx->drm_crtc);
214 ctx->drm_crtc = 0;
215 }
216 drm_destroy_window(ctx, ctx->gbm_surface);
217
218 if(ctx->gbm_dev) {
219 gbm_device_destroy(ctx->gbm_dev);
220 ctx->gbm_dev = NULL;
221 }
222 if(ctx->drm_connector) {
223 drmModeFreeConnector(ctx->drm_connector);
224 ctx->drm_connector = NULL;
225 }
226 if(ctx->drm_encoder) {
227 drmModeFreeEncoder(ctx->drm_encoder);
228 ctx->drm_encoder = NULL;
229 }
230 if(ctx->drm_resources) {
231 drmModeFreeResources(ctx->drm_resources);
232 ctx->drm_resources = NULL;
233 }
234 if(ctx->fd > 0) {
235 drmClose(ctx->fd);
236 }
237 ctx->fd = 0;
238 ctx->drm_mode = NULL;
239 lv_free(ctx);
240}

252static int drm_do_page_flip(lv_drm_ctx_t * ctx, int timeout_ms)
253{
254 fd_set fds;
255 FD_ZERO(&fds);
256 FD_SET(ctx->fd, &fds);
257
258 drmEventContext event_ctx;
259 lv_memset(&event_ctx, 0, sizeof(event_ctx));
260 event_ctx.version = 2;
261 event_ctx.page_flip_handler = drm_on_page_flip;
262
263 struct timeval timeout;
264 int status;
265 if(timeout_ms >= 0) {
266 timeout.tv_sec = timeout_ms / 1000;
267 timeout.tv_usec = (timeout_ms % 1000) * 1000;
268 status = select(ctx->fd + 1, &fds, NULL, NULL, &timeout);
269 }
270 else {
271 status = select(ctx->fd + 1, &fds, NULL, NULL, NULL);
272 }
273
274 if(status == 1) {
275 drmHandleEvent(ctx->fd, &event_ctx);
276 }
277 return status;
278}

295static drm_fb_state_t * drm_fb_state_create(lv_drm_ctx_t * ctx, struct gbm_bo * bo)
296{
297 LV_ASSERT(bo != NULL);
298 drm_fb_state_t * fb = (drm_fb_state_t *)gbm_bo_get_user_data(bo);
299
300 if(fb) {
301 return fb;
302 }
303
304 uint32_t width = gbm_bo_get_width(bo);
305 uint32_t height = gbm_bo_get_height(bo);
306 uint32_t handles[4] = {0};
307 uint32_t strides[4] = {0};
308 uint32_t offsets[4] = {0};
309 uint64_t modifiers[4] = {0};
310 uint32_t format = gbm_bo_get_format(bo);
311 uint64_t modifier = gbm_bo_get_modifier(bo);
312 uint32_t fb_id = 0;
313 uint64_t addfb2_mods = 0;
314 int32_t status;
315
316 drmGetCap(ctx->fd, DRM_CAP_ADDFB2_MODIFIERS, &addfb2_mods);
317
318 for(int i = 0; i < gbm_bo_get_plane_count(bo); i++) {
319 handles[i] = gbm_bo_get_handle_for_plane(bo, i).u32;
320 strides[i] = gbm_bo_get_stride_for_plane(bo, i);
321 offsets[i] = gbm_bo_get_offset(bo, i);
322 modifiers[i] = modifier;
323 }
324
325 if(addfb2_mods && modifier != DRM_FORMAT_MOD_INVALID) {
326 status = drmModeAddFB2WithModifiers(ctx->fd, width, height, format,
327 handles, strides, offsets, modifiers,
328 &fb_id, DRM_MODE_FB_MODIFIERS);
329 }
330 else {
331 status = drmModeAddFB2(ctx->fd, width, height, format,
332 handles, strides, offsets, &fb_id, 0);
333 }
334
335 if(status < 0) {
336 LV_LOG_ERROR("Failed to create drm_fb_state: %d", status);
337 return NULL;
338 }
339
340 fb = (drm_fb_state_t *)lv_malloc(sizeof(*fb));
341 if(!fb) {
342 LV_LOG_ERROR("Failed to allocate drmfb_state");
343 return NULL;
344 }
345
346 fb->fd = ctx->fd;
347 fb->bo = bo;
348 fb->fb_id = fb_id;
349
350 gbm_bo_set_user_data(bo, fb, drm_fb_state_destroy_cb);
351 return fb;
352}

354static void drm_flip_cb(void * driver_data, bool vsync)
355{
356 lv_drm_ctx_t * ctx = (lv_drm_ctx_t *) driver_data;
357
358 if(ctx->gbm_bo_pending) {
359 gbm_surface_release_buffer(ctx->gbm_surface, ctx->gbm_bo_pending);
360 }
361 ctx->gbm_bo_pending = gbm_surface_lock_front_buffer(ctx->gbm_surface);
362
363 if(!ctx->gbm_bo_pending) {
364 LV_LOG_ERROR("Failed to lock front buffer");
365 return;
366 }
367
368 drm_fb_state_t * pending_fb = drm_fb_state_create(ctx, ctx->gbm_bo_pending);
369
370 if(!ctx->gbm_bo_pending || !pending_fb) {
371 LV_LOG_ERROR("Failed to get gbm front buffer");
372 return;
373 }
374
375 if(vsync) {
376 while(ctx->gbm_bo_flipped && drm_do_page_flip(ctx, -1) >= 0)
377 continue;
378 }
379 else {
380 drm_do_page_flip(ctx, 0);
381 }
382
383 if(!ctx->gbm_bo_flipped) {
384 if(!ctx->crtc_isset) {
385 int status = drmModeSetCrtc(ctx->fd, ctx->drm_encoder->crtc_id, pending_fb->fb_id, 0, 0,
386 &(ctx->drm_connector->connector_id), 1, ctx->drm_mode);
387 if(status < 0) {
388 LV_LOG_ERROR("Failed to set crtc: %d", status);
389 return;
390 }
391 ctx->crtc_isset = true;
392 if(ctx->gbm_bo_presented)
393 gbm_surface_release_buffer(ctx->gbm_surface, ctx->gbm_bo_presented);
394 ctx->gbm_bo_presented = ctx->gbm_bo_pending;
395 ctx->gbm_bo_flipped = NULL;
396 ctx->gbm_bo_pending = NULL;
397 return;
398 }
399
400
401 uint32_t flip_flags = DRM_MODE_PAGE_FLIP_EVENT;
402 int status = drmModePageFlip(ctx->fd, ctx->drm_encoder->crtc_id, pending_fb->fb_id, flip_flags, ctx);
403 if(status < 0) {
404 LV_LOG_ERROR("Failed to enqueue page flip: %d", status);
405 return;
406 }
407 ctx->gbm_bo_flipped = ctx->gbm_bo_pending;
408 ctx->gbm_bo_pending = NULL;
409 }
410
411 /* We need to ensure our surface has a free buffer, otherwise GL will
412 * have no buffer to render on. */
413 while(!gbm_surface_has_free_buffers(ctx->gbm_surface) &&
414 drm_do_page_flip(ctx, -1) >= 0) {
415 continue;
416 }
417}

419static lv_result_t drm_device_init(lv_drm_ctx_t * ctx, const char * path)
420{
421 if(!path) {
422 LV_LOG_ERROR("Device path must not be NULL");
423 return LV_RESULT_INVALID;
424 }
425 int ret = open(path, O_RDWR);
426 if(ret < 0) {
427 LV_LOG_ERROR("Failed to open device path '%s'", path);
428 goto open_err;
429 }
430 ctx->fd = ret;
431
432 ret = drmSetClientCap(ctx->fd, DRM_CLIENT_CAP_UNIVERSAL_PLANES, 1);
433 if(ret < 0) {
434 LV_LOG_ERROR("Failed to set universal planes capability");
435 goto set_client_cap_err;
436 }
437
438 ctx->drm_resources = drmModeGetResources(ctx->fd);
439 if(!ctx->drm_resources) {
440 LV_LOG_ERROR("Failed to get card resources");
441 goto get_resources_err;
442 }
443
444 ctx->drm_connector = drm_get_connector(ctx);
445 if(!ctx->drm_connector) {
446 LV_LOG_ERROR("Failed to find a suitable connector");
447 goto get_connector_err;
448 }
449
450 ctx->drm_mode = drm_get_mode(ctx);
451 if(!ctx->drm_mode) {
452 LV_LOG_ERROR("Failed to find a suitable drm mode");
453 goto get_mode_err;
454 }
455
456 ctx->drm_encoder = drm_get_encoder(ctx);
457 if(!ctx->drm_encoder) {
458 LV_LOG_ERROR("Failed to find a suitable encoder");
459 goto get_encoder_err;
460 }
461
462 ctx->gbm_dev = gbm_create_device(ctx->fd);
463 if(!ctx->gbm_dev) {
464 LV_LOG_ERROR("Failed to create gbm device");
465 goto gbm_create_device_err;
466 }
467
468 if(drmSetMaster(ctx->fd) < 0) {
469 LV_LOG_ERROR("Failed to become DRM master");
470 goto set_master_err;
471 }
472
473 ctx->drm_crtc = drm_get_crtc(ctx);
474 if(!ctx->drm_crtc) {
475 LV_LOG_ERROR("Failed to get crtc");
476 goto get_crtc_err;
477 }
478
479 return LV_RESULT_OK;
480
481get_crtc_err:
482 gbm_device_destroy(ctx->gbm_dev);
483 ctx->gbm_dev = NULL;
484set_master_err:
485 /* Nothing special to do */
486gbm_create_device_err:
487 drmModeFreeEncoder(ctx->drm_encoder);
488 ctx->drm_encoder = NULL;
489get_encoder_err:
490 drmModeFreeConnector(ctx->drm_connector);
491 ctx->drm_connector = NULL;
492get_mode_err:
493 /* Nothing special to do */
494get_connector_err:
495 drmModeFreeResources(ctx->drm_resources);
496 ctx->drm_resources = NULL;
497get_resources_err:
498 /* Nothing special to do */
499set_client_cap_err:
500 close(ctx->fd);
501 ctx->fd = 0;
502open_err:
503 return LV_RESULT_INVALID;
504}

546static drmModeModeInfo * drm_get_mode(lv_drm_ctx_t * ctx)
547{
548 LV_ASSERT(ctx->drm_connector != NULL);
549 if(ctx->mode_select_cb) {
550 lv_linux_drm_mode_t * modes = lv_malloc(sizeof(lv_linux_drm_mode_t) * ctx->drm_connector->count_modes);
551 if(!modes) {
552 LV_LOG_WARN("Failed to allocate memory for drm modes");
553 return NULL;
554 }
555 for(int i = 0; i < ctx->drm_connector->count_modes; i++) {
556 modes[i].mode_info = &ctx->drm_connector->modes[i];
557 }
558 size_t mode_index = ctx->mode_select_cb(ctx->display, modes, (size_t)ctx->drm_connector->count_modes);
559 lv_free(modes);
560
561 if(mode_index >= (size_t)ctx->drm_connector->count_modes) {
562 LV_LOG_ERROR("Failed to select drm mode. User select callback return an invalid mode index");
563 return NULL;
564 }
565 return &ctx->drm_connector->modes[mode_index];
566 }
567
568 drmModeModeInfo * best_mode = NULL;
569 uint32_t best_area = 0;
570
571 for(int i = 0 ; i < ctx->drm_connector->count_modes; ++i) {
572 drmModeModeInfo * mode = &ctx->drm_connector->modes[i];
573 if(mode->type & DRM_MODE_TYPE_PREFERRED) {
574 return mode;
575 }
576 uint32_t area = mode->hdisplay * mode->vdisplay;
577 if(area > best_area) {
578 best_area = area;
579 best_mode = mode;
580 }
581 }
582 LV_LOG_WARN("Failed to find a drm mode with the TYPE_PREFERRED flag. Using the one with the biggest area");
583 return best_mode;
584}

586static drmModeCrtc * drm_get_crtc(lv_drm_ctx_t * ctx)
587{
588 drmModeCrtc * crtc = drmModeGetCrtc(ctx->fd, ctx->drm_encoder->crtc_id);
589 if(crtc) {
590 return crtc;
591 }
592
593 /* if there is no current CRTC, attach a suitable one */
594 for(int i = 0; i < ctx->drm_resources->count_crtcs; i++) {
595 if(ctx->drm_encoder->possible_crtcs & (1 << i)) {
596 ctx->drm_encoder->crtc_id = ctx->drm_resources->crtcs[i];
597 crtc = drmModeGetCrtc(ctx->fd, ctx->drm_encoder->crtc_id);
598 break;
599 }
600 }
601 return crtc;
602}

604static drmModeEncoder * drm_get_encoder(lv_drm_ctx_t * ctx)
605{
606 LV_ASSERT(ctx->drm_connector != NULL);
607 drmModeEncoder * encoder = NULL;
608 for(int i = 0; i < ctx->drm_resources->count_encoders; i++) {
609 encoder = drmModeGetEncoder(ctx->fd, ctx->drm_resources->encoders[i]);
610 if(!encoder) {
611 continue;
612 }
613 for(int j = 0; j < ctx->drm_connector->count_encoders; j++) {
614 if(encoder->encoder_id == ctx->drm_connector->encoders[j]) {
615 return encoder;
616 }
617 }
618 drmModeFreeEncoder(encoder);
619 encoder = NULL;
620 }
621 return encoder;
622}

645static void drm_destroy_window(void * driver_data, void * native_window)
646{
647 lv_drm_ctx_t * ctx = (lv_drm_ctx_t *)driver_data;
648 LV_ASSERT(native_window == ctx->gbm_surface);
649
650 if(!native_window) {
651 return;
652 }
653
654 if(ctx->gbm_bo_pending) {
655 gbm_surface_release_buffer(ctx->gbm_surface, ctx->gbm_bo_pending);
656 ctx->gbm_bo_pending = NULL;
