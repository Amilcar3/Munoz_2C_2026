42void lv_draw_nema_gfx_fill(lv_draw_task_t * t, const lv_draw_fill_dsc_t * dsc, const lv_area_t * coords)
43{
44 if(dsc->opa <= LV_OPA_MIN) return;
45
46 lv_draw_nema_gfx_unit_t * draw_nema_gfx_unit = (lv_draw_nema_gfx_unit_t *)t->draw_unit;
47
48 lv_layer_t * layer = t->target_layer;
49 lv_area_t rel_coords;
50 rel_coords = *coords;
51 lv_area_move(&rel_coords, -layer->buf_area.x1, -layer->buf_area.y1);
52
53 lv_area_t rel_clip_area;
54 rel_clip_area = t->clip_area;
55 lv_area_move(&rel_clip_area, -layer->buf_area.x1, -layer->buf_area.y1);
56
57 nema_set_clip(rel_clip_area.x1, rel_clip_area.y1, lv_area_get_width(&rel_clip_area),
58 lv_area_get_height(&rel_clip_area));
59
60 lv_area_t clipped_coords;
61 if(!lv_area_intersect(&clipped_coords, &rel_coords, &rel_clip_area))
62 return; /*Fully clipped, nothing to do*/
63
64 lv_color_format_t dst_cf = layer->draw_buf->header.cf;
65 uint32_t dst_nema_cf = lv_nemagfx_cf_to_nema(dst_cf);
66
67 /* the stride should be computed internally for NEMA_TSC images and images missing a stride value */
68 int32_t stride = (dst_cf >= LV_COLOR_FORMAT_NEMA_TSC_START && dst_cf <= LV_COLOR_FORMAT_NEMA_TSC_END) ?
69 -1 : lv_area_get_width(&(layer->buf_area)) * lv_color_format_get_size(dst_cf);
70
71 nema_bind_dst_tex((uintptr_t)NEMA_VIRT2PHYS(layer->draw_buf->data), lv_area_get_width(&(layer->buf_area)),
72 lv_area_get_height(&(layer->buf_area)), dst_nema_cf, stride);
73
74 int32_t coords_bg_w = lv_area_get_width(&rel_coords);
75 int32_t coords_bg_h = lv_area_get_height(&rel_coords);
76 int32_t short_side = LV_MIN(coords_bg_w, coords_bg_h);
77 int32_t radius = LV_MIN(dsc->radius, short_side >> 1);
78
79 if((dsc->grad.dir == (lv_grad_dir_t)LV_GRAD_DIR_NONE)) {
80
81 lv_color32_t col32 = lv_color_to_32(dsc->color, dsc->opa);
82 uint32_t bg_color = nema_rgba(col32.red, col32.green, col32.blue, col32.alpha);
83
84 if(col32.alpha < 255U) {
85 nema_set_blend_fill(NEMA_BL_SRC_OVER);
86 bg_color = nema_premultiply_rgba(bg_color);
87 }
88 else {
89 nema_set_blend_fill(NEMA_BL_SRC);
90 }
91
92 if(radius > 0.f)
93 nema_fill_rounded_rect_aa(rel_coords.x1, rel_coords.y1, coords_bg_w, coords_bg_h, radius, bg_color);
94 else
95 nema_fill_rect(rel_coords.x1, rel_coords.y1, coords_bg_w, coords_bg_h, bg_color);
96 }
97#if LV_USE_NEMA_VG
98 else {
99
100 nema_vg_paint_clear(draw_nema_gfx_unit->paint);
101
102 nema_vg_set_blend(NEMA_BL_SRC_OVER | NEMA_BLOP_SRC_PREMULT);
103
104 float stops[LV_GRADIENT_MAX_STOPS];
105 color_var_t colors[LV_GRADIENT_MAX_STOPS];
106
107 uint32_t cnt = LV_MIN(dsc->grad.stops_count, LV_GRADIENT_MAX_STOPS);
108
109 for(uint8_t i = 0; i < cnt; i++) {
110 stops[i] = (float)(dsc->grad.stops[i].frac) / 255.f;
111 colors[i].a = LV_OPA_MIX2(dsc->grad.stops[i].opa, dsc->opa);
112 colors[i].r = dsc->grad.stops[i].color.red;
113 colors[i].g = dsc->grad.stops[i].color.green;
114 colors[i].b = dsc->grad.stops[i].color.blue;
115 }
116
117 nema_vg_grad_set(draw_nema_gfx_unit->gradient, cnt, stops, colors);
118
119 nema_tex_mode_t extend_type;
120 switch(dsc->grad.extend) {
121 case LV_GRAD_EXTEND_REPEAT:
122 extend_type = NEMA_TEX_REPEAT;
123 break;
124 case LV_GRAD_EXTEND_REFLECT:
125 extend_type = NEMA_TEX_MIRROR;
126 break;
127 case LV_GRAD_EXTEND_PAD:
128 default:
129 extend_type = NEMA_TEX_CLAMP;
130 break;
131 }
132
133 if(dsc->grad.dir == LV_GRAD_DIR_VER || dsc->grad.dir == LV_GRAD_DIR_HOR
134 || dsc->grad.dir == LV_GRAD_DIR_LINEAR) {
135 nema_vg_paint_set_type(draw_nema_gfx_unit->paint, NEMA_VG_PAINT_GRAD_LINEAR);
136
137 float x0, y0, x1, y1;
138
139 if(dsc->grad.dir == LV_GRAD_DIR_HOR) {
140 x0 = rel_coords.x1;
141 x1 = rel_coords.x2;
142 y0 = rel_coords.y1;
143 y1 = rel_coords.y2;
144 }
145 else if(dsc->grad.dir == LV_GRAD_DIR_VER) {
146 x0 = rel_coords.x1;
147 x1 = rel_coords.x1;
148 y0 = rel_coords.y1;
149 y1 = rel_coords.y2;
150 }
151 else {
152 x0 = rel_coords.x1 + lv_pct_to_px(dsc->grad.params.linear.start.x, lv_area_get_width(coords));
153 x1 = rel_coords.x1 + lv_pct_to_px(dsc->grad.params.linear.end.x, lv_area_get_width(coords));
154 y0 = rel_coords.y1 + lv_pct_to_px(dsc->grad.params.linear.start.y, lv_area_get_height(coords));
155 y1 = rel_coords.y1 + lv_pct_to_px(dsc->grad.params.linear.end.y, lv_area_get_height(coords));
156 }
157
158 nema_vg_paint_set_grad_linear(draw_nema_gfx_unit->paint, draw_nema_gfx_unit->gradient, x0, y0, x1, y1,
159 extend_type | NEMA_FILTER_BL);
160 }
161 else if(dsc->grad.dir == LV_GRAD_DIR_RADIAL) {
162 nema_vg_paint_set_type(draw_nema_gfx_unit->paint, NEMA_VG_PAINT_GRAD_RADIAL);
163
164 nema_vg_paint_set_grad_radial(draw_nema_gfx_unit->paint, draw_nema_gfx_unit->gradient,
165 rel_coords.x1 + dsc->grad.params.radial.end.x,
166 rel_coords.y1 + dsc->grad.params.radial.end.y,
167 LV_ABS(dsc->grad.params.radial.end_extent.x - dsc->grad.params.radial.end.x),
168 extend_type | NEMA_FILTER_BL);
169 }
170
171 if(radius > 0.f)
172 nema_vg_draw_rounded_rect(rel_coords.x1, rel_coords.y1, coords_bg_w, coords_bg_h, radius, radius, NULL,
173 draw_nema_gfx_unit->paint);
174 else
175 nema_vg_draw_rect(rel_coords.x1, rel_coords.y1, coords_bg_w, coords_bg_h, NULL, draw_nema_gfx_unit->paint);
176 }
177#endif
178
179 nema_cl_submit(&(draw_nema_gfx_unit->cl));
180
181}
