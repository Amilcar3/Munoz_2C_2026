123static bool freetype_glyph_outline_create_cb(lv_freetype_outline_node_t * node, lv_freetype_font_dsc_t * dsc)
124{
125 LV_ASSERT(node != NULL);
126 LV_ASSERT(dsc != NULL);
127 LV_PROFILER_FONT_BEGIN;
128 lv_freetype_outline_t outline;
129
130 lv_mutex_lock(&dsc->cache_node->face_lock);
131 outline = outline_create(dsc->context,
132 dsc->cache_node->face,
133 node->glyph_index,
134 dsc->cache_node->ref_size,
135 (dsc->style & LV_FREETYPE_FONT_STYLE_BOLD) && !FT_HAS_MULTIPLE_MASTERS(dsc->cache_node->face) ? 1 : 0);
136 lv_mutex_unlock(&dsc->cache_node->face_lock);
137
138 if(!outline) {
139 LV_PROFILER_FONT_END;
140 return false;
141 }
142
143 LV_LOG_INFO("glyph_index = 0x%" LV_PRIx32, (uint32_t)node->glyph_index);
144
145 node->outline = outline;
146 LV_PROFILER_FONT_END;
147 return true;
148}

206static lv_cache_entry_t * lv_freetype_outline_lookup(lv_freetype_font_dsc_t * dsc, FT_UInt glyph_index)
207{
208 LV_ASSERT(dsc != NULL);
209 LV_PROFILER_FONT_BEGIN;
210 lv_freetype_cache_node_t * cache_node = dsc->cache_node;
211 LV_ASSERT(cache_node != NULL);
212
213 lv_freetype_outline_node_t tmp_node;
214 tmp_node.glyph_index = glyph_index;
215
216 lv_cache_entry_t * entry = lv_cache_acquire_or_create(cache_node->draw_data_cache, &tmp_node, dsc);
217 if(!entry) {
218 LV_LOG_ERROR("glyph outline lookup failed for glyph_index = 0x%" LV_PRIx32, (uint32_t)glyph_index);
219 LV_PROFILER_FONT_END;
220 return NULL;
221 }
222 LV_PROFILER_FONT_END;
223 return entry;
224}

326static lv_freetype_outline_t outline_create(
327 lv_freetype_context_t * ctx,
328 FT_Face face,
329 FT_UInt glyph_index,
330 uint32_t size,
331 uint32_t strength)
332{
333 LV_PROFILER_FONT_BEGIN;
334 LV_ASSERT(ctx != NULL);
335 FT_Error error;
336
337 error = FT_Set_Pixel_Sizes(face, 0, size);
338 if(error) {
339 FT_ERROR_MSG("FT_Set_Char_Size", error);
340 LV_PROFILER_FONT_END;
341 return NULL;
342 }
343
344
349 error = FT_Load_Glyph(face, glyph_index, FT_LOAD_DEFAULT | FT_LOAD_NO_BITMAP | FT_LOAD_NO_AUTOHINT);
350 if(error) {
351 FT_ERROR_MSG("FT_Load_Glyph", error);
352 LV_PROFILER_FONT_END;
353 return NULL;
354 }
355
356 if(strength > 0) {
357 error = FT_Outline_Embolden(&face->glyph->outline, FT_INT_TO_F26DOT6(strength));
358 if(error) {
359 FT_ERROR_MSG("FT_Outline_Embolden", error);
360 }
361 }
362
363
364 FT_Outline_Funcs outline_funcs = {
365 .move_to = outline_move_to_cb,
366 .line_to = outline_line_to_cb,
367 .conic_to = outline_conic_to_cb,
368 .cubic_to = outline_cubic_to_cb,
369 .shift = 0,
370 .delta = 0
371 };
372
373 lv_result_t res;
374 lv_freetype_outline_event_param_t param;
375 lv_memzero(&param, sizeof(param));
376
377 FT_Outline glyph_outline;
378 /* decompose glyph */
379 glyph_outline = face->glyph->outline;
380
381 /*Calculate Total Segments Before decompose */
382 int32_t tag_size = glyph_outline.n_points;
383 int32_t segments = 0;
384 int32_t vectors = 0;
385
386 for(int j = 0; j < tag_size; j++) {
387
388#if 0
389 if(j == 0 && (glyph_outline.tags[j] & 0x1) == 0) {
390 /* TODO handle the case where the first point is 'off curve'
391 https://stackoverflow.com/questions/3465809/how-to-interpret-a-freetype-glyph-outline-when-the-first-point-on-the-contour-is */
392 }
393#endif
394 if((glyph_outline.tags[j] & 0x1) == 0x1) {
395 segments++;
396 vectors++;
397 }
398 else {
399 int jj = j + 1 < tag_size ? j + 1 : 0;
400 if(glyph_outline.tags[jj] & 0x1) {
401 vectors++;
402 }
403 else {
404 segments++;
405 vectors += 2;
406 }
407 }
408 }
409
410 /*Also for every contour we may have a line for close*/
411 segments += glyph_outline.n_contours;
412 vectors += glyph_outline.n_contours;
413
414 param.sizes.data_size = vectors * 2;
415 param.sizes.segments_size = segments;
416
417 lv_freetype_outline_t outline;
418
419 res = outline_send_event(ctx, LV_EVENT_CREATE, &param);
420 outline = param.outline;
421
422 if(res != LV_RESULT_OK || !outline) {
423 LV_LOG_ERROR("Outline object create failed");
424 LV_PROFILER_FONT_END;
425 return NULL;
426 }
427
428 /* Run outline decompose again to fill outline data */
429 error = FT_Outline_Decompose(&glyph_outline, &outline_funcs, outline);
430 if(error) {
431 FT_ERROR_MSG("FT_Outline_Decompose", error);
432 outline_delete(ctx, outline);
433 LV_PROFILER_FONT_END;
434 return NULL;
435 }
436
437 /* Close the border glyph or the regular glyph */
438 res = outline_push_point(outline, LV_FREETYPE_OUTLINE_END, NULL, NULL, NULL);
439 if(res != LV_RESULT_OK) {
440 LV_LOG_ERROR("Outline object close failed");
441 outline_delete(ctx, outline);
442 LV_PROFILER_FONT_END;
443 return NULL;
444 }
445
446 LV_PROFILER_FONT_END;
447 return outline;
448}
