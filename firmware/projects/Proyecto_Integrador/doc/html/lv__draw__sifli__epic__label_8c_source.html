45void lv_draw_sifli_epic_label(lv_draw_task_t * task)
46{
47 const lv_draw_label_dsc_t * dsc = (const lv_draw_label_dsc_t *)task->draw_dsc;
48
49 if(dsc->opa <= LV_OPA_MIN || dsc->text == NULL || dsc->text[0] == '\0') {
50 return;
51 }
52
53#ifndef EPIC_SUPPORT_A8
54 LV_UNUSED(task);
55#else
56 lv_draw_label_iterate_characters(task, dsc, &task->area, draw_letter_cb);
57#endif
58}

64static void draw_letter_cb(lv_draw_task_t * task, lv_draw_glyph_dsc_t * glyph_draw_dsc,
65 lv_draw_fill_dsc_t * fill_draw_dsc, const lv_area_t * fill_area)
66{
67 if(glyph_draw_dsc) {
68 switch(glyph_draw_dsc->format) {
69 case LV_FONT_GLYPH_FORMAT_NONE: {
70#if LV_USE_FONT_PLACEHOLDER
71 draw_placeholder_border(task, glyph_draw_dsc);
72#endif
73 break;
74 }
75
76 case LV_FONT_GLYPH_FORMAT_A1:
77 case LV_FONT_GLYPH_FORMAT_A2:
78 case LV_FONT_GLYPH_FORMAT_A3:
79 case LV_FONT_GLYPH_FORMAT_A4:
80 case LV_FONT_GLYPH_FORMAT_A8:
81 if(glyph_draw_dsc->rotation % 3600 == 0) {
82 draw_bitmap_glyph(task, glyph_draw_dsc);
83 }
84 else {
85 draw_glyph_as_image(task, glyph_draw_dsc);
86 }
87 break;
88
89 case LV_FONT_GLYPH_FORMAT_IMAGE:
90 draw_glyph_as_image(task, glyph_draw_dsc);
91 break;
92
93 default:
94 break;
95 }
96 }
97
98 if(fill_draw_dsc && fill_area) {
99 draw_fill_part(task, fill_draw_dsc, fill_area);
100 }
101}

121static void draw_bitmap_glyph(lv_draw_task_t * task, lv_draw_glyph_dsc_t * glyph_draw_dsc)
122{
123 if(glyph_draw_dsc->opa <= LV_OPA_MIN || glyph_draw_dsc->g == NULL || glyph_draw_dsc->letter_coords == NULL) {
124 return;
125 }
126
127 glyph_draw_dsc->glyph_data = lv_font_get_glyph_bitmap(glyph_draw_dsc->g, glyph_draw_dsc->_draw_buf);
128 if(glyph_draw_dsc->glyph_data == NULL) {
129 return;
130 }
131
132 const lv_draw_buf_t * glyph_buf = (const lv_draw_buf_t *)glyph_draw_dsc->glyph_data;
133 if(glyph_buf->data == NULL) {
134 EPIC_ASSERT_MSG(false, "EPIC: Bitmap glyph buffer has no data");
135 return;
136 }
137
138 EPIC_LayerConfigTypeDef bg_layer;
139 EPIC_LayerConfigTypeDef output_layer;
140
141 if(lv_epic_setup_layers(&bg_layer, &output_layer, task, glyph_draw_dsc->letter_coords)) {
142 return;
143 }
144
145 EPIC_LayerConfigTypeDef input_layers[2];
146 input_layers[0] = bg_layer;
147 HAL_EPIC_LayerConfigInit(&input_layers[1]);
148
149 input_layers[1].data = (uint8_t *)glyph_buf->data;
150 input_layers[1].color_mode = lv_img_cf_to_epic_cf(glyph_buf->header.cf);
151 input_layers[1].width = glyph_buf->header.w;
152 input_layers[1].height = glyph_buf->header.h;
153 input_layers[1].total_width = lv_epic_stride_to_width(glyph_buf->header.stride, glyph_buf->header.cf);
154 input_layers[1].x_offset = glyph_draw_dsc->letter_coords->x1;
155 input_layers[1].y_offset = glyph_draw_dsc->letter_coords->y1;
156 input_layers[1].alpha = glyph_draw_dsc->opa;
157 input_layers[1].color_en = true;
158 input_layers[1].color_r = glyph_draw_dsc->color.red;
159 input_layers[1].color_g = glyph_draw_dsc->color.green;
160 input_layers[1].color_b = glyph_draw_dsc->color.blue;
161 input_layers[1].ax_mode = ALPHA_BLEND_RGBCOLOR;
162
163 (void)lv_epic_blend(input_layers, 2, &output_layer);
164}

166static void draw_glyph_as_image(lv_draw_task_t * task, lv_draw_glyph_dsc_t * glyph_draw_dsc)
167{
168 if(glyph_draw_dsc->opa <= LV_OPA_MIN || glyph_draw_dsc->g == NULL || glyph_draw_dsc->letter_coords == NULL) {
169 return;
170 }
171
172 glyph_draw_dsc->glyph_data = lv_font_get_glyph_bitmap(glyph_draw_dsc->g, glyph_draw_dsc->_draw_buf);
173 if(glyph_draw_dsc->glyph_data == NULL) {
174 return;
175 }
176
177 lv_draw_image_dsc_t img_dsc;
178 lv_draw_image_dsc_init(&img_dsc);
179 img_dsc.rotation = glyph_draw_dsc->rotation;
180 img_dsc.scale_x = LV_SCALE_NONE;
181 img_dsc.scale_y = LV_SCALE_NONE;
182 img_dsc.opa = glyph_draw_dsc->opa;
183 img_dsc.src = glyph_draw_dsc->glyph_data;
184 img_dsc.recolor = glyph_draw_dsc->color;
185 img_dsc.pivot.x = glyph_draw_dsc->pivot.x;
186 img_dsc.pivot.y = glyph_draw_dsc->g ? (glyph_draw_dsc->g->box_h + glyph_draw_dsc->g->ofs_y) : 0;
187
188 lv_draw_task_t img_task = *task;
189 img_task.draw_dsc = &img_dsc;
190 img_task.area = *glyph_draw_dsc->letter_coords;
191 lv_draw_sifli_epic_img(&img_task);
192}
