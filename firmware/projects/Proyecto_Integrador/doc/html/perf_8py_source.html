28def main() -> bool:
29 epilog = """This program runs LVGL performance tests
30 In order to provide timing consistency between host computers,
31 these runs are run in an ARM emulated environment inside QEMU.
32 For the runtime environment, SO3 is used which is a lightweight, ARM-based
33 operating system.
34 Right now, this script requires a host linux computer as we depend on
35 `losetup` which is used to set up and control loop devices.
36 """
37 parser = argparse.ArgumentParser(
38 description="Run LVGL performance tests.", epilog=epilog
39 )
40 parser.add_argument(
41 "--build-options",
42 nargs=1,
43 choices=perf_test_options.keys(),
44 help="""the perf test option name to run. When
45 omitted all build configurations are used.
46 """,
47 )
48 parser.add_argument("--test-suite", default=None, help="Select test suite to run")
49 parser.add_argument(
50 "--pull",
51 action="store_true",
52 default=False,
53 help="Pull latest images from registry before running tests",
54 )
55 parser.add_argument(
56 "--clean",
57 action="store_true",
58 default=False,
59 help="Clean existing build artifacts before operation",
60 )
61 parser.add_argument(
62 "--auto-clean",
63 action="store_true",
64 default=False,
65 help="Automatically clean build directories",
66 )
67 parser.add_argument(
68 "--debug",
69 action="store_true",
70 default=False,
71 help="Launch QEMU in debug mode allowing you to connect to gdb",
72 )
73 parser.add_argument(
74 "--debug-port",
75 default=5555,
76 help="Port used to connect to QEMU in debug mode",
77 )
78 parser.add_argument(
79 "actions",
80 nargs="*",
81 choices=["build", "generate", "test"],
82 help="build | generate: generates run files dependencies, test: run performance tests.",
83 )
84
85 args = parser.parse_args()
86
87 options = []
88 if args.build_options:
89 options = args.build_options
90 else:
91 options = perf_test_options.keys()
92
93 # Since test suites are run inside a virtual space, debugging them is tricky
94 # because every test suite will be loaded into memory at the same virtual address
95 # this means that when you try to set a breakpoint (e.g: in `main`), gdb will actually
96 # hit that breakpoint for EACH test_suite that runs, which can be very confusing
97 # Instead, only allow debugging if a test_suite is selected as we won't encounter that problem
98 if args.debug and not args.test_suite:
99 print("Please provide the test suite you want to debug")
100 exit(1)
101
102 is_error = False
103 for option_name in options:
104 if any(action in args.actions for action in ("generate", "build", "test")):
105 if args.clean:
106 clean(option_name)
107 generate_files(option_name, args.test_suite)
108
109 if "test" in args.actions:
110 ret = run_tests(
111 option_name,
112 "lv_test_perf_conf.h",
113 args.pull,
114 args.debug,
115 args.debug_port,
116 args.test_suite,
117 )
118 is_error = is_error or not ret
119
120 if args.auto_clean:
121 clean(option_name)
122

243def generate_so3_init_commands(runners: list[tuple[str, str]], path: str) -> None:
244 """
245 Generates the `commands.ini` file that will be mounted in `usr/out/commands.ini` replacing
246 the default `commands.ini` used by so3.
247 A `commands.ini` file declares some specific commands the `init` program should run
248 sequentially. We use this feature to run every perf test.
249 An example that runs tests A and B before exiting looks like this :
250 ```
251 run test_A.elf
252 run test_B.elf
253 exit
254 ```
255 """
256
257 output = []
258 for runner, _ in runners:
259 name_without_extension = Path(runner).stem
260 label = f"Running {name_without_extension}"
261 delimiter = "=" * len(label)
262 output.append(f"echo {delimiter}")
263 output.append(f"echo {label}")
264 output.append(f"echo {delimiter}")
265 output.append(f"run {name_without_extension}.elf")
266
267 output.append("exit")
268 write_lines_to_file(path, output)

272 """
273 Generates the CMakeLists.txt that will be mounted in `usr/src/test_src/CMakeLists.txt`

363def generate_test_runners(
364 output_folder: str, test_suite: Optional[str]
365) -> list[tuple[str, str]]:
366 """
367 Generates performance test runners by creating a runner for each `.c` file
368 located in the `test_cases/perf` directory.
369
370 If `test_suite` is provided, only `.c` files matching the suite will be used.
371 Otherwise, all `.c` files in the directory are included.
372
373 Each generated runner, along with its corresponding `.c` file, will be saved
374 in the specified `output_folder`.
375 """
376
377 runner_generator_script = os.path.join(
378 lvgl_test_dir, "unity", "generate_test_runner.rb"
379 )
380
381 # Get the necessary files in order to generate the runners
382 # This includes the test cases and the unity config
383 test_cases = find_c_files(os.path.join(lvgl_test_dir, "src", "test_cases_perf"))
384 unity_config_path = os.path.join(lvgl_test_dir, "config.yml")
385
386 runners = []
387
388 print("Generating test case runners")
389
390 for src_test_case_path in test_cases:
391 current_test_suite = Path(src_test_case_path).stem
392
393 if test_suite and test_suite not in current_test_suite:
394 continue
395
396 print(f"\t{current_test_suite}")
397
398 runner_file_name = current_test_suite + "_runner.c"
399 test_case_file_name = os.path.basename(src_test_case_path)
400
401 test_case_path = os.path.join(output_folder, test_case_file_name)
402 runner_path = os.path.join(output_folder, runner_file_name)
403
404 subprocess.check_call(
405 [
406 runner_generator_script,
407 src_test_case_path,
408 runner_path,
409 unity_config_path,
410 ]
411 )
412
413 # Copy the original test case as still need them in the build process
414 shutil.copy(src_test_case_path, test_case_path)
415 # Store a tuple of runner - test case so we can generate the cmakelists later
416 runners.append((runner_file_name, test_case_file_name))
417
418 return runners

474 There's no support for return codes when running qemu so we manually check the docker
475 logs to see if any case failed
476 """
477 stdout = subprocess.getoutput(f"docker logs {container_name}")

486def run_tests(
487 options_name: str,
488 lv_conf_name: str,
489 pull: bool,
490 debug: bool,
491 debug_port: int,
492 test_suite: Optional[str],
493) -> bool:
494 """
495 Runs the tests by running the docker image associated with `options_name`
496 while mounting the correct volumes from the previous generated files
497 """
498
499 def volume(src, dst):
500 return ["-v", f"{src}:{dst}"]
501
502 # The Infrabase images nest the repo at /so3, so the user space lives at
503 # /so3/so3/usr (kernel at /so3/so3/so3). All mount targets use that path.
504 def so3_usr_src(path):
505 return f"/so3/so3/usr/src/{path}"
506
507 def so3_usr_lib(path):
508 return f"/so3/so3/usr/lib/{path}"
509
510 def so3_usr_out(path):
511 return f"/so3/so3/usr/out/{path}"
512
513 so3_usr_build = f"/so3/so3/usr/build"
514 persistence_dir = f"/persistence"
515 container_name = get_container_name(options_name)
516 build_dir = get_build_dir(options_name)
517
518 for v in get_docker_volumes(options_name):
519 subprocess.check_call(["docker", "volume", "create", v])
520
521 unity_dir = os.path.join(build_dir, "unity")
522 test_src_dir = os.path.join(build_dir, "test_src")
523 main_cmakelists = os.path.join(build_dir, "CMakeLists.txt")
524 lvgl_src_dir = os.path.join(lvgl_test_dir, "..", "src")
525 lvgl_include_dir = os.path.join(lvgl_test_dir, "..", "include")
526 lv_conf_path = os.path.join(lvgl_test_dir, "src", lv_conf_name)
527 lvgl_h_path = os.path.join(lvgl_test_dir, "..", "lvgl.h")
528 lvgl_private_h_path = os.path.join(lvgl_test_dir, "..", "lvgl_private.h")
529 commands_ini_path = os.path.join(build_dir, "commands.ini")
530 docker_image_name = perf_test_options[options_name]["image_name"]
531
532 volumes = [
533 # This is necessary in order to create a loop device
534 # It is also the reason we only support linux for now.
535 volume("/dev", "/dev"),
536 # Replace container's lvgl source and lv_conf
537 volume(lvgl_include_dir, so3_usr_lib("lvgl/include")),
538 volume(lvgl_src_dir, so3_usr_lib("lvgl/src")),
539 volume(lv_conf_path, so3_usr_lib("lv_conf.h")),
540 volume(lvgl_h_path, so3_usr_lib("lvgl/lvgl.h")),
541 volume(lvgl_private_h_path, so3_usr_lib("lvgl/lvgl_private.h")),
542 # We also need to add the current "lvgl.h" and mount it in the correct path
543 # As there's a `#include "../../lvgl.h"` in the `unity_support.h` file
544 volume(lvgl_h_path, "/so3/so3/usr/lvgl.h"),
545 # Mount the test sources (test cases and runners)
546 volume(test_src_dir, so3_usr_src("test_src")),
547 # Mount the test framework
548 volume(unity_dir, so3_usr_src("unity")),
549 # Modify the default so3 CMakeLists and commands.ini
550 volume(main_cmakelists, so3_usr_src("CMakeLists.txt")),
551 volume(commands_ini_path, so3_usr_out("commands.ini")),
552 # Cache build and disk folders so we don't regenerate everything in consecutive runs
553 volume(get_build_cache_volume(options_name), so3_usr_build),
554 volume(get_disk_cache_volume(options_name), persistence_dir),
555 ]
556
557 interactive = "-it" if sys.stdout.isatty() else "-t"
558 command = ["docker", "run", "--privileged", "--name", container_name, interactive]
559
560 print()
561 print()
562 label = "Launching: %s: %s" % (
563 options_abbrev(options_name),
564 perf_test_options[options_name]["description"],
565 )
566 print("=" * len(label))
567 print(label)
568 print("=" * len(label))
569
570 if pull:
571 command.append("--pull=always")
572
573 if debug:
574 command.append("-e")
575 command.append("SO3_USR_DEBUG=1")
576 command.append("-p")
577 command.append(f"{debug_port}:1234")
578
579 for v in volumes:
580 command.extend(v)
581
582 command.append(docker_image_name)
583
584 try:
585 subprocess.check_call(command)
586 except subprocess.CalledProcessError:
587 subprocess.check_call(["docker", "rm", "-f", container_name])
588 return False
589 success = check_for_success(container_name)
590
591 # We can't use the `docker run --rm` syntax because we need access to the docker container
592 # after it exits in order to check for the success status of the run
593 subprocess.check_call(["docker", "rm", "-f", container_name])
594 return success
