249static void draw_event_cb(lv_event_t * e)
250{
251 lv_obj_t * obj = lv_event_get_target(e);
252 lv_draw_task_t * draw_task = lv_event_get_draw_task(e);
253 lv_draw_dsc_base_t * base_dsc = lv_draw_task_get_draw_dsc(draw_task);
254 lv_draw_label_dsc_t * label_draw_dsc = lv_draw_task_get_label_dsc(draw_task);
255 lv_draw_line_dsc_t * line_draw_dsc = lv_draw_task_get_line_dsc(draw_task);
256 if(label_draw_dsc) {
257 if(base_dsc->part == LV_PART_INDICATOR) {
258 const lv_color_t color_idx[7] = {
259 lv_palette_main(LV_PALETTE_RED),
260 lv_palette_main(LV_PALETTE_ORANGE),
261 lv_palette_main(LV_PALETTE_YELLOW),
262 lv_palette_main(LV_PALETTE_GREEN),
263 lv_palette_main(LV_PALETTE_CYAN),
264 lv_palette_main(LV_PALETTE_BLUE),
265 lv_palette_main(LV_PALETTE_PURPLE),
266 };
267 uint8_t major_tick = lv_scale_get_major_tick_every(obj);
268 label_draw_dsc->color = color_idx[base_dsc->id1 / major_tick];
269
270 /*Free the previously allocated text if needed*/
271 if(label_draw_dsc->text_local) lv_free((void *)label_draw_dsc->text);
272
273 /*Malloc the text and set text_local as 1 to make LVGL automatically free the text.
274 * (Local texts are malloc'd internally by LVGL. Mimic this behavior here too)*/
275 char tmp_buffer[20] = {0}; /* Big enough buffer */
276 lv_snprintf(tmp_buffer, sizeof(tmp_buffer), "%.1f", base_dsc->id2 * 1.0f);
277 label_draw_dsc->text = lv_strdup(tmp_buffer);
278 label_draw_dsc->text_local = 1;
279
280 lv_point_t size;
281 lv_text_attributes_t attributes = {0};
282 attributes.letter_space = 0;
283 attributes.line_space = 0;
284 attributes.max_width = 1000;
285 attributes.text_flags = LV_TEXT_FLAG_NONE;
286
287 lv_text_get_size_attributes(&size, label_draw_dsc->text, label_draw_dsc->font, &attributes);
288 int32_t new_w = size.x;
289 int32_t old_w = lv_area_get_width(&draw_task->area);
290
291 /* Distribute the new size equally on both sides */
292 draw_task->area.x1 -= (new_w - old_w) / 2;
293 draw_task->area.x2 += ((new_w - old_w) + 1) / 2; /* +1 for rounding */
294
295 }
296 }
297 else if(line_draw_dsc) {
298 if(base_dsc->part == LV_PART_INDICATOR || base_dsc->part == LV_PART_ITEMS) {
299 const lv_color_t color_idx[7] = {
300 lv_palette_main(LV_PALETTE_RED),
301 lv_palette_main(LV_PALETTE_ORANGE),
302 lv_palette_main(LV_PALETTE_YELLOW),
303 lv_palette_main(LV_PALETTE_GREEN),
304 lv_palette_main(LV_PALETTE_CYAN),
305 lv_palette_main(LV_PALETTE_BLUE),
306 lv_palette_main(LV_PALETTE_PURPLE),
307 };
308 uint32_t tick_idx = base_dsc->id1;
309 uint32_t tick_value = base_dsc->id2;
310
311 line_draw_dsc->color = color_idx[tick_idx % 7];
312
313 int32_t expected_tick_value = lv_map(tick_idx, 0, 31 - 1, 10, 40);
314 TEST_ASSERT_EQUAL(tick_value, expected_tick_value);
315 }
316 }
317}
