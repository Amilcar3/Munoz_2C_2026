8/* Subjects are owned by LVGL, so every case registers the ones it creates here

12static lv_subject_t * subject_create(lv_subject_type_t type)
13{
14 for(size_t i = 0; i < LV_ARRAYLEN(subjects); ++i) {
15 if(subjects[i] == NULL) {
16 subjects[i] = lv_subject_create(type);
17 TEST_ASSERT_NOT_NULL(subjects[i]);
18 return subjects[i];
19 }
20 }
21 TEST_FAIL_MESSAGE("subject pool exhausted");
22 return NULL;
23}

41 /* Group Subjects first: deleting a group dereferences its elements, so the
42 * elements have to outlive it. */
43 for(size_t i = 0; i < LV_ARRAYLEN(subjects); ++i) {

48 }

50 lv_subject_delete(subjects[i]);
