75void lv_draw_fill(lv_layer_t * layer, const lv_draw_fill_dsc_t * dsc, const lv_area_t * coords)
76{
77 LV_CHECK_ARG(layer != NULL, return);
78 LV_CHECK_ARG(dsc != NULL, return);
79 LV_CHECK_ARG(coords != NULL, return);
80
81 if(dsc->opa <= LV_OPA_MIN) return;
82
83 LV_PROFILER_DRAW_BEGIN;
84
85 if(dsc->base.drop_shadow_opa) {
86 lv_layer_t * ds_layer = lv_draw_layer_create_drop_shadow(layer, &dsc->base, coords);
87 LV_ASSERT_NULL(ds_layer);
88 lv_draw_fill_dsc_t ds_dsc = *dsc;
89 ds_dsc.base.drop_shadow_opa = 0; /*Disable drop shadow so rendering below will render plain fill*/
90 lv_draw_fill(ds_layer, &ds_dsc, coords);
91 lv_draw_layer_finish_drop_shadow(ds_layer, &dsc->base);
92 }
93
94 lv_draw_task_t * t = lv_draw_add_task(layer, coords, LV_DRAW_TASK_TYPE_FILL);
95
96 lv_memcpy(t->draw_dsc, dsc, sizeof(*dsc));
97
98 lv_draw_finalize_task_creation(layer, t);
99 LV_PROFILER_DRAW_END;

179void lv_draw_rect(lv_layer_t * layer, const lv_draw_rect_dsc_t * dsc, const lv_area_t * coords)
180{
181 LV_CHECK_ARG(layer != NULL, return);
182 LV_CHECK_ARG(dsc != NULL, return);
183 LV_CHECK_ARG(coords != NULL, return);
184
185 LV_PROFILER_DRAW_BEGIN;
186
187 bool has_shadow;
188 bool has_fill;
189 bool has_border;
190 bool has_outline;
191 bool has_bg_img;
192
193 if(dsc->shadow_width == 0 ||
194 dsc->shadow_opa <= LV_OPA_MIN ||
195 (dsc->shadow_width == 1 && dsc->shadow_spread <= 0 &&
196 dsc->shadow_offset_x == 0 && dsc->shadow_offset_y == 0)) {
197 has_shadow = false;
198 }
199 else {
200 has_shadow = true;
201 }
202
203 if(dsc->bg_opa <= LV_OPA_MIN) has_fill = false;
204 else has_fill = true;
205
206 if(dsc->bg_image_opa <= LV_OPA_MIN || dsc->bg_image_src == NULL) has_bg_img = false;
207 else has_bg_img = true;
208
209 if(dsc->border_opa <= LV_OPA_MIN
210 || dsc->border_width == 0
211 || dsc->border_post == true
212 || dsc->border_side == LV_BORDER_SIDE_NONE) has_border = false;
213 else has_border = true;
214
215 if(dsc->outline_opa <= LV_OPA_MIN || dsc->outline_width == 0) has_outline = false;
216 else has_outline = true;
217
218 bool bg_cover = true;
219 if(dsc->bg_opa < LV_OPA_COVER) bg_cover = false;
220 else if(dsc->bg_grad.dir != LV_GRAD_DIR_NONE) {
221 uint32_t s;
222 for(s = 0; s < dsc->bg_grad.stops_count; s++) {
223 if(dsc->bg_grad.stops[s].opa != LV_OPA_COVER) {
224 bg_cover = false;
225 break;
226 }
227 }
228 }
229
230 if(dsc->base.drop_shadow_opa && (has_fill || has_outline)) {
231 lv_layer_t * ds_layer = lv_draw_layer_create_drop_shadow(layer, &dsc->base, coords);
232 LV_ASSERT_NULL(ds_layer);
233 lv_draw_rect_dsc_t ds_dsc = *dsc;
234 ds_dsc.base.drop_shadow_opa = 0; /*Disable drop shadow so rendering below will render plain shadow*/
235 ds_dsc.shadow_opa = 0;
236 lv_draw_rect(ds_layer, &ds_dsc, coords);
237 lv_draw_layer_finish_drop_shadow(ds_layer, &dsc->base);
238 }
239
240 lv_draw_task_t * t;
241
242 /*Shadow*/
243 if(has_shadow) {
244 /*Check whether the shadow is visible*/
245 t = lv_draw_add_task(layer, coords, LV_DRAW_TASK_TYPE_BOX_SHADOW);
246 lv_draw_box_shadow_dsc_t * shadow_dsc = t->draw_dsc;
247
248 lv_area_increase(&t->_real_area, dsc->shadow_spread, dsc->shadow_spread);
249 lv_area_increase(&t->_real_area, dsc->shadow_width, dsc->shadow_width);
250 lv_area_move(&t->_real_area, dsc->shadow_offset_x, dsc->shadow_offset_y);
251 shadow_dsc->base = dsc->base;
252 shadow_dsc->base.dsc_size = sizeof(lv_draw_box_shadow_dsc_t);
253 shadow_dsc->radius = dsc->radius;
254 shadow_dsc->color = dsc->shadow_color;
255 shadow_dsc->width = dsc->shadow_width;
256 shadow_dsc->spread = dsc->shadow_spread;
257 shadow_dsc->opa = dsc->shadow_opa;
258 shadow_dsc->ofs_x = dsc->shadow_offset_x;
259 shadow_dsc->ofs_y = dsc->shadow_offset_y;
260 shadow_dsc->bg_cover = bg_cover;
261 lv_draw_finalize_task_creation(layer, t);
262 }
263
264 /*Background*/
265 if(has_fill) {
266 lv_area_t bg_coords = *coords;
267 /*If the border fully covers make the bg area 1px smaller to avoid artifacts on the corners*/
268 if(dsc->border_width > 1 && dsc->border_opa >= LV_OPA_MAX && dsc->radius != 0) {
269 bg_coords.x1 += (dsc->border_side & LV_BORDER_SIDE_LEFT) ? 1 : 0;
270 bg_coords.y1 += (dsc->border_side & LV_BORDER_SIDE_TOP) ? 1 : 0;
271 bg_coords.x2 -= (dsc->border_side & LV_BORDER_SIDE_RIGHT) ? 1 : 0;
272 bg_coords.y2 -= (dsc->border_side & LV_BORDER_SIDE_BOTTOM) ? 1 : 0;
273 }
274
275 t = lv_draw_add_task(layer, &bg_coords, LV_DRAW_TASK_TYPE_FILL);
276 lv_draw_fill_dsc_t * bg_dsc = t->draw_dsc;
277
278 lv_draw_fill_dsc_init(bg_dsc);
279 bg_dsc->base = dsc->base;
280 bg_dsc->base.dsc_size = sizeof(lv_draw_fill_dsc_t);
281 bg_dsc->radius = dsc->radius;
282 bg_dsc->color = dsc->bg_color;
283 bg_dsc->grad = dsc->bg_grad;
284 bg_dsc->opa = dsc->bg_opa;
285
286 lv_draw_finalize_task_creation(layer, t);
287 }
288
289 /*Background image*/
290 if(has_bg_img) {
291 lv_image_src_t src_type = lv_image_src_get_type(dsc->bg_image_src);
292 lv_result_t res = LV_RESULT_OK;
293 lv_image_header_t header;
294 if(src_type == LV_IMAGE_SRC_VARIABLE || src_type == LV_IMAGE_SRC_FILE) {
295 res = lv_image_decoder_get_info(dsc->bg_image_src, &header);
296 }
297 else {
298 lv_memzero(&header, sizeof(header));
299
300 if(src_type == LV_IMAGE_SRC_UNKNOWN) {
301 res = LV_RESULT_INVALID;
302 }
303 }
304
305 if(res == LV_RESULT_OK) {
306 if(src_type == LV_IMAGE_SRC_VARIABLE || src_type == LV_IMAGE_SRC_FILE) {
307
308 if(dsc->bg_image_tiled) {
309 t = lv_draw_add_task(layer, coords, LV_DRAW_TASK_TYPE_IMAGE);
310 }
311 else {
312 lv_area_t a = {0, 0, header.w - 1, header.h - 1};
313 lv_area_align(coords, &a, LV_ALIGN_CENTER, 0, 0);
314 t = lv_draw_add_task(layer, &a, LV_DRAW_TASK_TYPE_IMAGE);
315 }
316
317 lv_draw_image_dsc_t * bg_image_dsc = t->draw_dsc;
318
319 lv_draw_image_dsc_init(bg_image_dsc);
320 bg_image_dsc->base = dsc->base;
321 bg_image_dsc->base.dsc_size = sizeof(lv_draw_image_dsc_t);
322 bg_image_dsc->src = dsc->bg_image_src;
323 bg_image_dsc->opa = dsc->bg_image_opa;
324 bg_image_dsc->recolor = dsc->bg_image_recolor;
325 bg_image_dsc->recolor_opa = dsc->bg_image_recolor_opa;
326 bg_image_dsc->tile = dsc->bg_image_tiled;
327 bg_image_dsc->colorkey = dsc->bg_image_colorkey;
328 bg_image_dsc->header = header;
329 bg_image_dsc->clip_radius = dsc->radius;
330 bg_image_dsc->image_area = *coords;
331 lv_draw_finalize_task_creation(layer, t);
332 }
333 else {
334 lv_point_t s;
335
336 lv_text_attributes_t attributes = {0};
337 attributes.text_flags = LV_TEXT_FLAG_NONE;
338 attributes.max_width = LV_COORD_MAX;
339 attributes.line_space = 0;
340 attributes.letter_space = 0;
341
342 lv_text_get_size_attributes(&s, dsc->bg_image_src, dsc->bg_image_symbol_font, &attributes);
343
344 lv_area_t a = {0, 0, s.x - 1, s.y - 1};
345 lv_area_align(coords, &a, LV_ALIGN_CENTER, 0, 0);
346 t = lv_draw_add_task(layer, &a, LV_DRAW_TASK_TYPE_LABEL);
347
348 lv_draw_label_dsc_t * bg_label_dsc = t->draw_dsc;
349
350 lv_draw_label_dsc_init(bg_label_dsc);
351 bg_label_dsc->base = dsc->base;
352 bg_label_dsc->base.dsc_size = sizeof(lv_draw_label_dsc_t);
353 bg_label_dsc->color = dsc->bg_image_recolor;
354 bg_label_dsc->font = dsc->bg_image_symbol_font;
355 bg_label_dsc->text = dsc->bg_image_src;
356 t->type = LV_DRAW_TASK_TYPE_LABEL;
357 lv_draw_finalize_task_creation(layer, t);
358 }
359 }
360 }
361
362 /*Border*/
363 if(has_border) {
364 t = lv_draw_add_task(layer, coords, LV_DRAW_TASK_TYPE_BORDER);
365 lv_draw_border_dsc_t * border_dsc = t->draw_dsc;
366
367 border_dsc->base = dsc->base;
368 border_dsc->base.dsc_size = sizeof(lv_draw_border_dsc_t);
369 border_dsc->radius = dsc->radius;
370 border_dsc->color = dsc->border_color;
371 border_dsc->opa = dsc->border_opa;
372 border_dsc->width = dsc->border_width;
373 border_dsc->side = dsc->border_side;
374 lv_draw_finalize_task_creation(layer, t);
375 }
376
377 /*Outline*/
378 if(has_outline) {
379 lv_area_t outline_coords = *coords;
380 lv_area_increase(&outline_coords, dsc->outline_width + dsc->outline_pad, dsc->outline_width + dsc->outline_pad);
381 t = lv_draw_add_task(layer, &outline_coords, LV_DRAW_TASK_TYPE_BORDER);
382 lv_draw_border_dsc_t * outline_dsc = t->draw_dsc;
383 lv_area_increase(&t->_real_area, dsc->outline_width, dsc->outline_width);
384 lv_area_increase(&t->_real_area, dsc->outline_pad, dsc->outline_pad);
385 outline_dsc->base = dsc->base;
386 outline_dsc->base.dsc_size = sizeof(lv_draw_border_dsc_t);
387 outline_dsc->radius = dsc->radius == LV_RADIUS_CIRCLE ? LV_RADIUS_CIRCLE : dsc->radius + dsc->outline_width +
388 dsc->outline_pad;
389 outline_dsc->color = dsc->outline_color;
390 outline_dsc->opa = dsc->outline_opa;
391 outline_dsc->width = dsc->outline_width;
392 outline_dsc->side = LV_BORDER_SIDE_FULL;
393 lv_draw_finalize_task_creation(layer, t);
394 }
395
396 LV_ASSERT_MEM_INTEGRITY();
397
398 LV_PROFILER_DRAW_END;
399}
