66lv_display_t * lv_uefi_display_create(void * handle)
67{
68 LV_CHECK_ARG(handle != NULL, return NULL);
69 lv_display_t * display = NULL;
70 lv_uefi_display_context_t * display_ctx;
71
72 if(!lv_uefi_protocol_test(handle, &_uefi_guid_graphics_output)) return NULL;
73
74 display_ctx = lv_calloc(1, sizeof(lv_uefi_display_context_t));
75 LV_ASSERT_MALLOC(display_ctx);
76
77 display_ctx->handle = handle;
78 display_ctx->gop_protocol = (EFI_GRAPHICS_OUTPUT_PROTOCOL *)lv_uefi_protocol_open(handle, &_uefi_guid_graphics_output);
79 if(!_display_interface_is_valid(display_ctx->gop_protocol)) {
80 LV_LOG_WARN("[lv_uefi] The GOP interface is not valid.");
81 goto error;
82 }
83
84 // 4 bytes per pixel
85 display_ctx->buffer_size = 4 * display_ctx->gop_protocol->Mode->Info->HorizontalResolution *
86 display_ctx->gop_protocol->Mode->Info->VerticalResolution;
87 display_ctx->buffer = lv_malloc(display_ctx->buffer_size);
88 LV_ASSERT_MALLOC(display_ctx->buffer);
89
90 display = lv_display_create(display_ctx->gop_protocol->Mode->Info->HorizontalResolution,
91 display_ctx->gop_protocol->Mode->Info->VerticalResolution);
92 lv_display_add_event_cb(display, _display_event_cb, LV_EVENT_DELETE, display);
93 lv_display_set_flush_cb(display, _display_flush_cb);
94 lv_display_set_buffers(display, display_ctx->buffer, NULL, (uint32_t)display_ctx->buffer_size,
95 LV_DISPLAY_RENDER_MODE_DIRECT);
96 lv_display_set_user_data(display, display_ctx);
97
98 goto finish;
99
100error:
101 if(display != NULL) {
102 lv_display_set_user_data(display, NULL);
103 lv_display_delete(display);
104 display = NULL;
105 }
106
107 if(display_ctx != NULL) _display_ctx_free(display_ctx);
108
109finish:
110 return display;
111}

195

197

199 if(display == NULL) return;
200

202 lv_display_set_user_data(display, NULL);
203
204 if(display_ctx != NULL) _display_ctx_free(display_ctx);

207static void _display_flush_cb(lv_display_t * display, const lv_area_t * area, uint8_t * px_map)
208{
209 EFI_STATUS status;
210 int32_t w;
211 int32_t h;
212
213 lv_uefi_display_context_t * display_ctx = (lv_uefi_display_context_t *)lv_display_get_user_data(display);
214 LV_ASSERT(display_ctx != NULL);
215
216 w = (int32_t)area->x2 - (int32_t)area->x1 + 1;
217 h = (int32_t)area->y2 - (int32_t)area->y1 + 1;
218
219 if(w < 0 || h < 0) {
220 LV_LOG_ERROR("[lv_uefi] Invalid lv_display_flush_cb call (invalid rect).");
221 goto error;
222 }
223
224 if((uint32_t)(area->x1 + w) > display_ctx->gop_protocol->Mode->Info->HorizontalResolution) {
225 LV_LOG_ERROR("[lv_uefi] Invalid lv_display_flush_cb call (invalid width).");
226 goto error;
227 }
228
229 if((uint32_t)(area->y1 + h) > display_ctx->gop_protocol->Mode->Info->VerticalResolution) {
230 LV_LOG_ERROR("[lv_uefi] Invalid lv_display_flush_cb call (invalid height).");
231 goto error;
232 }
233
234 status = display_ctx->gop_protocol->Blt(
235 display_ctx->gop_protocol,
236 (EFI_GRAPHICS_OUTPUT_BLT_PIXEL *)px_map,
237 EfiBltBufferToVideo,
238 area->x1,
239 area->y1,
240 area->x1,
241 area->y1,
242 w,
243 h,
244 display_ctx->gop_protocol->Mode->Info->HorizontalResolution * sizeof(EFI_GRAPHICS_OUTPUT_BLT_PIXEL));
245 if(status != EFI_SUCCESS) {
246 LV_LOG_ERROR("[lv_uefi] Blt failed with error code: %llx.", status);
247 goto error;
248 }
249
250 goto finish;
251
252error:
253
254finish:
255 lv_display_flush_ready(display);
256}

270static bool _display_interface_is_valid(const EFI_GRAPHICS_OUTPUT_PROTOCOL * interface)
271{
272 if(interface == NULL) return FALSE;
273 if(interface->Mode == NULL) return FALSE;
274 if(interface->Mode->Info == NULL) return FALSE;
275 if(interface->Mode->Info->HorizontalResolution == 0) return FALSE;
276 if(interface->Mode->Info->HorizontalResolution >= 32767) return FALSE;
277 if(interface->Mode->Info->VerticalResolution == 0) return FALSE;
278 if(interface->Mode->Info->VerticalResolution >= 32767) return FALSE;
279
280 return TRUE;
281}
