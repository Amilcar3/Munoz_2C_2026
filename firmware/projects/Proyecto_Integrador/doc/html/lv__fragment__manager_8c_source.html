85{

88 lv_fragment_managed_states_t * states = NULL;

115void lv_fragment_manager_remove(lv_fragment_manager_t * manager, lv_fragment_t * fragment)
116{
117 LV_ASSERT_NULL(manager);
118 LV_ASSERT_NULL(fragment);
119 LV_ASSERT_NULL(fragment->managed);
120 LV_ASSERT(fragment->managed->manager == manager);
121 lv_fragment_managed_states_t * states = fragment->managed;
122 lv_fragment_managed_states_t * prev = NULL;
123 bool was_top = false;
124 if(states->in_stack) {
125 void * stack_top = lv_ll_get_tail(&manager->stack);
126 lv_fragment_stack_item_t * item = NULL;
127 LV_LL_READ_BACK(&manager->stack, item) {
128 if(item->states == states) {
129 was_top = stack_top == item;
130 void * stack_prev = lv_ll_get_prev(&manager->stack, item);
131 if(!stack_prev) break;
132 prev = ((lv_fragment_stack_item_t *) stack_prev)->states;
133 break;
134 }
135 }
136 if(item) {
137 lv_ll_remove(&manager->stack, item);
138 lv_free(item);
139 }
140 }
141 item_delete_obj(states);
142 item_delete_fragment(states);
143 lv_ll_remove(&manager->attached, states);
144 lv_free(states);
145 if(prev && was_top) {
146 item_create_obj(prev);
147 }
148}

150void lv_fragment_manager_push(lv_fragment_manager_t * manager, lv_fragment_t * fragment, lv_obj_t * const * container)
151{
152 lv_fragment_stack_item_t * top = lv_ll_get_tail(&manager->stack);
153 if(top != NULL) {
154 item_delete_obj(top->states);
155 }
156 lv_fragment_managed_states_t * states = fragment_attach(manager, fragment, container);
157 states->in_stack = true;
158 /*Add fragment to the top of the stack*/
159 lv_fragment_stack_item_t * item = lv_ll_ins_tail(&manager->stack);
160 lv_memzero(item, sizeof(lv_fragment_stack_item_t));
161 item->states = states;
162 item_create_obj(states);
163}

183bool lv_fragment_manager_send_event(lv_fragment_manager_t * manager, int code, void * userdata)
184{
185 LV_ASSERT_NULL(manager);
186 lv_fragment_managed_states_t * p = NULL;
187 LV_LL_READ_BACK(&manager->attached, p) {
188 if(!p->obj_created || p->destroying_obj) continue;
189 lv_fragment_t * instance = p->instance;
190 if(!instance) continue;
191 if(lv_fragment_manager_send_event(instance->child_manager, code, userdata)) return true;
192 if(p->cls->event_cb && p->cls->event_cb(instance, code, userdata)) return true;
193 }
194 return false;
195}

246static void item_delete_fragment(lv_fragment_managed_states_t * item)
247{
248 lv_fragment_t * instance = item->instance;
249 if(instance->cls->detached_cb) {
250 instance->cls->detached_cb(instance);

252 instance->managed = NULL;
253 lv_fragment_delete(instance);

257static lv_fragment_managed_states_t * fragment_attach(lv_fragment_manager_t * manager, lv_fragment_t * fragment,
258 lv_obj_t * const * container)
259{
260 LV_ASSERT(manager);
261 LV_ASSERT(fragment);
262 LV_ASSERT(fragment->managed == NULL);
263 lv_fragment_managed_states_t * states = lv_ll_ins_tail(&manager->attached);
264 lv_memzero(states, sizeof(lv_fragment_managed_states_t));
265 states->cls = fragment->cls;
266 states->manager = manager;
267 states->container = container;
268 states->instance = fragment;
269 fragment->managed = states;
270 if(fragment->cls->attached_cb) {
271 fragment->cls->attached_cb(fragment);
272 }
273 return states;
274}
