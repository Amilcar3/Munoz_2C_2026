68void lv_font_manager_recycle_delete(lv_font_manager_recycle_t * manager)

116void lv_font_manager_recycle_set_reuse(lv_font_manager_recycle_t * manager, lv_font_t * font,
117 const lv_font_info_t * ft_info)
118{
119 LV_ASSERT_NULL(manager);
120 LV_ASSERT_NULL(ft_info);
121
122 lv_ll_t * recycle_ll = &manager->recycle_ll;
123
124 /* check recycled size */
125 if(lv_ll_get_len(recycle_ll) >= manager->max_size) {
126 LV_LOG_INFO("recycle full, remove tail font...");
127 lv_font_manager_recycle_remove_tail(manager);
128 }
129
130 /* record reuse font */
131 lv_font_recycle_t * recycle = lv_ll_ins_head(recycle_ll);
132 LV_ASSERT_MALLOC(recycle);
133 lv_memzero(recycle, sizeof(lv_font_recycle_t));
134
135 lv_strncpy(recycle->name, ft_info->name, sizeof(recycle->name));
136 recycle->name[sizeof(recycle->name) - 1] = '\0';
137
138 recycle->font = font;
139 recycle->ft_info = *ft_info;
140 recycle->ft_info.name = recycle->name;
141
142 LV_LOG_INFO("insert font: %s(%d) to reuse list", ft_info->name, ft_info->size);
143}

145void lv_font_recycle_remove_fonts(lv_font_manager_recycle_t * manager, const char * name)
146{
147 LV_ASSERT_NULL(manager);
148 LV_ASSERT_NULL(name);
149
150 lv_font_recycle_t * next;
151 lv_font_recycle_t * recycle = lv_ll_get_head(&manager->recycle_ll);
152 while(recycle != NULL) {
153 next = lv_ll_get_next(&manager->recycle_ll, recycle);
154 if(lv_strcmp(recycle->name, name) == 0) {
155 lv_font_recycle_close(manager, recycle);
156 }
157 recycle = next;
158 }
159}
