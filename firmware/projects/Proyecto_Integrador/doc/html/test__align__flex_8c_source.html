316void test_col_grow_size_content(void)
317{
318 lv_obj_t * cont = lv_obj_create(lv_screen_active());
319 lv_obj_set_name(cont, "cont");
320 lv_obj_set_size(cont, LV_PCT(100), LV_SIZE_CONTENT);
321 lv_obj_set_flex_flow(cont, LV_FLEX_FLOW_COLUMN);
322 lv_obj_set_style_bg_color(cont, lv_color_hex(0xff0000), 0);
323 lv_obj_set_style_bg_opa(cont, LV_OPA_COVER, 0);
324
325 const int32_t pad_all = lv_obj_get_style_pad_top(cont, 0);
326 const int32_t pad_gap = lv_obj_get_style_pad_row(cont, 0);
327 const int32_t space_start = lv_obj_get_style_space_top(cont, LV_PART_MAIN);
328 const int32_t space_end = lv_obj_get_style_space_bottom(cont, LV_PART_MAIN);
329
330 /* Redundant but ensures all padding is as expected in case the theme sets different top & bottom padding */
331 lv_obj_set_style_pad_all(cont, pad_all, 0);
332 lv_obj_set_style_pad_gap(cont, pad_gap, 0);
333
334 lv_obj_t * header = lv_label_create(cont);
335 lv_obj_set_name(header, "header");
336 lv_label_set_text(header, "header");
337
338 lv_obj_t * item = lv_obj_create(cont);
339 lv_obj_set_name(item, "item");
340 lv_obj_set_width(item, LV_PCT(100));
341 lv_obj_set_flex_grow(item, 1);
342 lv_obj_set_style_bg_color(item, lv_color_hex(0x00ff00), 0);
343 lv_obj_set_style_bg_opa(item, LV_OPA_COVER, 0);
344
345 lv_obj_t * footer = lv_label_create(cont);
346 lv_obj_set_name(footer, "footer");
347 lv_label_set_text(footer, "footer");
348
349 lv_obj_update_layout(cont);
350 const int32_t header_height = lv_obj_get_height(header);
351 const int32_t footer_height = lv_obj_get_height(footer);
352
353 int32_t item_min_height = 0;
354 int32_t cont_min_height = 0;
355 int32_t item_max_height = LV_COORD_MAX;
356 int32_t cont_max_height = LV_COORD_MAX;
357
358 /* 2 * pad_gap because there are 3 objs in cont */
359 const int32_t fixed_size = space_start + header_height + 2 * pad_gap + footer_height + space_end;
360
361 TEST_ASSERT_GREATER_THAN_INT32(0, fixed_size);
362 TEST_ASSERT_EQUAL(fixed_size, lv_obj_get_height(cont));
363 TEST_ASSERT_EQUAL(0, lv_obj_get_height(item));
364
365 item_min_height = 200;
366 lv_obj_set_style_min_height(item, item_min_height, LV_PART_MAIN);
367
368 lv_obj_update_layout(cont);
369 TEST_ASSERT_EQUAL(item_min_height + fixed_size, lv_obj_get_height(cont));
370 TEST_ASSERT_EQUAL(item_min_height, lv_obj_get_height(item));
371
372 /* The min size of the cont should "override" the `LV_SIZE_CONTENT` height so item should be visible and grow to
373 * fill space */
374 item_min_height = 0;
375 cont_min_height = 400;
376 lv_obj_set_style_min_height(cont, cont_min_height, LV_PART_MAIN);
377 lv_obj_set_style_min_height(item, item_min_height, LV_PART_MAIN);
378
379 lv_obj_update_layout(cont);
380 TEST_ASSERT_EQUAL(cont_min_height, lv_obj_get_height(cont));
381 TEST_ASSERT_EQUAL(cont_min_height - fixed_size, lv_obj_get_height(item));
382
383 /* The min size of the cont should "override" the `LV_SIZE_CONTENT` height so item should be visible and grow to
384 * fill space up to the max size */
385 item_max_height = 200;
386 lv_obj_set_style_max_height(item, item_max_height, LV_PART_MAIN);
387
388 lv_obj_update_layout(cont);
389 TEST_ASSERT_LESS_THAN_INT32_MESSAGE(
390 cont_min_height - fixed_size,
391 item_max_height,
392 "For this test to function, `item_max_height` must be reduced otherwise the item size will not be clamped");
393 TEST_ASSERT_EQUAL(cont_min_height, lv_obj_get_height(cont));
394 TEST_ASSERT_EQUAL(item_max_height, lv_obj_get_height(item));
395
396 /* item height should be 0 since min size is not set for item or cont, cont should not be max size since max size is
397 * larger than content */
398 cont_min_height = 0;
399 cont_max_height = 500;
400 item_max_height = 0;
401 TEST_ASSERT_GREATER_THAN_INT32_MESSAGE(
402 fixed_size,
403 cont_max_height,
404 "For this test to function, `cont_max_height` must be increased otherwise the cont size will be clamped");
405 lv_obj_set_style_min_height(cont, cont_min_height, LV_PART_MAIN);
406 lv_obj_set_style_max_height(item, item_max_height, LV_PART_MAIN);
407 lv_obj_set_style_max_height(cont, cont_max_height, LV_PART_MAIN);
408
409 lv_obj_update_layout(cont);
410 TEST_ASSERT_EQUAL(0, lv_obj_get_height(item));
411 TEST_ASSERT_EQUAL(fixed_size, lv_obj_get_height(cont));
412
413 /* item height should be 0 since min size is not set for item or cont, cont should be max size since max size is
414 * smaller than content */
415 item_max_height = 0;
416 cont_max_height = 3 * fixed_size / 4;
417 lv_obj_set_style_max_height(item, item_max_height, LV_PART_MAIN);
418 lv_obj_set_style_max_height(cont, cont_max_height, LV_PART_MAIN);
419
420 lv_obj_update_layout(cont);
421 TEST_ASSERT_EQUAL(0, lv_obj_get_height(item));
422 TEST_ASSERT_EQUAL(cont_max_height, lv_obj_get_height(cont));
423
424 /* item should be min height (50) and cont will be max height so cont should be scrollable */
425 item_min_height = 50;
426 lv_obj_set_style_min_height(item, item_min_height, LV_PART_MAIN);
427
428 lv_obj_update_layout(cont);
429 TEST_ASSERT_EQUAL(item_min_height, lv_obj_get_height(item));
430 TEST_ASSERT_EQUAL(cont_max_height, lv_obj_get_height(cont));
431
432 /* max cont height should be ignored here */
433 cont_max_height = LV_PCT(70);
434 TEST_ASSERT_GREATER_THAN_INT32_MESSAGE(fixed_size + item_min_height,
435 cont_max_height,
436 "For this test to function, `cont_max_height` must be increased otherwise "
437 "the item size will be unintentionally clamped");
438 lv_obj_set_style_max_height(cont, cont_max_height, LV_PART_MAIN);
439
440 lv_obj_update_layout(cont);
441 TEST_ASSERT_EQUAL(item_min_height, lv_obj_get_height(item));
442 TEST_ASSERT_EQUAL(fixed_size + item_min_height, lv_obj_get_height(cont));
443}

579void test_nested_flex_grow_size_content(void)
580{
581 lv_obj_t * cont = lv_obj_create(lv_screen_active());
582 lv_obj_set_name(cont, "cont");
583 lv_obj_set_style_min_width(cont, LV_SIZE_CONTENT, LV_PART_MAIN);
584 lv_obj_set_size(cont, LV_PCT(100), LV_SIZE_CONTENT);
585 lv_obj_set_flex_flow(cont, LV_FLEX_FLOW_ROW);
586 lv_obj_set_style_bg_color(cont, lv_color_hex(0xff0000), 0);
587 lv_obj_set_style_bg_opa(cont, LV_OPA_COVER, 0);
588
589 const int32_t pad_all = lv_obj_get_style_pad_left(cont, 0);
590 const int32_t pad_gap = lv_obj_get_style_pad_row(cont, 0);
591 const int32_t space_start = lv_obj_get_style_space_left(cont, LV_PART_MAIN);
592 const int32_t space_end = lv_obj_get_style_space_right(cont, LV_PART_MAIN);
593
594 /* Redundant but ensures all padding is as expected in case the theme sets different top & bottom padding */
595 lv_obj_set_style_pad_all(cont, pad_all, 0);
596 lv_obj_set_style_pad_gap(cont, pad_gap, 0);
597
598 lv_obj_t * label = lv_label_create(cont);
599 lv_obj_set_name(label, "label");
600 lv_label_set_text(label, "label");
601
602 lv_obj_t * sub_cont = lv_obj_create(cont);
603 lv_obj_set_name(sub_cont, "sub_cont");
604 lv_obj_set_height(sub_cont, LV_SIZE_CONTENT);
605 lv_obj_set_flex_grow(sub_cont, 1);
606 lv_obj_set_flex_flow(sub_cont, LV_FLEX_FLOW_ROW_WRAP);
607 lv_obj_set_style_bg_color(sub_cont, lv_color_hex(0x00ff00), 0);
608 lv_obj_set_style_bg_opa(sub_cont, LV_OPA_COVER, 0);
609
610 /* Ensures all padding is as expected in case the theme sets different top & bottom padding */
611 lv_obj_set_style_pad_all(sub_cont, pad_all, 0);
612 lv_obj_set_style_pad_gap(sub_cont, pad_gap, 0);
613
614 lv_obj_update_layout(cont);
615 const int32_t label_width = lv_obj_get_width(label);
616 const int32_t cont_width = lv_obj_get_width(cont); /* LV_PCT(100) */
617 const int32_t fixed_size = space_start + label_width + pad_gap + space_end;
618
619 /* Check sub cont grows to fill expected space */
620 TEST_ASSERT_EQUAL(cont_width - fixed_size, lv_obj_get_width(sub_cont));
621 TEST_ASSERT_EQUAL(0,
622 lv_obj_get_height(sub_cont) - (lv_obj_get_style_space_top(sub_cont, LV_PART_MAIN) +
623 lv_obj_get_style_space_bottom(sub_cont, LV_PART_MAIN)));
624
625 const size_t items = 5;
626 size_t total_item_width = 0;
627 for(size_t i = 0; i < items; i++) {
628 lv_obj_t * item = lv_label_create(sub_cont);
629 lv_obj_set_name(item, "item_#");
630 lv_label_set_text_fmt(item, "item_%d", (int)i);
631 lv_obj_set_style_text_color(item, lv_color_black(), 0);
632
633 lv_obj_update_layout(item);
634 total_item_width += lv_obj_get_width(item);
635 }
636
637 lv_obj_set_style_max_width(sub_cont, LV_SIZE_CONTENT, LV_PART_MAIN);
638
639 lv_obj_update_layout(cont);
640
641 /* Manual check that the sub container has been clamped to the correct size of the content */
642 int32_t content_size = lv_obj_get_style_space_left(sub_cont, LV_PART_MAIN) + total_item_width +
643 (items - 1) * pad_gap + lv_obj_get_style_space_right(sub_cont, LV_PART_MAIN);
644 TEST_ASSERT_LESS_THAN_INT32_MESSAGE(cont_width - fixed_size,
645 content_size,
646 "For this test to work the content size must be less than the available grow "
647 "space, reduce the number of items or length of text in the items");
648 TEST_ASSERT_EQUAL(content_size, lv_obj_get_width(sub_cont));
649
650 /* Add more items to check that when the content size is calculated (which doesn't account for wrapping so max width
651 * should be greater than the available grow space), the width is not clamped */
652
653 const size_t new_items = 20;
654 for(size_t i = items; i < new_items; i++) {
655 lv_obj_t * item = lv_label_create(sub_cont);
656 lv_obj_set_name(item, "item_#");
657 lv_label_set_text_fmt(item, "item_%d", (int)i);
658 lv_obj_set_style_text_color(item, lv_color_black(), 0);
659
660 lv_obj_update_layout(item);
661 total_item_width += lv_obj_get_width(item);
662 }
663
664 lv_obj_update_layout(sub_cont);
665 int32_t new_content_size = lv_obj_get_style_space_left(sub_cont, LV_PART_MAIN) + total_item_width +
666 (new_items - 1) * pad_gap + lv_obj_get_style_space_right(sub_cont, LV_PART_MAIN);
667
668 TEST_ASSERT_GREATER_THAN_INT32_MESSAGE(
669 cont_width - fixed_size,
670 new_content_size,
671 "For this test to work the content size must be greater than the available grow "
672 "space, increase the number of items or length of text in the items");
673
674 TEST_ASSERT_EQUAL(cont_width - fixed_size, lv_obj_get_width(sub_cont));
675
676 /* This interaction was previously erroneous behaviour since the sub cont size would remain at the size before the
677 * new items were added. You previously had to manually override the max width to reset it */
678 lv_obj_set_style_max_width(sub_cont, LV_COORD_MAX, LV_PART_MAIN);
679 lv_obj_update_layout(sub_cont);
680 lv_obj_set_style_max_width(sub_cont, LV_SIZE_CONTENT, LV_PART_MAIN);
681
682 TEST_ASSERT_EQUAL(cont_width - fixed_size, lv_obj_get_width(sub_cont));
683}
