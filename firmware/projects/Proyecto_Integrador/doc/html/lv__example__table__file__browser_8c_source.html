129static bool path_join(char * buf, size_t buf_size, const char * base, const char * name)

131 int res = lv_fs_path_join(buf, buf_size, base, name);

141static void browser_open(const char * path)
142{
143 lv_fs_dir_t dir;
144 if(lv_fs_dir_open(&dir, path) != LV_FS_RES_OK) return;
145
146 lv_snprintf(browser_path, sizeof(browser_path), "%s", path);
147 lv_label_set_text(browser_path_label, browser_path);
148
149 lv_table_set_row_count(browser_table, 0);
150
151 /*First row goes up to the parent directory*/
152 uint32_t row = 0;
153 lv_table_set_cell_value(browser_table, row, 0, LV_SYMBOL_LEFT);
154 lv_table_set_cell_value(browser_table, row, 1, "..");
155 row++;
156
157 /*lv_fs_dir_read prefixes directory names with '/'; an empty name ends the list*/
158 char fn[LV_FS_MAX_PATH_LENGTH];
159 lv_fs_res_t res;
160 while((res = lv_fs_dir_read(&dir, fn, sizeof(fn))) == LV_FS_RES_OK && fn[0] != '\0') {
161 bool is_dir = fn[0] == '/';
162 const char * name = is_dir ? fn + 1 : fn;
163 lv_table_set_cell_value(browser_table, row, 0, is_dir ? LV_SYMBOL_DIRECTORY : LV_SYMBOL_FILE);
164 lv_table_set_cell_value(browser_table, row, 1, name);
165 row++;
166 }
167
168 if(res != LV_FS_RES_OK) LV_LOG_WARN("Reading %s failed, the listing is incomplete", path);
169
170 lv_fs_dir_close(&dir);
171}

176static void browser_up(void)
177{
178 char buf[LV_FS_MAX_PATH_LENGTH];
179 lv_snprintf(buf, sizeof(buf), "%s", browser_path);
180
181 size_t len = strlen(buf);
182 if(len > 0 && buf[len - 1] == '/') buf[len - 1] = '\0'; /*drop a trailing slash*/
183
184 char * slash = strrchr(buf, '/');
185 if(slash == NULL) return; /*already at the drive root*/
186 slash[1] = '\0'; /*keep the slash, cut the last component*/
187 browser_open(buf);
188}

190static void table_event(lv_event_t * e)
191{
192 LV_UNUSED(e);
193 uint32_t row, col;
194 lv_table_get_selected_cell(browser_table, &row, &col);
195 if(row == LV_TABLE_CELL_NONE) return;
196
197 const char * icon = lv_table_get_cell_value(browser_table, row, 0);
198 const char * name = lv_table_get_cell_value(browser_table, row, 1);
199
200 if(strcmp(name, "..") == 0) {
201 browser_up();
202 return;
203 }
204
205 char child[LV_FS_MAX_PATH_LENGTH];
206 if(!path_join(child, sizeof(child), browser_path, name)) {
207 LV_LOG_WARN("Path of %s is too long", name);
208 return;
209 }
210
211 /*Only folders can be opened; clicking a file just logs it*/
212 if(strcmp(icon, LV_SYMBOL_DIRECTORY) != 0) {
213 LV_LOG_USER("Selected file: %s", child);
214 return;
215 }
216
217 browser_open(child);
218}

241static void quick_add_shortcut(const char * icon, const char * name, const char * path)
242{
243 if(quick_access_cnt >= QUICK_ACCESS_MAX) {
244 LV_LOG_WARN("No room for the %s shortcut", name);
245 return;
246 }
247
248 /*The path is copied into static storage so no allocation needs to be freed later*/
249 char * stored_path = quick_access_paths[quick_access_cnt];
250 if(lv_strlcpy(stored_path, path, LV_FS_MAX_PATH_LENGTH) >= LV_FS_MAX_PATH_LENGTH) {
251 LV_LOG_WARN("Path of the %s shortcut is too long", name);
252 return;
253 }
254 quick_access_cnt++;
255
256 lv_obj_t * btn = lv_button_create(quick_access);
257 lv_obj_set_width(btn, lv_pct(100));
258 lv_obj_add_style(btn, &style_qa_btn, 0);
259 lv_obj_add_style(btn, &style_pressed, LV_STATE_PRESSED);
260
261 lv_obj_t * label = lv_label_create(btn);
262 lv_label_set_text_fmt(label, "%s %s", icon, name);
263
264 lv_obj_add_event_cb(btn, quick_access_event, LV_EVENT_CLICKED, stored_path);
265}

277void lv_example_table_file_browser(void)
278{
279 styles_init();
280 quick_access_cnt = 0; /*The sidebar buttons are created from scratch below*/
281
282 lv_obj_t * root = lv_obj_create(lv_screen_active());
283 lv_obj_set_size(root, lv_pct(100), lv_pct(100));
284 lv_obj_set_flex_flow(root, LV_FLEX_FLOW_ROW);
285 lv_obj_add_style(root, &style_bg, 0);
286
287 /*Quick-access sidebar (fixed width so the browser gets a predictable size)*/
288 quick_access = lv_obj_create(root);
289 lv_obj_set_size(quick_access, 100, lv_pct(100));
290 lv_obj_set_flex_flow(quick_access, LV_FLEX_FLOW_COLUMN);
291 lv_obj_add_style(quick_access, &style_sidebar, 0);
292
293 char home[LV_FS_MAX_PATH_LENGTH];
294#if LV_USE_FS_WIN32
295 lv_snprintf(home, sizeof(home), "C:C:/Users/Public");
296#else
297 /*Without HOME fall back to the drive root so the shortcuts still open something*/
298 const char * env = getenv("HOME");
299 lv_snprintf(home, sizeof(home), "A:%s", (env && env[0] != '\0') ? env : "/");
300#endif
301
302 char buf[LV_FS_MAX_PATH_LENGTH];
303 quick_add_header("PLACES", lv_palette_main(LV_PALETTE_LIME));
304 quick_add_shortcut(LV_SYMBOL_HOME, "Home", home);
305 if(path_join(buf, sizeof(buf), home, "Documents")) quick_add_shortcut(LV_SYMBOL_FILE, "Docs", buf);
306 if(path_join(buf, sizeof(buf), home, "Pictures")) quick_add_shortcut(LV_SYMBOL_IMAGE, "Pics", buf);
307 if(path_join(buf, sizeof(buf), home, "Music")) quick_add_shortcut(LV_SYMBOL_AUDIO, "Music", buf);
308
309 quick_add_header("DEVICE", lv_palette_main(LV_PALETTE_ORANGE));
310 quick_add_shortcut(LV_SYMBOL_DRIVE, "Files", BROWSER_ROOT);
311
312 /*Browser area: path header above the file table*/
313 lv_obj_t * browser = lv_obj_create(root);
314 lv_obj_set_height(browser, lv_pct(100));
315 lv_obj_set_flex_grow(browser, 1);
316 lv_obj_set_flex_flow(browser, LV_FLEX_FLOW_COLUMN);
317 lv_obj_add_style(browser, &style_panel, 0);
318
319 browser_path_label = lv_label_create(browser);
320 lv_obj_set_width(browser_path_label, lv_pct(100));
321 lv_label_set_long_mode(browser_path_label, LV_LABEL_LONG_MODE_DOTS);
322 lv_obj_add_style(browser_path_label, &style_header, 0);
323
324 browser_table = lv_table_create(browser);
325 lv_obj_set_width(browser_table, lv_pct(100));
326 lv_obj_set_flex_grow(browser_table, 1);
327 lv_obj_add_style(browser_table, &style_table, 0);
328 /*Tighten the cell padding; the theme default is large enough to clip the icon*/
329 lv_obj_set_style_pad_hor(browser_table, 6, LV_PART_ITEMS);
330 lv_obj_set_style_pad_ver(browser_table, 8, LV_PART_ITEMS);
331
332 /*Icon column fits a symbol; name column fills the rest of the browser area*/
333 int32_t icon_col = 30;
334 int32_t name_col = lv_display_get_horizontal_resolution(lv_display_get_default()) - 100 - icon_col - 16;
335 lv_table_set_column_count(browser_table, 2);
336 lv_table_set_column_width(browser_table, 0, icon_col);
337 lv_table_set_column_width(browser_table, 1, name_col);
338 lv_obj_add_event_cb(browser_table, table_event, LV_EVENT_VALUE_CHANGED, NULL);
339
340 browser_open(BROWSER_ROOT);
341}
