65lv_indev_t * lv_nuttx_mouse_create(const char * dev_path)
66{
67 lv_indev_t * indev;
68 int fd;
69
70 LV_CHECK_ARG(dev_path != NULL, return NULL);
71 LV_LOG_USER("mouse %s opening", dev_path);
72 fd = open(dev_path, O_RDONLY | O_NONBLOCK);
73 if(fd < 0) {
74 LV_LOG_ERROR("Error: cannot open mouse device");
75 return NULL;
76 }
77
78 LV_LOG_USER("mouse %s open success", dev_path);
79
80 indev = mouse_init(fd);
81
82 if(indev == NULL) {
83 close(fd);
84 }
85
86 return indev;
87}

93static void mouse_read(lv_indev_t * drv, lv_indev_data_t * data)
94{
95 lv_nuttx_mouse_t * mouse = drv->driver_data;
96 struct mouse_report_s sample;
97
98 /* Read one sample */
99
100 int nbytes = read(mouse->fd, &sample, sizeof(struct mouse_report_s));
101
102 /* Handle unexpected return values */
103
104 if(nbytes == sizeof(struct mouse_report_s)) {
105 lv_display_t * disp = lv_indev_get_display(drv);
106 int32_t hor_max = lv_display_get_horizontal_resolution(disp) - 1;
107 int32_t ver_max = lv_display_get_vertical_resolution(disp) - 1;
108
109 data->point.x =
110 LV_CLAMP(0,
111 data->point.x + (sample.x * LV_USE_NUTTX_MOUSE_MOVE_STEP),
112 hor_max);
113 data->point.y =
114 LV_CLAMP(0,
115 data->point.y + (sample.y * LV_USE_NUTTX_MOUSE_MOVE_STEP),
116 ver_max);
117
118 uint8_t mouse_buttons = sample.buttons;
119
120 if(mouse_buttons & MOUSE_BUTTON_1 || mouse_buttons & MOUSE_BUTTON_2 ||
121 mouse_buttons & MOUSE_BUTTON_3) {
122 mouse->last_state = LV_INDEV_STATE_PRESSED;
123 }
124 else {
125 mouse->last_state = LV_INDEV_STATE_RELEASED;
126 }
127 }
128 else {
129 if(nbytes == -1) {
130 if(errno != EAGAIN) {
131 LV_LOG_WARN("Read error: %s", strerror(errno));
132 }
133 }
134 else if(nbytes != 0) {
135 LV_LOG_WARN("Unexpected read size: %d", nbytes);
136 }
137 }
138
139 data->state = mouse->last_state;
140}

142static void mouse_delete_cb(lv_event_t * e)
143{
144 lv_indev_t * indev = lv_event_get_user_data(e);
145 lv_nuttx_mouse_t * mouse = lv_indev_get_driver_data(indev);
146 if(mouse) {
147 lv_indev_set_driver_data(indev, NULL);
148 lv_indev_set_read_cb(indev, NULL);
149
150 if(mouse->fd >= 0) {
151 close(mouse->fd);
152 mouse->fd = -1;
153 }
154 lv_free(mouse);
155 LV_LOG_USER("done");
156 }
157}

177static lv_indev_t * mouse_init(int fd)
178{
179 lv_nuttx_mouse_t * mouse;
180 lv_indev_t * indev = NULL;
181
182 mouse = lv_malloc_zeroed(sizeof(lv_nuttx_mouse_t));
183 LV_ASSERT_MALLOC(mouse);
184
185 mouse->fd = fd;
186 mouse->last_state = LV_INDEV_STATE_RELEASED;
187 indev = lv_indev_create();
188
189 if(indev == NULL) {
190 LV_LOG_ERROR("indev create failed");
191 lv_free(mouse);
192 return NULL;
193 }
194
195 lv_indev_set_type(indev, LV_INDEV_TYPE_POINTER);
196 lv_indev_set_read_cb(indev, mouse_read);
197 lv_indev_set_driver_data(indev, mouse);
198 lv_indev_add_event_cb(indev, mouse_delete_cb, LV_EVENT_DELETE, indev);
199
200 /* Set cursor icon */
201 mouse_set_cursor(indev);
202 return indev;
203}
