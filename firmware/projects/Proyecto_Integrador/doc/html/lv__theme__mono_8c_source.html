209lv_theme_t * lv_theme_mono_init(lv_display_t * disp, bool dark_bg, const lv_font_t * font)
210{
211 LV_CHECK_ARG(font != NULL, return NULL);
212
213 /*This trick is required only to avoid the garbage collection of
214 *styles' data if LVGL is used in a binding (e.g. MicroPython)
215 *In a general case styles could be in a simple `static lv_style_t my_style...` variables*/
216 if(!lv_theme_mono_is_inited()) {
217 theme_def = lv_malloc_zeroed(sizeof(my_theme_t));
218 LV_ASSERT_MALLOC(theme_def);
219 }
220
221 my_theme_t * theme = theme_def;
222
223 theme->base.disp = disp;
224 theme->base.font_small = LV_FONT_DEFAULT;
225 theme->base.font_normal = LV_FONT_DEFAULT;
226 theme->base.font_large = LV_FONT_DEFAULT;
227 theme->base.apply_cb = theme_apply;
228#if LV_USE_EXT_DATA
229 theme->base.ext_data.free_cb = NULL;
230 theme->base.ext_data.data = NULL;
231#endif
232
233 style_init(theme, dark_bg, font);
234
235 if(disp == NULL || lv_display_get_theme(disp) == (lv_theme_t *)theme) {
236 lv_obj_report_style_change(NULL);
237 }
238
239 theme->inited = true;
240
241 return (lv_theme_t *)theme_def;
242}
243
244bool lv_theme_mono_is_inited(void)
245{
246 my_theme_t * theme = theme_def;
247 if(theme == NULL) return false;
248 return theme->inited;
249}
250
251lv_theme_t * lv_theme_mono_get(void)
252{
253 if(!lv_theme_mono_is_inited()) {
254 return NULL;
255 }
256
257 return (lv_theme_t *)theme_def;
258}
259
260void lv_theme_mono_deinit(void)
261{
262 my_theme_t * theme = theme_def;
263 if(theme) {
264 if(theme->inited) {
265 lv_style_t * theme_styles = (lv_style_t *)(&(theme->styles));
266 uint32_t i;
267 for(i = 0; i < sizeof(my_theme_styles_t) / sizeof(lv_style_t); i++) {
268 lv_style_reset(theme_styles + i);
269 }
270 }
271#if LV_USE_EXT_DATA
272 if(theme->base.ext_data.free_cb) {
273 theme->base.ext_data.free_cb(theme->base.ext_data.data);
274 theme->base.ext_data.data = NULL;
275 }
276#endif
277 lv_free(theme_def);
278 theme_def = NULL;
279 }
