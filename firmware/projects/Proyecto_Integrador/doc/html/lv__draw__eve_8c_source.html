89static int32_t eve_dispatch(lv_draw_unit_t * draw_unit, lv_layer_t * layer)
90{
91 lv_draw_eve_unit_t * draw_eve_unit = (lv_draw_eve_unit_t *) draw_unit;
92
93 lv_draw_task_t * t = NULL;
94 t = lv_draw_get_next_available_task(layer, NULL, DRAW_UNIT_ID_EVE);
95 if(t == NULL) return LV_DRAW_UNIT_IDLE;
96
97
98 t->state = LV_DRAW_TASK_STATE_IN_PROGRESS;
99 draw_eve_unit->task_act = t;
100
101 eve_execute_drawing(draw_eve_unit);
102
103 draw_eve_unit->task_act->state = LV_DRAW_TASK_STATE_FINISHED;
104 draw_eve_unit->task_act = NULL;
105
106 /*The draw unit is free now. Request a new dispatching as it can get a new task*/
107 lv_draw_dispatch_request();
108
109
110 return 1;
111}

113static int32_t eve_evaluate(lv_draw_unit_t * draw_unit, lv_draw_task_t * task)
114{
115 LV_UNUSED(draw_unit);
116
117 if(lv_draw_eve_unit_g == NULL || lv_draw_eve_unit_g->disp == NULL)
118 return 0;
119
120 if(((lv_draw_dsc_base_t *)task->draw_dsc)->user_data == NULL) {
121 task->preference_score = 0;
122 task->preferred_draw_unit_id = DRAW_UNIT_ID_EVE;
123 }
124 return 0;
125}

127static void eve_execute_drawing(lv_draw_eve_unit_t * u)
128{
129 lv_draw_task_t * t = u->task_act;
130
131 switch(t->type) {
132 case LV_DRAW_TASK_TYPE_LINE:
133 lv_draw_line_iterate(t, t->draw_dsc, lv_draw_eve_line);
134 break;
135 case LV_DRAW_TASK_TYPE_BORDER:
136 lv_draw_eve_border(t, t->draw_dsc, &t->area);
137 break;
138 case LV_DRAW_TASK_TYPE_FILL:
139 lv_draw_eve_fill(t, t->draw_dsc, &t->area);
140 break;
141 case LV_DRAW_TASK_TYPE_IMAGE:
142 lv_draw_eve_image(t, t->draw_dsc, &t->area);
143 break;
144 case LV_DRAW_TASK_TYPE_LABEL:
145 lv_draw_eve_label(t, t->draw_dsc, &t->area);
146 break;
147 case LV_DRAW_TASK_TYPE_ARC:
148 lv_draw_eve_arc(t, t->draw_dsc, &t->area);
149 break;
150 case LV_DRAW_TASK_TYPE_TRIANGLE:
151 lv_draw_eve_triangle(t, t->draw_dsc);
152 break;
153 default:
154 break;
155 }
156}
