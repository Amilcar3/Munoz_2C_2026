8# Writes/updates the lvgl/.config configuration file by merging configuration
9# files passed as arguments
10#
11# When fragments haven't changed, lvgl/.config is both the input and the
12# output, which just updates it. This is handled in the CMake files.
13#
14# Also does various checks (most via Kconfiglib warnings).

44def main():
45 args = parse_args()
46
47 log("Parsing " + args.kconfig_file)
48 kconf = Kconfig(args.kconfig_file, warn_to_stderr=False, suppress_traceback=True)
49
50 if args.handwritten_input_configs:
51 # Warn for assignments to undefined symbols, but only for handwritten
52 # fragments, to avoid warnings-turned-errors when using an old
53 # configuration file together with updated Kconfig files
54 kconf.warn_assign_undef = True
55
56 # prj.conf may override settings from the board configuration, so
57 # disable warnings about symbols being assigned more than once
58 kconf.warn_assign_override = False
59 kconf.warn_assign_redun = False
60
61 if args.forced_input_configs:
62 # Do not warn on a redundant config.
63 # The reason is that a regular .config will be followed by the forced
64 # config which under normal circumstances should be identical to the
65 # configured setting.
66 # Only if user has modified to a value that gets overruled by the forced
67 # a warning shall be issued.
68 kconf.warn_assign_redun = False
69
70 # Load files
71 log(kconf.load_config(args.configs_in[0]))
72 for config in args.configs_in[1:]:
73 # replace=False creates a merged configuration
74 log(kconf.load_config(config, replace=False))
75
76 if args.handwritten_input_configs:
77 # Check that there are no assignments to promptless symbols, which
78 # have no effect.
79 #
80 # This only makes sense when loading handwritten fragments and not when
81 # loading lvgl/.config, because lvgl/.config is configuration
82 # output and also assigns promptless symbols.
83 check_no_promptless_assign(kconf)
84
85 # Print warnings for symbols that didn't get the assigned value. Only
86 # do this for handwritten input too, to avoid likely unhelpful warnings
87 # when using an old configuration and updating Kconfig files.
88 check_assigned_sym_values(kconf)
89 check_assigned_choice_values(kconf)
90
91 if kconf.syms.get("WARN_DEPRECATED", kconf.y).tri_value == 2:
92 check_deprecated(kconf)
93
94 if kconf.syms.get("WARN_EXPERIMENTAL", kconf.y).tri_value == 2:
95 check_experimental(kconf)
96
97 # Hack: Force all symbols to be evaluated, to catch warnings generated
98 # during evaluation. Wait till the end to write the actual output files, so
99 # that we don't generate any output if there are warnings-turned-errors.
100 #
101 # Kconfiglib caches calculated symbol values internally, so this is still
102 # fast.
103 kconf.write_config(os.devnull)
104
105 warn_only = r"warning:.*set more than once."
106
107 if kconf.warnings:
108 if args.forced_input_configs:
109 error_out = False
110 else:
111 error_out = True
112
113 # Put a blank line between warnings to make them easier to read
114 for warning in kconf.warnings:
115 print("\n" + warning, file=sys.stderr)
116
117 if not error_out and not re.search(warn_only, warning):
118 # The warning is not a warn_only, fail the Kconfig.
119 error_out = True
120
121 # Turn all warnings into errors, so that e.g. assignments to undefined
122 # Kconfig symbols become errors.
123 #
124 # A warning is generated by this script whenever a symbol gets a
125 # different value than the one it was assigned. Keep that one as just a
126 # warning for now.
127 if error_out:
128 err("Aborting due to Kconfig warnings")
129
130 # Write the merged configuration and the C header
131 log(kconf.write_config(args.config_out))
132 log(kconf.write_autoconf(args.header_out))
133
134 # Write the list of parsed Kconfig files to a file
135 write_kconfig_filenames(kconf, args.kconfig_list_out)

138def check_no_promptless_assign(kconf):
139 # Checks that no promptless symbols are assigned
140
141 for sym in kconf.unique_defined_syms:
142 if sym.user_value is not None and promptless(sym):
143 err(f"""\
144{sym.name_and_loc} is assigned in a configuration file, but is not directly
145user-configurable (has no prompt). It gets its value indirectly from other
146symbols. """ + SYM_INFO_HINT.format(sym))

149def check_assigned_sym_values(kconf):
150 # Verifies that the values assigned to symbols "took" (matches the value
151 # the symbols actually got), printing warnings otherwise. Choice symbols
152 # are checked separately, in check_assigned_choice_values().
153
154 for sym in kconf.unique_defined_syms:
155 if sym.choice:
156 continue
157
158 user_value = sym.user_value
159 if user_value is None:
160 continue
161
162 # Tristate values are represented as 0, 1, 2. Having them as "n", "m",
163 # "y" is more convenient here, so convert.
164 if sym.type in (BOOL, TRISTATE):
165 user_value = TRI_TO_STR[user_value]
166
167 if user_value != sym.str_value:
168 msg = (
169 f"{sym.name_and_loc} was assigned the value '{user_value}'"
170 f" but got the value '{sym.str_value}'. "
171 )
172
173 # List any unsatisfied 'depends on' dependencies in the warning
174 mdeps = missing_deps(sym)
175 if mdeps:
176 expr_strs = []
177 for expr in mdeps:
178 estr = expr_str(expr)
179 if isinstance(expr, tuple):
180 # Add () around dependencies that aren't plain symbols.
181 # Gives '(FOO || BAR) (=n)' instead of
182 # 'FOO || BAR (=n)', which might be clearer.
183 estr = f"({estr})"
184 expr_strs.append(f"{estr} " f"(={TRI_TO_STR[expr_value(expr)]})")
185
186 msg += (
187 "Check these unsatisfied dependencies: "
188 + ", ".join(expr_strs)
189 + ". "
190 )
191
192 warn(msg + SYM_INFO_HINT.format(sym))

195def missing_deps(sym):
196 # check_assigned_sym_values() helper for finding unsatisfied dependencies.
197 #
198 # Given direct dependencies
199 #
200 # depends on <expr> && <expr> && ... && <expr>
201 #
202 # on 'sym' (which can also come from e.g. a surrounding 'if'), returns a
203 # list of all <expr>s with a value less than the value 'sym' was assigned
204 # ("less" instead of "not equal" just to be general and handle tristates,
205 # even though lvgl doesn't use them).
206 #
207 # For string/int/hex symbols, just looks for <expr> = n.
208 #
209 # Note that <expr>s can be something more complicated than just a symbol,
210 # like 'FOO || BAR' or 'FOO = "string"'.
211
212 deps = split_expr(sym.direct_dep, AND)
213
214 if sym.type in (BOOL, TRISTATE):
215 return [dep for dep in deps if expr_value(dep) < sym.user_value]
216 # string/int/hex
217 return [dep for dep in deps if expr_value(dep) == 0]

220def check_assigned_choice_values(kconf):
221 # Verifies that any choice symbols that were selected (by setting them to
222 # y) ended up as the selection, printing warnings otherwise.
223 #
224 # We check choice symbols separately to avoid warnings when two different
225 # choice symbols within the same choice are set to y. This might happen if
226 # a choice selection from a board defconfig is overridden in a prj.conf,
227 # for example. The last choice symbol set to y becomes the selection (and
228 # all other choice symbols get the value n).
229 #
230 # Without special-casing choices, we'd detect that the first symbol set to
231 # y ended up as n, and print a spurious warning.
232
233 for choice in kconf.unique_choices:
234 if choice.user_selection and choice.user_selection is not choice.selection:
235
236 warn(f"""\
237The choice symbol {choice.user_selection.name_and_loc} was selected (set =y),
238but {choice.selection.name_and_loc if choice.selection else "no symbol"} ended
239up as the choice selection. """ + SYM_INFO_HINT.format(choice.user_selection))
240

242# Hint on where to find symbol information. Used like
243# SYM_INFO_HINT.format(sym).
244SYM_INFO_HINT = """\
245See https://docs.lvgl.io/master/integration/overview/configuration.html
246look up {0.name} in the menuconfig/guiconfig interface. The Application
247Development Primer, Setting Configuration Values, and Kconfig - Tips and Best
248Practices sections of the manual might be helpful too.\
249"""

252def check_deprecated(kconf):
253 deprecated = kconf.syms.get("DEPRECATED")
254 dep_expr = kconf.n if deprecated is None else deprecated.rev_dep
255
256 if dep_expr is not kconf.n:
257 selectors = [s for s in split_expr(dep_expr, OR) if expr_value(s) == 2]
258 for selector in selectors:
259 selector_name = split_expr(selector, AND)[0].name
260 warn(f"Deprecated symbol {selector_name} is enabled.")
261

264 experimental = kconf.syms.get("EXPERIMENTAL")

266
267 if dep_expr is not kconf.n:

274def promptless(sym):
275 # Returns True if 'sym' has no prompt. Since the symbol might be defined in
276 # multiple locations, we need to check all locations.
277
278 return not any(node.prompt for node in sym.nodes)

281def write_kconfig_filenames(kconf, kconfig_list_path):
282 # Writes a sorted list with the absolute paths of all parsed Kconfig files
283 # to 'kconfig_list_path'. The paths are realpath()'d, and duplicates are
284 # removed. This file is used by CMake to look for changed Kconfig files. It
285 # needs to be deterministic.
286
287 with open(kconfig_list_path, "w") as out:
288 for path in sorted(
289 {
290 os.path.realpath(os.path.join(kconf.srctree, path))
291 for path in kconf.kconfig_filenames
292 }
293 ):
294 print(path, file=out)

333def warn(msg):
