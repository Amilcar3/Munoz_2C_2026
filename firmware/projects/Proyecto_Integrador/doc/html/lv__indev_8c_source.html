88lv_indev_t * lv_indev_create(void)
89{
90 lv_display_t * disp = lv_display_get_default();
91 if(disp == NULL) {
92 LV_LOG_WARN("no display was created so far");
93 }
94
95 lv_indev_t * indev = lv_ll_ins_head(indev_ll_head);
96 LV_ASSERT_MALLOC(indev);
97 if(indev == NULL) {
98 return NULL;
99 }
100
101 lv_memzero(indev, sizeof(lv_indev_t));
102 indev->reset_query = 1;
103 indev->enabled = 1;
104
105 indev->read_timer = lv_timer_create(lv_indev_read_timer_cb, LV_DEF_REFR_PERIOD, indev);
106
107 indev->disp = lv_display_get_default();
108 indev->type = LV_INDEV_TYPE_NONE;
109 indev->mode = LV_INDEV_MODE_TIMER;
110 indev->scroll_limit = LV_INDEV_DEF_SCROLL_LIMIT;
111 indev->scroll_throw = LV_INDEV_DEF_SCROLL_THROW;
112 indev->long_press_time = LV_INDEV_DEF_LONG_PRESS_TIME;
113 indev->long_press_repeat_time = LV_INDEV_DEF_LONG_PRESS_REP_TIME;
114 indev->double_click_time = LV_INDEV_DEF_DOUBLE_CLICK_TIME;
115 indev->gesture_min_distance = LV_INDEV_DEF_GESTURE_LIMIT;
116 indev->gesture_min_velocity = LV_INDEV_DEF_GESTURE_MIN_VELOCITY;
117 indev->rotary_sensitivity = LV_INDEV_DEF_ROTARY_SENSITIVITY;
118 indev->key_remap_cb = NULL;
119#if LV_USE_EXT_DATA
120 indev->ext_data.free_cb = NULL;
121 indev->ext_data.data = NULL;
122#endif
123
124#if LV_USE_GESTURE_RECOGNITION
125 lv_indev_gesture_init(indev);
126#endif
127
128 return indev;

131void lv_indev_delete(lv_indev_t * indev)
132{
133 if(indev == NULL) return;
134
135 lv_indev_send_event(indev, LV_EVENT_DELETE, NULL);
136 lv_event_mark_deleted(indev);
137 lv_event_remove_all(&(indev->event_list));
138
139 /*Clean up the read timer first*/
140 if(indev->read_timer) lv_timer_delete(indev->read_timer);
141
142 /*Remove the input device from the list*/
143 lv_ll_remove(indev_ll_head, indev);
144
145#if LV_USE_EXT_DATA
146 if(indev->ext_data.free_cb) {
147 indev->ext_data.free_cb(indev->ext_data.data);
148 indev->ext_data.data = NULL;
149 }
150#endif
151
152 /*Free the memory of the input device*/
153 lv_free(indev);
154}

164static void indev_read_core(lv_indev_t * indev, lv_indev_data_t * data)
165{
166 LV_ASSERT(indev != NULL);
167 LV_ASSERT(data != NULL);
168 LV_PROFILER_INDEV_BEGIN;
169 lv_memzero(data, sizeof(lv_indev_data_t));
170
171 /* For touchpad sometimes users don't set the last pressed coordinate on release.
172 * So be sure a coordinates are initialized to the last point */
173 if(indev->type == LV_INDEV_TYPE_POINTER) {
174 data->point.x = indev->pointer.last_raw_point.x;
175 data->point.y = indev->pointer.last_raw_point.y;
176 }
177 /*Similarly set at least the last key in case of the user doesn't set it on release*/
178 else if(indev->type == LV_INDEV_TYPE_KEYPAD || indev->type == LV_INDEV_TYPE_ENCODER) {
179 data->key = indev->keypad.last_key;
180 }
181
182 if(indev->read_cb) {
183 LV_TRACE_INDEV("calling indev_read_cb");
184 indev->read_cb(indev, data);
185
186 /*Set the time stamp to the current time is it was not set in the read_cb*/
187 if(data->timestamp == 0) data->timestamp = lv_tick_get();
188 }
189 else {
190 LV_LOG_WARN("indev_read_cb is not registered");
191 }
192
193 LV_PROFILER_INDEV_END;
194}

202void lv_indev_read(lv_indev_t * indev)
203{
204 LV_CHECK_ARG(indev != NULL, return);
205
206 LV_TRACE_INDEV("begin");
207
208 indev_act = indev;
209
210 /*Read and process all indevs*/
211 if(indev->disp == NULL) return; /*Not assigned to any displays*/
212
213 /*Handle reset query before processing the point*/
214 indev_proc_reset_query_handler(indev);
215
216 if(indev->enabled == 0) return;
217 if(indev->disp->prev_scr != NULL) {
218 LV_TRACE_INDEV("input blocked while screen animation active");
219 return;
220 }
221
222 LV_PROFILER_INDEV_BEGIN;
223
224 bool continue_reading;
225 lv_indev_data_t data;
226
227 do {
228 /*Read the data*/
229 indev_read_core(indev, &data);
230 continue_reading = indev->mode != LV_INDEV_MODE_EVENT && data.continue_reading;
231
232 /*The active object might be deleted even in the read function*/
233 indev_proc_reset_query_handler(indev);
234 indev_obj_act = NULL;
235
236 indev->state = data.state;
237
238 /*Save the last activity time*/
239 indev->timestamp = data.timestamp;
240 if(indev->state == LV_INDEV_STATE_PRESSED) {
241 indev->disp->last_activity_time = data.timestamp;
242 }
243 else if(indev->type == LV_INDEV_TYPE_ENCODER && data.enc_diff) {
244 indev->disp->last_activity_time = data.timestamp;
245 }
246
247 if(indev->type == LV_INDEV_TYPE_POINTER) {
248 indev_pointer_proc(indev, &data);
249 }
250 else if(indev->type == LV_INDEV_TYPE_KEYPAD) {
251 indev_keypad_proc(indev, &data);
252 }
253 else if(indev->type == LV_INDEV_TYPE_ENCODER) {
254 indev_encoder_proc(indev, &data);
255 }
256 else if(indev->type == LV_INDEV_TYPE_BUTTON) {
257 indev_button_proc(indev, &data);
258 }
259 /*Handle reset query if it happened in during processing*/
260 indev_proc_reset_query_handler(indev);
261 } while(continue_reading);
262
263 /*End of indev processing, so no act indev*/
264 indev_act = NULL;
265 indev_obj_act = NULL;
266
267 LV_TRACE_INDEV("finished");
268 LV_PROFILER_INDEV_END;
269}

612lv_obj_t * lv_indev_search_obj(lv_obj_t * obj, lv_point_t * point)
613{
614 LV_CHECK_ARG(obj != NULL, return NULL);
615 LV_CHECK_ARG(point != NULL, return NULL);
616
617 lv_obj_t * found_p = NULL;
618
619 /*If this obj is hidden the children are hidden too so return immediately*/
620 if(lv_obj_is_hidden(obj)) return NULL;
621
622 lv_point_t p_trans = *point;
623 lv_obj_transform_point(obj, &p_trans, LV_OBJ_POINT_TRANSFORM_FLAG_INVERSE);
624
625 bool hit_test_ok = lv_obj_hit_test(obj, &p_trans);
626
627 /*If the point is on this object check its children too*/
628 lv_area_t obj_coords = obj->coords;
629 if(lv_obj_is_overflow_visible(obj)) {
630 int32_t ext_draw_size = lv_obj_get_ext_draw_size(obj);
631 lv_area_increase(&obj_coords, ext_draw_size, ext_draw_size);
632 }
633 if(lv_area_is_point_on(&obj_coords, &p_trans, 0)) {
634 int32_t i;
635 uint32_t child_cnt = lv_obj_get_child_count(obj);
636
637 /*If a child matches use it*/
638 for(i = child_cnt - 1; i >= 0; i--) {
639 lv_obj_t * child = obj->spec_attr->children[i];
640 found_p = lv_indev_search_obj(child, &p_trans);
641 if(found_p) return found_p;
642 }
643 }
644
645 /*If not return earlier for a clicked child and this obj's hittest was ok use it
646 *else return NULL*/
647 if(hit_test_ok) return obj;
648 else return NULL;
649}

755static void indev_pointer_proc(lv_indev_t * i, lv_indev_data_t * data)
756{
757 LV_ASSERT(i != NULL);
758 LV_ASSERT(data != NULL);
759 /*Save the raw points so they can be used again in indev_read_core*/
760 i->pointer.last_raw_point.x = data->point.x;
761 i->pointer.last_raw_point.y = data->point.y;
762
763 if(lv_indev_get_ccw(i)) {
764 lv_display_rotate_point_ccw(i->disp, &data->point);
765 }
766 else {
767 lv_display_rotate_point(i->disp, &data->point);
768 }
769
770 /*Simple sanity check*/
771 if(data->point.x < 0) {
772 LV_LOG_WARN("X is %d which is smaller than zero", (int)data->point.x);
773 }
774 if(data->point.x >= lv_display_get_horizontal_resolution(i->disp)) {
775 LV_LOG_WARN("X is %d which is greater than hor. res", (int)data->point.x);
776 }
777 if(data->point.y < 0) {
778 LV_LOG_WARN("Y is %d which is smaller than zero", (int)data->point.y);
779 }
780 if(data->point.y >= lv_display_get_vertical_resolution(i->disp)) {
781 LV_LOG_WARN("Y is %d which is greater than ver. res", (int)data->point.y);
782 }
783
784 /*Move the cursor if set and moved*/
785 if(i->cursor != NULL &&
786 (i->pointer.last_point.x != data->point.x || i->pointer.last_point.y != data->point.y)) {
787 lv_obj_set_pos(i->cursor, data->point.x, data->point.y);
788 }
789
790 i->pointer.act_point.x = data->point.x;
791 i->pointer.act_point.y = data->point.y;
792 i->pointer.diff = data->enc_diff;
793
794#if LV_USE_GESTURE_RECOGNITION
795 for(int gest = 0; gest < LV_INDEV_GESTURE_CNT; gest++) {
796 i->gesture_type[gest] = data->gesture_type[gest];
797 i->gesture_data[gest] = data->gesture_data[gest];
798 }
799#endif
800
801 /*Process the diff first as scrolling will be processed in indev_proc_release*/
802 indev_proc_pointer_diff(i);
803
804 if(i->state == LV_INDEV_STATE_PRESSED) {
805 indev_proc_press(i);
806 }
807 else {
808 indev_proc_release(i);
809 }
810
811 i->prev_state = i->state;
812 i->pointer.last_point.x = i->pointer.act_point.x;
813 i->pointer.last_point.y = i->pointer.act_point.y;
814}

821static void indev_keypad_proc(lv_indev_t * i, lv_indev_data_t * data)
822{
823 LV_ASSERT(i != NULL);
824 LV_ASSERT(data != NULL);
825 if(data->state == LV_INDEV_STATE_PRESSED && i->wait_until_release) return;
826
827 if(i->wait_until_release) {
828 i->wait_until_release = 0;
829 i->pr_timestamp = 0;
830 i->long_pr_sent = 0;
831 i->keypad.last_state = LV_INDEV_STATE_RELEASED; /*To skip the processing of release*/
832 }
833
834 /* Remap key using callback */
835 if(i->key_remap_cb) {
836 data->key = i->key_remap_cb(i, data->key);
837 }
838
839 /*Save the last key. *It must be done here else `lv_indev_get_key` will return the last key in events*/
840 uint32_t prev_key = i->keypad.last_key;
841 i->keypad.last_key = data->key;
842
843 /*Save the previous state so we can detect state changes below and also set the last state now
844 *so if any event handler on the way returns `LV_RESULT_INVALID` the last state is remembered
845 *for the next time*/
846 lv_indev_state_t prev_state = i->keypad.last_state;
847 i->keypad.last_state = data->state;
848
849
850 if(prev_key != data->key || prev_state != data->state) {
851 if(lv_indev_send_event(indev_act, LV_EVENT_KEY, NULL) == LV_RESULT_INVALID) {
852 return;
853 }
854 }
855
856 lv_group_t * g = i->group;
857
858 if(g != NULL) {
859 indev_obj_act = lv_group_get_focused(g);
860 if(indev_obj_act == NULL) return;
861 }
862 else {
863 indev_obj_act = NULL;
864 }
865
866 const bool is_enabled = (g == NULL) || !lv_obj_has_state(indev_obj_act, LV_STATE_DISABLED);
867
868 /*Key press happened*/
869 if(data->state == LV_INDEV_STATE_PRESSED && prev_state == LV_INDEV_STATE_RELEASED) {
870 LV_LOG_INFO("%" LV_PRIu32 " key is pressed", data->key);
871 i->pr_timestamp = i->timestamp;
872
873 if(g == NULL) {
874 if(send_event(LV_EVENT_PRESSED, indev_act) == LV_RESULT_INVALID) return;
875 }
876 /*Move the focus on NEXT*/
877 else if(data->key == LV_KEY_NEXT) {
878 lv_group_set_editing(g, false); /*Editing is not used by KEYPAD is be sure it is disabled*/
879 lv_group_focus_next(g);
880 if(indev_reset_check(i)) return;
881 }
882 /*Move the focus on PREV*/
883 else if(data->key == LV_KEY_PREV) {
884 lv_group_set_editing(g, false); /*Editing is not used by KEYPAD is be sure it is disabled*/
885 lv_group_focus_prev(g);
886 if(indev_reset_check(i)) return;
887 }
888 else if(is_enabled) {
889 /*Simulate a press on the object if ENTER was pressed*/
890 if(data->key == LV_KEY_ENTER) {
891 /*Send the ENTER as a normal KEY*/
892 if(lv_group_send_data(g, LV_KEY_ENTER) == LV_RESULT_INVALID) return;
893 if(indev_reset_check(i)) return;
894
895 if(send_event(LV_EVENT_PRESSED, indev_act) == LV_RESULT_INVALID) return;
896
897 }
898 else if(data->key == LV_KEY_ESC) {
899 /*Send the ESC as a normal KEY*/
900 if(lv_group_send_data(g, LV_KEY_ESC) == LV_RESULT_INVALID) return;
901 if(indev_reset_check(i)) return;
902
903 if(send_event(LV_EVENT_CANCEL, indev_act) == LV_RESULT_INVALID) return;
904 }
905 /*Just send other keys to the object (e.g. 'A' or `LV_GROUP_KEY_RIGHT`)*/
906 else {
907 if(lv_group_send_data(g, data->key) == LV_RESULT_INVALID) return;
908 if(indev_reset_check(i)) return;
909 }
910 }
911 }
912 /*Pressing*/
913 else if(is_enabled && data->state == LV_INDEV_STATE_PRESSED && prev_state == LV_INDEV_STATE_PRESSED) {
914
915 if(g == NULL || data->key == LV_KEY_ENTER) {
916 if(send_event(LV_EVENT_PRESSING, indev_act) == LV_RESULT_INVALID) return;
917 }
918
919 /*Long press time has elapsed?*/
920 if(i->long_pr_sent == 0 && lv_tick_diff(i->timestamp, i->pr_timestamp) >= i->long_press_time) {
921 i->long_pr_sent = 1;
922 if(g == NULL || data->key == LV_KEY_ENTER) {
923 i->longpr_rep_timestamp = i->timestamp;
924
925 if(send_event(LV_EVENT_LONG_PRESSED, indev_act) == LV_RESULT_INVALID) return;
926 }
927 }
928 /*Long press repeated time has elapsed?*/
929 else if(i->long_pr_sent != 0 &&
930 lv_tick_diff(i->timestamp, i->longpr_rep_timestamp) >= i->long_press_repeat_time) {
931
932 i->longpr_rep_timestamp = i->timestamp;
933
934 /*Send LONG_PRESS_REP on ENTER*/
935 if(g == NULL || data->key == LV_KEY_ENTER) {
936 if(send_event(LV_EVENT_LONG_PRESSED_REPEAT, indev_act) == LV_RESULT_INVALID) return;
937 }
938 /*Move the focus on NEXT again*/
939 else if(data->key == LV_KEY_NEXT) {
940 lv_group_set_editing(g, false); /*Editing is not used by KEYPAD is be sure it is disabled*/
941 lv_group_focus_next(g);
942 if(indev_reset_check(i)) return;
943 }
944 /*Move the focus on PREV again*/
945 else if(data->key == LV_KEY_PREV) {
946 lv_group_set_editing(g, false); /*Editing is not used by KEYPAD is be sure it is disabled*/
947 lv_group_focus_prev(g);
948 if(indev_reset_check(i)) return;
949 }
950 /*Just send other keys again to the object (e.g. 'A' or `LV_GROUP_KEY_RIGHT)*/
951 else {
952 lv_group_send_data(g, data->key);
953 if(indev_reset_check(i)) return;
954 }
955 }
956 }
957 /*Release happened*/
958 else if(is_enabled && data->state == LV_INDEV_STATE_RELEASED && prev_state == LV_INDEV_STATE_PRESSED) {
959 LV_LOG_INFO("%" LV_PRIu32 " key is released", data->key);
960
961 /*The user might clear the key when it was released. Always release the pressed key*/
962 data->key = prev_key;
963 if(g == NULL || data->key == LV_KEY_ENTER) {
964
965 if(send_event(LV_EVENT_RELEASED, indev_act) == LV_RESULT_INVALID) return;
966
967 if(i->long_pr_sent == 0) {
968 if(indev_proc_short_click(i) == LV_RESULT_INVALID) return;
969 }
970
971 if(send_event(LV_EVENT_CLICKED, indev_act) == LV_RESULT_INVALID) return;
972
973 }
974 i->pr_timestamp = 0;
975 i->long_pr_sent = 0;
976 }
977 indev_obj_act = NULL;
978}

987static void indev_encoder_proc(lv_indev_t * i, lv_indev_data_t * data)
988{
989 LV_ASSERT(i != NULL);
990 LV_ASSERT(data != NULL);
991 if(data->state == LV_INDEV_STATE_PRESSED && i->wait_until_release) return;
992
993 if(i->wait_until_release) {
994 i->wait_until_release = 0;
995 i->pr_timestamp = 0;
996 i->long_pr_sent = 0;
997 i->keypad.last_state = LV_INDEV_STATE_RELEASED; /*To skip the processing of release*/
998 }
999
1000 /*Save the last keys before anything else.
1001 *They need to be already saved if the function returns for any reason*/
1002 lv_indev_state_t last_state = i->keypad.last_state;
1003 i->keypad.last_state = data->state;
1004 i->keypad.last_key = data->key;
1005
1006 lv_group_t * g = i->group;
1007 if(g == NULL) return;
1008
1009 indev_obj_act = lv_group_get_focused(g);
1010 if(indev_obj_act == NULL) return;
1011
1012 /*Process the steps they are valid only with released button*/
1013 if(data->state != LV_INDEV_STATE_RELEASED) {
1014 data->enc_diff = 0;
1015 }
1016
1017 const bool is_enabled = !lv_obj_has_state(indev_obj_act, LV_STATE_DISABLED);
1018
1019 /*Button press happened*/
1020 if(data->state == LV_INDEV_STATE_PRESSED && last_state == LV_INDEV_STATE_RELEASED) {
1021 LV_LOG_INFO("pressed");
1022
1023 i->pr_timestamp = i->timestamp;
1024
1025 if(data->key == LV_KEY_ENTER) {
1026 bool editable_or_scrollable = lv_obj_is_editable(indev_obj_act) ||
1027 lv_obj_is_scrollable(indev_obj_act);
1028 if(lv_group_get_editing(g) == true || editable_or_scrollable == false) {
1029
1030 if(is_enabled) {
1031 if(send_event(LV_EVENT_PRESSED, indev_act) == LV_RESULT_INVALID) return;
1032 }
1033 }
1034 }
1035 else if(data->key == LV_KEY_LEFT) {
1036 /*emulate encoder left*/
1037 data->enc_diff--;
1038 }
1039 else if(data->key == LV_KEY_RIGHT) {
1040 /*emulate encoder right*/
1041 data->enc_diff++;
1042 }
1043 else if(data->key == LV_KEY_ESC) {
1044 /*Send the ESC as a normal KEY*/
1045 lv_group_send_data(g, LV_KEY_ESC);
1046 if(indev_reset_check(i)) return;
1047
1048 if(is_enabled) {
1049 if(send_event(LV_EVENT_CANCEL, indev_act) == LV_RESULT_INVALID) return;
1050 }
1051 }
1052 /*Just send other keys to the object (e.g. 'A' or `LV_GROUP_KEY_RIGHT`)*/
1053 else {
1054 lv_group_send_data(g, data->key);
1055 if(indev_reset_check(i)) return;
1056 }
1057 }
1058 /*Pressing*/
1059 else if(data->state == LV_INDEV_STATE_PRESSED && last_state == LV_INDEV_STATE_PRESSED) {
1060 /*Long press*/
1061 if(i->long_pr_sent == 0 && lv_tick_diff(i->timestamp, i->pr_timestamp) >= i->long_press_time) {
1062
1063 i->long_pr_sent = 1;
1064 i->longpr_rep_timestamp = i->timestamp;
1065
1066 if(data->key == LV_KEY_ENTER) {
1067 /* Always send event to indev callbacks*/
1068 lv_indev_send_event(indev_act, LV_EVENT_LONG_PRESSED, indev_obj_act);
1069 if(indev_reset_check(indev_act)) return;
1070
1071 bool editable_or_scrollable = lv_obj_is_editable(indev_obj_act) ||
1072 lv_obj_is_scrollable(indev_obj_act);
1073
1074 /*On enter long press toggle edit mode.*/
1075 if(editable_or_scrollable) {
1076 /*Don't leave edit mode if there is only one object (nowhere to navigate)*/
1077 if(lv_group_get_obj_count(g) > 1) {
1078 LV_LOG_INFO("toggling edit mode");
1079 lv_group_set_editing(g, lv_group_get_editing(g) ? false : true); /*Toggle edit mode on long press*/
1080 lv_obj_remove_state(indev_obj_act, LV_STATE_PRESSED); /*Remove the pressed state manually*/
1081 }
1082 }
1083 /*If not editable then just send a long press event*/
1084 else {
1085 if(is_enabled) {
1086 lv_obj_send_event(indev_obj_act, LV_EVENT_LONG_PRESSED, indev_act);
1087 if(indev_reset_check(indev_act)) return;
1088 }
1089 }
1090 }
1091
1092 i->long_pr_sent = 1;
1093 }
1094 /*Long press repeated time has elapsed?*/
1095 else if(i->long_pr_sent != 0 && lv_tick_diff(i->timestamp, i->longpr_rep_timestamp) >= i->long_press_repeat_time) {
1096
1097 i->longpr_rep_timestamp = i->timestamp;
1098
1099 if(data->key == LV_KEY_ENTER) {
1100 if(is_enabled) {
1101 if(send_event(LV_EVENT_LONG_PRESSED_REPEAT, indev_act) == LV_RESULT_INVALID) return;
1102 }
1103 }
1104 else if(data->key == LV_KEY_LEFT) {
1105 /*emulate encoder left*/
1106 data->enc_diff--;
1107 }
1108 else if(data->key == LV_KEY_RIGHT) {
1109 /*emulate encoder right*/
1110 data->enc_diff++;
1111 }
1112 else {
1113 lv_group_send_data(g, data->key);
1114 if(indev_reset_check(i)) return;
1115 }
1116
1117 }
1118
1119 }
1120 /*Release happened*/
1121 else if(data->state == LV_INDEV_STATE_RELEASED && last_state == LV_INDEV_STATE_PRESSED) {
1122 LV_LOG_INFO("released");
1123
1124 if(data->key == LV_KEY_ENTER) {
1125 bool editable_or_scrollable = lv_obj_is_editable(indev_obj_act) ||
1126 lv_obj_is_scrollable(indev_obj_act);
1127
1128 /*The button was released on a non-editable object. Just send enter*/
1129 if(editable_or_scrollable == false) {
1130 if(is_enabled) {
1131 if(send_event(LV_EVENT_RELEASED, indev_act) == LV_RESULT_INVALID) return;
1132 }
1133
1134 if(i->long_pr_sent == 0 && is_enabled) {
1135 if(indev_proc_short_click(i) == LV_RESULT_INVALID) return;
1136 }
1137
1138 if(is_enabled) {
1139 if(send_event(LV_EVENT_CLICKED, indev_act) == LV_RESULT_INVALID) return;
1140 }
1141
1142 }
1143 /*An object is being edited and the button is released.*/
1144 else if(lv_group_get_editing(g)) {
1145 /*Ignore long pressed enter release because it comes from mode switch*/
1146 if(!i->long_pr_sent || lv_group_get_obj_count(g) <= 1) {
1147 if(is_enabled) {
1148 if(send_event(LV_EVENT_RELEASED, indev_act) == LV_RESULT_INVALID) return;
1149 if(indev_proc_short_click(i) == LV_RESULT_INVALID) return;
1150 if(send_event(LV_EVENT_CLICKED, indev_act) == LV_RESULT_INVALID) return;
1151 }
1152
1153 lv_group_send_data(g, LV_KEY_ENTER);
1154 if(indev_reset_check(i)) return;
1155 }
1156 else {
1157 lv_obj_remove_state(indev_obj_act, LV_STATE_PRESSED); /*Remove the pressed state manually*/
1158 }
1159 }
1160 /*If the focused object is editable and now in navigate mode then on enter switch edit
1161 mode*/
1162 else if(!i->long_pr_sent) {
1163 LV_LOG_INFO("entering edit mode");
1164 lv_group_set_editing(g, true); /*Set edit mode*/
1165 }
1166 }
1167
1168 i->pr_timestamp = 0;
1169 i->long_pr_sent = 0;
1170 }
1171 indev_obj_act = NULL;
1172
1173 /*if encoder steps or simulated steps via left/right keys*/
1174 if(data->enc_diff != 0) {
1175 /*In edit mode send LEFT/RIGHT keys*/
1176 if(lv_group_get_editing(g)) {
1177 LV_LOG_INFO("rotated by %+d (edit)", data->enc_diff);
1178 int32_t s;
1179 if(data->enc_diff < 0) {
1180 for(s = 0; s < -data->enc_diff; s++) {
1181 lv_group_send_data(g, LV_KEY_LEFT);
1182 if(indev_reset_check(i)) return;
1183 }
1184 }
1185 else if(data->enc_diff > 0) {
1186 for(s = 0; s < data->enc_diff; s++) {
1187 lv_group_send_data(g, LV_KEY_RIGHT);
1188 if(indev_reset_check(i)) return;
1189 }
1190 }
1191 }
1192 /*In navigate mode focus on the next/prev objects*/
1193 else {
1194 LV_LOG_INFO("rotated by %+d (nav)", data->enc_diff);
1195 int32_t s;
1196 if(data->enc_diff < 0) {
1197 for(s = 0; s < -data->enc_diff; s++) {
1198 lv_group_focus_prev(g);
1199 if(indev_reset_check(i)) return;
1200 }
1201 }
1202 else if(data->enc_diff > 0) {
1203 for(s = 0; s < data->enc_diff; s++) {
1204 lv_group_focus_next(g);
1205 if(indev_reset_check(i)) return;
1206 }
1207 }
1208 }
1209 }

1217static void indev_button_proc(lv_indev_t * i, lv_indev_data_t * data)
1218{
1219 LV_ASSERT(i != NULL);
1220 LV_ASSERT(data != NULL);
1221 /*Die gracefully if i->btn_points is NULL*/
1222 if(i->btn_points == NULL) {
1223 LV_LOG_WARN("btn_points is NULL");
1224 return;
1225 }
1226
1227 int32_t x = i->btn_points[data->btn_id].x;
1228 int32_t y = i->btn_points[data->btn_id].y;
1229
1230 if(LV_INDEV_STATE_RELEASED != data->state) {
1231 if(data->state == LV_INDEV_STATE_PRESSED) {
1232 LV_LOG_INFO("button %" LV_PRIu32 " is pressed (x:%d y:%d)", data->btn_id, (int)x, (int)y);
1233 }
1234 else {
1235 LV_LOG_INFO("button %" LV_PRIu32 " is released (x:%d y:%d)", data->btn_id, (int)x, (int)y);
1236 }
1237 }
1238
1239 /*If a new point comes always make a release*/
1240 if(data->state == LV_INDEV_STATE_PRESSED) {
1241 if(i->pointer.last_point.x != x ||
1242 i->pointer.last_point.y != y) {
1243 indev_proc_release(i);
1244 }
1245 }
1246
1247 if(indev_reset_check(i)) return;
1248
1249 /*Save the new points*/
1250 i->pointer.act_point.x = x;
1251 i->pointer.act_point.y = y;
1252
1253 if(data->state == LV_INDEV_STATE_PRESSED) {
1254 indev_proc_press(i);
1255 }
1256 else {
1257 indev_proc_release(i);
1258 }

1260 i->prev_state = i->state;
1261
1262 if(indev_reset_check(i)) return;
1263
1264 i->pointer.last_point.x = i->pointer.act_point.x;
1265 i->pointer.last_point.y = i->pointer.act_point.y;
1266}

1276{
1277 if(t <= 0) return x;
1278 if(t >= 99) return 0;
1279 return x * (512 - (512 * t) / 99) / 512;

1286static void indev_proc_press(lv_indev_t * indev)
1287{
1288 LV_ASSERT(indev != NULL);
1289 LV_LOG_INFO("pressed at x:%d y:%d", (int)indev->pointer.act_point.x,
1290 (int)indev->pointer.act_point.y);
1291 indev_obj_act = indev->pointer.act_obj;
1292
1293 if(indev->wait_until_release != 0) return;
1294
1295 lv_display_t * disp = indev_act->disp;
1296 bool new_obj_searched = false;
1297
1298 /*If there is no last object then search*/
1299 if(indev_obj_act == NULL) {
1300 indev_obj_act = pointer_search_obj(disp, &indev->pointer.act_point);
1301 new_obj_searched = true;
1302 }
1303 /*If there is an active object it's not scrolled and not press locked also search*/
1304 else if(indev->pointer.scroll_obj == NULL &&
1305 lv_obj_is_press_lock(indev_obj_act) == false) {
1306 indev_obj_act = pointer_search_obj(disp, &indev->pointer.act_point);
1307 new_obj_searched = true;
1308 }
1309
1310 /*The scroll object might have scroll throw. Stop it manually*/
1311 if(new_obj_searched && indev->pointer.scroll_obj) {
1312 /*Attempt to stop scroll throw animation firstly*/
1313 if(indev->scroll_throw_anim) {
1314 lv_anim_delete(indev, indev_scroll_throw_anim_cb);
1315 indev->scroll_throw_anim = NULL;
1316 }
1317
1318 lv_indev_scroll_throw_handler(indev);
1319 if(indev_reset_check(indev)) return;
1320 }
1321
1322 /*If a new object was found reset some variables and send a pressed event handler*/
1323 if(indev_obj_act != indev->pointer.act_obj) {
1324 /*If no previous object was lost, overwrite the last point.*/
1325 if(indev->pointer.act_obj == NULL) {
1326 indev->pointer.last_point.x = indev->pointer.act_point.x;
1327 indev->pointer.last_point.y = indev->pointer.act_point.y;
1328 }
1329 indev->pointer.pressed = indev->prev_state == LV_INDEV_STATE_RELEASED;
1330
1331 /*Without `LV_OBJ_FLAG_PRESS_LOCK` new widget can be found while pressing.*/
1332 if(indev->pointer.last_hovered && indev->pointer.last_hovered != indev_obj_act) {
1333 lv_obj_send_event(indev->pointer.last_hovered, LV_EVENT_HOVER_LEAVE, indev);
1334 if(indev_reset_check(indev)) return;
1335
1336 lv_indev_send_event(indev, LV_EVENT_HOVER_LEAVE, indev->pointer.last_hovered);
1337 if(indev_reset_check(indev)) return;
1338
1339 indev->pointer.last_hovered = indev_obj_act;
1340 }
1341
1342 /*If a new object found the previous was lost, so send a PRESS_LOST event*/
1343 if(indev->pointer.act_obj != NULL) {
1344 /*Save the obj because in special cases `act_obj` can change in the event */
1345 lv_obj_t * prev_act_obj = indev->pointer.act_obj;
1346 lv_obj_send_event(prev_act_obj, LV_EVENT_PRESS_LOST, indev_act);
1347 if(indev_reset_check(indev)) return;
1348 }
1349
1350 indev->pointer.act_obj = indev_obj_act; /*Save the pressed object*/
1351
1352 if(indev_obj_act != NULL) {
1353
1354 /*Save the time when the obj pressed to count long press time.*/
1355 indev->pr_timestamp = indev->timestamp;
1356 indev->long_pr_sent = 0;
1357 indev->pointer.scroll_sum.x = 0;
1358 indev->pointer.scroll_sum.y = 0;
1359 indev->pointer.scroll_dir = LV_DIR_NONE;
1360 indev->pointer.scroll_obj = NULL;
1361 indev->pointer.gesture_dir = LV_DIR_NONE;
1362 indev->pointer.gesture_sent = 0;
1363 indev->pointer.gesture_sum.x = 0;
1364 indev->pointer.gesture_sum.y = 0;
1365 indev->pointer.press_moved = 0;
1366 indev->pointer.vect.x = 0;
1367 indev->pointer.vect.y = 0;
1368
1369
1370 /* If the indev was already in a pressed state it means that we got dragged here
1371 * so we shouldn't send any hover nor pressed events for a new object since the
1372 * originally pressed object didn't get released
1373 */
1374 if(indev->prev_state != LV_INDEV_STATE_PRESSED) {
1375 const bool is_enabled = !lv_obj_has_state(indev_obj_act, LV_STATE_DISABLED);
1376 if(is_enabled) {
1377 if(indev->pointer.last_hovered != indev_obj_act) {
1378 if(send_event(LV_EVENT_HOVER_OVER, indev_act) == LV_RESULT_INVALID) return;
1379 }
1380 if(send_event(LV_EVENT_PRESSED, indev_act) == LV_RESULT_INVALID) return;
1381 }
1382 }
1383
1384 if(indev_act->wait_until_release) return;
1385
1386 /*Handle focus*/
1387 indev_click_focus(indev_act);
1388 if(indev_reset_check(indev)) return;
1389
1390 }
1391 }
1392
1393 /*Update vector and scroll throw vector*/
1394 indev->pointer.vect.x = indev->pointer.act_point.x - indev->pointer.last_point.x;
1395 indev->pointer.vect.y = indev->pointer.act_point.y - indev->pointer.last_point.y;
1396
1397 indev->pointer.vect_hist[indev->pointer.vect_hist_index] = indev->pointer.vect;
1398 indev->pointer.vect_hist_timestamp[indev->pointer.vect_hist_index] = indev->timestamp;
1399 indev->pointer.vect_hist_index = (indev->pointer.vect_hist_index + 1) % LV_INDEV_VECT_HIST_SIZE;
1400
1401 indev->pointer.scroll_throw_vect.x = 0;
1402 indev->pointer.scroll_throw_vect.y = 0;
1403 for(int i = 0; i < LV_INDEV_VECT_HIST_SIZE; i++) {
1404 int32_t t = lv_tick_diff(indev->timestamp, indev->pointer.vect_hist_timestamp[i]);
1405 indev->pointer.scroll_throw_vect.x += indev_scroll_throw_decay(indev->pointer.vect_hist[i].x, t);
1406 indev->pointer.scroll_throw_vect.y += indev_scroll_throw_decay(indev->pointer.vect_hist[i].y, t);
1407 }
1408
1409 indev->pointer.scroll_throw_vect_ori = indev->pointer.scroll_throw_vect;
1410
1411 if(LV_ABS(indev->pointer.vect.x) > indev->scroll_limit || LV_ABS(indev->pointer.vect.y) > indev->scroll_limit) {
1412 indev->pointer.press_moved = 1;
1413 }
1414
1415#if LV_USE_GESTURE_RECOGNITION
1416 for(int i = 0; i < LV_INDEV_GESTURE_CNT; i++) {
1417 /* Send a gesture event to a potential indev cb callback, even if no object was found */
1418 if(indev->gesture_type[i] != LV_INDEV_GESTURE_NONE) {
1419 indev->cur_gesture = (lv_indev_gesture_type_t) i;
1420 lv_indev_send_event(indev, LV_EVENT_GESTURE, indev_act);
1421 break;
1422 }
1423 }
1424#endif
1425
1426 if(indev_obj_act) {
1427 const bool is_enabled = !lv_obj_has_state(indev_obj_act, LV_STATE_DISABLED);
1428
1429#if LV_USE_GESTURE_RECOGNITION
1430 for(int i = 0; i < LV_INDEV_GESTURE_CNT; i++) {
1431
1432 if(indev->gesture_type[i] != LV_INDEV_GESTURE_NONE) {
1433 indev->cur_gesture = (lv_indev_gesture_type_t) i;
1434 if(send_event(LV_EVENT_GESTURE, indev_act) == LV_RESULT_INVALID) return;
1435 break;
1436 }
1437 }
1438#endif
1439 if(is_enabled) {
1440 if(send_event(LV_EVENT_PRESSING, indev_act) == LV_RESULT_INVALID) return;
1441 }
1442
1443
1444 if(indev_act->wait_until_release) return;
1445
1446 if(indev->pointer.scroll_obj) {
1447 lv_obj_stop_scroll_anim(indev->pointer.scroll_obj);
1448 }
1449
1450 lv_indev_scroll_handler(indev);
1451 if(indev_reset_check(indev)) return;
1452 indev_gesture(indev);
1453 if(indev_reset_check(indev)) return;
1454
1455 /*In event driven mode resume the timer so that it can trigger long pressed and other time related events.
1456 *As a side effect it will also call read_cb periodically in event driven mode. */
1457 if(indev->mode == LV_INDEV_MODE_EVENT && indev->read_timer && lv_timer_get_paused(indev->read_timer)) {
1458 lv_timer_resume(indev->read_timer);
1459 }
1460
1461 /*If there is no scrolling then check for long press time*/
1462 if(indev->pointer.scroll_obj == NULL && indev->long_pr_sent == 0) {
1463 /*Send a long press event if enough time elapsed*/
1464 if(lv_tick_diff(indev->timestamp, indev->pr_timestamp) >= indev_act->long_press_time) {
1465 if(is_enabled) {
1466 if(send_event(LV_EVENT_LONG_PRESSED, indev_act) == LV_RESULT_INVALID) return;
1467 }
1468 /*Mark it to do not send the event again*/
1469 indev->long_pr_sent = 1;
1470
1471 /*Save the long press time stamp for the long press repeat handler*/
1472 indev->longpr_rep_timestamp = indev->timestamp;
1473 }
1474 }
1475
1476 if(indev->pointer.scroll_obj == NULL && indev->long_pr_sent == 1) {
1477 if(lv_tick_diff(indev->timestamp, indev->longpr_rep_timestamp) >= indev_act->long_press_repeat_time) {
1478 if(is_enabled) {
1479 if(send_event(LV_EVENT_LONG_PRESSED_REPEAT, indev_act) == LV_RESULT_INVALID) return;
1480 }
1481 indev->longpr_rep_timestamp = indev->timestamp;
1482 }
1483 }
1484 }

1491static void indev_proc_release(lv_indev_t * indev)
1492{
1493 LV_ASSERT(indev != NULL);
1494 if(indev->wait_until_release || /*Hover the new widget even if the coordinates didn't changed*/
1495 (indev->pointer.last_point.x != indev->pointer.act_point.x ||
1496 indev->pointer.last_point.y != indev->pointer.act_point.y)) {
1497 lv_obj_t ** last = &indev->pointer.last_hovered;
1498 lv_obj_t * hovered = pointer_search_obj(lv_display_get_default(), &indev->pointer.act_point);
1499 if(*last != hovered) {
1500
1501 if(hovered) {
1502 lv_obj_send_event(hovered, LV_EVENT_HOVER_OVER, indev);
1503 if(indev_reset_check(indev)) return;
1504 lv_indev_send_event(indev, LV_EVENT_HOVER_OVER, hovered);
1505 if(indev_reset_check(indev)) return;
1506 }
1507
1508 if(*last) {
1509 lv_obj_send_event(*last, LV_EVENT_HOVER_LEAVE, indev);
1510 if(indev_reset_check(indev)) return;
1511 lv_indev_send_event(indev, LV_EVENT_HOVER_LEAVE, *last);
1512 if(indev_reset_check(indev)) return;
1513 }
1514 *last = hovered;
1515 }
1516 }
1517
1518 if(indev->wait_until_release) {
1519 if(indev->pointer.act_obj) {
1520 lv_obj_send_event(indev->pointer.act_obj, LV_EVENT_PRESS_LOST, indev_act);
1521 }
1522 if(indev_reset_check(indev)) {
1523 indev->wait_until_release = 0;
1524 return;
1525 }
1526
1527 indev->pointer.act_obj = NULL;
1528 indev->pr_timestamp = 0;
1529 indev->longpr_rep_timestamp = 0;
1530 indev->wait_until_release = 0;
1531 }
1532 indev_obj_act = indev->pointer.act_obj;
1533 lv_obj_t * scroll_obj = indev->pointer.scroll_obj;
1534
1535 if(indev->mode == LV_INDEV_MODE_EVENT && indev->read_timer && !lv_timer_get_paused(indev->read_timer)) {
1536 lv_timer_pause(indev->read_timer);
1537 }
1538
1539#if LV_USE_GESTURE_RECOGNITION
1540 /* Send a gesture event to a potential indev cb callback, even if no object was found */
1541 for(int i = 0; i < LV_INDEV_GESTURE_CNT; i++) {
1542
1543 if(indev->gesture_type[i] != LV_INDEV_GESTURE_NONE) {
1544 indev_act->cur_gesture = (lv_indev_gesture_type_t) i;
1545 lv_indev_send_event(indev, LV_EVENT_GESTURE, indev_act);
1546 break;
1547 }
1548 }
1549#endif
1550
1551 if(indev_obj_act) {
1552 LV_LOG_INFO("released");
1553
1554 const bool is_enabled = !lv_obj_has_state(indev_obj_act, LV_STATE_DISABLED);
1555
1556#if LV_USE_GESTURE_RECOGNITION
1557 for(int i = 0; i < LV_INDEV_GESTURE_CNT; i++) {
1558 if(is_enabled && indev->gesture_type[i] != LV_INDEV_GESTURE_NONE) {
1559 indev_act->cur_gesture = (lv_indev_gesture_type_t) i;
1560 if(send_event(LV_EVENT_GESTURE, indev_act) == LV_RESULT_INVALID) return;
1561 break;
1562 }
1563 }
1564#endif
1565 if(is_enabled) {
1566 if(send_event(LV_EVENT_RELEASED, indev_act) == LV_RESULT_INVALID) return;
1567 }
1568
1569 if(is_enabled) {
1570 if(scroll_obj == NULL) {
1571 if(indev->pointer.pressed) {
1572 if(indev->long_pr_sent == 0) {
1573 if(indev_proc_short_click(indev) == LV_RESULT_INVALID) return;
1574 }
1575 if(send_event(LV_EVENT_CLICKED, indev_act) == LV_RESULT_INVALID) return;
1576 }
1577 }
1578 else {
1579 lv_obj_send_event(scroll_obj, LV_EVENT_SCROLL_THROW_BEGIN, indev_act);
1580 if(indev_reset_check(indev)) return;
1581 }
1582 }
1583 indev->pointer.act_obj = NULL;
1584 indev->pr_timestamp = 0;
1585 indev->longpr_rep_timestamp = 0;
1586
1587 /*Get the transformed vector with this object*/
1588 if(scroll_obj) {
1589 int16_t angle = 0;
1590 int16_t scale_x = 256;
1591 int16_t scale_y = 256;
1592 lv_point_t pivot = { 0, 0 };
1593 lv_obj_t * parent = scroll_obj;
1594 while(parent) {
1595 angle += lv_obj_get_style_transform_rotation_internal(parent, LV_PART_MAIN);
1596 int32_t zoom_act_x = lv_obj_get_style_transform_scale_x_safe_internal(parent, LV_PART_MAIN);
1597 int32_t zoom_act_y = lv_obj_get_style_transform_scale_y_safe_internal(parent, LV_PART_MAIN);
1598 scale_x = (scale_x * zoom_act_x) >> 8;
1599 scale_y = (scale_x * zoom_act_y) >> 8;
1600 parent = lv_obj_get_parent(parent);
1601 }
1602
1603 if(scale_x == 0) {
1604 scale_x = 1;
1605 }
1606
1607 if(scale_y == 0) {
1608 scale_y = 1;
1609 }
1610
1611 if(angle != 0 || scale_y != LV_SCALE_NONE || scale_x != LV_SCALE_NONE) {
1612 angle = -angle;
1613 scale_x = (256 * 256) / scale_x;
1614 scale_y = (256 * 256) / scale_y;
1615 lv_point_transform(&indev->pointer.scroll_throw_vect, angle, scale_x, scale_y, &pivot, false);
1616 lv_point_transform(&indev->pointer.scroll_throw_vect_ori, angle, scale_x, scale_y, &pivot, false);
1617 }
1618 }
1619 }
1620
1621 if(scroll_obj) {
1622 if(!indev->scroll_throw_anim) {
1623 indev_scroll_throw_anim_start(indev);
1624 }
1625
1626 if(indev_reset_check(indev)) return;
1627 }

1630static lv_result_t indev_proc_short_click(lv_indev_t * indev)
1631{
1632 LV_ASSERT(indev != NULL);
1633 /*Update streak for clicks within small distance and short time*/
1634 indev->pointer.short_click_streak++;
1635 if(lv_tick_diff(indev->timestamp, indev->pointer.last_short_click_timestamp) > indev->double_click_time) {
1636 indev->pointer.short_click_streak = 1;
1637 }
1638 else if(indev->type == LV_INDEV_TYPE_POINTER || indev->type == LV_INDEV_TYPE_BUTTON) {
1639 int32_t dx = indev->pointer.last_short_click_point.x - indev->pointer.act_point.x;
1640 int32_t dy = indev->pointer.last_short_click_point.y - indev->pointer.act_point.y;
1641 if(dx * dx + dy * dy > indev->scroll_limit * indev->scroll_limit) indev->pointer.short_click_streak = 1;
1642 }
1643
1644 indev->pointer.last_short_click_timestamp = indev->timestamp;
1645 lv_indev_get_point(indev, &indev->pointer.last_short_click_point);
1646
1647 /*Simple short click*/
1648 lv_result_t res = send_event(LV_EVENT_SHORT_CLICKED, indev_act);
1649 if(res == LV_RESULT_INVALID) {
1650 return res;
1651 }
1652
1653 /*Cycle through single/double/triple click*/
1654 switch((indev->pointer.short_click_streak - 1) % 3) {
1655 case 0:
1656 return send_event(LV_EVENT_SINGLE_CLICKED, indev_act);
1657 case 1:
1658 return send_event(LV_EVENT_DOUBLE_CLICKED, indev_act);
1659 case 2:
1660 return send_event(LV_EVENT_TRIPLE_CLICKED, indev_act);
1661 }
1662 return res;
1663}

1665static void indev_proc_pointer_diff(lv_indev_t * indev)
1666{
1667 LV_ASSERT(indev != NULL);
1668 lv_obj_t * obj = indev->pointer.last_pressed;
1669 if(obj == NULL) return;
1670 if(indev->pointer.diff == 0) return;
1671
1672 indev_obj_act = obj;
1673
1674 bool editable = lv_obj_is_editable(obj);
1675
1676 if(editable) {
1677 uint32_t indev_sensitivity = indev->rotary_sensitivity;
1678 uint32_t obj_sensitivity = lv_obj_get_style_rotary_sensitivity_internal(indev_obj_act, LV_PART_MAIN);
1679 int32_t diff = (int32_t)((int32_t)indev->pointer.diff * indev_sensitivity * obj_sensitivity + 32768) >> 16;
1680 send_event(LV_EVENT_ROTARY, &diff);
1681 }
1682 else {
1683
1684 int32_t vect = indev->pointer.diff > 0 ? indev->scroll_limit : -indev->scroll_limit;
1685 indev->pointer.vect.y = vect;
1686 indev->pointer.act_obj = obj;
1687 lv_obj_t * scroll_obj = lv_indev_find_scroll_obj(indev);
1688 if(scroll_obj == NULL) return;
1689 uint32_t indev_sensitivity = indev->rotary_sensitivity;
1690 uint32_t obj_sensitivity = lv_obj_get_style_rotary_sensitivity_internal(scroll_obj, LV_PART_MAIN);
1691 int32_t diff = (int32_t)((int32_t)indev->pointer.diff * indev_sensitivity * obj_sensitivity + 32768) >> 16;
1692
1693 indev->pointer.scroll_throw_vect.y = diff;
1694 indev->pointer.scroll_throw_vect_ori.y = diff;
1695 lv_indev_scroll_handler(indev);
1696 }
