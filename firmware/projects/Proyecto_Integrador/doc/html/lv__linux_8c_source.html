60uint32_t lv_os_get_idle_percent(void)
61{
62 lv_linux_proc_stat_t proc_stat;
63 {
64 lv_result_t err = lv_read_proc_stat(&proc_stat);
65
66 if(err == LV_RESULT_INVALID) {
67 return UINT32_MAX;
68 }
69
70 for(size_t i = 0; i < LV_PROC_STAT_PARAMS_LEN; ++i) {
71 uint32_t delta =
72 proc_stat.buffer[i] - last_proc_stat.buffer[i];
73 /* Update old for next call*/
74 last_proc_stat.buffer[i] = proc_stat.buffer[i];
75 /* Store delta in new */
76 proc_stat.buffer[i] = delta;
77 }
78 }
79
80 /* From here onwards, there's no risk of overflowing as long as we call this function regularly */
81 const uint32_t total = lv_proc_stat_get_total(&proc_stat);
82
83 if(total == 0) {
84 return 0;
85 }
86
87 return (proc_stat.fields.idle * 100) / total;
88}

92uint32_t lv_os_get_proc_idle_percent(void)
93{
94 uint64_t self_current_time_ticks = 0;
95 lv_linux_proc_stat_t stat_current_system_total_ticks;
96 lv_linux_proc_stat_t stat_delta_system_ticks;
97
98 FILE * self = fopen(LV_UPTIME_MONITOR_SELF_FILE, "r");
99 if(!self) {
100 LV_LOG_ERROR("Failed to open " LV_UPTIME_MONITOR_SELF_FILE);
101 return UINT32_MAX;
102 }
103
104 char self_stat_buffer[LV_SELF_PROC_STAT_BUFFER_SIZE];
105
106 if(!fgets(self_stat_buffer, sizeof(self_stat_buffer), self)) {
107 fclose(self);
108 LV_LOG_ERROR("Failed to read " LV_UPTIME_MONITOR_SELF_FILE);
109 return UINT32_MAX;
110 }
111
112 fclose(self);
113
114 /* The comm field can contain spaces and parentheses, so we find the last ')'
115 * Skip the whitespace after finding the last ')' */
116 char * p = strrchr(self_stat_buffer, ')');
117 if(!p) {
118 LV_LOG_ERROR(LV_UPTIME_MONITOR_SELF_FILE " is missing the closing ')'");
119 return UINT32_MAX;
120 }
121 p++; /* move past the ')' */
122 while(*p && (*p == ' ' || *p == '\t'))
123 p++; /* skip whitespace after ')' */
124 if(!*p) {
125 LV_LOG_ERROR(LV_UPTIME_MONITOR_SELF_FILE " unexpectedly ends after the closing ')'");
126 return UINT32_MAX;
127 }
128
129 uint64_t utime = 0;
130 uint64_t stime = 0;
131
132 int scanned_items = sscanf(p,
133 "%*c " // state (field 3)
134 "%*d %*d %*d %*d %*d " // ppid, pgrp, session, tty_nr, tpgid (fields 4-8)
135 "%*u " // flags (field 9)
136 "%*u %*u %*u %*u " // minflt, cminflt, majflt, cmajflt (fields 10-13)
137 "%" SCNu64 " %" SCNu64, // utime, stime (fields 14-15)
138 &utime, &stime);
139
140 if(scanned_items != 2) {
141 LV_LOG_ERROR("Failed to parse utime/stime");
142 return UINT32_MAX;
143 }
144
145 self_current_time_ticks = utime + stime;
146
147 if(lv_read_proc_stat(&stat_current_system_total_ticks) != LV_RESULT_OK) {
148 LV_LOG_ERROR("lv_read_proc_stat failed");
149 return UINT32_MAX;
150 }
151
152 /* no delta on the first call so return 0, next call will have actual values*/
153 if(last_self_ticks == 0) {
154 last_self_ticks = self_current_time_ticks;
155 last_system_total_ticks_stat = stat_current_system_total_ticks;
156
157 return 100; /* 100% idle = 0% CPU usage*/
158 }
159
160 uint64_t delta_self_proc_ticks = self_current_time_ticks - last_self_ticks;
161
162 for(size_t i = 0; i < LV_PROC_STAT_PARAMS_LEN; ++i) {
163 stat_delta_system_ticks.buffer[i] = stat_current_system_total_ticks.buffer[i] - last_system_total_ticks_stat.buffer[i];
164 }
165
166 uint64_t delta_total_system_ticks = lv_proc_stat_get_total(&stat_delta_system_ticks);
167
168 last_self_ticks = self_current_time_ticks;
169 last_system_total_ticks_stat = stat_current_system_total_ticks;
170
171 if(delta_total_system_ticks == 0) return 100;
172
173 uint32_t cpu_usage_percent = (uint32_t)((delta_self_proc_ticks * 100ULL) / delta_total_system_ticks);
174
175 return 100 - cpu_usage_percent;
176}

184static lv_result_t lv_read_proc_stat(lv_linux_proc_stat_t * result)
185{
186 FILE * fp = fopen(LV_UPTIME_MONITOR_FILE, "r");
187
188 if(!fp) {
189 LV_LOG_ERROR("Failed to open " LV_UPTIME_MONITOR_FILE);
190 return LV_RESULT_INVALID;
191 }
192 const char * fmt = "cpu " LV_PROC_STAT_VAR_FORMAT LV_PROC_STAT_VAR_FORMAT
193 LV_PROC_STAT_VAR_FORMAT LV_PROC_STAT_VAR_FORMAT
194 LV_PROC_STAT_IGNORE_VAR_FORMAT LV_PROC_STAT_VAR_FORMAT
195 LV_PROC_STAT_VAR_FORMAT LV_PROC_STAT_VAR_FORMAT;
196
197 int err = fscanf(fp, fmt, &result->fields.user, &result->fields.nice,
198 &result->fields.system, &result->fields.idle,
199 &result->fields.irq, &result->fields.softirq,
200 &result->fields.steal);
201
202 fclose(fp);
203
204 if(err != LV_PROC_STAT_PARAMS_LEN) {
205 LV_LOG_ERROR("Failed to parse " LV_UPTIME_MONITOR_FILE);
206 return LV_RESULT_INVALID;
207 }
208 return LV_RESULT_OK;
209}

213 uint32_t sum = 0;
214 for(size_t i = 0; i < LV_PROC_STAT_PARAMS_LEN; ++i) {
215 sum += p->buffer[i];
216 }
217 return sum;
218}
