32def parse_style_props(path: Path) -> dict[int, str]:
33 """Parse _lv_style_id_t enum from lv_style.h."""
34 text = path.read_text()
35 m = re.search(r"enum\s+_lv_style_id_t\s*\{(.*?)\}", text, re.DOTALL)
36 if not m:
37 raise RuntimeError("Cannot find _lv_style_id_t enum")
38
39 props = {}
40 current_val = 0
41 for line in m.group(1).splitlines():
42 line = line.strip().rstrip(",")
43 if not line or line.startswith(("/*", "//", "*")):
44 continue
45 match = re.match(r"(LV_STYLE_\w+)\s*=\s*(0x[\da-fA-F]+|\d+)", line)
46 if match:
47 name, current_val = match.group(1), int(match.group(2), 0)
48 else:
49 match = re.match(r"(LV_STYLE_\w+)", line)
50 if not match:
51 continue
52 name = match.group(1)
53 if name in SKIP_PROPS:
54 current_val += 1
55 continue
56 props[current_val] = name.removeprefix("LV_STYLE_")
57 current_val += 1
58 return props

61def parse_parts(path: Path) -> dict[int, str]:
62 """Parse lv_part_t enum from lv_obj_style.h."""
63 text = path.read_text()
64 m = re.search(r"typedef\s+enum\s*\{(.*?)\}\s*lv_part_t", text, re.DOTALL)
65 if not m:
66 raise RuntimeError("Cannot find lv_part_t enum")
67 parts = {}
68 for line in m.group(1).splitlines():
69 match = re.match(r"\s*(LV_PART_\w+)\s*=\s*(0x[\da-fA-F]+|\d+)", line)
70 if not match:
71 continue
72 val = int(match.group(2), 0)
73 parts[val >> 16] = match.group(1).removeprefix("LV_PART_")
74 return parts

77def _parse_int_expr(expr: str) -> int | None:
78 """Parse a C integer expression: decimal, hex, or '1 << N'."""
79 expr = expr.strip()
80 m = re.match(r"^(0x[\da-fA-F]+|\d+)$", expr)
81 if m:
82 return int(m.group(1), 0)
83 m = re.match(r"^(\d+)\s*<<\s*(\d+)$", expr)
84 if m:
85 return int(m.group(1)) << int(m.group(2))
86 return None

89def parse_states(path: Path) -> dict[int, str]:
90 """Parse lv_state_t enum from lv_obj_style.h."""
91 text = path.read_text()
92 m = re.search(r"typedef\s+enum\s*\{(.*?)\}\s*lv_state_t", text, re.DOTALL)
93 if not m:
94 raise RuntimeError("Cannot find lv_state_t enum")
95 states = {}
96 for line in m.group(1).splitlines():
97 match = re.match(r"\s*(LV_STATE_\w+)\s*=\s*(.+?)(?:,|/)", line)
98 if not match:
99 continue
100 name = match.group(1)
101 if name in ("LV_STATE_DEFAULT", "LV_STATE_ANY"):
102 continue
103 val = _parse_int_expr(match.group(2))
104 if val is not None:
105 states[val] = name.removeprefix("LV_STATE_")
106 return states

109def parse_color_props(style_gen_h: Path, prop_map: dict[int, str]) -> set[int]:
110 """Identify color properties from lv_style_gen.h setter signatures."""
111 text = style_gen_h.read_text()
112 name_to_id = {v: k for k, v in prop_map.items()}
113 color_ids = set()
114 for match in re.finditer(
115 r"void\s+lv_style_set_(\w+)\s*\‍([^,]+,\s*lv_color_t", text
116 ):
117 prop_name = match.group(1).upper()
118 if prop_name in name_to_id:
119 color_ids.add(name_to_id[prop_name])
120 return color_ids

167def parse_enum_props(
168 style_gen_h: Path, prop_map: dict[int, str]
169) -> dict[int, dict[int, str]]:
170 """Map each enum-valued property to its {value: name} table.
171
172 The setter signature names the enum type; the type's own header is found by
173 searching the include tree for its typedef.
174 """
175 text = style_gen_h.read_text()
176 name_to_id = {v: k for k, v in prop_map.items()}
177 headers = list(LVGL_INC.rglob("*.h"))
178 enum_tables: dict[str, dict[int, str]] = {}
179 result = {}
180
181 for match in re.finditer(
182 r"void\s+lv_style_set_(\w+)\s*\‍([^,]+,\s*(lv_\w+_t)\s+value\s*\‍)", text
183 ):
184 prop_name, enum_type = match.group(1).upper(), match.group(2)
185 if prop_name not in name_to_id or enum_type in ("lv_color_t", "lv_opa_t"):
186 continue
187 if enum_type not in enum_tables:
188 enum_tables[enum_type] = _parse_named_enum(headers, enum_type)
189 if enum_tables[enum_type]:
190 result[name_to_id[prop_name]] = enum_tables[enum_type]
191 return result

194def _parse_named_enum(headers: list[Path], enum_type: str) -> dict[int, str]:
195 """Find and parse an enum typedef by name, {} if it cannot be found."""
196 needle = f"}} {enum_type};"
197 prefix = enum_type.removesuffix("_t").upper() + "_"
198 for header in headers:
199 if needle not in header.read_text():
200 continue
201 try:
202 return parse_enum(header, enum_type, prefix)
203 except RuntimeError:
204 return {}
205 return {}

209 """Read LV_COORD_TYPE_SHIFT, which defines the LV_PCT()/content encoding."""

216def generate(props, parts, states, color_ids, pointer_ids, src_ids, coord_ids,
217 bool_ids, enum_values, coord_shift) -> str:
218 """Generate Python source for the style constants module."""
219 lines = [
220 '"""',
221 "Auto-generated style constants from LVGL headers.",
222 "",
223 "Do not edit manually. Regenerate from the GDB script root with:",
224 " python3 scripts/generate_all.py",
225 '"""',
226 "",
227 ]
228 lines.append("STYLE_PROP_NAMES = {")
229 for k in sorted(props):
230 lines.append(f' {k}: "{props[k]}",')
231 lines.append("}")
232 lines.append("")
233
234 lines.append("PART_NAMES = {")
235 for k in sorted(parts):
236 lines.append(f' 0x{k:02X}: "{parts[k]}",')
237 lines.append("}")
238 lines.append("")
239
240 lines.append("STATE_FLAGS = {")
241 for k in sorted(states):
242 lines.append(f' 0x{k:04X}: "{states[k]}",')
243 lines.append("}")
244 lines.append("")
245
246 if color_ids:
247 lines.append("COLOR_PROPS = {")
248 for v in sorted(color_ids):
249 lines.append(f" {v}, # {props.get(v, '?')}")
250 lines.append("}")
251 else:
252 lines.append("COLOR_PROPS = set()")
253 lines.append("")
254
255 for name, ids in (
256 ("POINTER_PROPS", pointer_ids),
257 ("SRC_PROPS", src_ids),
258 ("COORD_PROPS", coord_ids),
259 ("BOOL_PROPS", bool_ids),
260 ):
261 if ids:
262 lines.append(f"{name} = {{")
263 for v in sorted(ids):
264 lines.append(f" {v}, # {props.get(v, '?')}")
265 lines.append("}")
266 else:
267 lines.append(f"{name} = set()")
268 lines.append("")
269
270 lines.append("ENUM_PROP_VALUES = {")
271 for prop_id in sorted(enum_values):
272 lines.append(f" {prop_id}: {{ # {props.get(prop_id, '?')}")
273 for val in sorted(enum_values[prop_id]):
274 lines.append(f' {val}: "{enum_values[prop_id][val]}",')
275 lines.append(" },")
276 lines.append("}")
277 lines.append("")
278
279 lines.append("# From LV_COORD_TYPE_SHIFT in lv_area.h")
280 lines.append(f"COORD_TYPE_SHIFT = {coord_shift}")
281 lines.append("")
282
283 return "\n".join(lines)
