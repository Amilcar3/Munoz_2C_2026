133static bool defer_init(void)
134{
135 if(ctx->mem != NULL && ctx->heap != NULL) {
136 return true;
137 }
138
139 if(lv_image_cache_is_enabled() == false) {
140 LV_LOG_INFO("Image cache is not initialized yet. Skipping deferred initialization. Because max_size is 0.");
141 return false;
142 }
143
144 ctx->mem_size = img_cache_p->max_size;
145 ctx->mem = malloc(ctx->mem_size);
146 LV_ASSERT_MALLOC(ctx->mem);
147
148 if(ctx->mem == NULL) {
149 LV_LOG_ERROR("Failed to allocate memory for image cache");
150 ctx->initialized = false;
151 return false;
152 }
153
154 lv_snprintf(ctx->name, sizeof(ctx->name), HEAP_NAME "[%-4" LV_PRIu32 "]", (uint32_t)gettid());
155
156 ctx->heap = mm_initialize(
157 ctx->name,
158 ctx->mem,
159 ctx->mem_size
160 );
161
162 struct mallinfo info = mm_mallinfo(ctx->heap);
163 ctx->heap_size = info.arena;
164
165 LV_LOG_USER("heap info:");
166 LV_LOG_USER(" heap: %p", (void *)ctx->heap);
167 LV_LOG_USER(" mem: %p", ctx->mem);
168 LV_LOG_USER(" mem_size: %" LV_PRIu32, ctx->mem_size);
169 LV_LOG_USER(" arena: %d", info.arena);
170 LV_LOG_USER(" ordblks: %d", info.ordblks);
171 LV_LOG_USER(" aordblks: %d", info.aordblks);
172 LV_LOG_USER(" mxordblk: %d", info.mxordblk);
173 LV_LOG_USER(" uordblks: %d", info.uordblks);
174 LV_LOG_USER(" fordblks: %d", info.fordblks);
175
176 ctx->initialized = true;
177 return true;
178}

193static void * malloc_cb(size_t size_bytes, lv_color_format_t color_format)
194{
195 LV_UNUSED(color_format);
196 if(ctx->independent_image_heap == true && ctx->initialized == false) {
197 if(defer_init() == false) return NULL;
198 }
199
200 /*Allocate larger memory to be sure it can be aligned as needed*/
201 size_bytes += LV_DRAW_BUF_ALIGN - 1;
202 uint32_t cache_max_size = lv_cache_get_max_size(img_cache_p, NULL);
203
204 if(lv_cache_is_enabled(img_cache_p) && size_bytes > cache_max_size) {
205 LV_LOG_ERROR("data size (%" LV_PRIu32 ") is larger than max size (%" LV_PRIu32 ")",
206 (uint32_t)size_bytes,
207 cache_max_size);
208 return NULL;
209 }
210
211 while(1) {
212 void * mem = NULL;
213 if(ctx->independent_image_heap) {
214 mem = mm_malloc(ctx->heap, size_bytes);
215 }
216 else if((!lv_cache_is_enabled(img_cache_p)) || (lv_cache_get_size(img_cache_p, NULL) + size_bytes < cache_max_size)) {
217 mem = ctx->malloc_cb(size_bytes, color_format);
218 }
219 if(mem) return mem;
220 LV_LOG_INFO("appears to be out of memory. attempting to evict one cache entry. with allocated size %" LV_PRIu32,
221 (uint32_t)size_bytes);
222 bool evict_res = lv_cache_evict_one(img_cache_p, NULL);
223 if(evict_res == false) {
224 LV_LOG_ERROR("failed to evict one cache entry");
225 heap_memdump(ctx->heap);
226 return NULL;
227 }
228 }
229}
