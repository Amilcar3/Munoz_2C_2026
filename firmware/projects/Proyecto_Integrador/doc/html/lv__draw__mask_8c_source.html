51void LV_ATTRIBUTE_FAST_MEM lv_draw_mask_rect(lv_layer_t * layer, const lv_draw_mask_rect_dsc_t * dsc)
52{
53 LV_CHECK_ARG(layer != NULL, return);
54 LV_CHECK_ARG(dsc != NULL, return);
55
56 if(!lv_color_format_has_alpha(layer->color_format)) {
57 LV_LOG_WARN("Only layers with alpha channel can be masked");
58 return;
59 }
60 LV_PROFILER_DRAW_BEGIN;
61
62 lv_draw_task_t * t = lv_draw_add_task(layer, &layer->buf_area, LV_DRAW_TASK_TYPE_MASK_RECTANGLE);
63
64 lv_memcpy(t->draw_dsc, dsc, sizeof(*dsc));
65
66 lv_draw_dsc_base_t * base_dsc = t->draw_dsc;
67 base_dsc->layer = layer;
68
69 if(base_dsc->obj && lv_obj_is_send_draw_task_events(base_dsc->obj)) {
70 /*Disable sending LV_EVENT_DRAW_TASK_ADDED first to avoid triggering recursive
71 *event calls due draw task adds in the event*/
72 lv_obj_set_send_draw_task_events(base_dsc->obj, false);
73 lv_obj_send_event(dsc->base.obj, LV_EVENT_DRAW_TASK_ADDED, t);
74 lv_obj_set_send_draw_task_events(base_dsc->obj, true);
75 }
76
77 lv_draw_finalize_task_creation(layer, t);
78 LV_PROFILER_DRAW_END;
79}
