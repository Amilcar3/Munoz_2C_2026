51void test_keyboard_mode_switch_keeps_selection_in_place(void)
52{
53 /*Regression test for https://github.com/lvgl/lvgl/issues/10118:
54 *switching modes must keep the selection on the key in that position,
55 *not move it to an unrelated button (backspace).*/
56 lv_obj_t * keyboard = lv_keyboard_create(active_screen);
57 lv_obj_set_size(keyboard, LV_PCT(100), LV_PCT(50));
58 lv_keyboard_set_mode(keyboard, LV_KEYBOARD_MODE_SPECIAL);
59 lv_obj_update_layout(keyboard);
60
61 /*Select the "abc" mode-switch button and press it*/
62 uint32_t abc_btn = keyboard_find_button(keyboard, "abc");
63 TEST_ASSERT_NOT_EQUAL(LV_BUTTONMATRIX_BUTTON_NONE, abc_btn);
64 lv_buttonmatrix_set_selected_button(keyboard, abc_btn);
65 lv_obj_send_event(keyboard, LV_EVENT_VALUE_CHANGED, NULL);
66
67 /*The selection follows the position: it lands on the "ABC" toggle that
68 *took the "abc" button's spot, not the backspace key at the old index.*/
69 TEST_ASSERT_EQUAL_INT(LV_KEYBOARD_MODE_TEXT_LOWER, lv_keyboard_get_mode(keyboard));
70 uint32_t sel = lv_keyboard_get_selected_button(keyboard);
71 TEST_ASSERT_EQUAL_STRING("ABC", lv_keyboard_get_button_text(keyboard, sel));
72}
