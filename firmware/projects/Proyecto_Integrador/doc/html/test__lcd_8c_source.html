228static void lcd_ft81x_spi_cb(lv_display_t * disp, lv_ft81x_spi_operation_t operation, void * data, uint32_t length)
229{
230 TEST_ASSERT_NOT_NULL(disp);
231
232 test_disp_resolution_with_rotation(disp);
233
234 static uint32_t reg_addr = 0;
235
236 /* Mocked FT81X registers */
237#define REG_ID 0x302000UL
238#define REG_CPURESET 0x302020UL
239#define REG_CMD_READ 0x3020f8UL
240#define REG_CMD_WRITE 0x3020fcUL
241
242 switch(operation) {
243 case LV_FT81X_SPI_OPERATION_CS_ASSERT:
244 break;
245
246 case LV_FT81X_SPI_OPERATION_CS_DEASSERT:
247 break;
248
249 case LV_FT81X_SPI_OPERATION_SEND:
250 reg_addr = 0;
251 lv_memcpy(&reg_addr, data, LV_MIN(length, sizeof(reg_addr)));
252 break;
253
254 case LV_FT81X_SPI_OPERATION_RECEIVE:
255 if(reg_addr == lv_ft81x_encode_read_address(REG_ID)) {
256 *(uint8_t *)data = 0x7c;
257 }
258 else if(reg_addr == lv_ft81x_encode_read_address(REG_CPURESET)) {
259 *(uint8_t *)data = 0;
260 }
261 else if(reg_addr == lv_ft81x_encode_read_address(REG_CMD_READ)) {
262 *(uint16_t *)data = 0;
263 }
264 else if(reg_addr == lv_ft81x_encode_read_address(REG_CMD_WRITE)) {
265 *(uint16_t *)data = 0;
266 }
267 break;
268
269 default:
270 TEST_FAIL_MESSAGE("Invalid operation");
271 break;
272 }
273
274 if(data) {
275 TEST_ASSERT_GREATER_THAN_UINT32(0, length);
276 }
277}
