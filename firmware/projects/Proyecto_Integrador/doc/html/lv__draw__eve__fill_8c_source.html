28void lv_draw_eve_fill(lv_draw_task_t * t, const lv_draw_fill_dsc_t * dsc, const lv_area_t * coords)
29{
30
31 int32_t rad = dsc->radius;
32 int32_t bg_w = lv_area_get_width(coords);
33 int32_t bg_h = lv_area_get_height(coords);
34 int32_t real_radius = LV_MIN3(bg_w / 2, bg_h / 2, rad);
35
36 lv_eve_scissor(t->clip_area.x1, t->clip_area.y1, t->clip_area.x2, t->clip_area.y2);
37 lv_eve_save_context();
38
39 lv_eve_color(dsc->color);
40 lv_eve_color_opa(dsc->opa);
41
42 if(bg_w == bg_h && rad == LV_RADIUS_CIRCLE) {
43 lv_eve_draw_circle_simple(coords->x1 + (bg_w / 2), coords->y1 + (bg_h / 2), real_radius);
44 }
45 else {
46 lv_eve_draw_rect_simple(coords->x1, coords->y1, coords->x2, coords->y2, real_radius);
47 }
48
49 lv_eve_restore_context();
50}

59void lv_draw_eve_border(lv_draw_task_t * t, const lv_draw_border_dsc_t * dsc, const lv_area_t * coords)
60{
61
62 if(dsc->opa <= LV_OPA_MIN) return;
63 if(dsc->width == 0) return;
64 if(dsc->side == LV_BORDER_SIDE_NONE) return;
65
66 int32_t coords_w = lv_area_get_width(coords);
67 int32_t coords_h = lv_area_get_height(coords);
68 int32_t rout = dsc->radius;
69 int32_t short_side = LV_MIN(coords_w, coords_h);
70 if(rout > short_side >> 1) rout = short_side >> 1;
71
72 /*Get the inner area*/
73 lv_area_t area_inner;
74 area_inner = *coords;
75 area_inner.x1 += ((dsc->side & LV_BORDER_SIDE_LEFT) ? dsc->width : - (dsc->width));
76 area_inner.x2 -= ((dsc->side & LV_BORDER_SIDE_RIGHT) ? dsc->width : - (dsc->width));
77 area_inner.y1 += ((dsc->side & LV_BORDER_SIDE_TOP) ? dsc->width : - (dsc->width));
78 area_inner.y2 -= ((dsc->side & LV_BORDER_SIDE_BOTTOM) ? dsc->width : - (dsc->width));
79
80 int32_t rin = rout - dsc->width;
81 if(rin < 0) rin = 0;
82
83 lv_eve_save_context();
84
85 lv_eve_scissor(t->clip_area.x1, t->clip_area.y1, t->clip_area.x2, t->clip_area.y2);
86
87 lv_eve_color(dsc->color);
88 lv_eve_color_opa(dsc->opa);
89
90 lv_eve_color_mask(0, 0, 0, 1);
91 lv_eve_stencil_func(EVE_ALWAYS, 0, 1);
92 lv_eve_stencil_op(EVE_REPLACE, EVE_REPLACE);
93 lv_eve_draw_rect_simple(coords->x1, coords->y1, coords->x2, coords->y2, 0);
94
95 lv_eve_blend_func(EVE_ONE, EVE_ZERO);
96 lv_eve_draw_rect_simple(area_inner.x1 - 2, area_inner.y1 - 1, area_inner.x2 + 1, area_inner.y2 + 2, rin);
97
98 lv_eve_stencil_func(EVE_ALWAYS, 1, 1);
99 lv_eve_stencil_op(EVE_REPLACE, EVE_REPLACE);
100 lv_eve_blend_func(EVE_ZERO, EVE_ONE_MINUS_SRC_ALPHA);
101 lv_eve_color_opa(255);
102 lv_eve_draw_rect_simple(area_inner.x1, area_inner.y1, area_inner.x2, area_inner.y2, rin);
103
104 lv_eve_color_mask(1, 1, 1, 1);
105
106 if(dsc->side == LV_BORDER_SIDE_FULL) {
107 lv_eve_blend_func(EVE_DST_ALPHA, EVE_ONE_MINUS_DST_ALPHA);
108 lv_eve_draw_rect_simple(area_inner.x1, area_inner.y1, area_inner.x2, area_inner.y2, rin);
109 }
110
111 lv_eve_stencil_func(EVE_NOTEQUAL, 1, 255);
112 lv_eve_blend_func(EVE_SRC_ALPHA, EVE_ONE_MINUS_SRC_ALPHA);
113
114 lv_eve_color_opa(dsc->opa);
115 lv_eve_draw_rect_simple(coords->x1, coords->y1, coords->x2, coords->y2, rout);
116
117 lv_eve_restore_context();
118}
