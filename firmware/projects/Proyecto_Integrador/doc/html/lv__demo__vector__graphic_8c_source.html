51static void draw_gradient(lv_draw_vector_dsc_t * ctx, lv_vector_path_t * path)
52{
53 lv_vector_path_clear(path);
54 lv_draw_vector_dsc_identity(ctx);
55

59 lv_vector_path_close(path);
60

62 lv_memzero(stops, sizeof(stops));
63 stops[0].color = lv_color_hex(0xff0000);

73 lv_draw_vector_dsc_set_fill_transform(ctx, &mt);

76 lv_draw_vector_dsc_set_fill_gradient_color_stops(ctx, stops, 2);

188static void draw_arc(lv_draw_vector_dsc_t * ctx, lv_vector_path_t * path)
189{
190 lv_vector_path_clear(path);
191 lv_draw_vector_dsc_identity(ctx);
192
193 lv_area_t rect = {100, 0, 150, 50};
194 lv_draw_vector_dsc_set_fill_color(ctx, lv_color_lighten(lv_color_black(), 50));
195 lv_draw_vector_dsc_clear_area(ctx, &rect); // clear screen
196
197 lv_fpoint_t p = {100, 50}; /* Center */
198 lv_draw_vector_dsc_set_stroke_color(ctx, lv_color_make(0x00, 0xff, 0xff));
199 lv_draw_vector_dsc_set_stroke_opa(ctx, LV_OPA_COVER);
200 lv_draw_vector_dsc_set_stroke_width(ctx, 2.0f);
201 lv_draw_vector_dsc_set_stroke_dash(ctx, NULL, 0);
202 lv_draw_vector_dsc_set_fill_color32(ctx, lv_color32_make(0xFF, 0x00, 0x00, 0x80));
203
204 lv_vector_path_move_to(path, &p);
205 lv_fpoint_t temp = {p.x + 50, p.y};
206 lv_vector_path_line_to(path, &temp);
207 lv_vector_path_append_arc(path, &p, 50, 0, -90, false);
208 lv_vector_path_line_to(path, &p);
209 lv_vector_path_close(path);
210
211 lv_draw_vector_dsc_add_path(ctx, path);
212
213 /* Below code has same effect as above one. */
214 lv_vector_path_clear(path);
215 lv_vector_path_append_arc(path, &p, 50, 45, 45, true);
216 lv_draw_vector_dsc_add_path(ctx, path); // draw a path
217}

219static void draw_vector(lv_layer_t * layer)
220{
221 lv_draw_vector_dsc_t * ctx = lv_draw_vector_dsc_create(layer);
222
223 lv_area_t rect = {0, 100, 300, 300};
224 lv_draw_vector_dsc_set_fill_color(ctx, lv_color_lighten(lv_color_black(), 50));
225 lv_draw_vector_dsc_clear_area(ctx, &rect); // clear screen
226
227 lv_vector_path_t * path = lv_vector_path_create(LV_VECTOR_PATH_QUALITY_MEDIUM);
228
229 draw_shapes(ctx, path);
230 draw_lines(ctx, path);
231 draw_pattern(ctx, path);
232 draw_radial_gradient(ctx, path);
233 draw_gradient(ctx, path);
234 draw_blend(ctx, path);
235 draw_arc(ctx, path);
236 lv_draw_vector(ctx); // submit draw
237 lv_vector_path_delete(path);
238 lv_draw_vector_dsc_delete(ctx);
239}
