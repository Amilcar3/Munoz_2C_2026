55void lv_draw_eve_image(lv_draw_task_t * t, const lv_draw_image_dsc_t * draw_dsc, const lv_area_t * coords)
56{
57 if(!lv_draw_eve_image_src_check(draw_dsc->src)) {
58 return;
59 }
60
61 const lv_image_dsc_t * img_dsc = draw_dsc->src;
62
63 int32_t src_w = img_dsc->header.w;
64 int32_t src_h = img_dsc->header.h;
65 int32_t src_stride = img_dsc->header.stride;
66 lv_color_format_t src_cf = img_dsc->header.cf;
67
68 if(src_stride == 0) {
69 src_stride = src_w * lv_color_format_get_size(src_cf);
70 }
71
72 uint8_t eve_format;
73 int32_t eve_stride;
74
75 switch(src_cf) {
76 case LV_COLOR_FORMAT_L8:
77 eve_format = EVE_L8;
78 eve_stride = src_stride;
79 break;
80 case LV_COLOR_FORMAT_RGB565:
81 eve_format = EVE_RGB565;
82 eve_stride = src_stride;
83 break;
84 case LV_COLOR_FORMAT_RGB565A8:
85 case LV_COLOR_FORMAT_ARGB8888:
86 eve_format = EVE_ARGB4;
87 eve_stride = src_w * 2;
88 break;
89 default :
90 LV_ASSERT_FORMAT_MSG(0, "Invalid color format 0x%02x", src_cf);
91 return;
92 }
93
94 uint32_t ramg_addr = lv_draw_eve_image_upload_image(true, img_dsc);
95 if(ramg_addr == LV_DRAW_EVE_RAMG_OUT_OF_RAMG) {
96 LV_LOG_WARN("Could not load image because space could not be allocated in RAM_G.");
97 return;
98 }
99
100 lv_eve_scissor(t->clip_area.x1, t->clip_area.y1, t->clip_area.x2, t->clip_area.y2);
101
102 lv_eve_save_context();
103
104 lv_eve_color_opa(draw_dsc->opa);
105
106 if(draw_dsc->recolor_opa > LV_OPA_MIN) {
107 lv_eve_color(lv_color_mix(draw_dsc->recolor, lv_color_white(), draw_dsc->recolor_opa));
108 }
109
110 lv_eve_primitive(LV_EVE_PRIMITIVE_BITMAPS);
111 lv_eve_bitmap_source(ramg_addr);
112 /*real height and width is mandatory for rotation and scale (Clip Area)*/
113 lv_eve_bitmap_size(EVE_NEAREST, EVE_BORDER, EVE_BORDER, src_w, src_h);
114
115 lv_eve_bitmap_layout(eve_format, eve_stride, src_h);
116
117 if(draw_dsc->rotation || draw_dsc->scale_x != LV_SCALE_NONE || draw_dsc->scale_y != LV_SCALE_NONE) {
118 EVE_cmd_dl_burst(CMD_LOADIDENTITY);
119
120 EVE_cmd_translate_burst(F16(coords->x1 - t->clip_area.x1 + draw_dsc->pivot.x),
121 F16(coords->y1 - t->clip_area.y1 + draw_dsc->pivot.y));
122 if(draw_dsc->scale_x != LV_SCALE_NONE || draw_dsc->scale_y != LV_SCALE_NONE) {
123 /*Image Scale*/
124 EVE_cmd_scale_burst(F16_SCALE_DIV_256(draw_dsc->scale_x), F16_SCALE_DIV_256(draw_dsc->scale_y));
125 }
126 if(draw_dsc->rotation != 0) {
127 /*Image Rotate*/
128 EVE_cmd_rotate_burst(DEGREES(draw_dsc->rotation));
129 }
130 EVE_cmd_translate_burst(-F16(draw_dsc->pivot.x), -F16(draw_dsc->pivot.y));
131 EVE_cmd_dl_burst(CMD_SETMATRIX);
132 EVE_cmd_dl_burst(CMD_LOADIDENTITY);
133 lv_eve_vertex_2f(t->clip_area.x1, t->clip_area.y1);
134 }
135 else {
136 lv_eve_vertex_2f(coords->x1, coords->y1);
137 }
138 lv_eve_restore_context();
139}

141bool lv_draw_eve_image_src_check(const void * src)
142{
143 if(lv_image_src_get_type(src) != LV_IMAGE_SRC_VARIABLE) {
144 LV_LOG_WARN("lv_draw_eve can only render images from variables (not files or symbols) for now.");
145 return false;
146 }
147
148 const lv_image_dsc_t * img_dsc = src;
149
150 switch(img_dsc->header.cf) {
151 case LV_COLOR_FORMAT_L8:
152 case LV_COLOR_FORMAT_RGB565:
153 case LV_COLOR_FORMAT_RGB565A8:
154 case LV_COLOR_FORMAT_ARGB8888:
155 break;
156 default :
157 LV_LOG_WARN("lv_draw_eve can only render L8, RGB565, RGB565A8, and ARGB8888 images for now.");
158 return false;
159 }
160
161 return true;
162}

164uint32_t lv_draw_eve_image_upload_image(bool burst_is_active, const lv_image_dsc_t * img_dsc)
165{
166 const uint8_t * src_buf = img_dsc->data;
167 int32_t src_w = img_dsc->header.w;
168 int32_t src_h = img_dsc->header.h;
169 int32_t src_stride = img_dsc->header.stride;
170 lv_color_format_t src_cf = img_dsc->header.cf;
171
172 if(src_stride == 0) {
173 src_stride = src_w * lv_color_format_get_size(src_cf);
174 }
175
176 int32_t eve_stride;
177 uint8_t eve_alignment;
178
179 switch(src_cf) {
180 case LV_COLOR_FORMAT_L8:
181 eve_stride = src_stride;
182 eve_alignment = 1;
183 break;
184 case LV_COLOR_FORMAT_RGB565:
185 eve_stride = src_stride;
186 eve_alignment = 2;
187 break;
188 case LV_COLOR_FORMAT_RGB565A8:
189 case LV_COLOR_FORMAT_ARGB8888:
190 eve_stride = src_w * 2;
191 eve_alignment = 2;
192 break;
193 default :
194 LV_ASSERT_FORMAT_MSG(0, "Invalid color format 0x%02x", src_cf);
195 return 0;
196 }
197
198 int32_t eve_size = eve_stride * src_h;
199
200 uint32_t ramg_addr;
201 bool img_is_loaded = lv_draw_eve_ramg_get_addr(&ramg_addr, (uintptr_t) src_buf, eve_size, eve_alignment);
202
203 /* New image to load */
204 if(!img_is_loaded && ramg_addr != LV_DRAW_EVE_RAMG_OUT_OF_RAMG) {
205
206 /* Load image to RAM_G */
207
208 if(burst_is_active) {
209 EVE_end_cmd_burst();
210 }
211
212 switch(src_cf) {
213 case LV_COLOR_FORMAT_L8:
214 case LV_COLOR_FORMAT_RGB565:
215 EVE_memWrite_flash_buffer(ramg_addr, src_buf, eve_size);
216 break;
217 case LV_COLOR_FORMAT_RGB565A8: {
218 uint8_t * tmp_buf = lv_malloc(eve_stride);
219 LV_ASSERT_MALLOC(tmp_buf);
220 const uint8_t * src_alpha_buf = src_buf + src_h * src_stride;
221 int32_t src_alpha_stride = src_stride / 2;
222 for(uint32_t y = 0; y < src_h; y++) {
223 convert_row_rgb565a8_to_argb4444(src_buf + y * src_stride, src_alpha_buf + y * src_alpha_stride, tmp_buf, src_w);
224 EVE_memWrite_flash_buffer(ramg_addr + y * eve_stride, tmp_buf, eve_stride);
225 }
226 lv_free(tmp_buf);
227 break;
228 }
229 case LV_COLOR_FORMAT_ARGB8888: {
230 uint8_t * tmp_buf = lv_malloc(eve_stride);
231 LV_ASSERT_MALLOC(tmp_buf);
232 for(uint32_t y = 0; y < src_h; y++) {
233 convert_row_argb8888_to_argb4444(src_buf + y * src_stride, tmp_buf, src_w);
234 EVE_memWrite_flash_buffer(ramg_addr + y * eve_stride, tmp_buf, eve_stride);
235 }
236 lv_free(tmp_buf);
237 break;
238 }
239 default:
240 LV_ASSERT_FORMAT_MSG(0, "Invalid color format 0x%02x", src_cf);
241 return 0;
242 }
243
244 if(burst_is_active) {
245 EVE_start_cmd_burst();
246 }
247 }
248
249 return ramg_addr;
250}

259static void convert_row_rgb565a8_to_argb4444(const uint8_t * src, const uint8_t * src_alpha, uint8_t * dst,
260 uint32_t width)
261{
262 for(uint32_t x = 0; x < width; x++) {
263 uint16_t rgb565 = ((const uint16_t *) src)[x];
264
265 uint8_t r5 = (rgb565 >> 11) & 0x1F;
266 uint8_t g6 = (rgb565 >> 5) & 0x3F;
267 uint8_t b5 = rgb565 & 0x1F;
268 uint8_t alpha = src_alpha[x];
269
270 uint8_t r4 = r5 >> 1;
271 uint8_t g4 = g6 >> 2;
272 uint8_t b4 = b5 >> 1;
273 uint8_t a4 = alpha >> 4;
274
275 uint16_t argb4444 = (a4 << 12) | (r4 << 8) | (g4 << 4) | b4;
276
277 dst[2 * x] = argb4444 & 0xFF;
278 dst[2 * x + 1] = (argb4444 >> 8) & 0xFF;
279 }
280}

282static void convert_row_argb8888_to_argb4444(const uint8_t * src, uint8_t * dst, uint32_t width)
283{
284 for(uint32_t x = 0; x < width; x++) {
285 uint8_t blue = src[4 * x];
286 uint8_t green = src[4 * x + 1];
287 uint8_t red = src[4 * x + 2];
288 uint8_t alpha = src[4 * x + 3];
289
290 uint8_t r4 = red >> 4;
291 uint8_t g4 = green >> 4;
292 uint8_t b4 = blue >> 4;
293 uint8_t a4 = alpha >> 4;
294
295 uint16_t argb4444 = (a4 << 12) | (r4 << 8) | (g4 << 4) | b4;
296
297 dst[2 * x] = argb4444 & 0xFF;
298 dst[2 * x + 1] = (argb4444 >> 8) & 0xFF;
299 }
300}
