36void lv_draw_ppa_init(void)
37{
38 esp_err_t res;
39 ppa_client_config_t cfg = {0};
40
41 /* Create draw unit */
42 lv_draw_buf_ppa_init_handlers();
43 lv_draw_ppa_unit_t * draw_ppa_unit = lv_draw_create_unit(sizeof(lv_draw_ppa_unit_t));
44 draw_ppa_unit->base_unit.evaluate_cb = ppa_evaluate;
45 draw_ppa_unit->base_unit.dispatch_cb = ppa_dispatch;
46 draw_ppa_unit->base_unit.delete_cb = ppa_delete;
47 draw_ppa_unit->base_unit.name = "ESP_PPA";
48
49 /* Register SRM client */
50 cfg.oper_type = PPA_OPERATION_SRM;
51 cfg.max_pending_trans_num = 1;
52#if (LV_PPA_BURST_LENGTH == 128)
53 cfg.data_burst_length = PPA_DATA_BURST_LENGTH_128;
54#elif (LV_PPA_BURST_LENGTH == 64)
55 cfg.data_burst_length = PPA_DATA_BURST_LENGTH_64;
56#elif (LV_PPA_BURST_LENGTH == 32)
57 cfg.data_burst_length = PPA_DATA_BURST_LENGTH_32;
58#elif (LV_PPA_BURST_LENGTH == 16)
59 cfg.data_burst_length = PPA_DATA_BURST_LENGTH_16;
60#elif (LV_PPA_BURST_LENGTH == 8)
61 cfg.data_burst_length = PPA_DATA_BURST_LENGTH_8;
62#else
63#error "Invalid burst length selection for PPA"
64#endif
65
66 res = ppa_register_client(&cfg, &draw_ppa_unit->srm_client);
67 LV_ASSERT(res == ESP_OK);
68
69 /* Register Fill client */
70 cfg.oper_type = PPA_OPERATION_FILL;
71 res = ppa_register_client(&cfg, &draw_ppa_unit->fill_client);
72 LV_ASSERT(res == ESP_OK);
73
74 /* Register Blend client */
75 cfg.oper_type = PPA_OPERATION_BLEND;
76
77 res = ppa_register_client(&cfg, &draw_ppa_unit->blend_client);
78 LV_ASSERT(res == ESP_OK);
79}

89static int32_t ppa_evaluate(lv_draw_unit_t * u, lv_draw_task_t * t)
90{
91 LV_UNUSED(u);
92 const lv_draw_dsc_base_t * base = (lv_draw_dsc_base_t *)t->draw_dsc;
93
94 if(!ppa_dest_cf_supported(base->layer->color_format)) return 0;
95
96 switch(t->type) {
97 case LV_DRAW_TASK_TYPE_FILL: {
98 const lv_draw_fill_dsc_t * dsc = (lv_draw_fill_dsc_t *)t->draw_dsc;
99 if((dsc->radius != 0 || dsc->grad.dir != LV_GRAD_DIR_NONE)) return 0;
100 if(dsc->opa <= (lv_opa_t)LV_OPA_MAX) return 0;
101
102 if(t->preference_score > DRAW_UNIT_PPA_PREF_SCORE) {
103 t->preference_score = DRAW_UNIT_PPA_PREF_SCORE;
104 t->preferred_draw_unit_id = DRAW_UNIT_ID_PPA;
105 }
106 return 1;
107 }
108
109#if LV_USE_PPA_IMG
110 case LV_DRAW_TASK_TYPE_IMAGE: {
111 lv_draw_image_dsc_t * dsc = t->draw_dsc;
112 if(!(dsc->header.cf < LV_COLOR_FORMAT_PROPRIETARY_START
113 && dsc->clip_radius == 0
114 && dsc->bitmap_mask_src == NULL
115 && dsc->sup == NULL
116 && dsc->tile == 0
117 && dsc->blend_mode == LV_BLEND_MODE_NORMAL
118 && dsc->recolor_opa <= LV_OPA_MIN
119 && dsc->opa >= (lv_opa_t)LV_OPA_MAX
120 && dsc->skew_y == 0
121 && dsc->skew_x == 0
122 && dsc->scale_x == 256
123 && dsc->scale_y == 256
124 && dsc->rotation == 0
125 && lv_image_src_get_type(dsc->src) == LV_IMAGE_SRC_VARIABLE
126 && (dsc->header.cf == LV_COLOR_FORMAT_RGB888
127 || dsc->header.cf == LV_COLOR_FORMAT_RGB565)
128 && (dsc->base.layer->color_format == LV_COLOR_FORMAT_RGB888
129 || dsc->base.layer->color_format == LV_COLOR_FORMAT_RGB565))) {
130 return 0;
131 }
132
133 if(t->preference_score > DRAW_UNIT_PPA_PREF_SCORE) {
134 t->preference_score = DRAW_UNIT_PPA_PREF_SCORE;
135 t->preferred_draw_unit_id = DRAW_UNIT_ID_PPA;
136 }
137 return 1;
138 }
139#endif
140 default:
141 return 0;
142 }

145static bool other_unit_busy(const lv_layer_t * layer, const lv_draw_unit_t * self)
146{
147 lv_draw_task_t * t = layer->draw_task_head;
148 while(t) {
149 if(t->state == LV_DRAW_TASK_STATE_IN_PROGRESS && t->draw_unit != self) return true;
150 t = t->next;
151 }
152 return false;
153}

155static int32_t ppa_dispatch(lv_draw_unit_t * draw_unit, lv_layer_t * layer)
156{
157 lv_draw_ppa_unit_t * u = (lv_draw_ppa_unit_t *)draw_unit;
158 if(u->task_act) {
159 return LV_DRAW_UNIT_IDLE;
160 }
161
162 /* Row maintenance would discard another unit's pixels in the same rows */
163 if(other_unit_busy(layer, draw_unit)) {
164 return LV_DRAW_UNIT_IDLE;
165 }
166
167 lv_draw_task_t * t = lv_draw_get_available_task(layer, NULL, DRAW_UNIT_ID_PPA);
168 if(!t || t->preferred_draw_unit_id != DRAW_UNIT_ID_PPA) return LV_DRAW_UNIT_IDLE;
169 if(lv_draw_layer_alloc_buf(layer) == NULL) {
170 t->state = LV_DRAW_TASK_STATE_FAILED;
171 return LV_DRAW_UNIT_IDLE;
172 }
173
174 t->state = LV_DRAW_TASK_STATE_IN_PROGRESS;
175 u->task_act = t;
176 u->task_act->draw_unit = draw_unit;
177 u->img_sw_fallback = false;
178
179 ppa_execute_drawing(u);
180
181 u->task_act->state = LV_DRAW_TASK_STATE_FINISHED;
182 u->task_act = NULL;
183 lv_draw_dispatch_request();
184
185 return 1;
186}

197static void ppa_execute_drawing(lv_draw_ppa_unit_t * u)
198{
199 lv_draw_task_t * t = u->task_act;
200 lv_layer_t * layer = t->target_layer;
201 lv_draw_buf_t * buf = layer->draw_buf;
202 lv_area_t area;
203
204 if(!lv_area_intersect(&area, &t->area, &t->clip_area)) return;
205
206 /* The handlers take an area relative to the buffer */
207 lv_area_t buf_area = area;
208 lv_area_move(&buf_area, -layer->buf_area.x1, -layer->buf_area.y1);
209
210 lv_draw_buf_flush_cache(buf, &buf_area);
211
212 switch(t->type) {
213 case LV_DRAW_TASK_TYPE_FILL:
214 lv_draw_ppa_fill(t, (lv_draw_fill_dsc_t *)t->draw_dsc, &area);
215 break;
216 case LV_DRAW_TASK_TYPE_IMAGE:
217 lv_draw_ppa_img(t, (lv_draw_image_dsc_t *)t->draw_dsc, &t->area);
218 break;
219 default:
220 LV_ASSERT_FORMAT_MSG(false, "Invalid draw task type: %d", t->type);
221 return;
222 }
223
224 /* The software fallback wrote through the cache, so its lines must survive */
225 if(!u->img_sw_fallback) lv_draw_buf_invalidate_cache(buf, &buf_area);
226}
