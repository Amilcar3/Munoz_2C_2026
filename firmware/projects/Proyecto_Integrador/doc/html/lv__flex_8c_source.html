143 LV_ASSERT(cont != NULL);
144 LV_ASSERT(f != NULL);
145 lv_flex_flow_t flow = lv_obj_get_style_flex_flow_internal(cont, LV_PART_MAIN);
146 f->row = flow & LV_FLEX_COLUMN ? 0 : 1;
147 f->wrap = flow & LV_FLEX_WRAP ? 1 : 0;
148 f->rev = flow & LV_FLEX_REVERSE ? 1 : 0;
149 f->main_place = lv_obj_get_style_flex_main_place_internal(cont, LV_PART_MAIN);
150 f->cross_place = lv_obj_get_style_flex_cross_place_internal(cont, LV_PART_MAIN);

154static bool calc_min_size(lv_obj_t * cont, int32_t * req_size, bool width, void * user_data)
155{
156 LV_UNUSED(user_data);
157 LV_ASSERT(cont != NULL);
158 LV_ASSERT(req_size != NULL);
159
160 if(cont->spec_attr == NULL)
161 return false;
162
163 flex_t f;
164 get_flex_info(cont, &f);
165
166 if(f.row != width) {
167 return false;
168 }
169
170 /* Can't wrap if size is LV_SIZE_CONTENT */
171 f.wrap = false;
172
173 *req_size = 0;
174
175 int32_t item_gap =
176 f.row ? lv_obj_get_style_pad_column_internal(cont, LV_PART_MAIN) : lv_obj_get_style_pad_row_internal(cont,
177 LV_PART_MAIN);
178
179 int32_t cont_space_start =
180 (f.row ? lv_obj_get_style_space_left_internal(cont, LV_PART_MAIN) : lv_obj_get_style_space_top_internal(cont,
181 LV_PART_MAIN));
182 int32_t cont_space_end =
183 (f.row ? lv_obj_get_style_space_right_internal(cont, LV_PART_MAIN) : lv_obj_get_style_space_bottom_internal(cont,
184 LV_PART_MAIN));
185
186 int32_t track_first_item = f.rev ? cont->spec_attr->child_cnt - 1 : 0;
187 int32_t next_track_first_item;
188
189 while(track_first_item < (int32_t)cont->spec_attr->child_cnt && track_first_item >= 0) {
190 /*Search the first item of the next row*/
191 track_t t;
192 t.grow_dsc_calc = 0;
193 next_track_first_item = find_track_end(cont, &f, track_first_item, 0, item_gap, &t);
194
195 int32_t req_track_size = t.track_fix_main_size + t.track_grow_min_size;
196
197 track_first_item = next_track_first_item;
198
199 *req_size = LV_MAX(*req_size, req_track_size);
200 }
201
202 *req_size += (cont_space_start + cont_space_end);
203
204 // (*req_size)++;
205 return true;
206}

208static void flex_update(lv_obj_t * cont, void * user_data)
209{
210 LV_ASSERT(cont != NULL);
211 LV_LOG_INFO("update %p container", (void *)cont);
212 LV_UNUSED(user_data);
213
214 flex_t f;
215 get_flex_info(cont, &f);
216
217 bool rtl = lv_obj_get_style_base_dir_internal(cont, LV_PART_MAIN) == LV_BASE_DIR_RTL;
218 int32_t track_gap =
219 !f.row ? lv_obj_get_style_pad_column_internal(cont, LV_PART_MAIN) : lv_obj_get_style_pad_row_internal(cont,
220 LV_PART_MAIN);
221 int32_t item_gap =
222 f.row ? lv_obj_get_style_pad_column_internal(cont, LV_PART_MAIN) : lv_obj_get_style_pad_row_internal(cont,
223 LV_PART_MAIN);
224 int32_t max_main_size = (f.row ? lv_obj_get_content_width(cont) : lv_obj_get_content_height(cont));
225 int32_t abs_y = cont->coords.y1 + lv_obj_get_style_space_top_internal(cont, LV_PART_MAIN) - lv_obj_get_scroll_y(cont);
226 int32_t abs_x = cont->coords.x1 + lv_obj_get_style_space_left_internal(cont, LV_PART_MAIN) - lv_obj_get_scroll_x(cont);
227
228 lv_flex_align_t track_cross_place = f.track_place;
229 int32_t * cross_pos = (f.row ? &abs_y : &abs_x);
230
231 int32_t w_set = lv_obj_get_style_width_internal(cont, LV_PART_MAIN);
232 int32_t h_set = lv_obj_get_style_height_internal(cont, LV_PART_MAIN);
233
234 /*Content sized objects should squeeze the gap between the children, therefore any alignment will look like
235 * `START`*/
236 if((f.row && h_set == LV_SIZE_CONTENT && cont->h_layout == 0) ||
237 (!f.row && w_set == LV_SIZE_CONTENT && cont->w_layout == 0)) {
238 track_cross_place = LV_FLEX_ALIGN_START;
239 }
240
241 if(rtl && !f.row) {
242 if(track_cross_place == LV_FLEX_ALIGN_START)
243 track_cross_place = LV_FLEX_ALIGN_END;
244 else if(track_cross_place == LV_FLEX_ALIGN_END)
245 track_cross_place = LV_FLEX_ALIGN_START;
246 }
247
248 int32_t total_track_cross_size = 0;
249 int32_t gap = 0;
250 uint32_t track_cnt = 0;
251 int32_t track_first_item;
252 int32_t next_track_first_item;
253
254 if(track_cross_place != LV_FLEX_ALIGN_START) {
255 track_first_item = f.rev ? cont->spec_attr->child_cnt - 1 : 0;
256 track_t t;
257 while(track_first_item < (int32_t)cont->spec_attr->child_cnt && track_first_item >= 0) {
258 /*Search the first item of the next row*/
259 t.grow_dsc_calc = 0;
260 next_track_first_item = find_track_end(cont, &f, track_first_item, max_main_size, item_gap, &t);
261 total_track_cross_size += t.track_cross_size + track_gap;
262 track_cnt++;
263 track_first_item = next_track_first_item;
264 }
265
266 if(track_cnt)
267 total_track_cross_size -= track_gap; /*No gap after the last track*/
268
269 /*Place the tracks to get the start position*/
270 int32_t max_cross_size = (f.row ? lv_obj_get_content_height(cont) : lv_obj_get_content_width(cont));
271 place_content(track_cross_place, max_cross_size, total_track_cross_size, track_cnt, cross_pos, &gap);
272 }
273
274 track_first_item = f.rev ? cont->spec_attr->child_cnt - 1 : 0;
275
276 if(rtl && !f.row) {
277 *cross_pos += total_track_cross_size;
278 }
279
280 while(track_first_item < (int32_t)cont->spec_attr->child_cnt && track_first_item >= 0) {
281 track_t t;
282 t.grow_dsc_calc = 1;
283 /*Search the first item of the next row*/
284 next_track_first_item = find_track_end(cont, &f, track_first_item, max_main_size, item_gap, &t);
285
286 if(rtl && !f.row) {
287 *cross_pos -= t.track_cross_size;
288 }
289 children_repos(cont, &f, track_first_item, next_track_first_item, abs_x, abs_y, max_main_size, item_gap, &t);
290 track_first_item = next_track_first_item;
291 lv_free(t.grow_dsc);
292 t.grow_dsc = NULL;
293 if(rtl && !f.row) {
294 *cross_pos -= gap + track_gap;
295 }
296 else {
297 *cross_pos += t.track_cross_size + gap + track_gap;
298 }
299 }
300 LV_ASSERT_MEM_INTEGRITY();
301
302 if(w_set == LV_SIZE_CONTENT || h_set == LV_SIZE_CONTENT) {
303 lv_obj_refr_size(cont);
304 }
305
306 lv_obj_send_event(cont, LV_EVENT_LAYOUT_CHANGED, NULL);
307
308 LV_TRACE_LAYOUT("finished");
309}

314static int32_t find_track_end(lv_obj_t * cont, flex_t * f, int32_t item_start_id, int32_t max_main_size,
315 int32_t item_gap, track_t * t)
316{
317 LV_ASSERT(cont != NULL);
318 LV_ASSERT(f != NULL);
319 LV_ASSERT(t != NULL);
320 int32_t w_set = lv_obj_get_style_width_internal(cont, LV_PART_MAIN);
321 int32_t h_set = lv_obj_get_style_height_internal(cont, LV_PART_MAIN);
322
323 lv_obj_t * parent = lv_obj_get_parent(cont);
324 bool ignore_size_content = false;
325 if(parent != NULL) {
326 bool parent_is_flex = lv_obj_get_style_layout_internal(parent, LV_PART_MAIN) == LV_LAYOUT_FLEX;
327 uint8_t grow_value = lv_obj_get_style_flex_grow_internal(cont, LV_PART_MAIN);
328
329 /* If the obj is grown then the size in that direction is known and overrides LV_SIZE_CONTENT if it is set. In
330 * the next `if` statement we no longer need to prevent wrapping if the width/height (depending on flow) is
331 * `LV_SIZE_CONTENT`, since it is not used.
332 */
333 ignore_size_content = parent_is_flex && (grow_value > 0);
334 }
335
336 /*Can't wrap if the size is auto (i.e. the size depends on the children)*/
337 if(f->wrap && ((f->row && w_set == LV_SIZE_CONTENT) || (!f->row && h_set == LV_SIZE_CONTENT)) &&
338 !ignore_size_content) {
339 f->wrap = false;
340 }
341 int32_t (*get_main_size)(const lv_obj_t *) = (f->row ? lv_obj_get_width_with_margin : lv_obj_get_height_with_margin);
342 int32_t (*get_cross_size)(const lv_obj_t *) =
343 (!f->row ? lv_obj_get_width_with_margin : lv_obj_get_height_with_margin);
344
345 t->track_main_size = 0;
346 t->track_fix_main_size = 0;
347 t->track_grow_min_size = 0;
348 t->grow_item_cnt = 0;
349 t->track_cross_size = 0;
350 t->item_cnt = 0;
351 t->grow_dsc = NULL;
352
353 int32_t item_id = item_start_id;
354 lv_obj_t * item = lv_obj_get_child(cont, item_id);
355 bool first_item = true;
356 while(item) {
357 if(!(lv_obj_is_ignore_layout(item) || lv_obj_is_hidden(item) || lv_obj_is_floating(item))) {
358 if(!first_item && lv_obj_is_flex_in_new_track(item))
359 break;
360 uint8_t grow_value = lv_obj_get_style_flex_grow_internal(item, LV_PART_MAIN);
361 if(grow_value) {
362 int32_t min_size = f->row ? lv_obj_calc_dynamic_width(item, LV_STYLE_MIN_WIDTH)
363 : lv_obj_calc_dynamic_height(item, LV_STYLE_MIN_HEIGHT);
364 int32_t req_size = min_size;
365 if(!first_item) {
366 req_size += item_gap; /*No gap before the first item*/
367 }
368
369 /*Wrap if can't fit*/
370 if(f->wrap && t->track_fix_main_size + t->track_grow_min_size + req_size > max_main_size)
371 break;
372
373 t->track_grow_min_size += min_size;
374 if(!first_item) {
375 t->track_fix_main_size += item_gap; /*The gap is always taken from the space*/
376 }
377
378 t->grow_item_cnt++;
379
380 if(t->grow_dsc_calc) {
381 grow_dsc_t * new_dsc = lv_realloc(t->grow_dsc, sizeof(grow_dsc_t) * (t->grow_item_cnt));
382 LV_ASSERT_MALLOC(new_dsc);
383 if(new_dsc == NULL)
384 return item_id;
385
386 int32_t max_size = f->row ? lv_obj_calc_dynamic_width(item, LV_STYLE_MAX_WIDTH)
387 : lv_obj_calc_dynamic_height(item, LV_STYLE_MAX_HEIGHT);
388
389 new_dsc[t->grow_item_cnt - 1].item = item;
390 new_dsc[t->grow_item_cnt - 1].min_size = min_size;
391 new_dsc[t->grow_item_cnt - 1].max_size = max_size;
392 new_dsc[t->grow_item_cnt - 1].grow_value = grow_value;
393 new_dsc[t->grow_item_cnt - 1].clamped = 0;
394
395 t->grow_dsc = new_dsc;
396 }
397 }
398 else {
399 int32_t item_size = get_main_size(item);
400 int32_t req_size = item_size;
401 if(!first_item)
402 req_size += item_gap; /*No gap before the first item*/
403 if(f->wrap && t->track_fix_main_size + t->track_grow_min_size + req_size > max_main_size)
404 break;
405 t->track_fix_main_size += req_size;
406 }
407
408 first_item = false;
409 t->track_cross_size = LV_MAX(get_cross_size(item), t->track_cross_size);
410 t->item_cnt++;
411 }
412
413 item_id += f->rev ? -1 : +1;
414 if(item_id < 0)
415 break;
416 item = lv_obj_get_child(cont, item_id);
417 }
418
419 /*If there is at least one "grow item" the track takes the full space*/
420 t->track_main_size = t->grow_item_cnt ? max_main_size : t->track_fix_main_size;
421
422 /*Have at least one item in a row*/
423 if(item && item_id == item_start_id) {
424 item = cont->spec_attr->children[item_id];
425 get_next_item(cont, f->rev, &item_id);
426 if(item) {
427 t->track_cross_size = get_cross_size(item);
428 t->track_main_size = get_main_size(item);
429 t->item_cnt = 1;
430 }
431 }
432
433 return item_id;
434}

439static void children_repos(lv_obj_t * cont, flex_t * f, int32_t item_first_id, int32_t item_last_id, int32_t abs_x,
440 int32_t abs_y, int32_t max_main_size, int32_t item_gap, track_t * t)
441{
442 LV_ASSERT(cont != NULL);
443 LV_ASSERT(f != NULL);
444 LV_ASSERT(t != NULL);
445 void (*area_set_main_size)(lv_area_t *, int32_t) = (f->row ? lv_area_set_width : lv_area_set_height);
446 int32_t (*area_get_main_size)(const lv_area_t *) = (f->row ? lv_area_get_width : lv_area_get_height);
447 int32_t (*area_get_cross_size)(const lv_area_t *) = (!f->row ? lv_area_get_width : lv_area_get_height);
448
449 typedef int32_t (*margin_func_t)(const lv_obj_t *, lv_part_t);
450 margin_func_t get_margin_main_start = (f->row ? lv_obj_get_style_margin_left : lv_obj_get_style_margin_top);
451 margin_func_t get_margin_main_end = (f->row ? lv_obj_get_style_margin_right : lv_obj_get_style_margin_bottom);
452 margin_func_t get_margin_cross_start = (!f->row ? lv_obj_get_style_margin_left : lv_obj_get_style_margin_top);
453 margin_func_t get_margin_cross_end = (!f->row ? lv_obj_get_style_margin_right : lv_obj_get_style_margin_bottom);
454
455 /*Calculate the size of grow items first*/
456 uint32_t i;
457 bool grow_reiterate = true;
458 while(grow_reiterate && t->grow_item_cnt) {
459 grow_reiterate = false;
460 int32_t grow_value_sum = 0;
461 int32_t grow_max_size = t->track_main_size - t->track_fix_main_size;
462 for(i = 0; i < t->grow_item_cnt; i++) {
463 if(t->grow_dsc[i].clamped == 0) {
464 grow_value_sum += t->grow_dsc[i].grow_value;
465 }
466 else {
467 grow_max_size -= t->grow_dsc[i].final_size;
468 }
469 }
470
471 for(i = 0; i < t->grow_item_cnt; i++) {
472 if(t->grow_dsc[i].clamped == 0) {
473 LV_ASSERT(grow_value_sum != 0);
474 int32_t size = div_round_closest(grow_max_size * t->grow_dsc[i].grow_value, grow_value_sum);
475 int32_t size_clamp = LV_CLAMP(t->grow_dsc[i].min_size, size, t->grow_dsc[i].max_size);
476
477 if(size_clamp != size) {
478 t->grow_dsc[i].clamped = 1;
479 grow_reiterate = true;
480 }
481 t->grow_dsc[i].final_size = size_clamp;
482 grow_value_sum -= t->grow_dsc[i].grow_value;
483 grow_max_size -= t->grow_dsc[i].final_size;
484 }
485 }
486 }
487
488 bool rtl = lv_obj_get_style_base_dir_internal(cont, LV_PART_MAIN) == LV_BASE_DIR_RTL;
489
490 int32_t main_pos = 0;
491
492 int32_t place_gap = 0;
493 place_content(f->main_place, max_main_size, t->track_main_size, t->item_cnt, &main_pos, &place_gap);
494 if(f->row && rtl) main_pos = max_main_size - main_pos;
495
496 lv_obj_t * item = lv_obj_get_child(cont, item_first_id);
497 /*Reposition the children*/
498 while(item && item_first_id != item_last_id) {
499 if((lv_obj_is_ignore_layout(item) || lv_obj_is_hidden(item) || lv_obj_is_floating(item))) {
500 item = get_next_item(cont, f->rev, &item_first_id);
501 continue;
502 }
503
504 uint16_t item_w_layout = item->w_layout;
505 uint16_t item_h_layout = item->h_layout;
506
507 int32_t grow_size = lv_obj_get_style_flex_grow_internal(item, LV_PART_MAIN);
508 if(grow_size) {
509 int32_t s = 0;
510 for(i = 0; i < t->grow_item_cnt; i++) {
511 if(t->grow_dsc[i].item == item) {
512 s = t->grow_dsc[i].final_size;
513 break;
514 }
515 }
516
517 if(f->row) {
518 item->w_layout = 1;
519 item->h_layout = 0;
520 }
521 else {
522 item->h_layout = 1;
523 item->w_layout = 0;
524 }
525
526 if(s != area_get_main_size(&item->coords)) {
527 lv_obj_invalidate(item);
528
529 lv_area_t old_coords;
530 old_coords = item->coords;
531 area_set_main_size(&item->coords, s);
532 lv_obj_send_event(item, LV_EVENT_SIZE_CHANGED, &old_coords);
533 lv_obj_send_event(lv_obj_get_parent(item), LV_EVENT_CHILD_CHANGED, item);
534 lv_obj_invalidate(item);
535 }
536 }
537 else {
538 item->w_layout = 0;
539 item->h_layout = 0;
540 }
541
542 if(item->w_layout != item_w_layout || item->h_layout != item_h_layout) {
543 lv_obj_mark_layout_as_dirty(item);
544 }
545
546 int32_t cross_pos = 0;
547 switch(f->cross_place) {
548 case LV_FLEX_ALIGN_CENTER:
549 /*Round up the cross size to avoid rounding error when dividing by 2
550 *The issue comes up e,g, with column direction with center cross direction if an element's width changes*/
551 cross_pos = (((t->track_cross_size + 1) & (~1)) - area_get_cross_size(&item->coords)) / 2;
552 cross_pos += (get_margin_cross_start(item, LV_PART_MAIN) - get_margin_cross_end(item, LV_PART_MAIN)) / 2;
553 break;
554 case LV_FLEX_ALIGN_END:
555 cross_pos = t->track_cross_size - area_get_cross_size(&item->coords);
556 cross_pos -= get_margin_cross_end(item, LV_PART_MAIN);
557 break;
558 default:
559 cross_pos += get_margin_cross_start(item, LV_PART_MAIN);
560 break;
561 }
562
563 if(f->row && rtl)
564 main_pos -= area_get_main_size(&item->coords);
565
566 /*Handle percentage value of translate*/
567 int32_t tr_x = lv_obj_get_style_translate_x_internal(item, LV_PART_MAIN);
568 int32_t tr_y = lv_obj_get_style_translate_y_internal(item, LV_PART_MAIN);
569 int32_t w = lv_obj_get_width(item);
570 int32_t h = lv_obj_get_height(item);
571 if(LV_COORD_IS_PCT(tr_x))
572 tr_x = (w * LV_COORD_GET_PCT(tr_x)) / 100;
573 if(LV_COORD_IS_PCT(tr_y))
574 tr_y = (h * LV_COORD_GET_PCT(tr_y)) / 100;
575
576 int32_t diff_x = abs_x - item->coords.x1 + tr_x;
577 int32_t diff_y = abs_y - item->coords.y1 + tr_y;
578 diff_x += f->row ? main_pos + get_margin_main_start(item, LV_PART_MAIN) : cross_pos;
579 diff_y += f->row ? cross_pos : main_pos + get_margin_main_start(item, LV_PART_MAIN);
580
581 if(diff_x || diff_y) {
582 lv_obj_invalidate(item);
583 item->coords.x1 += diff_x;
584 item->coords.x2 += diff_x;
585 item->coords.y1 += diff_y;
586 item->coords.y2 += diff_y;
587 lv_obj_invalidate(item);
588 lv_obj_move_children_by(item, diff_x, diff_y, false);
589 }
590
591 if(!(f->row && rtl))
592 main_pos += area_get_main_size(&item->coords) + item_gap + place_gap +
593 get_margin_main_start(item, LV_PART_MAIN) + get_margin_main_end(item, LV_PART_MAIN);
594 else
595 main_pos -= item_gap + place_gap;
596
597 item = get_next_item(cont, f->rev, &item_first_id);
598 }
599}

604static void place_content(lv_flex_align_t place, int32_t max_size, int32_t content_size, int32_t item_cnt,
605 int32_t * start_pos, int32_t * gap)
606{
607 LV_ASSERT(start_pos != NULL);
608 LV_ASSERT(gap != NULL);
609 if(item_cnt <= 1) {
610 switch(place) {
611 case LV_FLEX_ALIGN_SPACE_AROUND:
612 case LV_FLEX_ALIGN_SPACE_EVENLY:
613 place = LV_FLEX_ALIGN_CENTER;
614 break;
615 default:
616 break;
617 }
618 }
619
620 switch(place) {
621 case LV_FLEX_ALIGN_CENTER:
622 *gap = 0;
623 *start_pos += (max_size - content_size) / 2;
624 break;
625 case LV_FLEX_ALIGN_END:
626 *gap = 0;
627 *start_pos += max_size - content_size;
628 break;
629 case LV_FLEX_ALIGN_SPACE_BETWEEN:
630 if(item_cnt > 1) *gap = (int32_t)(max_size - content_size) / (int32_t)(item_cnt - 1);
631 break;
632 case LV_FLEX_ALIGN_SPACE_AROUND:
633 *gap += (int32_t)(max_size - content_size) / (int32_t)(item_cnt);
634 *start_pos += *gap / 2;
635 break;
636 case LV_FLEX_ALIGN_SPACE_EVENLY:
637 *gap = (int32_t)(max_size - content_size) / (int32_t)(item_cnt + 1);
638 *start_pos += *gap;
639 break;
640 default:
641 *gap = 0;
642 }
643}
