60static lv_obj_t * win_create(lv_obj_t * parent)
61{
62 lv_obj_t * win = lv_obj_create(parent);
63 lv_obj_set_size(win, lv_obj_get_width(parent), lv_obj_get_height(parent));
64 lv_obj_set_flex_flow(win, LV_FLEX_FLOW_COLUMN);
65 lv_obj_add_style(win, &style_win, 0);
66
67 lv_obj_t * header = lv_obj_create(win);
68 lv_obj_set_size(header, lv_pct(100), lv_display_get_dpi(lv_obj_get_display(win)) / 2);
69 lv_obj_set_flex_flow(header, LV_FLEX_FLOW_ROW);
70 lv_obj_set_flex_align(header, LV_FLEX_ALIGN_START, LV_FLEX_ALIGN_CENTER, LV_FLEX_ALIGN_CENTER);
71 lv_obj_add_style(header, &style_header, 0);
72
73 lv_obj_t * content = lv_obj_create(win);
74 lv_obj_set_width(content, lv_pct(100));
75 lv_obj_set_flex_grow(content, 1);
76 lv_obj_add_style(content, &style_content, 0);
77
78 return win;
79}
