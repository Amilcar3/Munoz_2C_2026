110lv_result_t lv_obj_set_property(lv_obj_t * obj, const lv_property_t * value)
111{
112 LV_CHECK_ARG(obj != NULL, return LV_RESULT_INVALID);
113 LV_CHECK_ARG(value != NULL, return LV_RESULT_INVALID);
114
115 uint32_t index = LV_PROPERTY_ID_INDEX(value->id);
116 if(value->id == LV_PROPERTY_ID_INVALID || index > LV_PROPERTY_ID_ANY) {
117 LV_LOG_WARN("Invalid property id set to %p", obj);
118 return LV_RESULT_INVALID;
119 }
120
121 if(index < LV_PROPERTY_ID_START) {
122 lv_obj_set_local_style_prop(obj, index, value->style, value->selector);
123 return LV_RESULT_OK;
124 }
125
126 return obj_property(obj, value->id, (lv_property_t *)value, true);
127}

144lv_property_t lv_obj_get_property(lv_obj_t * obj, lv_prop_id_t id)
145{
146 LV_CHECK_ARG(obj != NULL, return (lv_property_t) {
147 .id = LV_PROPERTY_ID_INVALID
148 });
149
150 lv_result_t result;
151 lv_property_t value = { 0 };
152
153 uint32_t index = LV_PROPERTY_ID_INDEX(id);
154 if(id == LV_PROPERTY_ID_INVALID || index > LV_PROPERTY_ID_ANY) {
155 LV_LOG_WARN("Invalid property id to get from %p", obj);
156 value.id = LV_PROPERTY_ID_INVALID;
157 value.num = 0;
158 return value;
159 }
160
161 if(index < LV_PROPERTY_ID_START) {
162 value.style = lv_obj_get_style_prop(obj, LV_PART_MAIN, index);
163 value.id = id;
164 value.selector = LV_PART_MAIN | obj->state;
165 return value;
166 }
167
168 result = obj_property(obj, id, &value, false);
169 if(result != LV_RESULT_OK)
170 value.id = LV_PROPERTY_ID_INVALID;
171
172 return value;
173}

265static lv_result_t obj_property(lv_obj_t * obj, lv_prop_id_t id, lv_property_t * value, bool set)
266{
267 const lv_property_ops_t * properties;
268 const lv_property_ops_t * prop;
269
270 const lv_obj_class_t * clz;
271 uint32_t index = LV_PROPERTY_ID_INDEX(id);
272
273 for(clz = obj->class_p ; clz; clz = clz->base_class) {
274 properties = clz->properties;
275 if(properties == NULL) {
276 /* try base class*/
277 continue;
278 }
279
280 if(id != LV_PROPERTY_ID_ANY && (index < clz->prop_index_start || index > clz->prop_index_end)) {
281 /* try base class*/
282 continue;
283 }
284
285 /*Check if there's setter available for this class*/
286 for(uint32_t i = 0; i < clz->properties_count; i++) {
287 prop = &properties[i];
288
289 /*pass id and value directly to widget's property method*/
290 if(prop->id == LV_PROPERTY_ID_ANY) {
291 value->id = prop->id;
292 if(set) return ((lv_property_setter_t)prop->setter)(obj, id, value);
293 else return ((lv_property_getter_t)prop->getter)(obj, id, value);
294 }
295
296 /*Not this id, check next*/
297 if(prop->id != id)
298 continue;
299
300 /*id matched but we got null pointer to functions*/
301 if(set ? prop->setter == NULL : prop->getter == NULL) {
302 LV_LOG_WARN("NULL %s provided, id: 0x%" LV_PRIx32, set ? "setter" : "getter", id);
303 return LV_RESULT_INVALID;
304 }
305
306 /*Update value id if it's a read*/
307 if(!set) value->id = prop->id;
308
309 switch(LV_PROPERTY_ID_TYPE(prop->id)) {
310 case LV_PROPERTY_TYPE_INT:
311 HANDLE_PROPERTY_TYPE(integer, num);
312 break;
313 case LV_PROPERTY_TYPE_BOOL:
314 HANDLE_PROPERTY_TYPE(boolean, enable);
315 break;
316 case LV_PROPERTY_TYPE_PRECISE:
317 HANDLE_PROPERTY_TYPE(precise, precise);
318 break;
319 case LV_PROPERTY_TYPE_COLOR:
320 HANDLE_PROPERTY_TYPE(color, color);
321 break;
322 case LV_PROPERTY_TYPE_POINTER:
323 case LV_PROPERTY_TYPE_IMGSRC:
324 case LV_PROPERTY_TYPE_TEXT:
325 case LV_PROPERTY_TYPE_OBJ:
326 case LV_PROPERTY_TYPE_DISPLAY:
327 case LV_PROPERTY_TYPE_FONT:
328 HANDLE_PROPERTY_TYPE(pointer, ptr);
329 break;
330 case LV_PROPERTY_TYPE_POINT: {
331 lv_point_t * point = &value->point;
332 if(set)((lv_property_set_point_t)(prop->setter))(obj, point);
333 else *point = ((lv_property_get_point_t)(prop->getter))(obj);
334 break;
335 }
336 default: {
337 LV_LOG_WARN("Unknown property id: 0x%08" LV_PRIx32, prop->id);
338 return LV_RESULT_INVALID;
339 }
340 }
341
342 return LV_RESULT_OK;
343 }
344
345 /*If no setter found, try base class then*/
346 }
347
348 LV_LOG_WARN("Unknown property id: 0x%08" LV_PRIx32, id);
349 return LV_RESULT_INVALID;
350}
