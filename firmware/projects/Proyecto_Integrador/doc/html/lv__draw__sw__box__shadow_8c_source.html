55 /*Calculate the rectangle which is blurred to get the shadow in `shadow_area`*/
56 lv_area_t core_area;
57 core_area.x1 = coords->x1 + dsc->ofs_x - dsc->spread;
58 core_area.x2 = coords->x2 + dsc->ofs_x + dsc->spread;
59 core_area.y1 = coords->y1 + dsc->ofs_y - dsc->spread;
60 core_area.y2 = coords->y2 + dsc->ofs_y + dsc->spread;
61
62 /*Calculate the bounding box of the shadow*/
63 lv_area_t shadow_area;
64 shadow_area.x1 = core_area.x1 - dsc->width / 2 - 1;
65 shadow_area.x2 = core_area.x2 + dsc->width / 2 + 1;

67 shadow_area.y2 = core_area.y2 + dsc->width / 2 + 1;

69 lv_opa_t opa = dsc->opa;
70 if(opa > LV_OPA_MAX) opa = LV_OPA_COVER;

72 /*Get clipped draw area which is the real draw area.
73 *It is always the same or inside `shadow_area`*/
74 lv_area_t draw_area;
75 if(!lv_area_intersect(&draw_area, &shadow_area, &t->clip_area)) return;

77 /*Consider 1 px smaller bg to be sure the edge will be covered by the shadow*/
78 lv_area_t bg_area;
79 bg_area = *coords;
80 lv_area_increase(&bg_area, -1, -1);

82 /*Get the clamped radius*/
83 int32_t r_bg = dsc->radius;
84 int32_t short_side = LV_MIN(lv_area_get_width(&bg_area), lv_area_get_height(&bg_area));
85 if(r_bg > short_side >> 1) r_bg = short_side >> 1;
86
87 /*Get the clamped radius*/
88 int32_t r_sh = dsc->radius;
89 short_side = LV_MIN(lv_area_get_width(&core_area), lv_area_get_height(&core_area));
90 if(r_sh > short_side >> 1) r_sh = short_side >> 1;

92 /*Get how many pixels are affected by the blur on the corners*/

95 lv_opa_t * sh_buf;
96
97#if LV_DRAW_SW_SHADOW_CACHE_SIZE
98 lv_draw_sw_shadow_cache_t * cache = &shadow_cache;
99 if(cache->cache_size == corner_size && cache->cache_r == r_sh) {
100 /*Use the cache if available*/
101 sh_buf = lv_malloc(corner_size * corner_size);
102 LV_ASSERT_MALLOC(sh_buf);
103 if(!sh_buf) {
104 LV_LOG_WARN("Failed to allocate memory for shadow buffer");
105 return;
106 }
107 lv_memcpy(sh_buf, cache->cache, corner_size * corner_size);
108 }
109 else {
110 /*A larger buffer is required for calculation*/
111 sh_buf = lv_malloc(corner_size * corner_size * sizeof(uint16_t));
112 LV_ASSERT_MALLOC(sh_buf);
113 if(!sh_buf) {
114 LV_LOG_WARN("Failed to allocate memory for shadow buffer");
115 return;
116 }
117 shadow_draw_corner_buf(&core_area, (uint16_t *)sh_buf, dsc->width, r_sh);
118
119 /*Cache the corner if it fits into the cache size*/
120 if((uint32_t)corner_size * corner_size < sizeof(cache->cache)) {
121 lv_memcpy(cache->cache, sh_buf, corner_size * corner_size);
122 cache->cache_size = corner_size;
123 cache->cache_r = r_sh;
124 }
125 }
126#else
127 sh_buf = lv_malloc(corner_size * corner_size * sizeof(uint16_t));
128 LV_ASSERT_MALLOC(sh_buf);
129 if(!sh_buf) {
130 LV_LOG_WARN("Failed to allocate memory for shadow buffer");
131 return;
132 }
133 shadow_draw_corner_buf(&core_area, (uint16_t *)sh_buf, dsc->width, r_sh);
134#endif /*LV_DRAW_SW_SHADOW_CACHE_SIZE*/

511 !lv_area_is_in(&clip_area_sub, &bg_area, r_bg)) {
512 int32_t w = lv_area_get_width(&clip_area_sub);
513 sh_buf_tmp = sh_buf;
514 sh_buf_tmp += (blend_area.y2 - clip_area_sub.y2) * corner_size;
515 sh_buf_tmp += clip_area_sub.x1 - blend_area.x1;
516
517 /*Do not mask if out of the bg*/
518 if(simple && lv_area_is_out(&clip_area_sub, &bg_area, r_bg)) simple_sub = true;
519 else simple_sub = simple;
520 blend_dsc.mask_buf = mask_buf;
521 if(w > 0) {
522 blend_area.x1 = clip_area_sub.x1;
523 blend_area.x2 = clip_area_sub.x2;
524 blend_dsc.mask_res = LV_DRAW_SW_MASK_RES_CHANGED; /*In simple mode it won't be overwritten*/
525 for(y = clip_area_sub.y2; y >= clip_area_sub.y1; y--) {
526 blend_area.y1 = y;
527 blend_area.y2 = y;
528
529 if(!simple_sub) {
530 lv_memcpy(mask_buf, sh_buf_tmp, corner_size);
531 blend_dsc.mask_res = lv_draw_sw_mask_apply(masks, mask_buf, clip_area_sub.x1, y, w);
532 if(blend_dsc.mask_res == LV_DRAW_SW_MASK_RES_FULL_COVER) blend_dsc.mask_res = LV_DRAW_SW_MASK_RES_CHANGED;
533 }
534 else {
535 blend_dsc.mask_buf = sh_buf_tmp;
536 }
537 lv_draw_sw_blend(t, &blend_dsc);
538 sh_buf_tmp += corner_size;
539 }
540 }
541 }
542
543 /*Draw the center rectangle.*/
544 blend_area.x1 = shadow_area.x1 + corner_size ;
545 blend_area.x2 = shadow_area.x2 - corner_size;
546 blend_area.y1 = shadow_area.y1 + corner_size;
547 blend_area.y2 = shadow_area.y2 - corner_size;
548 blend_area.y1 = LV_MIN(blend_area.y1, h_half + 1);
549 blend_area.y2 = LV_MAX(blend_area.y2, h_half);
550 blend_dsc.mask_buf = mask_buf;
551
552 if(lv_area_intersect(&clip_area_sub, &blend_area, &t->clip_area) &&
553 !lv_area_is_in(&clip_area_sub, &bg_area, r_bg)) {
554 int32_t w = lv_area_get_width(&clip_area_sub);
555 if(w > 0) {
556 blend_area.x1 = clip_area_sub.x1;
557 blend_area.x2 = clip_area_sub.x2;
558 for(y = clip_area_sub.y1; y <= clip_area_sub.y2; y++) {
559 blend_area.y1 = y;
560 blend_area.y2 = y;
561
562 lv_memset(mask_buf, 0xff, w);
563 blend_dsc.mask_res = lv_draw_sw_mask_apply(masks, mask_buf, clip_area_sub.x1, y, w);
564 lv_draw_sw_blend(t, &blend_dsc);
565 }
566 }
567 }
568
569 if(!simple) {
570 lv_draw_sw_mask_free_param(&mask_rout_param);
571 }
572 lv_free(sh_buf);
573 lv_free(mask_buf);
574}

587static void LV_ATTRIBUTE_FAST_MEM shadow_draw_corner_buf(const lv_area_t * coords, uint16_t * sh_buf, int32_t sw,
588 int32_t r)
589{
590 int32_t sw_ori = sw;
591 int32_t size = sw_ori + r;
592
593 lv_area_t sh_area;
594 sh_area = *coords;
595 sh_area.x2 = sw / 2 + r - 1 - ((sw & 1) ? 0 : 1);
596 sh_area.y1 = sw / 2 + 1;
597
598 sh_area.x1 = sh_area.x2 - lv_area_get_width(coords);
599 sh_area.y2 = sh_area.y1 + lv_area_get_height(coords);
600
601 lv_draw_sw_mask_radius_param_t mask_param;
602 lv_draw_sw_mask_radius_init(&mask_param, &sh_area, r, false);
603
604#if SHADOW_ENHANCE
605 /*Set half shadow width because blur will be repeated*/
606 if(sw_ori == 1) sw = 1;
607 else sw = sw_ori >> 1;
608#endif /*SHADOW_ENHANCE*/
609
610 int32_t y;
611 lv_opa_t * mask_line = lv_malloc(size);
612 uint16_t * sh_ups_tmp_buf = (uint16_t *)sh_buf;
613 for(y = 0; y < size; y++) {
614 lv_memset(mask_line, 0xff, size);
615 lv_draw_sw_mask_res_t mask_res = mask_param.dsc.cb(mask_line, 0, y, size, &mask_param);
616 if(mask_res == LV_DRAW_SW_MASK_RES_TRANSP) {
617 lv_memzero(sh_ups_tmp_buf, size * sizeof(sh_ups_tmp_buf[0]));
618 }
619 else {
620 int32_t i;
621 sh_ups_tmp_buf[0] = (mask_line[0] << SHADOW_UPSCALE_SHIFT) / sw;
622 for(i = 1; i < size; i++) {
623 if(mask_line[i] == mask_line[i - 1]) sh_ups_tmp_buf[i] = sh_ups_tmp_buf[i - 1];
624 else sh_ups_tmp_buf[i] = (mask_line[i] << SHADOW_UPSCALE_SHIFT) / sw;
625 }
626 }
627
628 sh_ups_tmp_buf += size;
629 }
630 lv_free(mask_line);
631
632 lv_draw_sw_mask_free_param(&mask_param);
633
634 if(sw == 1) {
635 int32_t i;
636 lv_opa_t * res_buf = (lv_opa_t *)sh_buf;
637 for(i = 0; i < size * size; i++) {
638 res_buf[i] = (sh_buf[i] >> SHADOW_UPSCALE_SHIFT);
639 }
640 return;
641 }
642
643 shadow_blur_corner(size, sw, sh_buf);
644
645#if SHADOW_ENHANCE == 0
646 /*The result is required in lv_opa_t not uint16_t*/
647 uint32_t x;
648 lv_opa_t * res_buf = (lv_opa_t *)sh_buf;
649 for(x = 0; x < size * size; x++) {
650 res_buf[x] = sh_buf[x];
651 }
652#else
653 sw += sw_ori & 1;
654 if(sw > 1) {
655 uint32_t i;
656 uint32_t max_v_div = (LV_OPA_COVER << SHADOW_UPSCALE_SHIFT) / sw;
657 for(i = 0; i < (uint32_t)size * size; i++) {
658 if(sh_buf[i] == 0) continue;
659 else if(sh_buf[i] == LV_OPA_COVER) sh_buf[i] = max_v_div;
660 else sh_buf[i] = (sh_buf[i] << SHADOW_UPSCALE_SHIFT) / sw;
661 }
662
663 shadow_blur_corner(size, sw, sh_buf);
664 }
665 int32_t x;
666 lv_opa_t * res_buf = (lv_opa_t *)sh_buf;
667 for(x = 0; x < size * size; x++) {
668 res_buf[x] = (lv_opa_t) sh_buf[x];
669 }
670#endif
671
672}

674static void LV_ATTRIBUTE_FAST_MEM shadow_blur_corner(int32_t size, int32_t sw, uint16_t * sh_ups_buf)
675{
676 int32_t s_left = sw >> 1;
677 int32_t s_right = (sw >> 1);
678 if((sw & 1) == 0) s_left--;
679
680 /*Horizontal blur*/
681 uint16_t * sh_ups_blur_buf = lv_malloc(size * sizeof(uint16_t));
682
683 int32_t x;
684 int32_t y;
685
686 uint16_t * sh_ups_tmp_buf = sh_ups_buf;
687
688 for(y = 0; y < size; y++) {
689 int32_t v = sh_ups_tmp_buf[size - 1] * sw;
690 for(x = size - 1; x >= 0; x--) {
691 sh_ups_blur_buf[x] = v;
692
693 /*Forget the right pixel*/
694 uint32_t right_val = 0;
695 if(x + s_right < size) right_val = sh_ups_tmp_buf[x + s_right];
696 v -= right_val;
697
698 /*Add the left pixel*/
699 uint32_t left_val;
700 if(x - s_left - 1 < 0) left_val = sh_ups_tmp_buf[0];
701 else left_val = sh_ups_tmp_buf[x - s_left - 1];
702 v += left_val;
703 }
704 lv_memcpy(sh_ups_tmp_buf, sh_ups_blur_buf, size * sizeof(uint16_t));
705 sh_ups_tmp_buf += size;
706 }
707
708 /*Vertical blur*/
709 uint32_t i;
710 uint32_t max_v = LV_OPA_COVER << SHADOW_UPSCALE_SHIFT;
711 uint32_t max_v_div = max_v / sw;
712 for(i = 0; i < (uint32_t)size * size; i++) {
713 if(sh_ups_buf[i] == 0) continue;
714 else if(sh_ups_buf[i] == max_v) sh_ups_buf[i] = max_v_div;
715 else sh_ups_buf[i] = sh_ups_buf[i] / sw;
716 }
717
718 for(x = 0; x < size; x++) {
719 sh_ups_tmp_buf = &sh_ups_buf[x];
720 int32_t v = sh_ups_tmp_buf[0] * sw;
721 for(y = 0; y < size ; y++, sh_ups_tmp_buf += size) {
722 sh_ups_blur_buf[y] = v < 0 ? 0 : (v >> SHADOW_UPSCALE_SHIFT);
723
724 /*Forget the top pixel*/
725 uint32_t top_val;
726 if(y - s_right <= 0) top_val = sh_ups_tmp_buf[0];
727 else top_val = sh_ups_buf[(y - s_right) * size + x];
728 v -= top_val;
729
730 /*Add the bottom pixel*/
731 uint32_t bottom_val;
732 if(y + s_left + 1 < size) bottom_val = sh_ups_buf[(y + s_left + 1) * size + x];
733 else bottom_val = sh_ups_buf[(size - 1) * size + x];
734 v += bottom_val;
735 }
736
737 /*Write back the result into `sh_ups_buf`*/
738 sh_ups_tmp_buf = &sh_ups_buf[x];
739 for(y = 0; y < size; y++, sh_ups_tmp_buf += size) {
740 (*sh_ups_tmp_buf) = sh_ups_blur_buf[y];
741 }
742 }
743
744 lv_free(sh_ups_blur_buf);
745}
