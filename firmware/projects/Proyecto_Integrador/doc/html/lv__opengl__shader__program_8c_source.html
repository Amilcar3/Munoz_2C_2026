49lv_opengl_shader_program_t * lv_opengl_shader_program_create(unsigned int _program)
50{
51 lv_opengl_shader_program_t * program = lv_malloc(sizeof(*program));
52 LV_ASSERT_MALLOC(program);
53 if(!program) {
54 return NULL;
55 }
56 program->update_uniform_1i = &update_uniform_1i;
57 program->update_uniform_1f = &update_uniform_1f;
58 program->id = _program;
59
60 return program;
61}

63void lv_opengl_shader_program_destroy(lv_opengl_shader_program_t * program)
64{
65#ifndef __EMSCRIPTEN__
66 GLint shader_num = 0;
67 GL_CALL(glGetProgramiv(program->id, GL_ATTACHED_SHADERS, &shader_num));
68
69 if(shader_num > 0) {
70 GLuint * shader_names = lv_malloc(shader_num * sizeof(GLuint));
71
72 if(shader_names) {
73 GLsizei shader_count;
74 GL_CALL(glGetAttachedShaders(program->id, shader_num, &shader_count,
75 shader_names));
76
77 // Detach and delete each shader
78 for(GLsizei i = 0; i < shader_count && i < shader_num; ++i) {
79 if(shader_names[i] != 0)
80 GL_CALL(glDetachShader(program->id, shader_names[i]));
81 }
82 lv_free(shader_names);
83 }
84 }
85#endif

87 /* We should be able to call the function below without issue at this point
88 * but because of subtle issues regarding lazy updates of shader resources
89 * this induces significant pause on some platforms. Since the shaders
90 * have already been detached, we can safely skip this function and leave
91 * the empty programs in OpenGL's cache until the app shuts down, it's a
92 * very small amount of memory.
93 *
94 * To-do: Consider setting a flag at this point and if that flag is true
95 * when the app finally shuts down, then perform the glDeleteProgram calls
96 * if necessary. That is not really necessary, OpenGL will do that anyways
97 * when it shuts down. */
98
99 /* GL_CALL(glDeleteProgram(program->id)); */
100 lv_free(program);
101}

114static void update_uniform_1i(lv_opengl_shader_program_t * program, const char * prop,
115 int value)
116{
117 GLuint location = glGetUniformLocation(program->id, prop);
118 LV_ASSERT_FORMAT_MSG(location != INVALID_LOCATION,
119 "Uniform '%s' not found in program %d",
120 prop, program->id);
121 if(location == INVALID_LOCATION) {
122 LV_LOG_ERROR("Uniform '%s' not found in program %d", prop,
123 program->id);
124 return;
125 }
126 GL_CALL(glUniform1i(location, value));
127}

129static void update_uniform_1f(lv_opengl_shader_program_t * program, const char * prop,

138 LV_LOG_ERROR("Uniform '%s' not found in program %d", prop,
139 program->id);
140 return;
141 }
142 GL_CALL(glUniform1f(location, value));
143}
