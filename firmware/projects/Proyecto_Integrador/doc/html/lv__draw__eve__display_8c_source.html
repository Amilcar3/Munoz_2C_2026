51lv_display_t * lv_draw_eve_display_create(const lv_draw_eve_parameters_t * params, lv_draw_eve_operation_cb_t op_cb,
52 void * user_data)
53{
54 static uint32_t dummy_buf; /* It won't be used as it will send commands instead of draw pixels. */
55
56 LV_CHECK_ARG(params != NULL, return NULL);
57
58 lv_display_t * disp = lv_display_create(params->hor_res, params->ver_res);
59 lv_display_set_flush_cb(disp, flush_cb);
60 lv_display_set_buffers(disp, &dummy_buf, NULL,
61 params->hor_res * params->ver_res * LV_COLOR_FORMAT_GET_SIZE(LV_COLOR_FORMAT_DEFAULT),
62 LV_DISPLAY_RENDER_MODE_FULL); /* recreate the full display list each refresh */
63 lv_display_add_event_cb(disp, resolution_changed_cb, LV_EVENT_RESOLUTION_CHANGED, NULL);
64 lv_display_add_event_cb(disp, render_start_cb, LV_EVENT_RENDER_START, NULL);
65 lv_display_add_event_cb(disp, render_ready_cb, LV_EVENT_RENDER_READY, NULL);
66 lv_display_set_driver_data(disp, user_data);
67
68 lv_draw_eve_set_display_data(disp, params, op_cb);
69
70 EVE_init();
71 EVE_memWrite8(REG_PWM_DUTY, EVE_BACKLIGHT_PWM); /* 0 = off, 0x80 = max */
72
73 return disp;
74}

111void lv_draw_eve_pre_upload_font_range(lv_display_t * disp, const lv_font_t * font, uint32_t unicode_range_start,
112 uint32_t unicode_range_end)
113{
114 LV_CHECK_ARG(disp != NULL, return);
115 LV_CHECK_ARG(font != NULL, return);
116 LV_ASSERT_MSG(disp->flush_cb == flush_cb, "tried to do an LVGL EVE pre-upload without a draw_eve display");
117
118 if(!lv_draw_eve_label_font_check(font)) {
119 return;
120 }
121
122 for(uint32_t i = unicode_range_start; i <= unicode_range_end; i++) {
123 lv_font_glyph_dsc_t glyph_dsc;
124 bool found = lv_font_get_glyph_dsc_fmt_txt(font, &glyph_dsc, i, '\0');
125 if(!found) {
126 LV_LOG_INFO("Could not pre-upload glyph with unicode code point '0x%"LV_PRIX32"' "
127 "because it is not part of the font", i);
128 continue;
129 }
130 uint32_t ramg_addr = lv_draw_eve_label_upload_glyph(false, font->dsc, glyph_dsc.gid.index);
131 if(ramg_addr == LV_DRAW_EVE_RAMG_OUT_OF_RAMG) {
132 LV_LOG_WARN("Could not pre-upload glyph because space could not be allocated in RAM_G.");
133 /* don't return in case there are smaller glyphs that there is space for */
134 }
135 }

214static void flush_cb(lv_display_t * disp, const lv_area_t * area, uint8_t * px_map)
215{
216 if(lv_display_flush_is_last(disp)) {
217 EVE_cmd_dl_burst(DL_DISPLAY); /* instruct the co-processor to show the list */
218 EVE_cmd_dl_burst(CMD_SWAP); /* make this list active */
219 EVE_end_cmd_burst();
220
221 EVE_execute_cmd();
222
223 EVE_start_cmd_burst();
224 }
225
226 lv_display_flush_ready(disp);
227}

229static void resolution_changed_cb(lv_event_t * e)
230{
231 lv_display_t * disp = lv_event_get_target(e);
232
233 lv_display_rotation_t rotation = lv_display_get_rotation(disp);
234 uint32_t cmd_value;
235 switch(rotation) {
236 case LV_DISPLAY_ROTATION_0:
237 cmd_value = 0;
238 break;
239 case LV_DISPLAY_ROTATION_90:
240 cmd_value = 2;
241 break;
242 case LV_DISPLAY_ROTATION_180:
243 cmd_value = 1;
244 break;
245 case LV_DISPLAY_ROTATION_270:
246 cmd_value = 3;
247 break;
248 default:
249 return;
250 }
251
252 /* no need to rotate the touch coordinates with CMD_SETROTATE, as LVGL
253 * already rotates the input coordinates.
254 */
255 EVE_memWrite8(REG_ROTATE, cmd_value);
256}

258static void render_start_cb(lv_event_t * e)
259{
260 EVE_start_cmd_burst();

262 EVE_cmd_dl_burst(CMD_DLSTART); /* start the display list */
263 EVE_cmd_dl_burst(DL_CLEAR_COLOR_RGB | 0x000000);
264 EVE_cmd_dl_burst(DL_CLEAR | CLR_COL | CLR_STN | CLR_TAG);
265 EVE_cmd_dl_burst(VERTEX_FORMAT(0));
266}

273static void touch_read_cb(lv_indev_t * indev, lv_indev_data_t * data)
274{
275 lv_display_t * disp = lv_indev_get_display(indev);
276
277 if(disp == NULL || disp->flush_cb != flush_cb) return;
278
279 uint32_t xy = EVE_memRead32(REG_TOUCH_SCREEN_XY);
280 uint16_t x = xy >> 16;
281 uint16_t y = xy & 0xffff;
282
283 int32_t disp_w = lv_display_get_original_horizontal_resolution(disp);
284 int32_t disp_h = lv_display_get_original_vertical_resolution(disp);
285
286 if(x < disp_w && y < disp_h) {
287 data->state = LV_INDEV_STATE_PRESSED;
288 data->point.x = x;
289 data->point.y = y;
290 }
291 else {
292 data->state = LV_INDEV_STATE_RELEASED;
293 }
294}
