50static int32_t chord_length(int16_t radius, int16_t angle_degrees)
51{
52 angle_degrees %= 360;
53 if(angle_degrees < 0) angle_degrees += 360;
54 int32_t sin_value = lv_trigo_sin(angle_degrees / 2);
55 int64_t chord_length = 2 * radius * sin_value / 32768.0;
56 return (int32_t)chord_length ;
57}

60static lv_eve_primitive_t get_mask_direction(int16_t angle)
61{
62 if(angle >= 315 || angle < 45) {
63 return LV_EVE_PRIMITIVE_EDGE_STRIP_R;
64 }
65 if(angle >= 45 && angle < 135) {
66 return LV_EVE_PRIMITIVE_EDGE_STRIP_B;
67 }
68 if(angle >= 135 && angle < 225) {
69 return LV_EVE_PRIMITIVE_EDGE_STRIP_L;
70 }
71 if(angle >= 225 && angle < 315) {
72 return LV_EVE_PRIMITIVE_EDGE_STRIP_A;
73 }
74 return 0;
75}

87static bool is_same_quadrant(int16_t start_angle, int16_t end_angle)
88{
89 if(start_angle > end_angle) {
90 if((start_angle >= 0 && start_angle < 90) && (end_angle >= 0 && end_angle < 90)) {
91 return true;
92 }
93 else if((start_angle >= 90 && start_angle < 180) && (end_angle >= 90 && end_angle < 180)) {
94 return true;
95 }
96 else if((start_angle >= 180 && start_angle < 270) && (end_angle >= 180 && end_angle < 270)) {
97 return true;
98 }
99 else if((start_angle >= 270 && start_angle < 360) && (end_angle >= 270 && end_angle < 360)) {
100 return true;
101 }
102 else {
103 return false;
104 }
105 }
106 else {
107 return false;
108 }
109}

113static void lv_draw_eve_mask_angle(const lv_draw_arc_dsc_t * dsc, int32_t vertex_x, int32_t vertex_y,
114 int32_t start_angle, int32_t end_angle)
115{
116
117 /*Constrain the input angles*/
118
119
120 if(start_angle < 0)
121 start_angle = 0;
122 else if(start_angle > 359)
123 start_angle = 359;
124
125 if(end_angle < 0)
126 end_angle = 0;
127 else if(end_angle > 359)
128 end_angle = 359;
129
130 LV_ASSERT_MSG(start_angle >= 0 && start_angle <= 360, "Unexpected start angle");
131
132 int32_t mid_angle_op;
133 int32_t angle_range;
134 int32_t mask_dir_start;
135 int32_t mask_dir_end;
136 lv_point_t start;
137 lv_point_t end;
138 lv_point_t angle_range_op;
139
140 if(end_angle > start_angle) {
141 angle_range = LV_ABS(end_angle - start_angle);
142 }
143 else {
144 angle_range = 360 - start_angle + end_angle;
145 }
146
147 mid_angle_op = (angle_range / 2) + start_angle + 180;
148 mid_angle_op = mid_angle_op % 360;
149
150 mask_dir_end = LV_ABS(((360 - angle_range) / 4) + end_angle);
151 mask_dir_start = LV_ABS(((360 - angle_range) / 4) + mid_angle_op);
152
153 mask_dir_start = mask_dir_start % 360;
154 mask_dir_end = mask_dir_end % 360;
155
156 start.y = (lv_trigo_sin(start_angle) >> 5) + vertex_y;
157 start.x = (lv_trigo_cos(start_angle) >> 5) + vertex_x;
158
159 end.y = (lv_trigo_sin(end_angle) >> 5) + vertex_y;
160 end.x = (lv_trigo_cos(end_angle) >> 5) + vertex_x;
161
162 angle_range_op.y = (lv_trigo_sin(mid_angle_op) >> 5) + vertex_y;
163 angle_range_op.x = (lv_trigo_cos(mid_angle_op) >> 5) + vertex_x;
164
165 if(angle_range <= 180) {
166 /* Two sides mask and 6 vertex points */
167
168 /* Masking end angle */
169 lv_eve_primitive_t edge = get_mask_direction(mask_dir_end);
170 lv_eve_primitive(edge); /* Side one */
171 lv_eve_vertex_2f(angle_range_op.x, angle_range_op.y);
172 lv_eve_vertex_2f(vertex_x, vertex_y);
173 lv_eve_vertex_2f(end.x, end.y);
174
175 /* Masking start angle */
176 edge = get_mask_direction(mask_dir_start);
177 lv_eve_primitive(edge); /* Side two */
178 lv_eve_vertex_2f(angle_range_op.x, angle_range_op.y);
179 lv_eve_vertex_2f(vertex_x, vertex_y);
180 lv_eve_vertex_2f(start.x, start.y);
181
182 }
183
184 else {
185
186 if(is_same_quadrant(start_angle,
187 end_angle)) { /* "It is not an optimal implementation for the case where both angles (start and end) are in the same quadrant */
188 /* todo */
189 lv_point_t end_line_cntr;
190 lv_point_t start_line_cntr;
191
192 lv_point_t end_line_brd;
193 lv_point_t start_line_brd;
194
195 int16_t chord = chord_length(dsc->radius, 360 - angle_range);
196 int16_t w = ((chord / 4) < 1) ? 1 : chord / 4;
197 int16_t r_width = w;
198
199 end_line_brd.y = vertex_y + ((lv_trigo_sin(end_angle) * dsc->radius) >> LV_TRIGO_SHIFT);
200 end_line_brd.x = vertex_x + ((lv_trigo_cos(end_angle) * dsc->radius) >> LV_TRIGO_SHIFT);
201
202 start_line_brd.y = vertex_y + ((lv_trigo_sin(start_angle) * dsc->radius) >> LV_TRIGO_SHIFT);
203 start_line_brd.x = vertex_x + ((lv_trigo_cos(start_angle) * dsc->radius) >> LV_TRIGO_SHIFT);
204
205 lv_eve_draw_rect_simple(start_line_brd.x, start_line_brd.y, end_line_brd.x, end_line_brd.y, 0);
206
207 start_line_brd.y = start_line_brd.y + ((lv_trigo_sin(start_angle - 90) * r_width) >> LV_TRIGO_SHIFT);
208 start_line_brd.x = start_line_brd.x + ((lv_trigo_cos(start_angle - 90) * r_width) >> LV_TRIGO_SHIFT);
209
210 end_line_brd.y = end_line_brd.y + ((lv_trigo_sin(end_angle + 90) * r_width) >> LV_TRIGO_SHIFT);
211 end_line_brd.x = end_line_brd.x + ((lv_trigo_cos(end_angle + 90) * r_width) >> LV_TRIGO_SHIFT);
212
213 end_line_cntr.y = vertex_y + ((lv_trigo_sin(end_angle + 90) * r_width) >> LV_TRIGO_SHIFT);
214 end_line_cntr.x = vertex_x + ((lv_trigo_cos(end_angle + 90) * r_width) >> LV_TRIGO_SHIFT);
215
216 start_line_cntr.y = vertex_y + ((lv_trigo_sin(start_angle + 270) * r_width) >> LV_TRIGO_SHIFT);
217 start_line_cntr.x = vertex_x + ((lv_trigo_cos(start_angle + 270) * r_width) >> LV_TRIGO_SHIFT);
218
219 lv_eve_primitive(LV_EVE_PRIMITIVE_LINE_STRIP);
220 lv_eve_line_width(r_width * 16);
221 lv_eve_vertex_2f(start_line_cntr.x, start_line_cntr.y);
222 lv_eve_vertex_2f(start_line_brd.x, start_line_brd.y);
223 lv_eve_vertex_2f(end_line_brd.x, end_line_brd.y);
224 lv_eve_vertex_2f(end_line_cntr.x, end_line_cntr.y);
225
226 }
227 else { /* One side mask and 3 vertex points */
228 /* Masking end and start angles */
229 lv_eve_primitive_t edge = get_mask_direction(mid_angle_op);
230 lv_eve_primitive(edge);
231 lv_eve_vertex_2f(end.x, end.y);
232 lv_eve_vertex_2f(vertex_x, vertex_y);
233 lv_eve_vertex_2f(start.x, start.y);
234 }
235 }
236
237}

241static void draw_eve_arc(lv_draw_task_t * t, const lv_draw_arc_dsc_t * dsc, const lv_area_t * coords)
242{
243
244 if(dsc->opa <= LV_OPA_MIN)
245 return;
246 if(dsc->width == 0)
247 return;
248 if(dsc->start_angle == dsc->end_angle)
249 return;
250
251 lv_color_t color = dsc->color;
252 lv_opa_t opa = dsc->opa;
253 lv_point_t center = dsc->center;
254 int32_t width = dsc->width;
255 uint16_t radius_out = dsc->radius;
256 uint16_t radius_in = dsc->radius - dsc->width;
257 int32_t start_angle = (int32_t) dsc->start_angle;
258 int32_t end_angle = (int32_t) dsc->end_angle;
259
260 if(width > radius_out)
261 width = radius_out;
262
263 while(start_angle >= 360)
264 start_angle -= 360;
265 while(end_angle >= 360)
266 end_angle -= 360;
267
268 lv_eve_scissor(t->clip_area.x1, t->clip_area.y1, t->clip_area.x2, t->clip_area.y2);
269
270 lv_eve_save_context();
271
272 lv_eve_color(color);
273 lv_eve_color_opa(opa);
274
275 lv_eve_color_mask(0, 0, 0, 1);
276 lv_eve_stencil_func(EVE_ALWAYS, 0, 1);
277 lv_eve_stencil_op(EVE_REPLACE, EVE_REPLACE);
278 lv_eve_draw_circle_simple(center.x, center.y, radius_out); /* radius_out */
279
280 lv_eve_blend_func(EVE_ONE, EVE_ZERO);
281 lv_eve_draw_circle_simple(center.x, center.y, radius_in + 2); /* radius_in */
282
283 lv_eve_stencil_func(EVE_ALWAYS, 1, 1);
284 lv_eve_stencil_op(EVE_REPLACE, EVE_REPLACE);
285 lv_eve_blend_func(EVE_ZERO, EVE_ONE_MINUS_SRC_ALPHA);
286 lv_eve_color_opa(0XFF);
287
288 /* Start masking arc */
289
290 lv_draw_eve_mask_angle(dsc, center.x, center.y, start_angle, end_angle);
291
292 /* End masking arc */
293
294 lv_eve_draw_circle_simple(center.x, center.y, radius_in); /* radius_in */
295
296 lv_eve_color_mask(1, 1, 1, 1);
297 lv_eve_blend_func(EVE_DST_ALPHA, EVE_ONE_MINUS_DST_ALPHA);
298 lv_eve_draw_circle_simple(center.x, center.y, radius_in); /* radius_in */
299
300 lv_eve_stencil_func(EVE_NOTEQUAL, 1, 0XFF);
301 lv_eve_stencil_op(EVE_KEEP, EVE_KEEP);
302 lv_eve_blend_func(EVE_SRC_ALPHA, EVE_ONE_MINUS_SRC_ALPHA);
303
304 lv_eve_color_opa(opa);
305 lv_eve_draw_circle_simple(center.x, center.y, radius_out); /* radius_out */
306
307 if(dsc->rounded) {
308 lv_eve_stencil_func(EVE_EQUAL, 1, 0XFF);
309 if(opa < 255) {
310 lv_eve_stencil_op(EVE_ZERO, EVE_ZERO);
311 }
312
313 int32_t half_width = width / 2;
314 int32_t adjusted_radius = radius_out - half_width;
315 draw_rounded_end(center, adjusted_radius, end_angle, half_width);
316 draw_rounded_end(center, adjusted_radius, start_angle, half_width);
317 }
318
319 lv_eve_restore_context();
320}
