72

85
86 refr_ctx.disp->prev_scr = NULL;
87 refr_ctx.disp->draw_prev_over_act = 0;
88 lv_obj_delete(prev);

182void test_refr_simple_layer_is_split_into_chunks(void)
183{
184 /*A simple layer is rendered in chunks that fit into LV_DRAW_LAYER_SIMPLE_BUF_SIZE*/
185 const int32_t w = 200;
186 const int32_t h = 100;
187 refr_disp_create(w + 20, h + 20, LV_COLOR_FORMAT_RGB565, LV_DISPLAY_RENDER_MODE_DIRECT, 1, h + 20);
188 refr_screen_set_color(REFR_COLOR_BLACK);
189
190 lv_obj_t * obj = refr_rect_create(refr_screen(), 0, 0, w, h, REFR_COLOR_WHITE);
191 lv_obj_set_style_opa_layered(obj, 128, 0);
192 uint32_t draw_cnt = 0;
193 lv_obj_add_event_cb(obj, refr_draw_main_counter_cb, LV_EVENT_DRAW_MAIN, &draw_cnt);
194
195 lv_obj_invalidate(refr_screen());
196 refr_frame();
197
198 /*The object is opaque, so the chunks are rendered without an alpha channel*/
199 uint32_t px_size = lv_color_format_get_size(LV_COLOR_FORMAT_RGB565);
200 uint32_t max_rgb_rows = LV_DRAW_LAYER_SIMPLE_BUF_SIZE / (uint32_t)w / px_size;
201 uint32_t expected_rgb = ((uint32_t)h + max_rgb_rows - 1) / max_rgb_rows;
202 TEST_ASSERT_EQUAL_UINT32(expected_rgb, draw_cnt);
203
204 /*Every chunk is blended the same way*/
205 uint32_t reference = refr_screen_px(w / 2, 0);
206 ASSERT_PX_WITHIN(12, 128, (reference >> 16) & 0xFF);
207 int32_t y;
208 for(y = 0; y < h; y += 7) {
209 ASSERT_PX_EQ(reference, refr_screen_px(w / 2, y));
210 }
211
212 /*With a rounded corner the layer needs an alpha channel, so the chunks are smaller*/
213 draw_cnt = 0;
214 lv_obj_set_style_radius(obj, 20, 0);
215 lv_obj_invalidate(refr_screen());
216 refr_frame();
217
218 uint32_t max_argb_rows = LV_DRAW_LAYER_SIMPLE_BUF_SIZE / (uint32_t)w / sizeof(lv_color32_t);
219 uint32_t expected_argb = ((uint32_t)h + max_argb_rows - 1) / max_argb_rows;
220 TEST_ASSERT_EQUAL_UINT32(expected_argb, draw_cnt);
221 TEST_ASSERT_TRUE(expected_argb >= expected_rgb);
222}
