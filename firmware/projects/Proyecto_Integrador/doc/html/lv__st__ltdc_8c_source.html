108static lv_display_t * create(void * buf1, void * buf2, uint32_t buf_size, uint32_t layer_idx,
109 lv_display_render_mode_t mode)
110{
111 LTDC_LayerCfgTypeDef * layer_cfg = &hltdc.LayerCfg[layer_idx];
112 uint32_t layer_width = layer_cfg->ImageWidth;
113 uint32_t layer_height = layer_cfg->ImageHeight;
114 uint32_t layer_cf = layer_cfg->PixelFormat;
115 lv_color_format_t cf = get_lv_cf_from_layer_cf(layer_cf);
116
117 lv_display_t * disp = lv_display_create(layer_width, layer_height);
118 lv_display_set_color_format(disp, cf);
119 lv_display_set_flush_cb(disp, flush_cb);
120 lv_display_set_flush_wait_cb(disp, flush_wait_cb);
121 lv_display_set_driver_data(disp, (void *)(uintptr_t)layer_idx);
122
123 if(mode == LV_DISPLAY_RENDER_MODE_DIRECT) {
124 uint32_t cf_size = lv_color_format_get_size(cf);
125 lv_display_set_buffers(disp, buf1, buf2, layer_width * layer_height * cf_size, LV_DISPLAY_RENDER_MODE_DIRECT);
126
127 if(buf1 != NULL && buf2 != NULL) {
128 HAL_LTDC_RegisterCallback(&hltdc, HAL_LTDC_RELOAD_EVENT_CB_ID, reload_event_callback);
129 SYNC_INIT(layer_idx);
130 }
131 }
132 else {
133 lv_display_set_buffers(disp, buf1, buf2, buf_size, LV_DISPLAY_RENDER_MODE_PARTIAL);
134
135#if LV_ST_LTDC_USE_DMA2D_FLUSH
136 hdma2d.XferCpltCallback = transfer_complete_callback;
137 SYNC_INIT(layer_idx);
138#endif
139 }
140
141 return disp;
142}

144static void flush_cb(lv_display_t * disp, const lv_area_t * area, uint8_t * px_map)
145{
146 uint32_t layer_idx = (uint32_t)(uintptr_t)lv_display_get_driver_data(disp);
147 g_data.disp_flushed_in_flush_cb[layer_idx] = false;
148
149 if(disp->render_mode == LV_DISPLAY_RENDER_MODE_DIRECT) {
150 bool flush_is_last = lv_display_flush_is_last(disp);
151 if(flush_is_last) {
152 /* there is no ideal time to clean the cache (if present)
153 for **single-buffered** direct mode because the active buffer is drawn to
154 while LTDC is scanning it. Clean it in the last flush, at least,
155 but not every flush because it's expensive for not much visual improvement. */
156 clean_dcache();
157 }
158 if(flush_is_last && lv_display_is_double_buffered(disp)) {
159 HAL_LTDC_SetAddress_NoReload(&hltdc, (uint32_t)px_map, layer_idx);
160 g_data.layer_interrupt_is_owned[layer_idx] = true;
161 HAL_LTDC_Reload(&hltdc, LTDC_RELOAD_VERTICAL_BLANKING);
162 }
163 else {
164 g_data.disp_flushed_in_flush_cb[layer_idx] = true;
165 }
166 }
167 else {
168 LTDC_LayerCfgTypeDef * layer_cfg = &hltdc.LayerCfg[layer_idx];
169
170 lv_color_format_t cf = lv_display_get_color_format(disp);
171 int32_t disp_width = disp->hor_res;
172
173 uint8_t * fb = (uint8_t *) layer_cfg->FBStartAdress;
174 uint32_t px_size = lv_color_format_get_size(cf);
175 uint32_t fb_stride = px_size * disp_width;
176 lv_area_t rotated_area = *area;
177 lv_display_rotate_area(disp, &rotated_area);
178 uint8_t * first_pixel = fb + fb_stride * rotated_area.y1 + px_size * rotated_area.x1;
179
180 int32_t area_width = lv_area_get_width(area);
181 int32_t area_height = lv_area_get_height(area);
182
183 lv_display_rotation_t rotation = lv_display_get_rotation(disp);
184 if(rotation == LV_DISPLAY_ROTATION_0) {
185#if LV_ST_LTDC_USE_DMA2D_FLUSH
186 clean_dcache();
187
188 uint32_t dma2d_input_cf = get_dma2d_input_cf_from_lv_cf(cf);
189 uint32_t dma2d_output_cf = get_dma2d_output_cf_from_layer_cf(layer_cfg->PixelFormat);
190
191 while(DMA2D->CR & DMA2D_CR_START);
192 DMA2D->FGPFCCR = dma2d_input_cf;
193 DMA2D->FGMAR = (uint32_t)px_map;
194 DMA2D->FGOR = 0;
195 DMA2D->OPFCCR = dma2d_output_cf;
196 DMA2D->OMAR = (uint32_t)first_pixel;
197 DMA2D->OOR = disp_width - area_width;
198 DMA2D->NLR = (area_width << DMA2D_NLR_PL_Pos) | (area_height << DMA2D_NLR_NL_Pos);
199 g_data.dma2d_interrupt_owner = layer_idx + 1;
200 DMA2D->CR = DMA2D_CR_START | DMA2D_CR_TCIE | (0x1U << DMA2D_CR_MODE_Pos); /* memory-to-memory with PFC */
201#else
202 uint32_t area_stride = px_size * area_width;
203 uint8_t * fb_p = first_pixel;
204 uint8_t * px_map_p = px_map;
205 for(int i = 0; i < area_height; i++) {
206 lv_memcpy(fb_p, px_map_p, area_stride);
207 fb_p += fb_stride;
208 px_map_p += area_stride;
209 }
210 clean_dcache();
211 g_data.disp_flushed_in_flush_cb[layer_idx] = true;
212#endif
213 }
214 else {
215#if LV_USE_DRAW_SW
216 uint32_t area_stride = px_size * area_width;
217 lv_draw_rotate(px_map, first_pixel, area_width, area_height, area_stride, fb_stride, rotation, cf);
218 clean_dcache();
219 g_data.disp_flushed_in_flush_cb[layer_idx] = true;
220#else
221 LV_LOG_WARN("LV_USE_DRAW_SW needs to be enabled to rotate the display's content");
222 g_data.disp_flushed_in_flush_cb[layer_idx] = true;
223#endif
224 }
225 }

236static lv_color_format_t get_lv_cf_from_layer_cf(uint32_t cf)

238 switch(cf) {

242 return LV_COLOR_FORMAT_RGB888;

244 return LV_COLOR_FORMAT_RGB565;

246 return LV_COLOR_FORMAT_L8;

250 LV_ASSERT_MSG(0, "the LTDC color format is not supported");
251 }

254static void reload_event_callback(LTDC_HandleTypeDef * hltdc)
255{
256 uint32_t i;
257 for(i = 0; i < MAX_LAYER; i++) {
258 if(g_data.layer_interrupt_is_owned[i]) {

260 SYNC_SIGNAL_ISR(i);
261 }
