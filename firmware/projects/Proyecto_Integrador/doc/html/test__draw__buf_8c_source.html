121/* Verify that the draw buffer allocation size reported in `data_size`
122 * is rounded up to a multiple of LV_DRAW_BUF_ALIGN, as required by back-ends
123 * (e.g. the ESP32 PPA) that need the allocation length to match the cache-line
124 * boundary rather than the raw image content size. */
125void test_draw_buf_size_alignment(void)
126{
127 /* A set of (w, h, cf, stride) cases chosen so that the raw image size
128 * (stride*h [+ alpha/palette]) is deliberately not a multiple of
129 * LV_DRAW_BUF_ALIGN, so the round-up behavior is observable. */
130 struct {
131 uint32_t w;
132 uint32_t h;
133 lv_color_format_t cf;
134 uint32_t stride; /* 0 => let LVGL choose via lv_draw_buf_width_to_stride */
135 } cases[] = {
136 /* Small indexed: stride=1, h=1, palette=8 => raw=9 (not a multiple of ALIGN) */
137 { 5, 1, LV_COLOR_FORMAT_I1, 1 },
138 /* A8 with odd raw size: 1*3 = 3 */
139 { 1, 3, LV_COLOR_FORMAT_A8, 1 },
140 /* RGB565 with an intentionally non-aligned raw size: 6*1 = 6 */
141 { 3, 1, LV_COLOR_FORMAT_RGB565, 6 },
142 /* RGB888 with raw = 9*1 = 9 */
143 { 3, 1, LV_COLOR_FORMAT_RGB888, 9 },
144 /* RGB565A8: rgb=4*1=4, a8=(4/2)*1=2 => raw=6 */
145 { 2, 1, LV_COLOR_FORMAT_RGB565A8, 4 },
146 /* Larger ARGB8888 with raw already aligned: 10*10*4 = 400 (rounding should be a no-op) */
147 { 10, 10, LV_COLOR_FORMAT_ARGB8888, 40 },
148 /* Stride auto-computed (0) path */
149 { 7, 5, LV_COLOR_FORMAT_RGB565, 0 },
150 };
151
152 for(unsigned i = 0; i < sizeof(cases) / sizeof(cases[0]); i++) {
153 uint32_t w = cases[i].w;
154 uint32_t h = cases[i].h;
155 lv_color_format_t cf = cases[i].cf;
156 uint32_t stride = cases[i].stride;
157
158 lv_draw_buf_t * buf = lv_draw_buf_create(w, h, cf, stride);
159 TEST_ASSERT_NOT_NULL(buf);
160
161 /* Recompute the stride the same way `_calculate_draw_buf_size` would
162 * if the caller passed 0, so we can derive the expected raw size. */
163 uint32_t effective_stride = stride ? stride : lv_draw_buf_width_to_stride(w, cf);
164 uint32_t raw_size = effective_stride * h;
165 if(cf == LV_COLOR_FORMAT_RGB565A8) {
166 raw_size += (effective_stride / 2) * h;
167 }
168 else if(LV_COLOR_FORMAT_IS_INDEXED(cf)) {
169 raw_size += LV_COLOR_INDEXED_PALETTE_SIZE(cf) * 4;
170 }
171
172 uint32_t expected = LV_ROUND_UP(raw_size, LV_DRAW_BUF_ALIGN);
173
174 /* data_size must be rounded up to LV_DRAW_BUF_ALIGN ... */
175 TEST_ASSERT_EQUAL_UINT32(0, buf->data_size % LV_DRAW_BUF_ALIGN);
176 /* ... must cover at least the raw image content size ... */
177 TEST_ASSERT_TRUE(buf->data_size >= raw_size);
178 /* ... and must match the explicit LV_ROUND_UP value. */
179 TEST_ASSERT_EQUAL_UINT32(expected, buf->data_size);
180
181 lv_draw_buf_destroy(buf);
182 }
183}

196{

200 /* Test ARGB8888 format */
201 {
202 draw_buf = lv_draw_buf_create(2, 2, LV_COLOR_FORMAT_ARGB8888, 2 * 4);
203
204 /* Fill with test data: white with 50% alpha */
205 lv_color32_t * pixel = (lv_color32_t *)draw_buf->data;
206 for(i = 0; i < 4; i++) {
207 pixel[i] = (lv_color32_t) {
208 .alpha = 128, .red = 255, .green = 255, .blue = 255

210 }
211

214 TEST_ASSERT(draw_buf->header.flags & LV_IMAGE_FLAGS_PREMULTIPLIED);
215
216 /* Verify premultiplied result: color should be 127 = (255 * 128) >> 8, alpha should be 128 */
217 for(i = 0; i < 4; i++) {
218 TEST_ASSERT_EQUAL_UINT8(127, pixel[i].red);
219 TEST_ASSERT_EQUAL_UINT8(127, pixel[i].green);
220 TEST_ASSERT_EQUAL_UINT8(127, pixel[i].blue);
221 TEST_ASSERT_EQUAL_UINT8(128, pixel[i].alpha); /* Alpha should remain unchanged */
222 }

232 lv_color32_t * pixel = (lv_color32_t *)draw_buf->data;
233 for(i = 0; i < 4; i++) {
234 pixel[i] = (lv_color32_t) {

252 for(i = 0; i < 4; i++) {

254 alpha[i] = 128; /* 50% alpha */
255 }
256
257 res = lv_draw_buf_premultiply(draw_buf);
258 TEST_ASSERT_EQUAL(LV_RESULT_OK, res);
259 TEST_ASSERT(draw_buf->header.flags & LV_IMAGE_FLAGS_PREMULTIPLIED);
260
261 /* Verify premultiplied result: white (0xFFFF) with 50% alpha should become gray */
262 for(i = 0; i < 4; i++) {
263 /* RGB565: 5-6-5 bits, white premultiplied by 0.5 should be approximately half intensity */
264 TEST_ASSERT_EQUAL_UINT16(0x7BEF, pixel[i]); /* Approximate half of white in RGB565 */
265 }
266
267 lv_draw_buf_destroy(draw_buf);
268 }
269
270 /* Test ARGB8565 format */
271 {
272 draw_buf = lv_draw_buf_create(2, 2, LV_COLOR_FORMAT_ARGB8565, 2 * 3);
273
274 /* Fill with test data */
275 uint8_t * pixel = draw_buf->data;
276 for(i = 0; i < 4; i++) {
277 uint16_t c = lv_color_to_u16(lv_color_white()); /* White: 0xFFFF */
278 pixel[0] = c & 0xFF; /* Low byte of RGB565 */
279 pixel[1] = (c >> 8) & 0xFF; /* High byte of RGB565 */
280 pixel[2] = 128; /* Alpha */
281 pixel += 3;
282 }
283
284 res = lv_draw_buf_premultiply(draw_buf);
285 TEST_ASSERT_EQUAL(LV_RESULT_OK, res);
286 TEST_ASSERT(draw_buf->header.flags & LV_IMAGE_FLAGS_PREMULTIPLIED);
287
288 /* Verify premultiplied result */
289 pixel = draw_buf->data;
290 for(i = 0; i < 4; i++) {
291 uint16_t c = (pixel[1] << 8) | pixel[0]; /* Reconstruct RGB565 */
292 /* White premultiplied by 0.5 should be approximately half intensity */
293 TEST_ASSERT_EQUAL_UINT16(0x7BEF, c); /* Approximate half of white in RGB565 */
294 TEST_ASSERT_EQUAL_UINT8(128, pixel[2]); /* Alpha should remain unchanged */
295 pixel += 3;
296 }
297
298 lv_draw_buf_destroy(draw_buf);
299 }

328 TEST_ASSERT_EQUAL_UINT8(127, palette[i].green);
