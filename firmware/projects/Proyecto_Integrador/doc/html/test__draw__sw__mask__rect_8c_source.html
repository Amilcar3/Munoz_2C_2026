77/*
78 * The masked area is clipped to the target layer's buffer.
79 *
80 * `clip_area` may start above and to the left of `buf_area`, which used to make the
81 * offsets handed to lv_draw_layer_go_to_xy() negative. Those wrapped to huge unsigned
82 * values, the row pointer came back NULL, and the row loop dereferenced it.
83 */
84void test_mask_rect_area_starting_before_the_buffer(void)
85{
86 lv_draw_buf_t * buf = mask_rect_buf_create();
87
88 lv_area_t clip_area;
89 lv_area_set(&clip_area, 0, 0, 299, 299);
90
91 lv_area_t mask_area;
92 lv_area_set(&mask_area, 50, 50, 190, 190);
93
94 mask_rect_run(buf, &clip_area, &mask_area, 10);
95
96 /*Well inside the mask — left untouched*/
97 TEST_ASSERT_EQUAL_UINT8(0xff, mask_rect_alpha_at(buf, 0, 0));
98
99 /*(190,190) absolute is the rounded corner of the mask — alpha reduced*/
100 TEST_ASSERT_LESS_THAN_UINT8(0xff, mask_rect_alpha_at(buf, 90, 90));
101
102 /*Past the mask with keep_outside set — left untouched*/
103 TEST_ASSERT_EQUAL_UINT8(0xff, mask_rect_alpha_at(buf, 95, 95));
104
105 lv_draw_buf_destroy(buf);
106}
