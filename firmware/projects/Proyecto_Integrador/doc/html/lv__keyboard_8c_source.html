237void lv_keyboard_set_textarea(lv_obj_t * obj, lv_obj_t * ta)
238{
239 if(ta) {
240 LV_CHECK_OBJ(ta, &lv_textarea_class, return);
241 }
242
243 LV_CHECK_OBJ(obj, MY_CLASS, return);
244 lv_keyboard_t * keyboard = (lv_keyboard_t *)obj;
245
246 /*Hide the cursor of the old Text area if cursor management is enabled*/
247 if(keyboard->ta) {
248 lv_obj_remove_state(keyboard->ta, LV_STATE_FOCUSED);
249 }
250
251 keyboard->ta = ta;
252
253 /*Show the cursor of the new Text area if cursor management is enabled*/
254 if(keyboard->ta) {
255 lv_obj_add_state(keyboard->ta, LV_STATE_FOCUSED);
256 }
257}

326void lv_keyboard_def_event_cb(lv_event_t * e)
327{
328 LV_CHECK_ARG(e != NULL, return);
329
330 lv_obj_t * obj = lv_event_get_current_target(e);
331
332 LV_CHECK_OBJ(obj, MY_CLASS, return);
333 lv_keyboard_t * keyboard = (lv_keyboard_t *)obj;
334 uint32_t btn_id = lv_buttonmatrix_get_selected_button(obj);
335 if(btn_id == LV_BUTTONMATRIX_BUTTON_NONE) return;
336
337 const char * txt = lv_buttonmatrix_get_button_text(obj, btn_id);
338 if(txt == NULL) return;
339
340 if(lv_strcmp(txt, LV_KEYBOARD_CTRL_BUTTON_MODE_TEXT_LOWER) == 0) {
341 keyboard->mode = LV_KEYBOARD_MODE_TEXT_LOWER;
342 lv_buttonmatrix_set_map(obj, kb_map[LV_KEYBOARD_MODE_TEXT_LOWER]);
343 lv_keyboard_update_ctrl_map(obj);
344 return;
345 }
346#if LV_USE_ARABIC_PERSIAN_CHARS == 1
347 else if(lv_strcmp(txt, LV_KEYBOARD_CTRL_BUTTON_MODE_TEXT_ARABIC) == 0) {
348 keyboard->mode = LV_KEYBOARD_MODE_TEXT_ARABIC;
349 lv_buttonmatrix_set_map(obj, kb_map[LV_KEYBOARD_MODE_TEXT_ARABIC]);
350 lv_keyboard_update_ctrl_map(obj);
351 return;
352 }
353#endif
354 else if(lv_strcmp(txt, LV_KEYBOARD_CTRL_BUTTON_MODE_TEXT_UPPER) == 0) {
355 keyboard->mode = LV_KEYBOARD_MODE_TEXT_UPPER;
356 lv_buttonmatrix_set_map(obj, kb_map[LV_KEYBOARD_MODE_TEXT_UPPER]);
357 lv_keyboard_update_ctrl_map(obj);
358 return;
359 }
360 else if(lv_strcmp(txt, LV_KEYBOARD_CTRL_BUTTON_MODE_SPECIAL) == 0) {
361 keyboard->mode = LV_KEYBOARD_MODE_SPECIAL;
362 lv_buttonmatrix_set_map(obj, kb_map[LV_KEYBOARD_MODE_SPECIAL]);
363 lv_keyboard_update_ctrl_map(obj);
364 return;
365 }
366 else if(lv_strcmp(txt, LV_SYMBOL_CLOSE) == 0 || lv_strcmp(txt, LV_SYMBOL_KEYBOARD) == 0) {
367 lv_result_t res = lv_obj_send_event(obj, LV_EVENT_CANCEL, NULL);
368 if(res != LV_RESULT_OK) return;
369
370 if(keyboard->ta) {
371 res = lv_obj_send_event(keyboard->ta, LV_EVENT_CANCEL, NULL);
372 if(res != LV_RESULT_OK) return;
373 }
374 return;
375 }
376 else if(lv_strcmp(txt, LV_SYMBOL_OK) == 0) {
377 lv_result_t res = lv_obj_send_event(obj, LV_EVENT_READY, NULL);
378 if(res != LV_RESULT_OK) return;
379
380 if(keyboard->ta) {
381 res = lv_obj_send_event(keyboard->ta, LV_EVENT_READY, NULL);
382 if(res != LV_RESULT_OK) return;
383 }
384 return;
385 }
386
387 /*Add the characters to the text area if set*/
388 if(keyboard->ta == NULL) return;
389
390 if(lv_strcmp(txt, "Enter") == 0 || lv_strcmp(txt, LV_SYMBOL_NEW_LINE) == 0) {
391 lv_textarea_add_char(keyboard->ta, '\n');
392 if(lv_textarea_get_one_line(keyboard->ta)) {
393 lv_result_t res = lv_obj_send_event(keyboard->ta, LV_EVENT_READY, NULL);
394 if(res != LV_RESULT_OK) return;
395 }
396 }
397 else if(lv_strcmp(txt, LV_SYMBOL_LEFT) == 0) {
398 lv_textarea_cursor_left(keyboard->ta);
399 }
400 else if(lv_strcmp(txt, LV_SYMBOL_RIGHT) == 0) {
401 lv_textarea_cursor_right(keyboard->ta);
402 }
403 else if(lv_strcmp(txt, LV_SYMBOL_BACKSPACE) == 0) {
404 lv_textarea_delete_char(keyboard->ta);
405 }
406 else if(lv_strcmp(txt, "+/-") == 0) {
407 uint32_t cur = lv_textarea_get_cursor_pos(keyboard->ta);
408 const char * ta_txt = lv_textarea_get_text(keyboard->ta);
409 if(ta_txt[0] == '-') {
410 lv_textarea_set_cursor_pos(keyboard->ta, 1);
411 lv_textarea_delete_char(keyboard->ta);
412 lv_textarea_add_char(keyboard->ta, '+');
413 lv_textarea_set_cursor_pos(keyboard->ta, cur);
414 }
415 else if(ta_txt[0] == '+') {
416 lv_textarea_set_cursor_pos(keyboard->ta, 1);
417 lv_textarea_delete_char(keyboard->ta);
418 lv_textarea_add_char(keyboard->ta, '-');
419 lv_textarea_set_cursor_pos(keyboard->ta, cur);
420 }
421 else {
422 lv_textarea_set_cursor_pos(keyboard->ta, 0);
423 lv_textarea_add_char(keyboard->ta, '-');
424 lv_textarea_set_cursor_pos(keyboard->ta, cur + 1);
425 }
426 }
427 else {
428 lv_textarea_add_text(keyboard->ta, txt);
429 }
430}

457static void lv_keyboard_constructor(const lv_obj_class_t * class_p, lv_obj_t * obj)
458{
459 LV_ASSERT(obj != NULL);
460 LV_UNUSED(class_p);
461 lv_obj_set_click_focusable(obj, false);
462
463 lv_keyboard_t * keyboard = (lv_keyboard_t *)obj;
464 keyboard->ta = NULL;
465 keyboard->mode = LV_KEYBOARD_MODE_TEXT_LOWER;
466 keyboard->popovers = 0;
467
468 lv_obj_align(obj, LV_ALIGN_BOTTOM_MID, 0, 0);
469 lv_obj_add_event_cb(obj, lv_keyboard_def_event_cb, LV_EVENT_VALUE_CHANGED, NULL);
470 lv_obj_set_style_base_dir(obj, LV_BASE_DIR_LTR, 0);
471
472 lv_keyboard_update_map(obj);
473}

479static void lv_keyboard_update_map(lv_obj_t * obj)
480{
481 LV_ASSERT(obj != NULL);
482 lv_keyboard_t * keyboard = (lv_keyboard_t *)obj;
483 lv_buttonmatrix_set_map(obj, kb_map[keyboard->mode]);
484 lv_keyboard_update_ctrl_map(obj);
485}

491static void lv_keyboard_update_ctrl_map(lv_obj_t * obj)
492{
493 LV_ASSERT(obj != NULL);
494 lv_keyboard_t * keyboard = (lv_keyboard_t *)obj;
495
496 if(keyboard->popovers) {
497 /*Apply the current control map (already includes LV_BUTTONMATRIX_CTRL_POPOVER flags)*/
498 lv_buttonmatrix_set_ctrl_map(obj, kb_ctrl[keyboard->mode]);
499 }
500 else {
