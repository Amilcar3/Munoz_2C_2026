113void lv_canvas_set_px(lv_obj_t * obj, int32_t x, int32_t y, lv_color_t color, lv_opa_t opa)
114{
115 LV_CHECK_OBJ(obj, MY_CLASS, return);
116
117 lv_canvas_t * canvas = (lv_canvas_t *)obj;
118 lv_draw_buf_t * draw_buf = canvas->draw_buf;
119
120 if(draw_buf == NULL) return;
121
122 lv_color_format_t cf = draw_buf->header.cf;
123 uint8_t * data = lv_draw_buf_goto_xy(draw_buf, x, y);
124 if(data == NULL) return;
125
126 if(LV_COLOR_FORMAT_IS_INDEXED(cf)) {
127 uint8_t shift;
128 uint8_t c_int = color.blue;
129 switch(cf) {
130 case LV_COLOR_FORMAT_I1:
131 shift = 7 - (x & 0x7);
132 break;
133 case LV_COLOR_FORMAT_I2:
134 shift = 6 - 2 * (x & 0x3);
135 break;
136 case LV_COLOR_FORMAT_I4:
137 shift = 4 - 4 * (x & 0x1);
138 break;
139 case LV_COLOR_FORMAT_I8:
140 /*Indexed8 format is a easy case, process and return.*/
141 shift = 0;
142 *data = c_int;
143 default:
144 return;
145 }
146
147 uint8_t bpp = lv_color_format_get_bpp(cf);
148 uint8_t mask = (1 << bpp) - 1;
149 c_int &= mask;
150 *data = (*data & ~(mask << shift)) | (c_int << shift);
151 }
152 else if(cf == LV_COLOR_FORMAT_L8) {
153 *data = lv_color_luminance(color);
154 }
155 else if(cf == LV_COLOR_FORMAT_A8) {
156 *data = opa;
157 }
158 else if(cf == LV_COLOR_FORMAT_RGB565) {
159 lv_color16_t * buf = (lv_color16_t *)data;
160 buf->red = color.red >> 3;
161 buf->green = color.green >> 2;
162 buf->blue = color.blue >> 3;
163 }
164 else if(cf == LV_COLOR_FORMAT_RGB888) {
165 data[2] = color.red;
166 data[1] = color.green;
167 data[0] = color.blue;
168 }
169 else if(cf == LV_COLOR_FORMAT_XRGB8888) {
170 data[2] = color.red;
171 data[1] = color.green;
172 data[0] = color.blue;
173 data[3] = 0xFF;
174 }
175 else if(cf == LV_COLOR_FORMAT_ARGB8888) {
176 lv_color32_t * buf = (lv_color32_t *)data;
177 buf->red = color.red;
178 buf->green = color.green;
179 buf->blue = color.blue;
180 buf->alpha = opa;
181 }
182 else if(cf == LV_COLOR_FORMAT_AL88) {
183 lv_color16a_t * buf = (lv_color16a_t *)data;
184 buf->lumi = lv_color_luminance(color);
185 buf->alpha = 255;
186 }
187 lv_obj_invalidate(obj);
188}

214lv_color32_t lv_canvas_get_px(lv_obj_t * obj, int32_t x, int32_t y)
215{
216 lv_color32_t ret = { 0 };
217 LV_CHECK_OBJ(obj, MY_CLASS, return ret);
218 lv_canvas_t * canvas = (lv_canvas_t *)obj;
219 LV_CHECK_ARG(canvas->draw_buf != NULL, return ret);
220 LV_CHECK_ARG(lv_draw_buf_is_position_valid(canvas->draw_buf, x, y), return ret);
221
222 lv_image_header_t * header = &canvas->draw_buf->header;
223 const uint8_t * px = lv_draw_buf_goto_xy(canvas->draw_buf, x, y);
224 LV_ASSERT(px != NULL);
225
226 switch(header->cf) {
227 case LV_COLOR_FORMAT_ARGB8888:
228 ret = *(lv_color32_t *)px;
229 break;
230 case LV_COLOR_FORMAT_RGB888:
231 case LV_COLOR_FORMAT_XRGB8888:
232 ret.red = px[2];
233 ret.green = px[1];
234 ret.blue = px[0];
235 ret.alpha = 0xFF;
236 break;
237 case LV_COLOR_FORMAT_RGB565: {
238 lv_color16_t * c16 = (lv_color16_t *) px;
239 ret.red = (c16->red * 2106) >> 8; /*To make it rounded*/
240 ret.green = (c16->green * 1037) >> 8;
241 ret.blue = (c16->blue * 2106) >> 8;
242 ret.alpha = 0xFF;
243 break;
244 }
245 case LV_COLOR_FORMAT_A8: {
246 lv_color_t alpha_color = lv_obj_get_style_image_recolor_internal(obj, LV_PART_MAIN);
247 ret.red = alpha_color.red;
248 ret.green = alpha_color.green;
249 ret.blue = alpha_color.blue;
250 ret.alpha = px[0];
251 break;
252 }
253 case LV_COLOR_FORMAT_L8: {
254 ret.red = *px;
255 ret.green = *px;
256 ret.blue = *px;
257 ret.alpha = 0xFF;
258 break;
259 }
260 default:
261 lv_memzero(&ret, sizeof(lv_color32_t));
262 break;
263 }
264
265 return ret;
266}

304void lv_canvas_fill_bg(lv_obj_t * obj, lv_color_t color, lv_opa_t opa)
305{
306 LV_CHECK_OBJ(obj, MY_CLASS, return);
307 lv_canvas_t * canvas = (lv_canvas_t *)obj;
308 lv_draw_buf_t * draw_buf = canvas->draw_buf;
309 LV_CHECK_ARG(draw_buf != NULL, return);
310
311 lv_image_header_t * header = &draw_buf->header;
312 uint32_t x;
313 uint32_t y;
314
315 uint32_t stride = header->stride;
316 uint8_t * data = draw_buf->data;
317 if(header->cf == LV_COLOR_FORMAT_RGB565) {
318 uint16_t c16 = lv_color_to_u16(color);
319 for(y = 0; y < header->h; y++) {
320 uint16_t * buf16 = (uint16_t *)(data + y * stride);
321 for(x = 0; x < header->w; x++) {
322 buf16[x] = c16;
323 }
324 }
325 }
326 else if(header->cf == LV_COLOR_FORMAT_XRGB8888 || header->cf == LV_COLOR_FORMAT_ARGB8888) {
327 uint32_t c32 = lv_color_to_u32(color);
328 if(header->cf == LV_COLOR_FORMAT_ARGB8888) {
329 c32 &= 0x00ffffff;
330 c32 |= (uint32_t)opa << 24;
331 }
332 for(y = 0; y < header->h; y++) {
333 uint32_t * buf32 = (uint32_t *)(data + y * stride);
334 for(x = 0; x < header->w; x++) {
335 buf32[x] = c32;
336 }
337 }
338 }
339 else if(header->cf == LV_COLOR_FORMAT_RGB888) {
340 for(y = 0; y < header->h; y++) {
341 uint8_t * buf8 = (uint8_t *)(data + y * stride);
342 for(x = 0; x < header->w * 3; x += 3) {
343 buf8[x + 0] = color.blue;
344 buf8[x + 1] = color.green;
345 buf8[x + 2] = color.red;
346 }
347 }
348 }
349 else if(header->cf == LV_COLOR_FORMAT_L8) {
350 uint8_t c8 = lv_color_luminance(color);
351 for(y = 0; y < header->h; y++) {
352 uint8_t * buf = (uint8_t *)(data + y * stride);
353 for(x = 0; x < header->w; x++) {
354 buf[x] = c8;
355 }
356 }
357 }
358 else if(header->cf == LV_COLOR_FORMAT_AL88) {
359 lv_color16a_t c;
360 c.lumi = lv_color_luminance(color);
361 c.alpha = 255;
362 for(y = 0; y < header->h; y++) {
363 lv_color16a_t * buf = (lv_color16a_t *)(data + y * stride);
364 for(x = 0; x < header->w; x++) {
365 buf[x] = c;
366 }
367 }
368 }
369
370 else {
371 for(y = 0; y < header->h; y++) {
372 for(x = 0; x < header->w; x++) {
373 lv_canvas_set_px(obj, x, y, color, opa);
374 }
375 }
376 }
377
378 lv_draw_buf_flush_cache(canvas->draw_buf, NULL);
379 lv_obj_invalidate(obj);
380}

403void lv_canvas_finish_layer(lv_obj_t * canvas, lv_layer_t * layer)
404{
405 LV_CHECK_OBJ(canvas, MY_CLASS, return);
406 LV_CHECK_ARG(layer != NULL, return);
407
408 if(layer->draw_task_head == NULL) {
409 lv_draw_unit_send_event(NULL, LV_EVENT_CHILD_DELETED, layer);
410 return;
411 }
412
413 layer->all_tasks_added = true;
414
415 bool task_dispatched;
416 lv_display_t * display = layer->display;
417 LV_ASSERT(display != NULL);
418
419 while(layer->draw_task_head) {
420 lv_draw_dispatch_wait_for_request();
421
422 task_dispatched = lv_draw_dispatch_layer(display, layer);
423
424 /*TODO(v10): dispatching a layer can create new sub layers that the layer depends on.
425 * These sub layers are created in display's layer list because `lv_draw_layer_create`
426 * attaches this layer to the display layer head, so in here we need to iterate over that
427 * list and dispatchs any sub layers that are required by the main layer
428 */
429 lv_layer_t * sub_layer = display->layer_head;
430 while(sub_layer) {
431 if(layer_is_descendant(sub_layer, layer) && lv_draw_dispatch_layer(display, sub_layer)) {
432 task_dispatched = true;
433 }
434 sub_layer = sub_layer->next;
435 }
436
437 if(!task_dispatched) {
438 lv_draw_wait_for_finish();
439 lv_draw_dispatch_request();
440 }
441 }
442
443 lv_draw_unit_send_event(NULL, LV_EVENT_SCREEN_LOAD_START, layer);
444 lv_draw_unit_send_event(NULL, LV_EVENT_CHILD_DELETED, layer);
445 lv_obj_invalidate(canvas);
446}
