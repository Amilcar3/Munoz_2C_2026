100struct _lv_vg_lite_grad_ctx_t * lv_vg_lite_grad_ctx_create(uint32_t cache_cnt, struct _lv_draw_vg_lite_unit_t * unit)
101{
102 LV_ASSERT_MSG(cache_cnt > 0, "cache_cnt should be greater than 0");
103
104 lv_cache_ops_t ops = {
105 .compare_cb = (lv_cache_compare_cb_t)grad_compare_cb,
106 .create_cb = (lv_cache_create_cb_t)grad_create_cb,
107 .free_cb = (lv_cache_free_cb_t)grad_free_cb,
108 };
109
110 lv_vg_lite_grad_ctx_t * ctx = lv_malloc_zeroed(sizeof(lv_vg_lite_grad_ctx_t));
111 LV_ASSERT_MALLOC(ctx);
112 ctx->unit = unit;
113
114 lv_ll_init(&ctx->item_pool, sizeof(grad_item_t));
115
116 ctx->cache = lv_cache_create(&lv_cache_class_lru_ll_count, sizeof(grad_item_ref_t), cache_cnt, ops);
117 lv_cache_set_name(ctx->cache, "VG_GRAD");
118 ctx->pending = lv_vg_lite_pending_create(sizeof(lv_cache_entry_t *), 4);
119 lv_vg_lite_pending_set_free_cb(ctx->pending, grad_cache_release_cb, ctx->cache);
120
121 return ctx;
122}

147bool lv_vg_lite_draw_grad(
148 struct _lv_vg_lite_grad_ctx_t * ctx,
149 vg_lite_buffer_t * buffer,
150 vg_lite_path_t * path,
151 const lv_vector_gradient_t * grad,
152 const vg_lite_matrix_t * grad_matrix,
153 const vg_lite_matrix_t * matrix,
154 vg_lite_fill_t fill,
155 vg_lite_blend_t blend)
156{
157 LV_ASSERT_NULL(ctx);
158 LV_VG_LITE_ASSERT_DEST_BUFFER(buffer);
159 LV_VG_LITE_ASSERT_PATH(path);
160 LV_ASSERT_NULL(grad);
161 LV_VG_LITE_ASSERT_MATRIX(grad_matrix);
162 LV_VG_LITE_ASSERT_MATRIX(matrix);
163
164 /* check radial gradient is supported */
165 if(grad->style == LV_VECTOR_GRADIENT_STYLE_RADIAL) {
166 if(!vg_lite_query_feature(gcFEATURE_BIT_VG_RADIAL_GRADIENT)) {
167 LV_LOG_WARN("radial gradient is not supported");
168 return false;
169 }
170
171 /* check if the radius is valid */
172 if(grad->cr <= 0) {
173 LV_LOG_WARN("radius: %f is not valid", grad->cr);
174 return false;
175 }
176 }
177
178 /* check spread mode is supported */
179 if(grad->spread == LV_VECTOR_GRADIENT_SPREAD_REPEAT || grad->spread == LV_VECTOR_GRADIENT_SPREAD_REFLECT) {
180 if(!vg_lite_query_feature(gcFEATURE_BIT_VG_IM_REPEAT_REFLECT)) {
181 LV_LOG_WARN("repeat/reflect spread(%d) is not supported", grad->spread);
182 return false;
183 }
184 }
185
186 grad_item_t * grad_item = grad_get(ctx, grad);
187 if(!grad_item) {
188 LV_LOG_WARN("Failed to get gradient, style: %d", grad->style);
189 return false;
190 }
191 LV_ASSERT(grad_item->lv.style == grad->style);
192
193 switch(grad_item->type) {
194 case GRAD_TYPE_LINEAR: {
195 vg_lite_linear_gradient_t * linear_grad = &grad_item->vg.linear;
196 vg_lite_matrix_t * grad_mat_p = vg_lite_get_grad_matrix(linear_grad);
197 LV_ASSERT_NULL(grad_mat_p);
198 *grad_mat_p = *grad_matrix;
199 grad_point_to_matrix(grad_mat_p, grad->x1, grad->y1, grad->x2, grad->y2);
200
201 LV_PROFILER_DRAW_BEGIN_TAG("vg_lite_draw_grad");
202 LV_VG_LITE_CHECK_ERROR(vg_lite_draw_grad(
203 buffer,
204 path,
205 fill,
206 (vg_lite_matrix_t *)matrix,
207 linear_grad,
208 blend),
209 /* Dump parameters */
210 {
211 lv_vg_lite_buffer_dump_info(buffer);
212 lv_vg_lite_path_dump_info(path);
213 LV_LOG_USER("fill_rule: 0x%X", (int)fill);
214 lv_vg_lite_matrix_dump_info(matrix);
215 lv_vg_lite_linear_gradient_dump_info(linear_grad);
216 LV_LOG_USER("blend_mode: 0x%X", (int)blend);
217 });
218 LV_PROFILER_DRAW_END_TAG("vg_lite_draw_grad");
219 }
220 break;
221 case GRAD_TYPE_LINEAR_EXT: {
222 vg_lite_ext_linear_gradient_t * linear_grad = &grad_item->vg.linear_ext;
223 vg_lite_matrix_t * grad_mat_p = vg_lite_get_linear_grad_matrix(linear_grad);
224 LV_ASSERT_NULL(grad_mat_p);
225 *grad_mat_p = *grad_matrix;
226
227 LV_PROFILER_DRAW_BEGIN_TAG("vg_lite_draw_linear_grad");
228 LV_VG_LITE_CHECK_ERROR(vg_lite_draw_linear_grad(
229 buffer,
230 path,
231 fill,
232 (vg_lite_matrix_t *)matrix,
233 linear_grad,
234 0,
235 blend,
236 VG_LITE_FILTER_LINEAR),
237 /* Dump parameters */
238 {
239 lv_vg_lite_buffer_dump_info(buffer);
240 lv_vg_lite_path_dump_info(path);
241 LV_LOG_USER("fill_rule: 0x%X", (int)fill);
242 lv_vg_lite_matrix_dump_info(matrix);
243 lv_vg_lite_ext_linear_gradient_dump_info(linear_grad);
244 LV_LOG_USER("blend_mode: 0x%X", (int)blend);
245 });
246 LV_PROFILER_DRAW_END_TAG("vg_lite_draw_linear_grad");
247 }
248 break;
249
250 case GRAD_TYPE_RADIAL: {
251 vg_lite_radial_gradient_t * radial_gradient = &grad_item->vg.radial;
252 vg_lite_matrix_t * grad_mat_p = vg_lite_get_radial_grad_matrix(radial_gradient);
253 LV_ASSERT_NULL(grad_mat_p);
254 *grad_mat_p = *grad_matrix;
255
256 LV_PROFILER_DRAW_BEGIN_TAG("vg_lite_draw_radial_grad");
257 LV_VG_LITE_CHECK_ERROR(
258 vg_lite_draw_radial_grad(
259 buffer,
260 path,
261 fill,
262 (vg_lite_matrix_t *)matrix,
263 radial_gradient,
264 0,
265 blend,
266 VG_LITE_FILTER_LINEAR),
267 /* Dump parameters */
268 {
269 lv_vg_lite_buffer_dump_info(buffer);
270 lv_vg_lite_path_dump_info(path);
271 LV_LOG_USER("fill_rule: 0x%X", (int)fill);
272 lv_vg_lite_matrix_dump_info(matrix);
273 lv_vg_lite_radial_gradient_dump_info(radial_gradient);
274 LV_LOG_USER("blend_mode: 0x%X", (int)blend);
275 });
276 LV_PROFILER_DRAW_END_TAG("vg_lite_draw_radial_grad");
277 }
278 break;
279
280 default:
281 LV_LOG_ERROR("Unsupported gradient type: %d", grad_item->type);
282 return false;
283 }
284
285 return true;
286}

288bool lv_vg_lite_draw_grad_helper(
289 struct _lv_vg_lite_grad_ctx_t * ctx,
290 vg_lite_buffer_t * buffer,
291 vg_lite_path_t * path,
292 const lv_area_t * area,
293 const lv_grad_dsc_t * grad_dsc,
294 const vg_lite_matrix_t * matrix,
295 vg_lite_fill_t fill,
296 vg_lite_blend_t blend)
297{
298 LV_ASSERT_NULL(ctx);
299 LV_VG_LITE_ASSERT_DEST_BUFFER(buffer);
300 LV_VG_LITE_ASSERT_PATH(path);
301 LV_ASSERT_NULL(area);
302 LV_ASSERT_NULL(grad_dsc);
303 LV_VG_LITE_ASSERT_MATRIX(matrix);
304
305 lv_vector_gradient_t grad;
306 lv_memzero(&grad, sizeof(grad));
307
308 grad.style = LV_VECTOR_GRADIENT_STYLE_LINEAR;
309 grad.stops_count = grad_dsc->stops_count;
310 lv_memcpy(grad.stops, grad_dsc->stops, sizeof(lv_grad_stop_t) * grad_dsc->stops_count);
311
312 /*convert to spread mode*/
313 switch(grad_dsc->extend) {
314 case LV_GRAD_EXTEND_PAD:
315 grad.spread = LV_VECTOR_GRADIENT_SPREAD_PAD;
316 break;
317 case LV_GRAD_EXTEND_REPEAT:
318 grad.spread = LV_VECTOR_GRADIENT_SPREAD_REPEAT;
319 break;
320 case LV_GRAD_EXTEND_REFLECT:
321 grad.spread = LV_VECTOR_GRADIENT_SPREAD_REFLECT;
322 break;
323 default:
324 LV_LOG_WARN("Unsupported gradient extend mode: %d", grad_dsc->extend);
325 grad.spread = LV_VECTOR_GRADIENT_SPREAD_PAD;
326 break;
327 }
328
329 switch(grad_dsc->dir) {
330 case LV_GRAD_DIR_VER:
331 grad.x1 = area->x1;
332 grad.y1 = area->y1;
333 grad.x2 = area->x1;
334 grad.y2 = area->y2 + 1;
335 break;
336
337 case LV_GRAD_DIR_HOR:
338 grad.x1 = area->x1;
339 grad.y1 = area->y1;
340 grad.x2 = area->x2 + 1;
341 grad.y2 = area->y1;
342 break;
343
344 case LV_GRAD_DIR_LINEAR: {
345 int32_t w = lv_area_get_width(area);
346 int32_t h = lv_area_get_height(area);
347
348 grad.x1 = lv_pct_to_px(grad_dsc->params.linear.start.x, w) + area->x1;
349 grad.y1 = lv_pct_to_px(grad_dsc->params.linear.start.y, h) + area->y1;
350 grad.x2 = lv_pct_to_px(grad_dsc->params.linear.end.x, w) + area->x1;
351 grad.y2 = lv_pct_to_px(grad_dsc->params.linear.end.y, h) + area->y1;
352 }
353 break;
354
355 case LV_GRAD_DIR_RADIAL: {
356 grad.style = LV_VECTOR_GRADIENT_STYLE_RADIAL;
357 int32_t w = lv_area_get_width(area);
358 int32_t h = lv_area_get_height(area);
359
360 grad.cx = lv_pct_to_px(grad_dsc->params.radial.focal.x, w) + area->x1;
361 grad.cy = lv_pct_to_px(grad_dsc->params.radial.focal.y, h) + area->y1;
362 int32_t end_extent_x = lv_pct_to_px(grad_dsc->params.radial.end_extent.x, w) + area->x1;
363 int32_t end_extent_y = lv_pct_to_px(grad_dsc->params.radial.end_extent.y, h) + area->y1;
364 grad.cr = LV_MAX(end_extent_x - grad.cx, end_extent_y - grad.cy);
365 }
366 break;
367
368 default:
369 LV_LOG_WARN("Unsupported gradient direction: %d", grad_dsc->dir);
370 return false;
371 }
372
373 return lv_vg_lite_draw_grad(ctx, buffer, path, &grad, matrix, matrix, fill, blend);
374}

380static grad_item_t * grad_get(lv_vg_lite_grad_ctx_t * ctx, const lv_vector_gradient_t * grad)
381{
382 LV_PROFILER_DRAW_BEGIN;
383 LV_ASSERT_NULL(ctx);
384 LV_ASSERT_NULL(grad);
385
386 grad_item_ref_t search_key = &ctx->local_grad_item;
387 lv_memzero(search_key, sizeof(ctx->local_grad_item));
388 search_key->type = lv_grad_style_to_type(grad->style);
389 search_key->lv = *grad;
390
391 lv_cache_entry_t * cache_node_entry = lv_cache_acquire(ctx->cache, &search_key, ctx);
392 if(cache_node_entry == NULL) {
393 /* check if the cache is full */
394 size_t free_size = lv_cache_get_free_size(ctx->cache, NULL);
395 if(free_size == 0) {
396 LV_LOG_INFO("grad cache is full, release all pending cache entries");
397 lv_vg_lite_finish(ctx->unit);
398 }
399
400 cache_node_entry = lv_cache_acquire_or_create(ctx->cache, &search_key, ctx);
401 if(cache_node_entry == NULL) {
402 LV_LOG_ERROR("grad cache creating failed");
403 LV_PROFILER_DRAW_END;
404 return NULL;
405 }
406 }
407
408 /* Add the new entry to the pending list */
409 lv_vg_lite_pending_add(ctx->pending, &cache_node_entry);
410
411 grad_item_ref_t * grad_item_ref = lv_cache_entry_get_data(cache_node_entry);
412
413 LV_PROFILER_DRAW_END;
414 return *grad_item_ref;
415}

417static grad_item_t * grad_item_pool_alloc(lv_vg_lite_grad_ctx_t * ctx, grad_type_t type)
418{
419 LV_ASSERT_NULL(ctx);
420
421 grad_item_t * item = lv_ll_get_head(&ctx->item_pool);
422
423 /* Try to obtain a free node from the head */
424 if(item && item->type == GRAD_TYPE_FREE) {
425 lv_ll_move_before(&ctx->item_pool, item, NULL);
426 LV_LOG_TRACE("reuse item: %p, type: %d", (void *)item, type);
427 }
428 else {
429 /* Allocate a new node if the pool is empty or all nodes are in use */
430 item = lv_ll_ins_tail(&ctx->item_pool);
431 LV_ASSERT_MALLOC(item);
432 if(!item) {
433 LV_LOG_ERROR("alloc grad item failed");
434 return NULL;
435 }
436
437 LV_LOG_TRACE("alloc new item: %p, type: %d, pool size: %" LV_PRIu32,
438 (void *)item, type, lv_ll_get_len(&ctx->item_pool));
439 }
440
441 lv_memzero(item, sizeof(grad_item_t));
442 item->ctx = ctx;
443 item->type = type;
444 return item;
445}

448{
449 LV_ASSERT_NULL(item);
450 LV_ASSERT_NULL(item->ctx);

465static void convert_color_ramp(vg_lite_color_ramp_t * color_ramp, const lv_vector_gradient_t * grad)
466{
467 LV_ASSERT_NULL(color_ramp);
468 LV_ASSERT_NULL(grad);
469
470 for(uint16_t i = 0; i < grad->stops_count; i++) {
471 color_ramp[i].stop = grad->stops[i].frac / 255.0f;
472 lv_color_t c = grad->stops[i].color;
473
474 color_ramp[i].red = c.red / 255.0f;
475 color_ramp[i].green = c.green / 255.0f;
476 color_ramp[i].blue = c.blue / 255.0f;
477 color_ramp[i].alpha = grad->stops[i].opa / 255.0f;
478 }
479}

481static bool linear_grad_create(grad_item_t * item)
482{
483 LV_PROFILER_DRAW_BEGIN;
484
485 /* Capture error code from LV_VG_LITE_CHECK_ERROR */
486 vg_lite_error_t err = VG_LITE_SUCCESS;
487 LV_VG_LITE_CHECK_ERROR(
488 err = vg_lite_init_grad(&item->vg.linear), {
489 lv_vg_lite_linear_gradient_dump_info(&item->vg.linear);
490 });
491
492 if(err != VG_LITE_SUCCESS) {
493 LV_PROFILER_DRAW_END;
494 return false;
495 }
496
497 vg_lite_uint32_t colors[VLC_MAX_GRADIENT_STOPS];
498 vg_lite_uint32_t stops[VLC_MAX_GRADIENT_STOPS];
499
500 /* Gradient setup */
501 if(item->lv.stops_count > VLC_MAX_GRADIENT_STOPS) {
502 LV_LOG_WARN("Gradient stops limited: %d, max: %d", item->lv.stops_count, VLC_MAX_GRADIENT_STOPS);
503 item->lv.stops_count = VLC_MAX_GRADIENT_STOPS;
504 }
505
506 for(uint16_t i = 0; i < item->lv.stops_count; i++) {
507 stops[i] = item->lv.stops[i].frac;
508 const lv_color_t * c = &item->lv.stops[i].color;
509 lv_opa_t opa = item->lv.stops[i].opa;
510
511 /* lvgl color -> gradient color */
512 lv_color_t grad_color = lv_color_make(c->blue, c->green, c->red);
513 colors[i] = lv_vg_lite_color(grad_color, opa, true);
514 }
515
516 LV_VG_LITE_CHECK_ERROR(
517 vg_lite_set_grad(&item->vg.linear, item->lv.stops_count, colors, stops), {
518 lv_vg_lite_linear_gradient_dump_info(&item->vg.linear);
519 });
520
521 LV_PROFILER_DRAW_BEGIN_TAG("vg_lite_update_grad");
522 LV_VG_LITE_CHECK_ERROR(
523 vg_lite_update_grad(&item->vg.linear), {
524 lv_vg_lite_linear_gradient_dump_info(&item->vg.linear);
525 });
526 LV_PROFILER_DRAW_END_TAG("vg_lite_update_grad");
527
528 LV_PROFILER_DRAW_END;
529 return true;
530}

532static bool linear_ext_grad_create(grad_item_t * item, vg_lite_color_ramp_t * color_ramp)
533{
534 LV_PROFILER_DRAW_BEGIN;
535
536 if(item->lv.stops_count > VLC_MAX_COLOR_RAMP_STOPS) {
537 LV_LOG_WARN("Gradient stops limited: %d, max: %d", item->lv.stops_count, VLC_MAX_GRADIENT_STOPS);
538 item->lv.stops_count = VLC_MAX_COLOR_RAMP_STOPS;
539 }
540
541 const vg_lite_linear_gradient_parameter_t grad_param = {
542 .X0 = item->lv.x1,
543 .Y0 = item->lv.y1,
544 .X1 = item->lv.x2,
545 .Y1 = item->lv.y2,
546 };
547
548 convert_color_ramp(color_ramp, &item->lv);
549
550 LV_PROFILER_DRAW_BEGIN_TAG("vg_lite_set_linear_grad");
551 LV_VG_LITE_CHECK_ERROR(
552 vg_lite_set_linear_grad(
553 &item->vg.linear_ext,
554 item->lv.stops_count,
555 color_ramp,
556 grad_param,
557 lv_spread_to_vg(item->lv.spread),
558 1),
559 /* Dump parameters */
560 {
561 lv_vg_lite_ext_linear_gradient_dump_info(&item->vg.linear_ext);
562 });
563 LV_PROFILER_DRAW_END_TAG("vg_lite_set_linear_grad");
564
565 LV_PROFILER_DRAW_BEGIN_TAG("vg_lite_update_linear_grad");
566 vg_lite_error_t err = VG_LITE_SUCCESS;
567 vg_lite_identity(vg_lite_get_linear_grad_matrix(&item->vg.linear_ext));
568 LV_VG_LITE_CHECK_ERROR(err = vg_lite_update_linear_grad(&item->vg.linear_ext), {
569 lv_vg_lite_ext_linear_gradient_dump_info(&item->vg.linear_ext);
570 });
571 LV_PROFILER_DRAW_END_TAG("vg_lite_update_linear_grad");
572
573 LV_PROFILER_DRAW_END;
574 return err == VG_LITE_SUCCESS;
575}

577static bool radial_grad_create(grad_item_t * item, vg_lite_color_ramp_t * color_ramp)
578{
579 LV_PROFILER_DRAW_BEGIN;
580
581 if(item->lv.stops_count > VLC_MAX_COLOR_RAMP_STOPS) {
582 LV_LOG_WARN("Gradient stops limited: %d, max: %d", item->lv.stops_count, VLC_MAX_GRADIENT_STOPS);
583 item->lv.stops_count = VLC_MAX_COLOR_RAMP_STOPS;

640static void grad_point_to_matrix(vg_lite_matrix_t * grad_matrix, float x1, float y1, float x2, float y2)
641{
642 vg_lite_translate(x1, y1, grad_matrix);
643

645 vg_lite_rotate(angle, grad_matrix);

647 vg_lite_scale(length / 256.0f, 1, grad_matrix);

753static lv_cache_compare_res_t grad_compare_cb(const grad_item_ref_t * lhs_ref, const grad_item_ref_t * rhs_ref)
754{
755 const grad_item_t * lhs = *lhs_ref;
756 const grad_item_t * rhs = *rhs_ref;
757
758 /* compare type first */
759 if(lhs->type != rhs->type) {
760 return lhs->type > rhs->type ? 1 : -1;
761 }
762
763 /* compare spread mode */
764 if(lhs->lv.spread != rhs->lv.spread) {
765 return lhs->lv.spread > rhs->lv.spread ? 1 : -1;
766 }
767
768 /* compare gradient parameters */
769 switch(lhs->type) {
770 case GRAD_TYPE_LINEAR:
771 /* no extra compare needed */
772 break;
773
774 case GRAD_TYPE_LINEAR_EXT:
775 if(!math_equal(lhs->lv.x1, rhs->lv.x1)) {
776 return lhs->lv.x1 > rhs->lv.x1 ? 1 : -1;
777 }
778
779 if(!math_equal(lhs->lv.y1, rhs->lv.y1)) {
780 return lhs->lv.y1 > rhs->lv.y1 ? 1 : -1;
781 }
782
783 if(!math_equal(lhs->lv.x2, rhs->lv.x2)) {
784 return lhs->lv.x2 > rhs->lv.x2 ? 1 : -1;
785 }
786
787 if(!math_equal(lhs->lv.y2, rhs->lv.y2)) {
788 return lhs->lv.y2 > rhs->lv.y2 ? 1 : -1;
789 }
790 break;
791
792 case GRAD_TYPE_RADIAL:
793 if(!math_equal(lhs->lv.cx, rhs->lv.cx)) {
794 return lhs->lv.cx > rhs->lv.cx ? 1 : -1;
795 }
796
797 if(!math_equal(lhs->lv.cy, rhs->lv.cy)) {
798 return lhs->lv.cy > rhs->lv.cy ? 1 : -1;
799 }
800
801 if(!math_equal(lhs->lv.cr, rhs->lv.cr)) {
802 return lhs->lv.cr > rhs->lv.cr ? 1 : -1;
803 }
804 break;
805
806 default:
807 LV_LOG_ERROR("unknown gradient type: %d", lhs->type);
808 break;
809 }
810
811 /* compare stops count and stops */
812 if(lhs->lv.stops_count != rhs->lv.stops_count) {
813 return lhs->lv.stops_count > rhs->lv.stops_count ? 1 : -1;
814 }
815
816 int cmp_res = lv_memcmp(lhs->lv.stops, rhs->lv.stops,
817 sizeof(lv_grad_stop_t) * lhs->lv.stops_count);
818 if(cmp_res != 0) {
819 return cmp_res > 0 ? 1 : -1;
820 }
821
822 return 0;
823}
