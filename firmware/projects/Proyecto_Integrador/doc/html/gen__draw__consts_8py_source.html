21def parse_draw_unit_types(draw_dir: Path) -> dict[str, str]:
22 """Scan draw unit .c files for name-to-struct-type mappings.
23
24 Looks for patterns like: unit->base_unit.name = "SW";
25 Then finds the corresponding struct type from the variable declaration.
26 """
27 mappings = {}
28 for c_file in draw_dir.rglob("*.c"):
29 text = c_file.read_text()
30 for m in re.finditer(r'(\w+)->base_unit\.name\s*=\s*"(\w+)"', text):
31 var_name = m.group(1)
32 unit_name = m.group(2)
33 decl = re.search(
34 rf"(lv_draw_\w+_unit_t)\s*\*\s*{re.escape(var_name)}\b", text
35 )
36 if decl:
37 mappings[unit_name] = decl.group(1)
38 return mappings
