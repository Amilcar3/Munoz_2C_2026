42void lv_ll_init(lv_ll_t * ll_p, uint32_t node_size)
43{
44 LV_CHECK_ARG(ll_p != NULL, return);
45 LV_CHECK_ARG_MSG(node_size > 0, return, "cannot initialize a linked list with empty nodes");
46 ll_p->head = NULL;
47 ll_p->tail = NULL;
48#ifdef LV_ARCH_64
49 /*Round the size up to 8*/
50 node_size = (node_size + 7) & (~0x7);
51#else
52 /*Round the size up to 4*/
53 node_size = (node_size + 3) & (~0x3);
54#endif
55
56 ll_p->n_size = node_size;
57}

59void * lv_ll_ins_head(lv_ll_t * ll_p)
60{
61 LV_CHECK_ARG(ll_p != NULL, return NULL);
62
63 lv_ll_node_t * n_new;
64
65 n_new = lv_malloc(ll_p->n_size + LL_NODE_META_SIZE);
66
67 if(n_new != NULL) {
68 node_set_prev(ll_p, n_new, NULL); /*No prev. before the new head*/
69 node_set_next(ll_p, n_new, ll_p->head); /*After new comes the old head*/
70
71 if(ll_p->head != NULL) { /*If there is old head then before it goes the new*/
72 node_set_prev(ll_p, ll_p->head, n_new);
73 }
74
75 ll_p->head = n_new; /*Set the new head in the dsc.*/
76 if(ll_p->tail == NULL) { /*If there is no tail (1. node) set the tail too*/
77 ll_p->tail = n_new;
78 }
79 }
80
81 return n_new;
82}

84void * lv_ll_ins_prev(lv_ll_t * ll_p, void * n_act)
85{
86 LV_CHECK_ARG(ll_p != NULL, return NULL);
87 LV_CHECK_ARG(n_act != NULL, return NULL);
88
89 if(lv_ll_get_head(ll_p) == n_act) {
90 return lv_ll_ins_head(ll_p);
91 }
92 lv_ll_node_t * n_new = lv_malloc(ll_p->n_size + LL_NODE_META_SIZE);
93 if(n_new == NULL) return NULL;
94
95 lv_ll_node_t * n_prev;
96 n_prev = lv_ll_get_prev(ll_p, n_act);
97 node_set_next(ll_p, n_prev, n_new);
98 node_set_prev(ll_p, n_new, n_prev);
99 node_set_prev(ll_p, n_act, n_new);
100 node_set_next(ll_p, n_new, n_act);
101
102 return n_new;
103}

129void lv_ll_remove(lv_ll_t * ll_p, void * node_p)
130{
131 LV_CHECK_ARG(ll_p != NULL, return);
132 if(!node_p) {
133 return;
134 }
135
136 if(lv_ll_get_head(ll_p) == node_p) {
137 /*The new head will be the node after 'node_p'*/
138 ll_p->head = lv_ll_get_next(ll_p, node_p);
139 if(ll_p->head == NULL) {
140 ll_p->tail = NULL;
141 }
142 else {
143 node_set_prev(ll_p, ll_p->head, NULL);
144 }
145 }
146 else if(lv_ll_get_tail(ll_p) == node_p) {
147 /*The new tail will be the node before 'node_p'*/
148 ll_p->tail = lv_ll_get_prev(ll_p, node_p);
149 if(ll_p->tail == NULL) {
150 ll_p->head = NULL;
151 }
152 else {
153 node_set_next(ll_p, ll_p->tail, NULL);
154 }
155 }
156 else {
157 lv_ll_node_t * n_prev = lv_ll_get_prev(ll_p, node_p);
158 lv_ll_node_t * n_next = lv_ll_get_next(ll_p, node_p);
159
160 node_set_next(ll_p, n_prev, n_next);
161 node_set_prev(ll_p, n_next, n_prev);
162 }

165void lv_ll_clear_custom(lv_ll_t * ll_p, void(*cleanup)(void *))
166{
167 LV_CHECK_ARG(ll_p != NULL, return);
168
169 void * i;
170 void * i_next;
171
172 i = lv_ll_get_head(ll_p);
173 i_next = NULL;
174
175 while(i != NULL) {
176 i_next = lv_ll_get_next(ll_p, i);
177 if(cleanup == NULL) {
178 lv_ll_remove(ll_p, i);
179 lv_free(i);
180 }
181 else {
182 cleanup(i);
183 }
184 i = i_next;
185 }

188void lv_ll_chg_list(lv_ll_t * ll_ori_p, lv_ll_t * ll_new_p, void * node, bool head)
189{
190 LV_CHECK_ARG(ll_ori_p != NULL, return);
191 LV_CHECK_ARG(ll_new_p != NULL, return);
192 LV_CHECK_ARG(node != NULL, return);
193
194 lv_ll_remove(ll_ori_p, node);
195
196 if(head) {
197 /*Set node as head*/
198 node_set_prev(ll_new_p, node, NULL);
199 node_set_next(ll_new_p, node, ll_new_p->head);
200
201 if(ll_new_p->head != NULL) { /*If there is old head then before it goes the new*/
202 node_set_prev(ll_new_p, ll_new_p->head, node);
203 }
204
205 ll_new_p->head = node; /*Set the new head in the dsc.*/
206 if(ll_new_p->tail == NULL) { /*If there is no tail (first node) set the tail too*/
207 ll_new_p->tail = node;
208 }
209 }
210 else {
211 /*Set node as tail*/
212 node_set_prev(ll_new_p, node, ll_new_p->tail);
213 node_set_next(ll_new_p, node, NULL);
214
215 if(ll_new_p->tail != NULL) { /*If there is old tail then after it goes the new*/
216 node_set_next(ll_new_p, ll_new_p->tail, node);
217 }
218
219 ll_new_p->tail = node; /*Set the new tail in the dsc.*/
220 if(ll_new_p->head == NULL) { /*If there is no head (first node) set the head too*/
221 ll_new_p->head = node;
222 }
223 }

276void lv_ll_move_before(lv_ll_t * ll_p, void * n_act, void * n_after)
277{
278 LV_CHECK_ARG(ll_p != NULL, return);
279 LV_CHECK_ARG(n_act != NULL, return);
280
281 if(n_act == n_after) return; /*Can't move before itself*/
282
283 void * n_before;
284 if(n_after != NULL)
285 n_before = lv_ll_get_prev(ll_p, n_after);
286 else
287 n_before = lv_ll_get_tail(ll_p); /*if `n_after` is NULL `n_act` should be the new tail*/
288
289 if(n_act == n_before) return; /*Already before `n_after`*/
290
291 /*It's much easier to remove from the list and add again*/
292 lv_ll_remove(ll_p, n_act);
293
294 /*Add again by setting the prev. and next nodes*/
295 node_set_next(ll_p, n_before, n_act);
296 node_set_prev(ll_p, n_act, n_before);
297 node_set_prev(ll_p, n_after, n_act);
298 node_set_next(ll_p, n_act, n_after);
299
300 /*If `n_act` was moved before NULL then it become the new tail*/
301 if(n_after == NULL) ll_p->tail = n_act;
302
303 /*If `n_act` was moved before `NULL` then it's the new head*/
304 if(n_before == NULL) ll_p->head = n_act;
305}
