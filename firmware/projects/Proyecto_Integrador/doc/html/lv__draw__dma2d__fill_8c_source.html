37void lv_draw_dma2d_fill(lv_draw_task_t * t, void * first_pixel, int32_t w, int32_t h, int32_t stride)
38{
39 lv_draw_fill_dsc_t * dsc = t->draw_dsc;
40 lv_color_format_t cf = dsc->base.layer->color_format;
41 lv_opa_t opa = dsc->opa;
42
43 lv_draw_dma2d_output_cf_t output_cf = lv_draw_dma2d_cf_to_dma2d_output_cf(cf);
44 uint32_t cf_size = LV_COLOR_FORMAT_GET_SIZE(cf);
45
46 uint32_t output_offset = (stride / cf_size) - w;
47
48 lv_draw_dma2d_configuration_t conf = {
49 .mode = LV_DRAW_DMA2D_MODE_REGISTER_TO_MEMORY,
50 .w = w,
51 .h = h,
52
53 .output_address = first_pixel,
54 .output_offset = output_offset,
55 .output_cf = output_cf,
56 };
57
58 if(dsc->opa >= LV_OPA_MAX) {
59 conf.reg_to_mem_mode_color = lv_draw_dma2d_color_to_dma2d_color(output_cf, dsc->color);
60 }
61 else {
62 conf.mode = LV_DRAW_DMA2D_MODE_MEMORY_TO_MEMORY_WITH_BLENDING;
63 conf.fg_color = lv_color_to_u32(dsc->color);
64 conf.fg_address = first_pixel;
65 conf.fg_offset = output_offset;
66 conf.fg_alpha_mode = LV_DRAW_DMA2D_ALPHA_MODE_REPLACE_ALPHA_CHANNEL;
67 conf.fg_alpha = opa;
68 conf.fg_cf = LV_DRAW_DMA2D_FGBG_CF_A8;
69
70 conf.bg_address = first_pixel;
71 conf.bg_offset = output_offset;
72 conf.bg_alpha_mode = LV_DRAW_DMA2D_ALPHA_MODE_NO_MODIFY_IMAGE_ALPHA_CHANNEL;
73 conf.bg_alpha = opa;
74 conf.bg_cf = (lv_draw_dma2d_fgbg_cf_t) output_cf;
75
76 /* Background alpha channel should be treated as 0xFF if the cf is XRGB */
77 if(cf == LV_COLOR_FORMAT_XRGB8888) {
78 conf.bg_alpha_mode = LV_DRAW_DMA2D_ALPHA_MODE_REPLACE_ALPHA_CHANNEL;
79 conf.bg_alpha = 0xff;
80 }
81 }
82
83 lv_draw_dma2d_configure_and_start_transfer(&conf);
84}
