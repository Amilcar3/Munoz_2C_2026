120static lv_wl_egl_display_data_t * egl_create_display_data(lv_display_t * display,
121 int32_t width, int32_t height)
122{
123 lv_wl_egl_display_data_t * ddata = lv_zalloc(sizeof(*ddata));
124 if(!ddata) {
125 LV_LOG_WARN("Failed to allocate data for display");
126 return NULL;
127 }
128
129 /* Set the backend display data immediately as we will need it
130 * in the EGL window creation callback */
131 lv_wayland_set_backend_display_data(display, ddata);
132
133 lv_opengles_egl_set_display_color_format(display);
134
135 /* Create EGL context */
136 lv_egl_interface_t egl_interface = wl_egl_get_interface(display);
137 ddata->egl_ctx = lv_opengles_egl_context_create(&egl_interface);
138 if(!ddata->egl_ctx) {
139 LV_LOG_WARN("Failed to create EGL context");
140 goto egl_ctx_err;
141 }
142
143 /* Let the opengles texture driver handle the texture lifetime */
144 ddata->texture.is_texture_owner = true;
145
146 /*Initialize the draw buffers and texture*/
147 lv_result_t res = lv_opengles_texture_reshape(&ddata->texture, display, width, height);
148 if(res != LV_RESULT_OK) {
149 LV_LOG_WARN("Failed to create draw buffers");
150 goto texture_err;
151 }
152
153 has_display = true;
154 return ddata;
155
156texture_err:
157 lv_opengles_egl_context_destroy(ddata->egl_ctx);
158egl_ctx_err:
159 lv_wayland_set_backend_display_data(display, NULL);
160 lv_free(ddata);
161 return NULL;
162}

164static void egl_destroy_display_data(lv_wl_egl_display_data_t * ddata)
165{
166 if(!ddata) {
167 return;
168 }
169
170 lv_opengles_texture_deinit(&ddata->texture);
171
172 if(ddata->egl_ctx) {
173 lv_opengles_egl_context_destroy(ddata->egl_ctx);
174 ddata->egl_ctx = NULL;
175 }
176
177 LV_LOG_INFO("Deleted EGL display data");
178 lv_free(ddata);
179 has_display = false;
180}

189static void egl_flush_cb(lv_display_t * disp, const lv_area_t * area, uint8_t * px_map)
190{
191 LV_UNUSED(area);
192 LV_UNUSED(px_map);
193
194 if(!lv_display_flush_is_last(disp)) {
195 lv_display_flush_ready(disp);
196 return;
197 }
198
199 lv_wl_egl_display_data_t * ddata = lv_wayland_get_backend_display_data(disp);
200 struct wl_surface * surface = lv_wayland_get_window_surface(disp);
201
202 if(!surface) {
203 lv_display_flush_ready(disp);
204 return;
205 }
206
207 lv_opengles_texture_render_display(&ddata->texture, disp);
208
209 /* Swap buffers through EGL */
210 lv_opengles_egl_update(ddata->egl_ctx);
211
212 /* Request frame callback for vsync */
213 struct wl_callback * callback = wl_surface_frame(surface);
214 wl_callback_add_listener(callback, &frame_listener, disp);
215 wl_surface_damage(surface, 0, 0, lv_display_get_horizontal_resolution(disp),
216 lv_display_get_vertical_resolution(disp));
217 wl_surface_commit(surface);
218}

220static void * wl_egl_init_display(void * backend_ctx, lv_display_t * display, int32_t width, int32_t height)
221{
222 LV_UNUSED(backend_ctx);
223 if(has_display) {
224 LV_LOG_INFO("The EGL backend can only handle one display at a time");
225 return NULL;
226 }
227 lv_wl_egl_display_data_t * ddata = egl_create_display_data(display, width, height);
228 if(!ddata) {
229 LV_LOG_WARN("Failed to create display data");
230 return NULL;
231 }
232
233 lv_display_set_flush_cb(display, egl_flush_cb);
234 lv_display_set_flush_wait_cb(display, flush_wait_cb);
235 lv_display_set_render_mode(display, LV_USE_DRAW_NANOVG ? LV_DISPLAY_RENDER_MODE_FULL : LV_DISPLAY_RENDER_MODE_DIRECT);
236
237 has_display = true;
238 return ddata;
239}

241static void * wl_egl_resize_display(void * backend_ctx, lv_display_t * display)
242{
243 LV_UNUSED(backend_ctx);
244 lv_wl_egl_display_data_t * ddata = lv_wayland_get_backend_display_data(display);
245
246 int32_t width = lv_display_get_horizontal_resolution(display);
247 int32_t height = lv_display_get_vertical_resolution(display);
248 lv_result_t res = lv_opengles_texture_reshape(&ddata->texture, display, width, height);
249 if(res != LV_RESULT_OK) {
250 LV_LOG_ERROR("Failed to resize display");
251 return ddata;
252 }
253
254 wl_egl_window_resize(ddata->egl_window, width, height, 0, 0);
255 return ddata;
256}
