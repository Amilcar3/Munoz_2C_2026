90static lv_obj_t * font_stress_label_create(font_stress_ctx_t * ctx, const char * font_name, lv_obj_t * par, int size,
91 int x, int y)
92{
93 static const uint16_t font_style[] = {
94 LV_FREETYPE_FONT_STYLE_NORMAL,
95 LV_FREETYPE_FONT_STYLE_ITALIC,
96 LV_FREETYPE_FONT_STYLE_BOLD,
97 };
98
99 uint32_t index = lv_rand(0, sizeof(font_style) / sizeof(uint16_t) - 1);
100 uint32_t r = lv_rand(0, 0xFF);
101 uint32_t g = lv_rand(0, 0xFF);
102 uint32_t b = lv_rand(0, 0xFF);
103 lv_opa_t opa = lv_rand(0, LV_OPA_COVER);
104 lv_color_t color = lv_color_make(r, g, b);
105
106 lv_font_t * font = ctx->font_create_cb(ctx, font_name, LV_FREETYPE_FONT_RENDER_MODE_BITMAP, size, font_style[index]);
107 if(!font) {
108 return NULL;
109 }
110
111 lv_obj_t * label = lv_label_create(par);
112 lv_obj_set_style_text_font(label, font, 0);
113 lv_obj_set_style_text_opa(label, opa, 0);
114 lv_obj_set_style_text_color(label, color, 0);
115 lv_obj_set_pos(label, x, y);
116 lv_obj_set_style_outline_color(label, color, 0);
117 lv_obj_set_style_outline_width(label, 1, 0);
118
119 uint8_t str[128];
120 lv_random_utf8_chars(str, sizeof(str), LV_RND_UNICODE_ALPHANUM_AND_CJK_TABLE, LV_RND_UNICODE_ALPHANUM_AND_CJK_TABLE_LEN,
121 MAX_LABEL_CONTENT_LEN);
122 lv_label_set_text(label, (char *)str);
123 return label;
124}

145static void font_stress_update(font_stress_ctx_t * ctx)
146{
147 uint32_t label_index = lv_rand(0, ctx->config.label_cnt - 1);
148 uint32_t font_index = lv_rand(0, ctx->config.font_cnt - 1);
149 uint32_t font_size = lv_rand(0, MAX_FONT_SIZE);
150 uint32_t label_x = lv_rand(0, LV_HOR_RES) - LV_HOR_RES / 2;
151 uint32_t label_y = lv_rand(0, LV_VER_RES);
152
153 lv_obj_t * label = ctx->label_arr[label_index];
154 if(label) {
155 font_stress_label_delete(ctx, label);
156 ctx->label_arr[label_index] = NULL;
157 }
158 else {
159 const char * pathname = ctx->config.font_name_arr[font_index];
160 TEST_ASSERT_NOT_NULL(pathname);
161 label = font_stress_label_create(
162 ctx,
163 pathname,
164 ctx->par,
165 (int)font_size,
166 (int)label_x,
167 (int)label_y);
168 ctx->label_arr[label_index] = label;
169 }
170}
