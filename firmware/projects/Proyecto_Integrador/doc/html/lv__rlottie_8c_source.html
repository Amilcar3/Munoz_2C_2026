126static void lv_rlottie_constructor(const lv_obj_class_t * class_p, lv_obj_t * obj)
127{
128 LV_UNUSED(class_p);
129 lv_rlottie_t * rlottie = (lv_rlottie_t *) obj;
130
131 if(create_info.rlottie_desc) {
132 rlottie->animation = lottie_animation_from_data(create_info.rlottie_desc, create_info.rlottie_desc, "");
133 }
134 else if(create_info.path) {
135 rlottie->animation = lottie_animation_from_file(create_info.path);
136 }
137 if(rlottie->animation == NULL) {
138 LV_LOG_WARN("The animation can't be opened");
139 return;
140 }
141
142 rlottie->total_frames = lottie_animation_get_totalframe(rlottie->animation);
143 rlottie->framerate = (size_t)lottie_animation_get_framerate(rlottie->animation);
144 rlottie->current_frame = 0;
145
146 rlottie->scanline_width = create_info.width * LV_ARGB32 / 8;
147
148 size_t allocated_buf_size = (create_info.width * create_info.height * LV_ARGB32 / 8);
149 rlottie->allocated_buf = lv_malloc(allocated_buf_size);
150 if(rlottie->allocated_buf != NULL) {
151 rlottie->allocated_buffer_size = allocated_buf_size;
152 memset(rlottie->allocated_buf, 0, allocated_buf_size);
153 }
154
155 rlottie->imgdsc.header.cf = LV_COLOR_FORMAT_ARGB8888;
156 rlottie->imgdsc.header.h = create_info.height;
157 rlottie->imgdsc.header.w = create_info.width;
158 rlottie->imgdsc.data = (void *)rlottie->allocated_buf;
159 rlottie->imgdsc.data_size = allocated_buf_size;
160
161 lv_image_set_src(obj, &rlottie->imgdsc);
162
163 rlottie->play_ctrl = LV_RLOTTIE_CTRL_FORWARD | LV_RLOTTIE_CTRL_PLAY | LV_RLOTTIE_CTRL_LOOP;
164 rlottie->dest_frame = rlottie->total_frames; /* invalid destination frame so it's possible to pause on frame 0 */
165
166 rlottie->task = lv_timer_create(next_frame_task_cb, 1000 / rlottie->framerate, obj);
167
168 lv_obj_update_layout(obj);
169}

171static void lv_rlottie_destructor(const lv_obj_class_t * class_p, lv_obj_t * obj)
172{
173 LV_UNUSED(class_p);
174 lv_rlottie_t * rlottie = (lv_rlottie_t *) obj;
175

177 lottie_animation_destroy(rlottie->animation);
178 rlottie->animation = 0;
179 rlottie->current_frame = 0;
180 rlottie->framerate = 0;
181 rlottie->scanline_width = 0;
182 rlottie->total_frames = 0;
183 }
184
185 if(rlottie->task) {
186 lv_timer_delete(rlottie->task);
187 rlottie->task = NULL;
188 rlottie->play_ctrl = LV_RLOTTIE_CTRL_FORWARD;
189 rlottie->dest_frame = 0;
190 }
191
192 lv_image_cache_drop(&rlottie->imgdsc);
193
194 if(rlottie->allocated_buf) {
195 lv_free(rlottie->allocated_buf);
196 rlottie->allocated_buf = NULL;
197 rlottie->allocated_buffer_size = 0;
198 }

202static void next_frame_task_cb(lv_timer_t * t)
203{
204 lv_obj_t * obj = lv_timer_get_user_data(t);
205 lv_rlottie_t * rlottie = (lv_rlottie_t *) obj;
206
207 if((rlottie->play_ctrl & LV_RLOTTIE_CTRL_PAUSE) == LV_RLOTTIE_CTRL_PAUSE) {
208 if(rlottie->current_frame == rlottie->dest_frame) {
209 /* Pause the timer too when it has run once to avoid CPU consumption */
210 lv_timer_pause(t);
211 return;
212 }
213 rlottie->dest_frame = rlottie->current_frame;
214 }
215 else {
216 if((rlottie->play_ctrl & LV_RLOTTIE_CTRL_BACKWARD) == LV_RLOTTIE_CTRL_BACKWARD) {
217 if(rlottie->current_frame > 0)
218 --rlottie->current_frame;
219 else { /* Looping ? */
220 if((rlottie->play_ctrl & LV_RLOTTIE_CTRL_LOOP) == LV_RLOTTIE_CTRL_LOOP)
221 rlottie->current_frame = rlottie->total_frames - 1;
222 else {
223 lv_obj_send_event(obj, LV_EVENT_READY, NULL);
224 lv_timer_pause(t);
225 return;
226 }
227 }
228 }
229 else {
230 if(rlottie->current_frame < rlottie->total_frames)
231 ++rlottie->current_frame;
232 else { /* Looping ? */
233 if((rlottie->play_ctrl & LV_RLOTTIE_CTRL_LOOP) == LV_RLOTTIE_CTRL_LOOP)
234 rlottie->current_frame = 0;
235 else {
236 lv_obj_send_event(obj, LV_EVENT_READY, NULL);
237 lv_timer_pause(t);
238 return;
239 }
240 }
241 }
242 }
243
244 lottie_animation_render(
245 rlottie->animation,
246 rlottie->current_frame,
247 rlottie->allocated_buf,
248 rlottie->imgdsc.header.w,
249 rlottie->imgdsc.header.h,
250 rlottie->scanline_width
251 );
252
253 lv_obj_invalidate(obj);
254}
