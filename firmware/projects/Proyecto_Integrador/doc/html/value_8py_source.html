16 def normalize(val: "ValueInput", target_type: Optional[str] = None) -> "Value":
17 """Normalize input to a typed Value pointer.
18
19 Args:
20 val: Input value - str (GDB expression), int (address),
21 gdb.Value, or Value instance
22 target_type: C type name (e.g. "lv_obj_t"). If provided,
23 result is cast to target_type*
24
25 Returns:
26 Value instance, optionally cast to target_type*
27
28 Raises:
29 ValueError: If target_type lookup fails or cast fails
30 CorruptedError: If val is a CorruptedValue sentinel
31 """
32 if isinstance(val, CorruptedValue):
33 raise CorruptedError(val._addr, val._error)
34
35 if isinstance(val, str):
36 val = gdb.parse_and_eval(val)
37
38 if isinstance(val, int):
39 val = gdb.Value(val)
40
41 if not isinstance(val, Value):
42 val = Value(val)
43
44 # Auto-infer target_type from pointer type when not specified
45 if target_type is None:
46 typ = val.type.strip_typedefs()
47 if typ.code == gdb.TYPE_CODE_PTR:
48 target = typ.target().strip_typedefs()
49 if target.code in (gdb.TYPE_CODE_STRUCT, gdb.TYPE_CODE_UNION):
50 # Prefer .name or .tag over str() to avoid polluted
51 # anonymous struct expansions like "const struct {...}".
52 target_type = target.name or target.tag
53
54 if target_type is not None:
55 # If val is already a pointer whose target matches target_type,
56 # keep the original type to avoid gdb.lookup_type() which may
57 # resolve to a different compilation unit's definition when
58 # macro-controlled fields cause multiple struct definitions.
59 typ = val.type.strip_typedefs()
60 if typ.code == gdb.TYPE_CODE_PTR:
61 target = typ.target()
62 tname = target.name or target.tag or ""
63 if tname == target_type:
64 return val
65
66 try:
67 gdb.lookup_type(target_type)
68 except gdb.error:
69 raise ValueError(f"Type not found: {target_type}")
70
71 if typ.code != gdb.TYPE_CODE_PTR:
72 if typ.code in (gdb.TYPE_CODE_INT, gdb.TYPE_CODE_ENUM):
73 val = Value(val.cast(gdb.lookup_type(target_type).pointer()))
74 else:
75 val = Value(val.address)
76
77 result = val.cast(target_type, ptr=True)
78 if result is None:
79 raise ValueError(f"Failed to cast to {target_type}*")
80 val = result
81
