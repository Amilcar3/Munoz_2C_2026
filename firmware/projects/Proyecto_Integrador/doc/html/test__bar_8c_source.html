445static lv_obj_t * styled_bar_create(bool ver, int32_t start_value, int32_t end_value, lv_grad_dir_t grad_dir,
446 int32_t bg_radius, int32_t indic_radius, int32_t bg_pad)
447{
448 lv_obj_t * bar = lv_bar_create(lv_screen_active());

452 lv_bar_set_mode(bar, LV_BAR_MODE_RANGE);
453 lv_bar_set_value(bar, end_value, LV_ANIM_OFF);
454 lv_bar_set_start_value(bar, start_value, LV_ANIM_OFF);
455
456 lv_obj_set_style_bg_opa(bar, 255, LV_PART_MAIN);
457 lv_obj_set_style_bg_color(bar, lv_color_hex3(0x0ff), LV_PART_MAIN);
458 lv_obj_set_style_radius(bar, bg_radius, LV_PART_MAIN);
459 lv_obj_set_style_pad_all(bar, bg_pad, LV_PART_MAIN);
460
461 lv_obj_set_style_bg_opa(bar, LV_OPA_70, LV_PART_INDICATOR);
462 lv_obj_set_style_bg_color(bar, lv_color_hex3(0xf0f), LV_PART_INDICATOR);
463 lv_obj_set_style_bg_grad_color(bar, lv_color_hex3(0x8f8), LV_PART_INDICATOR);
464 lv_obj_set_style_bg_grad_dir(bar, grad_dir, LV_PART_INDICATOR);
465
466 lv_obj_set_style_border_width(bar, 2, LV_PART_INDICATOR);
467 lv_obj_set_style_border_color(bar, lv_color_hex3(0x0f0), LV_PART_INDICATOR);
468 lv_obj_set_style_border_opa(bar, LV_OPA_70, LV_PART_INDICATOR);
469 lv_obj_set_style_outline_width(bar, 2, LV_PART_INDICATOR);
470 lv_obj_set_style_outline_pad(bar, 4, LV_PART_INDICATOR);
471 lv_obj_set_style_outline_color(bar, lv_color_hex3(0xff0), LV_PART_INDICATOR);
472 lv_obj_set_style_outline_opa(bar, LV_OPA_70, LV_PART_INDICATOR);
473 lv_obj_set_style_shadow_width(bar, 20, LV_PART_INDICATOR);
474 lv_obj_set_style_shadow_spread(bar, 5, LV_PART_INDICATOR);
