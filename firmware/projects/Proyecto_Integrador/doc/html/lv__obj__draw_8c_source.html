49void lv_obj_init_draw_rect_dsc(lv_obj_t * obj, lv_part_t part, lv_draw_rect_dsc_t * draw_dsc)
50{
51 LV_CHECK_OBJ(obj, MY_CLASS, return);
52 LV_CHECK_ARG(draw_dsc != NULL, return);
53
54 LV_PROFILER_DRAW_BEGIN;
55 draw_dsc->base.obj = obj;
56 draw_dsc->base.part = part;
57
58 lv_opa_t opa = get_layer_opa(obj, part, &draw_dsc->base);
59 if(part != LV_PART_MAIN) {
60 if(opa <= LV_OPA_MIN) {
61 draw_dsc->bg_opa = LV_OPA_TRANSP;
62 draw_dsc->bg_image_opa = LV_OPA_TRANSP;
63 draw_dsc->border_opa = LV_OPA_TRANSP;
64 draw_dsc->outline_opa = LV_OPA_TRANSP;
65 draw_dsc->shadow_opa = LV_OPA_TRANSP;
66 LV_PROFILER_DRAW_END;
67 return;
68 }
69 }
70
71 draw_dsc->radius = lv_obj_get_style_radius_internal(obj, part);
72
73 if(draw_dsc->bg_opa != LV_OPA_TRANSP) {
74 draw_dsc->bg_opa = lv_obj_get_style_bg_opa_internal(obj, part);
75 if(draw_dsc->bg_opa > LV_OPA_MIN) {
76 lv_color_t bg_color = lv_obj_get_style_bg_color_filtered_internal(obj, part);
77 draw_dsc->bg_color = normal_apply_layer_recolor(obj, part, &draw_dsc->base, bg_color);
78 const lv_grad_dsc_t * grad = lv_obj_get_style_bg_grad_internal(obj, part);
79 if(grad && grad->dir != LV_GRAD_DIR_NONE) {
80 lv_memcpy(&draw_dsc->bg_grad, grad, sizeof(*grad));
81 }
82 else {
83 draw_dsc->bg_grad.dir = lv_obj_get_style_bg_grad_dir_internal(obj, part);
84 if(draw_dsc->bg_grad.dir != LV_GRAD_DIR_NONE) {
85 draw_dsc->bg_grad.stops[0].color = draw_dsc->bg_color;
86 lv_color_t bg_grad_color = lv_obj_get_style_bg_grad_color_filtered_internal(obj, part);
87 draw_dsc->bg_grad.stops[1].color = normal_apply_layer_recolor(obj, part, &draw_dsc->base, bg_grad_color);
88 draw_dsc->bg_grad.stops[0].frac = lv_obj_get_style_bg_main_stop_internal(obj, part);
89 draw_dsc->bg_grad.stops[1].frac = lv_obj_get_style_bg_grad_stop_internal(obj, part);
90 draw_dsc->bg_grad.stops[0].opa = lv_obj_get_style_bg_main_opa_internal(obj, part);
91 draw_dsc->bg_grad.stops[1].opa = lv_obj_get_style_bg_grad_opa_internal(obj, part);
92 }
93 }
94 }
95 }
96
97 if(draw_dsc->border_opa != LV_OPA_TRANSP) {
98 draw_dsc->border_width = lv_obj_get_style_border_width_internal(obj, part);
99 if(draw_dsc->border_width) {
100 draw_dsc->border_opa = lv_obj_get_style_border_opa_internal(obj, part);
101 if(draw_dsc->border_opa > LV_OPA_MIN) {
102 draw_dsc->border_side = lv_obj_get_style_border_side_internal(obj, part);
103 lv_color_t border_color = lv_obj_get_style_border_color_filtered_internal(obj, part);
104 draw_dsc->border_color = normal_apply_layer_recolor(obj, part, &draw_dsc->base, border_color);
105 }
106 }
107 }
108
109 if(draw_dsc->outline_opa != LV_OPA_TRANSP) {
110 draw_dsc->outline_width = lv_obj_get_style_outline_width_internal(obj, part);
111 if(draw_dsc->outline_width) {
112 draw_dsc->outline_opa = lv_obj_get_style_outline_opa_internal(obj, part);
113 if(draw_dsc->outline_opa > LV_OPA_MIN) {
114 draw_dsc->outline_pad = lv_obj_get_style_outline_pad_internal(obj, part);
115 lv_color_t outline_color = lv_obj_get_style_outline_color_filtered_internal(obj, part);
116 draw_dsc->outline_color = normal_apply_layer_recolor(obj, part, &draw_dsc->base, outline_color);
117 }
118 }
119 }
120
121 if(draw_dsc->bg_image_opa != LV_OPA_TRANSP) {
122 draw_dsc->bg_image_src = lv_obj_get_style_bg_image_src_internal(obj, part);
123 if(draw_dsc->bg_image_src) {
124 draw_dsc->bg_image_opa = lv_obj_get_style_bg_image_opa_internal(obj, part);
125 if(draw_dsc->bg_image_opa > LV_OPA_MIN) {
126 if(lv_image_src_get_type(draw_dsc->bg_image_src) == LV_IMAGE_SRC_SYMBOL) {
127 draw_dsc->bg_image_symbol_font = lv_obj_get_style_text_font_internal(obj, part);
128 lv_color_t text_color = lv_obj_get_style_text_color_filtered_internal(obj, part);
129 draw_dsc->bg_image_recolor = normal_apply_layer_recolor(obj, part, &draw_dsc->base, text_color);
130 }
131 else {
132 lv_color_t bg_image_recolor = lv_obj_get_style_bg_image_recolor_filtered_internal(obj, part);
133 lv_opa_t bg_image_recolor_opa = lv_obj_get_style_bg_image_recolor_opa_internal(obj, part);
134 lv_color32_t result = image_apply_layer_recolor(obj, part, &draw_dsc->base, bg_image_recolor, bg_image_recolor_opa);
135 draw_dsc->bg_image_recolor_opa = result.alpha;
136 draw_dsc->bg_image_recolor = lv_color_make(result.red, result.green, result.blue);
137 draw_dsc->bg_image_tiled = lv_obj_get_style_bg_image_tiled_internal(obj, part);
138 }
139
140 draw_dsc->bg_image_colorkey = lv_obj_get_style_image_colorkey_internal(obj, part);
141 }
142 }
143 }
144
145 if(draw_dsc->shadow_opa) {
146 draw_dsc->shadow_width = lv_obj_get_style_shadow_width_internal(obj, part);
147 if(draw_dsc->shadow_width) {
148 if(draw_dsc->shadow_opa > LV_OPA_MIN) {
149 draw_dsc->shadow_opa = lv_obj_get_style_shadow_opa_internal(obj, part);
150 if(draw_dsc->shadow_opa > LV_OPA_MIN) {
151 draw_dsc->shadow_offset_x = lv_obj_get_style_shadow_offset_x_internal(obj, part);
152 draw_dsc->shadow_offset_y = lv_obj_get_style_shadow_offset_y_internal(obj, part);
153 draw_dsc->shadow_spread = lv_obj_get_style_shadow_spread_internal(obj, part);
154 lv_color_t shadow_color = lv_obj_get_style_shadow_color_filtered_internal(obj, part);
155 draw_dsc->shadow_color = normal_apply_layer_recolor(obj, part, &draw_dsc->base, shadow_color);
156 }
157 }
158 }
159 }
160
161 if(opa < LV_OPA_MAX) {
162 draw_dsc->bg_opa = LV_OPA_MIX2(draw_dsc->bg_opa, opa);
163 draw_dsc->bg_image_opa = LV_OPA_MIX2(draw_dsc->bg_image_opa, opa);
164 draw_dsc->border_opa = LV_OPA_MIX2(draw_dsc->border_opa, opa);
165 draw_dsc->shadow_opa = LV_OPA_MIX2(draw_dsc->shadow_opa, opa);
166 draw_dsc->outline_opa = LV_OPA_MIX2(draw_dsc->outline_opa, opa);
167 }
168
169 drop_shadow_init(obj, part, &draw_dsc->base);
170
171 LV_PROFILER_DRAW_END;

378int32_t lv_obj_calculate_ext_draw_size(lv_obj_t * obj, lv_part_t part)
379{
380 LV_CHECK_OBJ(obj, MY_CLASS, return 0);
381
382 LV_PROFILER_DRAW_BEGIN;
383 int32_t s = 0;
384
385 int32_t sh_width = lv_obj_get_style_shadow_width_internal(obj, part);
386 if(sh_width) {
387 lv_opa_t sh_opa = lv_obj_get_style_shadow_opa_internal(obj, part);
388 if(sh_opa > LV_OPA_MIN) {
389 sh_width = sh_width / 2 + 1; /*The blur adds only half width*/
390 sh_width += lv_obj_get_style_shadow_spread_internal(obj, part);
391 int32_t sh_ofs_x = lv_obj_get_style_shadow_offset_x_internal(obj, part);
392 int32_t sh_ofs_y = lv_obj_get_style_shadow_offset_y_internal(obj, part);
393 sh_width += LV_MAX(LV_ABS(sh_ofs_x), LV_ABS(sh_ofs_y));
394 s = LV_MAX(s, sh_width);
395 }
396 }
397
398 int32_t outline_width = lv_obj_get_style_outline_width_internal(obj, part);
399 if(outline_width) {
400 lv_opa_t outline_opa = lv_obj_get_style_outline_opa_internal(obj, part);
401 if(outline_opa > LV_OPA_MIN) {
402 int32_t outline_pad = lv_obj_get_style_outline_pad_internal(obj, part);
403 s = LV_MAX(s, outline_pad + outline_width);
404 }
405 }
406
407 int32_t drop_shadow_size = 0;
408 if(lv_obj_get_style_drop_shadow_opa_internal(obj, part) > 0) {
409 drop_shadow_size += LV_MAX(LV_ABS(lv_obj_get_style_drop_shadow_offset_x_internal(obj, part)),
410 LV_ABS(lv_obj_get_style_drop_shadow_offset_y_internal(obj, part)));
411
412 drop_shadow_size += lv_obj_get_style_drop_shadow_radius_internal(obj, part) + 1;
413 }
414 s += drop_shadow_size;
415
416 int32_t w = lv_obj_get_style_transform_width_internal(obj, part);
417 int32_t h = lv_obj_get_style_transform_height_internal(obj, part);
418 int32_t wh = LV_MAX(w, h);
419 if(wh > 0) s += wh;
420
421 LV_PROFILER_DRAW_END;
422 return s;
423}

425void lv_obj_refresh_ext_draw_size(lv_obj_t * obj)
426{
427 LV_PROFILER_DRAW_BEGIN;
428 LV_CHECK_OBJ(obj, MY_CLASS, return);
429
430 int32_t s_old = lv_obj_get_ext_draw_size(obj);
431 int32_t s_new = 0;
432 lv_obj_send_event(obj, LV_EVENT_REFR_EXT_DRAW_SIZE, &s_new);
433
434 /*Store the result if the special attrs already allocated*/
435 if(obj->spec_attr) {
436 obj->spec_attr->ext_draw_size = s_new;
437 }
438 /*Allocate spec. attrs. only if the result is not zero.
439 *Zero is the default value if the spec. attr. are not defined.*/
440 else if(s_new != 0) {
441 if(!lv_obj_allocate_spec_attr(obj)) {
442 LV_PROFILER_DRAW_END;
443 return;
444 }
445 obj->spec_attr->ext_draw_size = s_new;
446 }
447
448 if(s_new != s_old) lv_obj_invalidate(obj);
449 LV_PROFILER_DRAW_END;
450}

472static inline lv_opa_t get_layer_opa(const lv_obj_t * obj, lv_part_t part, const lv_draw_dsc_base_t * base_dsc)
473{
474 if(base_dsc->layer) {
475 /* Accessing the layer opa directly is faster than using get style opa recursive */
476 if(part == LV_PART_MAIN) {
477 return base_dsc->layer->opa;
478 }
479
480 return LV_OPA_MIX2(base_dsc->layer->opa, lv_obj_get_style_opa_internal(obj, part));
481 }
482
483 /* fallback to old recursive style opa */
484 return lv_obj_get_style_opa_recursive(obj, part);
485}

488static lv_color_t normal_apply_layer_recolor(const lv_obj_t * obj, lv_part_t part, const lv_draw_dsc_base_t * base_dsc,
489 lv_color_t color)
490{
491 lv_color32_t recolor;
492
493 if(base_dsc->layer) {
494 recolor = base_dsc->layer->recolor;
495 if(part != LV_PART_MAIN) {
496 recolor = lv_obj_style_apply_recolor(obj, part, recolor);
497 }
498 }
499 else {
500 recolor = lv_obj_get_style_recolor_recursive(obj, part);
501 }
502
503 return lv_color_mix(lv_color_make(recolor.red, recolor.green, recolor.blue), color, recolor.alpha);
504}
