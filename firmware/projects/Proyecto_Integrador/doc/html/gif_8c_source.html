66 if (res != LV_FS_RES_OK) {

70 }

73 uint32_t pos;
74 lv_fs_tell(&pGIF->GIFFile.fHandle, &pos);
75 pGIF->GIFFile.iSize = pos;
76 lv_fs_seek(&pGIF->GIFFile.fHandle, 0, LV_FS_SEEK_SET);
77
78 int ret = GIFInit(pGIF);
79 if (!ret) {
80 LV_LOG_WARN("Failed to initialize GIF from file: %s, err: %d", szFilename, pGIF->iError);
81 lv_fs_close(&pGIF->GIFFile.fHandle);
82 }
83
84 return ret;
85} /* GIF_openFile() */

104//
105// Return value:
106// 1 = good decode, more frames exist
107// 0 = good decode, no more frames
108// -1 = error
109//
110int GIF_playFrame(GIFIMAGE *pGIF, int *delayMilliseconds, void *pUser)
111{
112int rc;
113
114 if (delayMilliseconds)
115 *delayMilliseconds = 0; // clear any old valid
116 if (pGIF->GIFFile.iPos >= pGIF->GIFFile.iSize-1) // no more data exists
117 {
118 (*pGIF->pfnSeek)(&pGIF->GIFFile, 0); // seek to start
119 }
120 if (GIFParseInfo(pGIF, 0))
121 {
122 pGIF->pUser = pUser;
123 if (pGIF->iError == GIF_EMPTY_FRAME) // don't try to decode it
124 return 0;
125 if (pGIF->pTurboBuffer) { // the presence of the Turbo buffer indicates Turbo mode
126 rc = DecodeLZWTurbo(pGIF, 0);
127 } else {
128 rc = DecodeLZW(pGIF, 0);
129 }
130 if (rc != 0) // problem
131 return 0;
132 }
133 else
134 {
135 return 0; // error parsing the frame info, we may be at the end of the file
136 }
137 // Return 1 for more frames or 0 if this was the last frame
138 if (delayMilliseconds) // if not NULL, return the frame delay time
139 *delayMilliseconds = pGIF->iFrameDelay;
140 return (pGIF->GIFFile.iPos < pGIF->GIFFile.iSize-1);
141} /* GIF_playFrame() */

251 return 0;
252 }
253 return 1;

256//
257// Parse the GIF header, gather the size and palette info
258// If called with bInfoOnly set to true, it will test for a valid file
259// and return the canvas size only
260// Returns 1 for success, 0 for failure
261//
262static int GIFParseInfo(GIFIMAGE *pPage, int bInfoOnly)
263{
264 int i, j, iColorTableBits;
265 int iBytesRead;
266 unsigned char c, *p;
267 int32_t iOffset = 0;
268 int32_t iStartPos = pPage->GIFFile.iPos; // starting file position
269 int iReadSize;
270
271 pPage->bUseLocalPalette = 0; // assume no local palette
272 pPage->bEndOfFrame = 0; // we're just getting started
273 pPage->iFrameDelay = 0; // may not have a gfx extension block
274 pPage->iRepeatCount = -1; // assume NETSCAPE loop count is not specified
275 iReadSize = MAX_CHUNK_SIZE;
276 // If you try to read past the EOF, the SD lib will return garbage data
277 if (iStartPos + iReadSize > pPage->GIFFile.iSize)
278 iReadSize = (pPage->GIFFile.iSize - iStartPos - 1);
279 p = pPage->ucFileBuf;
280 iBytesRead = (*pPage->pfnRead)(&pPage->GIFFile, pPage->ucFileBuf, iReadSize); // 255 is plenty for now
281
282 if (iBytesRead != iReadSize) // we're at the end of the file
283 {
284 pPage->iError = GIF_EARLY_EOF;
285 return 0;
286 }
287 if (iStartPos == 0) // start of the file
288 { // canvas size
289 if (lv_memcmp(p, "GIF89", 5) != 0 && lv_memcmp(p, "GIF87", 5) != 0) // not a GIF file
290 {
291 pPage->iError = GIF_BAD_FILE;
292 return 0;
293 }
294 pPage->iCanvasWidth = pPage->iWidth = INTELSHORT(&p[6]);
295 pPage->iCanvasHeight = pPage->iHeight = INTELSHORT(&p[8]);
296 pPage->iBpp = ((p[10] & 0x70) >> 4) + 1;
297 iColorTableBits = (p[10] & 7) + 1; // Log2(size) of the color table
298 pPage->ucBackground = p[11]; // background color
299 pPage->ucGIFBits = 0;
300 iOffset = 13;
301 if (p[10] & 0x80) // global color table?
302 { // by default, convert to byte-reversed RGB565 for immediate use
303 // Read enough additional data for the color table
304 iBytesRead += (*pPage->pfnRead)(&pPage->GIFFile, &pPage->ucFileBuf[iBytesRead], 3*(1<<iColorTableBits));
305 if (pPage->ucPaletteType == GIF_PALETTE_RGB565_LE || pPage->ucPaletteType == GIF_PALETTE_RGB565_BE) {
306 for (i=0; i<(1<<iColorTableBits); i++) {
307 uint16_t usRGB565;
308 usRGB565 = ((p[iOffset] >> 3) << 11); // R
309 usRGB565 |= ((p[iOffset+1] >> 2) << 5); // G
310 usRGB565 |= (p[iOffset+2] >> 3); // B
311 if (pPage->ucPaletteType == GIF_PALETTE_RGB565_LE)
312 pPage->pPalette[i] = usRGB565;
313 else
314 pPage->pPalette[i] = (usRGB565 << 8) | (usRGB565 >> 8); // SPI wants MSB first
315 iOffset += 3;
316 }
317 } else if (pPage->ucPaletteType == GIF_PALETTE_1BPP || pPage->ucPaletteType == GIF_PALETTE_1BPP_OLED) {
318 uint8_t *pPal1 = (uint8_t*)pPage->pPalette;
319 for (i=0; i<(1<<iColorTableBits); i++) {
320 uint16_t usGray;
321 usGray = p[iOffset]; // R
322 usGray += p[iOffset+1]*2; // G is twice as important
323 usGray += p[iOffset+2]; // B
324 pPal1[i] = (usGray >= 512); // bright enough = 1
325 iOffset += 3;
326 }
327 } else { // just copy it as-is (RGB888 & RGB8888 output)
328 lv_memcpy(pPage->pPalette, &p[iOffset], (1<<iColorTableBits) * 3);
329 iOffset += (1 << iColorTableBits) * 3;
330 }
331 }
332 }
333 while (p[iOffset] != ',' && p[iOffset] != ';') /* Wait for image separator */
334 {
335 if (p[iOffset] == '!') /* Extension block */
336 {
337 iOffset++;
338 switch(p[iOffset++]) /* Block type */
339 {
340 case 0xf9: /* Graphic extension */
341 if (p[iOffset] == 4) // correct length
342 {
343 pPage->ucGIFBits = p[iOffset+1]; // packed fields
344 pPage->iFrameDelay = (INTELSHORT(&p[iOffset+2]))*10; // delay in ms
345 if (pPage->iFrameDelay <= 1) // 0-1 is going to make it run at 60fps; use 100 (10fps) as a reasonable substitute
346 pPage->iFrameDelay = 100;
347 if (pPage->ucGIFBits & 1) // transparent color is used
348 pPage->ucTransparent = p[iOffset+4]; // transparent color index
349 iOffset += 6;
350 }
351 // else // error
352 break;
353 case 0xff: /* App extension */
354 c = 1;
355 while (c) /* Skip all data sub-blocks */
356 {
357 c = p[iOffset++]; /* Block length */
358 if ((iBytesRead - iOffset) < (c+32)) // need to read more data first
359 {
360 lv_memmove(pPage->ucFileBuf, &pPage->ucFileBuf[iOffset], (iBytesRead-iOffset)); // move existing data down
361 iBytesRead -= iOffset;
362 iStartPos += iOffset;
363 iOffset = 0;
364 iBytesRead += (*pPage->pfnRead)(&pPage->GIFFile, &pPage->ucFileBuf[iBytesRead], c+32);
365 }
366 if (c == 11) // fixed block length
367 { // Netscape app block contains the repeat count
368 if (lv_memcmp(&p[iOffset], "NETSCAPE2.0", 11) == 0)
369 {
370 if (p[iOffset+11] == 3 && p[iOffset+12] == 1) // loop count
371 pPage->iRepeatCount = INTELSHORT(&p[iOffset+13]);
372 }
373 }
374 iOffset += (int)c; /* Skip to next sub-block */
375 }
376 break;
377 case 0x01: /* Text extension */
378 c = 1;
379 j = 0;
380 while (c) /* Skip all data sub-blocks */
381 {
382 c = p[iOffset++]; /* Block length */
383 if (j == 0) // use only first block
384 {
385 j = c;
386 if (j > 127) // max comment length = 127
387 j = 127;
388 // memcpy(pPage->szInfo1, &p[iOffset], j);
389 // pPage->szInfo1[j] = '\0';
390 j = 1;
391 }
392 iOffset += (int)c; /* Skip this sub-block */
393 }
394 break;
395 case 0xfe: /* Comment */
396 c = 1;
397 while (c) /* Skip all data sub-blocks */
398 {
399 c = p[iOffset++]; /* Block length */
400 if ((iBytesRead - iOffset) < (c+32)) // need to read more data first
401 {
402 lv_memmove(pPage->ucFileBuf, &pPage->ucFileBuf[iOffset], (iBytesRead-iOffset)); // move existing data down
403 iBytesRead -= iOffset;
404 iStartPos += iOffset;
405 iOffset = 0;
406 iBytesRead += (*pPage->pfnRead)(&pPage->GIFFile, &pPage->ucFileBuf[iBytesRead], c+32);
407 }
408 if (pPage->iCommentPos == 0) // Save first block info
409 {
410 pPage->iCommentPos = iStartPos + iOffset;
411 pPage->sCommentLen = c;
412 }
413 iOffset += (int)c; /* Skip this sub-block */
414 }
415 break;
416 default:
417 /* Bad header info */
418 pPage->iError = GIF_DECODE_ERROR;
419 return 0;
420 } /* switch */
421 }
422 else // invalid byte, stop decoding
423 {
424 if (pPage->GIFFile.iSize - iStartPos < 32) // non-image bytes at end of file?
425 pPage->iError = GIF_EMPTY_FRAME;
426 else
427 /* Bad header info */
428 pPage->iError = GIF_DECODE_ERROR;
429 return 0;
430 }
431 } /* while */
432 if (bInfoOnly)
433 return 1; // we've got the info we needed, leave
434 if (p[iOffset] == ';') { // end of file, quit and return a correct error code
435 pPage->iError = GIF_EMPTY_FRAME;
436 return 1;
437 }
438
439 if (p[iOffset] == ',')
440 iOffset++;
441 // This particular frame's size and position on the main frame (if animated)
442 pPage->iX = INTELSHORT(&p[iOffset]);
443 pPage->iY = INTELSHORT(&p[iOffset+2]);
444 pPage->iWidth = INTELSHORT(&p[iOffset+4]);
445 pPage->iHeight = INTELSHORT(&p[iOffset+6]);
446 if (pPage->iWidth > pPage->iCanvasWidth || pPage->iHeight > pPage->iCanvasHeight ||
447 pPage->iWidth + pPage->iX > pPage->iCanvasWidth || pPage->iHeight + pPage->iY > pPage->iCanvasHeight) {
448 pPage->iError = GIF_DECODE_ERROR; // must be a corrupt file to encounter this error here
449 return 0;
450 }
451 iOffset += 8;
452
453 /* Image descriptor
454 7 6 5 4 3 2 1 0 M=0 - use global color map, ignore pixel
455 M I 0 0 0 pixel M=1 - local color map follows, use pixel
456 I=0 - Image in sequential order
457 I=1 - Image in interlaced order
458 pixel+1 = # bits per pixel for this image
459 */
460 pPage->ucMap = p[iOffset++];
461 if (pPage->ucMap & 0x80) // local color table?
462 {// by default, convert to byte-reversed RGB565 for immediate use
463 j = (1<<((pPage->ucMap & 7)+1));
464 // Read enough additional data for the color table
465 iBytesRead += (*pPage->pfnRead)(&pPage->GIFFile, &pPage->ucFileBuf[iBytesRead], j*3);
466 if (pPage->ucPaletteType == GIF_PALETTE_RGB565_LE || pPage->ucPaletteType == GIF_PALETTE_RGB565_BE)
467 {
468 for (i=0; i<j; i++)
469 {
470 uint16_t usRGB565;
471 usRGB565 = ((p[iOffset] >> 3) << 11); // R
472 usRGB565 |= ((p[iOffset+1] >> 2) << 5); // G
473 usRGB565 |= (p[iOffset+2] >> 3); // B
474 if (pPage->ucPaletteType == GIF_PALETTE_RGB565_LE)
475 pPage->pLocalPalette[i] = usRGB565;
476 else
477 pPage->pLocalPalette[i] = (usRGB565 << 8) | (usRGB565 >> 8); // SPI wants MSB first
478 iOffset += 3;
479 }
480 } else if (pPage->ucPaletteType == GIF_PALETTE_1BPP || pPage->ucPaletteType == GIF_PALETTE_1BPP_OLED) {
481 uint8_t *pPal1 = (uint8_t*)pPage->pLocalPalette;
482 for (i=0; i<j; i++) {
483 uint16_t usGray;
484 usGray = p[iOffset]; // R
485 usGray += p[iOffset+1]*2; // G is twice as important
486 usGray += p[iOffset+2]; // B
487 pPal1[i] = (usGray >= 512); // bright enough = 1
488 iOffset += 3;
489 }
490 } else { // just copy it as-is
491 lv_memcpy(pPage->pLocalPalette, &p[iOffset], j * 3);
492 iOffset += j*3;
493 }
494 pPage->bUseLocalPalette = 1;
495 }
496 pPage->ucCodeStart = p[iOffset++]; /* initial code size */
497 /* Since GIF can be 1-8 bpp, we only allow 1,4,8 */
498 pPage->iBpp = cGIFBits[pPage->ucCodeStart];
499 // we are re-using the same buffer turning GIF file data
500 // into "pure" LZW
501 pPage->iLZWSize = 0; // we're starting with no LZW data yet
502 c = 1; // get chunk length
503 while (c && iOffset < iBytesRead)
504 {
505// Serial.printf("iOffset=%d, iBytesRead=%d\n", iOffset, iBytesRead);
506 c = p[iOffset++]; // get chunk length
507// Serial.printf("Chunk size = %d\n", c);
508 if (c <= (iBytesRead - iOffset))
509 {
510 lv_memcpy(&pPage->ucLZW[pPage->iLZWSize], &p[iOffset], c);
511 pPage->iLZWSize += c;
512 iOffset += c;
513 }
514 else // partial chunk in our buffer
515 {
516 int iPartialLen = (iBytesRead - iOffset);
517 lv_memcpy(&pPage->ucLZW[pPage->iLZWSize], &p[iOffset], iPartialLen);
518 pPage->iLZWSize += iPartialLen;
519 iOffset += iPartialLen;
520 (*pPage->pfnRead)(&pPage->GIFFile, &pPage->ucLZW[pPage->iLZWSize], c - iPartialLen);
521 pPage->iLZWSize += (c - iPartialLen);
522 }
523 if (c == 0)
524 pPage->bEndOfFrame = 1; // signal not to read beyond the end of the frame
525 }
526// seeking on an SD card is VERY VERY SLOW, so use the data we've already read by de-chunking it
527// in this case, there's too much data, so we have to seek backwards a bit
528 if (iOffset < iBytesRead)
529 {
530// Serial.printf("Need to seek back %d bytes\n", iBytesRead - iOffset);
531 (*pPage->pfnSeek)(&pPage->GIFFile, iStartPos + iOffset); // position file to new spot
532 }
533 return 1; // we are now at the start of the chunk data
534} /* GIFParseInfo() */
535//
536// Gather info about an animated GIF file
537//
538int GIF_getInfo(GIFIMAGE *pPage, GIFINFO *pInfo)
539{
540 int iOff, iNumFrames;
541 int iDelay, iMaxDelay, iMinDelay, iTotalDelay;
542 int iReadAmount;
543 int iDataAvailable = 0;
544 int iDataRemaining = 0;
545 // uint32_t lFileOff = 0;
546 int bDone = 0;
547 int bExt;
548 uint8_t c, *cBuf;
549
550 iMaxDelay = iTotalDelay = 0;
551 iMinDelay = 10000;
552 iNumFrames = 1;
553 iDataRemaining = pPage->GIFFile.iSize;
554 cBuf = (uint8_t *) pPage->ucFileBuf;
555 (*pPage->pfnSeek)(&pPage->GIFFile, 0);
556 iDataAvailable = (*pPage->pfnRead)(&pPage->GIFFile, cBuf, FILE_BUF_SIZE);
557 iDataRemaining -= iDataAvailable;
558 // lFileOff += iDataAvailable;
559 iOff = 10;
560 c = cBuf[iOff]; // get info bits
561 iOff += 3; /* Skip flags, background color & aspect ratio */
562 if (c & 0x80) /* Deal with global color table */
563 {
564 c &= 7; /* Get the number of colors defined */
565 iOff += (2<<c)*3; /* skip color table */
566 }
567 while (!bDone) // && iNumFrames < MAX_FRAMES)
568 {
569 bExt = 1; /* skip extension blocks */
570 while (bExt && iOff < iDataAvailable)
571 {
572 if ((iDataAvailable - iOff) < 258) // need to read more data first
573 {
574 lv_memmove(cBuf, &cBuf[iOff], (iDataAvailable-iOff)); // move existing data down
575 iDataAvailable -= iOff;
576 iOff = 0;
577 iReadAmount = (*pPage->pfnRead)(&pPage->GIFFile, &cBuf[iDataAvailable], FILE_BUF_SIZE-iDataAvailable);
578 iDataAvailable += iReadAmount;
579 iDataRemaining -= iReadAmount;
580 // lFileOff += iReadAmount;
581 }
582 switch(cBuf[iOff])
583 {
584 case 0x3b: /* End of file */
585 /* we were fooled into thinking there were more pages */
586 iNumFrames--;
587 goto gifpagesz;
588 // F9 = Graphic Control Extension (fixed length of 4 bytes)
589 // FE = Comment Extension
590 // FF = Application Extension
591 // 01 = Plain Text Extension
592 case 0x21: /* Extension block */
593 if (cBuf[iOff+1] == 0xf9 && cBuf[iOff+2] == 4) // Graphic Control Extension
594 {
595 //cBuf[iOff+3]; // page disposition flags
596 iDelay = cBuf[iOff+4]; // delay low byte
597 iDelay |= ((uint16_t)(cBuf[iOff+5]) << 8); // delay high byte
598 if (iDelay < 2) // too fast, provide a default
599 iDelay = 2;
600 iDelay *= 10; // turn JIFFIES into milliseconds
601 iTotalDelay += iDelay;
602 if (iDelay > iMaxDelay) iMaxDelay = iDelay;
603 else if (iDelay < iMinDelay) iMinDelay = iDelay;
604 // (cBuf[iOff+6]; // transparent color index
605 }
606 iOff += 2; /* skip to length */
607 iOff += (int)cBuf[iOff]; /* Skip the data block */
608 iOff++;
609 // block terminator or optional sub blocks
610 c = cBuf[iOff++]; /* Skip any sub-blocks */
611 while (c)
612 {
613 iOff += (int)c;
614 c = cBuf[iOff++];
615 if ((iDataAvailable - iOff) < (c+258)) // need to read more data first
616 {
617 lv_memmove(cBuf, &cBuf[iOff], (iDataAvailable-iOff)); // move existing data down
618 iDataAvailable -= iOff;
619 iOff = 0;
620 iReadAmount = (*pPage->pfnRead)(&pPage->GIFFile, &cBuf[iDataAvailable], FILE_BUF_SIZE-iDataAvailable);
621 iDataAvailable += iReadAmount;
622 iDataRemaining -= iReadAmount;
623 // lFileOff += iReadAmount;
624 }
625 }
626 if (c != 0) // problem, we went past the end
627 {
628 iNumFrames--; // possible corrupt data; stop
629 goto gifpagesz;
630 }
631 break;
632 case 0x2c: /* Start of image data */
633 bExt = 0; /* Stop doing extension blocks */
634 break;
635 default:
636 /* Corrupt data, stop here */
637 iNumFrames--;
638 goto gifpagesz;
639 } // switch
640 } // while
641 if (iOff >= iDataAvailable) // problem
642 {
643 iNumFrames--; // possible corrupt data; stop
644 goto gifpagesz;
645 }
646 /* Start of image data */
647 c = cBuf[iOff+9]; /* Get the flags byte */
648 iOff += 10; /* Skip image position and size */
649 if (c & 0x80) /* Local color table */
650 {
651 c &= 7;
652 iOff += (2<<c)*3;
653 }
654 iOff++; /* Skip LZW code size byte */
655 if ((iDataAvailable - iOff) < (c+258)) // need to read more data first
656 {
657 if (iOff < iDataAvailable) {
658 lv_memmove(cBuf, &cBuf[iOff], (iDataAvailable-iOff)); // move existing data down
659 iDataAvailable -= iOff;
660 iOff = 0;
661 } else { // already points beyond end
662 iOff -= iDataAvailable;
663 iDataAvailable = 0;
664 }
665 iReadAmount = (*pPage->pfnRead)(&pPage->GIFFile, &cBuf[iDataAvailable], FILE_BUF_SIZE-iDataAvailable);
666 iDataAvailable += iReadAmount;
667 iDataRemaining -= iReadAmount;
668 // lFileOff += iReadAmount;
669 }
670 c = cBuf[iOff++];
671 while (c) /* While there are more data blocks */
672 {
673 if (iOff > (3*FILE_BUF_SIZE/4) && iDataRemaining > 0) /* Near end of buffer, re-align */
674 {
675 lv_memmove(cBuf, &cBuf[iOff], (iDataAvailable-iOff)); // move existing data down
676 iDataAvailable -= iOff;
677 iOff = 0;
678 iReadAmount = (FILE_BUF_SIZE - iDataAvailable);
679 if (iReadAmount > iDataRemaining)
680 iReadAmount = iDataRemaining;
681 iReadAmount = (*pPage->pfnRead)(&pPage->GIFFile, &cBuf[iDataAvailable], iReadAmount);
682 iDataAvailable += iReadAmount;
683 iDataRemaining -= iReadAmount;
684 // lFileOff += iReadAmount;
685 }
686 iOff += (int)c; /* Skip this data block */
687// if ((int)lFileOff + iOff > pPage->GIFFile.iSize) // past end of file, stop
688// {
689// iNumFrames--; // don't count this page
690// break; // last page is corrupted, don't use it
691// }
692 c = cBuf[iOff++]; /* Get length of next */
693 }
694 /* End of image data, check for more pages... */
695 if (cBuf[iOff] == 0x3b || (iDataRemaining == 0 && (iDataAvailable - iOff) < 32))
696 {
697 bDone = 1; /* End of file has been reached */
698 }
699 else /* More pages to scan */
700 {
701 iNumFrames++;
702 // read new page data starting at this offset
703 if (pPage->GIFFile.iSize > FILE_BUF_SIZE && iDataRemaining > 0) // since we didn't read the whole file in one shot
704 {
705 lv_memmove(cBuf, &cBuf[iOff], (iDataAvailable-iOff)); // move existing data down
706 iDataAvailable -= iOff;
707 iOff = 0;
708 iReadAmount = (FILE_BUF_SIZE - iDataAvailable);
709 if (iReadAmount > iDataRemaining)
710 iReadAmount = iDataRemaining;
711 iReadAmount = (*pPage->pfnRead)(&pPage->GIFFile, &cBuf[iDataAvailable], iReadAmount);
712 iDataAvailable += iReadAmount;
713 iDataRemaining -= iReadAmount;
714 // lFileOff += iReadAmount;
715 }
716 }
717 } /* while !bDone */
718gifpagesz:
719 pInfo->iFrameCount = iNumFrames;
720 pInfo->iMaxDelay = iMaxDelay;
721 pInfo->iMinDelay = iMinDelay;
722 pInfo->iDuration = iTotalDelay;
723 return 1;
724} /* GIF_getInfo() */

726//
727// Unpack more chunk data for decoding
728// returns 1 to signify more data available for this image
729// 0 indicates there is no more data
730//
731static int GIFGetMoreData(GIFIMAGE *pPage)
732{
733 int iDelta = (pPage->iLZWSize - pPage->iLZWOff);
734 int iLZWBufSize;
735 unsigned char c = 1;
736
737 // Turbo mode uses combined buffers to read more compressed data
738 iLZWBufSize = (pPage->pTurboBuffer) ? LZW_BUF_SIZE_TURBO : LZW_BUF_SIZE;
739 // move any existing data down
740 if (pPage->bEndOfFrame || iDelta >= (iLZWBufSize - MAX_CHUNK_SIZE) || iDelta <= 0)
741 return 1; // frame is finished or buffer is already full; no need to read more data
742 if (pPage->iLZWOff != 0)
743 {
744// NB: memcpy() fails on some systems because the src and dest ptrs overlap
745// so copy the bytes in a simple loop to avoid problems
746 for (int i=0; i<pPage->iLZWSize - pPage->iLZWOff; i++) {
747 pPage->ucLZW[i] = pPage->ucLZW[i + pPage->iLZWOff];
748 }
749 pPage->iLZWSize -= pPage->iLZWOff;
750 pPage->iLZWOff = 0;
751 }
752 while (c && pPage->GIFFile.iPos < pPage->GIFFile.iSize && pPage->iLZWSize < (iLZWBufSize-MAX_CHUNK_SIZE))
753 {
754 (*pPage->pfnRead)(&pPage->GIFFile, &c, 1); // current length
755 (*pPage->pfnRead)(&pPage->GIFFile, &pPage->ucLZW[pPage->iLZWSize], c);
756 pPage->iLZWSize += c;
757 }
758 if (c == 0) // end of frame
759 pPage->bEndOfFrame = 1;
760 return (c != 0 && pPage->GIFFile.iPos < pPage->GIFFile.iSize); // more data available?
761} /* GIFGetMoreData() */
762//
763// Draw and convert pixels when the user wants fully rendered output
764//
765static void DrawCooked(GIFIMAGE *pPage, GIFDRAW *pDraw, void *pDest)
766{
767 uint8_t c, *s, *d8, *pEnd;
768 uint8_t *pActivePalette;
769
770 pActivePalette = (pPage->bUseLocalPalette) ? (uint8_t *)pPage->pLocalPalette : (uint8_t *)pPage->pPalette;
771 // d8 points to the line in the full sized canvas where the new opaque pixels will be merged
772 d8 = &pPage->pFrameBuffer[pDraw->iX + (pDraw->iY + pDraw->y) * pPage->iCanvasWidth];
773 s = pDraw->pPixels; // s points to the newly decoded pixels of this line of the current frame
774 pEnd = s + pDraw->iWidth; // faster way to loop over the source pixels - eliminates a counter variable
775
776 if (pPage->ucPaletteType == GIF_PALETTE_1BPP || pPage->ucPaletteType == GIF_PALETTE_1BPP_OLED) { // 1-bit mono
777 uint8_t *d = NULL;
778 uint8_t *pPal = pActivePalette;
779 uint8_t uc, ucMask;
780 int iPitch = 0;
781 if (pPage->ucPaletteType == GIF_PALETTE_1BPP) { // horizontal pixels
782 d = pPage->pFrameBuffer;
783 iPitch = (pPage->iCanvasWidth+7)/8;
784 d += (pPage->iCanvasWidth * pPage->iCanvasHeight);
785 d += pDraw->iX/8; // starting column
786 d += (pDraw->iY + pDraw->y) * iPitch;
787 // Apply the new pixels to the main image and generate 1-bpp output
788 if (pDraw->ucHasTransparency) { // if transparency used
789 uint8_t ucTransparent = pDraw->ucTransparent;
790 if (pDraw->ucDisposalMethod == 2) { // restore to background color
791 uint8_t u8BG = pPal[pDraw->ucBackground];
792 if (u8BG == 1) u8BG = 0xff; // set all bits to use mask
793 uc = *d; ucMask = (0x80 >> (pDraw->iX & 7));;
794 while (s < pEnd) {
795 c = *s++;
796 if (c != ucTransparent) {
797 if (pPal[c])
798 uc |= ucMask;
799 else
800 uc &= ~ucMask;
801 *d8++ = c;
802 } else {
803 uc |= (u8BG & ucMask); // transparent pixel is restored to background color
804 *d8++ = pDraw->ucBackground;
805 }
806 ucMask >>= 1;
807 if (ucMask == 0) { // write the completed byte
808 *d++ = uc;
809 uc = *d;
810 ucMask = 0x80;
811 }
812 }
813 *d = uc; // write last partial byte
814 } else { // no disposal, just write non-transparent pixels
815 uc = *d; ucMask = (0x80 >> (pDraw->iX & 7));
816 while (s < pEnd) {
817 c = *s++;
818 if (c != ucTransparent) {
819 if (pPal[c])
820 uc |= ucMask;
821 else
822 uc &= ~ucMask;
823 *d8 = c;
824 }
825 d8++;
826 ucMask >>= 1;
827 if (ucMask == 0) {
828 *d++ = uc;
829 uc = *d;
830 ucMask = 0x80;
831 }
832 }
833 *d = uc;
834 }
835 } else { // convert everything as opaque
836 uc = *d; ucMask = (0x80 >> (pDraw->iX & 7)); // left pixel is MSB
837 while (s < pEnd) {
838 c = *d8++ = *s++; // just write the new opaque pixels over the old
839 if (pPal[c]) // if non-zero, set white pixel
840 uc |= ucMask;
841 else
842 uc &= ~ucMask;
843 ucMask >>= 1;
844 if (ucMask == 0) { // time to write the current byte
845 *d++ = uc;
846 uc = *d;
847 ucMask = 0x80;
848 }
849 }
850 *d = uc;
851 }
852 } else { // vertical pixels
853 d = pPage->pFrameBuffer;
854 d += (pPage->iCanvasWidth * pPage->iCanvasHeight);
855 d += pDraw->iX; // starting column
856 d += ((pDraw->iY + pDraw->y)>>3) * pPage->iCanvasWidth;
857 ucMask = 1 << ((pDraw->iY + pDraw->y) & 7);
858 // Apply the new pixels to the main image and generate 1-bpp output
859 if (pDraw->ucHasTransparency) { // if transparency used
860 uint8_t ucTransparent = pDraw->ucTransparent;
861 if (pDraw->ucDisposalMethod == 2) { // restore to background color
862 uint8_t u8BG = pPal[pDraw->ucBackground];
863 u8BG *= ucMask; // set the right bit
864 while (s < pEnd) {
865 c = *s++;
866 uc = *d & ~ucMask; // clear old pixel
867 if (c != ucTransparent) {
868 uc |= (pPal[c] * ucMask);
869 *d8++ = c;
870 } else {
871 uc |= u8BG; // transparent pixel is restored to background color
872 *d8++ = pDraw->ucBackground;
873 }
874 *d++ = uc; // write back the updated pixel
875 }
876 } else { // no disposal, just write non-transparent pixels
877 while (s < pEnd) {
878 c = *s++;
879 uc = *d & ~ucMask;
880 if (c != ucTransparent) {
881 *d = uc | (pPal[c] * ucMask);
882 *d8 = c;
883 }
884 d++;
885 d8++;
886 }
887 }
888 } else { // convert everything as opaque
889 while (s < pEnd) {
890 c = *d8++ = *s++; // just write the new opaque pixels over the old
891 uc = *d & ~ucMask;
892 *d++ = uc | (pPal[c] * ucMask);
893 }
894 }
895 }
896 } else if (pPage->ucPaletteType == GIF_PALETTE_RGB565_LE || pPage->ucPaletteType == GIF_PALETTE_RGB565_BE) {
897 uint16_t *d, *pPal = (uint16_t *)pActivePalette;
898 d = (uint16_t *)pDest; // dest pointer to the cooked pixels
899 // Apply the new pixels to the main image
900 if (pDraw->ucHasTransparency) { // if transparency used
901 uint8_t ucTransparent = pDraw->ucTransparent;
902 if (pDraw->ucDisposalMethod == 2) { // restore to background color
903 uint16_t u16BG = pPal[pDraw->ucBackground];
904 while (s < pEnd) {
905 c = *s++;
906 if (c != ucTransparent) {
907 *d++ = pPal[c];
908 *d8++ = c;
909 } else {
910 *d++ = u16BG; // transparent pixel is restored to background color
911 *d8++ = pDraw->ucBackground;
912 }
913 }
914 } else { // no disposal, just write non-transparent pixels
915 while (s < pEnd) {
916 c = *s++;
917 if (c != ucTransparent) {
918 *d = pPal[c];
919 *d8 = c;
920 }
921 d++;
922 d8++;
923 }
924 }
925 } else { // convert all pixels through the palette without transparency
926#if REGISTER_WIDTH == 64
927 // parallelize the writes
928 // optimizing for the write buffer helps; reading 4 bytes at a time vs 1 doesn't on M1
929 while (s < pEnd + 4) { // group 4 pixels
930 BIGUINT bu;
931 uint8_t s0, s1, s2, s3;
932 uint16_t d1, d2, d3;
933 *(uint32_t *)d8 = *(uint32_t *)s; // just copy new opaque pixels over the old
934 s0 = s[0]; s1 = s[1]; s2 = s[2]; s3 = s[3];
935 bu = pPal[s0]; // not much difference on Apple M1
936 d1 = pPal[s1]; // but other processors may gain
937 d2 = pPal[s2]; // from unrolling the reads
938 d3 = pPal[s3];
939 bu |= (BIGUINT)d1 << 16;
940 bu |= (BIGUINT)d2 << 32;
941 bu |= (BIGUINT)d3 << 48;
942 s += 4;
943 d8 += 4;
944 *(BIGUINT *)d = bu;
945 d += 4;
946 }
947#endif
948 while (s < pEnd) {
949 c = *d8++ = *s++; // just write the new opaque pixels over the old
950 *d++ = pPal[c]; // and create the cooked pixels through the palette
951 }
952 }
953 } else { // 24bpp or 32bpp
954 uint8_t pixel, *d, *pPal;
955 int x;
956 d = (uint8_t *)pDest;
957 pPal = pActivePalette;
958 if (pDraw->ucHasTransparency) {
959 uint8_t ucTransparent = pDraw->ucTransparent;
960 if (pDraw->ucDisposalMethod == 2) { // restore to background color
961 uint8_t * bg = &pPal[pDraw->ucBackground * 3];
962 if (pPage->ucPaletteType == GIF_PALETTE_RGB888) {
963 while (s < pEnd) {
964 pixel = *s++;
965 if (pixel != ucTransparent) {
966 *d8++ = pixel;
967 d[0] = pPal[(pixel * 3) + 2];
968 d[1] = pPal[(pixel * 3) + 1];
969 d[2] = pPal[(pixel * 3) + 0];
970 d += 3;
971 } else {
972 *d8++ = pDraw->ucBackground;
973 d[0] = bg[2];
974 d[1] = bg[1];
975 d[2] = bg[0];
976 d += 3;
977 }
978 }
979 } else { /* GIF_PALETTE_RGB8888 */
980 while (s < pEnd) {
981 pixel = *s++;
982 if (pixel != ucTransparent) {
983 *d8++ = pixel;
984 d[0] = pPal[(pixel * 3) + 2];
985 d[1] = pPal[(pixel * 3) + 1];
986 d[2] = pPal[(pixel * 3) + 0];
987 d[3] = 0xFF;
988 d += 4;
989 } else {
990 *d8++ = pDraw->ucBackground;
991 d[3] = 0x00;
992 d += 4;
993 }
994 }
995 }
996 } else { // no disposal, just write non-transparent pixels
997 if (pPage->ucPaletteType == GIF_PALETTE_RGB888) {
998 for (x=0; x<pPage->iWidth; x++) {
999 pixel = *s++;
1000 if (pixel != ucTransparent) {
1001 *d8 = pixel;
1002 d[0] = pPal[(pixel * 3) + 2]; // convert to RGB888 pixels
1003 d[1] = pPal[(pixel * 3) + 1];
1004 d[2] = pPal[(pixel * 3) + 0];
1005 }
1006 d8++;
1007 d += 3;
1008 }
1009 } else { // must be RGBA32
1010 for (x=0; x<pPage->iWidth; x++) {
1011 pixel = *s++;
1012 if (pixel != ucTransparent) {
1013 *d8 = pixel;
1014 d[0] = pPal[(pixel * 3) + 2]; // convert to RGB8888 pixels
1015 d[1] = pPal[(pixel * 3) + 1];
1016 d[2] = pPal[(pixel * 3) + 0];
1017 d[3] = 0xff;
1018 }
1019 d8++;
1020 d += 4;
1021 }
1022 }
1023 }
1024 } else { // no transparency
1025 if (pPage->ucPaletteType == GIF_PALETTE_RGB888) {
1026 for (x=0; x<pPage->iWidth; x++) {
1027 pixel = *d8++ = *s++;
1028 *d++ = pPal[(pixel * 3) + 2]; // convert to RGB888 pixels
1029 *d++ = pPal[(pixel * 3) + 1];
1030 *d++ = pPal[(pixel * 3) + 0];
1031 }
1032 } else { // must be RGBA32
1033 for (x=0; x<pPage->iWidth; x++) {

1036 *d++ = pPal[(pixel * 3) + 1];
1037 *d++ = pPal[(pixel * 3) + 0];
1038 *d++ = 0xff;
1039 }
1040 }
1041 } // opaque
1042 }

1074// LZWCopyBytes
1075//
1076// Output the bytes for a single code (checks for buffer len)
1077//
1078static int LZWCopyBytes(unsigned char *buf, int iOffset, uint32_t *pSymbols, uint16_t *pLengths)
1079{
1080int iLen;
1081uint8_t c, *s, *d, *pEnd;
1082uint32_t u32Offset;
1083
1084 iLen = *pLengths;
1085 u32Offset = *pSymbols;
1086 // The string data frequently writes past the end of the framebuffer (past last pixel)
1087 // ...but with the placement of our code tables AFTER the framebuffer, it doesn't matter
1088 // Adding a check for buffer overrun here slows everything down about 10%
1089 s = &buf[u32Offset & 0x7fffff];
1090 d = &buf[iOffset];
1091 pEnd = &d[iLen];
1092 while (d < pEnd) // most frequent are 1-8 bytes in length, copy 4 or 8 bytes in these cases too
1093 {
1094#ifdef ALLOWS_UNALIGNED
1095// This is a significant perf improvement compared to copying 1 byte at a time
1096// even though it will often copy too many bytes
1097 BIGUINT tmp = *(BIGUINT *) s;
1098 s += sizeof(BIGUINT);
1099 *(BIGUINT *)d = tmp;
1100 d += sizeof(BIGUINT);
1101#else
1102// CPUs which enforce unaligned address exceptions must do it 1 byte at a time
1103 *d++ = *s++;
1104#endif
1105 }
1106 if (u32Offset & 0x800000) // was a newly used code
1107 {
1108 d = pEnd; // in case we overshot
1109 c = (uint8_t)(u32Offset >> 24);
1110 iLen++;
1111 // since the code with extension byte has now been written to the output, fix the code
1112 *pSymbols = iOffset;
1113// pSymbols[SYM_EXTRAS] = 0xffffffff;
1114 *d = c;
1115 *pLengths = (uint16_t)iLen;
1116 }
1117 return iLen;
1118} /* LZWCopyBytes() */

1121//
1122#define GET_CODE_TURBO if (bitnum > (REGISTER_WIDTH - MAX_CODE_SIZE/*codesize*/)) { p += (bitnum >> 3); \
1123 bitnum &= 7; ulBits = INTELLONG(p); } \

1126
1127//
1128// DecodeLZWTurbo
1129//
1130// Theory of operation:
1131//
1132// The 'traditional' LZW decoder maintains a dictionary with a linked list of codes.
1133// These codes build into longer chains as more data is decoded. To output the pixels,
1134// the linked list is traversed backwards from the last node to the first, then these
1135// pixels are copied in reverse order to the output bitmap.
1136//
1137// My decoder takes a different approach. The output image becomes the dictionary and
1138// the tables keep track of where in the output image the 'run' begins and its length.
1139// ** NB **
1140// These tables cannot be 16-bit values because a single dictionary's output can be
1141// bigger than 64K
1142//
1143// I also work with the compressed data differently. Most decoders wind their way through
1144// the chunked data by constantly checking if the current chunk has run out of data. I
1145// take a different approach since modern machines have plenty of memory - I 'de-chunk'
1146// the data first so that the inner loop can just decode as fast as possible. I also keep
1147// a set of codes in a 64-bit local variable to minimize memory reads.
1148//
1149// These 2 changes result in a much faster decoder. For poorly compressed images, the
1150// speed gain is about 2.5x compared to giflib. For well compressed images (long runs)
1151// the speed can be as much as 30x faster. This is because it doesn't have to walk
1152// backwards through the linked list of codes when outputting pixels. It also doesn't
1153// have to copy pixels in reverse order, then unwind them.
1154//
1155static int DecodeLZWTurbo(GIFIMAGE *pImage, int iOptions)
1156{
1157int i, bitnum;
1158int iUncompressedLen;
1159uint32_t code, oldcode, codesize, nextcode, nextlim;
1160uint32_t cc, eoi;
1161uint32_t sMask;
1162uint8_t c, *p, *buf, codestart, *pHighWater;
1163BIGUINT ulBits;
1164int iLen, iColors;
1165int iErr = GIF_SUCCESS;
1166int iOffset;
1167uint32_t *pSymbols;
1168uint16_t *pLengths;
