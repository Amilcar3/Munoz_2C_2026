1"""`info widget`: everything known about a widget, in a readable list.
2
3`dump obj` walks the tree but prints one line per widget, and `info obj_class`
4answers about the class, not the instance in front of you. Between them the
5per-widget fields the wrappers collect - a label's text, an arc's angles - were
6only reachable through `dump dashboard`. This command prints the same data the
7dashboard shows, for one widget or for every widget on every display - and,
8given field names, only the fields asked for.
9
10`dump widget props` answers the question that comes right after: which field
11names does this widget have? It reads them from the generated wrapper, so a
12class name is enough - no live object needed.
13"""

29# --- finding a widget by name ------------------------------------------------
30#
31# The names are resolved the way lv_obj_get_name_resolved() resolves them, and
32# the search follows lv_obj_find_by_name() / lv_obj_get_child_by_name(): a
33# trailing '#' is an auto-index over the siblings sharing that name, and a
34# widget that was never named answers to `<class>_<n>`. Reimplementing this
35# rather than calling into the target keeps it working on a core dump, where
36# nothing can be called.
37
38
39def resolved_name(obj) -> str:
40 """The name `lv_obj_get_name_resolved()` would produce for this widget."""
41 raw = obj._get_name()
42 auto = raw is None
43 name = f"{obj.class_name}_#" if auto else raw
44
45 if not name.endswith("#"):
46 return name
47
48 parent = obj.super_value("parent")
49 parent = LVObject(parent) if parent.is_ok and int(parent) else None
50 # A screen has no parent, so its siblings are the other screens of its
51 # display. LVGL leaves the '#' unresolved there; indexing it as well is what
52 # makes a screen the typable first segment of a name path.
53 siblings = parent.children if parent is not None else _sibling_screens(obj)
54 if siblings is None:
55 return name
56
57 index = 0
58 for sibling in siblings:
59 if int(sibling) == int(obj):
60 return name[:-1] + str(index)
61 sibling_name = sibling._get_name()
62 if sibling_name is None:
63 # An unnamed sibling counts when the name we resolve is its own
64 # auto name, i.e. `<its class>_#`.
65 class_name = sibling.class_name
66 if len(name) > 3 and len(class_name) == len(name) - 2 \
67 and name.startswith(class_name):
68 index += 1
69 elif sibling_name == name:
70 index += 1
71 return name

74def _sibling_screens(obj):
75 """The screen list holding this screen, or None if it is not a screen."""
76 if not curr_inst().ensure_init():
77 return None
78 for disp in curr_inst().displays():
79 try:
80 found = list(disp.screens)

82 continue
83 if any(int(s) == int(obj) for s in found):
84 return found
85 return None

88def name_path(obj) -> str:
89 """The resolved names from the screen down to this widget, slash separated."""
90 parts = []
91 node = obj
92 seen = set()
93 while node is not None and int(node) not in seen:
94 seen.add(int(node))
95 try:
96 parts.append(resolved_name(node))
97 parent = node.super_value("parent")
98 except CorruptedError:
99 parts.append("?")
100 break
101 node = LVObject(parent) if parent.is_ok and int(parent) else None
102 return "/".join(reversed(parts))

118def find_by_name(name, parent=None):
119 """Widgets called *name* below *parent*, or below every screen.
120
121 Unlike `lv_obj_find_by_name()`, which stops at the first hit on the active
122 screen, every match is collected: in the debugger an ambiguous name is
123 worth seeing rather than silently resolving to one of several widgets.
124 """
125 found = []
126
127 def walk(obj, matchable=True):
128 try:
129 if matchable and _name_is(obj, name):
130 found.append(obj)
131 for child in obj.children:
132 walk(child)
133 except CorruptedError:
134 pass
135
136 if parent is not None:
137 # Like lv_obj_find_by_name(), the search starts below the parent: with
138 # `info widget dialog -p dialog` the parent itself is not the answer.
139 walk(parent, matchable=False)
140 else:
141 for root in screens():
142 walk(root)
143 return found

146def find_by_path(path, parent=None):
147 """The widget at a slash separated name path, or None.
148
149 With no *parent* the first segment names a screen, so
150 `screen_1/button_2/label3` is a whole path from the top.
151 """
152 segments = [seg for seg in path.split("/") if seg]
153 if not segments:
154 return None
155
156 if parent is None:
157 candidates = [s for s in screens() if _name_is(s, segments[0])]
158 segments = segments[1:]
159 else:
160 candidates = [parent]
161
162 for segment in segments:
163 nxt = None
164 for candidate in candidates:
165 try:
166 for child in candidate.children:
167 if _name_is(child, segment):
168 nxt = child
169 break
170 except CorruptedError:
171 continue
172 if nxt is not None:
173 break
174 if nxt is None:
175 return None
176 candidates = [nxt]
177
178 return candidates[0] if candidates else None

189def resolve_widget(token, parent_path=None):
190 """Turn what the user typed into one widget, or explain why it is not one.
191
192 Returns (widget, error). The token is a C expression when GDB can evaluate
193 it - a variable, a member chain, an address - and a name or a name path
194 otherwise.
195 """
196 parent = None
197 if parent_path:
198 parent, error = resolve_widget(parent_path)
199 if parent is None:
200 return None, f"--parent: {error}"
201
202 if "/" in token:
203 found = find_by_path(token, parent)
204 if found is None:
205 where = f" under {parent_path}" if parent_path else ""
206 return None, f"no widget at path '{token}'{where}"
207 return found, None
208
209 if parent is None:
210 try:
211 return LVObject(gdb.parse_and_eval(token)), None
212 except gdb.error:
213 pass # not an expression, so it is a name
214
215 matches = find_by_name(token, parent)
216 if not matches:
217 where = f" under {parent_path}" if parent_path else ""
218 return None, f"no widget named '{token}'{where}"
219 if len(matches) > 1:
220 lines = [f"'{token}' matches {len(matches)} widgets:"]
221 lines += [
222 f" {name_path(m)} ({m.class_name} @{hex(int(m))})" for m in matches
223 ]
224 lines.append("Narrow it down with --parent, or pass a name path.")
225 return None, "\n".join(lines)
226 return matches[0], None

240 if not addr:

256def _headline(obj, d) -> str:
257 gist = specs.summary(d.get("class_name", ""), d.get("widget_data"))
258 parts = [f"{d.get('class_name', '?')} @{d.get('addr', '?')}"]

268def _compact(obj) -> str:
269 """One line for the tree listing."""
270 d = _snapshot(obj)
271 line = _headline(obj, d)
272 c = d.get("coords") or {}
273 x1, y1 = c.get("x1", 0), c.get("y1", 0)
274 line += f" ({x1},{y1}) {c.get('x2', 0) - x1 + 1}x{c.get('y2', 0) - y1 + 1}"
275 states = [s for s in d.get("state_list", []) if s != "DEFAULT"]
276 if states:
277 line += f" state={'|'.join(states)}"
278 if d.get("error"):
279 line += f" (corrupted: {d['error']})"

340def _obj_field(d, name):
341 """An object-level field by name, accepting a few friendly aliases."""
342 aliases = {
343 "class": "class_name",
344 "state": "state_list",
345 "flags": "flags_list",
346 "parent": "parent_addr",
347 "group": "group_addr",
348 "children": "child_count",
349 "styles": "style_count",
350 }
351 key = aliases.get(name, name)
352 if key not in d or key == "widget_data":
353 return None, False
354 return d[key], True

357def _format_obj_field(key, value):
358 """Render an object-level field the way the full listing renders it."""
359 if key in ("state", "state_list"):
360 return "|".join(value or ["DEFAULT"])
361 if key in ("flags", "flags_list"):
362 return "|".join(value or []) or "-"
363 if key == "coords":
364 return _size_str({"coords": value})
365 if key in ("parent", "parent_addr", "group", "group_addr", "user_data"):
366 return _addr_str(value)
367 if key == "scroll_dir":
368 return format_enum("lv_dir_t", value, "LV_DIR_", bitmask=True)
369 if key in ("scroll_snap_x", "scroll_snap_y"):
370 return format_enum("lv_scroll_snap_t", value, "LV_SCROLL_SNAP_")
371 if key == "scrollbar_mode":
372 return format_enum("lv_scrollbar_mode_t", value, "LV_SCROLLBAR_MODE_")
373 if key == "layer_type":
374 return format_enum("lv_layer_type_t", value, "LV_LAYER_TYPE_")
375 return "-" if value is None else str(value)

378def print_widget_fields(obj, names):
379 """Print only the named fields of one widget."""
380 d = _snapshot(obj)
381 class_name = d.get("class_name", "")
382 data = d.get("widget_data") or {}
383 for name in names:
384 if name in data:
385 _field(name, specs.format_value(class_name, name, data[name]), indent=0)
386 continue
387 value, found = _obj_field(d, name)
388 if found:
389 _field(name, _format_obj_field(name, value), indent=0)
390 continue
391 print(f"{name}: no such field on {class_name}"
392 f" (see: dump widget props {class_name})")

395def print_widget(obj, expr=None):
396 """Print everything known about one widget."""
397 d = _snapshot(obj)
398
399 print(_headline(obj, d))
400 chain = []
401 cls = obj.obj_class
402 if cls:
403 chain = [c.name for c in cls]
404 _field("class", " -> ".join(chain) if chain else d.get("class_name", "?"))
405 _field("parent", _addr_str(d.get("parent_addr")))
406 _field("coords", _size_str(d))
407 _field("children", d.get("child_count", 0))
408 _field("state", "|".join(d.get("state_list") or ["DEFAULT"]))
409 _field("flags", "|".join(d.get("flags_list") or []) or "-")
410 _print_scroll(d)
411 if d.get("ext_click_pad") is not None or d.get("ext_draw_size") is not None:
412 _field(
413 "ext_pad",
414 f"click={d.get('ext_click_pad', 0)} draw={d.get('ext_draw_size', 0)}",
415 )
416 layer = d.get("layer_type")
417 _field(
418 "layout",
419 f"inv={d.get('layout_inv')} w={d.get('w_layout')} h={d.get('h_layout')}"
420 + (
421 f" layer={format_enum('lv_layer_type_t', layer, 'LV_LAYER_TYPE_')}"
422 if layer is not None
423 else ""
424 ),
425 )
426 _field(
427 "status",
428 f"rendered={d.get('rendered')} deleting={d.get('is_deleting')}"
429 f" skip_trans={d.get('skip_trans')}",
430 )
431 if d.get("group_addr"):
432 _field("group", d["group_addr"])
433 if d.get("user_data"):
434 _field("user_data", _addr_str(d["user_data"]))
435
436 style_cnt = d.get("style_count", 0)
437 hint = f" (info style --obj {expr})" if expr and style_cnt else ""
438 _field("styles", f"{style_cnt}{hint}")
439
440 _print_events(obj)
441 _print_widget_data(d)
442
443 try:
444 children = list(obj.children)
445 except CorruptedError:
446 print(" children = (corrupted)")
447 return
448 if children:
449 print(" children")
450 for i, child in enumerate(children):
451 try:
452 print(f" [{i}] {_compact(child)}")
453 except CorruptedError as e:
454 print(f" [{i}] (corrupted: {e})")

457def print_tree(obj, depth=0, limit=None):
458 """Print one compact line per widget, indented by tree depth."""
459 try:
460 print(" " * depth + _compact(obj))
461 except CorruptedError as e:
462 print(" " * depth + f"(corrupted: {e})")
463 return
464 if limit is not None and depth >= limit:
465 return
466 try:
467 for child in obj.children:
468 print_tree(child, depth + 1, limit=limit)
469 except CorruptedError:
470 print(" " * (depth + 1) + "(corrupted children)")

473def print_all_widgets(limit=None):
474 """List every widget of every display, screens labelled with their layer."""
475 if not curr_inst().ensure_init():
476 return
477 for disp in curr_inst().displays():
478 print(f"Display @{hex(int(disp))}")

486 except CorruptedError:
487 print(" (corrupted screens)")

490class InfoWidget(gdb.Command):
491 """show all details of a widget, or list every widget"""
492
493 def __init__(self):
494 super(InfoWidget, self).__init__(
495 "info widget", gdb.COMMAND_USER, gdb.COMPLETE_EXPRESSION
496 )
497
498 def invoke(self, args, from_tty):
499 parser = argparse.ArgumentParser(
500 prog="info widget",
501 description="Show widget details, or list all widgets.",
502 )
503 parser.add_argument(
504 "-t",
505 "--tree",
506 action="store_true",
507 help="List the subtree of the widget instead of its details.",
508 )
509 parser.add_argument(
510 "-L",
511 "--level",
512 type=int,
513 default=None,
514 help="Limit the depth of a listing.",
515 )
516 parser.add_argument(
517 "-p",
518 "--parent",
519 type=str,
520 default=None,
521 help="Search for the name below this widget, e.g. screen_1/button_2.",
522 )
523 parser.add_argument(
524 "widget_and_fields",
525 metavar="widget [field...]",
526 type=str,
527 nargs="*",
528 help="The widget - a name, a name path, an expression or an address"
529 " - then the field names to print instead of the whole widget.",
530 )
531 try:
532 # parse_known_args, because argparse fills a positional list in one
533 # go: in `info widget label3 -p screen_1 text` everything after the
534 # option would be dropped as unrecognised. The leftovers are the
535 # rest of the positionals, in the order they were typed.
536 opts, rest = parser.parse_known_args(gdb.string_to_argv(args))
537 except SystemExit:
538 return
539
540 flags = [word for word in rest if word.startswith("-")]
541 if flags:
542 print(f"Error: unknown option {flags[0]}")
543 return
544
545 positional = opts.widget_and_fields + rest
546 widget = positional[0] if positional else None
547 opts.fields = positional[1:]
548
549 if opts.parent and not widget:
550 print("Error: --parent needs a name to look for.")
551 return
552
553 if widget:
554 obj, error = resolve_widget(widget, opts.parent)
555 if obj is None:
556 print(f"Error: {error}")
557 return
558 if opts.tree:
559 print_tree(obj, limit=opts.level)
560 elif opts.fields:
561 print_widget_fields(obj, opts.fields)
562 else:
563 print_widget(obj, expr=widget)
564 return
565
566 print_all_widgets(limit=opts.level)

590def print_widget_props(class_name, live_fields=None):
591 """List the field names a widget has, grouped by the struct they come from."""
592 cls = _wrapper_class(class_name)
593 if cls is None:
594 print(f"{class_name} (no fields of its own)")
595 else:
596 print(f"{_class_name_of(cls)} ({cls.__name__})")
597
598 # Only the generated wrappers, so that LVObject's and Value's own helper
599 # properties - x1, obj_styles, is_ok - stay out of a list of widget fields.
600 for owner in (cls.__mro__ if cls else ()):
601 if not owner.__module__.startswith("lvglgdb.lvgl.widgets"):
602 continue
603 props = [
604 (name, value)
605 for name, value in vars(owner).items()
606 if isinstance(value, property)
607 ]
608 if not props:
609 continue
610 owner_name = _class_name_of(owner)
611 types = specs.field_types(owner_name)
612 print(f" from {owner_name}_t")
613 for name, prop in props:
614 doc = (prop.__doc__ or "").strip().splitlines()
615 desc = doc[0] if doc else ""
616 print(f" {name:{_LABEL_W + 4}s} {types.get(name, ''):<8s} {desc}".rstrip())
617
618 print(" common (any widget)")
619 fields = live_fields if live_fields is not None else _COMMON_FIELDS
620 line = " "
621 for name in fields:
622 if len(line) + len(name) > 76:
623 print(line)
624 line = " "
625 line += name + ", "
626 print(line.rstrip(", "))

662class DumpWidgetProps(gdb.Command):
663 """list the field names of a widget class or of a live widget"""
664
665 def __init__(self):
666 super(DumpWidgetProps, self).__init__(
667 "dump widget props", gdb.COMMAND_USER, gdb.COMPLETE_EXPRESSION
668 )
669
670 def invoke(self, args, from_tty):
671 parser = argparse.ArgumentParser(
672 prog="dump widget props",
673 description="List the fields 'info widget' can print.",
674 )
675 parser.add_argument(
676 "target",
677 type=str,
678 nargs="?",
679 default=None,
680 help="A widget class name (lv_label), or a widget itself:"
681 " a variable, a member chain, or its address.",
682 )
683 try:
684 opts = parser.parse_args(gdb.string_to_argv(args))
685 except SystemExit:
686 return
687
688 if not opts.target:
689 print_widget_classes()
690 return
691
692 if _wrapper_class(opts.target) is not None:
693 print_widget_props(opts.target)
694 return
695
696 obj, error = resolve_widget(opts.target)
697 if obj is None:
698 print(f"'{opts.target}' is neither a widget class nor a widget.")
699 print(f"({error})")
