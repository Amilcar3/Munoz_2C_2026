265{

267 LV_ASSERT(obj != NULL);
268 lv_slider_t * slider = (lv_slider_t *)obj;
269
270 /*Initialize the allocated 'slider'*/
271 slider->value_to_set = NULL;
272 slider->dragging = 0U;
273 slider->left_knob_focus = 0U;
274
275 lv_obj_set_scroll_chain_hor(obj, false);
276 lv_obj_set_scrollable(obj, false);
277 lv_obj_set_scroll_on_focus(obj, true);
278 lv_obj_set_ext_click_area(obj, LV_DPX(8));

281static void lv_slider_event(const lv_obj_class_t * class_p, lv_event_t * e)
282{
283 LV_UNUSED(class_p);
284 LV_ASSERT(e != NULL);
285
286 lv_result_t res;
287
288 /*Call the ancestor's event handler*/
289 res = lv_obj_event_base(MY_CLASS, e);
290 if(res != LV_RESULT_OK) return;
291
292 lv_event_code_t code = lv_event_get_code(e);
293 lv_obj_t * obj = lv_event_get_current_target(e);
294 LV_ASSERT(obj != NULL);
295 lv_slider_t * slider = (lv_slider_t *)obj;
296 lv_slider_mode_t type = lv_slider_get_mode(obj);
297
298 /*Advanced hit testing: react only on dragging the knob(s)*/
299 if(code == LV_EVENT_HIT_TEST) {
300 lv_hit_test_info_t * info = lv_event_get_param(e);
301 int32_t ext_click_area = obj->spec_attr ? obj->spec_attr->ext_click_pad : 0;
302
303 /*Ordinary slider: was the knob area hit?*/
304 lv_area_t a;
305 a = slider->right_knob_area;
306 lv_area_increase(&a, ext_click_area, ext_click_area);
307 info->res = lv_area_is_point_on(&a, info->point, 0);
308
309 /*There's still a chance that there is a hit if there is another knob*/
310 if((info->res == false) && (type == LV_SLIDER_MODE_RANGE)) {
311 a = slider->left_knob_area;
312 lv_area_increase(&a, ext_click_area, ext_click_area);
313 info->res = lv_area_is_point_on(&a, info->point, 0);
314 }
315 }
316 else if(code == LV_EVENT_PRESSED) {
317 /*Save the pressed coordinates*/
318 lv_indev_t * indev = lv_event_get_indev(e);
319 slider->pressed_point.x = 0;
320 slider->pressed_point.y = 0;
321 if(indev != NULL) lv_indev_get_point(indev, &slider->pressed_point);
322 lv_obj_transform_point(obj, &slider->pressed_point, LV_OBJ_POINT_TRANSFORM_FLAG_INVERSE_RECURSIVE);
323 }
324 else if(code == LV_EVENT_PRESSING) {
325 update_knob_pos(obj, true);
326 }
327 else if(code == LV_EVENT_RELEASED || code == LV_EVENT_PRESS_LOST) {
328 update_knob_pos(obj, false);
329 slider->dragging = false;
330 slider->value_to_set = NULL;
331
332 lv_obj_invalidate(obj);
333
334 /*Leave edit mode if released. (No need to wait for LONG_PRESS)*/
335 lv_group_t * g = lv_obj_get_group(obj);
336 bool editing = g != NULL && lv_group_get_editing(g);
337 lv_indev_t * indev = lv_event_get_indev(e);
338 lv_indev_type_t indev_type = indev != NULL ? lv_indev_get_type(indev) : LV_INDEV_TYPE_NONE;
339 if(indev_type == LV_INDEV_TYPE_ENCODER) {
340 if(editing) {
341 if(lv_slider_get_mode(obj) == LV_SLIDER_MODE_RANGE) {
342 if(slider->left_knob_focus == 0) slider->left_knob_focus = 1;
343 else {
344 slider->left_knob_focus = 0;
345 lv_group_set_editing(g, false);
346 }
347 }
348 else {
349 lv_group_set_editing(g, false);
350 }
351 }
352 }
353 else if(indev_type == LV_INDEV_TYPE_POINTER) {
354 if(is_slider_horizontal(obj)) lv_obj_set_scroll_chain_ver(obj, true);
355 else lv_obj_set_scroll_chain_hor(obj, true);
356 }
357 }
358 else if(code == LV_EVENT_FOCUSED) {
359 lv_indev_t * indev = lv_indev_active();
360 lv_indev_type_t indev_type = indev != NULL ? lv_indev_get_type(indev) : LV_INDEV_TYPE_NONE;
361 if(indev_type == LV_INDEV_TYPE_ENCODER || indev_type == LV_INDEV_TYPE_KEYPAD) {
362 slider->left_knob_focus = 0;
363 }
364 }
365 else if(code == LV_EVENT_SIZE_CHANGED) {
366 if(is_slider_horizontal(obj)) {
367 lv_obj_set_scroll_chain_ver(obj, true);
368 lv_obj_set_scroll_chain_hor(obj, false);
369 }
370 else {
371 lv_obj_set_scroll_chain_hor(obj, true);
372 lv_obj_set_scroll_chain_ver(obj, false);
373 }
374 lv_obj_refresh_ext_draw_size(obj);
375 }
376 else if(code == LV_EVENT_REFR_EXT_DRAW_SIZE) {
377 int32_t knob_left = lv_obj_get_style_pad_left_internal(obj, LV_PART_KNOB);
378 int32_t knob_right = lv_obj_get_style_pad_right_internal(obj, LV_PART_KNOB);
379 int32_t knob_top = lv_obj_get_style_pad_top_internal(obj, LV_PART_KNOB);
380 int32_t knob_bottom = lv_obj_get_style_pad_bottom_internal(obj, LV_PART_KNOB);
381
382 /*The smaller size is the knob diameter*/
383 int32_t trans_w = lv_obj_get_style_transform_width_internal(obj, LV_PART_KNOB);
384 int32_t trans_h = lv_obj_get_style_transform_height_internal(obj, LV_PART_KNOB);
385 int32_t knob_size = LV_MIN(lv_obj_get_width(obj) + 2 * trans_w, lv_obj_get_height(obj) + 2 * trans_h) >> 1;
386 knob_size += LV_MAX(LV_MAX(knob_left, knob_right), LV_MAX(knob_bottom, knob_top));
387 knob_size += 2; /*For rounding error*/
388 knob_size += lv_obj_calculate_ext_draw_size(obj, LV_PART_KNOB);
389
390 /*Indic. size is handled by bar*/
391 int32_t * s = lv_event_get_param(e);
392 *s = LV_MAX(*s, knob_size);
393
394 }
395 else if(code == LV_EVENT_KEY) {
396 uint32_t c = lv_event_get_key(e);
397
398 if(c == LV_KEY_RIGHT || c == LV_KEY_UP) {
399 if(!slider->left_knob_focus) lv_slider_set_value(obj, lv_slider_get_value(obj) + 1, LV_ANIM_ON);
400 else lv_slider_set_start_value(obj, lv_slider_get_left_value(obj) + 1, LV_ANIM_ON);
401 }
402 else if(c == LV_KEY_LEFT || c == LV_KEY_DOWN) {
403 if(!slider->left_knob_focus) lv_slider_set_value(obj, lv_slider_get_value(obj) - 1, LV_ANIM_ON);
404 else lv_slider_set_start_value(obj, lv_slider_get_left_value(obj) - 1, LV_ANIM_ON);
405 }
406 else {
407 return;
408 }
409
410 res = lv_obj_send_event(obj, LV_EVENT_VALUE_CHANGED, NULL);
411 if(res != LV_RESULT_OK) return;
412 }
413 else if(code == LV_EVENT_ROTARY) {
414 int32_t r = lv_event_get_rotary_diff(e);
415
416 if(!slider->left_knob_focus) lv_slider_set_value(obj, lv_slider_get_value(obj) + r, LV_ANIM_ON);
417 else lv_slider_set_start_value(obj, lv_slider_get_left_value(obj) + 1, LV_ANIM_ON);
418
419 res = lv_obj_send_event(obj, LV_EVENT_VALUE_CHANGED, NULL);
420 if(res != LV_RESULT_OK) return;
421 }
422 else if(code == LV_EVENT_DRAW_MAIN) {
423 draw_knob(e);
424 }
425}

427static void draw_knob(lv_event_t * e)
428{
429 LV_ASSERT(e != NULL);
430 lv_obj_t * obj = lv_event_get_current_target(e);
431 lv_layer_t * layer = lv_event_get_layer(e);
432 LV_ASSERT(obj != NULL);
433 LV_ASSERT(layer != NULL);
434 lv_slider_t * slider = (lv_slider_t *)obj;
435
436 const bool is_rtl = LV_BASE_DIR_RTL == lv_obj_get_style_base_dir_internal(obj, LV_PART_MAIN);
437 const bool is_horizontal = is_slider_horizontal(obj);
438 const bool is_reversed = slider->bar.val_reversed ^ (is_rtl && is_horizontal);
439
440 lv_area_t knob_area;
441 int32_t knob_size;
442 bool is_symmetrical = lv_slider_is_symmetrical(obj);
443
444 if(is_horizontal) {
445 knob_size = lv_obj_get_height(obj);
446 if(is_symmetrical &&
447 slider->bar.cur_value < 0) knob_area.x1 = LV_SLIDER_KNOB_COORD(!is_reversed, slider->bar.indic_area);
448 else knob_area.x1 = LV_SLIDER_KNOB_COORD(is_reversed, slider->bar.indic_area);
449 }
450 else {
451 knob_size = lv_obj_get_width(obj);
452 if(is_symmetrical &&
453 slider->bar.cur_value < 0) knob_area.y1 = LV_SLIDER_KNOB_COORD_VERTICAL(!is_reversed, slider->bar.indic_area);
454 else knob_area.y1 = LV_SLIDER_KNOB_COORD_VERTICAL(is_reversed, slider->bar.indic_area);
455 }
456 lv_draw_rect_dsc_t knob_rect_dsc;
457 lv_draw_rect_dsc_init(&knob_rect_dsc);
458 knob_rect_dsc.base.layer = layer;
459 lv_obj_init_draw_rect_dsc(obj, LV_PART_KNOB, &knob_rect_dsc);
460 /* Update knob area with knob style */
461 position_knob(obj, &knob_area, knob_size, is_horizontal);
462 /* Update right knob area with calculated knob area */
463 slider->right_knob_area = knob_area;
464
465 if(lv_slider_get_mode(obj) != LV_SLIDER_MODE_RANGE) {
466 lv_draw_rect(layer, &knob_rect_dsc, &slider->right_knob_area);
467 }
468 else {
469 /*Save the draw part_draw_dsc. because it can be modified in the event*/
470 lv_draw_rect_dsc_t knob_rect_dsc_tmp;
471 lv_memcpy(&knob_rect_dsc_tmp, &knob_rect_dsc, sizeof(lv_draw_rect_dsc_t));
472 /* Draw the right knob */
473 lv_draw_rect(layer, &knob_rect_dsc, &slider->right_knob_area);
474
475 /*Calculate the second knob area*/
476 if(is_horizontal) {
477 /*use !is_reversed to get the other knob*/
478 knob_area.x1 = LV_SLIDER_KNOB_COORD(!is_reversed, slider->bar.indic_area);
479 }
480 else {
481 knob_area.y1 = LV_SLIDER_KNOB_COORD_VERTICAL(!is_reversed, slider->bar.indic_area);
482 }
483 position_knob(obj, &knob_area, knob_size, is_horizontal);
484 slider->left_knob_area = knob_area;
485
486 lv_memcpy(&knob_rect_dsc, &knob_rect_dsc_tmp, sizeof(lv_draw_rect_dsc_t));
487
488 lv_draw_rect(layer, &knob_rect_dsc, &slider->left_knob_area);
489 }
490}

492static void position_knob(lv_obj_t * obj, lv_area_t * knob_area, const int32_t knob_size, const bool hor)
493{
494 LV_ASSERT(obj != NULL);
495 LV_ASSERT(knob_area != NULL);
496 if(hor) {
497 knob_area->x1 -= (knob_size >> 1);
498 knob_area->x2 = knob_area->x1 + knob_size - 1;
499 knob_area->y1 = obj->coords.y1;
500 knob_area->y2 = obj->coords.y2;
501 }
502 else {
503 knob_area->y1 -= (knob_size >> 1);
504 knob_area->y2 = knob_area->y1 + knob_size - 1;
505 knob_area->x1 = obj->coords.x1;
506 knob_area->x2 = obj->coords.x2;
507 }
508
509 int32_t knob_left = lv_obj_get_style_pad_left_internal(obj, LV_PART_KNOB);
510 int32_t knob_right = lv_obj_get_style_pad_right_internal(obj, LV_PART_KNOB);
511 int32_t knob_top = lv_obj_get_style_pad_top_internal(obj, LV_PART_KNOB);
512 int32_t knob_bottom = lv_obj_get_style_pad_bottom_internal(obj, LV_PART_KNOB);
513
514 int32_t transf_w = lv_obj_get_style_transform_width_internal(obj, LV_PART_KNOB);
515 int32_t transf_h = lv_obj_get_style_transform_height_internal(obj, LV_PART_KNOB);
516
517 /*Apply the paddings on the knob area*/
518 knob_area->x1 -= knob_left + transf_w;
519 knob_area->x2 += knob_right + transf_w;
520 knob_area->y1 -= knob_top + transf_h;
521 knob_area->y2 += knob_bottom + transf_h;
522}

524static bool is_slider_horizontal(lv_obj_t * obj)
525{
526 LV_ASSERT(obj != NULL);

534static void drag_start(lv_obj_t * obj)
535{
536 LV_ASSERT(obj != NULL);
537 lv_slider_t * slider = (lv_slider_t *)obj;
538 lv_slider_mode_t mode = lv_slider_get_mode(obj);
539 lv_point_t p = { 0, 0 };
540 slider->dragging = true;
541 if(mode == LV_SLIDER_MODE_NORMAL || mode == LV_SLIDER_MODE_SYMMETRICAL) {
542 slider->value_to_set = &slider->bar.cur_value;
543 }
544 else if(mode == LV_SLIDER_MODE_RANGE) {
545 lv_indev_t * indev = lv_indev_active();
546 if(indev != NULL) lv_indev_get_point(indev, &p);
547 lv_obj_transform_point(obj, &p, LV_OBJ_POINT_TRANSFORM_FLAG_INVERSE_RECURSIVE);
548 const bool is_rtl = LV_BASE_DIR_RTL == lv_obj_get_style_base_dir_internal(obj, LV_PART_MAIN);
549 const bool is_horizontal = is_slider_horizontal(obj);

569 slider->value_to_set = &slider->bar.start_value;

577 }
578 else if((!is_reversed && p.y > slider->left_knob_area.y2) || (is_reversed && p.y < slider->left_knob_area.y1)) {
579 slider->value_to_set = &slider->bar.start_value;
580 }

583 dist_left = LV_ABS((slider->left_knob_area.y1 + (slider->left_knob_area.y2 - slider->left_knob_area.y1) / 2) - p.y);
584 dist_right = LV_ABS((slider->right_knob_area.y1 + (slider->right_knob_area.y2 - slider->right_knob_area.y1) / 2) - p.y);
585

600static void update_knob_pos(lv_obj_t * obj, bool check_drag)
601{
602 LV_ASSERT(obj != NULL);
603 lv_slider_t * slider = (lv_slider_t *)obj;
604 lv_indev_t * indev = lv_indev_active();
605 if(indev == NULL) return;
606 if(lv_indev_get_type(indev) != LV_INDEV_TYPE_POINTER)
607 return;
608 if(lv_indev_get_scroll_obj(indev) != NULL && !slider->dragging)
609 return;
610
611 lv_point_t p;
612 lv_indev_get_point(indev, &p);
613 lv_obj_transform_point(obj, &p, LV_OBJ_POINT_TRANSFORM_FLAG_INVERSE_RECURSIVE);
614
615 bool is_hor = is_slider_horizontal(obj);
616
617 if(check_drag && !slider->dragging) {
618 int32_t ofs = is_hor ? (p.x - slider->pressed_point.x) : (p.y - slider->pressed_point.y);
619
620 /*Stop processing when offset is below scroll_limit*/
621 if(LV_ABS(ofs) < indev->scroll_limit) {
622 return;
623 }
624 }
625
626 if(!slider->value_to_set) {
627 /*Ready to start drag*/
628 drag_start(obj);
629 }
630
631 int32_t new_value = 0;
632 const int32_t range = slider->bar.max_value - slider->bar.min_value;
633 const bool is_rtl = LV_BASE_DIR_RTL == lv_obj_get_style_base_dir_internal(obj, LV_PART_MAIN);
634 const bool is_horizontal = is_slider_horizontal(obj);
635 const bool is_reversed = slider->bar.val_reversed ^ (is_rtl && is_horizontal);
636
637 if(is_hor) {
638 const int32_t bg_left = lv_obj_get_style_pad_left_internal(obj, LV_PART_MAIN);
639 const int32_t bg_right = lv_obj_get_style_pad_right_internal(obj, LV_PART_MAIN);
640 const int32_t w = lv_obj_get_width(obj);
641 const int32_t indic_w = w - bg_left - bg_right;
642
643 if(is_reversed) {
644 /*Make the point relative to the indicator*/
645 new_value = (obj->coords.x2 - bg_right) - p.x;
646 }
647 else {
648 /*Make the point relative to the indicator*/
649 new_value = p.x - (obj->coords.x1 + bg_left);
650 }
651 if(indic_w) {
652 new_value = (int32_t)(((int64_t)new_value * range + indic_w / 2) / indic_w);
653 new_value += slider->bar.min_value;
654 }
655 }
656 else {
657 const int32_t bg_top = lv_obj_get_style_pad_top_internal(obj, LV_PART_MAIN);
658 const int32_t bg_bottom = lv_obj_get_style_pad_bottom_internal(obj, LV_PART_MAIN);
659 const int32_t h = lv_obj_get_height(obj);
660 const int32_t indic_h = h - bg_bottom - bg_top;
661
662 if(is_reversed) {
663 /*Make the point relative to the indicator*/
664 new_value = p.y - (obj->coords.y1 + bg_top);
665 }
666 else {
667 /*Make the point relative to the indicator*/
668 new_value = p.y - (obj->coords.y2 + bg_bottom);
669 new_value = -new_value;
670 }
671 new_value = (int32_t)(((int64_t)new_value * range + indic_h / 2) / indic_h);
672 new_value += slider->bar.min_value;
673 }
674
675 int32_t real_max_value = slider->bar.max_value;
676 int32_t real_min_value = slider->bar.min_value;
677 /*Figure out the min. and max. for this mode*/
678 if(slider->value_to_set == &slider->bar.start_value) {
679 real_max_value = slider->bar.cur_value;
680 }
681 else {
682 real_min_value = slider->bar.start_value;
683 }
684
685 new_value = LV_CLAMP(real_min_value, new_value, real_max_value);
686 if(*slider->value_to_set != new_value) {
687 *slider->value_to_set = new_value;
688 if(is_hor)
689 lv_obj_set_scroll_chain_ver(obj, false);
690 else
691 lv_obj_set_scroll_chain_hor(obj, false);
692
693 lv_obj_invalidate(obj);
694 lv_result_t res = lv_obj_send_event(obj, LV_EVENT_VALUE_CHANGED, NULL);
695 if(res != LV_RESULT_OK)
696 return;
697 }
698}
