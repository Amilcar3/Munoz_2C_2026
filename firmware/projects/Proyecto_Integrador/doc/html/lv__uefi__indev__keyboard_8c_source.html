91lv_indev_t * lv_uefi_simple_text_input_indev_create(void)
92{

108 lv_indev_set_read_cb(indev, _simple_text_input_read_cb);

119bool lv_uefi_simple_text_input_indev_add_handle(lv_indev_t * indev, EFI_HANDLE handle)
120{
121 LV_CHECK_ARG(indev != NULL, return false);
122 EFI_SIMPLE_TEXT_INPUT_EX_PROTOCOL * interface = NULL;
123 lv_uefi_simple_text_input_handle_context_t * handle_ctx = NULL;
124
125 lv_uefi_simple_text_input_context_t * indev_ctx = (lv_uefi_simple_text_input_context_t *)lv_indev_get_user_data(indev);
126 LV_CHECK_ARG(indev_ctx != NULL, return false);
127
128 if(indev_ctx->signature != SIMPLE_TEXT_INPUT_INDEV_SIGNATURE) return false;
129
130 interface = (EFI_SIMPLE_TEXT_INPUT_EX_PROTOCOL *)lv_uefi_protocol_open(handle, &_uefi_guid_simple_text_input);
131 if(!_simple_text_input_interface_is_valid(interface)) {
132 lv_uefi_protocol_close(handle, &_uefi_guid_simple_text_input);
133 LV_LOG_WARN("[lv_uefi] The SIMPLE_TEXT_INPUT interface is not valid.");
134 return false;
135 }
136
137 handle_ctx = (lv_uefi_simple_text_input_handle_context_t *) lv_ll_ins_head(&indev_ctx->handles);
138 LV_ASSERT_MALLOC(handle_ctx);
139
140 handle_ctx->handle = handle;
141 handle_ctx->interface = interface;
142
143 return true;
144}

150void lv_uefi_simple_text_input_indev_add_all(lv_indev_t * indev)
151{
152 LV_CHECK_ARG(indev != NULL, return);
153 EFI_STATUS status;
154 EFI_HANDLE * handles = NULL;
155 UINTN no_handles;
156 UINTN index;
157
158 lv_uefi_simple_text_input_context_t * indev_ctx = (lv_uefi_simple_text_input_context_t *)lv_indev_get_user_data(indev);
159 LV_CHECK_ARG(indev_ctx != NULL, return);
160
161 if(indev_ctx->signature != SIMPLE_TEXT_INPUT_INDEV_SIGNATURE) return;
162
163 status = gLvEfiBS->LocateHandleBuffer(ByProtocol, &_uefi_guid_simple_text_input, NULL, &no_handles, &handles);
164 if(status != EFI_SUCCESS) {
165 LV_LOG_ERROR("[lv_uefi] LocateHandleBuffer(SIMPLE_TEXT_INPUT_EX) failed with error code %llx.", status);
166 return;
167 }
168
169 for(index = 0; index < no_handles; index++) {
170 lv_uefi_simple_text_input_indev_add_handle(indev, handles[index]);
171 }
172
173 if(handles != NULL) gLvEfiBS->FreePool(handles);
174}

196static void _simple_text_input_read_cb(lv_indev_t * indev, lv_indev_data_t * data)
197{
198 lv_uefi_simple_text_input_handle_context_t * handle_ctx = NULL;
199 lv_uefi_simple_text_input_key_cache_t * key_cache = NULL;
200 void * node = NULL;
201
202 lv_uefi_simple_text_input_context_t * indev_ctx = (lv_uefi_simple_text_input_context_t *)lv_indev_get_user_data(indev);
203 LV_ASSERT(indev_ctx != NULL);
204
205 /* Empty the buffer before reading new values */
206 if(lv_ll_is_empty(&indev_ctx->key_cache)) {
207 // Read from all registered devices
208 for(node = lv_ll_get_head(&indev_ctx->handles); node != NULL; node = lv_ll_get_next(&indev_ctx->handles, node)) {
209 handle_ctx = (lv_uefi_simple_text_input_handle_context_t *) node;
210 _simple_text_input_read(indev_ctx, handle_ctx);
211 }
212 }
213
214 /* Return the first value */
215 node = lv_ll_get_head(&indev_ctx->key_cache);
216 if(node != NULL) {
217 key_cache = (lv_uefi_simple_text_input_key_cache_t *)node;
218 data->state = key_cache->pressed ? LV_INDEV_STATE_PRESSED : LV_INDEV_STATE_RELEASED;
219 data->key = key_cache->key;
220 lv_ll_remove(&indev_ctx->key_cache, node);
221 lv_free(key_cache);
222 }
223
224 /* Continue reading if there are more values in the buffer */
225 data->continue_reading = !lv_ll_is_empty(&indev_ctx->key_cache);
226}

251static void _simple_text_input_read(lv_uefi_simple_text_input_context_t * indev_ctx,
252 lv_uefi_simple_text_input_handle_context_t * handle_ctx)
253{
254 EFI_STATUS status;
255 EFI_KEY_DATA state;
256 uint32_t key;
257 lv_uefi_simple_text_input_key_cache_t * cache = NULL;
258
259 LV_ASSERT(indev_ctx != NULL);
260 LV_ASSERT(handle_ctx != NULL);
261
262 status = handle_ctx->interface->ReadKeyStrokeEx(
263 handle_ctx->interface,
264 &state);
265 if(status == EFI_NOT_READY) return;
266 if(status != EFI_SUCCESS) {
267 LV_LOG_ERROR("[lv_uefi] ReadKeyStrokeEx failed.");
268 return;
269 }
270
271 key = _key_from_uefi_key(&state);
272
273 /* insert the press */
274 cache = (lv_uefi_simple_text_input_key_cache_t *) lv_ll_ins_tail(&indev_ctx->key_cache);
275 LV_ASSERT_MALLOC(cache);
276 cache->key = key;
277 cache->pressed = true;
278
279 /* insert the release */
280 cache = (lv_uefi_simple_text_input_key_cache_t *) lv_ll_ins_tail(&indev_ctx->key_cache);
281 LV_ASSERT_MALLOC(cache);
282 cache->key = key;
283 cache->pressed = false;
284}

286static uint32_t _utf8_from_unicode(UINT32 unicode)
287{
288 uint8_t bytes[4] = {0, 0, 0, 0};
289
290 /* unicode < 128 -> 1 byte */
291 if(unicode < 128) {
292 bytes[0] |= unicode;
293 }
294 /* unicode < 2048 -> 2 byte */
295 else if(unicode < 2048) {
296 bytes[0] = 0xC0;
297 bytes[0] |= unicode >> 6;
298 bytes[1] = 0x80;
299 bytes[1] |= (unicode & 0x003F);
300 }
301 /* unicode < 65536 -> 3 byte */
302 else if(unicode < 65536) {
303 bytes[0] = 0xE0;
304 bytes[0] |= unicode >> 12;
305 bytes[1] = 0x80;
306 bytes[1] |= ((unicode >> 6) & 0x003F);
307 bytes[2] = 0x80;
308 bytes[2] |= (unicode & 0x003F);
309 }
310
311 return *((uint32_t *)bytes);
312}

314static uint32_t _key_from_uefi_key(const EFI_KEY_DATA * key)
315{
316 LV_ASSERT(key != NULL);
317
318 switch(key->Key.ScanCode) {
319 case SCAN_UP:
320 return LV_KEY_UP;
321 case SCAN_DOWN:
322 return LV_KEY_DOWN;
323 case SCAN_LEFT:
324 return LV_KEY_LEFT;
325 case SCAN_RIGHT:
326 return LV_KEY_RIGHT;
327 case SCAN_DELETE:
328 return LV_KEY_DEL;
329 case SCAN_HOME:
330 return LV_KEY_HOME;
331 case SCAN_END:
332 return LV_KEY_END;
333 case SCAN_ESC:
334 return LV_KEY_ESC;
335 case SCAN_SUSPEND:
336 return LV_KEY_ENTER;
337 /* ignore all other scan codes */
338 default:
339 break;
340 }
341
342 switch(key->Key.UnicodeChar) {
343 case 0x09:
344 return (key->KeyState.KeyShiftState & EFI_SHIFT_STATE_VALID) &&
345 (key->KeyState.KeyShiftState & (EFI_RIGHT_SHIFT_PRESSED | EFI_LEFT_SHIFT_PRESSED)) ?
346 LV_KEY_PREV :
347 LV_KEY_NEXT;
348 case 0x08:
349 return LV_KEY_BACKSPACE;
350 case 0x0D:
351 return LV_KEY_ENTER;
352 case 0x18:
353 return LV_KEY_ESC;
354 default:
355 return _utf8_from_unicode(key->Key.UnicodeChar);
356 }
357}
