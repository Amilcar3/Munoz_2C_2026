104lv_obj_t * lv_msgbox_create(lv_obj_t * parent)
105{
106 LV_LOG_INFO("begin");
107 bool auto_parent = false;
108 if(parent == NULL) {
109 auto_parent = true;
110 parent = lv_obj_class_create_obj(&lv_msgbox_backdrop_class, lv_layer_top());
111 LV_ASSERT_MALLOC(parent);
112 lv_obj_class_init_obj(parent);
113 lv_obj_set_ignore_layout(parent, false);
114 lv_obj_set_size(parent, LV_PCT(100), LV_PCT(100));
115 }
116
117 lv_obj_t * obj = lv_obj_class_create_obj(&lv_msgbox_class, parent);
118 LV_ASSERT_MALLOC(obj);
119 if(obj == NULL) return NULL;
120 lv_obj_class_init_obj(obj);
121 lv_msgbox_t * mbox = (lv_msgbox_t *)obj;
122 lv_obj_set_flex_flow(obj, LV_FLEX_FLOW_COLUMN);
123
124 mbox->auto_parent = auto_parent;
125
126 mbox->content = lv_obj_class_create_obj(&lv_msgbox_content_class, obj);
127 LV_ASSERT_MALLOC(mbox->content);
128 if(mbox->content == NULL) {
129 lv_obj_delete(obj);
130 return NULL;
131 }
132 lv_obj_class_init_obj(mbox->content);
133 lv_obj_set_flex_flow(mbox->content, LV_FLEX_FLOW_COLUMN);
134 lv_obj_add_event_cb(obj, msgbox_size_changed_event_cb, LV_EVENT_SIZE_CHANGED, 0);
135
136 lv_obj_center(obj);
137 return obj;
138}

140lv_obj_t * lv_msgbox_add_title(lv_obj_t * obj, const char * title)
141{
142 LV_CHECK_OBJ(obj, MY_CLASS, return NULL);
143
144 lv_msgbox_t * mbox = (lv_msgbox_t *)obj;
145 if(mbox->header == NULL) {
146 mbox->header = lv_obj_class_create_obj(&lv_msgbox_header_class, obj);
147 LV_ASSERT_MALLOC(mbox->header);
148 if(mbox->header == NULL) return NULL;
149 lv_obj_class_init_obj(mbox->header);
150
151 lv_obj_set_size(mbox->header, lv_pct(100), lv_display_get_dpi(lv_obj_get_display(obj)) / 3);
152 lv_obj_set_flex_flow(mbox->header, LV_FLEX_FLOW_ROW);
153 lv_obj_set_flex_align(mbox->header, LV_FLEX_ALIGN_START, LV_FLEX_ALIGN_CENTER, LV_FLEX_ALIGN_CENTER);
154 lv_obj_set_scrollable(mbox->header, false);
155 lv_obj_move_to_index(mbox->header, 0);
156 }
157
158 if(mbox->title == NULL) {
159 mbox->title = lv_label_create(mbox->header);
160 lv_obj_set_flex_grow(mbox->title, 1);
161 }
162
163 lv_label_set_text(mbox->title, title);
164
165 return mbox->title;
166}
