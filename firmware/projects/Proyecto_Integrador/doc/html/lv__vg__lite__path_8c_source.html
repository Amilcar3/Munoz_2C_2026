187static void path_bounds_iter_cb(void * user_data, uint8_t op_code, const float * data, uint32_t len)
188{
189 LV_UNUSED(op_code);
190
191 if(len == 0) {
192 return;
193 }
194
195 typedef struct {
196 float x;
197 float y;
198 } point_t;
199
200 const int pt_len = sizeof(point_t) / sizeof(float);
201
202 LV_ASSERT(len % pt_len == 0);
203
204 const point_t * pt = (point_t *)data;
205 len /= pt_len;
206
207 lv_vg_lite_path_bounds_t * bounds = user_data;
208
209 for(uint32_t i = 0; i < len; i++) {
210 if(pt[i].x < bounds->min_x) bounds->min_x = pt[i].x;
211 if(pt[i].y < bounds->min_y) bounds->min_y = pt[i].y;
212 if(pt[i].x > bounds->max_x) bounds->max_x = pt[i].x;
213 if(pt[i].y > bounds->max_y) bounds->max_y = pt[i].y;
214 }
215}

217bool lv_vg_lite_path_update_bounding_box(lv_vg_lite_path_t * path)
218{
219 LV_ASSERT_NULL(path);
220
221 if(!path->format_len) {
222 return false;
223 }
224
225 LV_PROFILER_DRAW_BEGIN;
226
227 lv_vg_lite_path_bounds_t bounds;
228
229 /* init bounds */
230 bounds.min_x = FLT_MAX;
231 bounds.min_y = FLT_MAX;
232 bounds.max_x = -FLT_MAX;
233 bounds.max_y = -FLT_MAX;
234
235 /* calc bounds */
236 lv_vg_lite_path_for_each_data(lv_vg_lite_path_get_path(path), path_bounds_iter_cb, &bounds);
237
238 /* set bounds */
239 lv_vg_lite_path_set_bounding_box(path, bounds.min_x, bounds.min_y, bounds.max_x, bounds.max_y);
240
241 LV_PROFILER_DRAW_END;
242
243 return true;
244}

262void lv_vg_lite_path_reserve_space(lv_vg_lite_path_t * path, size_t len)
263{
264 bool need_reallocated = false;
265
266 /*Calculate new mem size until match the contidion*/
267 while(path->base.path_length + len > path->mem_size) {
268 if(path->mem_size == 0) {
269 path->mem_size = LV_MAX(len, PATH_MEM_SIZE_MIN);
270 }
271 else {
272 /* Increase memory size by 1.5 times */
273 path->mem_size = path->mem_size * 3 / 2;
274 }
275 need_reallocated = true;
276 }
277
278 if(!need_reallocated) {
279 return;
280 }
281
282 path->base.path = lv_realloc(path->base.path, path->mem_size);
283 LV_ASSERT_MALLOC(path->base.path);
284}

295static inline void lv_vg_lite_path_append_op(lv_vg_lite_path_t * path, uint32_t op)
296{
297 void * ptr = PATH_CURRENT_PTR(path);
298 switch(path->base.format) {
299 case VG_LITE_FP32:
300 case VG_LITE_S32:
301 LV_VG_LITE_PATH_SET_OP_CODE(ptr, uint32_t, op);
302 PATH_LENGTH_INC(path, sizeof(uint32_t));
303 break;
304 case VG_LITE_S16:
305 LV_VG_LITE_PATH_SET_OP_CODE(ptr, uint16_t, op);
306 PATH_LENGTH_INC(path, sizeof(uint16_t));
307 break;
308 case VG_LITE_S8:
309 LV_VG_LITE_PATH_SET_OP_CODE(ptr, uint8_t, op);
310 PATH_LENGTH_INC(path, sizeof(uint8_t));
311 break;
312 default:
313 LV_ASSERT_FORMAT_MSG(false, "Invalid format: %d", path->base.format);
314 break;
315 }

318static inline void lv_vg_lite_path_append_point(lv_vg_lite_path_t * path, float x, float y)
319{
320 if(path->has_transform) {
321 LV_VG_LITE_ASSERT_MATRIX(&path->matrix);
322 /* transform point */
323 float ori_x = x;
324 float ori_y = y;
325 x = ori_x * path->matrix.m[0][0] + ori_y * path->matrix.m[0][1] + path->matrix.m[0][2];
326 y = ori_x * path->matrix.m[1][0] + ori_y * path->matrix.m[1][1] + path->matrix.m[1][2];
327
328 /* perspective division, the matrix is not affine */
329 if(lv_vg_lite_matrix_has_perspective(&path->matrix)) {
330 float w = ori_x * path->matrix.m[2][0] + ori_y * path->matrix.m[2][1] + path->matrix.m[2][2];
331 if(w != 0.0f) {
332 x /= w;
333 y /= w;
334 }
335 }
336 }
337
338#define PATH_APPEND_POINT_DATA(X, Y, TYPE) \
339 do { \
340 TYPE * data = ptr; \
341 *data++ = (TYPE)(X); \
342 *data++ = (TYPE)(Y); \
343 PATH_LENGTH_INC(path, sizeof(TYPE) * 2); \
344 } while(0)
345
346 void * ptr = PATH_CURRENT_PTR(path);
347 switch(path->base.format) {
348 case VG_LITE_FP32:
349 PATH_APPEND_POINT_DATA(x, y, float);
350 break;
351 case VG_LITE_S32:
352 PATH_APPEND_POINT_DATA(x, y, int32_t);
353 break;
354 case VG_LITE_S16:
355 PATH_APPEND_POINT_DATA(x, y, int16_t);
356 break;
357 case VG_LITE_S8:
358 PATH_APPEND_POINT_DATA(x, y, int8_t);
359 break;
360 default:
361 LV_ASSERT_FORMAT_MSG(false, "Invalid format: %d", path->base.format);
362 break;
363 }
364}

427void lv_vg_lite_path_append_rect(
428 lv_vg_lite_path_t * path,
429 float x, float y,
430 float w, float h,
431 float r)
432{
433 LV_PROFILER_DRAW_BEGIN;
434 const float half_w = w / 2.0f;
435 const float half_h = h / 2.0f;
436
437 /*clamping cornerRadius by minimum size*/
438 const float r_max = LV_MIN(half_w, half_h);
439 if(r > r_max)
440 r = r_max;
441
442 /*rectangle*/
443 if(r <= 0) {
444 lv_vg_lite_path_move_to(path, x, y);
445 lv_vg_lite_path_line_to(path, x + w, y);
446 lv_vg_lite_path_line_to(path, x + w, y + h);
447 lv_vg_lite_path_line_to(path, x, y + h);
448 lv_vg_lite_path_close(path);
449 LV_PROFILER_DRAW_END;
450 return;
451 }
452
453 /*circle*/
454 if(math_equal(r, half_w) && math_equal(r, half_h)) {
455 lv_vg_lite_path_append_circle(path, x + half_w, y + half_h, r, r);
456 LV_PROFILER_DRAW_END;
457 return;
458 }
459
460 /* Get the control point offset for rounded cases */
461 const float offset = r * PATH_ARC_MAGIC;
462
463 /* Rounded rectangle case */
464 /* Starting point */
465 lv_vg_lite_path_move_to(path, x + r, y);
466
467 /* Top side */
468 lv_vg_lite_path_line_to(path, x + w - r, y);
469
470 /* Top-right corner */
471 lv_vg_lite_path_cubic_to(path, x + w - r + offset, y, x + w, y + r - offset, x + w, y + r);
472
473 /* Right side */
474 lv_vg_lite_path_line_to(path, x + w, y + h - r);
475
476 /* Bottom-right corner*/
477 lv_vg_lite_path_cubic_to(path, x + w, y + h - r + offset, x + w - r + offset, y + h, x + w - r, y + h);
478
479 /* Bottom side */
480 lv_vg_lite_path_line_to(path, x + r, y + h);
481
482 /* Bottom-left corner */
483 lv_vg_lite_path_cubic_to(path, x + r - offset, y + h, x, y + h - r + offset, x, y + h - r);
484
485 /* Left side*/
486 lv_vg_lite_path_line_to(path, x, y + r);
487
488 /* Top-left corner */
489 lv_vg_lite_path_cubic_to(path, x, y + r - offset, x + r - offset, y, x + r, y);
490
491 /* Ending point */
492 lv_vg_lite_path_close(path);
493 LV_PROFILER_DRAW_END;
494}

496void lv_vg_lite_path_append_circle(
497 lv_vg_lite_path_t * path,
498 float cx, float cy,
499 float rx, float ry)
500{
501 LV_PROFILER_DRAW_BEGIN;
502 /* https://learn.microsoft.com/zh-cn/xamarin/xamarin-forms/user-interface/graphics/skiasharp/curves/beziers */
503 float rx_kappa = rx * PATH_KAPPA;
504 float ry_kappa = ry * PATH_KAPPA;
505
506 lv_vg_lite_path_move_to(path, cx, cy - ry);
507 lv_vg_lite_path_cubic_to(path, cx + rx_kappa, cy - ry, cx + rx, cy - ry_kappa, cx + rx, cy);
508 lv_vg_lite_path_cubic_to(path, cx + rx, cy + ry_kappa, cx + rx_kappa, cy + ry, cx, cy + ry);
509 lv_vg_lite_path_cubic_to(path, cx - rx_kappa, cy + ry, cx - rx, cy + ry_kappa, cx - rx, cy);
510 lv_vg_lite_path_cubic_to(path, cx - rx, cy - ry_kappa, cx - rx_kappa, cy - ry, cx, cy - ry);
511 lv_vg_lite_path_close(path);
512 LV_PROFILER_DRAW_END;
513}

515void lv_vg_lite_path_append_arc_right_angle(lv_vg_lite_path_t * path,
516 float start_x, float start_y,
517 float center_x, float center_y,
518 float end_x, float end_y)
519{
520 LV_PROFILER_DRAW_BEGIN;
521 float dx1 = center_x - start_x;
522 float dy1 = center_y - start_y;
523 float dx2 = end_x - center_x;
524 float dy2 = end_y - center_y;
525
526 float c = SIGN(dx1 * dy2 - dx2 * dy1) * PATH_ARC_MAGIC;
527
528 lv_vg_lite_path_cubic_to(path,
529 start_x - c * dy1, start_y + c * dx1,
530 end_x - c * dy2, end_y + c * dx2,
531 end_x, end_y);
532 LV_PROFILER_DRAW_END;
533}

535void lv_vg_lite_path_append_arc(lv_vg_lite_path_t * path,
536 float cx, float cy,
537 float radius,
538 float start_angle,
539 float sweep,
540 bool pie)
541{
542 LV_PROFILER_DRAW_BEGIN;
543 /* just circle */
544 if(sweep >= 360.0f || sweep <= -360.0f) {
545 lv_vg_lite_path_append_circle(path, cx, cy, radius, radius);
546 LV_PROFILER_DRAW_END;
547 return;
548 }
549
550 start_angle = MATH_RADIANS(start_angle);
551 sweep = MATH_RADIANS(sweep);
552
553 int n_curves = (int)ceil(MATH_FABSF(sweep / MATH_HALF_PI));
554 float sweep_sign = sweep < 0 ? -1.f : 1.f;
555 float fract = fmodf(sweep, MATH_HALF_PI);
556 fract = (math_zero(fract)) ? MATH_HALF_PI * sweep_sign : fract;
557
558 /* Start from here */
559 float start_x = radius * MATH_COSF(start_angle);
560 float start_y = radius * MATH_SINF(start_angle);
561
562 if(pie) {
563 lv_vg_lite_path_move_to(path, cx, cy);
564 lv_vg_lite_path_line_to(path, start_x + cx, start_y + cy);
565 }
566
567 for(int i = 0; i < n_curves; ++i) {
568 float end_angle = start_angle + ((i != n_curves - 1) ? MATH_HALF_PI * sweep_sign : fract);
569 float end_x = radius * MATH_COSF(end_angle);
570 float end_y = radius * MATH_SINF(end_angle);
571
572 /* variables needed to calculate bezier control points */
573
577 float ax = start_x;
578 float ay = start_y;
579 float bx = end_x;
580 float by = end_y;
581 float q1 = ax * ax + ay * ay;
582 float q2 = ax * bx + ay * by + q1;
583 float k2 = (4.0f / 3.0f) * ((MATH_SQRTF(2 * q1 * q2) - q2) / (ax * by - ay * bx));
584
585 /* Next start point is the current end point */
586 start_x = end_x;
587 start_y = end_y;
588
589 end_x += cx;
590 end_y += cy;
591
592 float ctrl1_x = ax - k2 * ay + cx;
593 float ctrl1_y = ay + k2 * ax + cy;
594 float ctrl2_x = bx + k2 * by + cx;
595 float ctrl2_y = by - k2 * bx + cy;
596
597 lv_vg_lite_path_cubic_to(path, ctrl1_x, ctrl1_y, ctrl2_x, ctrl2_y, end_x, end_y);
598 start_angle = end_angle;
599 }
600
601 if(pie) {
602 lv_vg_lite_path_close(path);
603 }
604
605 LV_PROFILER_DRAW_END;
606}

608uint8_t lv_vg_lite_vlc_op_arg_len(uint8_t vlc_op)
609{
610 switch(vlc_op) {
611 VLC_OP_ARG_LEN(END, 0);
612 VLC_OP_ARG_LEN(CLOSE, 0);
613 VLC_OP_ARG_LEN(MOVE, 2);
614 VLC_OP_ARG_LEN(MOVE_REL, 2);

616 VLC_OP_ARG_LEN(LINE_REL, 2);
617 VLC_OP_ARG_LEN(QUAD, 4);
618 VLC_OP_ARG_LEN(QUAD_REL, 4);
619 VLC_OP_ARG_LEN(CUBIC, 6);
620 VLC_OP_ARG_LEN(CUBIC_REL, 6);

656void lv_vg_lite_path_for_each_data(const vg_lite_path_t * path, lv_vg_lite_path_iter_cb_t cb, void * user_data)
657{
658 LV_ASSERT_NULL(path);
659 LV_ASSERT_NULL(cb);
660 LV_PROFILER_DRAW_BEGIN;
661
662 uint8_t fmt_len = lv_vg_lite_path_format_len(path->format);
663 uint8_t * cur = path->path;
664 uint8_t * end = cur + path->path_length;
665 float tmp_data[8];
666
667 while(cur < end) {
668 /* get op code */
669 uint8_t op_code = LV_VG_LITE_PATH_GET_OP_CODE(cur);
670
671 /* get arguments length */
672 uint8_t arg_len = lv_vg_lite_vlc_op_arg_len(op_code);
673
674 /* skip op code */
675 cur += fmt_len;
676
677 /* print arguments */
678 for(uint8_t i = 0; i < arg_len; i++) {
679 switch(path->format) {
680 case VG_LITE_S8:
681 tmp_data[i] = *((int8_t *)cur);
682 break;
683 case VG_LITE_S16:
684 tmp_data[i] = *((int16_t *)cur);
685 break;
686 case VG_LITE_S32:
687 tmp_data[i] = *((int32_t *)cur);
688 break;
689 case VG_LITE_FP32:
690 tmp_data[i] = *((float *)cur);
691 break;
692 default:
693 LV_ASSERT_FORMAT_MSG(false, "Invalid format: %d", path->format);
694 break;
695 }
696
697 cur += fmt_len;
698 }
699
700 cb(user_data, op_code, tmp_data, arg_len);
701 }
702
703 LV_PROFILER_DRAW_END;
704}
