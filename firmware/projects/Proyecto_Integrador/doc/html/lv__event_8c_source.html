79lv_result_t lv_event_push_and_send(lv_event_list_t * event_list, lv_event_code_t code, void * original_target,
80 void * param)
81{
82 LV_ASSERT_NULL(event_list);
83 lv_event_t e;
84 lv_memzero(&e, sizeof(e));
85 e.code = code;
86 e.current_target = original_target;
87 e.original_target = original_target;
88 e.param = param;
89
90 lv_event_push(&e);
91 lv_result_t res = lv_event_send(event_list, &e, true);
92 if(res != LV_RESULT_OK || e.stop_processing) goto ret;
93
94 res = lv_event_send(event_list, &e, false);
95 if(res != LV_RESULT_OK) goto ret;
96
97ret:
98 lv_event_pop(&e);
99 return res;
100}

102lv_result_t lv_event_send(lv_event_list_t * list, lv_event_t * e, bool preprocess)
103{
104 if(list == NULL) return LV_RESULT_OK;
105 LV_CHECK_ARG(e != NULL, return LV_RESULT_INVALID);
106 if(e->deleted) return LV_RESULT_INVALID;
107
108 /* When obj is deleted in its own event, it will cause the `list->array` header to be released,
109 * but the content still exists, which leads to memory leakage.
110 * Therefore, back up the header in advance,
111 * which can strive to release the memory and prevent used-after-free. */
112 lv_array_t back_array_head = list->array;
113
114 /* Dealing with the problem of nested event deletion event */
115 const bool is_traversing = list->is_traversing;
116 list->is_traversing = true;
117
118 lv_result_t res = LV_RESULT_OK;
119 const uint32_t size = event_array_size(list);
120 for(uint32_t i = 0; i < size && !e->deleted; i++) {
121 lv_event_dsc_t * dsc = *event_array_at(list, i);
122 if(dsc->cb == NULL) continue;
123 if(event_is_marked_deleting(dsc)) continue;
124 const bool is_preprocessed = (dsc->filter & LV_EVENT_PREPROCESS) != 0;
125 if(is_preprocessed != preprocess) continue;
126 lv_event_code_t filter = dsc->filter & ~LV_EVENT_PREPROCESS;
127 if(filter == LV_EVENT_ALL || filter == e->code) {
128 e->user_data = dsc->user_data;
129#if LV_USE_EXT_DATA
130 e->ext_data.data = dsc->ext_data.data;
131#endif
132 dsc->cb(e);
133 if(e->stop_processing) break;
134
135 /*Stop if the object is deleted*/
136 if(e->deleted) {
137 res = LV_RESULT_INVALID;
138 break;
139 }
140 }
141 }
142
143 if(is_traversing) return res;
144
145 if(e->deleted) cleanup_event_list_core(&back_array_head);
146 else {
147 list->is_traversing = false;
148 cleanup_event_list(list);
149 }
150
151 return res;
152}

154lv_event_dsc_t * lv_event_add(lv_event_list_t * list, lv_event_cb_t cb, lv_event_code_t filter,
155 void * user_data)
156{
157 LV_CHECK_ARG(list != NULL, return NULL);
158 LV_CHECK_ARG(cb != NULL, return NULL);
159 lv_event_dsc_t * dsc = lv_malloc(sizeof(lv_event_dsc_t));
160 LV_ASSERT_NULL(dsc);
161 if(!dsc) {
162 return NULL;
163 }
164
165 dsc->cb = cb;
166 dsc->filter = filter;
167 dsc->user_data = user_data;
168#if LV_USE_EXT_DATA
169 dsc->ext_data.free_cb = NULL;
170 dsc->ext_data.data = NULL;
171#endif
172 lv_result_t res;
173 if(event_array_size(list) == 0) {
174 /*event list hasn't been initialized.*/
175 res = lv_array_init(&list->array, 1, sizeof(lv_event_dsc_t *));
176 if(res != LV_RESULT_OK) {
177 lv_free(dsc);
178 return NULL;
179 }
180 }
181
182 res = lv_array_push_back(&list->array, &dsc);
183 if(res != LV_RESULT_OK) {
184 lv_free(dsc);
185 return NULL;
186 }
187
188 return dsc;
189}

191bool lv_event_remove_dsc(lv_event_list_t * list, lv_event_dsc_t * dsc)
192{
193 LV_CHECK_ARG(list != NULL, return false);
194 LV_CHECK_ARG(dsc != NULL, return false);
195
196 const uint32_t size = event_array_size(list);
197 for(uint32_t i = 0; i < size; i++) {
198 lv_event_dsc_t * event = *event_array_at(list, i);
199 if(event == dsc) {
200#if LV_USE_EXT_DATA
201 if(dsc->ext_data.free_cb) {
202 dsc->ext_data.free_cb(dsc->ext_data.data);
203 dsc->ext_data.data = NULL;
204 }
205#endif
206 event_mark_deleting(list, event);
207 cleanup_event_list(list);
208 return true;
209 }
210 }
211
212 return false;
213}

475static void cleanup_event_list_core(lv_array_t * array)
476{
477 const uint32_t size = lv_array_size(array);
478 uint32_t kept_count = 0;
479 for(uint32_t i = 0; i < size; i++) {
480 lv_event_dsc_t ** dsc_i = lv_array_at(array, i);
481 lv_event_dsc_t ** dsc_kept = lv_array_at(array, kept_count);
482 if(event_is_marked_deleting(*dsc_i)) lv_free(*dsc_i);
483 else {
484 *dsc_kept = *dsc_i;
485 kept_count++;
486 }
487 }
488
489 if(kept_count == 0) lv_array_deinit(array);
490 else lv_array_resize(array, kept_count);
491}

493static void cleanup_event_list(lv_event_list_t * list)
