129 const char * ref_img_path)
130{
131 lv_obj_t * canvas = canvas_create(240, 240);
132
133 lv_font_t * font_normal = lv_freetype_font_create("./src/test_files/fonts/noto/NotoSansSC-Regular.ttf",

146 lv_area_set(&clip_area, 40, 40, 200 - 1, 200 - 1);
147 layer._clip_area = clip_area;
148
149 const int32_t offset_x = 40;
150 const int32_t offset_y = 70;
151
152 for(int i = 0; i < 9; i++) {

154 }

156 lv_draw_border_dsc_t draw_border_dsc;
157 lv_draw_border_dsc_init(&draw_border_dsc);
158 draw_border_dsc.width = 1;
159 draw_border_dsc.color = lv_color_black();
160 lv_draw_border(&layer, &draw_border_dsc, &clip_area);
161
162 lv_canvas_finish_layer(canvas, &layer);
163
164 TEST_ASSERT_EQUAL_LETTER_SCREENSHOT(ref_img_path);
165
166 lv_obj_delete(canvas);
167 lv_freetype_font_delete(font_normal);
168}

204void test_draw_letter_image_format_pivot(void)
205{
206 /*
207 * Regression test: vg_lite IMAGE format path was missing pivot in
208 * lv_draw_image_dsc_t, causing image glyphs to rotate around (0,0).
209 * Use screenshot comparison to detect pivot regression.
210 */
211 /* Create a properly aligned 20x20 red draw buf */
212 lv_draw_buf_t * dbuf = lv_draw_buf_create(20, 20, LV_COLOR_FORMAT_ARGB8888, LV_STRIDE_AUTO);
213 TEST_ASSERT_NOT_NULL(dbuf);
214 lv_draw_buf_clear(dbuf, NULL);
215 uint32_t stride = dbuf->header.stride;
216 for(int y = 0; y < 20; y++) {
217 uint8_t * row = dbuf->data + y * stride;
218 for(int x = 0; x < 20; x++) {
219 row[x * 4 + 0] = 0x00; /* B */
220 row[x * 4 + 1] = 0x00; /* G */
221 row[x * 4 + 2] = 0xFF; /* R */
222 row[x * 4 + 3] = 0xFF; /* A */
223 }
224 }
225
226 lv_font_t * imgfont = lv_imgfont_create(20, test_imgfont_path_cb, dbuf);
227 TEST_ASSERT_NOT_NULL(imgfont);
228
229 lv_obj_t * canvas = canvas_create(500, 360);
230 /* Fill canvas white so anti-aliasing edges blend cleanly */
231 lv_draw_buf_t * canvas_buf = lv_canvas_get_draw_buf(canvas);
232 lv_memset(canvas_buf->data, 0xFF, canvas_buf->header.stride * canvas_buf->header.h);
233
234 lv_layer_t layer;
235 lv_canvas_init_layer(canvas, &layer);
236
237 lv_draw_letter_dsc_t dsc;
238 lv_draw_letter_dsc_init(&dsc);
239 dsc.font = imgfont;
240 dsc.color = lv_color_hex(0xFFFFFF);
241 dsc.unicode = 'X';
242
243 /* Draw at multiple rotations to make pivot error clearly visible */
244 const int32_t cx = 250, cy = 180, r = 60;
245 for(int i = 0; i < 8; i++) {
246 int32_t angle_deg = i * 45;
247 int32_t x = cx + (lv_trigo_cos(angle_deg) * r / 32767);
248 int32_t y = cy + (lv_trigo_sin(angle_deg) * r / 32767);
249 dsc.rotation = (angle_deg + 90) * 10;
250 lv_draw_letter(&layer, &dsc, &(lv_point_t) {
251 .x = x, .y = y
252 });
253 }
254
255 lv_canvas_finish_layer(canvas, &layer);
256
257 TEST_ASSERT_EQUAL_SCREENSHOT("draw/letter_image_pivot.png");
258
259 lv_draw_buf_destroy(dbuf);
260 lv_imgfont_destroy(imgfont);
261}
