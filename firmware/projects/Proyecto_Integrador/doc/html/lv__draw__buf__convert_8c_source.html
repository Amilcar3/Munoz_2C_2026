57lv_result_t lv_draw_buf_convert_premultiply(lv_draw_buf_t * draw_buf)
58{
59 LV_PROFILER_DRAW_BEGIN;
60 LV_ASSERT_NULL(draw_buf);
61
62 /*Premultiply color with alpha, do case by case by judging color format*/
63 lv_color_format_t cf = draw_buf->header.cf;
64 if(LV_COLOR_FORMAT_IS_INDEXED(cf)) {
65 if(LV_RESULT_INVALID == LV_DRAW_CONVERT_PREMULTIPLY_INDEXED(draw_buf)) {
66 int size = LV_COLOR_INDEXED_PALETTE_SIZE(cf);
67 lv_color32_t * palette = (lv_color32_t *)draw_buf->data;
68 for(int i = 0; i < size; i++) {
69 lv_color_premultiply(&palette[i]);
70 }
71 }
72 }
73 else if(cf == LV_COLOR_FORMAT_ARGB8888) {
74 if(LV_RESULT_INVALID == LV_DRAW_CONVERT_PREMULTIPLY_ARGB8888(draw_buf)) {
75 uint32_t h = draw_buf->header.h;
76 uint32_t w = draw_buf->header.w;
77 uint32_t stride = draw_buf->header.stride;
78 uint8_t * line = (uint8_t *)draw_buf->data;
79 for(uint32_t y = 0; y < h; y++) {
80 lv_color32_t * pixel = (lv_color32_t *)line;
81 for(uint32_t x = 0; x < w; x++) {
82 lv_color_premultiply(pixel);
83 pixel++;
84 }
85 line += stride;
86 }
87 }
88 }
89 else if(cf == LV_COLOR_FORMAT_RGB565A8) {
90 if(LV_RESULT_INVALID == LV_DRAW_CONVERT_PREMULTIPLY_RGB565A8(draw_buf)) {
91 uint32_t h = draw_buf->header.h;
92 uint32_t w = draw_buf->header.w;
93 uint32_t stride = draw_buf->header.stride;
94 uint32_t alpha_stride = stride / 2;
95 uint8_t * line = (uint8_t *)draw_buf->data;
96 lv_opa_t * alpha = (lv_opa_t *)(line + stride * h);
97 for(uint32_t y = 0; y < h; y++) {
98 lv_color16_t * pixel = (lv_color16_t *)line;
99 for(uint32_t x = 0; x < w; x++) {
100 lv_color16_premultiply(pixel, alpha[x]);
101 pixel++;
102 }
103 line += stride;
104 alpha += alpha_stride;
105 }
106 }
107 }
108 else if(cf == LV_COLOR_FORMAT_ARGB8565) {
109 if(LV_RESULT_INVALID == LV_DRAW_CONVERT_PREMULTIPLY_ARGB8565(draw_buf)) {
110 uint32_t h = draw_buf->header.h;
111 uint32_t w = draw_buf->header.w;
112 uint32_t stride = draw_buf->header.stride;
113 uint8_t * line = (uint8_t *)draw_buf->data;
114 for(uint32_t y = 0; y < h; y++) {
115 uint8_t * pixel = line;
116 for(uint32_t x = 0; x < w; x++) {
117 uint8_t alpha = pixel[2];
118 lv_color16_premultiply((lv_color16_t *)pixel, alpha);
119 pixel += 3;
120 }
121 line += stride;
122 }
123 }
124 }
125 else {
126 LV_LOG_WARN("color format: %d not supported for premultiply", cf);
127 LV_PROFILER_DRAW_END;
128 return LV_RESULT_INVALID;
129 }
130
131 LV_PROFILER_DRAW_END;
132 return LV_RESULT_OK;
133}
