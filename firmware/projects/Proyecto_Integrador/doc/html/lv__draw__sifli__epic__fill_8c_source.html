46void lv_draw_sifli_epic_fill(lv_draw_task_t * task)
47{
48 const lv_draw_fill_dsc_t * dsc = (const lv_draw_fill_dsc_t *)task->draw_dsc;
49 const lv_area_t * coords = &task->area;
50
51 if(dsc->opa <= LV_OPA_MIN) {
52 return;
53 }
54
55 lv_layer_t * layer = task->target_layer;
56
57 lv_area_t blend_area;
58 if(!lv_area_intersect(&blend_area, coords, &task->clip_area)) {
59 return; /* Fully clipped */
60 }
61
62 if(!lv_area_intersect(&blend_area, &blend_area, &layer->buf_area)) {
63 return; /* Fully clipped */
64 }
65
66 lv_grad_dir_t grad_dir = dsc->grad.dir;
67 if(grad_dir == LV_GRAD_DIR_NONE) {
68 /* Solid color fill */
69 EPIC_LayerConfigTypeDef input_layers[1];
70 EPIC_LayerConfigTypeDef output_canvas;
71
72 if(lv_epic_setup_layers(&input_layers[0], &output_canvas, task, coords)) {
73 return; /* Fully clipped */
74 }
75
76 /* Set fill color */
77 output_canvas.color_r = dsc->color.red;
78 output_canvas.color_g = dsc->color.green;
79 output_canvas.color_b = dsc->color.blue;
80 output_canvas.color_en = true;
81 if(dsc->opa < LV_OPA_MAX) {
82 /* Alpha blending with background */
83 input_layers[0].alpha = (dsc->opa == 0) ? 255 : (256 - dsc->opa);
84 (void)lv_epic_fill(input_layers, 1, &output_canvas);
85 }
86 else {
87 /* Opaque fill */
88 (void)lv_epic_fill(NULL, 0, &output_canvas);
89 }
90 }
91 else {
92 /* Gradient fill */
93 lv_color_format_t dest_cf = layer->color_format;
94 uint32_t epic_cf = lv_img_cf_to_epic_cf(dest_cf);
95
96 LV_ASSERT_MSG(EPIC_SUPPROT_OUT_FORMAT(epic_cf),
97 "EPIC: Color format not supported for gradient fill");
98
99 uint8_t * dest_buf = lv_draw_layer_go_to_xy(layer,
100 blend_area.x1 - layer->buf_area.x1,
101 blend_area.y1 - layer->buf_area.y1);
102
103 EPIC_GradCfgTypeDef param;
104 HAL_EPIC_FillGradDataInit(&param);
105
106 param.start = dest_buf;
107 param.color_mode = epic_cf;
108 param.width = lv_area_get_width(&blend_area);
109 param.height = lv_area_get_height(&blend_area);
110 param.total_width = layer->draw_buf ? lv_epic_stride_to_width(layer->draw_buf->header.stride,
111 dest_cf) : lv_area_get_width(&layer->buf_area);
112 /* Setup gradient colors */
113 if(grad_dir == LV_GRAD_DIR_VER) {
114 /* Vertical gradient */
115 LV_ASSERT(dsc->grad.stops_count == 2);
116 EPIC_ColorDef color1 = lv_color_to_epic_color(
117 dsc->grad.stops[0].color,
118 LV_OPA_MIX2(dsc->grad.stops[0].opa, dsc->opa));
119 EPIC_ColorDef color2 = lv_color_to_epic_color(
120 dsc->grad.stops[1].color,
121 LV_OPA_MIX2(dsc->grad.stops[1].opa, dsc->opa));
122
123 param.color[0][0] = color1;
124 param.color[0][1] = color1;
125 param.color[1][0] = color2;
126 param.color[1][1] = color2;
127 }
128 else if(grad_dir == LV_GRAD_DIR_HOR) {
129 /* Horizontal gradient */
130 LV_ASSERT(dsc->grad.stops_count == 2);
131 EPIC_ColorDef color1 = lv_color_to_epic_color(
132 dsc->grad.stops[0].color,
133 LV_OPA_MIX2(dsc->grad.stops[0].opa, dsc->opa));
134 EPIC_ColorDef color2 = lv_color_to_epic_color(
135 dsc->grad.stops[1].color,
136 LV_OPA_MIX2(dsc->grad.stops[1].opa, dsc->opa));
137
138 param.color[0][0] = color1;
139 param.color[0][1] = color2;
140 param.color[1][0] = color1;
141 param.color[1][1] = color2;
142 }
143
144 (void)lv_epic_fill_grad(&param);
145 }
146}
