42void lv_draw_sw_triangle(lv_draw_task_t * t, const lv_draw_triangle_dsc_t * dsc)
43{
44#if LV_DRAW_SW_COMPLEX
45 lv_area_t tri_area;
46 tri_area.x1 = (int32_t)LV_MIN3(dsc->p[0].x, dsc->p[1].x, dsc->p[2].x);
47 tri_area.y1 = (int32_t)LV_MIN3(dsc->p[0].y, dsc->p[1].y, dsc->p[2].y);
48 tri_area.x2 = (int32_t)LV_MAX3(dsc->p[0].x, dsc->p[1].x, dsc->p[2].x);
49 tri_area.y2 = (int32_t)LV_MAX3(dsc->p[0].y, dsc->p[1].y, dsc->p[2].y);
50
51 bool is_common;
52 lv_area_t draw_area;
53 is_common = lv_area_intersect(&draw_area, &tri_area, &t->clip_area);
54 if(!is_common) return;
55
56 lv_point_t p[3];
57 /*If there is a vertical side use it as p[0] and p[1]*/
58 if(dsc->p[0].x == dsc->p[1].x) {
59 p[0] = lv_point_from_precise(&dsc->p[0]);
60 p[1] = lv_point_from_precise(&dsc->p[1]);
61 p[2] = lv_point_from_precise(&dsc->p[2]);
62 }
63 else if(dsc->p[0].x == dsc->p[2].x) {
64 p[0] = lv_point_from_precise(&dsc->p[0]);
65 p[1] = lv_point_from_precise(&dsc->p[2]);
66 p[2] = lv_point_from_precise(&dsc->p[1]);
67 }
68 else if(dsc->p[1].x == dsc->p[2].x) {
69 p[0] = lv_point_from_precise(&dsc->p[1]);
70 p[1] = lv_point_from_precise(&dsc->p[2]);
71 p[2] = lv_point_from_precise(&dsc->p[0]);
72 }
73 else {
74 p[0] = lv_point_from_precise(&dsc->p[0]);
75 p[1] = lv_point_from_precise(&dsc->p[1]);
76 p[2] = lv_point_from_precise(&dsc->p[2]);
77
78 /*Set the smallest y as p[0]*/
79 if(p[0].y > p[1].y) lv_point_swap(&p[0], &p[1]);
80 if(p[0].y > p[2].y) lv_point_swap(&p[0], &p[2]);
81
82 /*Set the greatest y as p[1]*/
83 if(p[1].y < p[2].y) lv_point_swap(&p[1], &p[2]);
84 }
85
86 /*Be sure p[0] is on top followed by lowest point (p[1]) and middle point last (p[2])*/
87 if(p[0].y > p[2].y) lv_point_swap(&p[0], &p[2]);
88 if(p[0].y > p[1].y) lv_point_swap(&p[0], &p[1]);
89 if(p[1].y < p[2].y) lv_point_swap(&p[1], &p[2]);
90
91 /*If right == true p[2] is on the right side of the p[0] p[1] line*/
92 bool right = ((p[1].x - p[0].x) * (p[2].y - p[0].y) - (p[1].y - p[0].y) * (p[2].x - p[0].x)) < 0;
93
94 void * masks[4] = {0};
95 lv_draw_sw_mask_line_param_t mask_left;
96 lv_draw_sw_mask_line_param_t mask_right;
97 lv_draw_sw_mask_line_param_t mask_bottom;
98
99 lv_draw_sw_mask_line_points_init(&mask_left, p[0].x, p[0].y,
100 p[1].x, p[1].y,
101 right ? LV_DRAW_SW_MASK_LINE_SIDE_RIGHT : LV_DRAW_SW_MASK_LINE_SIDE_LEFT);
102
103 lv_draw_sw_mask_line_points_init(&mask_right, p[0].x, p[0].y,
104 p[2].x, p[2].y,
105 right ? LV_DRAW_SW_MASK_LINE_SIDE_LEFT : LV_DRAW_SW_MASK_LINE_SIDE_RIGHT);
106
107 if(p[1].y == p[2].y) {
108 lv_draw_sw_mask_line_points_init(&mask_bottom, p[1].x, p[1].y,
109 p[2].x, p[2].y, LV_DRAW_SW_MASK_LINE_SIDE_TOP);
110 }
111 else {
112 lv_draw_sw_mask_line_points_init(&mask_bottom, p[1].x, p[1].y,
113 p[2].x, p[2].y,
114 right ? LV_DRAW_SW_MASK_LINE_SIDE_LEFT : LV_DRAW_SW_MASK_LINE_SIDE_RIGHT);
115 }
116
117 masks[0] = &mask_left;
118 masks[1] = &mask_right;
119 masks[2] = &mask_bottom;
120 int32_t area_w = lv_area_get_width(&draw_area);
121 lv_opa_t * mask_buf = lv_malloc(area_w);
122
123 lv_area_t blend_area = draw_area;
124 blend_area.y2 = blend_area.y1;
125 lv_draw_sw_blend_dsc_t blend_dsc;
126 blend_dsc.color = dsc->color;
127 blend_dsc.opa = dsc->opa;
128 blend_dsc.mask_buf = mask_buf;
129 blend_dsc.blend_area = &blend_area;
130 blend_dsc.mask_area = &blend_area;
131 blend_dsc.mask_stride = 0;
132 blend_dsc.blend_mode = LV_BLEND_MODE_NORMAL;
133 blend_dsc.src_stride = 0;
134 blend_dsc.src_buf = NULL;
135
136 lv_grad_dir_t grad_dir = dsc->grad.dir;
137
138 lv_draw_sw_grad_calc_t * grad = lv_draw_sw_grad_get(&dsc->grad, lv_area_get_width(&tri_area),
139 lv_area_get_height(&tri_area));
140 lv_opa_t * grad_opa_map = NULL;
141 if(grad && grad_dir == LV_GRAD_DIR_HOR) {
142 blend_dsc.src_area = &blend_area;
143 blend_dsc.src_buf = grad->color_map + draw_area.x1 - tri_area.x1;
144 grad_opa_map = grad->opa_map + draw_area.x1 - tri_area.x1;
145 blend_dsc.src_color_format = LV_COLOR_FORMAT_RGB888;
146 }
147
148 int32_t y;
149 for(y = draw_area.y1; y <= draw_area.y2; y++) {
150 blend_area.y1 = y;
151 blend_area.y2 = y;
152 lv_memset(mask_buf, 0xff, area_w);
153 blend_dsc.mask_res = lv_draw_sw_mask_apply(masks, mask_buf, draw_area.x1, y, area_w);
154 if(grad_dir == LV_GRAD_DIR_VER) {
155 LV_ASSERT_NULL(grad);
156 blend_dsc.color = grad->color_map[y - tri_area.y1];
157 blend_dsc.opa = grad->opa_map[y - tri_area.y1];
158 if(dsc->opa < LV_OPA_MAX) blend_dsc.opa = LV_OPA_MIX2(blend_dsc.opa, dsc->opa);
159 }
160 else if(grad_dir == LV_GRAD_DIR_HOR) {
161 if(grad_opa_map) {
162 int32_t i;
163 if(blend_dsc.mask_res == LV_DRAW_SW_MASK_RES_CHANGED) {
164 blend_dsc.mask_buf = mask_buf;
165 for(i = 0; i < area_w; i++) {
166 if(grad_opa_map[i] < LV_OPA_MAX) mask_buf[i] = LV_OPA_MIX2(mask_buf[i], grad_opa_map[i]);
167 }
168 }
169 else if(blend_dsc.mask_res == LV_DRAW_SW_MASK_RES_FULL_COVER) {
170 blend_dsc.mask_buf = grad_opa_map;
171 blend_dsc.mask_res = LV_DRAW_SW_MASK_RES_CHANGED;
172 }
173 else if(blend_dsc.mask_res == LV_DRAW_SW_MASK_RES_TRANSP) {
174 continue;
175 }
176 }
177 }
178 lv_draw_sw_blend(t, &blend_dsc);
179 }
180
181 lv_free(mask_buf);
182 lv_draw_sw_mask_free_param(&mask_bottom);
183 lv_draw_sw_mask_free_param(&mask_left);
184 lv_draw_sw_mask_free_param(&mask_right);
185
186 if(grad) {
187 lv_draw_sw_grad_cleanup(grad);
188 }
189
190#else
191 LV_UNUSED(t);
192 LV_UNUSED(dsc);
193 LV_LOG_WARN("Can't draw triangles with LV_DRAW_SW_COMPLEX == 0");
194#endif /*LV_DRAW_SW_COMPLEX*/
195}
