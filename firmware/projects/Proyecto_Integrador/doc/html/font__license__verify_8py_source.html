33def get_font_full_name(font_path: str) -> str:
34 font = TTFont(font_path)
35 name_records = font["name"].names
36 fallback = None
37 for record in name_records:
38 if record.nameID == 1: # ID 1 corresponds to the font family name and will be used if full name doesn't exists
39 fallback = record.toStr()
40 if record.nameID == 4: # ID 4 corresponds to the full font name
41 return record.toStr()
42 return fallback

58def has_intree_license(license_root_folder: str, font_name: str) -> bool:
59 if font_name in _EXCLUDED_FONTS:

62 # Prepare candidate folder names, ignoring casing

76 entry_path = os.path.join(license_root_folder, entry)

78 # Compare the directory name in lowercase with candidates

81 for item in os.listdir(entry_path):
