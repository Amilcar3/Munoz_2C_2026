53int32_t LV_ATTRIBUTE_FAST_MEM lv_trigo_sin(int16_t angle)
54{
55 int32_t ret = 0;
56 while(angle < 0) angle += 360;
57 while(angle >= 360) angle -= 360;
58
59 if(angle < 90) {
60 ret = sin0_90_table[angle];
61 }
62 else if(angle >= 90 && angle < 180) {
63 angle = 180 - angle;
64 ret = sin0_90_table[angle];
65 }
66 else if(angle >= 180 && angle < 270) {
67 angle = angle - 180;
68 ret = -sin0_90_table[angle];
69 }
70 else { /*angle >=270*/
71 angle = 360 - angle;
72 ret = -sin0_90_table[angle];
73 }
74
75 if(ret == 32767) return 32768;
76 else if(ret == -32767) return -32768;
77 else return ret;
78}

111static int32_t do_cubic_bezier(int32_t t, int32_t a, int32_t b, int32_t c)
112{
113 /*a * t^3 + b * t^2 + c * t*/
114#if CUBIC_PRECISION_BITS > 14
115 int64_t ret;
116#else
117 int32_t ret;
118#endif
119
120 ret = a;
121 ret = (ret * t) >> CUBIC_PRECISION_BITS;
122 ret = ((ret + b) * t) >> CUBIC_PRECISION_BITS;
123 ret = ((ret + c) * t) >> CUBIC_PRECISION_BITS;
124 return ret;
125}

127int32_t lv_cubic_bezier(int32_t x, int32_t x1, int32_t y1, int32_t x2, int32_t y2)
128{
129 /*x is not checked as the range is clamped below*/
130 LV_CHECK_ARG(x1 >= 0 && x1 <= LV_BEZIER_VAL_MAX, return 0);
131 LV_CHECK_ARG(x2 >= 0 && x2 <= LV_BEZIER_VAL_MAX, return 0);
132
133 int32_t ax, bx, cx, ay, by, cy;
134 int32_t tl, tr, t; /*t in cubic-bezier function, used for bisection */
135 int32_t xs; /*x sampled on curve */
136#if CUBIC_PRECISION_BITS > 14
137 int64_t d; /*slope value at specified t*/
138#else
139 int32_t d;
140#endif
141
142 if(x == 0 || x == LV_BEZIER_VAL_MAX) return x;
143
144 /* input is always LV_BEZIER_VAL_SHIFT bit precision */
145
146#if CUBIC_PRECISION_BITS != LV_BEZIER_VAL_SHIFT
147 x <<= CUBIC_PRECISION_BITS - LV_BEZIER_VAL_SHIFT;
148 x1 <<= CUBIC_PRECISION_BITS - LV_BEZIER_VAL_SHIFT;
149 x2 <<= CUBIC_PRECISION_BITS - LV_BEZIER_VAL_SHIFT;
150 y1 <<= CUBIC_PRECISION_BITS - LV_BEZIER_VAL_SHIFT;
151 y2 <<= CUBIC_PRECISION_BITS - LV_BEZIER_VAL_SHIFT;
152#endif
153
154 cx = 3 * x1;
155 bx = 3 * (x2 - x1) - cx;
156 ax = (1L << CUBIC_PRECISION_BITS) - cx - bx;
157
158 cy = 3 * y1;
159 by = 3 * (y2 - y1) - cy;
160 ay = (1L << CUBIC_PRECISION_BITS) - cy - by;
161
162 /*Try Newton's method firstly */
163 t = x; /*Make a guess*/
164 for(int i = 0; i < CUBIC_NEWTON_ITERATIONS; i++) {
165 /*Check if x on curve at t matches input x*/
166 xs = do_cubic_bezier(t, ax, bx, cx) - x;
167 if(LV_ABS(xs) <= 1) goto found;
168
169 /* get slop at t, d = 3 * ax * t^2 + 2 * bx + t + cx */
170 d = ax; /* use 64bit operation if needed. */
171 d = (3 * d * t) >> CUBIC_PRECISION_BITS;
172 d = ((d + 2 * bx) * t) >> CUBIC_PRECISION_BITS;
173 d += cx;
174
175 if(LV_ABS(d) <= 1) break;
176
177 d = ((int64_t)xs * (1L << CUBIC_PRECISION_BITS)) / d;
178 if(d == 0) break; /*Reached precision limits*/
179 t -= d;
180 }
181
182 /*Fallback to bisection method for reliability*/
183 tl = 0, tr = 1L << CUBIC_PRECISION_BITS, t = x;
184
185 if(t < tl) {
186 t = tl;
187 goto found;
188 }
189
190 if(t > tr) {
191 t = tr;
192 goto found;
193 }
194
195 while(tl < tr) {
196 xs = do_cubic_bezier(t, ax, bx, cx);
197 if(LV_ABS(xs - x) <= 1) goto found;
198 x > xs ? (tl = t) : (tr = t);
199 t = (tr - tl) / 2 + tl;
200 if(t == tl) break;
201 }
202
203 /*Failed to find suitable t for given x, return a value anyway.*/
204found:
205 /*Return y at t*/
206#if CUBIC_PRECISION_BITS != LV_BEZIER_VAL_SHIFT
207 return do_cubic_bezier(t, ay, by, cy) >> (CUBIC_PRECISION_BITS - LV_BEZIER_VAL_SHIFT);
208#else
209 return do_cubic_bezier(t, ay, by, cy);
210#endif
211}

213void LV_ATTRIBUTE_FAST_MEM lv_sqrt(uint32_t x, lv_sqrt_res_t * q, uint32_t mask)
214{
215 LV_CHECK_ARG(q != NULL, return);
216
217 x = x << 8; /*To get 4 bit precision. (sqrt(256) = 16 = 4 bit)*/
218
219 uint32_t root = 0;
220 uint32_t trial;
221 /*http://ww1.microchip.com/...en/AppNotes/91040a.pdf*/
222 do {
223 trial = root + mask;
224 if(trial * trial <= x) root = trial;
225 mask = mask >> 1;
226 } while(mask);
227
228 q->i = root >> 4;
229 q->f = (root & 0xf) << 4;
230}

304 }

306 return sqq_table[x] >> 4;

308 /* Run two iterations of the standard convergence formula */
309
310 xn = (xn + 1 + x / xn) / 2;
311nr1:
312 xn = (xn + 1 + x / xn) / 2;
313adj:
314
315 if(xn * xn > (int32_t)x) /* Correct rounding if necessary */
316 xn--;
317
318 return xn;

321uint16_t lv_atan2(int x, int y)
322{
334 unsigned char negflag;
335 unsigned char tempdegree;
336 unsigned char comp;
337 unsigned int degree; /*this will hold the result*/
338 unsigned int ux;
339 unsigned int uy;
340
341 /*Save the sign flags then remove signs and get XY as unsigned ints*/
342 negflag = 0;
343 if(x < 0) {
344 negflag += 0x01; /*x flag bit*/
345 x = (0 - x); /*is now +*/
346 }
347 ux = x; /*copy to unsigned var before multiply*/
348 if(y < 0) {
349 negflag += 0x02; /*y flag bit*/
350 y = (0 - y); /*is now +*/
351 }
352 uy = y; /*copy to unsigned var before multiply*/
353
354 /*1. Calc the scaled "degrees"*/
355 if(ux > uy) {
356 degree = (uy * 45) / ux; /*degree result will be 0-45 range*/
357 negflag += 0x10; /*octant flag bit*/
358 }
359 else {
360 degree = (ux * 45) / uy; /*degree result will be 0-45 range*/
361 }
362
363 /*2. Compensate for the 4 degree error curve*/
364 comp = 0;
365 tempdegree = degree; /*use an unsigned char for speed!*/
366 if(tempdegree > 22) { /*if top half of range*/
367 if(tempdegree <= 44) comp++;
368 if(tempdegree <= 41) comp++;
369 if(tempdegree <= 37) comp++;
370 if(tempdegree <= 32) comp++; /*max is 4 degrees compensated*/
371 }
372 else { /*else is lower half of range*/
373 if(tempdegree >= 2) comp++;
374 if(tempdegree >= 6) comp++;
375 if(tempdegree >= 10) comp++;
376 if(tempdegree >= 15) comp++; /*max is 4 degrees compensated*/
377 }
378 degree += comp; /*degree is now accurate to +/- 1 degree!*/
379
380 /*Invert degree if it was X>Y octant, makes 0-45 into 90-45*/
381 if(negflag & 0x10) degree = (90 - degree);
382
383 /*3. Degree is now 0-90 range for this quadrant,*/
384 /*need to invert it for whichever quadrant it was in*/
385 if(negflag & 0x02) { /*if -Y*/
386 if(negflag & 0x01) /*if -Y -X*/
387 degree = (180 + degree);
388 else /*else is -Y +X*/
389 degree = (180 - degree);
390 }
391 else { /*else is +Y*/
392 if(negflag & 0x01) /*if +Y -X*/
393 degree = (360 - degree);
394 }
395 return degree;
396}

398int64_t lv_pow(int64_t base, int8_t exp)
399{
400 int64_t result = 1;
401
402 /* Handle negative exponent: base^(-exp) = 1/(base^exp) */
403 /* In integer arithmetic, this is 0 for base > 1 */
404 if(exp < 0) {
405 if(base == 0) return 0; /* 0^(-n) is undefined, return 0 */
406 if(base == 1) return 1; /* 1^(-n) = 1 */
407 if(base == -1) return (exp & 1) ? -1 : 1; /* (-1)^(-n) alternates */
408 return 0; /* For all other cases, result is < 1, so 0 */
409 }
410
411 while(exp) {
412 if(exp & 1)
413 result *= base;
414 exp >>= 1;
415 base *= base;
416 }
417
418 return result;
419}

421int32_t lv_map(int32_t x, int32_t min_in, int32_t max_in, int32_t min_out, int32_t max_out)
422{
423 if(max_in == min_in) return min_out; /*Avoid division by zero later*/
424
425 if(max_in >= min_in && x >= max_in) return max_out;
426 if(max_in >= min_in && x <= min_in) return min_out;
427
428 if(max_in <= min_in && x <= max_in) return max_out;
429 if(max_in <= min_in && x >= min_in) return min_out;
430
437
438 int32_t delta_in = max_in - min_in;
439 int32_t delta_out = max_out - min_out;
440
441 return ((x - min_in) * delta_out) / delta_in + min_out;
442}

449uint32_t lv_rand(uint32_t min, uint32_t max)
450{
451 LV_CHECK_ARG(min <= max, return min);
452
453 /*Algorithm "xor" from p. 4 of Marsaglia, "Xorshift RNGs"*/
454 uint32_t x = rand_seed;
455 x ^= x << 13;
456 x ^= x >> 17;
457 x ^= x << 5;
458 rand_seed = x;
459
460 return (rand_seed % (max - min + 1)) + min;
461}
