40static void _gaussianExtendRegion(RenderRegion& region, int extra, int8_t direction)
41{
42 //bbox region expansion for feathering
43 if (direction != 2) {
44 region.x = -extra;
45 region.w = extra * 2;
46 }
47 if (direction != 1) {
48 region.y = -extra;
49 region.h = extra * 2;
50 }

54static int _gaussianRemap(int end, int idx, int border)
55{
56 //wrap
57 if (border == 1) return idx % end;
58
59 //duplicate
60 if (idx < 0) return 0;
61 else if (idx >= end) return end - 1;
62 return idx;
63}

67static void _gaussianBlur(uint8_t* src, uint8_t* dst, int32_t stride, int32_t w, int32_t h, const SwBBox& bbox, int32_t dimension, int border, bool flipped)
68{
69 if (flipped) {
70 src += ((bbox.min.x * stride) + bbox.min.y) << 2;
71 dst += ((bbox.min.x * stride) + bbox.min.y) << 2;
72 } else {
73 src += ((bbox.min.y * stride) + bbox.min.x) << 2;
74 dst += ((bbox.min.y * stride) + bbox.min.x) << 2;
75 }
76
77 auto iarr = 1.0f / (dimension + dimension + 1);
78
79 for (int x = 0; x < h; x++) {
80 auto p = x * stride;
81 auto i = p * 4; //current index
82 auto l = -(dimension + 1); //left index
83 auto r = dimension; //right index
84 int acc[4] = {0, 0, 0, 0}; //sliding accumulator
85
86 //initial acucmulation
87 for (int x2 = l; x2 < r; ++x2) {
88 auto id = (_gaussianRemap(w, x2, border) + p) * 4;
89 acc[0] += src[id++];
90 acc[1] += src[id++];
91 acc[2] += src[id++];
92 acc[3] += src[id];
93 }
94 //perform filtering
95 for (int x2 = 0; x2 < w; ++x2, ++r, ++l) {
96 auto rid = (_gaussianRemap(w, r, border) + p) * 4;
97 auto lid = (_gaussianRemap(w, l, border) + p) * 4;
98 acc[0] += src[rid++] - src[lid++];
99 acc[1] += src[rid++] - src[lid++];
100 acc[2] += src[rid++] - src[lid++];
101 acc[3] += src[rid] - src[lid];
102 dst[i++] = static_cast<uint8_t>(acc[0] * iarr + 0.5f);
103 dst[i++] = static_cast<uint8_t>(acc[1] * iarr + 0.5f);
104 dst[i++] = static_cast<uint8_t>(acc[2] * iarr + 0.5f);
105 dst[i++] = static_cast<uint8_t>(acc[3] * iarr + 0.5f);
106 }
107 }
108}

111static int _gaussianInit(int* kernel, float sigma, int level)
112{
113 const auto MAX_LEVEL = SwGaussianBlur::MAX_LEVEL;
114
115 //compute the kernel
116 auto wl = (int) sqrt((12 * sigma / MAX_LEVEL) + 1);
117 if (wl % 2 == 0) --wl;
118 auto wu = wl + 2;
119 auto mi = (12 * sigma - MAX_LEVEL * wl * wl - 4 * MAX_LEVEL * wl - 3 * MAX_LEVEL) / (-4 * wl - 4);
120 auto m = int(mi + 0.5f);
121 auto extends = 0;
122
123 for (int i = 0; i < level; i++) {
124 kernel[i] = ((i < m ? wl : wu) - 1) / 2;
125 extends += kernel[i];
126 }
127
128 return extends;
129}

132bool effectGaussianPrepare(RenderEffectGaussian* params)
133{
134 auto data = (SwGaussianBlur*)lv_malloc(sizeof(SwGaussianBlur));
135 LV_ASSERT_MALLOC(data);
136
137 //compute box kernel sizes
138 data->level = int(SwGaussianBlur::MAX_LEVEL * ((params->quality - 1) * 0.01f)) + 1;
139 auto extends = _gaussianInit(data->kernel, params->sigma * params->sigma, data->level);
140
141 //skip, if the parameters are invalid.
142 if (extends == 0) {
143 params->invalid = true;
144 lv_free(data);
145 return false;
146 }
147
148 _gaussianExtendRegion(params->extend, extends, params->direction);
149
150 params->rd = data;
151
152 return true;
153}

156/* It is best to take advantage of the Gaussian blur’s separable property
157 by dividing the process into two passes. horizontal and vertical.
158 We can expect fewer calculations. */
159bool effectGaussianBlur(SwImage& image, SwImage& buffer, const SwBBox& bbox, const RenderEffectGaussian* params)
160{
161 if (params->invalid) return false;
162
163 if (image.channelSize != sizeof(uint32_t)) {
164 TVGERR("SW_ENGINE", "Not supported grayscale Gaussian Blur!");
165 return false;
166 }
167
168 auto data = static_cast<SwGaussianBlur*>(params->rd);
169 auto w = (bbox.max.x - bbox.min.x);
170 auto h = (bbox.max.y - bbox.min.y);
171 auto stride = image.stride;
172 auto front = image.buf8;
173 auto back = buffer.buf8;
174 auto swapped = false;
175
176 //fine-tuning for low-quality (experimental)
177 auto threshold = (std::min(w, h) < 300) ? 2 : 1;
178
179 TVGLOG("SW_ENGINE", "GaussianFilter region(%ld, %ld, %ld, %ld) params(%f %d %d), level(%d)", bbox.min.x, bbox.min.y, bbox.max.x, bbox.max.y, params->sigma, params->direction, params->border, data->level);
180
181 //horizontal
182 if (params->direction == 0 || params->direction == 1) {
183 for (int i = 0; i < data->level; ++i) {
184 auto k = data->kernel[i] / threshold;
185 if (k == 0) continue;
186 _gaussianBlur(front, back, stride, w, h, bbox, k, params->border, false);
187 std::swap(front, back);
188 swapped = !swapped;
189 }
190 }
191
192 //vertical. x/y flipping and horionztal access is pretty compatible with the memory architecture.
193 if (params->direction == 0 || params->direction == 2) {
194 rasterXYFlip(reinterpret_cast<uint32_t*>(front), reinterpret_cast<uint32_t*>(back), stride, w, h, bbox, false);
195 std::swap(front, back);
196
197 for (int i = 0; i < data->level; ++i) {
198 auto k = data->kernel[i] / threshold;
199 if (k == 0) continue;
200 _gaussianBlur(front, back, stride, h, w, bbox, k, params->border, true);
201 std::swap(front, back);
202 swapped = !swapped;
203 }
204
205 rasterXYFlip(reinterpret_cast<uint32_t*>(front), reinterpret_cast<uint32_t*>(back), stride, h, w, bbox, true);
206 std::swap(front, back);
207 }
208
209 if (swapped) std::swap(image.buf8, buffer.buf8);
210
211 return true;
212}
