23/* Render `src` repeatedly and compare against `screenshot`. A descriptor the
24 * decoder rejects has to render as the empty image, without leaking. */
25static void bin_decoder(const void * src, const char * screenshot)
26{
27 lv_image_cache_drop(src);
28 create_image(src);
29 TEST_ASSERT_EQUAL_SCREENSHOT(screenshot);
30 lv_obj_clean(lv_screen_active());
31
32 size_t mem_before = lv_test_get_free_mem();
33 for(uint32_t i = 0; i < 20; i++) {
34 lv_obj_clean(lv_screen_active());
35 create_image(src);
36
37 lv_obj_invalidate(lv_screen_active());
38 lv_refr_now(NULL);
39 }
40 TEST_ASSERT_EQUAL_SCREENSHOT(screenshot);
41 lv_obj_clean(lv_screen_active());
42 TEST_ASSERT_MEM_LEAK_LESS_THAN(mem_before, 0);
43}

46static lv_image_dsc_t * get_image_dsc(void)
47{
48#define IMAGE_WIDTH 32
49#define IMAGE_HEIGHT 2
50 static LV_ATTRIBUTE_MEM_ALIGN uint8_t image_map[IMAGE_WIDTH * IMAGE_HEIGHT * sizeof(lv_color32_t)] = { 0 };
51 static lv_image_dsc_t image_dsc = { 0 };
52
53 image_dsc.header.magic = LV_IMAGE_HEADER_MAGIC,
54 image_dsc.header.cf = LV_COLOR_FORMAT_ARGB8888;
55 image_dsc.header.w = IMAGE_WIDTH;
56 image_dsc.header.h = IMAGE_HEIGHT;
57 image_dsc.header.stride = IMAGE_WIDTH * sizeof(lv_color32_t);
58 image_dsc.header.flags = 0;
59 image_dsc.data_size = sizeof(image_map);
60 image_dsc.data = image_map;
61
62 return &image_dsc;
63}

190void test_bin_decoder_get_area_outside_image(void)
191{
192 const char * src = "A:src/test_files/binimages/cogwheel.RGB565.bin";
193 lv_image_cache_drop(src);
194
195 const lv_image_decoder_args_t args = { .no_cache = true };
196 lv_image_decoder_dsc_t dsc;
197 TEST_ASSERT_EQUAL(LV_RESULT_OK, lv_image_decoder_open(&dsc, src, &args));
198
199 int32_t w = dsc.header.w;
200 int32_t h = dsc.header.h;
201
202 /*A transformed image is drawn on a larger area than the image itself. Such an area
203 *cannot be decoded line by line, so it must be rejected instead of reading out of bounds.*/
204 const lv_area_t outside[] = {
205 {-1, 0, w - 1, 0},
206 {0, -1, w - 1, 0},
207 {0, 0, w, 0},
208 {0, 0, w - 1, h},
209 {-4, -4, w + 3, h + 3},
210 };
211
212 for(uint32_t i = 0; i < sizeof(outside) / sizeof(outside[0]); i++) {
213 lv_area_t decoded_area = {LV_COORD_MIN, LV_COORD_MIN, LV_COORD_MIN, LV_COORD_MIN};
214 TEST_ASSERT_EQUAL(LV_RESULT_INVALID, lv_image_decoder_get_area(&dsc, &outside[i], &decoded_area));
215 }
216
217 /*The full image is a valid area*/
218 const lv_area_t inside = {0, 0, w - 1, h - 1};
219 lv_area_t decoded_area = {LV_COORD_MIN, LV_COORD_MIN, LV_COORD_MIN, LV_COORD_MIN};
220 lv_result_t res = lv_image_decoder_get_area(&dsc, &inside, &decoded_area);
221#if LV_BIN_DECODER_RAM_LOAD == 0
222 /*Read line by line*/
223 TEST_ASSERT_EQUAL(LV_RESULT_OK, res);
224 TEST_ASSERT_EQUAL(0, decoded_area.y1);
225 TEST_ASSERT_EQUAL(0, decoded_area.y2);
226 TEST_ASSERT_EQUAL(0, decoded_area.x1);
227 TEST_ASSERT_EQUAL(w - 1, decoded_area.x2);
228#else
229 LV_UNUSED(res); /*The whole image was decoded on open, no need to read it here*/
230#endif
231
232 lv_image_decoder_close(&dsc);
233}
