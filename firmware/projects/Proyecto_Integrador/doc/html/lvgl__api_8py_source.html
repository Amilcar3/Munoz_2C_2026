70def _split_args(argsstring: str) -> list[str] | None:
71 """Split Doxygen's normalised `(a, b, c)` into its parameters."""
72 s = (argsstring or "").strip()
73 if not s.startswith("("):
74 return None
75 end = s.rfind(")")
76 if end == -1:
77 return None
78 parts, cur, depth = [], "", 0
79 for ch in s[1:end]:
80 if ch in "([{":
81 depth += 1
82 elif ch in ")]}":
83 depth -= 1
84 if ch == "," and depth == 0:
85 parts.append(cur)
86 cur = ""
87 else:
88 cur += ch
89 parts.append(cur)
90 return parts

93def _name_of_arg(arg: str) -> str | None:
94 """The identifier a Doxygen-normalised parameter declares, if any."""
95 a = arg.strip()
96 if not a or a in ("void", "..."):
97 return None
98 fp = re.search(r"\‍(\s*\*+\s*(\w+)\s*\‍)", a) # function pointer
99 if fp:
100 return fp.group(1)
101 a = re.sub(r"\‍[[^\‍]]*\‍]\s*$", "", a).strip() # trailing []
102 m = re.search(r"(\w+)\s*$", a)
103 return m.group(1) if m else None

269 @classmethod
270 def _from_xml(cls, xml_dir: Path, repo_root: Path) -> "PublicApi":
271 api = cls()
272 root = repo_root.resolve()
273 if not xml_dir.is_dir():
274 raise RuntimeError(f"not a Doxygen XML directory: {xml_dir}")
275
276 paths = [
277 path
278 for path in sorted(xml_dir.glob("*.xml"))
279 if path.name not in ("index.xml", "Doxyfile.xml")
280 ]
281 if not paths:
282 raise RuntimeError(f"no Doxygen XML files in {xml_dir}")
283
284 for path in paths:
285 try:
286 tree = ET.parse(path)
287 except ET.ParseError as e:
288 raise RuntimeError(f"{path}: malformed Doxygen XML: {e}") from e
289 for member in tree.iter("memberdef"):
290 kind = member.get("kind")
291 if kind == "function":
292 func = _function_from_memberdef(member, root)
293 if func is not None:
294 api.functions[func.name] = _merge(
295 api.functions.get(func.name), func
296 )
297 elif kind == "enum":
298 en = _enum_from_memberdef(member, root)
299 if en is not None:
300 api.enums.setdefault(en.name, en)
301 elif kind == "typedef":
302 td = _typedef_from_memberdef(member, root)
303 if td is not None:
304 api.typedefs.setdefault(td.name, td)
305
306 for compound in tree.iter("compounddef"):
307 st = _struct_from_compounddef(compound, root)
308 if st is not None:
309 api.structs.setdefault(st.name, st)
310 return api

313def _relative(path: str | None, root: Path) -> str | None:
314 """Normalise a Doxygen path to one relative to `root`.
315
316 STRIP_FROM_PATH makes Doxygen emit paths already relative to the tree, so
317 they must be resolved against `root` -- not against the process working
318 directory, which is a different tree when auditing a worktree.
319 """
320 if not path:
321 return None
322 p = Path(path)
323 if not p.is_absolute():
324 p = root / p
325 try:
326 return os.path.relpath(p, root).replace(os.sep, "/")
327 except ValueError:
328 return str(path)

331def _function_from_memberdef(member, root: Path) -> Function | None:
332 name = member.findtext("name")
333 if not name:
334 return None
335 loc = member.find("location")
336 if loc is None:
337 return None
338
339 arg_parts = _split_args(member.findtext("argsstring") or "")
340 decl_params = member.findall("param")
341 docs = _param_docs(member)
342
343 params: list[Param] = []
344 for i, p in enumerate(decl_params):
345 ptype = _text(p.find("type"))
346 pname = p.findtext("declname") or p.findtext("defname")
347 if arg_parts and i < len(arg_parts):
348 from_args = _name_of_arg(arg_parts[i])
349 if from_args:
350 pname = from_args
351 if p.find("array") is not None or re.search(
352 r"\‍[[^\‍]]*\‍]\s*$", arg_parts[i].strip()
353 ):
354 ptype += "[]"
355 params.append(Param(name=pname, type=ptype, doc=docs.get(pname or "", "")))
356
357 declfile = _relative(loc.get("declfile") or loc.get("file"), root)
358 declline = loc.get("declline") or loc.get("line")
359 file_attr = loc.get("file")
360 bodyfile = loc.get("bodyfile")
361 return Function(
362 name=name,
363 return_type=_text(member.find("type")),
364 params=params,
365 header=declfile if (declfile or "").endswith((".h", ".hpp")) else None,
366 header_line=int(declline) if declline and declline.isdigit() else None,
367 body_file=_relative(loc.get("bodyfile"), root),
368 body_start=(
369 int(loc.get("bodystart"))
370 if (loc.get("bodystart") or "").isdigit()
371 else None
372 ),
373 body_end=(
374 int(loc.get("bodyend")) if (loc.get("bodyend") or "").isdigit() else None
375 ),
376 brief=_text(member.find("briefdescription")),
377 detail=_text(member.find("detaileddescription")),
378 from_definition=bool(file_attr and bodyfile and file_attr == bodyfile),
379 )

390def _enum_from_memberdef(member, root: Path) -> Enum | None:
391 name = member.findtext("name")
392 if not name or name.startswith("@"): # anonymous
393 return None

401 return Enum(name=name, members=members, header=_header_of(member, root))

415def _struct_from_compounddef(compound, root: Path) -> Struct | None:
416 kind = compound.get("kind")
417 if kind not in ("struct", "union"):
418 return None
419 name = compound.findtext("compoundname")
420 if not name or "@" in name:
421 return None
422 loc = compound.find("location")
423 header = None
424 if loc is not None:
425 path = _relative(loc.get("file"), root)
426 header = path if (path or "").endswith((".h", ".hpp")) else None
427 fields = []
428 for m in compound.iter("memberdef"):
429 if m.get("kind") != "variable":
430 continue
431 fname = m.findtext("name")
432 if not fname:
433 continue
434 ftype = _text(m.find("type")) + (m.findtext("argsstring") or "")
435 fields.append(Field(name=fname, type=" ".join(ftype.split())))
436 return Struct(name=name, kind=kind, fields=fields, header=header)

439def _combine_docs(have: str, extra: str) -> str:
440 """Fold `extra` into `have`, keeping whichever text says more."""
441 if not extra or extra in have:
442 return have
443 if have and have in extra:

448def _param_docs(member) -> dict[str, str]:
449 """Per-parameter documentation, keyed by the name used in the comment."""
450 out: dict[str, str] = {}
451 detail = member.find("detaileddescription")
452 if detail is None:
453 return out
454 for item in detail.iter("parameteritem"):
455 desc = _text(item.find("parameterdescription"))
456 for nm in item.iter("parametername"):
457 name = _text(nm)
458 if name:
459 out[name] = _combine_docs(out.get(name, ""), desc)
460 return out

463def _merge(prev: Function | None, new: Function) -> Function:
464 """Doxygen emits a function once per file that mentions it; combine them.
465
466 The declaration's entry knows the public header and carries the Doxygen
467 comment; the definition's entry carries the parameter names the body uses.
468 Take each from whichever entry actually has it.
469 """
470 if prev is None:
471 return new
472
473 # the public header always wins over a private one
474 def _better_header(a: Function, b: Function):
475 for cand in (a, b):
476 if cand.header and cand.header.startswith(PUBLIC_INCLUDE_DIR):
477 return cand.header, cand.header_line
478 for cand in (a, b):
479 if cand.header:
480 return cand.header, cand.header_line
481 return None, None
482
483 header, header_line = _better_header(prev, new)
484
485 definition = next((f for f in (prev, new) if f.from_definition), None)
486 declaration = next(
487 (
488 f
489 for f in (prev, new)
490 if not f.from_definition
491 and f.header
492 and f.header.startswith(PUBLIC_INCLUDE_DIR)
493 ),
494 None,
495 )
496 documented = next((f for f in (prev, new) if f.detail), None) or prev
497 with_body = next((f for f in (prev, new) if f.has_body), prev)
498
499 params = definition.params if definition else (prev.params or new.params)
500 decl_params = (
501 declaration.params
502 if declaration
503 else (prev.decl_params or new.decl_params or params)
504 )
505 decl_return_type = (
506 declaration.return_type
507 if declaration
508 else (prev.decl_return_type or new.decl_return_type)
509 )
510
511 for src in (declaration, documented):
512 if src is None or src is definition:
513 continue
514 for i, p in enumerate(params):
515 if i < len(src.params):
516 p.doc = _combine_docs(p.doc, src.params[i].doc)
517
518 return Function(
519 name=prev.name,
520 return_type=(definition or prev).return_type or new.return_type,
521 params=params,
522 header=header,
523 header_line=header_line,
524 body_file=with_body.body_file,
525 body_start=with_body.body_start,
526 body_end=with_body.body_end,
527 brief=documented.brief or prev.brief or new.brief,
528 detail=documented.detail or prev.detail or new.detail,
529 from_definition=bool(definition),
530 decl_params=decl_params,
531 decl_return_type=decl_return_type,
532 )
