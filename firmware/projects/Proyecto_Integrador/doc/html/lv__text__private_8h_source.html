163static inline bool lv_text_is_break_char(uint32_t letter)
164{
165 uint8_t i;
166 bool ret = false;
167
168 /*Compare the letter to TXT_BREAK_CHARS*/
169 for(i = 0; LV_TXT_BREAK_CHARS[i] != '\0'; i++) {
170 if(letter == (uint32_t)LV_TXT_BREAK_CHARS[i]) {
171 ret = true; /*If match then it is break char*/
172 break;
173 }
174 }
175
176 return ret;
177}

194static inline bool lv_text_is_a_word(uint32_t letter)
195{
196 /*Cheap check on invalid letter*/
197 if(letter == 0) return false;
198
199 /*CJK Unified Ideographs*/
200 if(letter >= 0x4E00 && letter <= 0x9FFF) {
201 return true;
202 }
203
204 /*Fullwidth ASCII variants*/
205 if(letter >= 0xFF01 && letter <= 0xFF5E) {
206 return true;
207 }
208
209 /*CJK symbols and punctuation*/
210 if(letter >= 0x3000 && letter <= 0x303F) {
211 return true;
212 }
213
214 /*CJK Radicals Supplement*/
215 if(letter >= 0x2E80 && letter <= 0x2EFF) {
216 return true;
217 }
218
219 /*CJK Strokes*/
220 if(letter >= 0x31C0 && letter <= 0x31EF) {
221 return true;
222 }
223
224 /*Hiragana and Katakana*/
225 if(letter >= 0x3040 && letter <= 0x30FF) {
226 return true;
227 }
228
229 /*Chinese Vertical Forms*/
230 if(letter >= 0xFE10 && letter <= 0xFE1F) {
231 return true;
232 }
233
234 /*CJK Compatibility Forms*/
235 if(letter >= 0xFE30 && letter <= 0xFE4F) {
236 return true;
237 }
238
239 return false;
240}

249static inline bool lv_text_is_marker(uint32_t letter)
250{
251 if(letter < 0x20) return true;
252
253 /*U+061C ARABIC LETTER MARK, see https://www.compart.com/en/unicode/block/U+0600*/
254 if(letter == 0x061C) return true;
255
256 /*U+115F HANGUL CHOSEONG FILLER, See https://www.compart.com/en/unicode/block/U+1100*/
257 if(letter == 0x115F) return true;
258 /*U+1160 HANGUL JUNGSEONG FILLER*/
259 if(letter == 0x1160) return true;
260
261 /*See https://www.compart.com/en/unicode/block/U+1800*/
262 if(letter >= 0x180B && letter <= 0x180E) return true;
263
264 /*See https://www.compart.com/en/unicode/block/U+2000*/
265 if(letter >= 0x200B && letter <= 0x200F) return true;
266 if(letter >= 0x2028 && letter <= 0x202F) return true;
267 if(letter >= 0x205F && letter <= 0x206F) return true;
268
269 /*U+FEFF ZERO WIDTH NO-BREAK SPACE, see https://www.compart.com/en/unicode/block/U+FE70*/
270 if(letter == 0xFEFF) return true;
271
272 if(letter == 0xF8FF) return true; /*LV_SYMBOL_DUMMY*/
273
274 return false;
275}

292static inline int32_t lv_font_get_top_trim_internal(const lv_font_t * font, lv_text_leading_trim_t trim)
293{
294 LV_ASSERT(font != NULL);
295 switch(trim) {
296 case LV_TEXT_LEADING_TRIM_CAPITAL_BASELINE:
297 case LV_TEXT_LEADING_TRIM_CAPITAL:
298 return (font->line_height - font->base_line) - font->cap_height;
299 case LV_TEXT_LEADING_TRIM_LOWER_BASELINE:
300 case LV_TEXT_LEADING_TRIM_LOWER:
301 return (font->line_height - font->base_line) - font->x_height;
302 case LV_TEXT_LEADING_TRIM_NONE:
303 return 0;
304 }
305 LV_UNREACHABLE();
306}
