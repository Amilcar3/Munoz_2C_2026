121static inline void tex_storage_2d_compat(GLenum target, GLsizei levels, GLenum internalformat,
122 GLsizei width, GLsizei height)
123{
124#ifdef glTexStorage2D
125 if(glad_glTexStorage2DEXT) {
126 glTexStorage2D(target, levels, internalformat, width, height);
127 return;
128 }
129#endif
130 /* Fallback: use glTexImage2D for each mipmap level */
131 GLenum format = GL_RGBA;
132 if(internalformat == GL_RGB8) {
133 format = GL_RGB;
134 }
135 for(GLsizei level = 0; level < levels; level++) {
136 glTexImage2D(target, level, internalformat, width, height, 0, format, GL_UNSIGNED_BYTE, NULL);
137 width = (width > 1) ? (width / 2) : 1;
138 height = (height > 1) ? (height / 2) : 1;
139 }

168lv_gltf_model_t * lv_gltf_data_load_internal(const void * data_source, size_t data_size,
169 lv_gltf_model_loader_t * loader)
170{
171 LV_ASSERT(data_source != NULL);
172 lv_gltf_model_t * model = NULL;
173 if(data_size > 0) {
174 model = create_data_from_bytes((const uint8_t *)data_source, data_size);
175 }
176 else {
177 model = create_data_from_file((const char *)data_source);
178 }
179
180 if(!model) {
181 LV_LOG_ERROR("Failed to create glTF data");
182 return NULL;
183 }
184
185 // Parse the visible node structure to get a world transform matrix for each mesh component
186 // instance per node, and apply that matrix to the min/max of the untransformed mesh, then
187 // grow a bounding volume to include those transformed points
188
189 int32_t scene_index = 0;
190 bool first_visible_mesh = true;
191 fastgltf::iterateSceneNodes(
192 model->asset, scene_index, fastgltf::math::fmat4x4(), [&](fastgltf::Node & node, fastgltf::math::fmat4x4 matrix) {
193 if(!node.meshIndex.has_value()) {
194 return;
195 }
196 if(first_visible_mesh) {
197 injest_set_initial_bounds(model, matrix, model->asset.meshes[node.meshIndex.value()]);
198 }
199 else {
200 injest_grow_bounds_to_include(model, matrix, model->asset.meshes[node.meshIndex.value()]);
201 }
202 first_visible_mesh = false;
203 });
204
205 /* Reserve enough space for model nodes */
206 lv_array_init(&model->nodes, model->asset.nodes.size(), sizeof(lv_gltf_model_node_t));
207 /*Virtually set size so that lv_array_assign will work*/
208 model->nodes.size = model->asset.nodes.size();
209
210 for(size_t node_index = 0; node_index < model->asset.nodes.size(); node_index++) {
211 lv_gltf_model_node_t model_node;
212 lv_gltf_model_node_init(model, &model_node, &model->asset.nodes[node_index], "", "");
213 lv_array_assign(&model->nodes, node_index, &model_node);
214 }
215
216 fastgltf::namegen_iterate_scene_nodes(model->asset, scene_index,
217 [&](fastgltf::Node & node, const std::string & node_path, const std::string & node_num_path,
218 size_t node_index, std::size_t child_index) {
219 LV_UNUSED(child_index);
220 lv_gltf_model_node_t model_node;
221 lv_gltf_model_node_init(model, &model_node, &node, node_path.c_str(), node_num_path.c_str());
222
223 /* Store the nodes in the same order as fastgltf
224 * This is a workaround as we can't assign any type of user data to fastgltf's types*/
225 lv_gltf_model_node_deinit((lv_gltf_model_node_t *)lv_array_at(&model->nodes, node_index));
226 lv_array_assign(&model->nodes, node_index, & model_node);
227 });
228
229 {
230 bool owns_loader = loader == NULL;
231 if(!loader) {
232 loader = lv_gltf_model_loader_create();
233 if(!loader) {
234 LV_LOG_ERROR("Failed to create gltf model loader");
235 lv_gltf_model_delete(model);
236 return NULL;
237 }
238 }
239 uint32_t i = 0;
240 for(auto & image : model->asset.images) {
241 injest_image(loader, model, image, i);
242 i++;
243 }
244 if(owns_loader) {
245 lv_gltf_model_loader_delete(loader);
246 }
247 }
248 uint16_t lightnum = 0;
249 for(auto & light : model->asset.lights) {
250 injest_light(model, lightnum, light, 0);
251 lightnum += 1;
252 }
253 for(auto & mesh : model->asset.meshes) {
254 injest_mesh(model, mesh);
255 }
256
257 if(model->asset.defaultScene.has_value()) {
258 LV_LOG_INFO("Default scene = #%zu", model->asset.defaultScene.value());
259 }
260
261 return model;
262}

268static lv_gltf_model_t * create_data_from_file(const char * path)
269{
270 LV_ASSERT(path != NULL);
271 lv_fs_file_t file;
272 lv_fs_res_t res = lv_fs_open(&file, path, LV_FS_MODE_RD);
273 if(res != LV_FS_RES_OK) {
274 LV_LOG_ERROR("Failed to open file '%s': %d", path, res);
275 return NULL;
276 }
277 res = lv_fs_seek(&file, 0, LV_FS_SEEK_END);
278 if(res != LV_FS_RES_OK) {
279 LV_LOG_ERROR("Failed to seek end of file '%s': %d", path, res);
280 lv_fs_close(&file);
281 return NULL;
282 }
283 uint32_t file_size;
284 res = lv_fs_tell(&file, &file_size);
285 if(res != LV_FS_RES_OK) {
286 LV_LOG_ERROR("Failed to get file count size '%s': %d", path, res);
287 lv_fs_close(&file);
288 return NULL;
289 }
290
291 res = lv_fs_seek(&file, 0, LV_FS_SEEK_SET);
292 if(res != LV_FS_RES_OK) {
293 LV_LOG_ERROR("Failed to seek start of file '%s': %d", path, res);
294 lv_fs_close(&file);
295 return NULL;
296 }
297
298 uint8_t * bytes = (uint8_t *) lv_malloc(file_size);
299 LV_ASSERT_MALLOC(bytes);
300 if(bytes == NULL) {
301 lv_fs_close(&file);
302 return NULL;
303 }
304
305 uint32_t bytes_read;
306 res = lv_fs_read(&file, bytes, file_size, &bytes_read);
307 lv_fs_close(&file);
308 if(res != LV_FS_RES_OK) {
309 LV_LOG_ERROR("Failed to read file '%s': %d", path, res);
310 lv_free(bytes);
311 return NULL;
312 }
313 if(bytes_read != file_size) {
314 LV_LOG_ERROR("Failed to read the entire gltf file '%s': %d", path, res);
315 lv_free(bytes);
316 return NULL;
317 }
318
319 lv_gltf_model_t * model = create_data_from_bytes(bytes, file_size);
320 lv_free(bytes);
321 if(model == NULL) {
322 return NULL;
323 }
324
325 model->filename = path;
326 return model;
327}

329static lv_gltf_model_t * create_data_from_bytes(const uint8_t * bytes, size_t data_size)
330{
331 LV_ASSERT(bytes != NULL);
332 LV_ASSERT(data_size > 0);
333 fastgltf::Parser parser(SUPPORTED_EXTENSIONS);
334 auto gltf_buffer = fastgltf::GltfDataBuffer::FromBytes(reinterpret_cast<const std::byte *>(bytes), data_size);
335 if(!gltf_buffer) {
336 LV_LOG_ERROR("Failed to create glTF buffer from bytes: %s",
337 std::string(fastgltf::getErrorMessage(gltf_buffer.error())).c_str());
338 return NULL;
339 }
340 auto asset = parser.loadGltf(gltf_buffer.get(), ".", GLTF_OPTIONS);
341 if(!asset) {
342 LV_LOG_ERROR("Failed to decode glTF bytes: %s", std::string(fastgltf::getErrorMessage(asset.error())).c_str());
343 return NULL;
344 }
345 return lv_gltf_data_create_internal("from_bytes", std::move(asset.get()));
346}

348static void make_small_magenta_texture(uint32_t new_magenta_tex)
349{
350 GL_CALL(glBindTexture(GL_TEXTURE_2D, new_magenta_tex));
351 unsigned char clearBytes[4] = { 255, 0, 255, 255 }; // RGBA format
352 GL_CALL(glTexImage2D(GL_TEXTURE_2D, 0, GL_RGBA, 1, 1, 0, GL_RGBA, GL_UNSIGNED_BYTE, clearBytes));
353 // Set texture parameters (optional but recommended)
354 GL_CALL(glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_NEAREST));
355 GL_CALL(glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_NEAREST));
356 GL_CALL(glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_S, GL_CLAMP_TO_EDGE));
357 GL_CALL(glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_T, GL_CLAMP_TO_EDGE));
358 GL_CALL(glGenerateMipmap(GL_TEXTURE_2D));
359 GL_CALL(glBindTexture(GL_TEXTURE_2D, 0));
360 return;
361}

444static void injest_grow_bounds_to_include(lv_gltf_model_t * data, const fastgltf::math::fmat4x4 & matrix,
445 const fastgltf::Mesh & mesh)
446{
447 LV_ASSERT(data != NULL);
448 /* Grow the bounds to include the specified mesh. */
449 fastgltf::math::fvec3 v_min{ data->vertex_min[0], data->vertex_min[1], data->vertex_min[2] };
450
451 fastgltf::math::fvec3 v_max{
452 data->vertex_max[0],
453 data->vertex_max[1],
454 data->vertex_max[2],
455 };
456 fastgltf::math::fvec3 v_cen{ data->vertex_cen[0], data->vertex_cen[1], data->vertex_cen[2] };
457
458 float new_bound_radius = data->bound_radius;
459 if(mesh.primitives.size() > 0) {
460 set_bounds_info(data, v_min, v_max, v_cen, new_bound_radius);
461 return;
462 }
463 size_t accessor_index = mesh.primitives[0].findAttribute("POSITION")->accessorIndex;
464 const auto & accessor = data->asset.accessors[accessor_index];
465
466 if(!accessor.bufferViewIndex.has_value() || !(accessor.min.has_value() && accessor.max.has_value())) {
467 set_bounds_info(data, v_min, v_max, v_cen, new_bound_radius);
468 return;
469 }
470
471 fastgltf::math::fvec4 t_min{ (float)(accessor.min.value().get<double>(0)), (float)(accessor.min.value().get<double>(1)),
472 (float)(accessor.min.value().get<double>(2)), 1.f };
473 fastgltf::math::fvec4 t_max{ (float)(accessor.max.value().get<double>(0)), (float)(accessor.max.value().get<double>(1)),
474 (float)(accessor.max.value().get<double>(2)), 1.f };
475
476 t_min = matrix * t_min;
477 t_max = matrix * t_max;
478 v_max[0] = LV_MAX(LV_MAX(v_max[0], t_min.x()), t_max.x());
479 v_max[1] = LV_MAX(LV_MAX(v_max[1], t_min.y()), t_max.y());
480 v_max[2] = LV_MAX(LV_MAX(v_max[2], t_min.z()), t_max.z());
481 v_min[0] = LV_MIN(LV_MIN(v_min[0], t_min.x()), t_max.x());
482 v_min[1] = LV_MIN(LV_MIN(v_min[1], t_min.y()), t_max.y());
483 v_min[2] = LV_MIN(LV_MIN(v_min[2], t_min.z()), t_max.z());
484 v_cen[0] = (v_max[0] + v_min[0]) / 2.0f;
485 v_cen[1] = (v_max[1] + v_min[1]) / 2.0f;
486 v_cen[2] = (v_max[2] + v_min[2]) / 2.0f;
487
488 float size_x = v_max[0] - v_min[0];
489 float size_y = v_max[1] - v_min[1];
490 float size_z = v_max[2] - v_min[2];
491 new_bound_radius = std::sqrt((size_x * size_x) + (size_y * size_y) + (size_z * size_z)) / 2.0f;
492
493 set_bounds_info(data, v_min, v_max, v_cen, new_bound_radius);
494}
495static void injest_set_initial_bounds(lv_gltf_model_t * data, const fastgltf::math::fmat4x4 & matrix,
496 const fastgltf::Mesh & mesh)
497{
498 LV_ASSERT(data != NULL);
499 fastgltf::math::fvec3 v_min, v_max, v_cen;
500 float radius = 0.f;
501 if(mesh.primitives.size() == 0) {
502 set_bounds_info(data, v_min, v_max, v_cen, radius);
503 return;
504 }
505
506 size_t accessor_index = mesh.primitives[0].findAttribute("POSITION")->accessorIndex;
507 const auto & accessor = data->asset.accessors[accessor_index];
508
509 if(!accessor.bufferViewIndex.has_value() || !(accessor.min.has_value() && accessor.max.has_value())) {
510 set_bounds_info(data, v_min, v_max, v_cen, radius);
511 return;
512 }
513
514 fastgltf::math::fvec4 t_min{ (float)(accessor.min.value().get<double>(0)), (float)(accessor.min.value().get<double>(1)),
515 (float)(accessor.min.value().get<double>(2)), 1.f };
516
517 fastgltf::math::fvec4 t_max{ (float)(accessor.max.value().get<double>(0)), (float)(accessor.max.value().get<double>(1)),
518 (float)(accessor.max.value().get<double>(2)), 1.f };
519
520 t_min = matrix * t_min;
521 t_max = matrix * t_max;
522
523 v_max[0] = LV_MAX(t_min.x(), t_max.x());
524 v_max[1] = LV_MAX(t_min.y(), t_max.y());
525 v_max[2] = LV_MAX(t_min.z(), t_max.z());
526 v_min[0] = LV_MIN(t_min.x(), t_max.x());
527 v_min[1] = LV_MIN(t_min.y(), t_max.y());
528 v_min[2] = LV_MIN(t_min.z(), t_max.z());
529 v_cen[0] = (v_max[0] + v_min[0]) / 2.0f;
530 v_cen[1] = (v_max[1] + v_min[1]) / 2.0f;
531 v_cen[2] = (v_max[2] + v_min[2]) / 2.0f;
532 const float size_x = v_max[0] - v_min[0];
533 const float size_y = v_max[1] - v_min[1];
534 const float size_z = v_max[2] - v_min[2];
535 radius = std::sqrt((size_x * size_x) + (size_y * size_y) + (size_z * size_z)) / 2.0f;
536
537 set_bounds_info(data, v_min, v_max, v_cen, radius);
538}

540static bool injest_image(lv_gltf_model_loader_t * loader, lv_gltf_model_t * data, fastgltf::Image & image,
541 uint32_t index)
542{
543 LV_ASSERT(loader != NULL);
544 LV_ASSERT(data != NULL);
545 std::string _tex_id = std::string(lv_gltf_get_filename(data)) + "_IMG" + std::to_string(index);
546
547 char tmp[512];
548 lv_snprintf(tmp, sizeof(tmp), "%s_img_%u", data->filename, index);
549 const uint32_t hash = lv_opengl_shader_hash(tmp);
550 GLuint texture_id = lv_gltf_model_loader_get_texture(loader, hash);
551
552 if(texture_id != GL_NONE) {
553 LV_LOG_TRACE("Emplacing back already cached texture from previous injest iteration %u", texture_id);
554 data->textures.emplace_back(texture_id);
555 return true;
556 }
557
558 LV_LOG_TRACE("Image (%s) [%d] [%u]", image.name.c_str(), texture_id, hash);
559 glGenTextures(1, &texture_id);
560 glBindTexture(GL_TEXTURE_2D, texture_id);
561 GL_CALL(glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_LINEAR_MIPMAP_LINEAR));
562 GL_CALL(glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_LINEAR));
563 GL_CALL(glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAX_LEVEL, 20));
564
565 GL_CALL(glPixelStorei(GL_UNPACK_ALIGNMENT, 1));

567 std::visit(fastgltf::visitor{
568 [](auto & arg)
569 {
570 LV_UNUSED(arg);
571 LV_LOG_ERROR("Unexpected image source");
572 },
573 [&](fastgltf::sources::URI & file_path)
574 {

576 LV_ASSERT_MSG(file_path.uri.isLocalPath(), "We're only capable of loading local files.");

583 glTexSubImage2D(GL_TEXTURE_2D, 0, 0, 0, width, height, GL_RGBA, GL_UNSIGNED_BYTE, data);
584 stbi_image_free(data);
585 },
586 [&](fastgltf::sources::Array & vector)
587 {
588 int32_t width, height, nrChannels;
589 LV_LOG_TRACE("Unpacking image data: %s", image.name.c_str());
590
591 unsigned char * data = stbi_load_from_memory(
592 reinterpret_cast<const stbi_uc *>(vector.bytes.data()),
593 static_cast<int32_t>(vector.bytes.size()), &width, &height, &nrChannels, 4);
594 tex_storage_2d_compat(GL_TEXTURE_2D, get_level_count(width, height), GL_RGBA8, width, height);
595 glTexSubImage2D(GL_TEXTURE_2D, 0, 0, 0, width, height, GL_RGBA, GL_UNSIGNED_BYTE, data);
596 stbi_image_free(data);
597 },
598 [&](fastgltf::sources::BufferView & view)
599 {
600 LV_LOG_TRACE("Injesting image from bufferview: %s", image.name.c_str());
601 image_invalidated |= injest_image_from_buffer_view(data, view, texture_id);
602 },
603 }, image.data);
604
605 if(!image_invalidated) {
606 glGenerateMipmap(GL_TEXTURE_2D);
607 }
608 else {
609 LV_LOG_ERROR("Failed to load image %s", image.name.c_str());
610 }
611 LV_LOG_TRACE("Storing texture with hash: %u %u", hash, texture_id);
612 lv_gltf_model_loader_store_texture(loader, hash, texture_id);
613 GL_CALL(glBindTexture(GL_TEXTURE_2D, 0));
614 data->textures.emplace_back(texture_id);
615 return true;
616}

618static bool injest_image_from_buffer_view(lv_gltf_model_t * data, fastgltf::sources::BufferView & view,
619 GLuint texture_id)
620{
621 LV_ASSERT(data != NULL);
622 /* Yes, we've already loaded every buffer into some GL buffer. However, with GL it's simpler
623 to just copy the buffer data again for the texture. Besides, this is just an example. */
624 auto & buffer_view = data->asset.bufferViews[view.bufferViewIndex];
625 auto & buffer = data->asset.buffers[buffer_view.bufferIndex];
626 LV_LOG_INFO("Unpacking image bufferView: %s from %zu bytes", buffer_view.name.c_str(), buffer.byteLength);
627 return std::visit(
628 fastgltf::visitor{
629 // We only care about VectorWithMime here, because we specify LoadExternalBuffers, meaning
630 // all buffers are already loaded into a vector.
631 [](auto & arg)
632 {
633 LV_UNUSED(arg);
634 LV_LOG_ERROR("Unexpected image source");
635 return false;
636 },
637 [&](fastgltf::sources::Array & vector)
638 {
639 int32_t width, height, nrChannels;
640 int32_t webpRes = WebPGetInfo(
641 reinterpret_cast<const uint8_t *>(vector.bytes.data() + buffer_view.byteOffset),
642 static_cast<std::size_t>(buffer_view.byteLength), &width, &height);
643
644 LV_LOG_TRACE("[WEBP] width: %d height: %d", width, height);
645 if(!webpRes) {
646 unsigned char * data = stbi_load_from_memory(
647 reinterpret_cast<const stbi_uc *>(vector.bytes.data() + buffer_view.byteOffset),
648 static_cast<int32_t>(buffer_view.byteLength), &width, &height, &nrChannels, 4);
649 if((width <= 0) || (height <= 0)) {
650 LV_LOG_ERROR("Failed to load image from memory");
651 make_small_magenta_texture(texture_id);
652 return true;
653 }
654 LV_LOG_TRACE("[WEBP] width: %d height: %d", width, height);
655 tex_storage_2d_compat(GL_TEXTURE_2D, get_level_count(width, height), GL_RGBA8, width, height);
656 glTexSubImage2D(GL_TEXTURE_2D, 0, 0, 0, width, height, GL_RGBA, GL_UNSIGNED_BYTE, data);
657 stbi_image_free(data);
658 return false;
659 }
660 WebPBitstreamFeatures features = WebPBitstreamFeatures();
661 auto status_code = WebPGetFeatures(
662 reinterpret_cast<const uint8_t *>(vector.bytes.data() + buffer_view.byteOffset),
663 static_cast<std::size_t>(buffer_view.byteLength), &features);
664 if(status_code != VP8_STATUS_OK) {
665 LV_LOG_ERROR("Failed to load webp image %d", status_code);
666 make_small_magenta_texture(texture_id);
667 return true;
668 }
669 if(features.has_alpha) {
670 uint8_t * unpacked = WebPDecodeRGBA(
671 reinterpret_cast<const uint8_t *>(vector.bytes.data() + buffer_view.byteOffset),
672 static_cast<std::size_t>(buffer_view.byteLength), &width, &height);
673 tex_storage_2d_compat(GL_TEXTURE_2D, get_level_count(width, height), GL_RGBA8, width, height);
674 glTexSubImage2D(GL_TEXTURE_2D, 0, 0, 0, width, height, GL_RGBA, GL_UNSIGNED_BYTE,
675 unpacked);
676 WebPFree(unpacked);
677 }
678 else {
679 uint8_t * unpacked = WebPDecodeRGB(
680 reinterpret_cast<const uint8_t *>(vector.bytes.data() + buffer_view.byteOffset),
681 static_cast<std::size_t>(buffer_view.byteLength), &width, &height);
682 tex_storage_2d_compat(GL_TEXTURE_2D, get_level_count(width, height), GL_RGB8, width, height);
683 glTexSubImage2D(GL_TEXTURE_2D, 0, 0, 0, width, height, GL_RGB, GL_UNSIGNED_BYTE,
684 unpacked);
685 WebPFree(unpacked);
686 }
687 return false;
688 } },
689 buffer.data);
690}

958template <typename T, typename Func>
959static size_t injest_vec_attribute(uint8_t vec_size, int32_t current_attrib_index, lv_gltf_model_t * data,
960 const fastgltf::Primitive * prim, const char * attrib_id, GLuint primitive_vertex_buffer,
961 size_t offset, bool normals, Func && functor
962
963 )
964{
965 LV_ASSERT(data != NULL);
966 LV_ASSERT(prim != NULL);
967 LV_ASSERT(attrib_id != NULL);
968 const auto & asset = data->asset;
969 if(const auto * _attrib = prim->findAttribute(std::string(attrib_id)); _attrib != prim->attributes.end()) {
970 auto & accessor = asset.accessors[_attrib->accessorIndex];
971 if(accessor.bufferViewIndex.has_value()) {
972 glBindBuffer(GL_ARRAY_BUFFER, primitive_vertex_buffer);
973 fastgltf::iterateAccessorWithIndex<T>(asset, accessor, functor);
974 // Specify the layout of the vertex data
975 glVertexAttribPointer(current_attrib_index, // Attribute index
976 vec_size, // Number of components per vertex
977 GL_FLOAT, // Data type
978 GL_FALSE, // Normalized
979 normals ? sizeof(vertex_t) : sizeof(vertex_flat_t), // Stride (size of one vertex)
980 (void *)offset); // Offset in the buffer
981 glEnableVertexAttribArray(current_attrib_index);
982 }
983 else {
984 glDisableVertexAttribArray(current_attrib_index);
985 }
986 current_attrib_index++;
987 }
988 return current_attrib_index;
