87static lv_result_t decoder_info(lv_image_decoder_t * decoder, lv_image_decoder_dsc_t * dsc, lv_image_header_t * header)
88{
89 LV_UNUSED(decoder); /*Unused*/
90 lv_image_src_t src_type = dsc->src_type; /*Get the source type*/
91
92 /*If it's a webp file...*/
93 if(src_type == LV_IMAGE_SRC_FILE) {
94 uint8_t buf[WEBP_HEADER_SIZE];
95 int width;
96 int height;
97 uint32_t rn;
98 lv_fs_res_t res = lv_fs_read(&dsc->file, buf, sizeof(buf), &rn);
99
100 /* The max header size = RIFF + VP8X + (optional chunks) + VP8(L), 64 bytes is enough to get the width and height.
101 * If the file is smaller than 64 bytes, it's maybe a valid webp file,
102 * so we don't check the result length here.
103 * VP8X : RIFF(12) + VP8X(18) = 30bytes;
104 * VP8(L): RIFF(12) + VP8(L) chunk header(8) + VP8(L) frame header(5) = 23bytes;
105 * VP8: RIFF(12) + VP8(L) chunk header(8) + VP8(L) frame header(10) = 28bytes;
106 */
107 if(res != LV_FS_RES_OK) return LV_RESULT_INVALID;
108
109 if(WebPGetInfo(buf, rn, &width, &height) == 0) {
110 return LV_RESULT_INVALID;
111 }
112
113 /*Default decoder color format is ARGB8888*/
114 header->cf = LV_COLOR_FORMAT_ARGB8888;
115 header->w = width;
116 header->h = height;
117 header->stride = width * sizeof(lv_color32_t);
118 return LV_RESULT_OK;
119 }
120
121 return LV_RESULT_INVALID; /*If it didn't succeed earlier then it's an error*/
122}

130static lv_result_t decoder_open(lv_image_decoder_t * decoder, lv_image_decoder_dsc_t * dsc)
131{
132 LV_UNUSED(decoder); /*Unused*/
133
134 /*If it's a webp file...*/
135 if(dsc->src_type == LV_IMAGE_SRC_FILE) {
136 const char * fn = dsc->src;
137 lv_draw_buf_t * decoded = decode_webp_file(dsc, fn);
138 if(decoded == NULL) {
139 return LV_RESULT_INVALID;
140 }
141
142 lv_draw_buf_t * adjusted = lv_image_decoder_post_process(dsc, decoded);
143 if(adjusted == NULL) {
144 lv_draw_buf_destroy(decoded);
145 return LV_RESULT_INVALID;
146 }
147
148 /*The adjusted draw buffer is newly allocated.*/
149 if(adjusted != decoded) {
150 lv_draw_buf_destroy(decoded);
151 decoded = adjusted;
152 }
153
154 dsc->decoded = decoded;
155
156 if(dsc->args.no_cache) {
157 return LV_RESULT_OK;
158 }
159
160 /*If the image cache is disabled, just return the decoded image*/
161 if(!lv_image_cache_is_enabled()) {
162 return LV_RESULT_OK;
163 }
164
165 lv_image_cache_data_t search_key;
166 search_key.src_type = dsc->src_type;
167 search_key.src = dsc->src;
168 search_key.slot.size = decoded->data_size;
169
170 lv_cache_entry_t * entry = lv_image_decoder_add_to_cache(decoder, &search_key, decoded, NULL);
171
172 if(entry == NULL) {
173 lv_draw_buf_destroy(decoded);
174 return LV_RESULT_INVALID;
175 }
176 dsc->cache_entry = entry;
177
178 return LV_RESULT_OK; /*The image is fully decoded. Return with its pointer*/
179 }
180
181 return LV_RESULT_INVALID; /*If not returned earlier then it failed*/
182}

195static lv_draw_buf_t * decode_webp_file(lv_image_decoder_dsc_t * dsc, const char * filename)
196{
197 LV_PROFILER_DECODER_BEGIN;
198 uint32_t data_size;
199 uint8_t * data = lv_fs_load_with_alloc(filename, &data_size);
200 if(data == NULL) {
201 LV_LOG_WARN("can't load file: %s", filename);
202 LV_PROFILER_DECODER_END;
203 return NULL;
204 }
205
206 /*Alloc image buffer*/
207 lv_draw_buf_t * decoded;
208 decoded = lv_draw_buf_create_ex(image_cache_draw_buf_handlers, dsc->header.w, dsc->header.h, dsc->header.cf,
209 LV_STRIDE_AUTO);
210 if(decoded == NULL) {
211 LV_LOG_ERROR("alloc draw buffer failed: %s", filename);
212 lv_free(data);
213 LV_PROFILER_DECODER_END;
214 return NULL;
215 }
216
217 WebPDecoderConfig config;
218 WebPInitDecoderConfig(&config);
219
220 config.output.colorspace = MODE_BGRA;
221 config.output.u.RGBA.rgba = (uint8_t *) decoded->data;
222 config.output.u.RGBA.stride = decoded->header.stride;
223 config.output.u.RGBA.size = decoded->data_size;
224 config.output.is_external_memory = 1;
225
226 LV_PROFILER_DECODER_BEGIN_TAG("WebPDecode");
227 int status = WebPDecode(data, data_size, &config);
228 LV_PROFILER_DECODER_END_TAG("WebPDecode");
229
230 if(status != VP8_STATUS_OK) {
231 LV_LOG_ERROR("decode webp failed: %s, status: %d", filename, status);
232 lv_draw_buf_destroy(decoded);
233 decoded = NULL;
234 }
235
236 lv_free(data);
237 LV_PROFILER_DECODER_END;
238 return decoded;
239}
