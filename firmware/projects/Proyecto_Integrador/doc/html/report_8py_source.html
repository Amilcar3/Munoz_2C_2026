85def symbol_table(pr_symbols, master_symbols):
86 """The symbols that moved most, so a regression says where and not just how much."""
87 names = set(pr_symbols) | set(master_symbols)
88 rows = sorted(
89 ((pr_symbols.get(n, 0) - master_symbols.get(n, 0), n) for n in names),
90 key=lambda r: -abs(r[0]),
91 )
92 rows = [r for r in rows if r[0]][:TOP_SYMBOLS]
93 if not rows:
94 return []
95
96 lines = ["", "| Symbol | master | PR | Δ |", "|---|---:|---:|---:|"]
97 for d, name in rows:
98 before = master_symbols.get(name, 0)
99 now = pr_symbols.get(name, 0)
100 lines.append(f"| `{name}` | {before} B | {now} B | {d:+d} B |")
101 lines.append("")
102 return lines

242def metric_summary(rows, key, label):
243 """One clause for one metric: what it did, and the target where it did it worst.
244
245 Returns the clause, the delta it describes and whether that delta is notable.
246 """
247 base_key = f"{key}_base"
248 comparable = [r for r in rows if r[key] is not None and r[base_key] is not None]
249 if not comparable:
250 return "", 0, False
251
252 worst = max(comparable, key=lambda r: r[key] - r[base_key])
253 best = min(comparable, key=lambda r: r[key] - r[base_key])
254 grew = worst[key] - worst[base_key]
255 shrank = best[key] - best[base_key]
256
257 if grew == 0 and shrank == 0:
258 return f"{label} unchanged", 0, False
259
260 row, delta = (worst, grew) if grew > 0 else (best, shrank)
261 return (
262 f"{label} {delta:+d} B ({pct(row[key], row[base_key]):+.1f}%) on "
263 f"{row['target']}/{row['config']}",
264 delta,
265 is_notable(worst[key], worst[base_key]),
266 )

269def summarise(rows):
270 """One line for the shared PR report table.
271
272 Every metric reports itself. Picking one to headline is what used to make this
273 claim things about the metric it had not picked - "flash unchanged" on a change
274 that shrank flash, for one.
275
276 Each clause names the worst target and configuration rather than an average: the
277 failure this test exists to catch is a single target running out of flash, and a
278 mean or a median hides exactly that. The median comes along for flash because it
279 says for free whether a change is broad or isolated.
280 """
281 if not rows:
282 return "info", "no baseline to compare against"
283
284 flash, flash_delta, flash_notable = metric_summary(rows, "flash", "flash")
285 ram, ram_delta, ram_notable = metric_summary(rows, "ram", "RAM")
286 lib, lib_delta, lib_notable = metric_summary(rows, "lib", "library")
287
288 if not flash_delta and not ram_delta and not lib_delta:
289 return "stable", "no change"
290
291 parts = [flash, ram]
292 if flash_delta:
293 median = int(statistics.median(r["flash"] - r["flash_base"] for r in rows))
294 parts[0] += f", {median:+d} B median"
295
296 # Worth its own clause when it says something flash does not: either it grew more
297 # than the linked image did - the case that costs the bindings flash while every
298 # linked demo looks unchanged - or it is the only thing that moved at all.
299 if lib and (lib_delta > max(flash_delta, 0) or not (flash_delta or ram_delta)):
300 parts.append(lib)
301
302 if flash_notable or ram_notable or lib_notable:
303 icon = "warn"
304 elif max(flash_delta, ram_delta, lib_delta) > 0:
305 icon = "up"
306 else:
307 icon = "down"
308
309 return icon, " · ".join(p for p in parts if p)
