31// String hashing function.
32static inline uint32_t djb2_hash(const char *s)
33{
34 unsigned long hash = 5381;
35
36 while (*s) {
37 /* hash = hash * 33 ^ c */
38 hash = ((hash << 5) + hash) ^ *s++;
39 }
40
41 return hash;
42}

55frogfs_fs_t *frogfs_init(const frogfs_config_t *conf)
56{
57 frogfs_fs_t *fs = lv_calloc(1, sizeof(frogfs_fs_t));
58 if (fs == NULL) {
59 LV_LOG_ERROR("calloc failed");
60 return NULL;
61 }
62
63 LV_LOG_TRACE("%p", fs);
64
65 fs->head = (const void *) conf->addr;
66 if (fs->head == NULL) {
67 LV_LOG_ERROR("flash mmap not enabled and addr is NULL");
68 goto err_out;
69 }
70
71 if (fs->head->magic != FROGFS_MAGIC) {
72 LV_LOG_ERROR("frogfs magic not found");
73 goto err_out;
74 }
75
76 if (fs->head->ver_major != FROGFS_VER_MAJOR) {
77 LV_LOG_ERROR("frogfs major version mismatch. filesystem is v%d.%d and this "
78 "library is v%d.%d", fs->head->ver_major, fs->head->ver_minor,
79 FROGFS_VER_MAJOR, FROGFS_VER_MINOR);
80 goto err_out;
81 }
82
83 fs->num_entries = fs->head->num_entries;
84 fs->hash = (const void *) fs->head + sizeof(frogfs_head_t);
85 fs->root = (const void *) fs->hash + (sizeof(frogfs_hash_t) * fs->num_entries);
86
87 return fs;
88
89err_out:
90 frogfs_deinit(fs);
91 return NULL;
92}

101const frogfs_entry_t *frogfs_get_entry(const frogfs_fs_t *fs, const char *path)
102{
103 LV_ASSERT_NULL(fs);
104 LV_ASSERT_NULL(path);
105
106 while (*path == '/') {
107 path++;
108 }
109 LV_LOG_TRACE("'%s'", path);
110
111 uint32_t hash = djb2_hash(path);
112 LV_LOG_TRACE("hash %08"PRIx32, hash);
113
114 int first = 0;
115 int last = fs->num_entries - 1;
116 int middle;
117 const frogfs_hash_t *e;
118
119 while (first <= last) {
120 middle = first + (last - first) / 2;
121 e = &fs->hash[middle];
122 if (e->hash == hash) {
123 break;
124 } else if (e->hash < hash) {
125 first = middle + 1;
126 } else {
127 last = middle - 1;
128 }
129 }
130
131 if (first > last) {
132 LV_LOG_TRACE("no match");
133 return NULL;
134 }
135
136 /* move e to the first match */
137 while (middle > 0) {
138 e = fs->hash + middle;
139 if ((e - 1)->hash != hash) {
140 break;
141 }
142 middle--;
143 }
144
145 /* walk through canidates and look for a match */
146 const frogfs_entry_t *entry;
147 do {
148 entry = (const void *) fs->head + e->offs;
149 char *match = frogfs_get_path(fs, entry);
150 if (lv_strcmp(path, match) == 0) {
151 lv_free(match);
152 LV_LOG_TRACE("entry %d", middle);
153 return entry;
154 }
155 lv_free(match);
156 entry++;
157 middle++;
158 } while ((middle < last) && (e->hash == hash));
159
160 LV_LOG_WARN("unable to find entry");
161 return NULL;
162}

172char *frogfs_get_path(const frogfs_fs_t *fs, const frogfs_entry_t *entry)
173{
174 LV_ASSERT_NULL(entry);
175
176 char *path = lv_calloc(LV_FS_MAX_PATH_LENGTH, 1);
177 size_t len = 0;
178 if (!path) {
179 return NULL;
180 }
181
182 if (entry->parent == 0) {
183 return path;
184 }
185
186 while (entry->parent != 0 && len + entry->seg_sz + 1 < LV_FS_MAX_PATH_LENGTH - 1) {
187 const frogfs_entry_t *parent = (const void *) fs->head + entry->parent;
188 if ((const void *) parent == (const void *) fs->root) {
189 lv_memmove(path + entry->seg_sz, path, len);
190 lv_memcpy(path, get_name(entry), entry->seg_sz);
191 len += entry->seg_sz;
192 break;
193 } else {
194 lv_memmove(path + entry->seg_sz + 1, path, len + 1);
195 path[0] = '/';
196 lv_memcpy(path + 1, get_name(entry), entry->seg_sz);
197 len += entry->seg_sz + 1;
198 }
199 entry = parent;
200 }
201
202 return path;
203}

239frogfs_fh_t *frogfs_open(const frogfs_fs_t *fs, const frogfs_entry_t *entry,
240 unsigned int flags)
241{
242 LV_ASSERT_NULL(fs);
243 LV_ASSERT_NULL(entry);
244
245 if (FROGFS_IS_DIR(entry)) {
246 return NULL;
247 }
248
249 const frogfs_file_t *file = (const void *) entry;
250
251 frogfs_fh_t *fh = lv_calloc(1, sizeof(frogfs_fh_t));
252 if (fh == NULL) {
253 LV_LOG_ERROR("calloc failed");
254 goto err_out;
255 }
256
257 LV_LOG_TRACE("%p", fh);
258
259 fh->fs = fs;
260 fh->file = file;
261 fh->data_start = (const void *) fs->head + file->data_offs;
262 fh->data_ptr = fh->data_start;
263 fh->data_sz = file->data_sz;
264 fh->flags = flags;
265
266 if (entry->u.compression == 0 || flags & FROGFS_OPEN_RAW) {
267 fh->real_sz = file->data_sz;
268 fh->decomp_funcs = &frogfs_decomp_raw;
269 }
270#if CONFIG_FROGFS_USE_MINIZ == 1
271 else if ((entry->u.compression == FROGFS_COMP_ALGO_GZIP) ||
272 (entry->u.compression == FROGFS_COMP_ALGO_ZLIB)) {
273 fh->real_sz = ((frogfs_comp_t *) file)->real_sz;
274 fh->decomp_funcs = &frogfs_decomp_miniz;
275 }
276#endif
277#if CONFIG_FROGFS_USE_ZLIB == 1
278 else if ((entry->u.compression == FROGFS_COMP_ALGO_GZIP) ||
279 (entry->u.compression == FROGFS_COMP_ALGO_ZLIB)) {
280 fh->real_sz = ((frogfs_comp_t *) file)->real_sz;
281 fh->decomp_funcs = &frogfs_decomp_zlib;
282 }
283#endif
284#if CONFIG_FROGFS_USE_HEATSHRINK == 1
285 else if (entry->u.compression == FROGFS_COMP_ALGO_HEATSHRINK) {
286 fh->real_sz = ((frogfs_comp_t *) file)->real_sz;
287 fh->decomp_funcs = &frogfs_decomp_heatshrink;
288 }
289#endif
290 else {
291 LV_LOG_ERROR("unsupported compression type %d", entry->u.compression);
292 goto err_out;
293 }
294
295 if (fh->decomp_funcs->open) {
296 if (fh->decomp_funcs->open(fh, flags) < 0) {
297 LV_LOG_ERROR("decomp_funcs->fopen");
298 goto err_out;
299 }
300 }
301
302 return fh;
303
304err_out:
305 frogfs_close(fh);
306 return NULL;
307}
