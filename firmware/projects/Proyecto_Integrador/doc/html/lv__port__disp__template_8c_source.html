120/*Flush the content of the internal buffer the specific area on the display.
121 *`px_map` contains the rendered image as raw pixel map and it should be copied to `area` on the display.
122 *You can use DMA or any hardware acceleration to do this operation in the background but
123 *'lv_display_flush_ready()' has to be called when it's finished.*/
124static void disp_flush(lv_display_t * disp_drv, const lv_area_t * area, uint8_t * px_map)

126 if(disp_flush_enabled) {

130 int32_t y;
131 for(y = area->y1; y <= area->y2; y++) {
132 for(x = area->x1; x <= area->x2; x++) {
133 /*Put a pixel to the display. For example:*/
134 /*put_px(x, y, *px_map)*/
135 px_map++;
136 }
137 }
138 }

140 /*IMPORTANT!!!
141 *Inform the graphics library that you are ready with the flushing*/
142 lv_display_flush_ready(disp_drv);
