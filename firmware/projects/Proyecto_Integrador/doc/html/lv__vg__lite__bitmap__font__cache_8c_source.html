59void lv_vg_lite_bitmap_font_cache_init(struct _lv_draw_vg_lite_unit_t * unit, uint32_t cache_cnt)
60{
61 LV_ASSERT_NULL(unit);
62 LV_ASSERT(unit->bitmap_font_cache == NULL);
63 LV_ASSERT(unit->bitmap_font_pending == NULL);
64 LV_ASSERT(cache_cnt > 0);
65
66 const lv_cache_ops_t ops = {
67 .compare_cb = (lv_cache_compare_cb_t)cache_compare_cb,
68 .create_cb = (lv_cache_create_cb_t)cache_create_cb,
69 .free_cb = (lv_cache_free_cb_t)cache_free_cb,
70 };
71
72 unit->bitmap_font_cache = lv_cache_create(&lv_cache_class_lru_rb_count, sizeof(cache_item_t), cache_cnt, ops);
73 lv_cache_set_name(unit->bitmap_font_cache, "VG_BITMAP_FONT");
74 unit->bitmap_font_pending = lv_vg_lite_pending_create(sizeof(lv_cache_entry_t *), 8);
75 lv_vg_lite_pending_set_free_cb(unit->bitmap_font_pending, cache_release_cb, unit->bitmap_font_cache);
76}

91lv_draw_buf_t * lv_vg_lite_bitmap_font_cache_get(struct _lv_draw_vg_lite_unit_t * unit,
92 const lv_font_glyph_dsc_t * g_dsc)
93{
94 LV_PROFILER_FONT_BEGIN;
95 LV_ASSERT_NULL(unit);
96 LV_ASSERT_NULL(g_dsc);
97
98 uint32_t gid = g_dsc->gid.index;
99 if(!gid) {
100 LV_PROFILER_FONT_END;
101 return NULL;
102 }
103
104 cache_item_t search_key = { 0 };
105 search_key.g_dsc = *g_dsc;
106
107 lv_cache_entry_t * cache_node_entry = lv_cache_acquire(unit->bitmap_font_cache, &search_key, NULL);
108
109 if(cache_node_entry == NULL) {
110 /* check if the cache is full */
111 size_t free_size = lv_cache_get_free_size(unit->bitmap_font_cache, NULL);
112 if(free_size == 0) {
113 LV_LOG_INFO("bitmap font cache is full, release all pending cache entries");
114 lv_vg_lite_finish(unit);
115 }
116
117 cache_node_entry = lv_cache_acquire_or_create(unit->bitmap_font_cache, &search_key, NULL);
118 if(cache_node_entry == NULL) {
119 LV_LOG_ERROR("bitmap cache creating failed");
120 LV_PROFILER_FONT_END;
121 return NULL;
122 }
123 }
124
125 /* Add the new entry to the pending list */
126 lv_vg_lite_pending_add(unit->bitmap_font_pending, &cache_node_entry);
127
128 cache_item_t * cache_item = lv_cache_entry_get_data(cache_node_entry);
129
130 LV_PROFILER_FONT_END;
131 return cache_item->draw_buf;
132}

145static bool cache_create_cb(cache_item_t * item, void * user_data)
146{
147 LV_UNUSED(user_data);
148 LV_PROFILER_FONT_BEGIN;
149
150 LV_LOG_TRACE("gid: %" LV_PRIu32 ", W%" LV_PRIu32 "xH%" LV_PRIu32,
151 item->g_dsc.gid.index, item->g_dsc.box_w, item->g_dsc.box_h);
152
153 lv_draw_buf_t * draw_buf = lv_draw_buf_create_ex(font_draw_buf_handlers,
154 item->g_dsc.box_w,
155 item->g_dsc.box_h,
156 LV_COLOR_FORMAT_A8,
157 LV_STRIDE_AUTO);
158 if(!draw_buf) {
159 LV_LOG_ERROR("Failed to create draw buffer for bitmap font cache");
160 LV_PROFILER_FONT_END;
161 return false;
162 }
163
164 if(!lv_font_get_glyph_bitmap_internal(&item->g_dsc, draw_buf)) {
165 LV_LOG_WARN("Failed to get glyph bitmap for bitmap font cache");
166 lv_draw_buf_destroy(draw_buf);
167 LV_PROFILER_FONT_END;
168 return false;
169 }
170
171 item->draw_buf = draw_buf;
172
173 LV_PROFILER_FONT_END;
174 return true;
175}
