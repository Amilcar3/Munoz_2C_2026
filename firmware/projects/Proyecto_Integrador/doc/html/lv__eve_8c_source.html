89void lv_eve_scissor(uint16_t x1, uint16_t y1, uint16_t x2, uint16_t y2)
90{
91 if(x1 != scissor_x1 || y1 != scissor_y1 || x2 != scissor_x2 || y2 != scissor_y2) {
92 EVE_cmd_dl_burst(SCISSOR_XY(x1, y1));
93 scissor_x1 = x1;
94 scissor_y1 = y1;
95 uint16_t w = x2 - x1 + 1;
96 uint16_t h = y2 - y1 + 1;
97 EVE_cmd_dl_burst(SCISSOR_SIZE(w, h));
98 scissor_x2 = x2;
99 scissor_y2 = y2;
100 }
101}

193void lv_eve_draw_rect_simple(int16_t coord_x1, int16_t coord_y1, int16_t coord_x2, int16_t coord_y2, uint16_t radius)
194{
195 lv_eve_primitive(LV_EVE_PRIMITIVE_RECTS);
196 if(radius > 1) {
197 lv_eve_line_width(radius * 16);
198 }
199 if(radius > 0) {
200 /* With radius 1, the EVE engine draws one half-opacity
201 * pixel outline around the rectangle. Subtracting one
202 * here ensures that the full-opacity rectangle reaches
203 * the coordinate pixels we were passed in. */
204 radius--;
205 }
206
207 lv_eve_vertex_2f(coord_x1 + radius, coord_y1 + radius);
208 lv_eve_vertex_2f(coord_x2 - radius, coord_y2 - radius);
209}

220}
