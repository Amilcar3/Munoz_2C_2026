64SPEC = 1 << COORD_TYPE_SHIFT

69def lv_pct(value: int) -> int:
70 """The LV_PCT() macro, so the tests encode the way LVGL does."""
71 stored = PCT_POS_MAX - value if value < 0 else value
72 return SPEC | stored

104 def test_a_negative_value_whose_type_bits_read_as_spec(self):
105 """LV_COORD_PLAIN() keeps bit 31, so `plain` can exceed the pct range.
106
107 LVGL guards the same way - LV_COORD_IS_PCT() also requires
108 LV_COORD_PLAIN(x) <= LV_PCT_STORED_MAX - and without the guard this
109 comes out as a nonsense percentage rather than the number it is.
110 """
111 raw = ((1 << 31) | SPEC | 100) - (1 << 32)
112 self.assertEqual(decode_coord(raw), str(raw))
