333void lv_chart_get_point_pos_by_id(lv_obj_t * obj, lv_chart_series_t * ser, uint32_t id, lv_point_t * p_out)
334{
335 LV_CHECK_OBJ(obj, MY_CLASS, return);
336 LV_CHECK_ARG(ser != NULL, return);
337 LV_CHECK_ARG(p_out != NULL, return);
338
339 lv_chart_t * chart = (lv_chart_t *)obj;
340 if(id >= chart->point_cnt) {
341 LV_LOG_WARN("Invalid index: %"LV_PRIu32, id);
342 p_out->x = 0;
343 p_out->y = 0;
344 return;
345 }
346
347 int32_t w = lv_obj_get_content_width(obj);
348 int32_t h = lv_obj_get_content_height(obj);
349
350 if(chart->type == LV_CHART_TYPE_LINE || chart->type == LV_CHART_TYPE_CURVE) {
351 if(chart->point_cnt > 1) {
352 p_out->x = (w * id) / (chart->point_cnt - 1);
353 }
354 else {
355 p_out->x = 0;
356 }
357 int32_t temp_y = value_to_y(obj, ser, ser->y_points[id], h);
358 p_out->y = h - temp_y;
359 }
360 else if(chart->type == LV_CHART_TYPE_SCATTER) {
361 p_out->x = lv_map(ser->x_points[id], chart->xmin[ser->x_axis_sec], chart->xmax[ser->x_axis_sec], 0, w);
362 int32_t temp_y = value_to_y(obj, ser, ser->y_points[id], h);
363 p_out->y = h - temp_y;
364 }
365 else if(chart->type == LV_CHART_TYPE_BAR) {
366 uint32_t ser_cnt = lv_ll_get_len(&chart->series_ll);
367 int32_t ser_gap = lv_obj_get_style_pad_column_internal(obj, LV_PART_ITEMS);
368
369 /*Gap between the columns on adjacent X ticks*/
370 int32_t block_gap = lv_obj_get_style_pad_column_internal(obj, LV_PART_MAIN);
371
372 int32_t block_w = (w - ((chart->point_cnt - 1) * block_gap)) / chart->point_cnt;
373
374 if(chart->point_cnt > 1) {
375 p_out->x = (int32_t)((int32_t)(w - block_w) * id) / (chart->point_cnt - 1);
376 }
377 else {
378 p_out->x = 0;
379 }
380
381 lv_chart_series_t * ser_i = NULL;
382 uint32_t ser_idx = 0;
383 LV_LL_READ(&chart->series_ll, ser_i) {
384 if(ser_i == ser) break;
385 ser_idx++;
386 }
387
388 p_out->x = (int32_t)((int32_t)(w + block_gap) * id) / chart->point_cnt;
389 if(ser_cnt > 0) {
390 p_out->x += block_w * ser_idx / ser_cnt;
391
392 int32_t col_w = (block_w - (ser_gap * (ser_cnt - 1))) / ser_cnt;
393 p_out->x += col_w / 2;
394 }
395 else {
396 LV_LOG_WARN("bar chart series count is zero");
397 }
398
399 int32_t temp_y = value_to_y(obj, ser, ser->y_points[id], h);
400 p_out->y = h - temp_y;
401 }
402 else if(chart->type == LV_CHART_TYPE_STACKED) {
403 /*Gap between the columns on adjacent X ticks*/
404 int32_t block_gap = lv_obj_get_style_pad_column_internal(obj, LV_PART_MAIN);
405
406 int32_t block_w = (w - ((chart->point_cnt - 1) * block_gap)) / chart->point_cnt;
407
408 if(chart->point_cnt > 1) {
409 p_out->x = (int32_t)((int32_t)(w - block_w) * id) / (chart->point_cnt - 1);
410 }
411 else {
412 p_out->x = 0;
413 }
414
415 p_out->x += block_w / 2;
416
417 int32_t v_sum = 0;
418 lv_chart_series_t * s;
419 LV_LL_READ(&chart->series_ll, s) {
420 if(s->hidden) continue;
421 int32_t start_point = chart->update_mode == LV_CHART_UPDATE_MODE_SHIFT ? s->start_point : 0;
422 int32_t p_act = (start_point + id) % chart->point_cnt;
423 int32_t v_act = s->y_points[p_act];
424 if(s->y_points[p_act] == LV_CHART_POINT_NONE) continue;
425
426 /* Skip negative values in stacked charts. Negative values are not supported
427 * in stacked charts as they cannot be visually represented in the stacking logic. */
428 if(v_act <= 0) {
429 LV_LOG_WARN("Stacked chart doesn't support negative values.");
430 continue;
431 }
432 v_sum += v_act;
433 if(s == ser) break;
434 }
435
436 int32_t temp_y = value_to_y(obj, ser, v_sum, h);
437 p_out->y = h - temp_y;
438 }
439 /*LV_CHART_TYPE_NONE*/
440 else {
441 p_out->x = 0;
442 p_out->y = 0;
443 }
444
445 int32_t border_width = lv_obj_get_style_border_width_internal(obj, LV_PART_MAIN);
446 p_out->x += lv_obj_get_style_pad_left_internal(obj, LV_PART_MAIN) + border_width;
447 p_out->x -= lv_obj_get_scroll_left(obj);
448
449 uint32_t start_point = chart->update_mode == LV_CHART_UPDATE_MODE_SHIFT ? ser->start_point : 0;
450 id = ((int32_t)start_point + id) % chart->point_cnt;
451
452 p_out->y += lv_obj_get_style_pad_top_internal(obj, LV_PART_MAIN) + border_width;
453 p_out->y -= lv_obj_get_scroll_top(obj);
454}

467lv_chart_series_t * lv_chart_add_series(lv_obj_t * obj, lv_color_t color, lv_chart_axis_t axis)
468{
469 LV_LOG_INFO("begin");
470
471 LV_CHECK_OBJ(obj, MY_CLASS, return NULL);
472
473 lv_chart_t * chart = (lv_chart_t *)obj;
474
475 /* Allocate space for a new series and add it to the chart series linked list */
476 lv_chart_series_t * ser = lv_ll_ins_tail(&chart->series_ll);
477 LV_ASSERT_MALLOC(ser);
478 if(ser == NULL) return NULL;
479 lv_memzero(ser, sizeof(lv_chart_series_t));
480
481 /* Allocate memory for point_cnt points, handle failure below */
482 ser->y_points = lv_malloc(sizeof(int32_t) * chart->point_cnt);
483 LV_ASSERT_MALLOC(ser->y_points);
484
485 if(chart->type == LV_CHART_TYPE_SCATTER) {
486 ser->x_points = lv_malloc(sizeof(int32_t) * chart->point_cnt);
487 LV_ASSERT_MALLOC(ser->x_points);
488 if(NULL == ser->x_points) {
489 lv_free(ser->y_points);
490 lv_ll_remove(&chart->series_ll, ser);
491 lv_free(ser);
492 return NULL;
493 }
494 }
495 else {
496 ser->x_points = NULL;
497 }
498
499 if(ser->y_points == NULL) {
500 if(ser->x_points) {
501 lv_free(ser->x_points);
502 ser->x_points = NULL;
503 }
504
505 lv_ll_remove(&chart->series_ll, ser);
506 lv_free(ser);
507 return NULL;
508 }
509
510 /* Set series properties on successful allocation */
511 ser->color = color;
512 ser->start_point = 0;
513 ser->y_ext_buf_assigned = false;
514 ser->hidden = 0;
515 ser->x_axis_sec = axis & LV_CHART_AXIS_SECONDARY_X ? 1 : 0;
516 ser->y_axis_sec = axis & LV_CHART_AXIS_SECONDARY_Y ? 1 : 0;
517
518 uint32_t i;
519 const int32_t def = LV_CHART_POINT_NONE;
520 int32_t * p_tmp = ser->y_points;
521 for(i = 0; i < chart->point_cnt; i++) {
522 *p_tmp = def;
523 p_tmp++;
524 }
525
526 return ser;
527}

920static void lv_chart_event(const lv_obj_class_t * class_p, lv_event_t * e)
921{
922 LV_UNUSED(class_p);
923 LV_ASSERT(e != NULL);
924
925 /*Call the ancestor's event handler*/
926 lv_result_t res;
927
928 res = lv_obj_event_base(MY_CLASS, e);
929 if(res != LV_RESULT_OK) return;
930
931 lv_event_code_t code = lv_event_get_code(e);
932 lv_obj_t * obj = lv_event_get_current_target(e);
933
934 lv_chart_t * chart = (lv_chart_t *)obj;
935 if(code == LV_EVENT_PRESSED) {
936 lv_indev_t * indev = lv_indev_active();
937 lv_point_t p;
938 lv_indev_get_point(indev, &p);
939
940 p.x -= obj->coords.x1;
941 uint32_t id = get_index_from_x(obj, p.x + lv_obj_get_scroll_left(obj));
942 if(id != (uint32_t)chart->pressed_point_id) {
943 invalidate_point(obj, id);
944 invalidate_point(obj, chart->pressed_point_id);
945 chart->pressed_point_id = id;
946 lv_obj_send_event(obj, LV_EVENT_VALUE_CHANGED, NULL);
947 }
948 }
949 else if(code == LV_EVENT_RELEASED) {
950 invalidate_point(obj, chart->pressed_point_id);
951 chart->pressed_point_id = LV_CHART_POINT_NONE;
952 }
953 else if(code == LV_EVENT_DRAW_MAIN) {
954 lv_layer_t * layer = lv_event_get_layer(e);
955
956 lv_area_t ext_coords;
957 lv_obj_get_coords(obj, &ext_coords);
958 int32_t ext_draw_size = lv_obj_get_ext_draw_size(obj);
959 lv_area_increase(&ext_coords, ext_draw_size, ext_draw_size);
960
961 lv_area_t clip_area;
962 if(lv_area_intersect(&clip_area, &ext_coords, &layer->_clip_area)) {
963 const lv_area_t clip_area_ori = layer->_clip_area;
964 layer->_clip_area = clip_area;
965
966 draw_div_lines(obj, layer);
967
968 if(lv_ll_is_empty(&chart->series_ll) == false) {
969 if(chart->type == LV_CHART_TYPE_LINE) draw_series_line(obj, layer);
970 else if(chart->type == LV_CHART_TYPE_CURVE) draw_series_curve(obj, layer);
971 else if(chart->type == LV_CHART_TYPE_BAR) draw_series_bar(obj, layer);
972 else if(chart->type == LV_CHART_TYPE_STACKED) draw_series_stacked(obj, layer);
973 else if(chart->type == LV_CHART_TYPE_SCATTER) draw_series_scatter(obj, layer);
974 }
975
976 draw_cursors(obj, layer);
977
978 layer->_clip_area = clip_area_ori;
979 }
980 }

983static void draw_div_lines(lv_obj_t * obj, lv_layer_t * layer)
984{
985 LV_ASSERT(obj != NULL);
986 LV_ASSERT(layer != NULL);
987 lv_chart_t * chart = (lv_chart_t *)obj;
988
989 int16_t i;
990 int16_t i_start;
991 int16_t i_end;
992 int32_t border_width = lv_obj_get_style_border_width_internal(obj, LV_PART_MAIN);
993 int32_t pad_left = lv_obj_get_style_pad_left_internal(obj, LV_PART_MAIN) + border_width;
994 int32_t pad_top = lv_obj_get_style_pad_top_internal(obj, LV_PART_MAIN) + border_width;
995 int32_t w = lv_obj_get_content_width(obj);
996 int32_t h = lv_obj_get_content_height(obj);
997
998 lv_draw_line_dsc_t line_dsc;
999 lv_draw_line_dsc_init(&line_dsc);
1000 line_dsc.base.layer = layer;
1001 lv_obj_init_draw_line_dsc(obj, LV_PART_MAIN, &line_dsc);
1002
1003 lv_opa_t border_opa = lv_obj_get_style_border_opa_internal(obj, LV_PART_MAIN);
1004 int32_t border_w = lv_obj_get_style_border_width_internal(obj, LV_PART_MAIN);
1005 lv_border_side_t border_side = lv_obj_get_style_border_side_internal(obj, LV_PART_MAIN);
1006
1007 int32_t scroll_left = lv_obj_get_scroll_left(obj);
1008 int32_t scroll_top = lv_obj_get_scroll_top(obj);
1009 if(chart->hdiv_cnt > 1) {
1010 int32_t y_ofs = obj->coords.y1 + pad_top - scroll_top;
1011 line_dsc.p1.x = obj->coords.x1;
1012 line_dsc.p2.x = obj->coords.x2;
1013
1014 i_start = 0;
1015 i_end = chart->hdiv_cnt;
1016 if(border_opa > LV_OPA_MIN && border_w > 0) {
1017 if((border_side & LV_BORDER_SIDE_TOP) && (lv_obj_get_style_pad_top_internal(obj, LV_PART_MAIN) == 0)) i_start++;
1018 if((border_side & LV_BORDER_SIDE_BOTTOM) && (lv_obj_get_style_pad_bottom_internal(obj, LV_PART_MAIN) == 0)) i_end--;
1019 }
1020
1021 for(i = i_start; i < i_end; i++) {
1022 line_dsc.p1.y = (int32_t)((int32_t)h * i) / (chart->hdiv_cnt - 1);
1023 line_dsc.p1.y += y_ofs;
1024 line_dsc.p2.y = line_dsc.p1.y;
1025 line_dsc.base.id1 = i;
1026
1027 lv_draw_line(layer, &line_dsc);
1028 }
1029 }
1030
1031 if(chart->vdiv_cnt > 1) {
1032 int32_t x_ofs = obj->coords.x1 + pad_left - scroll_left;
1033 line_dsc.p1.y = obj->coords.y1;
1034 line_dsc.p2.y = obj->coords.y2;
1035 i_start = 0;
1036 i_end = chart->vdiv_cnt;
1037 if(border_opa > LV_OPA_MIN && border_w > 0) {
1038 if((border_side & LV_BORDER_SIDE_LEFT) && (lv_obj_get_style_pad_left_internal(obj, LV_PART_MAIN) == 0)) i_start++;
1039 if((border_side & LV_BORDER_SIDE_RIGHT) && (lv_obj_get_style_pad_right_internal(obj, LV_PART_MAIN) == 0)) i_end--;
1040 }
1041
1042 for(i = i_start; i < i_end; i++) {
1043 line_dsc.p1.x = (int32_t)((int32_t)w * i) / (chart->vdiv_cnt - 1);
1044 line_dsc.p1.x += x_ofs;
1045 line_dsc.p2.x = line_dsc.p1.x;
1046 line_dsc.base.id1 = i;
1047
1048 lv_draw_line(layer, &line_dsc);
1049 }
1050 }

1053static void draw_series_line(lv_obj_t * obj, lv_layer_t * layer)
1054{
1055 LV_ASSERT(obj != NULL);
1056 LV_ASSERT(layer != NULL);
1057 lv_chart_t * chart = (lv_chart_t *)obj;
1058 if(chart->point_cnt < 2) return;
1059
1060 uint32_t ser_cnt = lv_ll_get_len(&chart->series_ll);
1061 if(ser_cnt == 0) return;
1062
1063 int32_t border_width = lv_obj_get_style_border_width_internal(obj, LV_PART_MAIN);
1064 int32_t pad_left = lv_obj_get_style_pad_left_internal(obj, LV_PART_MAIN) + border_width;
1065 int32_t pad_top = lv_obj_get_style_pad_top_internal(obj, LV_PART_MAIN) + border_width;
1066 int32_t w = lv_obj_get_content_width(obj);
1067 int32_t h = lv_obj_get_content_height(obj);
1068 int32_t x_ofs = obj->coords.x1 + pad_left - lv_obj_get_scroll_left(obj);
1069 int32_t y_ofs = obj->coords.y1 + pad_top - lv_obj_get_scroll_top(obj);
1070 lv_chart_series_t * ser;
1071
1072 lv_draw_line_dsc_t line_dsc;
1073 lv_draw_line_dsc_init(&line_dsc);
1074 line_dsc.base.layer = layer;
1075 lv_obj_init_draw_line_dsc(obj, LV_PART_ITEMS, &line_dsc);
1076 line_dsc.base.id1 = ser_cnt - 1;
1077
1078 /*If there are at least as many points as pixels then draw only vertical lines*/
1079 bool crowded_mode = (int32_t)chart->point_cnt >= w;
1080
1081 int32_t bullet_w = lv_obj_get_style_width_internal(obj, LV_PART_INDICATOR) / 2;
1082 int32_t bullet_h = lv_obj_get_style_height_internal(obj, LV_PART_INDICATOR) / 2;
1083 int32_t extra_space_x;
1084 if(chart->point_cnt <= 1) extra_space_x = 0;
1085 else extra_space_x = w / (chart->point_cnt - 1) + bullet_w + line_dsc.width;
1086
1087 lv_draw_rect_dsc_t point_draw_dsc;
1088 if(crowded_mode == false) {
1089 lv_draw_rect_dsc_init(&point_draw_dsc);
1090 lv_obj_init_draw_rect_dsc(obj, LV_PART_INDICATOR, &point_draw_dsc);
1091 point_draw_dsc.base.id1 = line_dsc.base.id1;
1092 }
1093
1094 lv_point_precise_t * points = NULL;
1095 if(crowded_mode) {
1096 points = lv_malloc((w + 2 * extra_space_x) * 3 * sizeof(lv_point_precise_t));
1097 }
1098 else {
1099 points = lv_malloc(chart->point_cnt * sizeof(lv_point_precise_t));
1100 }
1101
1102 if(points == NULL) {
1103 LV_LOG_WARN("Couldn't allocate the points array");
1104 return;
1105 }
1106
1107 line_dsc.points = points;
1108
1109 /*Go through all data lines*/
1110 LV_LL_READ_BACK(&chart->series_ll, ser) {
1111 if(ser->hidden) {
1112 if(line_dsc.base.id1 > 0) {
1113 line_dsc.base.id1--;
1114 }
1115 continue;
1116 }
1117 line_dsc.color = ser->color;
1118 line_dsc.base.drop_shadow_color = ser->color;
1119
1120 int32_t start_point = chart->update_mode == LV_CHART_UPDATE_MODE_SHIFT ? ser->start_point : 0;
1121 int32_t p_act = start_point;
1122 int32_t p_prev = start_point;
1123
1124 lv_value_precise_t y_min = obj->coords.y2;
1125 lv_value_precise_t y_max = obj->coords.y1;
1126 lv_value_precise_t x_prev = -10000;
1127 line_dsc.p1.x = x_ofs;
1128 line_dsc.p2.x = x_ofs;
1129 line_dsc.point_cnt = 0;
1130
1131 uint32_t i;
1132 for(i = 0; i < chart->point_cnt; i++) {
1133 lv_value_precise_t p_x = (int32_t)((w * i) / (chart->point_cnt - 1)) + x_ofs;
1134 if(p_x > layer->_clip_area.x2 + extra_space_x + 1) break;
1135 if(p_x < layer->_clip_area.x1 - extra_space_x - 1) {
1136 p_prev = p_act;
1137 continue;
1138 }
1139 p_act = (start_point + i) % chart->point_cnt;
1140
1141 lv_value_precise_t p_y;
1142 if(ser->y_points[p_act] == LV_CHART_POINT_NONE) {
1143 p_y = LV_DRAW_LINE_POINT_NONE;
1144 }
1145 else {
1146 int32_t v = ser->y_points[p_act];
1147 int32_t min_v = chart->ymin[ser->y_axis_sec];
1148 int32_t max_v = chart->ymax[ser->y_axis_sec];
1149 p_y = (int32_t)lv_map(v, min_v, max_v, y_ofs + h, y_ofs);
1150 }
1151
1152 /*In normal mode just collect the points here*/
1153 if(crowded_mode == false) {
1154 points[line_dsc.point_cnt].x = p_x;
1155 points[line_dsc.point_cnt].y = p_y;
1156 line_dsc.point_cnt++;
1157 }
1158 /*In crowded mode draw vertical lines from the min/max on the same X coordinate*/
1159 else {
1160 if(ser->y_points[p_prev] != LV_CHART_POINT_NONE && ser->y_points[p_act] != LV_CHART_POINT_NONE) {
1161 /*Draw only one vertical line between the min and max y-values on the same x-value*/
1162 y_max = LV_MAX(y_max, p_y);
1163 y_min = LV_MIN(y_min, p_y);
1164 if(x_prev != p_x) {
1165 line_dsc.points[line_dsc.point_cnt].y = y_min;
1166 line_dsc.points[line_dsc.point_cnt].x = p_x;
1167 line_dsc.points[line_dsc.point_cnt + 1].y = y_max;
1168 line_dsc.points[line_dsc.point_cnt + 1].x = p_x;
1169 line_dsc.points[line_dsc.point_cnt + 2].y = LV_DRAW_LINE_POINT_NONE;
1170 line_dsc.points[line_dsc.point_cnt + 2].x = p_x;
1171
1172 /*If they are the same no line would be drawn*/
1173 if(line_dsc.points[line_dsc.point_cnt].y == line_dsc.points[line_dsc.point_cnt + 1].y) {
1174 line_dsc.points[line_dsc.point_cnt + 1].y++;
1175 }
1176 y_min = p_y; /*Start the line of the next x from the current last y*/
1177 y_max = p_y;
1178 x_prev = p_x;
1179 line_dsc.point_cnt += 3;
1180 }
1181 }
1182 }
1183
1184 p_prev = p_act;
1185 }
1186
1187 /*Draw the line from the accumulated points*/
1188 lv_draw_line(layer, &line_dsc);
1189 if(!crowded_mode) {
1190 point_draw_dsc.bg_color = ser->color;
1191 point_draw_dsc.base.id1 = line_dsc.base.id1;
1192 /*Add the bullets too*/
1193 if(bullet_w > 0 && bullet_h > 0) {
1194 point_draw_dsc.base.id2 = i - 1; /*Start from the last rendered point*/
1195 int32_t j;
1196 for(j = line_dsc.point_cnt - 1; j >= 0; j--) {
1197 if(points[j].y == LV_DRAW_LINE_POINT_NONE) continue;
1198
1199 lv_area_t point_area;
1200 point_area.x1 = (int32_t)points[j].x - bullet_w;
1201 point_area.x2 = (int32_t)points[j].x + bullet_w;
1202 point_area.y1 = (int32_t)points[j].y - bullet_h;
1203 point_area.y2 = (int32_t)points[j].y + bullet_h;
1204
1205 lv_draw_rect(layer, &point_draw_dsc, &point_area);
1206 point_draw_dsc.base.id2--;
1207 }
1208 }
1209 }
1210 line_dsc.base.id1--;
1211 }
1212
1213 if(points) lv_free(points);

1364 }
1365 valid_point_cnt++;
1366 }

1372 dsc->base.id1--;
1373 }
1374 }

1376 lv_draw_vector(dsc);
1377 lv_vector_path_delete(path);
1378 lv_draw_vector_dsc_delete(dsc);

1381 draw_series_line(obj, layer);
1382#endif /*LV_USE_VECTOR_GRAPHIC*/

1662 bar_full_area.y1 = obj->coords.y2 - y_ofs - total_bar_height + 1;

1665 int32_t y_prev = obj->coords.y2 + 1;

1667 /*Draw the current point of all data line*/
1668 int32_t v_sum_act = 0;
1669 LV_LL_READ(&chart->series_ll, ser) {
1670 int32_t start_point = chart->update_mode == LV_CHART_UPDATE_MODE_SHIFT ? ser->start_point : 0;
1671
1672 col_dsc.bg_color = ser->color;
1673
1674 int32_t p_act = (start_point + i) % chart->point_cnt;
1675 int32_t v_act = ser->y_points[p_act];
1676 /*Can't show negative values on a stacked chart*/
1677 if(ser->y_points[p_act] == LV_CHART_POINT_NONE ||
1678 v_act <= 0 ||
1679 ser->hidden) {
1680 col_dsc.base.id1++;
1681 continue;
1682 }
1683
1684 v_sum_act += v_act; /*Use the summed value to get its `y` to avoid rounding errors*/
1685
1686 int32_t segment_y = value_to_y(obj, ser, v_sum_act, h);
1687 bar_clip_area.y2 = y_prev - 1;
1688 bar_clip_area.y1 = obj->coords.y2 - y_ofs - segment_y + 1;
1689 y_prev = bar_clip_area.y1;
1690 if(lv_area_intersect(&layer->_clip_area, &clip_area_ori, &bar_clip_area)) {
1691 lv_draw_rect(layer, &col_dsc, &bar_full_area);

1693 col_dsc.base.id1++;

1832static void invalidate_point(lv_obj_t * obj, uint32_t i)
1833{
1834 LV_ASSERT(obj != NULL);
1835 lv_chart_t * chart = (lv_chart_t *)obj;
1836 if(i >= chart->point_cnt) return;
1837
1838
1839 /*In shift mode the whole chart changes so the whole object*/
1840 if(chart->update_mode == LV_CHART_UPDATE_MODE_SHIFT) {
1841 lv_obj_invalidate(obj);
1842 return;
1843 }
1844 int32_t w = lv_obj_get_content_width(obj);
1845 int32_t scroll_left = lv_obj_get_scroll_left(obj);
1846 int32_t bwidth = lv_obj_get_style_border_width_internal(obj, LV_PART_MAIN);
1847 int32_t pleft = lv_obj_get_style_pad_left_internal(obj, LV_PART_MAIN);
1848 int32_t x_ofs = obj->coords.x1 + pleft + bwidth - scroll_left;
1849
1850 if(chart->type == LV_CHART_TYPE_LINE || chart->type == LV_CHART_TYPE_CURVE) {
1851 int32_t line_width = lv_obj_get_style_line_width_internal(obj, LV_PART_ITEMS);
1852 int32_t point_w = lv_obj_get_style_width_internal(obj, LV_PART_INDICATOR);
1853
1854 lv_area_t coords;
1855 coords = obj->coords;
1856 coords.y1 -= line_width + point_w;
1857 coords.y2 += line_width + point_w;
1858
1859 /*Invalidate the area between the previous and the next points*/
1860 if(i < chart->point_cnt - 1) {
1861 coords.x1 = ((w * i) / (chart->point_cnt - 1)) + x_ofs - line_width - point_w;
1862 coords.x2 = ((w * (i + 1)) / (chart->point_cnt - 1)) + x_ofs + line_width + point_w;
1863 lv_obj_invalidate_area(obj, &coords);
1864 }
1865
1866 if(i > 0) {
1867 coords.x1 = ((w * (i - 1)) / (chart->point_cnt - 1)) + x_ofs - line_width - point_w;
1868 coords.x2 = ((w * i) / (chart->point_cnt - 1)) + x_ofs + line_width + point_w;
1869 lv_obj_invalidate_area(obj, &coords);
1870 }
1871 }
1872 else if(chart->type == LV_CHART_TYPE_BAR || chart->type == LV_CHART_TYPE_STACKED) {
1873 lv_area_t col_a;
1874 /*Gap between the column on ~adjacent X*/
1875 int32_t block_gap = lv_obj_get_style_pad_column_internal(obj, LV_PART_MAIN);
1876 int32_t block_w = (w - ((chart->point_cnt - 1) * block_gap)) / chart->point_cnt;
1877
1878 int32_t x_act;
1879 if(chart->point_cnt > 1) {
1880 x_act = (int32_t)((int32_t)(w - block_w) * i) / (chart->point_cnt - 1);
1881 }
1882 else {
1883 x_act = 0;
1884 }
1885
1886 lv_obj_get_coords(obj, &col_a);
1887 col_a.x1 = x_act + x_ofs;
1888 col_a.x2 = col_a.x1 + block_w + block_gap;
1889 col_a.x1 -= block_gap;
1890
1891 lv_obj_invalidate_area(obj, &col_a);
1892 }
1893 else {
1894 lv_obj_invalidate(obj);
1895 }
