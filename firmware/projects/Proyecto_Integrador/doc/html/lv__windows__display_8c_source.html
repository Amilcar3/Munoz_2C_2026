44lv_display_t * lv_windows_create_display(
45 const wchar_t * title,
46 int32_t hor_res,
47 int32_t ver_res,
48 int32_t zoom_level,
49 bool allow_dpi_override,
50 bool simulator_mode)
51{
52 LV_CHECK_ARG(title != NULL, return NULL);
53 lv_windows_create_display_data_t data;
54
55 lv_memzero(&data, sizeof(lv_windows_create_display_data_t));
56 data.title = title;
57 data.hor_res = hor_res;
58 data.ver_res = ver_res;
59 data.zoom_level = zoom_level;
60 data.allow_dpi_override = allow_dpi_override;
61 data.simulator_mode = simulator_mode;
62 data.mutex = CreateEventExW(NULL, NULL, 0, EVENT_ALL_ACCESS);
63 data.display = NULL;
64 if(!data.mutex) {
65 return NULL;
66 }
67
68 HANDLE thread = (HANDLE)_beginthreadex(
69 NULL,
70 0,
71 lv_windows_display_thread_entrypoint,
72 &data,
73 0,
74 NULL);
75 LV_ASSERT(thread);
76
77 WaitForSingleObjectEx(data.mutex, INFINITE, FALSE);
78
79 if(thread) {
80 CloseHandle(thread);
81 }
82
83 if(data.mutex) {
84 CloseHandle(data.mutex);
85 }
86
87 return data.display;
88}

120static unsigned int __stdcall lv_windows_display_thread_entrypoint(
121 void * parameter)
122{
123 lv_windows_create_display_data_t * data = parameter;
124 LV_ASSERT(data != NULL);
125
126 DWORD window_style = WS_OVERLAPPEDWINDOW;
127 DWORD ext_window_style = WS_EX_CLIENTEDGE;
128 RECT rect = { 0, 0, data->hor_res, data->ver_res };
129
130 if(data->simulator_mode) {
131 window_style &= ~(WS_SIZEBOX | WS_MAXIMIZEBOX | WS_THICKFRAME);
132 }
133 else {
134 /* Have Windows compute window size so, regardless of window style,
135 * the CLIENT AREA has dimensions [data->hor_res, data->ver_res].
136 * This is the area needed for LVGL to render to. */
137 AdjustWindowRectEx(&rect, window_style, false, ext_window_style);
138 }
139
140 HWND window_handle = CreateWindowExW(
141 ext_window_style,
142 L"LVGL.Window",
143 data->title,
144 window_style,
145 CW_USEDEFAULT,
146 0,
147 rect.right - rect.left,
148 rect.bottom - rect.top,
149 NULL,
150 NULL,
151 NULL,
152 data);
153
154 if(!window_handle) {
155 return 0;
156 }
157
158 lv_windows_window_context_t * context = lv_windows_get_window_context(
159 window_handle);
160 if(!context) {
161 return 0;
162 }
163
164 data->display = context->display_device_object;
165
166 ShowWindow(window_handle, SW_SHOW);
167 UpdateWindow(window_handle);
168
169 BOOL event_set = SetEvent(data->mutex);
170 LV_ASSERT(event_set);
171 LV_UNUSED(event_set);
172
173 data = NULL;
174
175 MSG message;
176 while(GetMessageW(&message, NULL, 0, 0)) {
177 TranslateMessage(&message);
178 DispatchMessageW(&message);
179 }
180
181 return 0;
182}
