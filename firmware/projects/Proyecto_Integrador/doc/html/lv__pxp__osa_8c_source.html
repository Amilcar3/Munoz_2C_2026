123static void _pxp_interrupt_init(void)
124{
125#if LV_USE_OS
126 if(lv_thread_sync_init(&pxp_sync) != LV_RESULT_OK) {
127 PXP_ASSERT_MSG(false, "Failed to init thread_sync.");
128 }
129#endif
130
131#if defined(__ZEPHYR__)
132 IRQ_CONNECT(DT_IRQN(DT_NODELABEL(pxp)), CONFIG_LV_Z_PXP_INTERRUPT_PRIORITY, _pxp_zephyr_irq_handler, NULL, 0);
133 irq_enable(DT_IRQN(DT_NODELABEL(pxp)));
134#elif defined(SDK_OS_FREE_RTOS)
135 NVIC_SetPriority(PXP_IRQ_ID, configLIBRARY_MAX_SYSCALL_INTERRUPT_PRIORITY + 1);
136#endif
137
138#if !defined(__ZEPHYR__)
139 NVIC_EnableIRQ(PXP_IRQ_ID);
140#endif
141
142 ucPXPIdle = true;
143}
