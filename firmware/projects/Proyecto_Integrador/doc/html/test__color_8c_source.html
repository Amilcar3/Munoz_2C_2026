168void test_color_hsv_to_rgb(void)
169{
170 /* HSV(0, 0, 0) should be black */
171 lv_color_t black = lv_color_hsv_to_rgb(0, 0, 0);
172 TEST_ASSERT_EQUAL_UINT8(0, black.red);
173 TEST_ASSERT_EQUAL_UINT8(0, black.green);
174 TEST_ASSERT_EQUAL_UINT8(0, black.blue);
175
176 /*HSV(0,0,100) should be white*/
177 lv_color_t white = lv_color_hsv_to_rgb(0, 0, 100);
178 TEST_ASSERT_EQUAL_UINT8(255, white.red);
179 TEST_ASSERT_EQUAL_UINT8(255, white.green);
180 TEST_ASSERT_EQUAL_UINT8(255, white.blue);
181
182 /*HSV(0,100,100) should be red*/
183 lv_color_t red = lv_color_hsv_to_rgb(0, 100, 100);
184 TEST_ASSERT_EQUAL_UINT8(255, red.red);
185 TEST_ASSERT_EQUAL_UINT8(0, red.green);
186 TEST_ASSERT_EQUAL_UINT8(0, red.blue);
187
188 /* Due to precision limitations, a tolerance has been added.*/
189 /*HSV(120,100,100) should be green*/
190 lv_color_t green = lv_color_hsv_to_rgb(120, 100, 100);
191 TEST_ASSERT_UINT8_WITHIN(3, 0, green.red);
192 TEST_ASSERT_EQUAL_UINT8(255, green.green);
193 TEST_ASSERT_EQUAL_UINT8(0, green.blue);
194
195 /*HSV(240,100,100) should be blue*/
196 lv_color_t blue = lv_color_hsv_to_rgb(240, 100, 100);
197 TEST_ASSERT_EQUAL_UINT8(0, blue.red);
198 TEST_ASSERT_UINT8_WITHIN(9, 0, blue.green);
199 TEST_ASSERT_EQUAL_UINT8(255, blue.blue);
200
201 /*Test boundary values*/
202 lv_color_t c1 = lv_color_hsv_to_rgb(360, 100, 100);
203 TEST_ASSERT_EQUAL_UINT8(255, c1.red);
204 TEST_ASSERT_EQUAL_UINT8(0, c1.green);
205 TEST_ASSERT_UINT8_WITHIN(15, 0, c1.blue);
206
207 lv_color_t c2 = lv_color_hsv_to_rgb(180, 0, 50);
208 TEST_ASSERT_EQUAL_UINT8(127, c2.blue);
209 TEST_ASSERT_EQUAL_UINT8(127, c2.green);
210 TEST_ASSERT_EQUAL_UINT8(127, c2.red);
211}
